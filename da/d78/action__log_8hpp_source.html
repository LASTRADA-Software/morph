16/// @brief One recorded execution of an action against a model instance.
17///
18/// Produced automatically by `morph::model::detail::IModelHolder::recordIfAttached`
19/// — application and model code never construct or append these directly.
20struct LogEntry {
21 /// @brief Monotonic order assigned by the sink on `append()`. Callers pass `0`.
22 uint64_t seq = 0;
23
24 /// @brief String type-id of the model the action ran against (`ModelTraits<M>::typeId()`).
25 std::string modelType;
26
27 /// @brief Stable identity of the model instance (e.g. an account id), stamped
28 /// from the value passed to `attachActionLog()`. Empty if none was set.
29 std::string entityKey;
30
31 /// @brief String type-id of the executed action (`ActionTraits<A>::typeId()`).
32 std::string actionType;
33
34 /// @brief JSON-encoded request (`ActionTraits<A>::toJson`).
35 std::string payload;
36
37 /// @brief JSON-encoded result (`ActionTraits<A>::resultToJson`), captured after
38 /// successful execution.
39 std::string result;
40
41 /// @brief Auth principal from `morph::session::current()`, if any. Empty if unset.
42 std::string principal;
43
44 /// @brief Wall-clock time of execution, milliseconds since the Unix epoch.
45 int64_t timestampMs = 0;
46};

55/// @brief Converts a Glaze error into a `SerializationError`, or does nothing
56/// if @p errCode reports success.
57///
58/// Shared by `toJson` and `fromJson` on purpose, not just for DRY: this is one
59/// non-template function, so both call through the exact same compiled branch.
60/// `fromJson`'s failure path is easy to exercise for real (malformed JSON is
61/// everyday input); `toJson`'s is not — Glaze's buffer-writer has no reachable
62/// failure mode for a flat struct of strings/integers like `LogEntry` (its
63/// only write-relevant error codes are for recursion-depth limits `LogEntry`
64/// can't hit, and `dump_int_error`, which nothing in Glaze's own source ever
65/// actually raises). Routing both through here means `toJson`'s error branch
66/// is the same branch `fromJson`'s test already exercises, rather than a
67/// second, structurally-unreachable copy of the same three lines.
68/// @param errCode Result of a `glz::write_json`/`glz::read_json` call.
69/// @param context Buffer or input passed to `glz::format_error` for the message.
70inline void throwOnGlazeError(const glz::error_ctx& errCode, std::string_view context) {
71 if (errCode) {
72 throw SerializationError{glz::format_error(errCode, context)};
73 }
74}

76} // namespace detail

78/// @brief Encodes @p entry as JSON.
79///
80/// `LogEntry` is a plain aggregate, so Glaze reflects it without a `glz::meta`
81/// specialisation — the same automatic reflection `BRIDGE_REGISTER_ACTION`
82/// relies on for user action structs. Used by sinks that need an opaque
83/// string representation (`FileActionLog`).
84/// @throws SerializationError on encode failure (see `detail::throwOnGlazeError`
85/// for why this is not realistically reachable for `LogEntry`).
86inline std::string toJson(const LogEntry& entry) {
87 std::string out;
88 detail::throwOnGlazeError(glz::write_json(entry, out), out);
89 return out;
90}
91
92/// @brief Decodes @p json into a `LogEntry`.
93/// @throws SerializationError if @p json is not a valid `LogEntry`.
94inline LogEntry fromJson(std::string_view json) {
95 LogEntry entry{};
96 detail::throwOnGlazeError(glz::read_json(entry, json), json);
97 return entry;
98}

100/// @brief Interface for durable storage of executed-action entries.
101///
102/// Entries are never removed by the framework — this is a permanent, append-only
103/// record, unlike `morph::offline::IOfflineQueue` (whose `markDone()` deletes
104/// items once retried successfully). Implementations range from in-memory
105/// (`InMemoryActionLog`) to file, SQL, or network-backed stores supplied by the
106/// host application.
107// NOLINTBEGIN(cppcoreguidelines-special-member-functions)
108struct IActionLog {
109 virtual ~IActionLog() = default;
110
111 /// @brief Appends @p entry. Implementations assign `entry.seq`.
112 /// @param entry Entry to append.
113 virtual void append(LogEntry entry) = 0;
114
115 /// @brief Pushes any buffered entries to the durable backend. No-op for sinks
116 /// with nothing to buffer (e.g. `InMemoryActionLog`).
117 virtual void flush() = 0;
118
119 /// @brief Returns recorded entries in append order.
120 /// @param entityKey If non-empty, restricts the result to that entity's entries.
121 /// @return Matching entries, in append order.
122 [[nodiscard]] virtual std::vector<LogEntry> entries(std::string_view entityKey = {}) const = 0;
123};
124// NOLINTEND(cppcoreguidelines-special-member-functions)
125
126/// @brief Thread-safe in-memory implementation of `IActionLog`.
127///
128/// Suitable for testing and for applications that do not need cross-process
129/// durability. Mirrors `morph::offline::InMemoryOfflineQueue`'s shape.
130class InMemoryActionLog : public IActionLog {
131public:
132 /// @brief Appends @p entry, assigning a monotonically increasing `seq`. Thread-safe.
133 /// @param entry Entry to append; `seq` is overwritten regardless of the input value.
134 void append(LogEntry entry) override {
135 std::scoped_lock const lock{_mtx};
136 entry.seq = ++_nextSeq;
137 _entries.push_back(std::move(entry));
138 }
139
140 /// @brief No-op — there is no external backend to flush to.
141 void flush() override {}
142
143 /// @brief Returns a snapshot of matching entries in append order. Thread-safe.
144 /// @param entityKey If non-empty, restricts the result to that entity's entries.
145 /// @return Matching entries, in append order.
146 [[nodiscard]] std::vector<LogEntry> entries(std::string_view entityKey = {}) const override {
147 std::scoped_lock const lock{_mtx};
148 if (entityKey.empty()) {
149 return _entries;
150 }
151 std::vector<LogEntry> out;
152 for (const auto& entry : _entries) {
153 if (entry.entityKey == entityKey) {
154 out.push_back(entry);
155 }
156 }
157 return out;
158 }
159
160private:
161 mutable std::mutex _mtx;
162 std::vector<LogEntry> _entries;
163 uint64_t _nextSeq{0};
164};

178/// @brief Installs @p log as the process-wide default action log.
179///
180/// Every model instance created via `morph::model::detail::ModelFactory::create<Model>()`
181/// — which is every model registered the ordinary way, whether the active
182/// backend ends up being local or remote — automatically gets @p log attached
183/// (with an empty `entityKey`) from that point on. Call this once at startup;
184/// no per-model or per-handler wiring is needed for the common case.
185///
186/// Application code that needs a specific instance identity (e.g. per-account
187/// auditing) can still call `IModelHolder::attachActionLog` explicitly on that
188/// instance afterward — an explicit call always overrides whatever the
189/// default attached. Thread-safe.
190///
191/// @param log Sink to attach automatically, or `nullptr` to stop auto-attaching
192/// (existing instances keep whatever they already have).
193inline void setActionLog(std::shared_ptr<IActionLog> log) {
194 auto& [mtx, slot] = detail::defaultActionLogState();
195 std::scoped_lock const lock{mtx};
196 slot = std::move(log);
197}
198
199/// @brief Returns the currently installed default action log, or `nullptr`
200/// if none has been set. Thread-safe.
201[[nodiscard]] inline std::shared_ptr<IActionLog> defaultActionLog() {
202 auto& [mtx, slot] = detail::defaultActionLogState();
203 std::scoped_lock const lock{mtx};
204 return slot;
205}

207/// @brief RAII helper that installs a default action log for its lifetime and
208/// restores the previous one on destruction.
209///
210/// Mirrors `morph::log::ScopedLoggerOverride`. Intended for tests (so one test
211/// case's sink never leaks into the next) and for applications that need to
212/// temporarily redirect auto-attached logging within a scope.
213///
214/// @code
215/// {
216/// morph::journal::ScopedActionLog guard{std::make_shared<morph::journal::InMemoryActionLog>()};
217/// // ... models created in this scope auto-attach guard's log ...
218/// } // previous default restored here
219/// @endcode
220class ScopedActionLog {
221public:
222 /// @brief Installs @p log as the default, saving whatever was there before.
223 /// @param log New default for the lifetime of this object.
224 explicit ScopedActionLog(std::shared_ptr<IActionLog> log) : _previous{defaultActionLog()} {
225 setActionLog(std::move(log));
226 }
227
228 /// @brief Restores the saved default.
229 ~ScopedActionLog() { setActionLog(std::move(_previous)); }
230
231 ScopedActionLog(const ScopedActionLog&) = delete;
232 ScopedActionLog& operator=(const ScopedActionLog&) = delete;
233 ScopedActionLog(ScopedActionLog&&) = delete;
234 ScopedActionLog& operator=(ScopedActionLog&&) = delete;
235
236private:
237 std::shared_ptr<IActionLog> _previous;
238};
