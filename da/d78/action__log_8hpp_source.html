17/// @brief Current line-format version stamped on every newly-written
18/// `LogEntry` (`LogEntry::v`'s default).
19///
20/// Bumped only on a **breaking** change to `LogEntry`'s on-disk/on-wire shape;
21/// an additive key (tolerated by `fromJson`'s lenient decode) does not bump
22/// it. A reader refuses to decode a line whose `v` exceeds this constant —
23/// see `fromJson`.
24inline constexpr std::uint32_t kLogFormatVersion = 1;
25
26/// @brief Discriminates a recorded `LogEntry` as a successful execution or one
27/// rejected by the validator / thrown by `Model::execute`.
28///
29/// A failed entry has an empty `result` (there was none) and a non-empty
30/// `error`; a succeeded entry is the reverse. Serialises as the string
31/// `"Succeeded"`/`"Failed"` (see the `glz::meta` specialisation below) so the
32/// audit trail stays human-readable without cross-referencing the enum.
33enum class Outcome : std::uint8_t { Succeeded, Failed };
34
35/// @brief One recorded execution of an action against a model instance.
36///
37/// Produced automatically by `morph::model::detail::IModelHolder::recordIfAttached`
38/// — application and model code never construct or append these directly.
39struct LogEntry {
40 /// @brief Monotonic order assigned by the sink on `append()`. Callers pass `0`.
41 uint64_t seq = 0;
42
43 /// @brief String type-id of the model the action ran against (`ModelTraits<M>::typeId()`).
44 std::string modelType;
45
46 /// @brief Stable identity of the model instance (e.g. an account id), stamped
47 /// from the value passed to `attachActionLog()`. Empty if none was set.
48 std::string entityKey;
49
50 /// @brief String type-id of the executed action (`ActionTraits<A>::typeId()`).
51 std::string actionType;
52
53 /// @brief JSON-encoded request (`ActionTraits<A>::toJson`).
54 std::string payload;
55
56 /// @brief JSON-encoded result (`ActionTraits<A>::resultToJson`), captured after
57 /// successful execution. Empty when `outcome == Outcome::Failed`.
58 std::string result;
59
60 /// @brief Whether this execution succeeded or was rejected/threw. Defaults to
61 /// `Succeeded` so a pre-existing on-disk entry (written before this
62 /// field existed) decodes unchanged under the lenient reader — an
63 /// absent key is indistinguishable from an explicit `Succeeded`.
64 Outcome outcome = Outcome::Succeeded;
65
66 /// @brief `std::exception::what()` from the exception that rejected the
67 /// action. Empty unless `outcome == Outcome::Failed`.
68 std::string error{};
69
70 /// @brief Auth principal from `morph::session::current()`, if any. Empty if unset.
71 std::string principal;
72
73 /// @brief Wall-clock time of execution, milliseconds since the Unix epoch.
74 int64_t timestampMs = 0;
75
76 /// @brief Optional dedup token for outbox-relayed entries. Empty by default;
77 /// ordinary auto-appended entries (from `ActionDispatcher`'s runner or
78 /// `Bridge::executeVia`'s local op) never set it. Mirrors
79 /// `morph::offline::QueueItem::idempotencyKey`'s exact contract: opaque,
80 /// stored verbatim, stable across restarts for one logical outbox row.
81 /// See `journal::OutboxRelay` (`outbox.hpp`) for how it's used.
82 std::string idempotencyKey{};
83
84 /// @brief Line-format version this entry was written at.
85 ///
86 /// Defaults to `kLogFormatVersion`, so every freshly-constructed entry
87 /// already carries the current version with no separate stamping step. A
88 /// legacy line (written before this field existed) has no `v` key; under
89 /// `fromJson`'s lenient decode that is just an absent key, so it decodes
90 /// with this same default — i.e. legacy data reads as `v == 1`, which is
91 /// correct: v1 is today's shape, `kLogFormatVersion` merely names it.
92 std::uint32_t v = kLogFormatVersion;
93
94 /// @brief Identity of the "trigger" entry that caused this entry to be
95 /// recorded, or empty (the sentinel) if this entry was not caused
96 /// by another one.
97 ///
98 /// Set by application code that journals a cascaded mutation — e.g. an
99 /// automation rule that reacts to one recorded action by executing a
100 /// further one — to the triggering entry's own stable identity, so
101 /// `replay()` and an activity view can both recover "what caused this."
102 /// Empty by default: an ordinary, non-cascaded entry never sets it.
103 ///
104 /// @warning **Must not be a `LogEntry::seq` value.** `seq` is sink-local
105 /// and re-stamped by every sink's `append()` (and again by
106 /// `SessionLog::checkpoint()` when forwarding) — it is not a stable,
107 /// cross-sink or cross-restart identifier (see the Invariants section of
108 /// `docs/spec/journal/journal.md`). A `causalParentId` wired to a raw
109 /// `seq` would stop matching anything the moment the trigger entry is
110 /// forwarded to another sink or the process restarts. Application code
111 /// must instead mint its own opaque/UUID-style identity for the trigger
112 /// entry at the point it is created, independent of whatever `seq` any
113 /// sink later assigns it, and reuse that same identity as every cascaded
114 /// entry's `causalParentId`.
115 std::string causalParentId{};
116};

139/// @brief Converts a Glaze error into a `SerializationError`, or does nothing
140/// if @p errCode reports success.
141///
142/// Shared by `toJson` and `fromJson` on purpose, not just for DRY: this is one
143/// non-template function, so both call through the exact same compiled branch.
144/// `fromJson`'s failure path is easy to exercise for real (malformed JSON is
145/// everyday input); `toJson`'s is not — Glaze's buffer-writer has no reachable
146/// failure mode for a flat struct of strings/integers like `LogEntry` (its
147/// only write-relevant error codes are for recursion-depth limits `LogEntry`
148/// can't hit, and `dump_int_error`, which nothing in Glaze's own source ever
149/// actually raises). Routing both through here means `toJson`'s error branch
150/// is the same branch `fromJson`'s test already exercises, rather than a
151/// second, structurally-unreachable copy of the same three lines.
152/// @param errCode Result of a `glz::write_json`/`glz::read_json` call.
153/// @param context Buffer or input passed to `glz::format_error` for the message.
154inline void throwOnGlazeError(const glz::error_ctx& errCode, std::string_view context) {
155 if (errCode) {
156 throw SerializationError{glz::format_error(errCode, context)};
157 }
158}

160/// @brief Write options that escape ASCII control bytes as `\\uXXXX` sequences.
161///
162/// glaze 7.4 leaves control bytes (0x00-0x1F) unescaped by default, which
163/// breaks a `LogEntry` carrying one in `entityKey`/`payload`/`error`/`principal`/
164/// `idempotencyKey` two ways: RFC 8259 requires those bytes escaped, so the raw
165/// byte alone yields JSON `fromJson`'s `glz::read` throws on; worse, once the
166/// same string also contains an escaped `\` or `"`, glaze's chunked writer path
167/// silently rewrites the control byte as two 0x00 bytes, corrupting the payload
168/// before it ever reaches disk. This mirrors `morph::wire::detail::EscapingWriteOpts`
169/// (`core/wire.hpp`) exactly; duplicated here (rather than shared) so this header
170/// stays free of a `core/` dependency for `MORPH_CLIENT_ONLY`-style consumers
171/// that only want the journal. Escaping is lossless, so any such byte still
172/// round-trips through `fromJson` unchanged.
173struct EscapingWriteOpts : glz::opts {
174 /// @brief Emit control bytes as `\\uXXXX` rather than raw.
175 // NOLINTNEXTLINE(readability-identifier-naming) — glaze's option name, matched by name.
176 bool escape_control_characters = true;
177};

181/// @brief Encodes @p entry as JSON.
182///
183/// `LogEntry` is a plain aggregate, so Glaze reflects it without a `glz::meta`
184/// specialisation of its own — the same automatic reflection
185/// `BRIDGE_REGISTER_ACTION` relies on for user action structs. (Its `outcome`
186/// field is `Outcome`, which does have a `glz::meta` — see above — so it reads
187/// back as `"Succeeded"`/`"Failed"`, not `0`/`1`.) Used by sinks that need an
188/// opaque string representation (`FileActionLog`).
189///
190/// Writes with `detail::EscapingWriteOpts` so a raw ASCII control byte in any
191/// string field (`entityKey`, `payload`, `error`, `principal`, `idempotencyKey`)
192/// round-trips through `fromJson` instead of producing invalid or silently
193/// corrupted JSON — see that struct's doc comment.
194/// @throws SerializationError on encode failure (see `detail::throwOnGlazeError`
195/// for why this is not realistically reachable for `LogEntry`).
196inline std::string toJson(const LogEntry& entry) {
197 std::string out;
198 detail::throwOnGlazeError(glz::write<detail::EscapingWriteOpts{}>(entry, out), out);
199 return out;
200}

202/// @brief Decodes @p json into a `LogEntry`.
203///
204/// Reads leniently: `glz::read<glz::opts{.error_on_unknown_keys = false}>`,
205/// the same stance `morph::wire::decode` takes (`include/morph/core/wire.hpp`) —
206/// an unknown/extra JSON key (e.g. an additive field written by a newer morph
207/// build) is ignored rather than rejected. Same duplicate-key caveat as
208/// `wire::decode`: last-wins, not a security boundary (glaze offers no reject
209/// option). Syntactically malformed JSON still throws. After a successful
210/// decode, also enforces the line-format version rule: `v <= kLogFormatVersion`
211/// decodes normally, `v` greater than this build's `kLogFormatVersion` throws
212/// — a build refuses to guess at a line format newer than any it has seen.
213/// @throws SerializationError if @p json is not valid JSON, does not decode
214/// into a `LogEntry`, or decodes with `v` greater than
215/// `kLogFormatVersion`.
216inline LogEntry fromJson(std::string_view json) {
217 LogEntry entry{};
218 // `null_terminated = false`: `json` is a caller-supplied view (a line read from a
219 // journal file, with no guaranteed trailing '\0') — see the identical rationale on
220 // `morph::wire::decode` (`wire.hpp`), whose fuzz harness found the resulting
221 // heap-buffer-overflow in glaze's `skip_ws`.
222 static constexpr glz::opts kLenient{.null_terminated = false, .error_on_unknown_keys = false};
223 detail::throwOnGlazeError(glz::read<kLenient>(entry, json), json);
224 if (entry.v > kLogFormatVersion) {
225 throw SerializationError{
226 "journal::fromJson: line format v" + std::to_string(entry.v) +
227 " is newer than this build supports (kLogFormatVersion = " + std::to_string(kLogFormatVersion) + ")"};
228 }
229 return entry;
230}

232/// @brief Interface for durable storage of executed-action entries.
233///
234/// Entries are never removed by the framework — this is a permanent, append-only
235/// record, unlike `morph::offline::IOfflineQueue` (whose `markDone()` deletes
236/// items once retried successfully). Implementations range from in-memory
237/// (`InMemoryActionLog`) to file, SQL, or network-backed stores supplied by the
238/// host application.
239///
240/// @par Idempotency-key dedup (optional)
241/// An implementation MAY treat a non-empty `LogEntry::idempotencyKey` as a dedup
242/// key on `append()`: if an entry with the same key was already recorded, treat
243/// the call as a no-op. This is not required by the interface, but
244/// `InMemoryActionLog` and `FileActionLog` both do it, which is what makes them
245/// safe choices for `journal::OutboxRelay::sink` (see `outbox.hpp`) — a
246/// re-relayed row after a crash between `append()` and marking it relayed lands
247/// here twice but is stored once. An entry with an empty `idempotencyKey` is
248/// never deduped.
249// NOLINTBEGIN(cppcoreguidelines-special-member-functions)
250struct IActionLog {
251 virtual ~IActionLog() = default;
252
253 /// @brief Appends @p entry. Implementations assign `entry.seq`.
254 ///
255 /// An implementation that can fail to record the entry must throw. Returning
256 /// normally is the sink's promise that the entry is recorded (or, for a
257 /// buffering sink, that it will be by the next successful `flush()`); see
258 /// `flush()` for why silence is not an option here.
259 ///
260 /// @param entry Entry to append.
261 /// @throws std::exception (implementation-defined) if the entry could not be recorded.
262 virtual void append(LogEntry entry) = 0;
263
264 /// @brief Pushes any buffered entries to the durable backend. No-op for sinks
265 /// with nothing to buffer (e.g. `InMemoryActionLog`).
266 ///
267 /// **Must throw if the data did not reach the backend.** The return type is
268 /// `void`, so throwing is the only channel an implementation has, and
269 /// callers rely on it: `OutboxRelay::relay()` calls `markRelayed()` directly
270 /// after this, and a silently-failed flush would mark rows relayed in the
271 /// model's own store while nothing was durably written — dropping them from
272 /// the outbox *and* from the log, with no error anywhere. An implementation
273 /// that cannot fail (nothing to buffer) simply never throws.
274 ///
275 /// @throws std::exception (implementation-defined) if buffered entries could
276 /// not be made durable.
277 virtual void flush() = 0;
278
279 /// @brief Returns recorded entries in append order.
280 /// @param entityKey If non-empty, restricts the result to that entity's entries.
281 /// @return Matching entries, in append order.
282 [[nodiscard]] virtual std::vector<LogEntry> entries(std::string_view entityKey = {}) const = 0;
283};
284// NOLINTEND(cppcoreguidelines-special-member-functions)

286/// @brief Thread-safe in-memory implementation of `IActionLog`.
287///
288/// Suitable for testing and for applications that do not need cross-process
289/// durability. Mirrors `morph::offline::InMemoryOfflineQueue`'s shape. Dedups
290/// `append()` on a non-empty `LogEntry::idempotencyKey` — see `IActionLog`'s
291/// class docs.
292class InMemoryActionLog : public IActionLog {
293public:
294 /// @brief Appends @p entry, assigning a monotonically increasing `seq`. Thread-safe.
295 /// @param entry Entry to append; `seq` is overwritten regardless of the input value.
296 void append(LogEntry entry) override {
297 std::scoped_lock const lock{_mtx};
298 if (!entry.idempotencyKey.empty() && !_seenIdempotencyKeys.insert(entry.idempotencyKey).second) {
299 return; // already recorded once; a re-relayed duplicate is a safe no-op
300 }
301 entry.seq = ++_nextSeq;
302 _entries.push_back(std::move(entry));
303 }
304
305 /// @brief No-op — there is no external backend to flush to.
306 void flush() override {}
307
308 /// @brief Returns a snapshot of matching entries in append order. Thread-safe.
309 /// @param entityKey If non-empty, restricts the result to that entity's entries.
310 /// @return Matching entries, in append order.
311 [[nodiscard]] std::vector<LogEntry> entries(std::string_view entityKey = {}) const override {
312 std::scoped_lock const lock{_mtx};
313 if (entityKey.empty()) {
314 return _entries;
315 }
316 std::vector<LogEntry> out;
317 for (const auto& entry : _entries) {
318 if (entry.entityKey == entityKey) {
319 out.push_back(entry);
320 }
321 }
322 return out;
323 }
324
325private:
326 mutable std::mutex _mtx;
327 std::vector<LogEntry> _entries;
328 uint64_t _nextSeq{0};
329 std::unordered_set<std::string> _seenIdempotencyKeys;
330};

344/// @brief Installs @p log as the process-wide default action log.
345///
346/// Every model instance created via `morph::model::detail::ModelFactory::create<Model>()`
347/// — which is every model registered the ordinary way, whether the active
348/// backend ends up being local or remote — automatically gets @p log attached
349/// (with an empty `entityKey`) from that point on. Call this once at startup;
350/// no per-model or per-handler wiring is needed for the common case.
351///
352/// Application code that needs a specific instance identity (e.g. per-account
353/// auditing) can still call `IModelHolder::attachActionLog` explicitly on that
354/// instance afterward — an explicit call always overrides whatever the
355/// default attached. Thread-safe.
356///
357/// @param log Sink to attach automatically, or `nullptr` to stop auto-attaching
358/// (existing instances keep whatever they already have).
359inline void setActionLog(std::shared_ptr<IActionLog> log) {
360 auto& [mtx, slot] = detail::defaultActionLogState();
361 std::scoped_lock const lock{mtx};
362 slot = std::move(log);
363}

365/// @brief Returns the currently installed default action log, or `nullptr`
366/// if none has been set. Thread-safe.
367[[nodiscard]] inline std::shared_ptr<IActionLog> defaultActionLog() {
368 auto& [mtx, slot] = detail::defaultActionLogState();
369 std::scoped_lock const lock{mtx};
370 return slot;
371}
