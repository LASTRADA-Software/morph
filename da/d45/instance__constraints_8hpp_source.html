5/// @file forms/instance_constraints.hpp
6/// @brief Per-*instance* constraints on a form field: the seam by which data
7/// that lives in a database row — not in a template parameter — reaches
8/// the framework-meaningful keys of a served schema, and is checked
9/// against a submitted value by the same declaration that served it.
10///
11/// `schemaJson<A>()` is a pure function of the compiled action type `A`: every
12/// key it emits is derived from `A`'s members, so two rows that describe the
13/// *same* action differently — an analysis catalogue whose version 1 declares
14/// three decimal places and whose version 2 declares one — cannot be told
15/// apart by anything the framework enforces. Before this header the only way
16/// out was for the application to patch a *second*, app-private key beside the
17/// framework's (`x-versionDecimalPlaces` next to `x-decimalPlaces`) and
18/// re-implement the check by hand, which leaves a renderer with two keys for
19/// one concept and no way to know which to believe.
20///
21/// `InstanceConstraints` replaces that with one declaration that does both
22/// jobs:
23///
24/// - `decorate()` rewrites the served schema so the *framework's own* keys
25/// carry the instance's values (`x-decimalPlaces`, `x-minimum`,
26/// `x-maximum`), and stamps the document-level `x-instanceConstraints` array
27/// naming every field whose keys came from instance data rather than from
28/// the compiled type — so a renderer can tell the two apart.
29/// - `checkAction()` / `checkValue()` report what a submitted value violates,
30/// against the identical declaration. Serving a bound and checking it can no
31/// longer drift apart, because there is only one place the bound is written.
32///
33/// @par What this deliberately does not do
34/// The framework does **not** decide what a violation means. An out-of-range
35/// value is a rejection in one domain and a value to *flag* in another (a lab
36/// result outside its specification range is reportable, not refusable), so
37/// `checkAction`/`checkValue` return violations and the model applies its own
38/// policy. Nor is any of this wired into the dispatch runners: those decode an
39/// action from bytes with no access to the row a constraint lives in (see
40/// docs/spec/forms/instance_constraints.md, "Why dispatch cannot apply these
41/// automatically"). A model that decorates a schema is responsible for
42/// checking against the same constraints; the `x-instanceConstraints` stamp is
43/// what makes a schema that was decorated visible to a reviewer and a
44/// renderer.
45///
46/// @par Structure is still compile-time
47/// Only *values* of existing keys vary per instance. Which fields exist, which
48/// are required, and the `x-rules` list all remain functions of the compiled
49/// type — an instance cannot add a field. See the spec's "Limitations".

64/// @brief Which framework-meaningful key a submitted value fell foul of.
65enum class ConstraintViolationKind : std::uint8_t {
66 /// @brief The value is strictly below the declared inclusive `minimum`.
67 BelowMinimum,
68 /// @brief The value is strictly above the declared inclusive `maximum`.
69 AboveMaximum,
70 /// @brief The value is not exactly representable at the declared
71 /// `decimalPlaces` — it carries more precision than the instance
72 /// says the field has.
73 PrecisionExceeded,
74};
75
76/// @brief One violation: which field, and which declared key it broke.
77struct ConstraintViolation {
78 /// @brief The offending field's wire (JSON) name.
79 std::string field;
80 /// @brief What was violated.
81 ConstraintViolationKind kind{ConstraintViolationKind::BelowMinimum};
82
83 /// @brief Value equality, so violations can be compared in tests.
84 /// @param other The violation to compare against.
85 /// @return `true` when both the field and the kind match.
86 [[nodiscard]] bool operator==(const ConstraintViolation& other) const noexcept = default;
87};
88
89/// @brief A stable, machine-readable name for a violation kind — for log
90/// lines and model-level error messages.
91/// @param kind The violation kind to name.
92/// @return `"belowMinimum"`, `"aboveMaximum"` or `"precisionExceeded"`.
93[[nodiscard]] inline std::string_view violationKindName(ConstraintViolationKind kind) noexcept {
94 switch (kind) {
95 case ConstraintViolationKind::BelowMinimum:
96 return "belowMinimum";
97 case ConstraintViolationKind::AboveMaximum:
98 return "aboveMaximum";
99 case ConstraintViolationKind::PrecisionExceeded:
100 return "precisionExceeded";
101 default:
102 return "unknown";
103 }
104}

106/// @brief The per-instance constraint declared for one field of an action.
107///
108/// Every member other than `field` is optional and defaults to "not declared":
109/// an absent `decimalPlaces` leaves the compiled `x-decimalPlaces` alone, and
110/// an absent bound emits nothing and checks nothing. A constraint naming a
111/// field the action does not have is ignored (schema decoration never throws
112/// over an application's declaration mistake, exactly like `formLayout`).
113struct FieldConstraint {
114 /// @brief Wire (JSON) key of the member this entry constrains.
115 std::string field;
116
117 /// @brief The instance's decimal precision for this field. Overwrites
118 /// `x-decimalPlaces`, and a submitted value that is not exactly
119 /// representable at this many places is `PrecisionExceeded`.
120 std::optional<std::uint32_t> decimalPlaces{};
121
122 /// @brief Inclusive lower bound; emitted as `x-minimum`.
123 std::optional<math::Rational> minimum{};
124
125 /// @brief Inclusive upper bound; emitted as `x-maximum`.
126 std::optional<math::Rational> maximum{};
127};

131/// @brief `10^places`, for the exact-representability test below.
132///
133/// `Rational` bounds its own `decimalPlaces` to `kMaxDecimalPlaces` (18, the
134/// largest power of ten that fits an `std::int64_t`), and callers clamp to the
135/// same range before calling this, so it cannot overflow.
136/// @param places Decimal places, in `[0, morph::math::kMaxDecimalPlaces]`.
137/// @return `10^places`.
138[[nodiscard]] constexpr std::int64_t powerOfTenForConstraint(std::uint32_t places) noexcept {
139 std::int64_t result = 1;
140 for (std::uint32_t i = 0; i < places; ++i) {
141 result *= 10;
142 }
143 return result;
144}
145
146/// @brief Whether @p value is *exactly* representable at @p places decimals.
147///
148/// `Rational` keeps `gcd(|numerator|, denominator) == 1`, so `numerator *
149/// 10^places` is divisible by `denominator` exactly when `10^places` is — the
150/// test needs no multiplication and therefore cannot overflow.
151/// @param value The submitted exact value.
152/// @param places The instance's declared decimal places.
153/// @return `true` when the value needs no rounding at that precision.
154[[nodiscard]] constexpr bool exactAtDecimals(const math::Rational& value, std::uint32_t places) noexcept {
155 if (places > math::kMaxDecimalPlaces) {
156 return false;
157 }
158 return powerOfTenForConstraint(places) % value.denominator == 0;
159}
160
161/// @brief Renders one exact bound as the same `{"num","den","dp"}` node shape
162/// a `Rational` uses on the wire, so a renderer parses a bound exactly
163/// the way it parses the value it bounds.
164/// @param bound The exact bound.
165/// @return The DOM node.
166[[nodiscard]] inline glz::generic_u64 boundNode(const math::Rational& bound) {
167 glz::generic_u64 node{};
168 // NOLINTBEGIN(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) — glaze DOM requires operator[]
169 node["num"] = bound.numerator;
170 node["den"] = bound.denominator;
171 node["dp"] = static_cast<std::uint64_t>(bound.decimalPlaces.value);
172 // NOLINTEND(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
173 return node;
174}

178/// @brief A set of per-field constraints sourced from one instance (a database
179/// row, a tenant record, a catalogue version) — the single declaration
180/// that both decorates a served schema and checks a submitted value.
181///
182/// @code{.cpp}
183/// morph::forms::InstanceConstraints constraints;
184/// constraints.declare({.field = "value",
185/// .decimalPlaces = version.decimalPlaces,
186/// .minimum = version.specLow,
187/// .maximum = version.specHigh});
188///
189/// // Serving: the framework's own keys now carry the version's values.
190/// const auto schema = morph::forms::instanceSchemaJson<CaptureConcentration>(constraints);
191///
192/// // Checking: the same object, so the served bound and the checked bound
193/// // cannot disagree.
194/// for (const auto& violation : constraints.checkAction(action)) { ... }
195/// @endcode
196class InstanceConstraints {
197public:
198 /// @brief Declares (or replaces, by field name) one field's constraint.
199 /// @param constraint The constraint to record.
200 /// @return `*this`, so declarations can be chained.
201 InstanceConstraints& declare(FieldConstraint constraint) {
202 for (auto& existing : _fields) {
203 if (existing.field == constraint.field) {
204 existing = std::move(constraint);
205 return *this;
206 }
207 }
208 _fields.push_back(std::move(constraint));
209 return *this;
210 }
211
212 /// @brief The constraint declared for @p field, if any.
213 /// @param field The wire (JSON) field name to look up.
214 /// @return A pointer to the entry, or `nullptr` when the field is
215 /// unconstrained. Invalidated by any later `declare()`.
216 [[nodiscard]] const FieldConstraint* forField(std::string_view field) const noexcept {
217 for (const auto& entry : _fields) {
218 if (entry.field == field) {
219 return &entry;
220 }
221 }
222 return nullptr;
223 }
224
225 /// @brief Whether nothing has been declared.
226 /// @return `true` when no field carries a constraint.
227 [[nodiscard]] bool empty() const noexcept { return _fields.empty(); }
228
229 /// @brief Every declared constraint, in declaration order.
230 /// @return The constraint entries.
231 [[nodiscard]] const std::vector<FieldConstraint>& fields() const noexcept { return _fields; }
232
233 /// @brief Rewrites @p schema so the framework's own keys carry this
234 /// instance's values.
235 ///
236 /// For each declared field that the schema actually has a property for:
237 /// `x-decimalPlaces` is **overwritten** with the declared precision (when
238 /// one is declared), and `x-minimum` / `x-maximum` are added as
239 /// `{"num","den","dp"}` nodes. The document then carries
240 /// `x-instanceConstraints`: the array of field names whose keys came from
241 /// instance data, which is how a renderer (and a reviewer) tells a
242 /// decorated schema from the compiled one.
243 ///
244 /// Never throws and never mangles: a @p schema that is not readable JSON
245 /// is returned verbatim, and a constraint naming a field the schema has no
246 /// property for is skipped — the same "an author's declaration mistake is
247 /// not worth a crash" rule `mergeSchemaExtras` follows.
248 /// @param schema The compiled schema text, typically `schemaJson<A>()`.
249 /// @return The decorated schema text.
250 [[nodiscard]] std::string decorate(std::string schema) const {
251 if (_fields.empty()) {
252 return schema;
253 }
254 // NOLINTBEGIN(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) — glaze DOM requires operator[]
255 // u64 number mode, for the same reason mergeSchemaExtras uses it: the
256 // schema carries int64/uint64 bounds in $defs that a double-only DOM
257 // would silently round on the round trip.
258 glz::generic_u64 dom{};
259 if (glz::read_json(dom, schema)) {
260 return schema;
261 }
262 if (!dom.contains("properties")) {
263 return schema;
264 }
265 auto& properties = dom["properties"];
266 glz::generic_u64::array_t decorated{};
267 for (const auto& entry : _fields) {
268 if (!properties.contains(entry.field)) {
269 continue; // names a field the action does not have: ignored, never thrown
270 }
271 auto& property = properties[entry.field];
272 if (entry.decimalPlaces.has_value()) {
273 property["x-decimalPlaces"] =
274 static_cast<std::uint64_t>(math::detail::clampWireDecimalPlaces(*entry.decimalPlaces));
275 }
276 if (entry.minimum.has_value()) {
277 property["x-minimum"] = detail::boundNode(*entry.minimum);
278 }
279 if (entry.maximum.has_value()) {
280 property["x-maximum"] = detail::boundNode(*entry.maximum);
281 }
282 decorated.emplace_back(entry.field);
283 }
284 if (decorated.empty()) {
285 return schema;
286 }
287 dom["x-instanceConstraints"] = decorated;
288 // NOLINTEND(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
289 return glz::write_json(dom).value_or(schema);
290 }

292 /// @brief Checks one exact value against the constraint declared for
293 /// @p field.
294 ///
295 /// The value-level entry point, for a model whose stored value is
296 /// *derived* from the payload rather than taken from it verbatim (a
297 /// reading multiplied by a dilution factor, a total recomputed from line
298 /// items): the constraint governs the value the model is about to store,
299 /// which is not a member of the action at all.
300 /// @param field The wire (JSON) field name the value belongs to.
301 /// @param value The exact value to check.
302 /// @return Every violation, in `minimum`, `maximum`, `decimalPlaces`
303 /// order; empty when the field is unconstrained or the value
304 /// satisfies every declared key.
305 [[nodiscard]] std::vector<ConstraintViolation> checkValue(std::string_view field,
306 const math::Rational& value) const {
307 std::vector<ConstraintViolation> violations{};
308 const auto* entry = forField(field);
309 if (entry == nullptr) {
310 return violations;
311 }
312 if (entry->minimum.has_value() && (value <=> *entry->minimum) == std::strong_ordering::less) {
313 violations.push_back({.field = std::string{field}, .kind = ConstraintViolationKind::BelowMinimum});
314 }
315 if (entry->maximum.has_value() && (value <=> *entry->maximum) == std::strong_ordering::greater) {
316 violations.push_back({.field = std::string{field}, .kind = ConstraintViolationKind::AboveMaximum});
317 }
318 if (entry->decimalPlaces.has_value() && !detail::exactAtDecimals(value, *entry->decimalPlaces)) {
319 violations.push_back({.field = std::string{field}, .kind = ConstraintViolationKind::PrecisionExceeded});
320 }
321 return violations;
322 }
323
324 /// @brief Checks every constrained `Quantity` member of @p action.
325 ///
326 /// Walks the action's reflected members in declaration order and runs
327 /// `checkValue` for each engaged `Quantity` whose wire name carries a
328 /// constraint. An empty `Quantity` is skipped (emptiness is `required`'s
329 /// business, not a bound's), as is any member that is not a `Quantity` —
330 /// bounds and decimal precision are exact-value concepts, and `Quantity`
331 /// is the only field type morph forms carry an exact value in. A no-op for
332 /// an action type glaze cannot reflect.
333 /// @tparam A Action type (a reflectable aggregate).
334 /// @param action The decoded action to check.
335 /// @return Every violation across every constrained field, in member
336 /// declaration order.
337 template <typename A>
338 [[nodiscard]] std::vector<ConstraintViolation> checkAction(const A& action) const {
339 using Plain = std::remove_cvref_t<A>;
340 std::vector<ConstraintViolation> violations{};
341 if constexpr (glz::reflectable<Plain> || glz::glaze_object_t<Plain>) {
342 forms::detail::forEachNamedMember(action, [&]<std::size_t I>(std::string_view name, const auto& member) {
343 static_cast<void>(I);
344 using Member = std::remove_cvref_t<decltype(member)>;
345 if constexpr (units::isQuantity<Member>) {
346 if (member.hasValue()) {
347 for (auto& violation : checkValue(name, *member)) {
348 violations.push_back(std::move(violation));
349 }
350 }
351 } else {
352 static_cast<void>(name);
353 static_cast<void>(member);
354 }
355 });
356 } else {
357 static_cast<void>(action);
358 }
359 return violations;
360 }
361
362private:
363 std::vector<FieldConstraint> _fields{};
364};

366/// @brief `schemaJson<A>()` decorated with @p constraints — the served form
367/// for one instance.
368///
369/// Not memoised: unlike `schemaJson<A>()` (one cached string per compiled
370/// type), the result varies per instance, so the caller caches it if it wants
371/// caching. The compiled half is still memoised inside `schemaJson<A>()`, so
372/// the per-call cost is the DOM round trip, not schema generation.
373/// @tparam A Action type (a reflectable aggregate).
374/// @param constraints The instance's constraints.
375/// @return The decorated schema text.
376template <typename A>
377[[nodiscard]] std::string instanceSchemaJson(const InstanceConstraints& constraints) {
378 return constraints.decorate(schemaJson<A>());
379}
