11namespace morph::model {

13/// @brief Whether @p Action declares a guard: a public member alias `guard` naming the guard type.
14///
15/// Read off the action's own definition, so every translation unit that names the action gives the same answer.
16template <typename Action>
17concept DeclaresGuard = requires { typename Action::guard; };
18
19/// @brief Whether @p Model declares `void onGuardRefused(Action const&, std::exception const&)`, which every
20/// execution site and `executeChecked` call after the action's guard refuses it.
21template <typename Model, typename Action>
22concept GuardRefusalObserver = requires(Model& model, Action const& action, std::exception const& refusal) {
23 { model.onGuardRefused(action, refusal) } -> std::same_as<void>;
24};

26namespace detail {

28/// @brief Whether @p Guard checks @p Action on its payload alone.
29template <typename Guard, typename Action>
30concept GuardChecksAction = requires(Action const& action) {
31 { Guard::check(action) } -> std::same_as<void>;
32};
33
34/// @brief Whether @p Guard checks @p Action against an instance of @p Model.
35template <typename Guard, typename Model, typename Action>
36concept GuardChecksInModel = requires(Model const& model, Action const& action) {
37 { Guard::check(model, action) } -> std::same_as<void>;
38};
39
40/// @brief Runs @p Action's guard on @p action in the form it declares; nothing for an unguarded action.
41/// @tparam Model The model type the action is executed on.
42/// @tparam Action The action type.
43/// @param model The instance the action is addressed to.
44/// @param action The decoded, recomputed, validated action.
45template <typename Model, typename Action>
46void guardAction(Model const& model, Action const& action) {
47 if constexpr (DeclaresGuard<Action>) {
48 using Guard = typename Action::guard;
49 // A hard error, never a false: an execution site compiled without the guard's definition must not compile,
50 // and the form checks below are only stable once the type is complete.
51 // NOLINTNEXTLINE(bugprone-sizeof-expression): sizeof is the completeness test here, not a size.
52 static_assert(sizeof(Guard) > 0, "an action's guard must be complete where its execution sites are compiled");
53 constexpr bool payloadForm = GuardChecksAction<Guard, Action>;
54 constexpr bool modelForm = GuardChecksInModel<Guard, Model, Action>;
55 static_assert(payloadForm != modelForm,
56 "an action's guard provides exactly one callable check for this model: "
57 "check(Action const&) or check(Model const&, Action const&)");
58 if constexpr (modelForm) {
59 Guard::check(model, action);
60 } else {
61 Guard::check(action);
62 }
63 } else {
64 static_cast<void>(model);
65 static_cast<void>(action);
66 }
67}
