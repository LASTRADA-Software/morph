91/// @brief The catalog as a list a person opens screens from, with a title search and pinned shortcuts.
92///
93/// Entries are grouped by module, then group, in the order each first appears; each group is a column of its own
94/// holding its heading and its entries, and the groups are a spacing step apart. Each entry is a row as wide as the
95/// navigator holding a list item (`ButtonStyle::Item`) whose content is the title and, while `setBadge` gives it one,
96/// a badge at its end, so the item's selection spans the badge; `setCurrent` marks the open screen's. An unavailable
97/// entry is listed disabled, with its reason as its tooltip. The search shows the entries one of whose titles contains
98/// the text, ignoring ASCII case, and the headings of the groups that still show one. `menuEntries` and `choose` give
99/// the same list as a menu, for a frame too narrow for the navigator. Neither copyable nor movable; owned by the owner
100/// thread of its backend.
101///
102/// Test ids: `navigator`, `navigator_search`, `navigator_pins`, `nav_row_<screen id>`, `nav_<screen id>`,
103/// `nav_content_<screen id>`, `nav_badge_<screen id>`, `pin_<screen id>`, `navigator_group_<module>/<group>` for a
104/// heading and `navigator_group_<module>/<group>_section` for its column.

340 // One search for both constnesses of itemOf: `Groups` is the const or the mutable group list, and the item
341 // pointer it returns carries the same constness.
342 template <typename Groups>
343 [[nodiscard]] static auto itemIn(Groups& groups, std::string_view screenId)
344 -> decltype(&groups.front().items.front()) {
345 for (auto& group : groups) {
346 auto const found = std::ranges::find_if(
347 group.items, [screenId](Item const& item) { return !screenId.empty() && item.entry->id == screenId; });
348 if (found != group.items.end()) {
349 return &*found;
350 }
351 }
352 return nullptr;
353 }
