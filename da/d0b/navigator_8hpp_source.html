40/// @brief How a `Navigator` lists the catalog and what it reports.
41struct NavigatorOptions {
42 /// @brief The catalog's entries, in the order the navigator lists them within a group.
43 std::vector<CatalogEntry> entries;
44 /// @brief An entry's titles, each one a search can match: one per locale (spec 6 §19). Empty uses the entry's
45 /// title, a string or a message reference's `fallback`. The first title is the one shown.

60/// @brief The catalog as a list a person opens screens from, with a title search and pinned shortcuts.
61///
62/// Entries are grouped by module, then group, in the order each first appears; each group is a column of its own
63/// holding its heading and its entries, and the groups are a spacing step apart. Each entry is a row as wide as the
64/// navigator holding a list item (`ButtonStyle::Item`) whose content is the title and, while `setBadge` gives it one,
65/// a badge at its end, so the item's selection spans the badge; `setCurrent` marks the open screen's. An unavailable
66/// entry is listed disabled, with its reason as its tooltip. The search shows the entries one of whose titles contains
67/// the text, ignoring ASCII case, and the headings of the groups that still show one. `menuEntries` and `choose` give
68/// the same list as a menu, for a frame too narrow for the navigator. Neither copyable nor movable; owned by the owner
69/// thread of its backend.
70///
71/// Test ids: `navigator`, `navigator_search`, `navigator_pins`, `nav_row_<screen id>`, `nav_<screen id>`,
72/// `nav_content_<screen id>`, `nav_badge_<screen id>`, `pin_<screen id>`, `navigator_group_<module>/<group>` for a
73/// heading and `navigator_group_<module>/<group>_section` for its column.
74class Navigator {
75public:
76 /// @brief The navigator, appended to @p parent.
77 /// @param backend The backend its widgets are made with. Borrowed.
78 /// @param parent The container it is appended to; null makes it a backend root.
79 /// @param options The entries, the pins and the handlers.
80 Navigator(IViewBackend& backend MORPH_LIFETIMEBOUND, ContainerWidget* parent, NavigatorOptions options)
81 : _backend{&backend}, _options{std::move(options)}, _root{backend.createStack(parent, Axis::Vertical)} {
82 _root->setTestId("navigator");
83 // The search, the pins and the groups stand a spacing step apart; a hidden pin row or group takes no gap.
84 _root->setGap(3);
85 _search = backend.createTextInput(_root.get(), TextInputMode::SingleLine);
86 _search->setTestId("navigator_search");
87 _search->setPlaceholder("Search");
88 _search->setAccessibleName("Search screens");
89 _search->setOnChange([this](std::string const& text) { filter(text); });
90 _pinRow = backend.createStack(_root.get(), Axis::Horizontal);
91 _pinRow->setTestId("navigator_pins");
92 listEntries();
93 std::vector<std::string> stored = std::move(_options.pins);
94 _options.pins.clear();
95 for (std::string& screenId : stored) {
96 if (_pins.size() < _options.maxPins && known(screenId) && !pinned(screenId)) {
97 _pins.push_back(std::move(screenId));
98 }
99 }
100 showPins();
101 }

166 /// @brief Puts a count at the end of a screen's entry, or takes it away. The entry stays one button to assistive
167 /// technology, and its accessible name carries the count: "Review, 4".
168 /// @param screenId The screen whose entry it goes on.
169 /// @param badge The count, or `std::nullopt` to take it away.
170 /// @return Whether the navigator lists the screen: `false`, changing nothing, for one it does not list.
171 bool setBadge(std::string_view screenId, std::optional<NavigatorBadge> const& badge) {
172 Item* const item = itemOf(screenId);
173 if (item == nullptr) {
174 return false;
175 }
176 if (item->shown == badge) {
177 return true;
178 }
179 item->shown = badge;
180 std::string const& title = item->titles.front();
181 if (badge) {
182 item->badge->setText(badge->text);
183 item->badge->setTone(badge->tone);
184 item->button->setAccessibleName(title + ", " + badge->text);
185 } else {
186 item->button->setAccessibleName(title);
187 }
188 item->badge->setVisible(badge.has_value());
189 changed();
190 return true;
191 }

201 /// @brief The navigator as the entries of a menu: the pins first, then a separator, then each group as a
202 /// submenu labelled by its heading (the entries of a group with no heading at the top level instead).
203 /// An entry's label is its title with its count after it ("Review · 4"), the current screen's entry is
204 /// checked, and an unavailable one is disabled.
205 /// @return The entries; `choose` takes a path into them.
206 [[nodiscard]] std::vector<MenuEntry> menuEntries() const { return menu().entries; }
207
208 /// @brief Opens the entry at @p path of `menuEntries()`, as choosing it in the navigator does.
209 /// @param path The entry's index at each level from the top, separators counted.
210 /// @return Whether it named an available entry, which was opened.
211 bool choose(std::vector<std::size_t> const& path) {
212 Menu const shown = menu();
213 std::vector<MenuSlot> const* level = &shown.targets;
214 MenuSlot const* slot = nullptr;
215 for (std::size_t const index : path) {
216 if (index >= level->size()) {
217 return false;
218 }
219 slot = &level->at(index);
220 level = &slot->items;
221 }
222 if (slot == nullptr || slot->screenId.empty() || !slot->items.empty()) {
223 return false;
224 }
225 CatalogEntry const* const entry = entryOf(slot->screenId);
226 if (entry == nullptr || !entry->available) {
227 return false;
228 }
229 if (_options.onOpen) {
230 _options.onOpen(slot->screenId);
231 }
232 return true;
233 }

272 // One entry of the menu: its title, then its count.
273 [[nodiscard]] MenuEntry menuEntry(CatalogEntry const& entry) const {
274 MenuEntry made;
275 made.label = titlesOf(entry).front();
276 if (Item const* const item = itemOf(entry.id); item != nullptr && item->shown) {
277 made.label += " · " + item->shown->text;
278 }
279 if (entry.id == _current) {
280 made.checked = true;
281 }
282 made.enabled = entry.available;
283 return made;
284 }

286 [[nodiscard]] Menu menu() const {
287 Menu out;
288 for (std::string const& screenId : _pins) {
289 out.entries.push_back(menuEntry(*entryOf(screenId)));
290 out.targets.push_back(MenuSlot{.screenId = screenId, .items = {}});
291 }
292 if (!_pins.empty() && !_groups.empty()) {
293 out.entries.push_back(MenuEntry{.label = {},
294 .icon = {},
295 .keys = {},
296 .checked = std::nullopt,
297 .enabled = true,
298 .items = {},
299 .separator = true,
300 .description = {}});
301 out.targets.push_back(MenuSlot{});
302 }
303 for (Group const& group : _groups) {
304 std::string const heading = headingOf(group);
305 MenuEntry submenu;
306 MenuSlot slot;
307 for (Item const& item : group.items) {
308 (heading.empty() ? out.entries : submenu.items).push_back(menuEntry(*item.entry));
309 (heading.empty() ? out.targets : slot.items)
310 .push_back(MenuSlot{.screenId = item.entry->id, .items = {}});
311 }
312 if (!heading.empty()) {
313 submenu.label = heading;
314 out.entries.push_back(std::move(submenu));
315 out.targets.push_back(std::move(slot));
316 }
317 }
318 return out;
319 }

364 // One search for both constnesses of itemOf: `Groups` is the const or the mutable group list, and the item
365 // pointer it returns carries the same constness.
366 template <typename Groups>
367 [[nodiscard]] static auto itemIn(Groups& groups, std::string_view screenId)
368 -> decltype(&groups.front().items.front()) {
369 for (auto& group : groups) {
370 auto const found = std::ranges::find_if(
371 group.items, [screenId](Item const& item) { return !screenId.empty() && item.entry->id == screenId; });
372 if (found != group.items.end()) {
373 return &*found;
374 }
375 }
376 return nullptr;
377 }

404 void listEntries() {
405 for (CatalogEntry const& entry : _options.entries) {
406 bool const seen = std::ranges::any_of(_groups, [&entry](Group const& candidate) {
407 return candidate.module == entry.module && candidate.group == entry.group;
408 });
409 if (!seen) {
410 _groups.push_back(
411 Group{.module = entry.module, .group = entry.group, .section = {}, .heading = {}, .items = {}});
412 }
413 }
414 for (Group& group : _groups) {
415 // The heading and its entries are a column of their own, a step apart from the next group's; the heading
416 // sits a step under whatever precedes it, and the entries' own heights keep them apart.
417 group.section = _backend->createStack(_root.get(), Axis::Vertical);
418 group.section->setGap(1);
419 group.section->setTestId("navigator_group_" + group.module + "/" + group.group + "_section");
420 group.heading = _backend->createText(group.section.get());
421 // An overline, which a theme may draw as small capitals; by default it looks as a section does.
422 group.heading->setRole(TextRole::Overline);
423 group.heading->setTestId("navigator_group_" + group.module + "/" + group.group);
424 group.heading->setText(headingOf(group));
425 for (CatalogEntry const& entry : _options.entries) {
426 if (entry.module != group.module || entry.group != group.group) {
427 continue;
428 }
429 Item item{.entry = &entry,
430 .titles = titlesOf(entry),
431 .row = _backend->createStack(group.section.get(), Axis::Horizontal),
432 .button = {},
433 .content = {},
434 .label = {},
435 .badge = {},
436 .shown = std::nullopt};
437 item.row->setTestId("nav_row_" + entry.id);
438 item.row->setLayout({.width = Sizing::stretch(), .height = {}});
439 item.row->setAlign(Align::Center);
440 item.button = _backend->createButton(item.row.get());
441 item.button->setTestId("nav_" + entry.id);
442 item.button->setStyle(ButtonStyle::Item);
443 item.button->setLayout({.width = Sizing::stretch(), .height = {}});
444 wire(*item.button, entry, item.titles.front());
445 // The title and the badge are the button's content, so its selection spans both. The badge is hidden
446 // until a count is set; the button's accessible name carries the count, so the entry stays one
447 // control to assistive technology.
448 ContainerWidget& content = item.button->content();
449 item.content = _backend->createStack(&content, Axis::Horizontal);
450 item.content->setTestId("nav_content_" + entry.id);
451 item.content->setGap(1);
452 item.content->setAlign(Align::Center);
453 item.content->setLayout({.width = Sizing::stretch(), .height = {}});
454 item.label = _backend->createText(item.content.get());
455 item.label->setText(item.titles.front());
456 item.label->setLayout({.width = Sizing::stretch(), .height = {}});
457 item.badge = _backend->createBadge(item.content.get());
458 item.badge->setTestId("nav_badge_" + entry.id);
459 item.badge->setVisible(false);
460 group.items.push_back(std::move(item));
461 }
462 }
463 }

465 void filter(std::string const& text) {
466 std::string const wanted = lower(text);
467 for (Group& group : _groups) {
468 bool any = false;
469 for (Item& item : group.items) {
470 bool const shown =
471 wanted.empty() || std::ranges::any_of(item.titles, [&wanted](std::string const& title) {
472 return lower(title).contains(wanted);
473 });
474 item.row->setVisible(shown);
475 any = any || shown;
476 }
477 group.section->setVisible(any);
478 }
479 }

481 void showPins() {
482 _pinButtons.clear();
483 for (std::string const& screenId : _pins) {
484 CatalogEntry const* const entry = entryOf(screenId);
485 auto button = _backend->createButton(_pinRow.get());
486 button->setTestId("pin_" + screenId);
487 wire(*button, *entry, titlesOf(*entry).front());
488 _pinButtons.push_back(std::move(button));
489 }
490 }
