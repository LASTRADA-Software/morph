83/// @brief Connects the card document and `tour.start` to the frame's controller through a `HostVocabulary`.
84///
85/// For the card document alone (`kTourCardId`): the values `tour.title`, `tour.stepTitle`, `tour.body`,
86/// `tour.waitingFor` (strings), `tour.position`, `tour.count` (ints), `tour.waiting`, `tour.canBack`, `tour.last`
87/// (flags), and the commands `tour.next`, `tour.back` and `tour.end`. For every document: `tour.start` with `{id}`,
88/// refused, and reported by the interpreter, for an id the app offers no tour for. Every command is safe for an
89/// untrusted connection: it changes nothing on the server and shows only what the application declared. The commands'
90/// work is posted, since a step can open a screen and a screen must not be replaced inside one of its widgets'
91/// handlers; when the posted `tour.start` finds no controller or a controller that refuses the id (the frame was
92/// rebuilt after the refusal check), that is reported too. A shell makes one binding, which outlives frames, and calls
93/// `refresh` after each change of the controller's view. The registered commands stay in the host vocabulary after the
94/// binding is gone, and posted work may still be queued: both check a token the destructor clears, so after
95/// destruction a command does nothing and a posted command's work is dropped, and a refusal check allows nothing.
96/// Neither copyable nor movable; on the runtime's owner.
97class TourCardBinding {
98public:
99 /// @brief Registers the values and the commands.
100 /// @param host The vocabulary. Borrowed: it must outlive this binding's constructor and `refresh` calls; its
101 /// commands may outlive the binding.
102 /// @param controller The frame's controller, read at each command and each `refresh`; null between frames.
103 /// @param post Runs a command's work after the handler that asked returns; empty runs it at once.
104 /// @param report Called with the sentence for the developer when a posted `tour.start` could not start its
105 /// tour; empty logs it at warn.
106 TourCardBinding(HostVocabulary& host, std::function<TourController*()> controller,
107 std::function<void(std::function<void()>)> post = {},
108 std::function<void(std::string const&)> report = {})
109 : _host{&host}, _controller{std::move(controller)}, _post{std::move(post)}, _report{std::move(report)} {
110 std::set<std::string, std::less<>> const card{std::string{kTourCardId}};
111 auto const addValues = [&host, &card](std::initializer_list<std::string_view> names, std::string const& type,
112 Value const& initial) {
113 for (std::string_view const name : names) {
114 host.add(HostValue{
115 .name = std::string{name},
116 .version = 1,
117 .type = type,
118 .initial = initial,
119 .documents = card,
120 });
121 }
122 };
123 addValues({"tour.title", "tour.stepTitle", "tour.body", "tour.waitingFor"}, "string", Value{std::string{}});
124 addValues({"tour.position", "tour.count"}, "int", Value{0});
125 addValues({"tour.waiting", "tour.canBack", "tour.last"}, "bool", Value{false});
126 auto const command = [this, &card](std::string name, std::function<void(TourController&)> work) {
127 return HostCommand{
128 .name = std::move(name),
129 .version = 1,
130 .args = {},
131 .safeForUntrusted = true,
132 .run =
133 [this, token = std::weak_ptr<bool>{_alive}, work = std::move(work)](ValueRecord const& /*args*/) {
134 if (auto const live = token.lock(); live != nullptr && *live) {
135 later(work);
136 }
137 },
138 .documents = card,
139 .refusal = {}};
140 };
141 host.add(command("tour.next", [](TourController& tours) { tours.next(); }));
142 host.add(command("tour.back", [](TourController& tours) { tours.back(); }));
143 host.add(command("tour.end", [](TourController& tours) { tours.end(); }));
144 host.add(startCommand());
145 refresh();
146 }
147
148 TourCardBinding(TourCardBinding const&) = delete;
149 TourCardBinding& operator=(TourCardBinding const&) = delete;
150 TourCardBinding(TourCardBinding&&) = delete;
151 TourCardBinding& operator=(TourCardBinding&&) = delete;
152 /// @brief Makes the commands it registered, and work it posted, do nothing from now on.
153 ~TourCardBinding() { *_alive = false; }
154
155 /// @brief Copies the controller's view into the values; with no tour running, every value is back to its initial
156 /// one.
157 void refresh() {
158 TourController const* const tours = _controller ? _controller() : nullptr;
159 TourView const shown = tours != nullptr && tours->view() ? *tours->view() : TourView{};
160 _host->set("tour.title", Value{shown.title});
161 _host->set("tour.stepTitle", Value{shown.stepTitle});
162 _host->set("tour.body", Value{shown.body});
163 _host->set("tour.waitingFor", Value{shown.waitingFor});
164 _host->set("tour.position", Value{shown.position});
165 _host->set("tour.count", Value{shown.count});
166 _host->set("tour.waiting", Value{shown.waiting});
167 _host->set("tour.canBack", Value{shown.canBack});
168 _host->set("tour.last", Value{shown.last});
169 }
170
171private:
172 // Runs @p work on the controller of the moment it runs, posted when a post was given; @p missing runs instead when
173 // there is none then.
174 void later(std::function<void(TourController&)> work, std::function<void()> missing = {}) {
175 auto run = [this, token = std::weak_ptr<bool>{_alive}, work = std::move(work), missing = std::move(missing)] {
176 if (auto const live = token.lock(); live == nullptr || !*live) {
177 return;
178 }
179 if (TourController* const tours = _controller ? _controller() : nullptr) {
180 work(*tours);
181 } else if (missing) {
182 missing();
183 }
184 };
185 if (_post) {
186 _post(std::move(run));
187 } else {
188 run();
189 }
190 }
191
192 // `tour.start`: every document may use it, with the id of a tour of the app document.
193 [[nodiscard]] HostCommand startCommand() {
194 auto const run = [this, token = std::weak_ptr<bool>{_alive}](ValueRecord const& args) {
195 if (auto const live = token.lock(); live == nullptr || !*live) {
196 return;
197 }
198 std::string const tourId = *args.at("id").asString();
199 std::function<void(TourController&)> work = [this, tourId](TourController& tours) {
200 if (auto const started = tours.start(tourId); !started) {
201 tell(started.error().message());
202 }
203 };
204 std::function<void()> missing = [this, tourId] {
205 tell("the tour `" + tourId + "` did not start: no frame is shown");
206 };
207 later(std::move(work), std::move(missing));
208 // NOLINTNEXTLINE(clang-analyzer-cplusplus.NewDeleteLeaks): the analyzer loses the std::function blocks.
209 };
210 auto const refusal =
211 [this, token = std::weak_ptr<bool>{_alive}](ValueRecord const& args) -> std::optional<std::string> {
212 if (auto const live = token.lock(); live == nullptr || !*live) {
213 return std::string{"no tour can start now: the window is gone"};
214 }
215 TourController const* const tours = _controller ? _controller() : nullptr;
216 std::string const& tourId = *args.at("id").asString();
217 if (tours == nullptr) {
218 return std::string{"no tour can start now: no frame is shown"};
219 }
220 if (!tours->declares(tourId)) {
221 return "the app offers no tour `" + tourId + "`";
222 }
223 return std::nullopt;
224 };
225 return HostCommand{
226 .name = "tour.start",
227 .version = 1,
228 .args = {HostArg{.name = "id", .type = "string"}},
229 .safeForUntrusted = true,
230 .run = run,
231 .documents = {},
232 .refusal = refusal,
233 };
234 }
235
236 // Reports a posted `tour.start` that could not start its tour: to the shell's callback, or at warn without one.
237 void tell(std::string const& text) const {
238 if (_report) {
239 _report(text);
240 } else {
241 morph::log::logWarn("[tours] {}", text);
242 }
243 }
244
245 HostVocabulary* _host;
246 std::function<TourController*()> _controller;
247 std::function<void(std::function<void()>)> _post;
248 std::function<void(std::string const&)> _report;
249 // Cleared by the destructor; the commands and posted work hold it weakly.
250 std::shared_ptr<bool> _alive = std::make_shared<bool>(true);
251};
