28/// @brief A dotted numeric version such as `6.11.0`, ordered component by component.
29///
30/// The value means nothing to morph apart from its order. Components are compared from the left as unsigned numbers,
31/// and a missing trailing component counts as 0, so `6.11` and `6.11.0` are equal and `6.11` is below `6.11.1`. A
32/// default-constructed version is the **first version**, written `0`: what a field or an element without a `since`
33/// belongs to, and below every version that has a non-zero component.
34class FieldVersion {
35public:
36 /// @brief The most bytes a version's text may have, so a version quoted in a refusal stays short.
37 static constexpr std::size_t kMaxTextBytes = 64;
38
39 /// @brief The first version, `0`.
40 FieldVersion() = default;
41
42 /// @brief Parses a dotted numeric version.
43 ///
44 /// The text is one or more components separated by single dots. Each component is decimal digits, without a
45 /// leading zero unless it is `0` itself (so `6.01` cannot silently equal `6.1`), and at most 4294967295. The
46 /// whole text is at most `kMaxTextBytes` bytes. Anything else is refused.
47 /// @param text The version as written, e.g. `"6.11.0"`.
48 /// @return The version, or why @p text is not one, a sentence that quotes it.
49 [[nodiscard]] static std::expected<FieldVersion, std::string> parse(std::string_view text) {
50 if (text.empty()) {
51 return std::unexpected{std::string{"a version must not be empty; write dotted numbers such as 6.11.0"}};
52 }
53 if (text.size() > kMaxTextBytes) {
54 return std::unexpected{"a version is at most " + std::to_string(kMaxTextBytes) + " bytes; this one has " +
55 std::to_string(text.size())};
56 }
57 FieldVersion version;
58 version._text = std::string{text};
59 std::string_view rest = text;
60 while (true) {
61 auto const dot = rest.find('.');
62 std::string_view const component = rest.substr(0, dot);
63 auto parsed = parseComponent(text, component);
64 if (!parsed.has_value()) {
65 return std::unexpected{std::move(parsed.error())};
66 }
67 version._components.push_back(*parsed);
68 if (dot == std::string_view::npos) {
69 break;
70 }
71 rest.remove_prefix(dot + 1);
72 }
73 return version;
74 }
75
76 /// @brief The version as written; `0` for the first version.
77 /// @return The text this version was parsed from.
78 [[nodiscard]] std::string const& text() const noexcept { return _text; }
79
80 /// @brief The numeric components, as written (trailing zeros kept); empty for the first version.
81 /// @return The components, most significant first.
82 [[nodiscard]] std::span<std::uint32_t const> components() const noexcept { return _components; }
83
84 /// @brief Orders two versions component by component, a missing component counting as 0.
85 /// @param lhs The left version.
86 /// @param rhs The right version.
87 /// @return How @p lhs orders against @p rhs.
88 friend std::strong_ordering operator<=>(FieldVersion const& lhs, FieldVersion const& rhs) noexcept {
89 std::size_t const count = std::max(lhs._components.size(), rhs._components.size());
90 for (std::size_t index = 0; index < count; ++index) {
91 std::uint32_t const left = index < lhs._components.size() ? lhs._components.at(index) : 0U;
92 std::uint32_t const right = index < rhs._components.size() ? rhs._components.at(index) : 0U;
93 if (left != right) {
94 return left <=> right;
95 }
96 }
97 return std::strong_ordering::equal;
98 }
99
100 /// @brief Whether two versions are the same version (`6.11` equals `6.11.0`).
101 /// @param lhs The left version.
102 /// @param rhs The right version.
103 /// @return `true` when neither orders before the other.
104 friend bool operator==(FieldVersion const& lhs, FieldVersion const& rhs) noexcept {
105 return (lhs <=> rhs) == std::strong_ordering::equal;
106 }
107
108private:
109 /// @brief Parses one component of @p whole.
110 /// @param whole The whole version text, quoted in a refusal.
111 /// @param component The component's digits.
112 /// @return The component's value, or why it is not one.
113 // NOLINTBEGIN(bugprone-easily-swappable-parameters): the whole text, then the component of it.
114 [[nodiscard]] static std::expected<std::uint32_t, std::string> parseComponent(std::string_view whole,
115 std::string_view component) {
116 // NOLINTEND(bugprone-easily-swappable-parameters)
117 auto const refuse = [whole](std::string_view why) {
118 return std::unexpected{"'" + std::string{whole} + "' is not a version: " + std::string{why} +
119 "; write dotted numbers such as 6.11.0"};
120 };
121 if (component.empty()) {
122 return refuse("it has an empty component");
123 }
124 if (component.size() > 1 && component.front() == '0') {
125 return refuse("a component has a leading zero");
126 }
127 std::uint32_t value = 0;
128 char const* const last = std::to_address(component.end());
129 auto const [end, error] = std::from_chars(std::to_address(component.begin()), last, value);
130 if (error == std::errc::result_out_of_range) {
131 return refuse("a component is above 4294967295");
132 }
133 if (error != std::errc{} || end != last) {
134 return refuse("a component is not decimal digits");
135 }
136 return value;
137 }
138
139 std::vector<std::uint32_t> _components;
140 std::string _text{"0"};
141};

143/// @brief Reads a `since` as written in a declaration or a document: empty is the first version.
144/// @param text The text, possibly empty.
145/// @return The version, or why @p text is not one.
146[[nodiscard]] inline std::expected<FieldVersion, std::string> parseSince(std::string_view text) {
147 if (text.empty()) {
148 return FieldVersion{};
149 }
150 return FieldVersion::parse(text);
151}

153/// @brief One field of an action's inventory: its wire name and the version it first shipped in.
154///
155/// A plain aggregate, so an application can keep a snapshot as JSON with glaze as it is.
156struct FieldInventoryEntry {
157 /// @brief The field's wire name.
158 std::string field;
159 /// @brief The version the field first ships in, as written; empty for the first version.
160 std::string since;
161 /// @brief Whether the field was removed. Only a snapshot keeps removed fields, so a name is never used twice.
162 bool removed = false;
163};
164
165/// @brief What is wrong with an action's fields against the snapshot of the last release.
166enum class FieldInventoryProblemKind : std::uint8_t {
167 /// An entry's `since` is not a version.
168 InvalidSince,
169 /// A field that is not in the snapshot has no `since` above the snapshot's highest.
170 AddedWithoutNewerSince,
171 /// A field of the snapshot declares another `since` than it shipped with.
172 SinceChanged,
173 /// A field the snapshot records as removed is back: a wire name is never reused.
174 Reappeared,
175};
176
177/// @brief One problem `checkFieldInventory` found.
178struct FieldInventoryProblem {
179 /// @brief What kind of problem it is.
180 FieldInventoryProblemKind kind{FieldInventoryProblemKind::InvalidSince};
181 /// @brief The field it is about.
182 std::string field;
183 /// @brief A sentence for a test's failure output, naming the field and the versions.
184 std::string message;
185};

189/// @brief Finds the entry named @p field in @p entries.
190/// @param entries The inventory to search.
191/// @param field The wire name.
192/// @return The entry, or `nullptr` when none is named @p field.
193[[nodiscard]] inline FieldInventoryEntry const* findInventoryEntry(std::span<FieldInventoryEntry const> entries,
194 std::string_view field) noexcept {
195 auto const found =
196 std::ranges::find_if(entries, [field](FieldInventoryEntry const& entry) { return entry.field == field; });
197 return found == entries.end() ? nullptr : &*found;
198}

202/// @brief Checks an action's current fields against the snapshot of its last release.
203///
204/// The application keeps the snapshot (`nextFieldInventory` gives the one to keep at a release) and runs this in its
205/// own tests; morph keeps none. A problem is reported for:
206/// - an entry of either list whose `since` is not a version (`InvalidSince`);
207/// - a current field absent from the snapshot whose `since` is not above the highest `since` the snapshot records,
208/// removed fields included (`AddedWithoutNewerSince`); an empty snapshot is the first release and checks no
209/// addition;
210/// - a current field the snapshot holds whose `since` is another version (`SinceChanged`);
211/// - a current field the snapshot records as removed (`Reappeared`).
212///
213/// A field of the snapshot that is gone from @p current is a removal, which is allowed. An entry of @p current marked
214/// `removed` is not a field of the action and is skipped.
215/// @param previous The snapshot of the last release, removed fields included.
216/// @param current The action's fields now, `fieldInventory<A>()`.
217/// @return Every problem, in the order of @p current (invalid snapshot entries first); empty when there is none.
218// NOLINTBEGIN(bugprone-easily-swappable-parameters): the snapshot, then the fields now, as the check reads them.
219[[nodiscard]] inline std::vector<FieldInventoryProblem> checkFieldInventory(
220 std::span<FieldInventoryEntry const> previous, std::span<FieldInventoryEntry const> current) {
221 // NOLINTEND(bugprone-easily-swappable-parameters)
222 std::vector<FieldInventoryProblem> problems;
223 FieldVersion highest;
224 for (FieldInventoryEntry const& entry : previous) {
225 auto const since = parseSince(entry.since);
226 if (!since.has_value()) {
227 problems.push_back({.kind = FieldInventoryProblemKind::InvalidSince,
228 .field = entry.field,
229 .message = "the snapshot's '" + entry.field + "': " + since.error()});
230 continue;
231 }
232 highest = std::max(highest, *since);
233 }
234 for (FieldInventoryEntry const& entry : current) {
235 if (entry.removed) {
236 continue;
237 }
238 auto const since = parseSince(entry.since);
239 if (!since.has_value()) {
240 problems.push_back({.kind = FieldInventoryProblemKind::InvalidSince,
241 .field = entry.field,
242 .message = "'" + entry.field + "': " + since.error()});
243 continue;
244 }
245 FieldInventoryEntry const* const shipped = detail::findInventoryEntry(previous, entry.field);
246 if (shipped == nullptr) {
247 if (!previous.empty() && *since <= highest) {
248 problems.push_back({.kind = FieldInventoryProblemKind::AddedWithoutNewerSince,
249 .field = entry.field,
250 .message = "'" + entry.field + "' is new, so its since must be above " +
251 highest.text() + ", the highest the last release had; it is " +
252 since->text()});
253 }
254 continue;
255 }
256 if (shipped->removed) {
257 problems.push_back({.kind = FieldInventoryProblemKind::Reappeared,
258 .field = entry.field,
259 .message = "'" + entry.field +
260 "' was removed and is back; a wire name is never reused, so give the "
261 "field a new name"});
262 continue;
263 }
264 auto const before = parseSince(shipped->since);
265 if (before.has_value() && *before != *since) {
266 problems.push_back({.kind = FieldInventoryProblemKind::SinceChanged,
267 .field = entry.field,
268 .message = "'" + entry.field + "' shipped with since " + before->text() +
269 " and now declares " + since->text() + "; a field's since never changes"});
270 }
271 }
272 return problems;
273}

275/// @brief The snapshot to keep at a release: the current fields, then every field of @p previous that @p current
276/// no longer has, marked removed, so a later `checkFieldInventory` catches a removed name coming back.
277/// @param previous The snapshot of the last release.
278/// @param current The action's fields now, `fieldInventory<A>()`.
279/// @return The new snapshot.
280// NOLINTBEGIN(bugprone-easily-swappable-parameters): the snapshot, then the fields now, as the check reads them.
281[[nodiscard]] inline std::vector<FieldInventoryEntry> nextFieldInventory(
282 std::span<FieldInventoryEntry const> previous, std::span<FieldInventoryEntry const> current) {
283 // NOLINTEND(bugprone-easily-swappable-parameters)
284 std::vector<FieldInventoryEntry> next{current.begin(), current.end()};
285 for (FieldInventoryEntry const& entry : previous) {
286 if (detail::findInventoryEntry(current, entry.field) == nullptr) {
287 next.push_back({.field = entry.field, .since = entry.since, .removed = true});
288 }
289 }
290 return next;
291}
