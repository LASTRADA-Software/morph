12/// @brief Severity levels for the logging system.
13enum class LogLevel : std::uint8_t {
14 /// @brief Fine-grained diagnostic output.
15 debug = 0,
16 /// @brief General informational messages.
17 info = 1,
18 /// @brief Recoverable conditions worth noting.
19 warn = 2,
20 /// @brief Errors that should be investigated.
21 error = 3,
22 /// @brief Suppresses all output when used as the minimum level.
23 off = 4,
24};

26namespace detail {
27
28/// @brief Returns the human-readable name for a log level.
29constexpr std::string_view levelName(LogLevel level) noexcept {
30 switch (level) {
31 case LogLevel::debug:
32 return "DEBUG";
33 case LogLevel::info:
34 return "INFO ";
35 case LogLevel::warn:
36 return "WARN ";
37 case LogLevel::error:
38 return "ERROR";
39 case LogLevel::off:
40 return "OFF ";
41 }
42 return "? ";
43}
44
45/// @brief Sink function type used internally for log output.
46using Logger = std::function<void(LogLevel, std::string_view)>;
47
48struct LogState {
49 Logger sink = [](LogLevel lvl, std::string_view msg) { std::println(stderr, "[{}] {}", levelName(lvl), msg); };
50 LogLevel minLevel = LogLevel::debug;
51 std::mutex mtx;
52};
53
54inline LogState& logState() {
55 static LogState state;
56 return state;
57}
58
59/// @brief Internal emit entry point used by the public level helpers.
60inline void log(LogLevel level, std::string_view msg) {
61 std::scoped_lock const lock{logState().mtx};
62 auto& state = logState();
63 if (state.sink && level >= state.minLevel) {
64 state.sink(level, msg);
65 }
66}

68} // namespace detail

72/// @brief Replaces the global log sink.
73///
74/// Thread-safe. The new sink is called for every message whose level meets the
75/// current minimum. Pass a no-op lambda to silence all output.
76/// @param logger New sink function.
77inline void setLogger(std::function<void(LogLevel, std::string_view)> logger) {
78 std::scoped_lock const lock{detail::logState().mtx};
79 detail::logState().sink = std::move(logger);
80}
81
82/// @brief Sets the minimum log level.
83///
84/// Messages below this level are silently dropped. Thread-safe.
85/// @param level Minimum level to emit.
86inline void setLogLevel(LogLevel level) {
87 std::scoped_lock const lock{detail::logState().mtx};
88 detail::logState().minLevel = level;
89}
90
91/// @brief Returns the current minimum log level. Thread-safe.
92/// @return The active minimum level.
93inline LogLevel getLogLevel() {
94 std::scoped_lock const lock{detail::logState().mtx};
95 return detail::logState().minLevel;
96}

142/// @brief RAII helper that swaps the global logger and level for the lifetime
143/// of the object, and restores them in the destructor.
144///
145/// Designed for tests that want to capture log output without leaking the
146/// custom sink into other tests. Thread-safe construction and destruction —
147/// the global mutex is acquired briefly during each.
148///
149/// @par Example
150/// @code
151/// {
152/// std::vector<std::string> captured;
153/// morph::log::ScopedLoggerOverride guard{
154/// [&](morph::log::LogLevel, std::string_view msg) { captured.emplace_back(msg); },
155/// morph::log::LogLevel::debug,
156/// };
157/// // ... run code that logs ...
158/// } // previous sink + level restored here
159/// @endcode
160class ScopedLoggerOverride {
161public:
162 /// @brief Snapshots the current sink + level so they can be restored.
163 ///
164 /// Use this when test code will install its own sink mid-test via
165 /// `setLogger()` / `setLogLevel()` and just wants automatic restoration.
166 ScopedLoggerOverride() : _savedSink(detail::logState().sink), _savedLevel(detail::logState().minLevel) {
167 std::scoped_lock const lock{detail::logState().mtx};
168
169
170 }
171
172 /// @brief Installs @p sink and @p level; saves whatever was there before.
173 ///
174 /// @param sink New logger sink. Pass a no-op lambda to suppress output entirely.
175 /// @param level New minimum level. Defaults to `debug` so every message reaches @p sink.
176 explicit ScopedLoggerOverride(std::function<void(LogLevel, std::string_view)> sink,
177 LogLevel level = LogLevel::debug) : _savedSink(std::move(detail::logState().sink)), _savedLevel(detail::logState().minLevel) {
178 std::scoped_lock const lock{detail::logState().mtx};
179
180
181 detail::logState().sink = std::move(sink);
182 detail::logState().minLevel = level;
183 }
184
185 /// @brief Restores the saved sink and level.
186 ~ScopedLoggerOverride() {
187 std::scoped_lock const lock{detail::logState().mtx};
188 detail::logState().sink = std::move(_savedSink);
189 detail::logState().minLevel = _savedLevel;
190 }
191
192 ScopedLoggerOverride(const ScopedLoggerOverride&) = delete;
193 ScopedLoggerOverride& operator=(const ScopedLoggerOverride&) = delete;
194 ScopedLoggerOverride(ScopedLoggerOverride&&) = delete;
195 ScopedLoggerOverride& operator=(ScopedLoggerOverride&&) = delete;
196
197private:
198 std::function<void(LogLevel, std::string_view)> _savedSink;
199 LogLevel _savedLevel{LogLevel::debug};
200};
