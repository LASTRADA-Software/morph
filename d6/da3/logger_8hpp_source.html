15/// @brief Severity levels for the logging system.

48/// @brief Sink function type used internally for log output.

51/// @brief Escapes newline and control characters so a message cannot forge log
52/// lines (log injection).
53///
54/// A message is a single logical record; the default sink emits it as one line.
55/// User-controlled text containing `\n`, `\r`, or other C0 control bytes could
56/// otherwise splice a forged `[ERROR] ...` line into the stream or corrupt a
57/// line-oriented log parser. This replaces CR/LF/TAB with their C-style escapes
58/// (`\n`, `\r`, `\t`) and any remaining control byte (`< 0x20`, or `0x7f` DEL)
59/// with a `\xHH` escape, leaving printable text (including non-ASCII UTF-8
60/// continuation bytes `>= 0x80`) untouched. It is a cheap single pass with no
61/// per-byte escaping work on the common (clean) path (just one string copy).
62/// @param msg Raw message to sanitize.
63/// @return @p msg with control characters escaped; the same bytes if already clean.
64inline std::string sanitizeLogLine(std::string_view msg) {
65 auto needsEscape = [](unsigned char chr) { return chr < 0x20 || chr == 0x7f; };
66 std::size_t firstBad = std::string_view::npos;
67 for (std::size_t i = 0; i < msg.size(); ++i) {
68 if (needsEscape(static_cast<unsigned char>(msg[i]))) {
69 firstBad = i;
70 break;
71 }
72 }
73 if (firstBad == std::string_view::npos) {
74 return std::string{msg}; // clean: one copy, no per-char work
75 }
76 std::string out;
77 out.reserve(msg.size() + 8);
78 out.append(msg.substr(0, firstBad));
79 static constexpr std::string_view kHex = "0123456789abcdef";
80 for (std::size_t i = firstBad; i < msg.size(); ++i) {
81 const auto chr = static_cast<unsigned char>(msg[i]);
82 switch (chr) {
83 case '\n':
84 out += "\\n";
85 break;
86 case '\r':
87 out += "\\r";
88 break;
89 case '\t':
90 out += "\\t";
91 break;
92 default:
93 if (needsEscape(chr)) {
94 out += "\\x";
95 out.push_back(kHex[chr >> 4]);
96 out.push_back(kHex[chr & 0x0f]);
97 } else {
98 out.push_back(static_cast<char>(chr));
99 }
100 break;
101 }
102 }
103 return out;
104}

106struct LogState {
107 Logger sink = [](LogLevel lvl, std::string_view msg) {
108 std::println(stderr, "[{}] {}", levelName(lvl), sanitizeLogLine(msg));
109 };
110 // Atomic so the level check is a lock-free fast path: `logFormat` and `log`
111 // can reject a suppressed message without touching `mtx` or formatting it.
112 std::atomic<LogLevel> minLevel{LogLevel::debug};
113 std::mutex mtx; // guards `sink` (and serialises sink invocation)
114};
115
116inline LogState& logState() {
117 static LogState state;
118 return state;
119}
120
121/// @brief Internal emit entry point used by the public level helpers.
122///
123/// The sink is invoked while `mtx` is held, so sink calls are serialised. A sink
124/// must therefore not call back into `morph::log` (log, `setLogger`, or
125/// construct a `ScopedLoggerOverride`) — `std::mutex` is non-recursive and that
126/// would self-deadlock — and should not block for long.
127inline void log(LogLevel level, std::string_view msg) {
128 auto& state = logState();
129 if (level < state.minLevel.load(std::memory_order_relaxed)) {
130 return; // lock-free reject of suppressed messages
131 }
132 std::scoped_lock const lock{state.mtx};
133 if (state.sink) {
134 state.sink(level, msg);
135 }
136}

142/// @brief Replaces the global log sink.
143///
144/// Thread-safe. The new sink is called for every message whose level meets the
145/// current minimum. Pass a no-op lambda to silence all output.
146/// @param logger New sink function.
147inline void setLogger(std::function<void(LogLevel, std::string_view)> logger) {
148 std::scoped_lock const lock{detail::logState().mtx};
149 detail::logState().sink = std::move(logger);
150}

152/// @brief Sets the minimum log level.
153///
154/// Messages below this level are silently dropped. Thread-safe.
155/// @param level Minimum level to emit.
156inline void setLogLevel(LogLevel level) {
157 detail::logState().minLevel.store(level, std::memory_order_relaxed);
158}

160/// @brief Returns the current minimum log level. Thread-safe.
161/// @return The active minimum level.
162inline LogLevel getLogLevel() {
163 return detail::logState().minLevel.load(std::memory_order_relaxed);
164}

170/// @brief Formats @p fmt with @p args and forwards to `log()`.
171///
172/// The level is checked *before* formatting, so a suppressed call pays no
173/// `std::format` / allocation cost.
174template <typename... Args>
175void logFormat(LogLevel level, std::format_string<Args...> fmt, Args&&... args) {
176 if (level < logState().minLevel.load(std::memory_order_relaxed)) {
177 return;
178 }
179 log(level, std::format(fmt, std::forward<Args>(args)...));
180}

216/// @brief RAII helper that swaps the global logger and level for the lifetime
217/// of the object, and restores them in the destructor.
218///
219/// Designed for tests that want to capture log output without leaking the
220/// custom sink into other tests. Thread-safe construction and destruction —
221/// the global mutex is acquired briefly during each.
222///
223/// @par Example
224/// @code
225/// {
226/// std::vector<std::string> captured;
227/// morph::log::ScopedLoggerOverride guard{
228/// [&](morph::log::LogLevel, std::string_view msg) { captured.emplace_back(msg); },
229/// morph::log::LogLevel::debug,
230/// };
231/// // ... run code that logs ...
232/// } // previous sink + level restored here
233/// @endcode
234class ScopedLoggerOverride {
235public:
236 /// @brief Snapshots the current sink + level so they can be restored.
237 ///
238 /// Use this when test code will install its own sink mid-test via
239 /// `setLogger()` / `setLogLevel()` and just wants automatic restoration.
240 ScopedLoggerOverride() {
241 // NOLINTBEGIN(cppcoreguidelines-prefer-member-initializer) — these must be
242 // read while holding the lock; a member-initializer list would read the
243 // global state before the mutex is acquired (a data race).
244 std::scoped_lock const lock{detail::logState().mtx};
245 _savedSink = detail::logState().sink;
246 _savedLevel = detail::logState().minLevel;
247 // NOLINTEND(cppcoreguidelines-prefer-member-initializer)
248 }
249
250 /// @brief Installs @p sink and @p level; saves whatever was there before.
251 ///
252 /// @param sink New logger sink. Pass a no-op lambda to suppress output entirely.
253 /// @param level New minimum level. Defaults to `debug` so every message reaches @p sink.
254 explicit ScopedLoggerOverride(std::function<void(LogLevel, std::string_view)> sink,
255 LogLevel level = LogLevel::debug) {
256 // NOLINTBEGIN(cppcoreguidelines-prefer-member-initializer) — snapshot and
257 // install happen under one lock; a member-initializer list would read the
258 // previous state outside the mutex, racing a concurrent setLogger.
259 std::scoped_lock const lock{detail::logState().mtx};
260 _savedSink = std::move(detail::logState().sink);
261 _savedLevel = detail::logState().minLevel;
262 detail::logState().sink = std::move(sink);
263 detail::logState().minLevel = level;
264 // NOLINTEND(cppcoreguidelines-prefer-member-initializer)
265 }
266
267 /// @brief Restores the saved sink and level.
268 ~ScopedLoggerOverride() {
269 std::scoped_lock const lock{detail::logState().mtx};
270 detail::logState().sink = std::move(_savedSink);
271 detail::logState().minLevel = _savedLevel;
272 }
273
274 ScopedLoggerOverride(const ScopedLoggerOverride&) = delete;
275 ScopedLoggerOverride& operator=(const ScopedLoggerOverride&) = delete;
276 ScopedLoggerOverride(ScopedLoggerOverride&&) = delete;
277 ScopedLoggerOverride& operator=(ScopedLoggerOverride&&) = delete;
278
279private:
280 std::function<void(LogLevel, std::string_view)> _savedSink;
281 LogLevel _savedLevel{LogLevel::debug};
282};
