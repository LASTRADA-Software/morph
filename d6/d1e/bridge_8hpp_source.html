39/// @brief Type-erased, JSON-in/JSON-out execute path for actions whose
40/// concrete C++ type is only known by its registered string id at
41/// the call site (e.g. a schema-driven GUI that reads action names
42/// out of a JSON Schema at runtime).
43///
44/// Populated automatically by `BRIDGE_REGISTER_ACTION` — no action-specific
45/// code is required at any call site. Every entry calls through the real
46/// `BridgeHandler<Model>::execute<Action>()` (so sessions, backend
47/// switches, and completions all behave exactly as they do for hand-written
48/// call sites), unlike `morph::model::detail::ActionDispatcher`, which
49/// calls `Model::execute` directly against an already-owned model holder
50/// and is only ever used server-side.
51///
52/// HARD REQUIREMENT (other direction of the same constraint documented on
53/// `morph::model::detail::registerActionExecutorOnce` in `registry.hpp`): registration into
54/// this registry happens via `registerActionExecutorOnce<Model, Action>`, which
55/// `BRIDGE_REGISTER_ACTION` calls unconditionally but which is only defined here, in
56/// `bridge.hpp`. Every translation unit that calls `BRIDGE_REGISTER_ACTION` must therefore
57/// include this header (directly or transitively), or that translation unit's static
58/// initializer will fail to link.
59class ActionExecuteRegistry {
60public:
61 /// @brief Deserialises `bodyJson`, dispatches through the handler's
62 /// `Bridge`, and resolves with the JSON-encoded result.
63 using Executor = std::function<::morph::async::Completion<std::string>(void*, std::string_view)>;
64
65 /// @brief Registers the executors for `(Model, Action)` under the given string ids.
66 ///
67 /// Populates one `Executor` per `Sharing` policy the framework defines
68 /// (`NoSharing` and `AllowShared`), keyed by
69 /// `(modelId, actionId, typeid(Sharing))`, rather than a single executor
70 /// that unconditionally assumes `NoSharing`. A handler whose `Sharing` is
71 /// anything else must be dispatched through the *matching* executor: the
72 /// wrong one `static_cast`s `handlerVoid` to the wrong `BridgeHandler<Model,
73 /// Sharing>` instantiation, so `kShared` resolves incorrectly inside it and
74 /// a payload-/result-keyed action's attach-or-promote step silently never
75 /// runs. Both executors' bodies are otherwise byte-for-byte identical
76 /// (built once, from the same generic lambda template, instantiated for
77 /// `NoSharing` and again for `AllowShared` — the only two sharing tags
78 /// `morph::bridge` defines), so this costs one extra closure per
79 /// registered action, not per call.
80 /// Defined out-of-line after BridgeHandler to avoid forward reference issues.
81 /// @tparam Model Model type whose handler will execute the action.
82 /// @tparam Action Action type to register.
83 /// @param modelId String id the model is registered under.
84 /// @param actionId String id the action is registered under.
85 template <typename Model, typename Action>
86 void registerAction(std::string_view modelId, std::string_view actionId);
87
88 /// @brief Looks up and invokes the executor for `(modelId, actionId)`,
89 /// specialised for the caller's own `Sharing` policy.
90 /// @tparam Sharing `NoSharing` or `AllowShared` — the caller's own sharing policy.
91 /// @param modelId String id of the target model.
92 /// @param actionId String id of the action to execute.
93 /// @param handler Type-erased `BridgeHandler<Model, Sharing>*` matching `modelId`.
94 /// @param bodyJson JSON-encoded action payload.
95 /// @return Completion that resolves with the JSON-encoded action result.
96 /// @throws std::runtime_error if no executor was registered for that triple.
97 template <typename Sharing>
98 [[nodiscard]] ::morph::async::Completion<std::string> execute(std::string_view modelId, std::string_view actionId,
99 void* handler, std::string_view bodyJson) const {
100 // See `ActionDispatcher::dispatch` -- the registration-phase latch,
101 // closed on the first read of a process-level registry so a later
102 // registration can assert. Debug builds only.
103 ::morph::model::detail::noteRegistryRead(this == &instance());
104 auto iter = _executors.find(
105 KeyView{.modelId = modelId, .actionId = actionId, .sharing = std::type_index{typeid(Sharing)}});
106 if (iter == _executors.end()) {
107 throw std::runtime_error("unknown action for executeJson: " + std::string{modelId} + "/" +
108 std::string{actionId});
109 }
110 return iter->second(handler, bodyJson);
111 }
112
113 /// @brief Returns the process-level singleton registry.
114 /// @return Reference to the singleton `ActionExecuteRegistry`.
115 static ActionExecuteRegistry& instance();
116
117private:
118 // The stored key: the registry owns its ids, because nothing else outlives
119 // a registration.
120 struct Key {
121 std::string modelId;
122 std::string actionId;
123 std::type_index sharing;
124 bool operator==(const Key&) const = default;
125 };
126 // The key a caller looks an entry up *with*. Every id reaching `execute`
127 // arrives as a `string_view` -- a schema-driven GUI's decoded action name,
128 // or a `constexpr` `ModelTraits<M>::typeId()` -- so materialising `Key`
129 // just to hash it would charge every `executeJson` two `std::string`
130 // constructions. Deliberately not
131 // `morph::model::detail::PairKeyView`: this key carries a `std::type_index`
132 // as well as the two ids, so it needs its own view type and its own
133 // functors rather than a reuse that would silently drop the sharing tag.
134 struct KeyView {
135 std::string_view modelId;
136 std::string_view actionId;
137 std::type_index sharing;
138 };
139 struct KeyHash {
140 // Marks the functor transparent; `KeyEqual` below is the other half
141 // `unordered_map` needs before a heterogeneous `find` compiles at all.
142 using is_transparent = void;
143
144 std::size_t operator()(KeyView key) const noexcept {
145 // Spelled with the view type rather than a braced list: `PairKeyHash`
146 // is transparent now, so a braced `{modelId, actionId}` is equally
147 // convertible to both of its overloads and would be ambiguous.
148 std::size_t const modelHash =
149 ::morph::model::detail::PairKeyHash{}(::morph::model::detail::PairKeyView{key.modelId, key.actionId});
150 return modelHash ^ (key.sharing.hash_code() + 0x9e3779b9U + (modelHash << 6) + (modelHash >> 2));
151 }
152
153 // Routed through the view body rather than hashing the stored strings
154 // separately, for the reason `PairKeyHash`'s own comment gives: two
155 // bodies that happen to agree are not a property a heterogeneous
156 // lookup may rest on. If they disagreed, a transparent `find()` would
157 // hash into the wrong bucket and report a registered action as
158 // unknown, with no diagnostic anywhere.
159 std::size_t operator()(const Key& key) const noexcept {
160 return (*this)(KeyView{.modelId = key.modelId, .actionId = key.actionId, .sharing = key.sharing});
161 }
162 };
163 struct KeyEqual {
164 // Marks the functor transparent, enabling heterogeneous lookup. A
165 // transparent hash alone is not enough: `unordered_map` requires both
166 // before a heterogeneous `find` compiles, and naming only the hash
167 // leaves every lookup silently materialising a `Key`.
168 using is_transparent = void;
169
170 // Accepts any mix of `Key` and `KeyView` on either side.
171 template <typename LhsT, typename RhsT>
172 bool operator()(const LhsT& lhs, const RhsT& rhs) const noexcept {
173 return lhs.sharing == rhs.sharing && std::string_view{lhs.modelId} == std::string_view{rhs.modelId} &&
174 std::string_view{lhs.actionId} == std::string_view{rhs.actionId};
175 }
176 };
177 std::unordered_map<Key, Executor, KeyHash, KeyEqual> _executors;
178};

190inline bool registerActionExecutorOnce(std::string_view modelId, std::string_view actionId) noexcept {
191 // clang-tidy's misc-static-assert fires on any `assert(!f())` whose `f` takes no
192 // argument, whether or not `f` is constexpr -- and this one loads an atomic.
193 // Applying the check's own fix does not compile: `static_assert` on this
194 // condition is "static assertion expression is not an integral constant
195 // expression / non-constexpr function 'registrationPhaseClosed' cannot be used
196 // in a constant expression".
197 //
198 // Last re-checked at clang-tidy 22.1.8, the version CI pins: the diagnostic
199 // still fires and the suggested fix still does not compile. The stamp is here
200 // because nothing re-checks it for you -- no script flags a directive that has
201 // stopped suppressing anything, and nothing triggers a re-read on a
202 // CLANG_VERSION bump. Re-read this when the pin moves; if the finding is gone,
203 // delete all four directives together.
204 // NOLINTNEXTLINE(misc-static-assert,cert-dcl03-c)
205 assert(!::morph::model::registrationPhaseClosed() &&
206 "registerActionExecutorOnce: registration after the registration phase closed. The "
207 "process-level registries are unsynchronised and are read-only once dispatch begins -- "
208 "registering now races their internals against concurrent lookups (see "
209 "docs/spec/core/registry.md, \"Thread safety\"). Load and register plugin modules before "
210 "the first dispatch.");
211 ::morph::bridge::ActionExecuteRegistry::instance().registerAction<Model, Action>(modelId, actionId);
212 return true;
213}

239/// @brief Internal linkage record between a model type and its active backend id.
240///
241/// Shared between `Bridge` and `BridgeHandler`. `Bridge::switchBackend()` updates
242/// `currentId` atomically under its mutex so that concurrent `executeVia()` calls
243/// always see a consistent value.
244struct HandlerBinding {
245 /// @brief String type-id of the model.
246 std::string typeId;
247
248 /// @brief Factory used to re-register the model on a backend switch.
249 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> modelFactory;
250
251 /// @brief Stable identity of this model instance (e.g. an account id).
252 ///
253 /// Empty by default — the common local-mode case needs nothing here, since
254 /// `modelFactory` can already call `IModelHolder::attachActionLog` directly
255 /// with whatever identity it captures. Set this when the active backend may
256 /// be remote (`SimulatedRemoteBackend`, `QtWebSocketBackend`): it travels in
257 /// the `register` wire envelope so a server-side `RemoteServer::LogProvider`
258 /// can attach a log to the instance it creates. See `IBackend::registerModelWithContext`.
259 std::string contextKey;
260
261 /// @brief Canonical string encoding of this instance's primary key.
262 ///
263 /// Empty until a keyed action or an explicit `attach` supplies one. Only
264 /// meaningful when `shared` is set; a private binding never consults it.
265 /// Mutated only under `Bridge::_attachMtx` (or, during `switchBackend()`
266 /// and the reconnect handler, both `_attachMtx` and `_mtx` together).
267 std::string primary;
268
269 /// @brief Whether this binding participates in the shared instance directory.
270 ///
271 /// Set once at construction from `BridgeHandler`'s `Sharing` template
272 /// argument and never changed. A shared binding defers its backend
273 /// registration until it has a primary, so `currentId` stays `0` — and
274 /// `executeVia` fails fast on "handler not bound" — until then.
275 bool shared = false;
276
277 /// @brief Current `ModelId` value in the active backend (0 = unbound).
278 std::atomic<uint64_t> currentId{0};
279
280 /// @brief Registration-settled seam (see `Bridge::whenBound`).
281 ///
282 /// `registrationInFlight` is `true` from just *before* `registerHandlerImpl`
283 /// calls the backend until that call's continuation resolves. It is set
284 /// unconditionally on every path, the blocking one included — see the
285 /// comment at the assignment for why it must be set before the backend call
286 /// rather than after. A backend that settles inside the dispatch frame does
287 /// not *leave* it set: the continuation runs before `registerHandlerImpl`
288 /// returns and resolves the waiters (clearing the flag). `whenBound()`
289 /// checks it to distinguish "an async reply is coming, queue a waiter"
290 /// from "nothing is in flight, resolve false now". `registrationWaiters`
291 /// holds callbacks queued by `whenBound()` while `registrationInFlight` is
292 /// `true`; the resolving callback invokes and clears every one of them
293 /// exactly once, in the same call that clears `registrationInFlight`.
294 /// Guarded by `registrationMtx`, deliberately separate from
295 /// `Bridge::_mtx`/`_attachMtx`: a waiter can be queued or resolved from
296 /// either the registering thread or the backend's reply-delivering
297 /// thread, and must never block on the bridge's own locks.
298 std::mutex registrationMtx;
299 bool registrationInFlight = false;
300 std::vector<std::pair<std::function<void(bool)>, std::function<void(std::exception_ptr)>>> registrationWaiters;
301};

304///
305/// The `Completion` surface carries an `exception_ptr`; the one remaining place
306/// that needs text rather than a rethrowable failure is a log message. Kept

359/// @brief Handoff slot between an async attach/bind dispatch and its callback.
360///
361/// `Bridge::attachHandlerAsync`/`ensureBoundAsync` dispatch to the backend while
362/// holding `_attachMtx`, and both promise to release it before invoking their
363/// `onDone`. A backend whose `bindModel` settles its `Completion` *inline* —
364/// synchronously, before the dispatch call returned, as `QtWebSocketBackend`
365/// does on its `!_connected` error branch — would otherwise break that promise
366/// from inside the dispatch frame, with the lock still held (the executor
367/// those call sites name is `inlineExecutor()`, so an inline settle runs the
368/// continuation right there).
369///
370/// So instead of acting, such a callback parks its outcome here and returns; the
371/// dispatching frame picks it up after the dispatch call returns, publishes it
372/// under the lock it already holds, releases the lock, and only then reports.
373/// `mtx` makes the handover race-free even for a backend that replies from
374/// another thread *while* its own dispatch call is still on this stack, and the
375/// `fired` flag keeps `onDone` invoked exactly once on every interleaving.
376struct AsyncDispatchHandoff {
377 /// @brief Guards every other field; never held across `onDone` or `_attachMtx`.
378 std::mutex mtx;
379 /// @brief Signalled once `fired` is set, for `awaitHandoff`'s benefit.
380 ///
381 /// Only a dispatcher that deliberately waits (`awaitHandoff`) ever blocks on
382 /// this; the three asynchronous dispatch sites never do, so for them the
383 /// `notify_all` in `parkIfInFrame` is an uncontended no-op.
384 std::condition_variable settled;
385 /// @brief `true` while the backend's dispatch call is still on the caller's stack.
386 bool inFrame = true;
387 /// @brief Set once either callback has claimed the outcome.
388 bool fired = false;
389 /// @brief `true` when the parked outcome is a success, `false` for a failure.
390 bool succeeded = false;
391 /// @brief Instance id the success callback reported.
392 ::morph::exec::detail::ModelId modelId{};
393 /// @brief Diagnostic the error callback reported; null on success.
394 std::exception_ptr failure;
395};

397/// @brief Records a backend callback's outcome in @p handoff.
398///
399/// Called first thing by both completion callbacks of an async attach/bind
400/// dispatch.
401///
402/// @param handoff Handoff slot created by the dispatching frame.
403/// @param succeeded `true` for the success callback, `false` for the error one.
404/// @param modelId Instance id, for the success callback; ignored otherwise.
405/// @param failure Diagnostic, for the error callback; null otherwise.
406/// @return `true` if the caller must **not** act — either because the dispatch
407/// call is still on the dispatcher's stack (which owns the outcome from
408/// here on) or because another callback already claimed this dispatch.
409/// `false` if the caller owns the outcome and should deliver it itself.
410///
411/// @par Reachability of the double-claim arm
412/// No backend can reach it, and that is a property of the callers rather
413/// than of this function. Every one of the eight call sites below
414/// is a `.then`/`.onError` on one `Completion`, and a `CompletionState` settles
415/// once — the second `resolve`/`reject` is a documented no-op — so exactly one
416/// of the two lambdas runs, exactly once, and `fired` is always `false` on
417/// entry. Measured, not assumed: replacing the arm's `return true` with an
418/// `abort()` runs the whole suite (1556 cases) plus `morph_net_tests` (191)
419/// without firing. The arm is kept because the invariant that makes it dead is
420/// every *current* caller's, and a ninth site that does not park a single
421/// `Completion`'s outcome would need it again; `tests/test_async_registration.cpp`
422/// calls this function directly so the arm is pinned by a test rather than
423/// merely unreached.
424inline bool parkIfInFrame(AsyncDispatchHandoff& handoff, bool succeeded, ::morph::exec::detail::ModelId modelId,
425 std::exception_ptr failure) {
426 bool inFrame = false;
427 {
428 std::scoped_lock const guard{handoff.mtx};
429 if (handoff.fired) {
430 // A backend is contractually allowed exactly one callback per dispatch;
431 // swallow a second one rather than reporting twice. Unreachable from
432 // a backend, because every dispatch goes behind one `Completion` --
433 // see @par Reachability above for what that rests on and why the arm
434 // stays.
435 return true;
436 }
437 handoff.fired = true;
438 handoff.succeeded = succeeded;
439 handoff.modelId = modelId;
440 handoff.failure = std::move(failure);
441 inFrame = handoff.inFrame;
442 }
443 // Outside the lock: a waiter in `awaitHandoff` re-acquires `mtx` the moment
444 // it wakes, and notifying while still holding it makes it wake only to block
445 // again.
446 handoff.settled.notify_all();
447 return inFrame;
448}

450/// @brief Closes the inline window and takes whatever a callback parked.
451///
452/// Called by the dispatching frame immediately after the backend's dispatch call
453/// returns. After this, a callback that has not yet run delivers its own outcome.
454///
455/// @param handoff Handoff slot created by the dispatching frame.
456/// @return The parked outcome if the backend completed inline (or concurrently,
457/// before this frame closed the window); `std::nullopt` if the frame won
458/// the race and the reply, if any, is still to come.
459inline std::optional<ParkedOutcome> claimHandoff(AsyncDispatchHandoff& handoff) {
460 std::scoped_lock const guard{handoff.mtx};
461 handoff.inFrame = false;
462 if (!handoff.fired) {
463 return std::nullopt;
464 }
465 return ParkedOutcome{.succeeded = handoff.succeeded, .modelId = handoff.modelId, .failure = handoff.failure};
466}

468/// @brief Holds the inline window open until a callback parks an outcome in it,
469/// then closes it and takes that outcome.
470///
471/// `claimHandoff`'s blocking twin, and the whole of what
472/// `BindWait::kCallerMayBlock` buys: the dispatching frame stops and waits, so a
473/// reply that lands on the backend's own thread is still delivered *by this
474/// frame*, on this thread, before the dispatching call returns — which is
475/// exactly what the frame does for a backend that settled inline. The outcome
476/// therefore takes the same path in both cases (published here, rethrown here),
477/// instead of one path for "settled in frame" and another for "settled later".
478///
479/// Only ever called on a backend that returned `BindWait::kCallerMayBlock`,
480/// whose contract is that the completion settles without this thread doing
481/// anything. There is deliberately no timeout: a bounded wait would make
482/// "is the handler bound when `registerHandler` returns" depend on how fast the
483/// network was, which is the non-determinism the synchronous contract exists to
484/// exclude. A backend that violates its own settle-exactly-once contract hangs
485/// here rather than silently returning an unbound handler, and a hang is the
486/// diagnosable failure of those two.
487///
488/// @param handoff Handoff slot created by the dispatching frame.
489/// @return The outcome a callback parked; never `std::nullopt`.
490inline std::optional<ParkedOutcome> awaitHandoff(AsyncDispatchHandoff& handoff) {
491 std::unique_lock guard{handoff.mtx};
492 handoff.settled.wait(guard, [&handoff] { return handoff.fired; });
493 handoff.inFrame = false;
494 return ParkedOutcome{.succeeded = handoff.succeeded, .modelId = handoff.modelId, .failure = handoff.failure};
495}

497/// @brief Delivers a registration continuation that arrived *after* its
498/// dispatching frame closed the handoff window.
499///
500/// The counterpart to `parkIfInFrame` returning `false`: nobody is left on the
501/// dispatching stack to publish this outcome, so the callback publishes it
502/// itself. With a null @p exec it publishes on whichever thread the backend
503/// happened to settle the `Completion` on, because the executor every dispatch
504/// site names is `exec::detail::inlineExecutor()`. That thread is where the
505/// use-after-free lives: each of these callbacks asks "is the `Bridge` still
506/// alive" and then touches it, and a `~Bridge` running concurrently on another
507/// thread can land between the two steps.
508///
509/// Routing only the late delivery through an executor the `Bridge` was given
510/// closes that window structurally for an embedder whose executor runs tasks
511/// on the thread that also runs `~Bridge`: the continuation and the destructor
512/// are then two tasks on one thread and cannot interleave at all. The in-frame
513/// case deliberately does not go through @p exec — see `Bridge`'s
514/// `bridgeExec` constructor parameter for why posting it would turn
515/// synchronous registration asynchronous and deadlock a caller that waits on
516/// `awaitHandoff` from the executor's own thread.
517///
518/// A template rather than a `std::function` parameter, and that is not
519/// incidental: with a null @p exec the callable is invoked **in place**, so the
520/// default path type-erases nothing and allocates nothing. Taking a
521/// `std::function` would put a heap allocation — sometimes a large one, since
522/// these closures carry a primary key — on that inline path;
523/// `tests/test_async_registration.cpp`'s allocation-failure case detects such a
524/// regression by catching the wrong allocation.
525///
526/// @tparam Action Callable of no arguments; convertible to `std::function` only
527/// on the posting path.
528/// @param exec Executor to deliver on. Borrowed, and may be null, which is
529/// the default a `Bridge` constructed without one carries: a
530/// null @p exec runs @p action inline, on the settling thread.
531/// Non-null, it must outlive every in-flight registration,
532/// because a reply can land after `~Bridge`.
533/// @param action Work to run. Must be safe to run after `~Bridge` — every
534/// caller here gates on a `CallbackToken` or a
535/// `detail::BridgeLifetime` before touching the bridge.
536template <typename Action>
537void deliverLate(::morph::exec::IExecutor* exec, Action&& action) {
538 if (exec == nullptr) {
539 std::forward<Action>(action)();
540 return;
541 }
542 exec->post(std::function<void()>{std::forward<Action>(action)});
543}

545/// @brief The gate that makes "the `Bridge` is still there" and "call into it"
546/// a single, indivisible step.
547///
548/// `morph::async::CallbackToken` answers only *"was the scope active at the
549/// moment of the check"* — advisory across threads by design
550/// (docs/spec/core/callback_scope.md, "Boundary of the guarantee"). That is
551/// enough to gate **delivery** of a callback, because a suppressed callback is
552/// simply not run and the check being stale costs nothing. It is *not* enough
553/// to gate a **member call on the `Bridge`**: the bridge can be destroyed in
554/// the instructions between the check and the call, and the call then runs on
555/// destroyed memory. Concretely, a `~BridgeHandler` running on a worker thread
556/// sees an active token, and `Bridge::deregisterHandler` then iterates a
557/// `_handlers` vector whose `Bridge` the owning thread has already finished
558/// destroying.
559///
560/// This type closes that window structurally rather than per call site: the
561/// answer is only ever read while `mtx` is held, and `~Bridge` flips it while
562/// holding the same mutex exclusively. A caller that observes `alive == true`
563/// therefore *keeps* the bridge alive for as long as it stays inside the
564/// shared lock, because `~Bridge` cannot get past its own first statement
565/// until that lock is released.
566///
567/// Owned through a `shared_ptr`, so it outlives the `Bridge` for exactly as
568/// long as some handler still holds a reference to it and can ask the
569/// question.
570///
571/// @par Why a blocking gate is acceptable here
572/// `~Bridge` waits only for work that is provably non-blocking and bounded.
573/// There are three guarded regions: `~BridgeHandler`'s `deregisterHandler`
574/// call, `executeVia`'s `.then` continuation around `onResult`, and
575/// `registerHandlerImpl`'s async `onRegistered` callback. The argument below is
576/// about the first; each of the other two carries its own at its call site.
577/// `IBackend::deregisterModel` never blocks on another thread on any shipped
578/// backend — `LocalBackend` erases map entries under its own mutex,
579/// `SimulatedRemoteBackend` runs the envelope inline via
580/// `RemoteServer::handleInline`, and `QtWebSocketBackend`/`SocketBackend` are
581/// documented fire-and-forget sends precisely so that destruction never spins
582/// a nested event loop. A destructor that blocks on a bounded predicate is
583/// also the framework's existing idiom for exactly this class of hazard —
584/// `~StrandExecutor` blocks until `_inFlight == 0`
585/// (docs/spec/concurrency_and_lifetimes.md, "Destruction ordering").
586struct BridgeLifetime {
587 /// Held shared by a caller for the whole of its call into the `Bridge`,
588 /// and exclusively by `~Bridge` while it retires the bridge.
589 std::shared_mutex mtx;
590 /// `true` until `~Bridge` begins. Read and written only under `mtx`.
591 bool alive = true;
592};

594/// @brief The typed completion state a backend settles directly.
595///
596/// This object *is* the typed completion state `Bridge::executeVia` hands the
597/// caller, and is also the `ISettleSink` the backend settles. One object rather
598/// than a typed completion plus an erased `Completion<std::shared_ptr<void>>`
599/// with a `.then`/`.onError` pair forwarding between them, which costs one
600/// allocation per dispatch instead of six — the two states, the two forwarding
601/// closures, and their two handler vectors.
602///
603/// @par The four invariants the settle path holds
604/// Each is a property of *this* class's settle methods, and each is
605/// load-bearing on its own.
606/// - **The deadline disarm happens first**, before any other settle work, so a
607/// slow `onResult`/`publishResult` cannot give the timer a window to resolve
608/// this completion with `ClientTimeoutError` while the real result is in
609/// hand — `settleOnce`, called at the top of both settle methods.
610/// - **`_pendingCalls` is decremented on exactly one of two mutually exclusive
611/// paths**, whether or not the work that follows then throws — `settleOnce`
612/// again, which is also what makes a `cancelPending` racing a reply decrement
613/// once rather than twice.
614/// - **`lifetime->alive` is read once** under the gate and that one snapshot
615/// decides both `onResult` and `publishResult`, so the two cannot disagree.
616/// - **The value forwarding is guarded**, so a throwing move of `R` routes to
617/// this state's own error sink instead of escaping the callback executor,
618/// where `ThreadPoolExecutor` swallows it (hanging the completion) and
619/// `QtExecutor` lets it reach the event loop.
620///
621/// @par Lifetime
622/// Everything this holds is pinned: `_pendingCalls`, `_subscriptions` and
623/// `_lifetime` are the `Bridge`'s own heap-allocated, shared members, and
624/// `_schedulerRef` is a private `shared_ptr` copy taken under
625/// `_executeDeadlineMtx`. Nothing here touches the `Bridge` itself, so a
626/// dispatch settling after `~Bridge` is safe by construction rather than by
627/// check — except `onResult`, which genuinely needs the current bridge and is
628/// therefore the one thing gated on `alive`.
629///
630/// @tparam R The action's result type.
631template <typename R>
632class BridgeSink final : public ::morph::async::detail::CompletionState<R>,
633 public ::morph::async::detail::ISettleSink {
634public:
635 /// @brief Builds the sink for one dispatch.
636 /// @param mid Instance id this dispatch targeted, for `publishResult`.
637 /// @param onResult Optional observer run on the typed result before it is
638 /// moved into this state and before the caller's `.then`.
639 /// @param pendingCalls The bridge's pinned in-flight counter.
640 /// @param subscriptions The bridge's pinned subscription registry.
641 /// @param lifetime The bridge's lifetime gate.
642 BridgeSink(::morph::exec::detail::ModelId mid, std::function<void(const R&)> onResult,
643 std::shared_ptr<std::atomic<std::size_t>> pendingCalls,
644 std::shared_ptr<SubscriptionRegistry<HandlerBinding>> subscriptions,
645 std::shared_ptr<BridgeLifetime> lifetime)
646 : _mid{mid},
647 _onResult{std::move(onResult)},
648 _pendingCalls{std::move(pendingCalls)},
649 _subscriptions{std::move(subscriptions)},
650 _lifetime{std::move(lifetime)} {}
651
652 /// @brief Hands the sink the deadline entry it must disarm when it settles.
653 ///
654 /// Called by `executeVia` *before* the dispatch, so the write is sequenced
655 /// before anything that could settle this sink on another thread. The
656 /// scheduler is held as a `shared_ptr` copy taken under the bridge's
657 /// `_executeDeadlineMtx`, never as `Bridge::_timeoutScheduler`: while any
658 /// copy is alive `~TimeoutScheduler` cannot run, which is what makes
659 /// `cancel()` safe from a callback that may outlive the bridge.
660 ///
661 /// @param handle The scheduled entry.
662 /// @param scheduler The scheduler that owns it.
663 void armDeadline(::morph::async::detail::TimeoutScheduler::Handle handle,
664 std::shared_ptr<::morph::async::detail::TimeoutScheduler> scheduler) {
665 _deadlineHandle = handle;
666 _schedulerRef = std::move(scheduler);
667 }
668
669 /// @brief Undoes `armDeadline` and the pending count for a dispatch that
670 /// never started, because `IBackend::executeInto` threw.
671 ///
672 /// Without this, a throw out of the backend leaves `_pendingCalls` inflated
673 /// — and `pendingCalls()` is documented as a quiescence gate — and the timer
674 /// entry stranded. Routed through `settleOnce` so it cannot
675 /// double-count against a sink the backend had already settled before it
676 /// threw.
677 void abandon() {
678 if (settleOnce()) {
679 _pendingCalls->fetch_sub(1, std::memory_order_relaxed);
680 }
681 }
682
683 /// @brief Settles with the backend's opaque result.
684 /// @param opaque The result, which must point at an `R`. Spelled
685 /// `opaque` rather than `value` only because `CompletionState`
686 /// already has a member of that name, and -Wshadow-field is on.
687 void settleValue(std::shared_ptr<void> opaque) override {
688 if (!settleOnce()) {
689 return;
690 }
691 _pendingCalls->fetch_sub(1, std::memory_order_relaxed);
692 // Guard the value-forwarding: if R's move/copy throws (or the cast is
693 // somehow wrong), route the exception to this state's own error sink
694 // rather than letting it escape into the strand task or transport
695 // thread that called us -- where ThreadPoolExecutor swallows it (and
696 // the completion then hangs forever) and QtExecutor lets it reach the
697 // event loop and std::terminate. Mirrors the forwarding guard in
698 // remote.hpp's SimulatedRemoteBackend::execute. See
699 // docs/spec/core/bridge.md.
700 try {
701 auto* const typedResult = static_cast<R*>(opaque.get());
702 // `bridgeAlive` is read once, fresh, under `_lifetime`'s own gate --
703 // the same `detail::BridgeLifetime` `~BridgeHandler` uses (see its
704 // doc comment for why a `CallbackToken` cannot carry this weight) --
705 // and reused below to decide whether to publish, so both decisions
706 // come from one consistent snapshot. Only `onResult` actually runs
707 // inside the gate: it is the one piece of this method that is not
708 // pinned and genuinely needs the *current* bridge/backend (for a
709 // result-keyed action it calls `assignHandlerPrimary`, which
710 // touches `_attachMtx` and `loadBackend()`) -- and it runs before
711 // the value is moved out and before the caller's own `.then`, so a
712 // result-sourced primary key is adopted before any user code
713 // observes the result. `assignHandlerPrimary`'s own synchronous
714 // path is non-blocking by construction (its doc comment: it prefers
715 // the backend's async registration precisely to avoid a
716 // nested-event-loop block), so holding the gate across it does not
717 // expose `~Bridge` to the unbounded-block hazard that a guarded
718 // region calling into consumer-supplied code would carry.
719 bool bridgeAlive = false;
720 {
721 std::shared_lock const gate{_lifetime->mtx};
722 bridgeAlive = _lifetime->alive;
723 if (bridgeAlive && _onResult) {
724 _onResult(*typedResult);
725 }
726 }
727 // Fan the result out to everything attached to this instance before
728 // the value is moved away. `_subscriptions` is pinned, so this call
729 // is safe regardless of the Bridge's lifetime
730 // (`SubscriptionRegistry` snapshots its own sinks under its own lock
731 // and invokes them outside it -- see that class -- so it cannot
732 // deadlock against a subscriber that re-enters, which is exactly
733 // the hazard that rules out gating this call the way `onResult`
734 // above is gated). Still conditioned on `bridgeAlive`: a Bridge
735 // already torn down should not go on fanning results out to its
736 // instance subscribers.
737 if constexpr (std::is_copy_constructible_v<R>) {
738 if (bridgeAlive && _subscriptions->hasSubscribers()) {
739 _subscriptions->publishResult(_mid, std::type_index{typeid(R)}, std::any{*typedResult});
740 }
741 }
742 this->setValue(std::move(*typedResult));
743 } catch (...) {
744 this->setException(std::current_exception());
745 }
746 }
747
748 /// @brief Settles with a failure from the backend.
749 /// @param exc The exception to deliver to the caller's `.onError`.
750 void settleException(const std::exception_ptr& exc) override {
751 if (!settleOnce()) {
752 // Already settled -- but `CompletionState::setException` is
753 // first-result-wins and silently drops a late one, so calling it
754 // anyway costs nothing and keeps the behaviour identical to the
755 // pre-sink shape, where `cancelPending` reached the state directly.
756 this->setException(exc);
757 return;
758 }
759 _pendingCalls->fetch_sub(1, std::memory_order_relaxed);
760 this->setException(exc);
761 }
762
763private:
764 /// @brief Claims the one-time settle work; `true` for exactly one caller.
765 ///
766 /// Two things happen once per dispatch and must not happen twice: the
767 /// deadline disarm and the `_pendingCalls` decrement. Before this class
768 /// they were carried by the fact that `.then` and `.onError` are mutually
769 /// exclusive on one `CompletionState`. A sink has no such guarantee --
770 /// `IBackend::cancelPending` settles it from one thread while a reply may
771 /// be settling it from another -- so the latch is explicit.
772 ///
773 /// The deadline disarm rides along here rather than in the callers,
774 /// because "first" is exactly what it needs to be: it must happen before
775 /// any forwarding work, so a slow `onResult`/`publishResult` cannot give
776 /// the timer a window to fire and resolve this completion with
777 /// `ClientTimeoutError` while the real result is already in hand.
778 ///
779 /// @return `true` if this call claimed the settle, `false` if another did.
780 bool settleOnce() {
781 if (_settled.test_and_set(std::memory_order_acq_rel)) {
782 return false;
783 }
784 if (_deadlineHandle && _schedulerRef) {
785 try {
786 _schedulerRef->cancel(*_deadlineHandle);
787 } catch (...) { // NOLINT(bugprone-empty-catch)
788 // Best-effort, by contract and not only on a throw:
789 // `TimeoutScheduler::cancel` stops an entry that has not
790 // started but returns without waiting for one that already has.
791 // A deadline callback already mid-flight still runs its
792 // `setException`, which this state discards as an already-ready
793 // one -- first result wins. That, not the disarm, is what makes
794 // the race harmless.
795 }
796 }
797 return true;
798 }
799
800 ::morph::exec::detail::ModelId _mid;
801 std::function<void(const R&)> _onResult;
802 std::shared_ptr<std::atomic<std::size_t>> _pendingCalls;
803 std::shared_ptr<SubscriptionRegistry<HandlerBinding>> _subscriptions;
804 std::shared_ptr<BridgeLifetime> _lifetime;
805 std::optional<::morph::async::detail::TimeoutScheduler::Handle> _deadlineHandle;
806 std::shared_ptr<::morph::async::detail::TimeoutScheduler> _schedulerRef;
807 std::atomic_flag _settled = ATOMIC_FLAG_INIT;
808};

812/// @brief Central dispatcher that routes typed actions to an `IBackend`.
813///
814/// `Bridge` owns exactly one active backend at a time. It tracks all registered
815/// `HandlerBinding` instances and re-registers them automatically when
816/// `switchBackend()` is called, enabling seamless local ↔ remote transitions.
817///
818/// @par Thread safety
819/// All public methods are thread-safe. `executeVia()` takes a short snapshot of
820/// the backend `shared_ptr` under the dedicated `_backendMtx` (never `_mtx`), so
821/// it does not block `switchBackend()`.
822class Bridge {
823public:
824 /// @brief Constructs a bridge that dispatches through @p backend.
825 ///
826 /// Installs a reconnect handler so backends with a recoverable transport
827 /// (e.g. `QtWebSocketBackend`) can ask the bridge to re-register every live
828 /// handler against the freshly reconnected peer.
829 ///
830 /// @par The bridge's own executor
831 /// Every other completion in the framework is delivered on an executor its
832 /// caller named — `BridgeHandler` supplies `guiExec`, `executeVia` takes a
833 /// `cbExec`. The registrations the bridge issues *on its own behalf* have no
834 /// such caller: their five dispatch sites name
835 /// `exec::detail::inlineExecutor()`, which is "deliver wherever the backend
836 /// settled" written as a value rather than as a sentence in a doc comment.
837 /// @p bridgeExec is where that choice lives.
838 ///
839 /// It is used for **one** thing: a registration reply that arrives after
840 /// its dispatching frame has gone (`detail::deliverLate`). That is the only
841 /// case with a thread to choose — a reply that settles while the dispatch
842 /// call is still on the stack is parked in a
843 /// `detail::AsyncDispatchHandoff` and published by the dispatching frame
844 /// itself, on the dispatching thread, whatever @p bridgeExec says.
845 ///
846 /// The `bindModel`/`promoteModel` calls therefore keep naming
847 /// `inlineExecutor()`, and that is deliberate rather than an omission.
848 /// Naming a posting executor there would break two things at once:
849 /// `registerHandler()` would stop being synchronous for every backend that
850 /// binds inline (the settle would become a queued task, so the handler
851 /// would be unbound on return), and a `kCallerMayBlock` backend would
852 /// deadlock outright whenever the dispatching thread *is* the executor's
853 /// thread — `detail::awaitHandoff` would sit waiting for a task only that
854 /// same thread can run. A GUI embedder passing its GUI executor is exactly
855 /// that case.
856 ///
857 /// **What the caller must guarantee, and what it buys.** @p bridgeExec must
858 /// outlive this bridge and every registration still in flight when it is
859 /// destroyed, because a late reply can land after `~Bridge` (the same
860 /// requirement `BridgeHandler`'s `guiExec` already carries). The
861 /// check-then-touch window is closed only if @p bridgeExec runs its tasks
862 /// on a thread that cannot run `~Bridge` concurrently — for a Qt embedder, the
863 /// GUI thread that both owns the `Bridge` and pumps the executor. Supplying
864 /// an executor on some *other* thread satisfies the type and does not close
865 /// the window; it is not made worse than the default either, since the
866 /// callbacks' `CallbackToken`/`detail::BridgeLifetime` gates still apply.
867 /// Left null — the default — delivery is inline, on whichever thread the
868 /// backend settled the reply.
869 ///
870 /// @param backend Initial backend. Ownership is transferred.
871 /// @param bridgeExec Executor for late registration continuations, or null
872 /// (the default) to keep delivering them inline.
873 /// Borrowed: it must outlive this bridge.
874 explicit Bridge(std::unique_ptr<::morph::backend::detail::IBackend> backend,
875 ::morph::exec::IExecutor* bridgeExec MORPH_LIFETIMEBOUND = nullptr)
876 : _backend{std::shared_ptr<::morph::backend::detail::IBackend>(std::move(backend))}, _bridgeExec{bridgeExec} {
877 installReconnectHandler(_backend);
878 // A null initial backend is tolerated (see installReconnectHandler's own
879 // null check just above) — there is nothing yet to stamp a session onto.
880 if (_backend) {
881 _backend->setSession(_defaultSession);
882 }
883 }

885 /// @brief Cancels every still-pending completion on the active backend.
886 ///
887 /// Each `.onError(...)` callback receives a `BridgeDestroyedError`. In-flight
888 /// server replies that arrive after destruction are no-ops because each
889 /// `CompletionState::setValue`/`setException` is idempotent.
890 ///
891 /// Before cancelling, the active backend's reconnect handler is cleared: the
892 /// installed handler captures `this`, and a co-owned backend that outlives the
893 /// `Bridge` could otherwise fire it after destruction and dereference freed
894 /// memory. Clearing it here (outside `_mtx` — `setReconnectHandler` only stores
895 /// a callback and never re-enters the bridge) closes the common case; the
896 /// handler additionally guards on `_callbacks` so a reconnect already in flight
897 /// on the transport thread becomes a no-op too. See docs/spec/concurrency_and_lifetimes.md.
898 ///
899 /// The **first** statement retires the lifetime gate (`detail::BridgeLifetime`),
900 /// and it has to stay first: everything below it — clearing the reconnect
901 /// handler, cancelling pending completions, and then the implicit destruction
902 /// of every member — is state a concurrently-destroyed `BridgeHandler` would
903 /// otherwise still be walking. `closeLifetime()` both publishes "this bridge is
904 /// retired" and **waits out** any deregistration already inside the gate, so
905 /// this destructor never overlaps one. That wait is bounded and cannot
906 /// deadlock: see `detail::BridgeLifetime`.
907 ~Bridge() {
908 closeLifetime();
909 if (auto active = loadBackend()) {
910 active->setReconnectHandler(nullptr);
911 active->cancelPending(std::make_exception_ptr(::morph::backend::BridgeDestroyedError{}));
912 }
913 }

963 /// @brief Attaches (or re-points) @p binding to the shared instance for @p primary.
964 ///
965 /// Idempotent: attaching to the primary a binding already holds is a no-op,
966 /// so a keyed action repeated against the same instance costs nothing. A
967 /// different primary re-points the binding — the previous instance is
968 /// released and survives only if another handler still holds it.
969 ///
970 /// The binding's `contextKey` is set to @p primary as well, so a keyed
971 /// model's journal entries carry the entity key without the caller
972 /// arranging it separately.
973 ///
974 /// @tparam Model Concrete model type.
975 /// @param binding Shared binding, as returned by `registerSharedHandler<Model>()`.
976 /// @param primary Canonical string encoding of the primary key to attach to.
977 template <typename Model>
978 void attachHandler(const std::shared_ptr<detail::HandlerBinding>& binding, std::string primary) {
979 std::scoped_lock const lock{_attachMtx};
980 if (binding->primary == primary && binding->currentId.load() != 0U) {
981 return;
982 }
983 auto const previous = ::morph::exec::detail::ModelId{binding->currentId.load()};
984 // IBackend::attachModel's default (local) implementation and
985 // RemoteServer's wire-level attach handling both acquire the
986 // replacement before releasing `previous`, so a throwing acquire
987 // leaves `previous` untouched except in one narrow, harmless case: a
988 // remote re-point whose connection scope closes concurrently with
989 // the attach, where `previous` may already be released by the time
990 // the reply reports failure -- but no further request on this
991 // binding will ever run at that point either, so nothing is actually
992 // lost from the caller's perspective. Unbind first is therefore
993 // unnecessary: publish only on success.
994 auto newId = loadBackend()->attachModel(binding->typeId, binding->modelFactory,
995 {.contextKey = primary, .primary = primary}, previous);
996 binding->contextKey = primary;
997 binding->primary = std::move(primary);
998 binding->currentId.store(newId.v);
999 }

1001 /// @brief Applies an out-of-frame bind reply to a binding and reports the
1002 /// outcome, under the guards every such reply needs.
1003 ///
1004 /// The one shape `attachHandlerAsync` and `ensureBoundAsync` share, and
1005 /// the reason it is here rather than written twice: both dispatch through
1006 /// `IBackend::bindModel`, both park an in-frame reply, and both are left
1007 /// with the same four-step job when the reply instead arrives *after* the
1008 /// dispatching frame has gone -- is the `Bridge` still there, is the
1009 /// binding still there, is the backend that answered still the active one,
1010 /// and report through `onDone` either way. They differ only in what a
1011 /// successful reply publishes, which is what @p publish is.
1012 ///
1013 /// `assignHandlerPrimary`'s continuation deliberately does **not** go
1014 /// through here. It has no `onDone` -- a stale reply there is dropped
1015 /// silently, because no caller is waiting on it -- so routing it through a
1016 /// helper whose contract is "report exactly once" would mean giving that
1017 /// helper a second mode, and the two behaviours are different on purpose.
1018 ///
1019 /// @par The liveness check and the `this` touch are two steps
1020 /// That window is closed not by a gate but by the thread this body runs on,
1021 /// which is the bridge's own executor: a non-null one running `~Bridge`'s
1022 /// thread closes it outright, and the null default delivers inline on
1023 /// whichever thread the backend settled the reply. Gating instead would
1024 /// block `~Bridge` behind `_attachMtx`, which `attachHandler` holds across
1025 /// a full `attachModel` round trip.
1026 ///
1027 /// @par Locking
1028 /// @p publish runs under `_attachMtx`, because `contextKey`/`primary` are
1029 /// plain `std::string`s that every other site reads under that lock --
1030 /// publishing them without it would be a data race, not merely a stale
1031 /// read. (`registerHandlerImpl`'s read during registration is the one
1032 /// documented carve-out; see its own comment.) @p onDone is
1033 /// invoked **after** the lock is released, on every path that invokes it at
1034 /// all: what a caller does from inside it is dispatch the action, which can
1035 /// re-enter `_attachMtx` through `assignHandlerPrimary`.
1036 ///
1037 /// @tparam Publish Callable taking `detail::HandlerBinding&`. Runs under
1038 /// `_attachMtx` on success only. May throw: the exception
1039 /// is caught and reported through @p onDone rather than
1040 /// escaping onto an executor's thread.
1041 /// @param liveness Token for the `Bridge`; nothing is touched once it
1042 /// reports inactive.
1043 /// @param weakBinding Binding the reply belongs to. A expired one means the
1044 /// `BridgeHandler` is gone and there is nothing to
1045 /// publish to, so @p onDone is not called.
1046 /// @param weakBackend Backend the reply came from, compared against the
1047 /// active one before anything is published.
1048 /// @param onDone Caller's continuation: `nullptr` on success, a
1049 /// non-null `exception_ptr` on failure. Invoked at most
1050 /// once, outside `_attachMtx`.
1051 /// @param publish Applies the reply to the binding. See @p Publish.
1052 template <typename Publish>
1053 void publishLateBindReply(const ::morph::async::CallbackToken& liveness,
1054 const std::weak_ptr<detail::HandlerBinding>& weakBinding,
1055 const std::weak_ptr<::morph::backend::detail::IBackend>& weakBackend,
1056 const std::function<void(std::exception_ptr)>& onDone, Publish&& publish) {
1057 if (!liveness.active()) {
1058 return; // The Bridge is gone; publishing this id would be pointless.
1059 }
1060 auto strongBinding = weakBinding.lock();
1061 if (!strongBinding) {
1062 return; // The BridgeHandler (and its binding) is gone.
1063 }
1064 std::exception_ptr failure;
1065 {
1066 std::scoped_lock const guard{_attachMtx};
1067 auto pinned = weakBackend.lock();
1068 if (!pinned || pinned != loadBackend()) {
1069 // A switchBackend() already moved past this reply (see
1070 // registerHandlerImpl's identical guard) and its own
1071 // re-registration loop already handled this binding on the
1072 // *new* backend -- applying the stale reply now would
1073 // overwrite that with a dangling id from a backend nothing
1074 // uses any more. Unlike registerHandlerImpl's fire-and-forget

1080 std::runtime_error("attach reply arrived from a backend switchBackend() already replaced"));

1082 try {
1083 std::forward<Publish>(publish)(*strongBinding);
1084 } catch (...) {
1085 failure = std::current_exception();
1086 }
1087 }
1088 }
1089 onDone(failure); // Outside the lock -- see @par Locking.

1092 /// @brief Async counterpart to `attachHandler`: dispatches the attach and
1093 /// invokes @p onDone once attached (or failed), instead of blocking.
1094 ///
1095 /// Reaches the backend through `IBackend::bindModel` — the structural
1096 /// registration surface, and the only one. There is
1097 /// exactly one dispatch and exactly one continuation: a
1098 /// backend with no non-blocking attach settles inside the `bindModel` call,
1099 /// having blocked for the same round trip the synchronous `attachHandler`
1100 /// would have, and @p onDone is then invoked from this thread before this
1101 /// method returns — so a caller that always goes through this method
1102 /// behaves identically to calling `attachHandler` directly on such a
1103 /// backend.
1104 ///
1105 /// @par Locking
1106 /// `_attachMtx` is held around the guard check and the *dispatch* — matching
1107 /// `attachHandler`'s existing lock scope — but is **released before
1108 /// @p onDone is ever invoked**, on every path, unconditionally. That is not
1109 /// a nicety: what `execute()` does from inside @p onDone is dispatch the
1110 /// action, and a result-keyed dispatch promotes its binding through
1111 /// `assignHandlerPrimary`, which takes `_attachMtx` itself. Invoking
1112 /// @p onDone under the lock therefore self-deadlocks the moment the
1113 /// completion is delivered on the calling thread — which is exactly what a
1114 /// blocking backend's `bindModel` does, since the executor this call site
1115 /// names runs the continuation inline. This is `registerHandlerImpl`'s
1116 /// existing rule ("the backend call must not run under `_mtx`") applied to
1117 /// `_attachMtx`.
1118 ///
1119 /// The guarantee holds even for a backend that settles *inline*, from inside
1120 /// the dispatch call itself, while this frame still holds the lock: such a
1121 /// continuation parks its outcome in a `detail::AsyncDispatchHandoff` and
1122 /// returns without acting, and this frame applies it after the dispatch call
1123 /// has returned and the lock is gone. See that struct's doc comment.
1124 ///
1125 /// An out-of-frame success callback re-acquires `_attachMtx` for the two
1126 /// `std::string` fields it publishes (`contextKey`/`primary`, which
1127 /// `HandlerBinding` documents as readable only under that lock) and drops it
1128 /// again before calling @p onDone. That re-acquisition is safe precisely
1129 /// because the inline case never reaches it.
1130 ///
1131 /// @par Known gap
1132 /// Two calls for the same key issued before the first one's reply arrives
1133 /// are **not** deduplicated: the guard below reads `binding->primary`/
1134 /// `currentId`, neither of which is updated until the reply lands, so both
1135 /// calls pass it and both dispatch an `attach`. This is a real behaviour
1136 /// difference from the synchronous `attachHandler` it replaces, not merely
1137 /// something inherent to asynchrony — `attachHandler` held `_attachMtx`
1138 /// across the whole blocking round trip, which serialised concurrent
1139 /// callers for free. It needs no second thread to hit: two `execute()`
1140 /// calls in one event-loop turn are enough. The server answers both with
1141 /// the same `ModelId` but counts two attachments, so one attach reference
1142 /// leaks; the leak is bounded, not unbounded — the connection scope
1143 /// releases every reference it holds when it closes. Closing this properly
1144 /// needs in-flight tracking on the binding (coalescing the second caller
1145 /// onto the first dispatch's completion); tracked as a follow-up, not fixed
1146 /// here. Same gap, same reasoning, on `ensureBoundAsync`.
1147 ///
1148 /// @tparam Model Concrete model type.
1149 /// @param binding Shared binding, as returned by `registerSharedHandler<Model>()`.
1150 /// @param primary Canonical string encoding of the primary key to attach to.
1151 /// @param onDone Invoked with `nullptr` on success, or a non-null
1152 /// `exception_ptr` on failure — always exactly once,
1153 /// synchronously if the fallback path is taken.
1154 template <typename Model>
1155 void attachHandlerAsync(const std::shared_ptr<detail::HandlerBinding>& binding, std::string primary,
1156 const std::function<void(std::exception_ptr)>& onDone) {
1157 std::unique_lock lock{_attachMtx};
1158 if (binding->primary == primary && binding->currentId.load() != 0U) {
1159 lock.unlock();
1160 onDone(nullptr);
1161 return;
1162 }
1163 auto const previous = ::morph::exec::detail::ModelId{binding->currentId.load()};
1164 auto backend = loadBackend();
1165 auto primaryCopy = std::move(primary);
1166 std::weak_ptr<::morph::backend::detail::IBackend> const weakBackend{backend};
1167 auto const weakLiveness = _callbacks.token();
1168 std::weak_ptr<detail::HandlerBinding> const weakBinding{binding};
1169 auto handoff = std::make_shared<detail::AsyncDispatchHandoff>();
1170 auto* const bridgeExec = _bridgeExec;
1171 auto onAttached = [this, weakBackend, weakLiveness, weakBinding, primaryCopy, onDone, handoff,
1172 bridgeExec](::morph::exec::detail::ModelId newId) mutable {
1173 if (detail::parkIfInFrame(*handoff, true, newId, nullptr)) {
1174 return; // Completed inline: the dispatching frame will finish this.
1175 }
1176 // Nobody is left on the dispatching stack, so this callback
1177 // publishes the outcome itself -- on `bridgeExec` if the embedder
1178 // named one, inline otherwise. See `detail::deliverLate`.
1179 //
1180 // `mutable`, so `primaryCopy` is *moved* into the body's closure
1181 // rather than copied: this callback runs exactly once (one
1182 // `Completion`, settled once), and copying the primary key here
1183 // would put a fresh allocation on a path that has none -- which is
1184 // both a cost and, for `tests/test_async_registration.cpp`'s
1185 // allocation-failure case, the wrong allocation for its injector to
1186 // catch. `onDone` cannot be moved
1187 // the same way: it is captured from a `const` reference parameter,
1188 // so the capture itself is const.
1189 detail::deliverLate(bridgeExec, [this, weakBackend, weakLiveness, weakBinding,
1190 primaryCopy = std::move(primaryCopy), onDone, newId] {
1191 // Guards, locking and the liveness reasoning all live in
1192 // `publishLateBindReply`, which `ensureBoundAsync` shares.
1193 // What is this site's own is the three fields a successful
1194 // attach publishes -- and that they can throw, which is why
1195 // the helper wraps this in a try/catch.
1196 publishLateBindReply(weakLiveness, weakBinding, weakBackend, onDone,
1197 [&primaryCopy, newId](detail::HandlerBinding& target) {
1198 target.contextKey = primaryCopy;
1199 target.primary = primaryCopy;
1200 target.currentId.store(newId.v);
1201 });
1202 });
1203 };
1204 auto onFailed = [onDone, handoff, bridgeExec](const std::exception_ptr& failure) {
1205 if (detail::parkIfInFrame(*handoff, false, {}, failure)) {
1206 return;
1207 }
1208 // Reported on the bridge's executor for the same reason the
1209 // success path is: `onDone` is the caller's continuation, and a
1210 // bridge that named an executor wants it delivered there.
1211 detail::deliverLate(bridgeExec, [onDone, failure] { onDone(failure); });
1212 };
1213 try {
1214 // The structural surface (`IBackend::bindModel`). A backend with a
1215 // genuinely non-blocking attach settles the returned `Completion`
1216 // when its reply lands; one without settles it from inside this
1217 // call, having blocked for the same round trip the synchronous
1218 // `attachModel` would. Either way the continuation exists, so there
1219 // is no second path here.
1220 //
1221 // `inlineExecutor()` names, as a value, "deliver on whichever
1222 // thread the backend settled the reply on"; see
1223 // `exec::detail::InlineExecutor`. An inline settle therefore
1224 // reaches `onAttached`/`onFailed` while `_attachMtx` is still
1225 // held, which is precisely the case `handoff` exists for.
1226 auto completion =
1227 backend->bindModel(::morph::backend::detail::BindRequest{.typeId = binding->typeId,
1228 .factory = binding->modelFactory,
1229 .contextKey = primaryCopy,
1230 .primary = primaryCopy,
1231 .current = previous},
1232 ::morph::exec::detail::inlineExecutor());
1233 completion.then(onAttached).onError(onFailed);
1234 } catch (...) {
1235 // `IBackend::bindModel`'s own default converts a throwing
1236 // synchronous verb into a rejection, and every backend in the tree
1237 // does the same. This guard is for an out-of-tree override that
1238 // throws out of the dispatch call itself (e.g. a `wire::encode()`
1239 // failing before send): report it like any other failure rather
1240 // than let it escape execute()'s documented never-throws contract.
1241 lock.unlock();
1242 onDone(std::current_exception());
1243 return;
1244 }
1245 if (auto parked = detail::claimHandoff(*handoff)) {
1246 // The backend answered on this very stack, with `_attachMtx` still
1247 // held -- switchBackend() cannot have run concurrently (it takes
1248 // the same lock), so no staleness check is needed here. Publish
1249 // under the lock we already own, then release it and report --
1250 // @p onDone never runs inside the dispatch frame.
1251 std::exception_ptr failure = parked->failure;
1252 if (parked->succeeded) {
1253 try {
1254 binding->contextKey = primaryCopy;
1255 binding->primary = std::move(primaryCopy);
1256 binding->currentId.store(parked->modelId.v);
1257 } catch (...) {
1258 failure = std::current_exception();
1259 }
1260 }
1261 lock.unlock();
1262 onDone(failure);
1263 return;
1264 }
1265 // Nothing parked: the reply is still to come, and `onAttached`/
1266 // `onFailed` will deliver it themselves once it does.
1267 }

1297 /// @param binding Shared binding to bind.
1298 /// @param onDone Invoked exactly once: `nullptr` on success, or a

1376 /// @brief Files @p binding's current instance under @p primary, in place.
1377 ///
1378 /// The instance keeps everything the creating action just did — nothing is
1379 /// re-created and nothing is stranded. A no-op if @p binding already holds
1380 /// a different real primary (the locally cached primary must not race
1381 /// ahead of the backend's own refusal to re-key an already-keyed
1382 /// instance — see `IBackend::assignPrimary`).
1383 ///
1384 /// Known gap: if @p binding was still anonymous but the *target* key is
1385 /// already held by a different instance, the backend's `assignPrimary`
1386 /// silently declines to promote (the existing holder always wins), but
1387 /// this method has no way to learn that and still caches @p primary as
1388 /// though the promotion succeeded — `binding->primary()` can then report
1389 /// a key the backend never actually filed this instance under, and a
1390 /// same-key `attach()` after that becomes a silent no-op (the "already
1391 /// primary == primary" guard in `attachHandler`), so the binding can
1392 /// never reach the instance actually holding that key. Closing this
1393 /// requires `assignPrimary`'s outcome to become observable (e.g. a
1394 /// `bool` return threaded across every `IBackend` implementation and, for
1395 /// wire backends, a reply field) — tracked as a follow-up, not fixed
1396 /// here.
1397 /// Reaches the backend through `IBackend::promoteModel` — the structural
1398 /// registration surface, and the only one.
1399 /// The same "avoid a nested-event-loop block that aborts a WASM
1400 /// main thread" rationale `Bridge::registerHandler()` follows for the
1401 /// initial bind step applies here: this method is invoked from inside the
1402 /// result `Completion`'s callback chain (`BridgeHandler::execute`'s
1403 /// `onResult`), not from the original call stack, so there is no caller
1404 /// left blocked waiting on it either way — a non-blocking promote simply
1405 /// avoids parking the Qt event loop for the round trip.
1406 /// `binding->contextKey`/`primary` are only published once the reply
1407 /// confirms the call, so a caller reading `binding->primary()` never sees a
1408 /// promotion the backend has not actually completed. A backend with no
1409 /// non-blocking promote settles inside the `promoteModel` call, publishing
1410 /// before this method returns; a failure is logged rather than thrown,
1411 /// since `promoteModel` reports through its `Completion` by contract.
1412 ///
1413 /// `_attachMtx` is released *before* the dispatch — not held across it —
1414 /// mirroring `registerHandlerImpl`'s discipline: the success callback
1415 /// below re-acquires
1416 /// `_attachMtx`, so a backend that ever invoked it synchronously (none
1417 /// documented here do, but nothing prevents one from doing so) would
1418 /// otherwise self-deadlock re-acquiring a mutex this same call stack
1419 /// still held.
1420 /// @tparam Model Concrete model type.
1421 /// @param binding Shared binding whose instance is being promoted.
1422 /// @param primary Canonical string encoding of the key to file it under.
1423 template <typename Model>
1424 void assignHandlerPrimary(const std::shared_ptr<detail::HandlerBinding>& binding, std::string primary) {
1425 std::shared_ptr<::morph::backend::detail::IBackend> backend;
1426 uint64_t raw = 0;
1427 {
1428 std::scoped_lock const lock{_attachMtx};
1429 raw = binding->currentId.load();
1430 if (raw == 0U || primary.empty() || !binding->primary.empty()) {
1431 return;
1432 }
1433 backend = loadBackend();
1434 }
1435 auto const weakLiveness = _callbacks.token();
1436 std::weak_ptr<::morph::backend::detail::IBackend> const weakBackend{backend};
1437 std::weak_ptr<detail::HandlerBinding> const weakBinding{binding};
1438 auto onPromoted = [this, weakLiveness, weakBackend, weakBinding, primary] {
1439 // This check and the `this` touch below it are two steps. That
1440 // window is closed not by a gate but by the thread this body runs
1441 // on: a reply arriving after this frame goes to the bridge's own
1442 // executor, and an in-frame reply is published by this frame.
1443 // Gating instead would block `~Bridge` behind `_attachMtx`, which
1444 // `attachHandler` holds across a full `attachModel` round trip.
1445 if (!weakLiveness.active()) {
1446 return; // The Bridge is gone; do not touch `this`.
1447 }
1448 auto strongBinding = weakBinding.lock();
1449 if (!strongBinding) {
1450 return; // The BridgeHandler (and its binding) is gone.
1451 }
1452 std::scoped_lock const attachLock{_attachMtx};
1453 auto pinned = weakBackend.lock();
1454 if (!pinned || pinned != loadBackend()) {
1455 // A switchBackend() already moved past this promotion;
1456 // applying a stale reply now would overwrite whatever
1457 // state the new backend's re-registration already
1458 // established -- same reasoning as registerHandlerImpl's
1459 // async callback.
1460 return;
1461 }
1462 // A binding whose primary is already set (by a concurrent
1463 // attach/assign that raced ahead of this async reply, or
1464 // simply already promoted) must not be overwritten here.
1465 if (!strongBinding->primary.empty()) {
1466 return;
1467 }
1468 strongBinding->contextKey = primary;
1469 strongBinding->primary = primary;
1470 };
1471 auto onFailed = [typeId = binding->typeId](const std::string& message) {
1472 ::morph::log::logError("[assignHandlerPrimary] async promotion of '" + typeId + "' failed: " + message);
1473 };
1474 // The structural surface (`IBackend::promoteModel`). A backend with
1475 // no non-blocking promote settles the returned `Completion` from
1476 // inside this call, having run the synchronous `assignPrimary`;
1477 // `inlineExecutor()` then delivers the reply on this thread, and the
1478 // `claimHandoff` below publishes it before this method returns, which
1479 // `BridgeHandler::execute`'s `onResult` relies on ("the binding is
1480 // already promoted by the time user code sees the result"). A failure
1481 // is logged rather than thrown: this runs inside the result
1482 // `Completion`'s callback chain, where an escaping exception is
1483 // swallowed by `CompletionState` anyway, and `promoteModel` reports
1484 // through the `Completion` by contract.
1485 //
1486 // The handoff is what separates the two delivery cases, and is why
1487 // this site has one at all: a reply that
1488 // lands in this frame is published by this frame, while one that
1489 // lands later goes to the bridge's executor. Without it the callback
1490 // could not tell the two apart, and posting *both* would make an
1491 // inline promote asynchronous.
1492 auto handoff = std::make_shared<detail::AsyncDispatchHandoff>();
1493 auto* const bridgeExec = _bridgeExec;
1494 auto completion = backend->promoteModel(
1495 ::morph::backend::detail::PromoteRequest{
1496 .mid = ::morph::exec::detail::ModelId{raw}, .typeId = binding->typeId, .primary = primary},
1497 ::morph::exec::detail::inlineExecutor());
1498 completion
1499 .then([onPromoted, handoff, bridgeExec](::morph::exec::detail::ModelId newId) {
1500 if (detail::parkIfInFrame(*handoff, true, newId, nullptr)) {
1501 return; // Settled inline: the dispatching frame owns the outcome.
1502 }
1503 detail::deliverLate(bridgeExec, onPromoted);
1504 })
1505 .onError([onFailed, handoff, bridgeExec](const std::exception_ptr& failure) mutable {
1506 if (detail::parkIfInFrame(*handoff, false, {}, failure)) {
1507 return;
1508 }
1509 detail::deliverLate(bridgeExec, [onFailed = std::move(onFailed), failure] {
1510 onFailed(detail::describeFailure(failure));
1511 });
1512 });
1513 if (auto parked = detail::claimHandoff(*handoff)) {
1514 if (parked->succeeded) {
1515 onPromoted();
1516 } else {
1517 onFailed(detail::describeFailure(parked->failure));
1518 }
1519 }
1520 }

1543 /// @brief Resolves once @p binding's initial registration settles.
1544 ///
1545 /// Closes the gap `executeVia`'s fast-fail leaves for a caller that
1546 /// constructs a `BridgeHandler` against a backend that answers
1547 /// `BindWait::kCallerMustNotBlock` (see `backend.md`, "Waiting for a
1548 /// bind") and wants to dispatch the
1549 /// moment registration completes, rather than failing fast with "handler
1550 /// not bound" or polling `isBound()` in a loop of its own devising.
1551 ///
1552 /// - If @p binding is already bound, resolves immediately with `true`.
1553 /// - If an async registration is still in flight, resolves with `true`
1554 /// once `onRegistered` fires, or with the registration's error via
1555 /// `.onError(...)` if it fails.
1556 /// - If @p binding was never handed an async registration at all (the
1557 /// synchronous fallback path, or a binding still awaiting its very
1558 /// first `registerHandlerImpl` call to even attempt one) and is not yet
1559 /// bound, resolves immediately with `false` — there is nothing in
1560 /// flight to wait for. This can only happen for a `shared` binding that
1561 /// has not yet been given a primary (`ensureBound`/`attachHandler`
1562 /// never ran) or in the narrow window before `registerHandlerImpl`
1563 /// itself has run; ordinary (non-shared) bindings are always either
1564 /// bound or mid-registration by the time a caller can observe them.
1565 /// @param binding Binding to wait on.
1566 /// @param cbExec Executor the resolution is delivered on, exactly like
1567 /// every other `Completion`-returning call in this class.
1568 /// @return `Completion<bool>` resolving `true` once bound, `false` if
1569 /// nothing is in flight, or an error if registration failed.
1570 [[nodiscard]] ::morph::async::Completion<bool> whenBound(const std::shared_ptr<detail::HandlerBinding>& binding,
1571 ::morph::exec::IExecutor* cbExec) {
1572 auto state = std::make_shared<::morph::async::detail::CompletionState<bool>>();
1573 ::morph::async::Completion<bool> comp{state, cbExec};
1574 if (isBound(binding)) {
1575 state->setValue(true);
1576 return comp;
1577 }
1578 std::scoped_lock const lock{binding->registrationMtx};
1579 // Re-check under the lock: registerHandlerImpl's callback may have
1580 // resolved (and bound) between the lock-free check above and here.
1581 if (isBound(binding)) {
1582 state->setValue(true);
1583 return comp;
1584 }
1585 if (!binding->registrationInFlight) {

1799 /// @note Lock ordering: `Bridge::_mtx` is acquired before
1800 /// `LocalBackend::_regMtx`. `notifyBackendChanged()` runs under `_mtx`
1801 /// but only **posts** each model's `onBackendChanged()` onto that
1802 /// model's strand (see `LocalBackend::notifyBackendChanged`); the model

1804 /// `onBackendChanged()` implementation **may** re-enter the bridge
1805 /// (`switchBackend`/`registerHandler`/`deregisterHandler`) — it acquires
1806 /// `_mtx` freshly on the strand thread rather than deadlocking on the
1807 /// switch caller's still-held lock. It is also strand-serialised against

1811 /// `unique_ptr<IBackend>` directly) so it is an exact match for a
1812 /// `unique_ptr<Concrete>` argument (e.g. `std::make_unique<LocalBackend>(...)`)
1813 /// and is preferred over the `shared_ptr<IBackend>` overload during overload
1814 /// resolution — both would otherwise require an equally-ranked user-defined
1815 /// conversion (unique_ptr<Concrete> -> unique_ptr<IBackend> vs. unique_ptr<Concrete>
1816 /// -> shared_ptr<IBackend>), making every existing call site ambiguous.
1817 /// @tparam Backend Concrete backend type; must derive from `IBackend`.
1818 /// @param newBackend Replacement backend. Ownership is transferred.
1819 template <typename Backend>

1825 /// @brief Atomically replaces the active backend with @p newBackend.
1826 ///
1827 /// Identical to the `unique_ptr` overload, except the caller keeps shared
1828 /// ownership of @p newBackend — the backend can be re-installed later
1829 /// (e.g. switching back to a long-lived remote backend after a temporary
1830 /// fallback to a local one) without reconstructing it.
1831 ///
1832 /// @param newBackend Replacement backend, shared with the caller.
1833 void switchBackend(std::shared_ptr<::morph::backend::detail::IBackend> newBackend) {
1834 auto newShared = std::move(newBackend);
1835 // Stamp the current default session onto the new backend before phase 1
1836 // below builds a single `register`/`registerShared` envelope per live
1837 // binding — otherwise every control envelope re-registering handlers on
1838 // the new backend would carry a default-constructed (unauthenticated)
1839 // session, which is the gap this hook closes. Read under
1840 // `_sessionMtx` alone (a leaf mutex never held while calling into this
1841 // backend), mirroring `executeVia`'s copy-then-release pattern.
1842 {
1843 std::scoped_lock const lock{_sessionMtx};
1844 newShared->setSession(_defaultSession);
1845 }
1846 std::shared_ptr<::morph::backend::detail::IBackend> previous;
1847 // Settled after the mutexes are released, never under them: resolving a
1848 // waiter runs consumer code, which is free to call back into this
1849 // `Bridge`.
1850 std::vector<std::shared_ptr<detail::HandlerBinding>> settleBound;
1851 std::vector<std::shared_ptr<detail::HandlerBinding>> settleFailed;
1852 std::exception_ptr staging;
1853 // Whether this frame may stop and wait for each bind to settle. A
1854 // backend that answers `kCallerMustNotBlock` delivers its replies
1855 // through the calling thread's own event loop, so waiting here is a
1856 // deadlock rather than a delay — on a WASM main thread, a page abort.
1857 // See `IBackend::bindWaitPolicy`.
1858 bool const mayBlock = newShared->bindWaitPolicy() == ::morph::backend::detail::BindWait::kCallerMayBlock;
1859 {
1860 // Both mutexes: this phase reads/writes every live binding's
1861 // `primary`/`contextKey` (via `_attachMtx`'s ownership of those
1862 // fields) as well as `_handlers` itself (via `_mtx`), and must not
1863 // race a concurrent attachHandler()/ensureBound()/
1864 // assignHandlerPrimary() call on any one of them.
1865 std::scoped_lock const lock{_mtx, _attachMtx};
1866
1867 detail::StagedRebinds rebinds;
1868 staging = stageRebinds(newShared, mayBlock, rebinds);
1869 if (staging) {
1870 rollbackStaged(*newShared, rebinds.staged);
1871 settleFailed = std::move(rebinds.armed);
1872 } else {
1873 previous = commitRebinds(newShared, mayBlock, rebinds, settleBound);
1874 }
1875 }
1876 for (const auto& binding : settleFailed) {
1877 resolveRegistrationWaiters(*binding, /*ok=*/false, staging);
1878 }
1879 for (const auto& binding : settleBound) {
1880 resolveRegistrationWaiters(*binding, /*ok=*/true, nullptr);
1881 }
1882 if (staging) {
1883 // Rethrown here rather than from inside the locked block, so the
1884 // waiters above are settled outside `_mtx`/`_attachMtx`. Everything
1885 // below stays unreached on this path: a staging failure leaves the
1886 // outgoing backend's reconnect handler and its pending completions
1887 // alone, because the switch did not happen.
1888 std::rethrow_exception(staging);
1889 }
1890 if (previous && previous != newShared) {
1891 previous->setReconnectHandler(nullptr);
1892 }
1893 // Drain the outgoing backend outside the bridge mutex: cancelPending
1894 // delivers callbacks through the caller's gui executor, and we never want
1895 // to hold _mtx while user code runs.
1896 if (previous && previous != newShared) {
1897 previous->cancelPending(std::make_exception_ptr(::morph::backend::BackendChangedError{}));
1898 }
1899 }

1906 void deregisterHandler(const std::shared_ptr<detail::HandlerBinding>& binding) {

1925 /// @brief Dispatches @p action against the model identified by @p binding.
1926 ///
1927 /// Snapshots the backend `shared_ptr` under `_backendMtx` and reads the
1928 /// binding's `currentId` lock-free, so the call does not block
1929 /// `switchBackend()`. If a backend switch happens concurrently,
1930 /// the old backend still exists (its `shared_ptr` refcount is > 0) and the
1931 /// call either succeeds or fails with "model not found" — both are safe.
1932 ///
1933 /// On `LocalBackend`, the `localOp` this method builds first overwrites any
1934 /// declared computed fields from their inputs (`morph::forms::recomputeAll`,
1935 /// a no-op for actions with no `computedFields`), then enforces
1936 /// `morph::model::ActionValidator<Action>::ready(action)` before calling
1937 /// `Model::execute`, mirroring `ActionDispatcher::registerAction`'s runner
1938 /// (`registry.hpp`) for the in-process path. A `false` result resolves the
1939 /// returned `Completion` through `onError` with a `morph::model::ValidationError`
1940 /// instead of executing the action — see docs/spec/core/registry.md and
1941 /// docs/spec/forms/forms.md.
1942 ///
1943 /// @tparam Model Model type that owns the handler.
1944 /// @tparam Action Action type to dispatch.
1945 /// @param binding Binding returned by `registerHandler<Model>()`.
1946 /// @param action Action to execute (moved in).
1947 /// @param cbExec Executor on which the `Completion` callbacks are posted.
1948 /// @param onResult Optional observer run on the typed result *before* it is
1949 /// moved into the returned `Completion` and before the
1950 /// caller's own `.then`. Used to adopt a result-sourced
1951 /// primary key so the binding is already promoted by the time
1952 /// user code sees the result; empty for every other call.
1953 /// @return Completion that resolves with the typed result or an exception
1954 /// (including `ValidationError` on `LocalBackend` when the action
1955 /// fails its validator).
1956 template <typename Model, typename Action>
1957 // A dispatch function with a carefully reasoned safety argument threaded through its branches (see
1958 // the .then/.onError comments below); extracting helpers would separate that reasoning from the code
1959 // it justifies more than it would simplify either half. Same treatment as remote.hpp's
1960 // comparably-shaped dispatch functions.
1961 // NOLINTNEXTLINE(readability-function-cognitive-complexity)
1962 ::morph::async::Completion<typename ::morph::model::ActionTraits<Action>::Result> executeVia(
1963 const std::shared_ptr<detail::HandlerBinding>& binding, Action action, ::morph::exec::IExecutor* cbExec,
1964 std::function<void(const typename ::morph::model::ActionTraits<Action>::Result&)> onResult = {}) {
1965 using R = ::morph::model::ActionTraits<Action>::Result;
1966
1967 auto backend = loadBackend();
1968 uint64_t const raw = binding->currentId.load();
1969
1970 // One allocation, not six: this object *is* the typed completion state
1971 // the caller gets, and is also the `ISettleSink` the backend settles --
1972 // so there is no second, erased completion and no `.then`/`.onError`
1973 // pair forwarding one into the other. See `detail::BridgeSink` for the
1974 // four invariants its settle path holds.
1975 auto sink = std::make_shared<detail::BridgeSink<R>>(::morph::exec::detail::ModelId{raw}, std::move(onResult),
1976 _pendingCalls, _subscriptions, _lifetime);
1977 ::morph::async::Completion<R> typed{sink, cbExec};
1978 if (raw == 0U) {
1979 // Resolved synchronously, before any dispatch -- never counted as
1980 // pending, so nothing to decrement here (see pendingCalls()).
1981 sink->setException(std::make_exception_ptr(std::runtime_error("handler not bound")));
1982 return typed;
1983 }
1984 // Counted from here on: every path below either reaches the backend
1985 // dispatch or falls through to one of the two mutually-exclusive
1986 // resolution continuations attached to anyCompletion further down,
1987 // each of which decrements exactly once (setValue/setException are
1988 // first-result-wins, so only one of the two ever actually fires).
1989 _pendingCalls->fetch_add(1, std::memory_order_relaxed);
1990 // Arm the client-side deadline (setExecuteDeadline) only for real
1991 // dispatches -- the fast-failed "handler not bound" completion above is
1992 // already resolved and needs no timer. Reading the deadline and arming
1993 // it happen under one lock so a concurrent setExecuteDeadline() cannot
1994 // interleave between the two.
1995 // A private shared_ptr copy, obtained under the same lock as the
1996 // schedule() call -- not `this->_timeoutScheduler`, and not gated on
1997 // `alive` -- so the sink's own disarm can
1998 // call `cancel()` on a scheduler that is provably still alive,
1999 // regardless of whether ~Bridge() has run or is running concurrently
2000 // on another thread. `_executeDeadlineMtx` alone does not establish
2001 // that: ~Bridge()'s own body never acquires it, so a plain
2002 // `!alive.active()` check followed by `_timeoutScheduler->cancel()`
2003 // a few instructions later is a check-then-use race against
2004 // ~Bridge()'s implicit member destruction (which joins
2005 // TimeoutScheduler's thread). Holding a shared_ptr for the
2006 // callback's own lifetime turns that into a non-issue by
2007 // construction: while any copy of it is alive, ~TimeoutScheduler()
2008 // cannot run at all.
2009 std::shared_ptr<::morph::async::detail::TimeoutScheduler> schedulerRef;
2010 {
2011 std::scoped_lock const lock{_executeDeadlineMtx};
2012 if (_executeDeadline.count() > 0 && _timeoutScheduler) {
2013 schedulerRef = _timeoutScheduler;
2014 // The callback captures the sink alone -- never `this` -- so
2015 // it stays safe to fire even while ~Bridge() is running, and
2016 // equally safe to fire *after* its own `cancel()`:
2017 // `TimeoutScheduler::cancel` stops a callback that has not
2018 // started but returns without waiting for one that already has
2019 // (see that function's comment), so the sink's disarm is
2020 // best-effort by contract and not only
2021 // when `cancel()` throws. A deadline callback that is already
2022 // mid-flight when the real reply lands still runs its
2023 // `setException`, which `CompletionState` discards on an
2024 // already-ready state -- first result wins. That, not the
2025 // disarm, is what makes the race harmless.
2026 // The callback settles the *state*, not the sink: a deadline
2027 // that fires is not one of the two mutually-exclusive
2028 // resolution paths and must not decrement `_pendingCalls`,
2029 // which stays inflated until the real reply lands.
2030 auto const handle = schedulerRef->schedule(_executeDeadline, [sink] {
2031 sink->setException(std::make_exception_ptr(::morph::backend::ClientTimeoutError{}));
2032 });
2033 // Handed to the sink here, before the dispatch below, so the
2034 // write is sequenced before anything that could settle it on
2035 // another thread.
2036 sink->armDeadline(handle, schedulerRef);
2037 }
2038 }

2041 // addressed rather than copied: this whole block allocates exactly
2042 // once (the action itself). Copying instead would cost four

2056 call.deserializeResult = [](std::string_view jsonStr) -> std::shared_ptr<void> {

2058 };
2059 call.localOp = [](::morph::model::detail::IModelHolder& holder, void* actionPtr) -> std::shared_ptr<void> {
2060 // The action `ActionCall::action` owns, handed back typed. The
2061 // backend that invokes this keeps that handle alive across the
2062 // call (LocalBackend carries it onto the strand with `localOp`).
2063 Action& actionRef = *static_cast<Action*>(actionPtr);
2064 // Enforce the action's validator on the local execution path too, so
2065 // a caller that constructs an Action by hand and calls
2066 // BridgeHandler<Model>::execute<Action>() directly is rejected the
2067 // same way a hand-built wire envelope is rejected by
2068 // ActionDispatcher::registerAction's runner (registry.hpp). No JSON is
2069 // involved on this path, so there is no declared-precision
2070 // reconciliation step here (that only applies to decoded wire
2071 // payloads); the Quantity fields carry whatever precision the caller
2072 // constructed them with. ActionValidator<Action>::ready defaults to
2073 // `true` for actions with no validator, so this is a no-op for
2074 // unvalidated actions (zero behavior change). The thrown exception is
2075 // caught by LocalBackend::execute's strand task (backend.hpp) and
2076 // resolves this Completion through onError.
2077 //
2078 // Overwrite any computed fields from their declared inputs before the
2079 // validator runs and the model ever sees the action -- the same
2080 // authoritative recompute ActionDispatcher::registerAction's runner

2588 auto parked = rebindThroughSurface(binding, newBackend, mayBlock, "[switchBackend]");
2589 if (!parked) {
2590 out.deferred.push_back(binding);
2591 out.live.push_back(weak);
2592 continue;
2593 }
2594 if (!parked->succeeded) {
2595 // A rejected `Completion`: the structural surface reports
2596 // failure through the completion, not by throwing, so a
2597 // bare `catch (...)` cannot see it. The rollback keys on
2598 // this.
2599 return parked->failure;
2600 }
2601 out.staged.emplace_back(binding, parked->modelId.v);
2602 out.live.push_back(weak);
2603 }
2604 } catch (...) {
2605 // `bindModel` rejects rather than throws by contract, so this
2606 // guards only what the contract does not cover — an allocation in
2607 // this loop, or a backend that throws out of `bindModel` anyway.
2608 return std::current_exception();
2609 }
2610 return nullptr;
2611 }

2631 /// @brief `switchBackend`'s phase 2: publish the staged ids and swap the
2632 /// backend in. Caller holds `_mtx` and `_attachMtx`.
2633 ///
2634 /// A deferred binding is published as **unbound**: the id it held belonged
2635 /// to the backend being replaced and means nothing on the new one, while
2636 /// the new id is not here yet. Unbound is the honest state, and it is the
2637 /// one `executeVia` fails fast on and `whenBound()` gates on — strictly
2638 /// better than a dangling id from a backend nothing uses any more, which
2639 /// `executeVia` would happily dispatch against.
2640 /// @param newBackend Backend to install.
2641 /// @param mayBlock Whether staging was allowed to wait; when it was, no
2642 /// binding was armed and none needs settling.
2643 /// @param rebinds What staging accumulated.
2644 /// @param settleBound Out: bindings whose waiters the caller must resolve
2645 /// with `true`, once it has released the mutexes.
2646 /// @return The backend just replaced, for the caller to retire.
2647 [[nodiscard]] std::shared_ptr<::morph::backend::detail::IBackend> commitRebinds(
2648 const std::shared_ptr<::morph::backend::detail::IBackend>& newBackend, bool mayBlock,
2649 detail::StagedRebinds& rebinds, std::vector<std::shared_ptr<detail::HandlerBinding>>& settleBound) {
2650 for (const auto& [binding, newId] : rebinds.staged) {
2651 binding->currentId.store(newId);
2652 }
2653 for (const auto& binding : rebinds.deferred) {
2654 binding->currentId.store(0);
2655 }
2656 _handlers = std::move(rebinds.live);
2657
2658 auto previous = exchangeBackend(newBackend);
2659 newBackend->notifyBackendChanged();
2660 installReconnectHandler(newBackend);
2661 if (!mayBlock) {
2662 settleBound.reserve(rebinds.staged.size());
2663 for (const auto& [binding, newId] : rebinds.staged) {
2664 settleBound.push_back(binding);
2665 }
2666 }

2670 /// @brief Re-acquires an instance for every live binding on a backend that
2671 /// has just reconnected. Caller holds `_mtx` and `_attachMtx`.
2672 ///
2673 /// The reconnect counterpart of `stageRebinds`, and deliberately not the
2674 /// same function: there is nothing to stage here. The connection those ids
2675 /// belonged to is gone either way, so each binding is published as it
2676 /// settles and a failure costs only that binding.
2677 /// @param pinned The reconnected backend, already checked to be the active one.
2678 /// @param mayBlock Whether this frame may wait for each bind to settle.
2679 /// @return `(binding, failure)` pairs whose waiters the caller must resolve
2680 /// once it has released the mutexes; a null failure means success.
2681 [[nodiscard]] std::vector<std::pair<std::shared_ptr<detail::HandlerBinding>, std::exception_ptr>> reregisterLive(
2682 const std::shared_ptr<::morph::backend::detail::IBackend>& pinned, bool mayBlock) {
2683 std::vector<std::pair<std::shared_ptr<detail::HandlerBinding>, std::exception_ptr>> settle;
2684 for (auto& weak : _handlers) {
2685 auto binding = weak.lock();
2686 if (!binding) {
2687 continue;
2688 }
2689 // Same branch as switchBackend()'s staging phase: a shared binding
2690 // that never attached has no instance to re-create on the
2691 // reconnected backend either, and a shared binding that did attach
2692 // must come back through the `registerModelShared` request shape (a
2693 // non-empty `primary`), or it re-registers as a non-shared instance
2694 // and silently drops the sharing.
2695 if (binding->shared && binding->primary.empty()) {
2696 continue;
2697 }
2698 if (!mayBlock) {
2699 armRegistration(*binding);
2700 }
2701 auto parked = rebindThroughSurface(binding, pinned, mayBlock, "[reconnect]");
2702 if (!parked) {
2703 // The reply is still to come and will publish itself. The id
2704 // this binding holds belongs to the connection that just
2705 // dropped, so it is cleared rather than left to be dispatched
2706 // against: `executeVia` fails fast on an unbound handler, and
2707 // `whenBound()` — which this path is the first re-registration
2708 // ever to arm — lets a caller wait the reconnect out instead.
2709 binding->currentId.store(0);
2710 continue;
2711 }
2712 if (!parked->succeeded) {
2713 // A rejected `Completion` is reported per binding: this one is
2714 // left unbound and the loop carries on, rather than throwing
2715 // out of the handler onto the transport thread and taking
2716 // every binding after it along.
2717 binding->currentId.store(0);
2718 settle.emplace_back(binding, parked->failure);
2719 continue;
2720 }
2721 binding->currentId.store(parked->modelId.v);
2722 if (!mayBlock) {
2723 settle.emplace_back(binding, nullptr);
2724 }
2725 }
2726 return settle;
2727 }

2737 void installReconnectHandler(const std::shared_ptr<::morph::backend::detail::IBackend>& backend) {
2738 if (!backend) {
2739 return;
2740 }
2741 // The handler is invoked on the backend's transport thread. We keep a
2742 // weak_ptr to the same shared backend so a stale callback fired after a
2743 // switchBackend is a no-op, and a `CallbackToken` for the bridge
2744 // so a callback fired after the Bridge is destroyed (a co-owned backend
2745 // outliving its bridge) is also a no-op instead of a use-after-free on
2746 // `this`. `~Bridge` additionally clears the handler; this guard covers a
2747 // reconnect that is already in flight when the Bridge is torn down.
2748 std::weak_ptr<::morph::backend::detail::IBackend> const weakBackend{backend};
2749 auto const weakLiveness = _callbacks.token();
2750 backend->setReconnectHandler([this, weakBackend, weakLiveness] {
2751 if (!weakLiveness.active()) {
2752 return; // The Bridge is gone; do not touch `this`.
2753 }
2754 auto pinned = weakBackend.lock();
2755 // Settled after the mutexes are released, for the reason
2756 // switchBackend() gives: resolving a waiter runs consumer code.
2757 std::vector<std::pair<std::shared_ptr<detail::HandlerBinding>, std::exception_ptr>> settle;
2758 {
2759 // Both mutexes: reads `_handlers` (guarded by `_mtx`) and each
2760 // binding's `contextKey` (guarded by `_attachMtx`), same
2761 // reasoning as switchBackend() above.
2762 std::scoped_lock const lock{_mtx, _attachMtx};
2763 if (!pinned || pinned != loadBackend()) {
2764 return; // We've moved on to a different backend; ignore.
2765 }
2766 // The same question switchBackend() now asks, for the same
2767 // reason, and it matters more here: this handler runs on the
2768 // backend's *transport* thread, and for a `QtWebSocketBackend`
2769 // with `asyncRegistrationEnabled` that thread is the one whose
2770 // event loop has to deliver the reply. Calling the blocking
2771 // verb here would park that thread against itself, and on a
2772 // WASM main thread abort the page. See
2773 // `IBackend::bindWaitPolicy`.
2774 settle = reregisterLive(
2775 pinned, pinned->bindWaitPolicy() == ::morph::backend::detail::BindWait::kCallerMayBlock);
2776 }

2778 if (failure) {
2779 ::morph::log::logError("[reconnect] registration of '" + binding->typeId +
2780 "' failed: " + detail::describeFailure(failure));
2781 }
2782 resolveRegistrationWaiters(*binding, /*ok=*/failure == nullptr, failure);
2783 }
2784 });
2785 }

2792 // value into the continuation, never read *from* the callback: the

2883/// @brief `BridgeHandler` sharing policy: joins the shared instance directory.
2884///

2886/// instance the first time a keyed action or an explicit `attach()` names a

2897/// @brief RAII wrapper that binds a single model type to a `Bridge`.
2898///

2900/// deregisters it automatically. The handler is non-copyable.
2901///
2902/// @par Instance subscriptions
2903/// Beyond the one-shot `execute(action) -> Completion<R>` API, a handler can
2904/// observe *the instance it is attached to*:
2905///
2906/// - `subscribe<R>(cb)` fires whenever an `R` is produced on that instance, by
2907/// any handler attached to it.
2908/// - `unsubscribe<R>()` drops the callback.

2912// NOLINTNEXTLINE(cppcoreguidelines-special-member-functions)

2914public:
2915 /// @brief Whether this handler participates in the shared instance directory.

2919 ///
2920 /// @param bridge The bridge to register on. Borrowed, not owned: it must
2921 /// outlive every *call* made on this handler. Destruction
2922 /// order itself is unconstrained, on any thread —
2923 /// `~BridgeHandler` takes the bridge's `detail::BridgeLifetime`
2924 /// gate and deregisters nothing if the bridge is already gone
2925 /// (see "Lifetime & ownership" in `docs/spec/core/bridge.md`).

2927 /// GUI thread). Borrowed: it must outlive this handler.

2930 static_assert(!kShared || ::morph::model::KeyedModel<Model>,
2931 "BridgeHandler<Model, AllowShared> requires Model to declare a PrimaryKey alias");
2932 }

2934 /// @brief Constructs the handler with a pre-built binding (for dependency injection).
2935 ///
2936 /// @param bridge The bridge to register on. Borrowed, on the same terms as
2937 /// the constructor above.
2938 /// @param guiExec Executor for callback delivery. Borrowed: it must outlive
2939 /// this handler.
2940 /// @param binding Pre-built binding whose factory captures injected dependencies.

2948 ///
2949 /// If the `Bridge` has already been destroyed, this is a no-op: there is
2950 /// nothing to deregister from and dereferencing the dangling `Bridge&` would
2951 /// be undefined behaviour. Destroying the bridge before its handlers is still
2952 /// discouraged, but it is safe — in either order, and **from any thread**.
2953 ///
2954 /// The gate is held across the whole call, not merely across the question.
2955 /// A handler is routinely released on a thread that is not the bridge's

2977 /// @brief Dispatches @p action via the underlying `Bridge` and returns a `Completion`.
2978 ///
2979 /// The bridge's currently-installed default session (set once at startup or
2980 /// after login via `Bridge::setDefaultSession`) is attached automatically —
2981 /// callers never thread the session through individual call sites.
2982 ///
2983 /// @tparam Action Concrete action type registered with `BRIDGE_REGISTER_ACTION`.
2984 /// @param action Action to execute (moved into the dispatch).
2985 /// @return Completion that resolves on the GUI executor. A payload- or
2986 /// result-keyed action's attach/promote step never throws out of
2987 /// this call, even when the backend refuses it (e.g. a remote
2988 /// server at `LimitPolicy::maxLiveModels`, a transport error, or
2989 /// an unauthorized attach) — the failure is instead delivered
2990 /// through the returned Completion's `.onError(...)`, exactly
2991 /// like any other dispatch failure.
2992 template <typename Action>
2993 ::morph::async::Completion<typename ::morph::model::ActionTraits<Action>::Result> execute(Action action) {
2994 using R = ::morph::model::ActionTraits<Action>::Result;
2995 // Three mutually exclusive routes, chained rather than sequential: an
2996 // action is payload-keyed or result-keyed or neither (`PayloadKeyed`
2997 // and `ResultKeyed` differ only in `fromResult`, so no action can
2998 // satisfy both), and an unkeyed action — or any action at all on a
2999 // `NoSharing` handler — always lands in the final `else`. Chaining the
3000 // `if constexpr`s (rather than leaving the payload-keyed branch to
3001 // fall through to that `else`, as it did while the attach step was
3002 // synchronous) is what lets the keyed branches own their dispatch: the
3003 // attach now completes asynchronously, so the dispatch it precedes has
3004 // to happen from inside its completion callback, not on this stack.
3005 if constexpr (kShared && ::morph::model::detail::PayloadKeyed<Action>) {
3006 // The action names its instance: attach (or re-point) before
3007 // dispatching, so the call lands on the instance it asked for. A
3008 // remote backend's refusal (LimitPolicy::maxLiveModels, a
3009 // transport error, unauthorized) must surface through the
3010 // returned Completion's onError, exactly like every other
3011 // dispatch failure — not as a synchronous throw out of execute().
3012 //
3013 // The attach goes through Bridge::attachHandlerAsync, which
3014 // dispatches `IBackend::bindModel`: a backend with a genuinely
3015 // non-blocking attach settles later, one without settles inline,
3016 // having run the identical synchronous attach and called back
3017 // before returning — so a backend with no non-blocking path
3018 // still behaves synchronously here, just through the async
3019 // interface.
3020 auto state = std::make_shared<::morph::async::detail::CompletionState<R>>();
3021 ::morph::async::Completion<R> pending{state, _guiExec};
3022 auto* const bridgePtr = &_bridge;
3023 auto binding = _binding;
3024 // Key extraction is user code (ActionKeyTraits + keyToString), so it
3025 // is inside the same no-throw-out-of-execute() promise the attach
3026 // itself makes: a throw here resolves the Completion, it does not
3027 // escape.
3028 std::string key;
3029 try {
3030 key = ::morph::model::ActionKeyTraits<Action>::key(action);
3031 } catch (...) {
3032 state->setException(std::current_exception());
3033 return pending;
3034 }
3035 auto sharedAction = std::make_shared<Action>(std::move(action));
3036 bridgePtr->template attachHandlerAsync<Model>(
3037 binding, std::move(key),
3038 [bridgePtr, binding, sharedAction, state, guiExec = _guiExec](std::exception_ptr err) {
3039 if (err) {
3040 state->setException(err);
3041 return;
3042 }
3043 bridgePtr->template executeVia<Model, Action>(binding, std::move(*sharedAction), guiExec)
3044 .then([state](R value) { state->setValue(std::move(value)); })
3045 .onError([state](std::exception_ptr exc) { state->setException(exc); });
3046 });
3047 return pending;
3048 } else if constexpr (kShared && ::morph::model::detail::ResultKeyed<Action>) {
3049 // The action *creates* the instance and its result carries the
3050 // generated key, exactly as a database insert returns its primary
3051 // key. Adopt it before any user callback observes the result, so a
3052 // .then() can immediately run further actions on the new instance.
3053 //
3054 // The action generates its own key, so it must run before the key
3055 // exists. Give the handler an anonymous instance to run on, then
3056 // promote *that* instance once the reply names it — re-pointing to a
3057 // fresh one instead would strand whatever the create just did.
3058 // Same async/fallback contract as the payload-keyed branch above,
3059 // via Bridge::ensureBoundAsync.
3060 auto state = std::make_shared<::morph::async::detail::CompletionState<R>>();
3061 ::morph::async::Completion<R> pending{state, _guiExec};
3062 auto* const bridgePtr = &_bridge;
3063 auto binding = _binding;
3064 auto sharedAction = std::make_shared<Action>(std::move(action));
3065 bridgePtr->ensureBoundAsync(
3066 binding, [bridgePtr, binding, sharedAction, state, guiExec = _guiExec](std::exception_ptr err) {
3067 if (err) {
3068 state->setException(err);
3069 return;
3070 }
3071 bridgePtr
3072 ->template executeVia<Model, Action>(
3073 binding, std::move(*sharedAction), guiExec,
3074 [bridgePtr, binding](const R& result) {
3075 bridgePtr->template assignHandlerPrimary<Model>(
3076 binding, ::morph::model::ActionKeyTraits<Action>::template keyOfResult<R>(result));
3077 })
3078 .then([state](R value) { state->setValue(std::move(value)); })
3079 .onError([state](std::exception_ptr exc) { state->setException(exc); });
3080 });

3317/// (`NoSharing`, `AllowShared`) from the same generic-lambda template,
3318/// `static_cast`ing `handlerVoid` to the matching `BridgeHandler<Model,
3319/// Sharing>*` in each — see bridge.md's design-decision entry for why a single
3320/// `NoSharing`-only executor is unsound for a shared handler.

3322inline void ActionExecuteRegistry::registerAction(std::string_view modelId, std::string_view actionId) {
3323 auto makeExecutor = []<typename Sharing>() {
3324 return [](void* handlerVoid, std::string_view bodyJson) -> ::morph::async::Completion<std::string> {
3325 auto* handler = static_cast<BridgeHandler<Model, Sharing>*>(handlerVoid);
3326 auto resultState = std::make_shared<::morph::async::detail::CompletionState<std::string>>();
3327 try {
3328 Action action = ::morph::model::ActionTraits<Action>::fromJson(bodyJson);
3329 // Retag any Quantity fields to their declared precision so the stored
3330 // value matches the schema's advertised `x-decimalPlaces`, rather than
3331 // silently keeping whatever runtime `dp` the client sent. No-op for
3332 // actions with no Quantity members. See docs/spec/forms/forms.md.

3335 // engaged value falls outside its unit's declared bounds
3336 // (UnitTraits<E>::bounds), before the validator/business-rule
3337 // check below. No-op for actions with no Quantity members, or
3338 // whose units declare no bounds(). See docs/spec/forms/forms.md,
3339 // "Pre-decode wire validation". Thrown as QuantityDecodeError,
3340 // caught by the same catch block as every other decode/validation
3341 // failure on this path.
3342 ::morph::forms::enforceQuantityBounds(action);
3343 // Overwrite any computed fields from their declared inputs -- a
3344 // computed field is never trusted from the client, on any path.
3345 // No-op for actions with no computedFields. See docs/spec/forms/forms.md.
3346 ::morph::forms::recomputeAll(action);
3347 // Enforce the action's validator on the request/reply dispatch path,
3348 // just as the reactive `set<>` path does in
3349 // `morph::flows::FlowSession::set<>` (forms/flows.hpp), which fires a
