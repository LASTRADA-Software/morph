34/// @brief Type-erased, JSON-in/JSON-out execute path for actions whose
35/// concrete C++ type is only known by its registered string id at
36/// the call site (e.g. a schema-driven GUI that reads action names
37/// out of a JSON Schema at runtime).
38///
39/// Populated automatically by `BRIDGE_REGISTER_ACTION` — no action-specific
40/// code is required at any call site. Every entry calls through the real
41/// `BridgeHandler<Model>::execute<Action>()` (so sessions, backend
42/// switches, and completions all behave exactly as they do for hand-written
43/// call sites), unlike `morph::model::detail::ActionDispatcher`, which
44/// calls `Model::execute` directly against an already-owned model holder
45/// and is only ever used server-side.
46///
47/// HARD REQUIREMENT (other direction of the same constraint documented on
48/// `morph::model::detail::registerActionExecutorOnce` in `registry.hpp`): registration into
49/// this registry happens via `registerActionExecutorOnce<Model, Action>`, which
50/// `BRIDGE_REGISTER_ACTION` calls unconditionally but which is only defined here, in
51/// `bridge.hpp`. Every translation unit that calls `BRIDGE_REGISTER_ACTION` must therefore
52/// include this header (directly or transitively), or that translation unit's static
53/// initializer will fail to link.
54class ActionExecuteRegistry {
55public:
56 /// @brief Deserialises `bodyJson`, dispatches through the handler's
57 /// `Bridge`, and resolves with the JSON-encoded result.
58 using Executor = std::function<::morph::async::Completion<std::string>(void*, std::string_view)>;
59
60 /// @brief Registers the executors for `(Model, Action)` under the given string ids.
61 ///
62 /// Populates one `Executor` per `Sharing` policy the framework defines
63 /// (`NoSharing` and `AllowShared`), keyed by
64 /// `(modelId, actionId, typeid(Sharing))`, rather than a single executor
65 /// that unconditionally assumes `NoSharing`. A handler whose `Sharing` is
66 /// anything else must be dispatched through the *matching* executor: the
67 /// wrong one `static_cast`s `handlerVoid` to the wrong `BridgeHandler<Model,
68 /// Sharing>` instantiation, so `kShared` resolves incorrectly inside it and
69 /// a payload-/result-keyed action's attach-or-promote step silently never
70 /// runs. Both executors' bodies are otherwise byte-for-byte identical
71 /// (built once, from the same generic lambda template, instantiated for
72 /// `NoSharing` and again for `AllowShared` — the only two sharing tags
73 /// `morph::bridge` defines), so this costs one extra closure per
74 /// registered action, not per call.
75 /// Defined out-of-line after BridgeHandler to avoid forward reference issues.
76 /// @tparam Model Model type whose handler will execute the action.
77 /// @tparam Action Action type to register.
78 /// @param modelId String id the model is registered under.
79 /// @param actionId String id the action is registered under.
80 template <typename Model, typename Action>
81 void registerAction(std::string_view modelId, std::string_view actionId);
82
83 /// @brief Looks up and invokes the executor for `(modelId, actionId)`,
84 /// specialised for the caller's own `Sharing` policy.
85 /// @tparam Sharing `NoSharing` or `AllowShared` — the caller's own sharing policy.
86 /// @param modelId String id of the target model.
87 /// @param actionId String id of the action to execute.
88 /// @param handler Type-erased `BridgeHandler<Model, Sharing>*` matching `modelId`.
89 /// @param bodyJson JSON-encoded action payload.
90 /// @return Completion that resolves with the JSON-encoded action result.
91 /// @throws std::runtime_error if no executor was registered for that triple.
92 template <typename Sharing>
93 [[nodiscard]] ::morph::async::Completion<std::string> execute(std::string_view modelId, std::string_view actionId,
94 void* handler, std::string_view bodyJson) const {
95 auto iter =
96 _executors.find(Key{std::string{modelId}, std::string{actionId}, std::type_index{typeid(Sharing)}});
97 if (iter == _executors.end()) {
98 throw std::runtime_error("unknown action for executeJson: " + std::string{modelId} + "/" +
99 std::string{actionId});
100 }
101 return iter->second(handler, bodyJson);
102 }
103
104 /// @brief Returns the process-level singleton registry.
105 /// @return Reference to the singleton `ActionExecuteRegistry`.
106 static ActionExecuteRegistry& instance();
107
108private:
109 struct Key {
110 std::string modelId;
111 std::string actionId;
112 std::type_index sharing;
113 bool operator==(const Key&) const = default;
114 };
115 struct KeyHash {
116 std::size_t operator()(const Key& key) const noexcept {
117 std::size_t const modelHash = ::morph::model::detail::PairKeyHash{}({key.modelId, key.actionId});
118 return modelHash ^ (key.sharing.hash_code() + 0x9e3779b9U + (modelHash << 6) + (modelHash >> 2));
119 }
120 };
121 std::unordered_map<Key, Executor, KeyHash> _executors;
122};

124inline ActionExecuteRegistry& ActionExecuteRegistry::instance() {
125 static ActionExecuteRegistry inst;
126 return inst;
127}

163/// @brief Internal linkage record between a model type and its active backend id.
164///
165/// Shared between `Bridge` and `BridgeHandler`. `Bridge::switchBackend()` updates
166/// `currentId` atomically under its mutex so that concurrent `executeVia()` calls
167/// always see a consistent value.
168struct HandlerBinding {
169 /// @brief String type-id of the model.
170 std::string typeId;
171
172 /// @brief Factory used to re-register the model on a backend switch.
173 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> modelFactory;
174
175 /// @brief Stable identity of this model instance (e.g. an account id).
176 ///
177 /// Empty by default — the common local-mode case needs nothing here, since
178 /// `modelFactory` can already call `IModelHolder::attachActionLog` directly
179 /// with whatever identity it captures. Set this when the active backend may
180 /// be remote (`SimulatedRemoteBackend`, `QtWebSocketBackend`): it travels in
181 /// the `register` wire envelope so a server-side `RemoteServer::LogProvider`
182 /// can attach a log to the instance it creates. See `IBackend::registerModelWithContext`.
183 std::string contextKey;
184
185 /// @brief Canonical string encoding of this instance's primary key.
186 ///
187 /// Empty until a keyed action or an explicit `attach` supplies one. Only
188 /// meaningful when `shared` is set; a private binding never consults it.
189 /// Mutated only under `Bridge::_attachMtx` (or, during `switchBackend()`
190 /// and the reconnect handler, both `_attachMtx` and `_mtx` together).
191 std::string primary;
192
193 /// @brief Whether this binding participates in the shared instance directory.
194 ///
195 /// Set once at construction from `BridgeHandler`'s `Sharing` template
196 /// argument and never changed. A shared binding defers its backend
197 /// registration until it has a primary, so `currentId` stays `0` — and
198 /// `executeVia` fails fast on "handler not bound" — until then.
199 bool shared = false;
200
201 /// @brief Current `ModelId` value in the active backend (0 = unbound).
202 std::atomic<uint64_t> currentId{0};
203
204 /// @brief Registration-settled seam (see `Bridge::whenBound`).
205 ///
206 /// `registrationInFlight` is `true` from the moment `registerHandlerImpl`
207 /// hands this binding's initial registration to
208 /// `IBackend::registerModelAsync` (and that call returns `true`) until its
209 /// `onRegistered`/`onError` callback resolves — the synchronous fallback
210 /// path never sets it, since that call has already returned bound (or
211 /// thrown) by the time anyone could observe it in flight. `whenBound()`
212 /// checks it to distinguish "an async reply is coming, queue a waiter"
213 /// from "nothing is in flight, resolve false now". `registrationWaiters`
214 /// holds callbacks queued by `whenBound()` while `registrationInFlight` is
215 /// `true`; the resolving callback invokes and clears every one of them
216 /// exactly once, in the same call that clears `registrationInFlight`.
217 /// Guarded by `registrationMtx`, deliberately separate from
218 /// `Bridge::_mtx`/`_attachMtx`: a waiter can be queued or resolved from
219 /// either the registering thread or the backend's reply-delivering
220 /// thread, and must never block on the bridge's own locks.
221 std::mutex registrationMtx;
222 bool registrationInFlight = false;
223 std::vector<std::pair<std::function<void(bool)>, std::function<void(std::exception_ptr)>>> registrationWaiters;
224};

226/// @brief Outcome a backend's inline completion parked for its dispatcher.
227struct ParkedOutcome {
228 /// @brief `true` when the parked outcome is a success.
229 bool succeeded = false;
230 /// @brief Instance id the success callback reported.
231 ::morph::exec::detail::ModelId modelId{};
232 /// @brief Diagnostic the error callback reported; null on success.
233 std::exception_ptr failure;

236/// @brief Handoff slot between an async attach/bind dispatch and its callback.
237///
238/// `Bridge::attachHandlerAsync`/`ensureBoundAsync` dispatch to the backend while
239/// holding `_attachMtx`, and both promise to release it before invoking their
240/// `onDone`. A backend whose `attachModelAsync`/`registerModelSharedAsync`
241/// completed its callback *inline* — synchronously, before the dispatch call
242/// returned, as `QtWebSocketBackend` does on its `!_connected` error branch —
243/// would otherwise break that promise from inside the dispatch frame, with the
244/// lock still held.
245///
246/// So instead of acting, such a callback parks its outcome here and returns; the
247/// dispatching frame picks it up after the dispatch call returns, publishes it
248/// under the lock it already holds, releases the lock, and only then reports.
249/// `mtx` makes the handover race-free even for a backend that replies from
250/// another thread *while* its own dispatch call is still on this stack, and the
251/// `fired` flag keeps `onDone` invoked exactly once on every interleaving.
252struct AsyncDispatchHandoff {
253 /// @brief Guards every other field; never held across `onDone` or `_attachMtx`.
254 std::mutex mtx;
255 /// @brief `true` while the backend's dispatch call is still on the caller's stack.
256 bool inFrame = true;
257 /// @brief Set once either callback has claimed the outcome.
258 bool fired = false;
259 /// @brief `true` when the parked outcome is a success, `false` for a failure.
260 bool succeeded = false;
261 /// @brief Instance id the success callback reported.
262 ::morph::exec::detail::ModelId modelId{};
263 /// @brief Diagnostic the error callback reported; null on success.
264 std::exception_ptr failure;
265};
266
267/// @brief Records a backend callback's outcome in @p handoff.
268///
269/// Called first thing by both completion callbacks of an async attach/bind
270/// dispatch.
271///
272/// @param handoff Handoff slot created by the dispatching frame.
273/// @param succeeded `true` for the success callback, `false` for the error one.
274/// @param modelId Instance id, for the success callback; ignored otherwise.
275/// @param failure Diagnostic, for the error callback; null otherwise.
276/// @return `true` if the caller must **not** act — either because the dispatch
277/// call is still on the dispatcher's stack (which owns the outcome from
278/// here on) or because another callback already claimed this dispatch.
279/// `false` if the caller owns the outcome and should deliver it itself.
280inline bool parkIfInFrame(AsyncDispatchHandoff& handoff, bool succeeded, ::morph::exec::detail::ModelId modelId,
281 std::exception_ptr failure) {
282 std::scoped_lock const guard{handoff.mtx};
283 if (handoff.fired) {
284 // A backend is contractually allowed exactly one callback per dispatch;
285 // swallow a second one rather than reporting twice.
286 return true;
287 }
288 handoff.fired = true;
289 handoff.succeeded = succeeded;
290 handoff.modelId = modelId;
291 handoff.failure = std::move(failure);
292 return handoff.inFrame;
293}
294
295/// @brief Closes the inline window and takes whatever a callback parked.
296///
297/// Called by the dispatching frame immediately after the backend's dispatch call
298/// returns. After this, a callback that has not yet run delivers its own outcome.
299///
300/// @param handoff Handoff slot created by the dispatching frame.
301/// @return The parked outcome if the backend completed inline (or concurrently,
302/// before this frame closed the window); `std::nullopt` if the frame won
303/// the race and the reply, if any, is still to come.
304inline std::optional<ParkedOutcome> claimHandoff(AsyncDispatchHandoff& handoff) {
305 std::scoped_lock const guard{handoff.mtx};
306 handoff.inFrame = false;
307 if (!handoff.fired) {
308 return std::nullopt;
309 }
310 return ParkedOutcome{.succeeded = handoff.succeeded, .modelId = handoff.modelId, .failure = handoff.failure};
311}

315/// @brief Central dispatcher that routes typed actions to an `IBackend`.
316///
317/// `Bridge` owns exactly one active backend at a time. It tracks all registered
318/// `HandlerBinding` instances and re-registers them automatically when
319/// `switchBackend()` is called, enabling seamless local ↔ remote transitions.
320///
321/// @par Thread safety
322/// All public methods are thread-safe. `executeVia()` uses a lock-free snapshot
323/// of the backend pointer so it does not block `switchBackend()`.
324class Bridge {
325public:
326 /// @brief Constructs a bridge that dispatches through @p backend.
327 ///
328 /// Installs a reconnect handler so backends with a recoverable transport
329 /// (e.g. `QtWebSocketBackend`) can ask the bridge to re-register every live
330 /// handler against the freshly reconnected peer.
331 ///
332 /// @param backend Initial backend. Ownership is transferred.
333 explicit Bridge(std::unique_ptr<::morph::backend::detail::IBackend> backend)
334 : _backend{std::shared_ptr<::morph::backend::detail::IBackend>(std::move(backend))} {
335 installReconnectHandler(_backend);
336 // A null initial backend is tolerated (see installReconnectHandler's own
337 // null check just above) — there is nothing yet to stamp a session onto.
338 if (_backend) {
339 _backend->setSession(_defaultSession);
340 }
341 }
342
343 /// @brief Cancels every still-pending completion on the active backend.
344 ///
345 /// Each `.onError(...)` callback receives a `BridgeDestroyedError`. In-flight
346 /// server replies that arrive after destruction are no-ops because each
347 /// `CompletionState::setValue`/`setException` is idempotent.
348 ///
349 /// Before cancelling, the active backend's reconnect handler is cleared: the
350 /// installed handler captures `this`, and a co-owned backend that outlives the
351 /// `Bridge` could otherwise fire it after destruction and dereference freed
352 /// memory. Clearing it here (outside `_mtx` — `setReconnectHandler` only stores
353 /// a callback and never re-enters the bridge) closes the common case; the
354 /// handler additionally guards on `_callbacks` so a reconnect already in flight
355 /// on the transport thread becomes a no-op too. See docs/spec/concurrency_and_lifetimes.md.
356 ~Bridge() {
357 if (auto active = loadBackend()) {
358 active->setReconnectHandler(nullptr);
359 active->cancelPending(std::make_exception_ptr(::morph::backend::BridgeDestroyedError{}));
360 }
361 }
362
363 Bridge(const Bridge&) = delete;
364 Bridge& operator=(const Bridge&) = delete;
365 Bridge(Bridge&&) = delete;
366 Bridge& operator=(Bridge&&) = delete;

411 /// @brief Attaches (or re-points) @p binding to the shared instance for @p primary.
412 ///
413 /// Idempotent: attaching to the primary a binding already holds is a no-op,
414 /// so a keyed action repeated against the same instance costs nothing. A
415 /// different primary re-points the binding — the previous instance is
416 /// released and survives only if another handler still holds it.
417 ///
418 /// The binding's `contextKey` is set to @p primary as well, so a keyed
419 /// model's journal entries carry the entity key without the caller
420 /// arranging it separately.
421 ///
422 /// @tparam Model Concrete model type.
423 /// @param binding Shared binding, as returned by `registerSharedHandler<Model>()`.
424 /// @param primary Canonical string encoding of the primary key to attach to.
425 template <typename Model>
426 void attachHandler(const std::shared_ptr<detail::HandlerBinding>& binding, std::string primary) {
427 std::scoped_lock const lock{_attachMtx};
428 if (binding->primary == primary && binding->currentId.load() != 0U) {
429 return;
430 }
431 auto const previous = ::morph::exec::detail::ModelId{binding->currentId.load()};
432 // IBackend::attachModel's default (local) implementation and
433 // RemoteServer's wire-level attach handling both acquire the
434 // replacement before releasing `previous`, so a throwing acquire
435 // leaves `previous` untouched except in one narrow, harmless case: a
436 // remote re-point whose connection scope closes concurrently with
437 // the attach, where `previous` may already be released by the time
438 // the reply reports failure -- but no further request on this
439 // binding will ever run at that point either, so nothing is actually
440 // lost from the caller's perspective. Unbind first is therefore
441 // unnecessary: publish only on success.
442 auto newId = loadBackend()->attachModel(binding->typeId, binding->modelFactory,
443 {.contextKey = primary, .primary = primary}, previous);
444 binding->contextKey = primary;
445 binding->primary = std::move(primary);
446 binding->currentId.store(newId.v);
447 }

449 /// @brief Async counterpart to `attachHandler`: prefers the backend's
450 /// `attachModelAsync` when available, invoking @p onDone once
451 /// attached (or failed) instead of blocking.
452 ///
453 /// Falls back to the synchronous `attachHandler` body (and calls @p onDone
454 /// immediately, from this thread) when the backend offers no async
455 /// path — so a caller that always goes through this method behaves
456 /// identically to calling `attachHandler` directly, on every backend
457 /// that has not opted in to `attachModelAsync`.
458 ///
459 /// @par Locking
460 /// `_attachMtx` is held around the guard check, the async branch's
461 /// *dispatch*, and the synchronous branch's own state mutation — matching
462 /// `attachHandler`'s existing lock scope — but is **released before
463 /// @p onDone is ever invoked**, on every path, unconditionally. That is not
464 /// a nicety: what `execute()` does from inside @p onDone is dispatch the
465 /// action, and a result-keyed dispatch promotes its binding through
466 /// `assignHandlerPrimary`, which takes `_attachMtx` itself. Invoking
467 /// @p onDone under the lock therefore self-deadlocks the moment the
468 /// completion is delivered on the calling thread — which is exactly what
469 /// the synchronous fallback below does, and what an inline executor does
470 /// for every callback. This is `registerHandlerImpl`'s existing rule ("the
471 /// backend call must not run under `_mtx`") applied to `_attachMtx`.
472 ///
473 /// The guarantee holds even for a backend that completes its callback
474 /// *inline*, from inside `attachModelAsync` itself, while this frame still
475 /// holds the lock: such a callback parks its outcome in a
476 /// `detail::AsyncDispatchHandoff` and returns without acting, and this frame
477 /// applies it after the dispatch call has returned and the lock is gone.
478 /// See that struct's doc comment.
479 ///
480 /// An out-of-frame success callback re-acquires `_attachMtx` for the two
481 /// `std::string` fields it publishes (`contextKey`/`primary`, which
482 /// `HandlerBinding` documents as readable only under that lock) and drops it
483 /// again before calling @p onDone. That re-acquisition is safe precisely
484 /// because the inline case never reaches it.
485 ///
486 /// @par Known gap
487 /// Two calls for the same key issued before the first one's reply arrives
488 /// are **not** deduplicated: the guard below reads `binding->primary`/
489 /// `currentId`, neither of which is updated until the reply lands, so both
490 /// calls pass it and both dispatch an `attach`. This is a real behaviour
491 /// difference from the synchronous `attachHandler` it replaces, not merely
492 /// something inherent to asynchrony — `attachHandler` held `_attachMtx`
493 /// across the whole blocking round trip, which serialised concurrent
494 /// callers for free. It needs no second thread to hit: two `execute()`
495 /// calls in one event-loop turn are enough. The server answers both with
496 /// the same `ModelId` but counts two attachments, so one attach reference
497 /// leaks; the leak is bounded, not unbounded — the connection scope
498 /// releases every reference it holds when it closes. Closing this properly
499 /// needs in-flight tracking on the binding (coalescing the second caller
500 /// onto the first dispatch's completion); tracked as a follow-up, not fixed
501 /// here. Same gap, same reasoning, on `ensureBoundAsync`.
502 ///
503 /// @tparam Model Concrete model type.
504 /// @param binding Shared binding, as returned by `registerSharedHandler<Model>()`.
505 /// @param primary Canonical string encoding of the primary key to attach to.
506 /// @param onDone Invoked with `nullptr` on success, or a non-null
507 /// `exception_ptr` on failure — always exactly once,
508 /// synchronously if the fallback path is taken.
509 template <typename Model>
510 void attachHandlerAsync(const std::shared_ptr<detail::HandlerBinding>& binding, std::string primary,
511 const std::function<void(std::exception_ptr)>& onDone) {
512 std::unique_lock lock{_attachMtx};
513 if (binding->primary == primary && binding->currentId.load() != 0U) {
514 lock.unlock();
515 onDone(nullptr);
516 return;
517 }
518 auto const previous = ::morph::exec::detail::ModelId{binding->currentId.load()};
519 auto backend = loadBackend();
520 auto primaryCopy = primary;
521 std::weak_ptr<::morph::backend::detail::IBackend> const weakBackend{backend};
522 auto const weakLiveness = _callbacks.token();
523 std::weak_ptr<detail::HandlerBinding> const weakBinding{binding};
524 auto handoff = std::make_shared<detail::AsyncDispatchHandoff>();
525 bool started = false;
526 try {
527 started = backend->attachModelAsync(
528 binding->typeId, binding->modelFactory, {.contextKey = primaryCopy, .primary = primaryCopy}, previous,
529 [this, weakBackend, weakLiveness, weakBinding, primaryCopy, onDone,
530 handoff](::morph::exec::detail::ModelId newId) {
531 if (detail::parkIfInFrame(*handoff, true, newId, nullptr)) {
532 return; // Completed inline: the dispatching frame will finish this.
533 }
534 if (!weakLiveness.active()) {
535 return; // The Bridge is gone; publishing this id would be pointless.
536 }
537 auto strongBinding = weakBinding.lock();
538 if (!strongBinding) {
539 return; // The BridgeHandler (and its binding) is gone.
540 }
541 std::exception_ptr failure;
542 {
543 // contextKey/primary are plain std::strings that five
544 // other sites read under `_attachMtx`; publishing them
545 // without it would be a data race, not just a stale
546 // read.
547 std::scoped_lock const guard{_attachMtx};
548 auto pinned = weakBackend.lock();
549 if (!pinned || pinned != loadBackend()) {
550 // A switchBackend() already moved past this attach
551 // (see registerHandlerImpl's identical guard) and
552 // its own re-registration loop already handled
553 // `binding` on the *new* backend -- applying this
554 // stale reply now would overwrite that with a
555 // dangling id from a backend nothing uses any
556 // more. Unlike registerHandlerImpl's fire-and-
557 // forget re-registration, a real execute() call is
558 // synchronously waiting on `onDone` here, so the
559 // stale reply must still be reported -- silently
560 // dropping it would hang that caller forever.
561 failure = std::make_exception_ptr(std::runtime_error(
562 "attach reply arrived from a backend switchBackend() already replaced"));
563 } else {
564 try {
565 strongBinding->contextKey = primaryCopy;
566 strongBinding->primary = primaryCopy;
567 strongBinding->currentId.store(newId.v);
568 } catch (...) {
569 failure = std::current_exception();
570 }
571 }

573 onDone(failure); // Outside the lock -- see @par Locking.

575 [onDone, handoff](const std::string& message) {
576 auto failure = std::make_exception_ptr(std::runtime_error(message));
577 if (detail::parkIfInFrame(*handoff, false, {}, failure)) {
578 return;
579 }
580 onDone(failure);
581 });
582 } catch (...) {

584 // QtWebSocketBackend::attachModelAsync's wire::encode() failing
585 // before send) -- report it like any other failure instead of
586 // letting it escape execute()'s documented never-throws contract.
587 lock.unlock();
588 onDone(std::current_exception());
589 return;
590 }

592 // The backend answered on this very stack, with `_attachMtx` still

595 // under the lock we already own, then release it and report --
596 // @p onDone never runs inside the dispatch frame.

602 binding->currentId.store(parked->modelId.v);
603 } catch (...) {
604 failure = std::current_exception();
605 }

607 lock.unlock();
608 onDone(failure);
609 return;
610 }

612 return;
613 }

615 // `attachHandler` would have run, under the same lock, then report the
616 // outcome only once the lock is gone (see @par Locking above).
617 std::exception_ptr failure;

620 {.contextKey = primary, .primary = primary}, previous);
621 binding->contextKey = primary;
622 binding->primary = std::move(primary);
623 binding->currentId.store(newId.v);
624 } catch (...) {
625 failure = std::current_exception();
626 }
627 lock.unlock();
628 onDone(failure);

648 /// @brief Async counterpart to `ensureBound`. See `attachHandlerAsync`'s
649 /// doc comment for the fallback and locking contract, including the
650 /// inline-completion handling and the in-flight dedup gap, both of
651 /// which apply here identically (two result-keyed `execute()` calls
652 /// on the same still-unbound handler each bind their own anonymous
653 /// instance; the first is then stranded until the connection scope
654 /// closes).
655 ///
656 /// The one difference: this method's success callback publishes only
657 /// `currentId`, which is a `std::atomic`, so — unlike `attachHandlerAsync`'s
658 /// — it needs no `_attachMtx` of its own to do it.
659 /// @param binding Shared binding to bind.
660 /// @param onDone Invoked exactly once: `nullptr` on success, or a
661 /// non-null `exception_ptr` on failure.
662 void ensureBoundAsync(const std::shared_ptr<detail::HandlerBinding>& binding,

666 lock.unlock();
667 onDone(nullptr);
668 return;
669 }
670 auto backend = loadBackend();

672 auto const weakLiveness = _callbacks.token();
673 std::weak_ptr<detail::HandlerBinding> const weakBinding{binding};
674 auto handoff = std::make_shared<detail::AsyncDispatchHandoff>();

676 try {

678 binding->typeId, binding->modelFactory, {.contextKey = binding->contextKey, .primary = {}},
679 [this, weakBackend, weakLiveness, weakBinding, onDone, handoff](::morph::exec::detail::ModelId newId) {
680 if (detail::parkIfInFrame(*handoff, true, newId, nullptr)) {
681 return; // Completed inline: the dispatching frame will finish this.
682 }

684 return; // The Bridge is gone; publishing this id would be pointless.
685 }
686 auto strongBinding = weakBinding.lock();
687 if (!strongBinding) {
688 return; // The BridgeHandler (and its binding) is gone.
689 }
690 std::exception_ptr failure;
691 {

693 // check against a concurrent switchBackend() (which
694 // takes the same lock) -- `currentId` itself is an
695 // atomic and needs no lock to store.
696 std::scoped_lock const guard{_attachMtx};
697 auto pinned = weakBackend.lock();
698 if (!pinned || pinned != loadBackend()) {
699 // See attachHandlerAsync's identical guard: a
700 // stale reply from a backend switchBackend()
701 // already replaced must still resolve `onDone`
702 // (a real execute() call is waiting), not be
703 // silently dropped.
704 failure = std::make_exception_ptr(std::runtime_error(
705 "attach reply arrived from a backend switchBackend() already replaced"));
706 } else {
707 strongBinding->currentId.store(newId.v);
708 }
709 }
710 onDone(failure);
711 },
712 [onDone, handoff](const std::string& message) {
713 auto failure = std::make_exception_ptr(std::runtime_error(message));
714 if (detail::parkIfInFrame(*handoff, false, {}, failure)) {
715 return;
716 }
717 onDone(failure);
718 });
719 } catch (...) {
720 // See attachHandlerAsync's identical guard: the backend's own
721 // dispatch call can throw synchronously before send.
722 lock.unlock();
723 onDone(std::current_exception());
724 return;
725 }
726 if (auto parked = detail::claimHandoff(*handoff)) {
727 // Completed on this stack, under `_attachMtx`: publish here, then
728 // release the lock before reporting (see `attachHandlerAsync`).
729 if (parked->succeeded) {
730 binding->currentId.store(parked->modelId.v);
731 }
732 lock.unlock();

734 return;
735 }
736 if (started) {
737 return;
738 }
739 // No async path on this backend: run the identical synchronous
740 // registration `ensureBound` would have run, under the same lock, then
741 // report the outcome only once the lock is gone (see
742 // `attachHandlerAsync`'s "@par Locking").
743 std::exception_ptr failure;
744 try {
745 auto newId = backend->registerModelShared(binding->typeId, binding->modelFactory,
746 {.contextKey = binding->contextKey, .primary = {}});

752 onDone(failure);

755 /// @brief Files @p binding's current instance under @p primary, in place.
756 ///
757 /// The instance keeps everything the creating action just did — nothing is
758 /// re-created and nothing is stranded. A no-op if @p binding already holds
759 /// a different real primary (the locally cached primary must not race
760 /// ahead of the backend's own refusal to re-key an already-keyed
761 /// instance — see `IBackend::assignPrimary`).
762 ///
763 /// Known gap: if @p binding was still anonymous but the *target* key is
764 /// already held by a different instance, the backend's `assignPrimary`
765 /// silently declines to promote (the existing holder always wins), but
766 /// this method has no way to learn that and still caches @p primary as
767 /// though the promotion succeeded — `binding->primary()` can then report
768 /// a key the backend never actually filed this instance under, and a
769 /// same-key `attach()` after that becomes a silent no-op (the "already
770 /// primary == primary" guard in `attachHandler`), so the binding can
771 /// never reach the instance actually holding that key. Closing this
772 /// requires `assignPrimary`'s outcome to become observable (e.g. a
773 /// `bool` return threaded across every `IBackend` implementation and, for
774 /// wire backends, a reply field) — tracked as a follow-up, not fixed
775 /// here.
776 /// Prefers the backend's `IBackend::assignPrimaryAsync` when it offers
777 /// one — the same "avoid a nested-event-loop block that aborts a WASM
778 /// main thread" rationale `Bridge::registerHandler()` follows for the
779 /// initial bind step (`backend.md`, "Asynchronous registration") applies
780 /// identically here: this method is invoked from inside the result
781 /// `Completion`'s callback chain (`BridgeHandler::execute`'s `onResult`),
782 /// not from the original call stack, so there is no caller left blocked
783 /// waiting on it either way — the async path simply avoids parking the
784 /// Qt event loop for the round trip. `binding->contextKey`/`primary` are
785 /// only published once the (possibly async) reply confirms the call, so
786 /// a caller reading `binding->primary()` never sees a promotion that the
787 /// backend has not actually completed. Falls back to the synchronous
788 /// `assignPrimary` when the backend offers no async path, publishing
789 /// immediately exactly as before this method existed.
790 ///
791 /// `_attachMtx` is released *before* calling `assignPrimaryAsync` — not
792 /// held across it — mirroring `registerHandlerImpl`'s discipline for
793 /// `registerModelAsync`: the success/error callback below re-acquires
794 /// `_attachMtx`, so a backend that ever invoked it synchronously (none
795 /// documented here do, but nothing prevents one from doing so) would
796 /// otherwise self-deadlock re-acquiring a mutex this same call stack
797 /// still held.
798 /// @tparam Model Concrete model type.
799 /// @param binding Shared binding whose instance is being promoted.
800 /// @param primary Canonical string encoding of the key to file it under.
801 template <typename Model>
802 void assignHandlerPrimary(const std::shared_ptr<detail::HandlerBinding>& binding, std::string primary) {
803 std::shared_ptr<::morph::backend::detail::IBackend> backend;
804 uint64_t raw = 0;
805 {
806 std::scoped_lock const lock{_attachMtx};
807 raw = binding->currentId.load();
808 if (raw == 0U || primary.empty() || !binding->primary.empty()) {
809 return;
810 }
811 backend = loadBackend();
812 }
813 auto const weakLiveness = _callbacks.token();
814 std::weak_ptr<::morph::backend::detail::IBackend> const weakBackend{backend};
815 std::weak_ptr<detail::HandlerBinding> const weakBinding{binding};
816 bool const started = backend->assignPrimaryAsync(
817 ::morph::exec::detail::ModelId{raw}, binding->typeId, primary,
818 [this, weakLiveness, weakBackend, weakBinding, primary](::morph::exec::detail::ModelId) {
819 if (!weakLiveness.active()) {
820 return; // The Bridge is gone; do not touch `this`.
821 }
822 auto strongBinding = weakBinding.lock();
823 if (!strongBinding) {
824 return; // The BridgeHandler (and its binding) is gone.
825 }
826 std::scoped_lock const attachLock{_attachMtx};
827 auto pinned = weakBackend.lock();
828 if (!pinned || pinned != loadBackend()) {
829 // A switchBackend() already moved past this promotion;
830 // applying a stale reply now would overwrite whatever
831 // state the new backend's re-registration already
832 // established -- same reasoning as registerHandlerImpl's
833 // async callback.
834 return;
835 }
836 // A binding whose primary is already set (by a concurrent
837 // attach/assign that raced ahead of this async reply, or
838 // simply already promoted) must not be overwritten here.
839 if (!strongBinding->primary.empty()) {
840 return;
841 }
842 strongBinding->contextKey = primary;
843 strongBinding->primary = primary;
844 },
845 [typeId = binding->typeId](const std::string& message) {
846 ::morph::log::logError("[assignHandlerPrimary] async promotion of '" + typeId +
847 "' failed: " + message);
848 });
849 if (!started) {
850 backend->assignPrimary(::morph::exec::detail::ModelId{raw}, binding->typeId, primary);
851 std::scoped_lock const lock{_attachMtx};
852 // Re-check under the lock: a concurrent attach/assign could have
853 // raced ahead while the (now-established-synchronous) backend
854 // call above ran without _attachMtx held.
855 if (binding->primary.empty()) {
856 binding->contextKey = primary;
857 binding->primary = std::move(primary);
858 }
859 }
860 }

882 /// @brief Resolves once @p binding's initial registration settles.
883 ///
884 /// Closes the gap `executeVia`'s fast-fail leaves for a caller that
885 /// constructs a `BridgeHandler` through the async registration path (see
886 /// `backend.md`, "Asynchronous registration") and wants to dispatch the
887 /// moment registration completes, rather than failing fast with "handler
888 /// not bound" or polling `isBound()` in a loop of its own devising.
889 ///
890 /// - If @p binding is already bound, resolves immediately with `true`.
891 /// - If an async registration is still in flight, resolves with `true`
892 /// once `onRegistered` fires, or with the registration's error via
893 /// `.onError(...)` if it fails.
894 /// - If @p binding was never handed an async registration at all (the
895 /// synchronous fallback path, or a binding still awaiting its very
896 /// first `registerHandlerImpl` call to even attempt one) and is not yet
897 /// bound, resolves immediately with `false` — there is nothing in
898 /// flight to wait for. This can only happen for a `shared` binding that
899 /// has not yet been given a primary (`ensureBound`/`attachHandler`
900 /// never ran) or in the narrow window before `registerHandlerImpl`
901 /// itself has run; ordinary (non-shared) bindings are always either
902 /// bound or mid-registration by the time a caller can observe them.
903 /// @param binding Binding to wait on.
904 /// @param cbExec Executor the resolution is delivered on, exactly like
905 /// every other `Completion`-returning call in this class.
906 /// @return `Completion<bool>` resolving `true` once bound, `false` if
907 /// nothing is in flight, or an error if registration failed.
908 [[nodiscard]] ::morph::async::Completion<bool> whenBound(const std::shared_ptr<detail::HandlerBinding>& binding,
909 ::morph::exec::IExecutor* cbExec) {
910 auto state = std::make_shared<::morph::async::detail::CompletionState<bool>>();
911 ::morph::async::Completion<bool> comp{state, cbExec};
912 if (isBound(binding)) {
913 state->setValue(true);
914 return comp;
915 }
916 std::scoped_lock const lock{binding->registrationMtx};
917 // Re-check under the lock: registerHandlerImpl's callback may have
918 // resolved (and bound) between the lock-free check above and here.
919 if (isBound(binding)) {
920 state->setValue(true);
921 return comp;
922 }
923 if (!binding->registrationInFlight) {
924 // Nothing to wait for: no async registration was ever started for
925 // this binding (synchronous fallback already ran to completion,
926 // or a shared binding with no primary yet), and it is still
927 // unbound. Resolve false rather than hang forever.
928 state->setValue(false);
929 return comp;
930 }
931 binding->registrationWaiters.emplace_back([state](bool ok) { state->setValue(ok); },
932 [state](std::exception_ptr err) { state->setException(err); });
933 return comp;
934 }

946 /// The subscription is stored against the *binding*, not against a fixed
947 /// instance id, and is matched at publish time by comparing the binding's
948 /// current instance. Re-pointing a handler therefore moves its subscriptions
949 /// with it, which is what makes "tell me about the account I am looking at"
950 /// keep working when the user switches accounts.
951 ///
952 /// @param binding Handler binding that owns the subscription.
953 /// @param type Result type being subscribed to.
954 /// @param sink Type-erased delivery callback; receives the boxed result.

963 entry.exec = exec;

965 }
966 }
967 _subscriptions.push_back({.binding = binding, .type = type, .sink = std::move(sink), .exec = exec});

997 /// @brief Delivers @p value to every subscriber attached to instance @p mid.
998 ///
999 /// Called for every successful action result. Subscribers are matched on
1000 /// *the instance the result was produced on*, so a handler hears about work
1001 /// another handler — or, with a shared instance, another screen entirely —
1002 /// did on the model it is attached to.
1003 ///
1004 /// The producing handler is notified too: suppressing the echo would force
1005 /// every subscriber to special-case "was this mine", which is exactly the
1006 /// bookkeeping the feature exists to remove.
1007 ///
1008 /// Sinks are snapshotted under the lock and invoked outside it, so a
1009 /// subscriber that re-enters the bridge cannot deadlock.
1010 ///
1011 /// @param mid Instance the result was produced on.
1012 /// @param type Result type produced.
1013 /// @param value Boxed result.
1014 void publishResult(::morph::exec::detail::ModelId mid, std::type_index type, const std::any& value) {
1015 std::vector<std::pair<std::function<void(const std::any&)>, ::morph::exec::IExecutor*>> targets;
1016 {
1017 std::scoped_lock const lock{_subMtx};
1018 // Prune while we are already holding the lock and walking the list:
1019 // a handler that is destroyed without unsubscribing would otherwise
1020 // leave its entry behind until some *other* handler happened to call
1021 // add/removeSubscription, which in a long-lived app with many
1022 // transient handlers is never.
1023 std::erase_if(_subscriptions, [](const InstanceSubscription& entry) { return entry.binding.expired(); });
1024 _subscriptionCount.store(_subscriptions.size(), std::memory_order_relaxed);
1025 for (const auto& entry : _subscriptions) {
1026 auto owner = entry.binding.lock();
1027 if (owner && entry.type == type && owner->currentId.load() == mid.v && entry.sink) {
1028 targets.emplace_back(entry.sink, entry.exec);
1029 }
1030 }
1031 }
1032 for (auto& [sink, exec] : targets) {
1033 if (exec != nullptr) {
1034 exec->post([sink, value] { sink(value); });
1035 } else {
1036 sink(value);
1037 }
1038 }

1172 /// `LocalBackend::_regMtx`. `notifyBackendChanged()` runs under `_mtx`
1173 /// but only **posts** each model's `onBackendChanged()` onto that
1174 /// model's strand (see `LocalBackend::notifyBackendChanged`); the model
1175 /// body runs later on a pool thread, off `_mtx`. So an
1176 /// `onBackendChanged()` implementation **may** re-enter the bridge

1178 /// `_mtx` freshly on the strand thread rather than deadlocking on the

1180 /// `execute` on the same model, so it needs no locking of model state.

1183 /// `unique_ptr<IBackend>` directly) so it is an exact match for a

1186 /// resolution — both would otherwise require an equally-ranked user-defined
1187 /// conversion (unique_ptr<Concrete> -> unique_ptr<IBackend> vs. unique_ptr<Concrete>
1188 /// -> shared_ptr<IBackend>), making every existing call site ambiguous.
1189 /// @tparam Backend Concrete backend type; must derive from `IBackend`.
1190 /// @param newBackend Replacement backend. Ownership is transferred.

1197 /// @brief Atomically replaces the active backend with @p newBackend.
1198 ///
1199 /// Identical to the `unique_ptr` overload, except the caller keeps shared
1200 /// ownership of @p newBackend — the backend can be re-installed later
1201 /// (e.g. switching back to a long-lived remote backend after a temporary
1202 /// fallback to a local one) without reconstructing it.
1203 ///
1204 /// @param newBackend Replacement backend, shared with the caller.
1205 void switchBackend(std::shared_ptr<::morph::backend::detail::IBackend> newBackend) {
1206 auto newShared = std::move(newBackend);
1207 // Stamp the current default session onto the new backend before phase 1
1208 // below builds a single `register`/`registerShared` envelope per live
1209 // binding — otherwise every control envelope re-registering handlers on
1210 // the new backend would carry a default-constructed (unauthenticated)
1211 // session, exactly the #63 gap this hook closes. Read under
1212 // `_sessionMtx` alone (a leaf mutex never held while calling into this
1213 // backend), mirroring `executeVia`'s copy-then-release pattern.
1214 {
1215 std::scoped_lock const lock{_sessionMtx};
1216 newShared->setSession(_defaultSession);
1217 }
1218 std::shared_ptr<::morph::backend::detail::IBackend> previous;
1219 {
1220 // Both mutexes: this phase reads/writes every live binding's
1221 // `primary`/`contextKey` (via `_attachMtx`'s ownership of those
1222 // fields) as well as `_handlers` itself (via `_mtx`), and must not
1223 // race a concurrent attachHandler()/ensureBound()/
1224 // assignHandlerPrimary() call on any one of them.
1225 std::scoped_lock const lock{_mtx, _attachMtx};
1226
1227 // Phase 1 — register every live binding on the new backend WITHOUT
1228 // mutating any `currentId` yet, staging (binding, newId) pairs. If a
1229 // registration throws partway (a plausible remote/transport failure),
1230 // roll back the ones already registered and rethrow, leaving the old
1231 // backend and every `currentId` untouched — so the switch is atomic:
1232 // it either fully succeeds or is a no-op.
1233 std::vector<std::weak_ptr<detail::HandlerBinding>> live;
1234 std::vector<std::pair<std::shared_ptr<detail::HandlerBinding>, uint64_t>> staged;
1235 try {
1236 for (auto& weak : _handlers) {
1237 auto binding = weak.lock();
1238 if (!binding) {
1239 continue;
1240 }
1241 // A shared binding that never attached has no instance to
1242 // re-create: it stays live and unbound, and acquires one on
1243 // the new backend the first time it is attached.
1244 if (binding->shared && binding->primary.empty()) {
1245 live.push_back(weak);
1246 continue;
1247 }
1248 auto newId = binding->shared
1249 ? newShared->registerModelShared(
1250 binding->typeId, binding->modelFactory,
1251 {.contextKey = binding->contextKey, .primary = binding->primary})
1252 : newShared->registerModelWithContext(binding->typeId, binding->modelFactory,
1253 binding->contextKey);
1254 staged.emplace_back(binding, newId.v);
1255 live.push_back(weak);
1256 }
1257 } catch (...) {
1258 for (const auto& [binding, newId] : staged) {
1259 try {
1260 newShared->deregisterModel(::morph::exec::detail::ModelId{newId});
1261 } catch (const std::exception& exc) {
1262 ::morph::log::logError(std::string{"[switchBackend] rollback deregister failed: "} +
1263 exc.what());
1264 }
1265 }
1266 throw;
1267 }
1268
1269 // Phase 2 — commit. Every registration succeeded, so it is now safe to
1270 // publish the new ids and swap the backend in.
1271 for (const auto& [binding, newId] : staged) {
1272 binding->currentId.store(newId);
1273 }
1274 _handlers = std::move(live);
1275
1276 previous = exchangeBackend(newShared);
1277 newShared->notifyBackendChanged();
1278 installReconnectHandler(newShared);
1279 }
1280 if (previous && previous != newShared) {
1281 previous->setReconnectHandler(nullptr);
1282 }
1283 // Drain the outgoing backend outside the bridge mutex: cancelPending
1284 // delivers callbacks through the caller's gui executor, and we never want
1285 // to hold _mtx while user code runs.
1286 if (previous && previous != newShared) {
1287 previous->cancelPending(std::make_exception_ptr(::morph::backend::BackendChangedError{}));
1288 }
1289 }

1291 /// @brief Deregisters @p binding from the active backend and removes it from tracking.
1292 ///
1293 /// After this call, any future `executeVia()` using this binding will
1294 /// complete with an error.
1295 /// @param binding Binding to remove. Must have been returned by `registerHandler()`.
1296 void deregisterHandler(const std::shared_ptr<detail::HandlerBinding>& binding) {
1297 std::scoped_lock const lock{_mtx};
1298 uint64_t const raw = binding->currentId.load();
1299 if (raw != 0U) {
1300 loadBackend()->deregisterModel(::morph::exec::detail::ModelId{raw});
1301 }

1303 // this binding fails fast on the documented guard rather than sending a
1304 // now-destroyed ModelId to the backend.
1305 binding->currentId.store(0);

1311 _handlers.erase(iter);
1312 }

1553 // delivered to the caller's own Completion either way -- only
1554 // the bridge-touching side effects are skipped.

1561 // itself throws and routes to setException.

1566 // Runs before the value is moved out and before the caller's
1567 // own .then, so a result-sourced primary key is adopted by
1568 // the binding before any user code observes the result.
1569 if (onResult && bridgeAlive) {
1570 onResult(*typedResult);
1571 }

1573 // instance before the value is moved away. Guarded on the
1574 // bridge's liveness token: a completion can in principle
1575 // resolve after the Bridge is gone.
1576 if constexpr (std::is_copy_constructible_v<R>) {
1577 if (bridgeAlive && hasSubscribers()) {
1578 publishResult(::morph::exec::detail::ModelId{raw}, std::type_index{typeid(R)},
1579 std::any{*typedResult});
1580 }
1581 }
1582 typedState->setValue(std::move(*typedResult));
1583 } catch (...) {
1584 typedState->setException(std::current_exception());
1585 }
1586 })
1587 .onError(

1589 // Same disarm-first reasoning (and the same schedulerRef-based
1590 // safety, not a liveness-then-use race) as the success branch
1591 // above: a real error reply settles the completion, so the
1592 // deadline must not also fire.
1593 if (deadlineHandle && schedulerRef) {
1594 try {
1595 schedulerRef->cancel(*deadlineHandle);
1596 } catch (...) {
1597 // Best-effort: a failed cancel leaves the deadline's own
1598 // entry to fire later and find nothing (setValue/
1599 // setException below are idempotent), which is exactly
1600 // what an uncancelled entry already does.
1601 }
1602 }
1603 // The other of the two mutually-exclusive resolution paths --
1604 // see the .then continuation above. Same liveness guard: this
1605 // touches `this` and must not run once the Bridge might be gone.
1606 if (alive.active()) {
1607 this->_pendingCalls.fetch_sub(1, std::memory_order_relaxed);
1608 }
1609 typedState->setException(err);
1610 });
1611 return typed;

1938 /// @brief Dispatches @p action via the underlying `Bridge` and returns a `Completion`.
1939 ///
1940 /// The bridge's currently-installed default session (set once at startup or
1941 /// after login via `Bridge::setDefaultSession`) is attached automatically —
1942 /// callers never thread the session through individual call sites.
1943 ///
1944 /// @tparam Action Concrete action type registered with `BRIDGE_REGISTER_ACTION`.
1945 /// @param action Action to execute (moved into the dispatch).
1946 /// @return Completion that resolves on the GUI executor. A payload- or
1947 /// result-keyed action's attach/promote step never throws out of
1948 /// this call, even when the backend refuses it (e.g. a remote
1949 /// server at `LimitPolicy::maxLiveModels`, a transport error, or
1950 /// an unauthorized attach) — the failure is instead delivered
1951 /// through the returned Completion's `.onError(...)`, exactly
1952 /// like any other dispatch failure.
1953 template <typename Action>
1954 ::morph::async::Completion<typename ::morph::model::ActionTraits<Action>::Result> execute(Action action) {
1955 using R = ::morph::model::ActionTraits<Action>::Result;
1956 // Three mutually exclusive routes, chained rather than sequential: an
1957 // action is payload-keyed or result-keyed or neither (`PayloadKeyed`
1958 // and `ResultKeyed` differ only in `fromResult`, so no action can
1959 // satisfy both), and an unkeyed action — or any action at all on a
1960 // `NoSharing` handler — always lands in the final `else`. Chaining the
1961 // `if constexpr`s (rather than leaving the payload-keyed branch to
1962 // fall through to that `else`, as it did while the attach step was
1963 // synchronous) is what lets the keyed branches own their dispatch: the
1964 // attach now completes asynchronously, so the dispatch it precedes has
1965 // to happen from inside its completion callback, not on this stack.
1966 if constexpr (kShared && ::morph::model::detail::PayloadKeyed<Action>) {
1967 // The action names its instance: attach (or re-point) before
1968 // dispatching, so the call lands on the instance it asked for. A
1969 // remote backend's refusal (LimitPolicy::maxLiveModels, a
1970 // transport error, unauthorized) must surface through the
1971 // returned Completion's onError, exactly like every other
1972 // dispatch failure — not as a synchronous throw out of execute().
1973 //
1974 // The attach goes through Bridge::attachHandlerAsync, which uses
1975 // the backend's `attachModelAsync` when it has one and otherwise
1976 // runs the identical synchronous attach inline and calls back
1977 // before returning — so a backend that has not opted in behaves
1978 // exactly as it did before this path existed.
1979 auto state = std::make_shared<::morph::async::detail::CompletionState<R>>();
1980 ::morph::async::Completion<R> pending{state, _guiExec};
1981 auto* const bridgePtr = &_bridge;
1982 auto binding = _binding;
1983 // Key extraction is user code (ActionKeyTraits + keyToString), so it
1984 // is inside the same no-throw-out-of-execute() promise the attach
1985 // itself makes: a throw here resolves the Completion, it does not
1986 // escape.
1987 std::string key;
1988 try {
1989 key = ::morph::model::ActionKeyTraits<Action>::key(action);
1990 } catch (...) {
1991 state->setException(std::current_exception());
1992 return pending;
1993 }
1994 auto sharedAction = std::make_shared<Action>(std::move(action));
1995 bridgePtr->template attachHandlerAsync<Model>(
1996 binding, std::move(key),
1997 [bridgePtr, binding, sharedAction, state, guiExec = _guiExec](std::exception_ptr err) {
1998 if (err) {
1999 state->setException(err);
2000 return;
2001 }
2002 bridgePtr->template executeVia<Model, Action>(binding, std::move(*sharedAction), guiExec)
2003 .then([state](R value) { state->setValue(std::move(value)); })
2004 .onError([state](std::exception_ptr exc) { state->setException(exc); });
2005 });
2006 return pending;
2007 } else if constexpr (kShared && ::morph::model::detail::ResultKeyed<Action>) {
2008 // The action *creates* the instance and its result carries the
2009 // generated key, exactly as a database insert returns its primary
2010 // key. Adopt it before any user callback observes the result, so a
2011 // .then() can immediately run further actions on the new instance.
2012 //
2013 // The action generates its own key, so it must run before the key
2014 // exists. Give the handler an anonymous instance to run on, then
2015 // promote *that* instance once the reply names it — re-pointing to a
2016 // fresh one instead would strand whatever the create just did.
2017 // Same async/fallback contract as the payload-keyed branch above,
2018 // via Bridge::ensureBoundAsync.
2019 auto state = std::make_shared<::morph::async::detail::CompletionState<R>>();
2020 ::morph::async::Completion<R> pending{state, _guiExec};
2021 auto* const bridgePtr = &_bridge;
2022 auto binding = _binding;
2023 auto sharedAction = std::make_shared<Action>(std::move(action));
2024 bridgePtr->ensureBoundAsync(
2025 binding, [bridgePtr, binding, sharedAction, state, guiExec = _guiExec](std::exception_ptr err) {
2026 if (err) {
2027 state->setException(err);
2028 return;
2029 }
2030 bridgePtr
2031 ->template executeVia<Model, Action>(
2032 binding, std::move(*sharedAction), guiExec,
2033 [bridgePtr, binding](const R& result) {
2034 bridgePtr->template assignHandlerPrimary<Model>(
2035 binding, ::morph::model::ActionKeyTraits<Action>::template keyOfResult<R>(result));
2036 })
2037 .then([state](R value) { state->setValue(std::move(value)); })
2038 .onError([state](std::exception_ptr exc) { state->setException(exc); });
2039 });
2040 return pending;
2041 } else {
2042 return _bridge.template executeVia<Model, Action>(_binding, std::move(action), _guiExec);
2043 }
2044 }

2261/// Builds one executor per `Sharing` policy the framework defines
2262/// (`NoSharing`, `AllowShared`) from the same generic-lambda template,
2263/// `static_cast`ing `handlerVoid` to the matching `BridgeHandler<Model,
2264/// Sharing>*` in each — see issue #68 / bridge.md's design-decision entry for
2265/// why a single `NoSharing`-only executor is unsound for a shared handler.
2266template <typename Model, typename Action>
2267inline void ActionExecuteRegistry::registerAction(std::string_view modelId, std::string_view actionId) {
2268 auto makeExecutor = []<typename Sharing>() {
2269 return [](void* handlerVoid, std::string_view bodyJson) -> ::morph::async::Completion<std::string> {
2270 auto* handler = static_cast<BridgeHandler<Model, Sharing>*>(handlerVoid);
2271 auto resultState = std::make_shared<::morph::async::detail::CompletionState<std::string>>();
2272 try {
2273 Action action = ::morph::model::ActionTraits<Action>::fromJson(bodyJson);
2274 // Retag any Quantity fields to their declared precision so the stored
2275 // value matches the schema's advertised `x-decimalPlaces`, rather than
2276 // silently keeping whatever runtime `dp` the client sent. No-op for
2277 // actions with no Quantity members. See docs/spec/forms/forms.md.
2278 ::morph::forms::reconcileDeclaredPrecision(action);
2279 // Pre-decode wire validation seam: reject a Quantity field whose
2280 // engaged value falls outside its unit's declared bounds
2281 // (UnitTraits<E>::bounds), before the validator/business-rule
2282 // check below. No-op for actions with no Quantity members, or
2283 // whose units declare no bounds(). See docs/spec/forms/forms.md,
2284 // "Pre-decode wire validation". Thrown as QuantityDecodeError,
2285 // caught by the same catch block as every other decode/validation
2286 // failure on this path.
2287 ::morph::forms::enforceQuantityBounds(action);
2288 // Overwrite any computed fields from their declared inputs -- a
2289 // computed field is never trusted from the client, on any path.
2290 // No-op for actions with no computedFields. See docs/spec/forms/forms.md.
2291 ::morph::forms::recomputeAll(action);
2292 // Enforce the action's validator on the request/reply dispatch path,
2293 // just as the reactive `set<>` path does via `tryFireImpl`. Without
2294 // this, a submitted action that fails its readiness/validity check
2295 // (empty required Quantity, out-of-range field, …) would reach the
2296 // handler and either produce a silently wrong result or force every
2297 // handler to re-check by hand. `ActionValidator<Action>::ready`
2298 // auto-detects a `bool validate() const` member and defaults to
2299 // `true` for actions with no validator, so this is a no-op for
2300 // unvalidated actions and a hard gate for validated ones. An invalid
2301 // action resolves the completion through `setException` (a proper
2302 // error reply upstream), never a bad execution.
2303 if (!::morph::model::ActionValidator<Action>::ready(action)) {
2304 throw std::invalid_argument{"action failed validation: " +
2305 std::string{::morph::model::ActionTraits<Action>::typeId()}};
2306 }
2307 handler
2308 ->template execute<Action>(std::move(action))
2309 // NOLINTNEXTLINE(performance-unnecessary-value-param) — lambda captures the result by value
2310 .then([resultState](auto result) {
2311 // Guard the JSON forwarding for the same reason as executeVia:
2312 // a throwing resultToJson (or the move it does) must land on
2313 // the error sink, not escape the callback executor and either
2314 // hang the completion or terminate the Qt loop.
2315 try {
2316 resultState->setValue(::morph::model::ActionTraits<Action>::resultToJson(result));
2317 } catch (...) {
2318 resultState->setException(std::current_exception());
2319 }
2320 })
2321 .onError([resultState](const std::exception_ptr& err) { resultState->setException(err); });
2322 } catch (...) {
2323 resultState->setException(std::current_exception());
2324 }
2325 return {resultState, handler->guiExecutor()};
2326 };
