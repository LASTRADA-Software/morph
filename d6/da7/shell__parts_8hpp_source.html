180/// @brief The parts of a shell an application declares.
181struct ShellParts {

183 std::optional<ContextSwitch> context;
184 /// @brief A count per module.
185 std::vector<ModuleCount> modules;
186 /// @brief The modules the global search reads.
187 std::vector<SearchSource> search;

189 std::optional<InboxEntry> inbox;
190 /// @brief The signed-in person's menu.
191 std::optional<PrincipalMenu> principal;
192 /// @brief The search field's placeholder: a catalogue key or text.
193 std::string searchHint;
194 /// @brief The product mark and version.
195 std::optional<BrandMark> brand;
196 /// @brief The help button.
197 std::optional<HelpEntry> help;
198
199 /// @brief Memberwise equality.
200 /// @return Whether every member matches.
201 bool operator==(ShellParts const&) const = default;
202
203 /// @brief Whether any part is declared.
204 /// @return `false` for a default `ShellParts`.
205 [[nodiscard]] bool any() const noexcept {
206 return context || !modules.empty() || !search.empty() || inbox || principal || brand || help;
207 }

234/// @brief Whether a name is one the shell's parts declare: the search text, the switcher and the person's queries, and
235/// a model alias, count, search or inbox query followed by its number.
236/// @param name A name declared in the app document's scope.
237/// @return True when the lowering of the parts would declare it.
238[[nodiscard]] inline bool reserved(std::string_view name) {
239 if (name == kSearchText || name == kContext || name == kPrincipal || name == kHelp) {
240 return true;
241 }
242 for (std::string_view const prefix : {"shellModel", "shellCount", "shellSearch", "shellInbox"}) {
243 if (name.starts_with(prefix)) {
244 std::string_view const digits = name.substr(prefix.size());
245 if (!digits.empty() && std::ranges::all_of(digits, [](char c) { return c >= '0' && c <= '9'; })) {
246 return true;
247 }
248 }
249 }
250 return false;
251}

270/// @brief A request body expression: an object of the declared request fields.
271/// @param members The `(member, expression)` pairs; a pair with an empty member is left out.
272/// @return `{"obj": {...}}`, or an empty string when no member remains.
273[[nodiscard]] inline std::string bodyOf(std::vector<std::pair<std::string, std::string>> const& members) {
274 std::string out;
275 for (auto const& [name, expression] : members) {
276 if (name.empty()) {
277 continue;
278 }
279 out += (out.empty() ? "" : ",") + jsonString(name) + ":" + expression;
280 }
281 return out.empty() ? out : R"({"obj":{)" + out + "}}";
282}

286/// @brief Lowers the parts to the declarations of an app document's scope: one shared model alias per distinct model,
287/// one query per read, and the search text as state.
288/// @param parts The parts.
289/// @return A JSON object with `models`, `state` and `queries`, in the document's own shape.
290[[nodiscard]] inline std::string lowerShellParts(ShellParts const& parts) {
291 std::map<std::string, std::string, std::less<>> aliases;
292 std::string models;
293 auto alias = [&](std::string const& model) {
294 auto const found = aliases.find(model);
295 if (found != aliases.end()) {
296 return found->second;
297 }
298 std::string const name = "shellModel" + std::to_string(aliases.size());
299 aliases.emplace(model, name);
300 models += (models.empty() ? "" : ",") + detail::jsonString(name) + ":{\"type\":" + detail::jsonString(model) +
301 ",\"instance\":\"shared\"}";
302 return name;
303 };
304 std::string queries;
305 auto add = [&](std::string const& name, std::string const& model, std::string const& action,
306 std::string const& body, std::string const& extra) {
307 std::string const owner = alias(model);
308 queries += (queries.empty() ? "" : ",") + detail::jsonString(name) +
309 ":{\"model\":" + detail::jsonString(owner) + ",\"action\":" + detail::jsonString(action) +
310 (body.empty() ? "" : ",\"body\":" + body) + extra + "}";
311 };
312 auto env = [](std::string_view name) { return std::string{R"({"env":")"} + std::string{name} + "\"}"; };
313 std::string const engagedContext = R"({"fn":"engaged","args":[{"env":"context"}]})";
314 auto read = [&](std::string const& name, std::string const& model, std::string const& action,
315 std::string const& contextField, std::string const& localeField) {
316 // A read that sends the scope waits until there is one, so nothing is asked unscoped.
317 std::string const when =
318 parts.context && !contextField.empty() ? ",\"when\":" + engagedContext : std::string{};
319 add(name, model, action, detail::bodyOf({{contextField, env("context")}, {localeField, env("locale")}}), when);
320 };
321 if (parts.context) {
322 add(std::string{shell_names::kContext}, parts.context->model, parts.context->action,
323 detail::bodyOf({{parts.context->localeField, env("locale")}}), std::string{});
324 }
325 for (std::size_t index = 0; index < parts.modules.size(); ++index) {
326 ModuleCount const& count = parts.modules.at(index);
327 read(shell_names::count(index), count.model, count.action, count.contextField, count.localeField);
328 }
329 for (std::size_t index = 0; index < parts.search.size(); ++index) {
330 SearchSource const& source = parts.search.at(index);
331 std::string const text = R"({"ref":")" + std::string{shell_names::kSearchText} + "\"}";
332 std::string const typed = R"({"fn":"ne","args":[)" + text + R"(,""]})";
333 std::string const searchWhen = parts.context && !source.contextField.empty()
334 ? R"({"fn":"and","args":[)" + typed + "," + engagedContext + "]}"
335 : typed;
336 add(shell_names::search(index), source.model, source.action,
337 detail::bodyOf({{source.queryField, text},
338 {source.contextField, env("context")},
339 {source.localeField, env("locale")}}),
340 ",\"when\":" + searchWhen + ",\"debounce\":" + std::to_string(kSearchPause));
341 }
342 if (parts.inbox) {
343 for (std::size_t index = 0; index < parts.inbox->sources.size(); ++index) {
344 ModuleCount const& count = parts.inbox->sources.at(index);
345 read(shell_names::inbox(index), count.model, count.action, count.contextField, count.localeField);
346 }
347 }
348 if (parts.principal) {
349 PrincipalMenu const& menu = *parts.principal;
350 read(std::string{shell_names::kPrincipal}, menu.model, menu.action, menu.contextField, menu.localeField);
351 }
352 // The button runs the action with its params as the request, so the mutation declares only where it goes.
353 std::string mutations;
354 if (parts.help && !parts.help->action.empty()) {
355 mutations = ",\"mutations\":{" + detail::jsonString(shell_names::kHelp) +
356 ":{\"model\":" + detail::jsonString(alias(parts.help->model)) +
357 ",\"action\":" + detail::jsonString(parts.help->action) + "}}";
358 }
359 std::string const state =
360 parts.search.empty() ? std::string{} : "\"" + std::string{shell_names::kSearchText} + "\":{\"init\":\"\"}";
361 return "{\"models\":{" + models + "},\"state\":{" + state + "},\"queries\":{" + queries + "}" + mutations + "}";
362}
