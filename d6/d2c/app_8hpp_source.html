112/// @brief Generates the `app-*` JSON document for app-shell type @p AppT.
113///
114/// Emits `app-title`, an ordered `app-menu` array (`{label, screen}` per
115/// entry), and an `app-screens` object mapping each screen's id to
116/// `{kind, ref}`. See docs/spec/forms/workflows_navigation.md for the full
117/// key vocabulary.
118/// @tparam AppT Concrete `morph::app::App<Title, Menu, Screens>` type.
119/// @return The app's JSON document. Empty string only if glaze's own JSON
120/// writer fails on the assembled DOM (schema generation never throws).
121template <typename AppT>
122[[nodiscard]] std::string appSchemaJson() {
123 glz::generic_u64 dom{};
124 dom["app-title"] = std::string{AppT::title()};
125
126 glz::generic_u64::array_t menu{};
127 ::morph::flows::detail::forEachTupleElement<typename AppT::menu>([&]<typename Entry, std::size_t I>() {
128 static_cast<void>(I);
129 glz::generic_u64 entry{};
130 entry["label"] = std::string{Entry::label()};
131 entry["screen"] = std::string{Entry::screen()};
132 menu.emplace_back(std::move(entry));
133 });
134 dom["app-menu"] = menu;
135
136 auto& screensNode = dom["app-screens"];
137 ::morph::flows::detail::forEachTupleElement<typename AppT::screens>([&]<typename S, std::size_t I>() {
138 static_cast<void>(I);
139 auto& screenNode = screensNode[std::string{S::id()}];
140 screenNode["kind"] = std::string{S::kind()};
141 screenNode["ref"] = std::string{S::ref()};
142 });
143
144 return glz::write_json(dom).value_or(std::string{});
145}

149/// @brief Specialises `morph::app::AppTraits<A>` with the string type-id @p NAME.
150///
151/// Metadata only, exactly like `BRIDGE_REGISTER_WIZARD`: an app-shell
152/// descriptor is never itself executed, only the actions/wizards its screens

154/// @param A Concrete `morph::app::App<...>` type.
