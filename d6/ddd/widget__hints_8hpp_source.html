5/// @file forms/widget_hints.hpp
6/// @brief Typed wrappers that carry rendering-control intent for a form field.
7///
8/// `morph::forms::Multiline` and `morph::forms::Ranged<Min, Max, Step>` are
9/// thin field-type wrappers in the same family as `morph::forms::Choice`
10/// (choice.hpp): each carries a rendering *control* preference in the C++
11/// type itself — `Multiline` says "edit me as a text area", `Ranged` says
12/// "edit me as a slider with these bounds" — and `morph::forms::schemaJson`
13/// surfaces that preference as the `x-widget` (and, for `Ranged`, `x-min` /
14/// `x-max` / `x-step`) schema annotation(s). See docs/spec/forms/widget_hints.md
15/// for the full design.
16///
17/// Neither wrapper changes what travels on the wire: the `glz::meta`
18/// specialisations below reflect each type's bare payload directly, exactly
19/// as `Choice` does, so `Multiline` serialises as a plain JSON string and
20/// `Ranged` as a nullable number.
21

32/// @brief A string field edited as a multi-line text area.
33///
34/// Wire form: a plain JSON string (via the `glz::meta` specialisation below),
35/// exactly like an unwrapped `std::string` member — `Multiline` changes only
36/// the *rendering intent*, never what travels. Because the payload is a bare
37/// `std::string` (not a `std::optional`), `Multiline` does **not** define
38/// `hasValue()`: like a plain `std::string` member, it is always considered
39/// engaged by `allRequiredEngaged` (see docs/spec/forms/forms.md, "Empty state").
40struct Multiline {
41 /// @brief The text content.
42 std::string value;
43
44 /// @brief Constructs the empty string.
45 constexpr Multiline() noexcept = default;
46
47 /// @brief Engages with @p text.
48 /// @param text The initial text content.
49 Multiline(std::string text) noexcept(std::is_nothrow_move_constructible_v<std::string>) : value{std::move(text)} {}
50
51 /// @brief The preferred control id for this field.
52 /// @return `"textarea"`.
53 [[nodiscard]] static constexpr std::string_view widget() noexcept { return "textarea"; }
54
55 /// @brief Equality on the text content.
56 /// @param other Multiline to compare against.
57 /// @return `true` when both hold the same text.
58 [[nodiscard]] constexpr bool operator==(const Multiline& other) const = default;
59};

61/// @brief A bounded numeric field edited as a slider.
62/// @tparam Min Inclusive lower bound of the control track.
63/// @tparam Max Inclusive upper bound of the control track (must be > `Min`).
64/// @tparam Step Track increment (default `1`; must be > 0). Must be the same
65/// arithmetic type as `Min`/`Max` — a floating-point `Ranged`
66/// must name `Step` explicitly (the default `1` is `int`).
67template <auto Min, auto Max, auto Step = 1>
68 requires(std::is_arithmetic_v<decltype(Min)> && std::same_as<decltype(Min), decltype(Max)> &&
69 std::same_as<decltype(Min), decltype(Step)>)
70struct Ranged {
71 static_assert(Min < Max, "Ranged: Min must be less than Max");
72 static_assert(Step > decltype(Step){0}, "Ranged: Step must be strictly positive");
73
74 /// @brief The payload; `std::nullopt` means "not entered".
75 std::optional<decltype(Min)> value;
76
77 /// @brief Constructs the empty state.
78 constexpr Ranged() noexcept = default;
79
80 /// @brief Engages with @p selected.
81 /// @param selected The selected value.
82 constexpr Ranged(decltype(Min) selected) noexcept : value{selected} {}
83
84 /// @brief Adopts an optional payload as-is.
85 /// @param payload Engaged or empty payload.
86 constexpr Ranged(std::optional<decltype(Min)> payload) noexcept : value{payload} {}
87
88 /// @brief Whether a value has been entered.
89 /// @return `true` if the payload is engaged.
90 [[nodiscard]] constexpr bool hasValue() const noexcept { return value.has_value(); }
91
92 /// @brief Unchecked access to the engaged value (UB when empty, exactly
93 /// like `std::optional::operator*`).
94 /// @return The engaged value.
95 // NOLINTNEXTLINE(bugprone-unchecked-optional-access)
96 [[nodiscard]] constexpr decltype(Min) operator*() const noexcept { return *value; }
97
98 /// @brief The slider's inclusive lower bound.
99 /// @return `Min`.
100 [[nodiscard]] static constexpr auto min() noexcept { return Min; }
101
102 /// @brief The slider's inclusive upper bound.
103 /// @return `Max`.
104 [[nodiscard]] static constexpr auto max() noexcept { return Max; }
105
106 /// @brief The slider's track increment.
107 /// @return `Step`.
108 [[nodiscard]] static constexpr auto step() noexcept { return Step; }
109
110 /// @brief The preferred control id for this field.
111 /// @return `"slider"`.
112 [[nodiscard]] static constexpr std::string_view widget() noexcept { return "slider"; }
113
114 /// @brief Equality on the payload; empty equals only empty.
115 /// @param other Ranged to compare against.
116 /// @return `true` when both are empty or both hold equal values.
117 [[nodiscard]] constexpr bool operator==(const Ranged& other) const = default;
118};
