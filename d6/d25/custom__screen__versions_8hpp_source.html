18/// @file
19/// @brief The versions of a custom screen: the base version a person's copy knows, what its base gained or lost since,

26/// @brief The schema JSON of an action by its id, as `forms::schemaJson<A>()` emits it; nothing for an action it does
27/// not know.
28using ActionSchemaLookup = std::function<std::optional<std::string>(std::string_view actionId)>;

30/// @brief The member of a custom screen's document that records the base version it was last saved or updated

34/// @brief What a change is about.
35enum class LayoutItemKind : std::uint8_t {
36 Element, ///< An element of the document: a node or a part with an `id`.
37 Field, ///< A field of the action a `form` binds.
38};
39
40/// @brief Where an element sits in the base.
41struct LayoutPlace {
42 /// @brief The id of the nearest element that holds it; empty when no element does and it sits in the document
43 /// itself.
44 std::string parent;
45 /// @brief The JSON Pointer, from that element (or the document), of the member that holds it: `/children`,
46 /// `/tabs`, `/view/children`, `/actions`.
47 std::string path;
48 /// @brief Its index in that member's array; absent when the member holds the element itself.
49 std::optional<std::size_t> index;
50};
51
52/// @brief Something the base has that the person's copy does not, newer than the version the copy knows.
53struct LayoutAddition {
54 /// @brief An element, or a form's field.
55 LayoutItemKind kind = LayoutItemKind::Element;
56 /// @brief What a choice names it by (`LayoutChoices`): an element's id, or `<form>/<field>`, where `<form>` is
57 /// the form's id, or its action for a form without one.
58 std::string key;
59 /// @brief The element's id, or the field's name.
60 std::string id;
61 /// @brief The version it first ships in.
62 std::string since;
63 /// @brief An element's place in the base; empty for a field.
64 LayoutPlace place;
65 /// @brief A field's form: its id, empty for a form without one.
66 std::string form;
67 /// @brief A field's form: the action it binds.
68 std::string action;
69};
70
71/// @brief Something the person's copy refers to that the base no longer has.
72struct LayoutRemoval {
73 /// @brief An element, or a form's field overlay.
74 LayoutItemKind kind = LayoutItemKind::Element;
75 /// @brief What a choice names it by, as `LayoutAddition::key`.
76 std::string key;
77 /// @brief The element's id, or the field's name.
78 std::string id;
79 /// @brief A field's form: its id, empty for a form without one.
80 std::string form;
81 /// @brief A field's form: the action it binds.
82 std::string action;
83};
84
85/// @brief What changed in a base screen since the version a person's copy of it knows.
86struct LayoutChanges {
87 /// @brief The base's version: the highest `since` among its elements and the fields of the actions its forms bind.
88 forms::FieldVersion baseVersion;
89 /// @brief The version the copy knows, its `knownVersion`; the first version when it records none.
90 forms::FieldVersion knownVersion;
91 /// @brief What the base gained, in the base's document order, forms' fields after the elements.
92 std::vector<LayoutAddition> added;
93 /// @brief What the copy refers to that the base lost: elements in the copy's document order, then forms' field
94 /// overlays.
95 std::vector<LayoutRemoval> removed;
96
97 /// @brief Whether the copy knows an older version than its base's.
98 /// @return `knownVersion < baseVersion`.
99 [[nodiscard]] bool needsUpdate() const noexcept { return knownVersion < baseVersion; }
100
101 /// @brief Whether the copy refers to something the base no longer has.
102 /// @return Whether `removed` is not empty.
103 [[nodiscard]] bool needsAttention() const noexcept { return !removed.empty(); }
104};
105
106/// @brief What the person chose for one addition.
107enum class AddedChoice : std::uint8_t {
108 TakeOver, ///< Take it over: an element at its place in the base, a field shown as the form shows it.
109 LeaveOut, ///< Leave it out: an element stays absent, a field gets an `omit` overlay.
110};
111
112/// @brief What the person chose for one removal.
113enum class RemovedChoice : std::uint8_t {
114 Drop, ///< Drop the entry from the copy.
115 Keep, ///< Keep it as it is.
116};
117
118/// @brief The person's choices over a `LayoutChanges`, by `LayoutAddition::key` and `LayoutRemoval::key`. An addition
119/// without a choice is left out and a removal without one is kept, so the copy changes only where the person
120/// said; a key that names no change is refused.
121struct LayoutChoices {
122 /// @brief The choice for each addition, by key.
123 std::map<std::string, AddedChoice, std::less<>> added;
124 /// @brief The choice for each removal, by key.
125 std::map<std::string, RemovedChoice, std::less<>> removed;
126};
127
128/// @brief The version of a base screen: the highest `since` among its elements and the `x-since` of the fields of the
129/// actions its forms bind.
130/// @param base The base screen's JSON text.
131/// @param schemas The action schemas; an action it does not know adds no fields.
132/// @return The version, or the refusal: a base that does not load, or a schema whose `x-since` is not a version.
133[[nodiscard]] MORPH_API std::expected<forms::FieldVersion, Refusal> baseVersion(std::string_view base,
134 ActionSchemaLookup const& schemas);
135
136/// @brief What changed in @p base since the version @p layout knows.
137///
138/// Added: each element of the base with a `since` above the copy's `knownVersion` whose id the copy does not have,
139/// with its place (an element inside another added element comes with it and is not listed), and each field of a
140/// base form's action with an `x-since` above it, for a form the copy has (the same id, or for a form without one the
141/// first form of the same action) and gives no overlay for that field. Removed: each element of the copy with a
142/// `since` whose id the base no longer has (an element a person adds has none), and each field overlay of a copy's
143/// form naming a field its action no longer has.
144/// @param base The base screen's JSON text.
145/// @param baseSchemas The action schemas; a form whose action it does not know is not compared.
146/// @param layout The person's copy's JSON text.
147/// @return The changes, or the refusal: a document that does not load, or a schema whose `x-since` is not a version.
148[[nodiscard]] MORPH_API std::expected<LayoutChanges, Refusal> layoutChanges(std::string_view base,
149 ActionSchemaLookup const& baseSchemas,
150 std::string_view layout);
151
152/// @brief @p document with its `knownVersion` set to @p version.
153/// @param document A document's JSON text.
154/// @param version The version to record.
155/// @return The rewritten text, or the refusal of text that is not a JSON object.
156[[nodiscard]] MORPH_API std::expected<std::string, Refusal> withKnownVersion(std::string_view document,
157 forms::FieldVersion const& version);
158
159/// @brief The person's copy updated by their choices, recording the base's version as `knownVersion`.
160///
161/// Each removal chosen `Drop` is removed from the copy; each field addition left out gets `"omit": true` in its form's
162/// `fields`; then each element addition taken over is inserted at its place: into the copy's element of the same id
163/// as its parent (the document itself for none), in the same member, at the base's index or at the end when the
164/// member is shorter. The result is loaded as a save loads it, so what comes back is a document; the server's checks
165/// still decide when it is saved.
166/// @param base The base screen's JSON text.
167/// @param baseSchemas The action schemas, as `layoutChanges` takes them.
168/// @param layout The person's copy's JSON text.
169/// @param choices The person's choices.
170/// @return The updated copy, or the refusal: one `layoutChanges` gives, a choice for a key that names no change, an
171/// element whose place the copy does not have, or a result that does not load.
172[[nodiscard]] MORPH_API std::expected<std::string, Refusal> applyLayoutChoices(std::string_view base,
173 ActionSchemaLookup const& baseSchemas,
174 std::string_view layout,
175 LayoutChoices const& choices);
