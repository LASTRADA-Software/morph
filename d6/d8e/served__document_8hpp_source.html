18/// @brief Why a server's catalog entry may not be mounted as a screen, when it may not.
19///
20/// The chat panel's id carries host values and commands no other document may reach, so the panel is the client's
21/// own document and never a server's.
22/// @param screenId The catalog entry's id.
23/// @return The refusal, naming @p screenId, when it is the chat panel's id; otherwise nothing.
24[[nodiscard]] inline std::optional<Refusal> servedScreenRefusal(std::string_view screenId) {
25 if (screenId == chat::kChatPanelId) {
26 return Refusal{.document = std::string{screenId},
27 .path = {},
28 .reason = "the catalog lists the id `" + std::string{chat::kChatPanelId} +
29 "`, which is the chat panel's; the entry is refused"};
30 }
31 return std::nullopt;
32}

34/// @brief Why a loaded server document may not be mounted under its catalog entry, when it may not.
35///
36/// Host names are scoped by the id a mounted document carries, so a document that names another id than its entry
37/// could take that id's scope.
38/// @param screenId The catalog entry's id.
39/// @param documentId The id the loaded document names.
40/// @return The refusal, naming @p screenId, when the two ids differ; otherwise nothing.
41[[nodiscard]] inline std::optional<Refusal> servedDocumentRefusal(std::string_view screenId,
42 std::string_view documentId) {
43 if (documentId != screenId) {
44 return Refusal{.document = std::string{screenId},
45 .path = {},
46 .reason = "the document of `" + std::string{screenId} + "` names the id `" +
47 std::string{documentId} + "`; it is refused"};
48 }
49 return std::nullopt;
50}
