29/// @brief All the information needed to dispatch one action call through a backend.
30///
31/// Backends that execute locally use `localOp` directly. Remote backends
32/// serialize via `serializeAction` and deserialize replies via `deserializeResult`.
33struct ActionCall {
34 /// @brief String id of the target model type (from `ModelTraits`).
35 std::string modelTypeId;
36
37 /// @brief String id of the action type (from `ActionTraits`).
38 std::string actionTypeId;
39
40 /// @brief Serialises the action to JSON. Called only on the remote path.
41 std::function<std::string()> serializeAction;
42
43 /// @brief Deserialises a JSON reply into the opaque result `shared_ptr<void>`.
44 std::function<std::shared_ptr<void>(std::string_view)> deserializeResult;
45
46 /// @brief Executes the action directly against a model holder. Used on the local path.
47 std::function<std::shared_ptr<void>(::morph::model::detail::IModelHolder&)> localOp;
48
49 /// @brief Session context attached to this call.
50 ///
51 /// Local backends thread it through a thread-local before invoking `localOp`;
52 /// remote backends serialise it into the wire envelope.
53 ::morph::session::Context session;
54};

56/// @brief The two identities a model instance can carry, passed together.
57///
58/// Bundled into one struct rather than passed as two adjacent `string_view`
59/// parameters because they are trivially swappable at a call site and mean
60/// entirely different things: transposing them would silently file journal
61/// entries under the directory key and share instances under the log's entity
62/// key. Keeping them named at every call site makes that mistake unwritable.
63struct InstanceIdentity {
64 /// @brief Entity key for the action log; empty if none. See `journal::LogEntry::entityKey`.
65 std::string_view contextKey;
66
67 /// @brief Canonical string encoding of the primary key; empty if the
68 /// instance is anonymous and therefore unshareable.
69 std::string_view primary;
70};

85 /// @brief Registers a new model instance, additionally passing @p contextKey —
86 /// the instance's stable identity (e.g. an account id) — through to

88 ///
89 /// Default implementation forwards to `registerModel()` and drops @p contextKey,
90 /// which is exactly correct for `LocalBackend`: the caller's own @p factory
91 /// closure already captures whatever identity it needs directly (see
92 /// `IModelHolder::attachActionLog`), so there is nothing for the backend to
93 /// forward. Backends whose model instances live behind a wire protocol
94 /// (`SimulatedRemoteBackend`) override this to carry @p contextKey across —
95 /// see `wire::Envelope::contextKey` and `RemoteServer::setLogProvider`.
96 /// @param typeId String type-id of the model to instantiate.
97 /// @param factory Callable that constructs the `IModelHolder` (local path only).
98 /// @param contextKey Stable identity of the new instance; empty if none.
99 /// @return Newly assigned `ModelId`.
100 virtual ::morph::exec::detail::ModelId registerModelWithContext(
101 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
102 std::string_view contextKey) {
103 (void)contextKey;
104 return registerModel(typeId, std::move(factory));

107 /// @brief Optional non-blocking counterpart to `registerModelWithContext`.
108 ///
109 /// `registerModelWithContext`/`registerModel` are synchronous: a backend
110 /// whose registration requires a round-trip (a socket backend) can only
111 /// implement that by blocking the calling thread until the reply arrives —
112 /// `QtWebSocketBackend` does this via a nested `QEventLoop`. On a WASM main
113 /// thread, Qt refuses to spin a nested loop at all
114 /// (`WaitForMoreEvents is not supported on the main thread without asyncify`),
115 /// so that blocking call aborts the page — the very first `registerModel`
116 /// a WASM client makes.
117 ///
118 /// Overriding this lets such a backend register without blocking: send the
119 /// request and return `true` immediately, then invoke exactly one of
120 /// @p onRegistered / @p onError once the reply arrives, on the backend's own
121 /// thread (unless the backend is destroyed first, in which case neither
122 /// fires). `Bridge::registerHandler()` prefers this path when it is
123 /// available and falls back to the synchronous `registerModelWithContext`
124 /// otherwise, so every backend that has not opted in (every backend as of
125 /// this writing, other than `QtWebSocketBackend`) is unaffected.
126 ///
127 /// The default implementation offers no async path and returns `false`
128 /// without calling either callback — the caller (`Bridge::registerHandler`)
129 /// falls back to `registerModelWithContext` in that case, matching every
130 /// caller's behavior before this method existed.
131 ///
132 /// @note Scope: only `Bridge::registerHandler()`'s plain (non-shared)
133 /// registration path — a `BridgeHandler`'s initial construction —
134 /// uses this. Shared/keyed registration has its own opt-in async
135 /// pair, `registerModelSharedAsync`/`attachModelAsync` below,
136 /// preferred by `Bridge::ensureBoundAsync`/`attachHandlerAsync`.
137 /// The re-registration `switchBackend()`/the reconnect handler
138 /// perform after a backend swap remains synchronous; see
139 /// docs/spec/core/backend.md.
140 /// @param typeId String type-id of the model to instantiate.
141 /// @param factory Callable that constructs the `IModelHolder` (local path only).
142 /// @param contextKey Stable identity of the new instance; empty if none.
143 /// @param onRegistered Invoked with the assigned `ModelId` on success.
144 /// @param onError Invoked with a diagnostic message on failure.
145 /// @return `true` if this backend accepted the request and will invoke
146 /// exactly one callback later; `false` if it has no async path
147 /// (neither callback is invoked in that case).
148 virtual bool registerModelAsync(
149 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
150 std::string_view contextKey, std::function<void(::morph::exec::detail::ModelId)> onRegistered,
151 std::function<void(const std::string&)> onError) {
152 (void)typeId;
153 (void)factory;
154 (void)contextKey;
155 (void)onRegistered;
156 (void)onError;
157 return false;
158 }

165 /// aborts a WASM main thread the moment a shared/keyed handler makes its
166 /// first attach. A backend that overrides this sends the request and
167 /// returns `true` immediately, then invokes exactly one of
168 /// @p onRegistered / @p onError once the reply arrives, on the backend's
169 /// own thread (unless the backend is destroyed first, in which case
170 /// neither fires).
171 ///
172 /// The default implementation offers no async path and returns `false`
173 /// without calling either callback — the caller (`Bridge::ensureBoundAsync`)
174 /// falls back to the synchronous `registerModelShared` in that case,
175 /// matching every caller's behavior before this method existed.
176 ///
177 /// @param typeId String type-id of the model.
178 /// @param factory Callable that constructs the `IModelHolder` (local path only).
179 /// @param identity Entity key for the action log plus the directory primary key.
180 /// @param onRegistered Invoked with the assigned/attached `ModelId` on success.
181 /// @param onError Invoked with a diagnostic message on failure.
182 /// @return `true` if this backend accepted the request and will invoke
183 /// exactly one callback later; `false` if it has no async path.

190 std::function<void(const std::string&)> onError) {

192 (void)factory;
193 (void)identity;

200 /// @brief Registers or attaches to the shared instance holding @p primary.
201 ///
202 /// A *register-or-attach*: if an instance for `(typeId, primary)` is already
203 /// live in the backend's shared directory, its id is returned and its attach
204 /// count incremented — no new instance is created and @p factory is not
205 /// called. Otherwise a new instance is created, entered in the directory,
206 /// and returned with an attach count of one.
207 ///
208 /// An empty @p primary means "no identity": the call degrades to
209 /// `registerModelWithContext`, producing a private instance that never
210 /// enters the directory and can never be shared.
211 ///
212 /// The default implementation ignores @p primary and forwards to
213 /// `registerModelWithContext`, so a backend that has not implemented sharing
214 /// keeps its existing one-instance-per-caller behaviour rather than silently
215 /// handing two callers the same instance.
216 ///
217 /// @param typeId String type-id of the model.
218 /// @param factory Callable that constructs the `IModelHolder` (local path only).
219 /// @param identity Entity key for the action log plus the directory primary key.
220 /// @return Id of the shared (or newly created) instance.
221 virtual ::morph::exec::detail::ModelId registerModelShared(
222 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
223 InstanceIdentity identity) {
224 return registerModelWithContext(typeId, std::move(factory), identity.contextKey);
225 }

227 /// @brief Re-points from @p current to the shared instance holding @p primary.
228 ///
229 /// The default implementation acquires the replacement via
230 /// `registerModelShared` first and only then releases @p current (when
231 /// non-zero), so a same-key re-attach never destroys and recreates the
232 /// instance it already holds, and a throwing acquire never strands the
233 /// caller with neither instance. Backends behind a wire protocol override
234 /// this with the single `attach` request so a re-pointing client cannot
235 /// lose its slot to `LimitPolicy::maxLiveModels` between the release and
236 /// the acquire.
237 ///
238 /// @param typeId String type-id of the model.
239 /// @param factory Callable that constructs the `IModelHolder` (local path only).
240 /// @param identity Entity key for the action log plus the directory primary key.
241 /// @param current Instance currently held, or `ModelId{0}` if none.
242 /// @return Id of the instance now attached to.
243 virtual ::morph::exec::detail::ModelId attachModel(
244 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
245 InstanceIdentity identity, ::morph::exec::detail::ModelId current) {
246 // Acquire the replacement before releasing `current`, not after: a
247 // same-key re-attach (registerModelShared finds `current` already
248 // live in the directory and takes a second reference to it) then
249 // hands back the identical id instead of destroying and recreating
250 // it, and a throwing acquire never touches `current` at all, so the
251 // caller's existing instance is never stranded by a failed attach.
252 // Either way, exactly one reference on `current` needs releasing
253 // afterward: the genuinely old instance's, if this re-pointed to a
254 // different key; or the redundant one registerModelShared just took,
255 // if it did not.
256 auto next = registerModelShared(typeId, std::move(factory), identity);
257 if (current.v != 0U) {
258 deregisterModel(current);
259 }
260 return next;
261 }

301 /// @brief Enters an already-live instance into the directory under @p primary.
302 ///
303 /// The *promotion* half of keyed instances, and what makes a result-sourced
304 /// key work without losing state: an action that creates its own entity runs
305 /// on an instance that does not yet have a key, and the key only exists once
306 /// the result comes back. Re-pointing to a freshly created instance would
307 /// strand everything the create just did, so instead the instance the action
308 /// ran on is given the generated key in place.
309 ///
310 /// A no-op when @p primary is empty, when @p mid is not live, when another
311 /// instance already holds that key, or when @p mid itself already holds a
312 /// *different* real key. The existing holder of a key always wins (a
313 /// promotion can never silently displace a directory entry other handlers
314 /// are attached to), and an already-keyed instance never changes key (a
315 /// promotion can never silently move one out from under handlers already
316 /// attached to it) — only a still-anonymous instance can ever be promoted.
317 ///
318 /// @param mid Live instance to promote.
319 /// @param typeId Model type id — the directory's first key component.
320 /// @param primary Canonical string encoding of the key to file it under.
321 virtual void assignPrimary(::morph::exec::detail::ModelId mid, const std::string& typeId,
322 std::string_view primary) {
323 (void)mid;
324 (void)typeId;
325 (void)primary;
326 }

328 /// @brief Optional non-blocking counterpart to `assignPrimary`.
329 ///
330 /// `assignPrimary` is synchronous: a backend whose promote step requires a
331 /// round-trip (a socket backend) can only implement that by blocking the
332 /// calling thread until the reply arrives — `QtWebSocketBackend` does this
333 /// via a nested `QEventLoop`, exactly the blocking shape
334 /// `registerModelAsync` exists to let a caller avoid for the bind step.
335 /// The promote step (the second half of a shared handler's result-keyed
336 /// `execute()`, after `Bridge::ensureBound`) reaches the same nested loop
337 /// on a single-threaded WASM build, which cannot spin one at all.
338 ///
339 /// Overriding this lets such a backend promote without blocking: send the
340 /// request and return `true` immediately, then invoke exactly one of
341 /// @p onRegistered / @p onError once the reply arrives, on the backend's
342 /// own thread (unless the backend is destroyed first, in which case
343 /// neither fires). `Bridge::assignHandlerPrimary` prefers this path when
344 /// it is available and falls back to the synchronous `assignPrimary`
345 /// otherwise, so every backend that has not opted in (every backend as of
346 /// this writing, other than `QtWebSocketBackend`) is unaffected.
347 ///
348 /// The default implementation offers no async path and returns `false`
349 /// without calling either callback — the caller (`Bridge::assignHandlerPrimary`)
350 /// falls back to `assignPrimary` in that case, matching every caller's
351 /// behavior before this method existed.
352 ///
353 /// @param mid Live instance to promote.
354 /// @param typeId Model type id — the directory's first key component.
355 /// @param primary Canonical string encoding of the key to file it under.
356 /// @param onRegistered Invoked with @p mid (echoed back, for symmetry with
357 /// `registerModelAsync`'s callback shape) on success —
358 /// including the no-op cases `assignPrimary` documents
359 /// (empty primary, dead `mid`, key already taken, `mid`
360 /// already keyed differently): those are not backend
361 /// failures, so they resolve `onRegistered` exactly as
362 /// the synchronous path returns normally for them.
363 /// @param onError Invoked with a diagnostic message on a genuine
364 /// backend/transport failure.
365 /// @return `true` if this backend accepted the request and will invoke
366 /// exactly one callback later; `false` if it has no async path
367 /// (neither callback is invoked in that case).
368 virtual bool assignPrimaryAsync(::morph::exec::detail::ModelId mid, const std::string& typeId,
369 std::string_view primary,
370 std::function<void(::morph::exec::detail::ModelId)> onRegistered,
371 std::function<void(const std::string&)> onError) {
372 (void)mid;
373 (void)typeId;
374 (void)primary;
375 (void)onRegistered;
376 (void)onError;
377 return false;
378 }

422 /// @brief Installs a callback invoked when the backend reconnects to its peer.
423 ///
424 /// Used by backends that may lose and re-establish their transport (e.g.
425 /// `QtWebSocketBackend`). `Bridge` installs a handler that re-registers every
426 /// live `HandlerBinding` so model ids stay valid after the reconnect.
427 ///
428 /// Deliberately fires only on the *second and later* connects, never the
429 /// first — re-registering handlers only makes sense after a drop; on the
430 /// first connect there is nothing yet to re-register. See
431 /// `setConnectHandler` for a hook that also covers the first connect.
432 ///
433 /// Default implementation: store-and-ignore. Backends with no transport (e.g.
434 /// `LocalBackend`) never invoke it.
435 /// @param handler Callable invoked on the backend's transport thread after a
436 /// successful reconnect. Pass `nullptr` to clear.
437 virtual void setReconnectHandler(const std::function<void()>& handler) { (void)handler; }

439 /// @brief Installs a callback invoked on every successful connect, including the first.
440 ///
441 /// `setReconnectHandler` deliberately skips the first connect (there is
442 /// nothing to re-register yet); this is the complementary hook for UI that
443 /// needs to know the transport is up at all — a "connecting… / connected /
444 /// offline" status indicator, for instance. `waitForConnected()` (where a
445 /// concrete backend offers one, e.g. `QtWebSocketBackend`) answers the same
446 /// question but blocks the calling thread, which is unusable on a
447 /// browser/WASM main thread and undesirable even on desktop if it means
448 /// blocking startup on a network round-trip; this hook is fired
449 /// asynchronously instead.
450 ///
451 /// Default implementation: store-and-ignore. Backends with no transport (e.g.
452 /// `LocalBackend`) never invoke it.
453 /// @param handler Callable invoked on the backend's transport thread after
454 /// every successful connect (first and subsequent). Pass
455 /// `nullptr` to clear.
456 virtual void setConnectHandler(const std::function<void()>& handler) { (void)handler; }
457
458 /// @brief Installs a callback invoked whenever the transport drops.
459 ///
460 /// Fires before any reconnect is scheduled, so an observer sees the
461 /// disconnected state even when a retry follows immediately — a status
462 /// indicator that skipped straight from "connected" to a fresh "connected"
463 /// (after an instant reconnect) would misreport an outage that did happen.
464 /// Without this hook a client learns the socket dropped only indirectly,
465 /// when a later action fails.
466 ///
467 /// Default implementation: store-and-ignore. Backends with no transport (e.g.
468 /// `LocalBackend`) never invoke it.
469 /// @param handler Callable invoked on the backend's transport thread whenever
470 /// the connection drops. Pass `nullptr` to clear.
471 virtual void setDisconnectHandler(const std::function<void()>& handler) { (void)handler; }

473 /// @brief Installs the session `Bridge` stamps onto every control envelope
474 /// this backend builds (`register`, `registerShared`, `attach`,
475 /// `assign`, `deregister`).
476 ///
477 /// `Bridge::executeVia` already stamps `Bridge::defaultSession()` onto the
478 /// `ActionCall` passed to `execute()`, so the session reaches `execute`
479 /// envelopes regardless of this hook. Control messages are different: they
480 /// are built directly by the concrete backend (`registerModelWithContext`,
481 /// `registerModelShared`, `attachModel`, `assignPrimary`, `deregisterModel`),
482 /// which has no other way to learn the `Bridge`'s current session except
483 /// this hook. Stamping the stored session onto those envelopes is what lets
484 /// `RemoteServer::authorizeRegister` see a caller's identity and the owner
485 /// principal recorded at `register` time reflect the registering session —
486 /// which `authorizeInstance`'s ownership check relies on for every instance
487 /// a `Bridge` registers.
488 ///
489 /// `Bridge::setDefaultSession()` calls this immediately, and
490 /// `Bridge::switchBackend()` calls it on the new backend before any
491 /// re-registration runs, so every control envelope built afterward carries
492 /// the current session. Default implementation: store-and-ignore, matching
493 /// `setReconnectHandler`'s pattern. `LocalBackend` needs no override — the
494 /// local path never serialises a session onto a wire envelope in the first
495 /// place (see docs/spec/session/session.md).
496 /// @param session Session to stamp onto every subsequently built control envelope.
497 virtual void setSession(::morph::session::Context session) { (void)session; }
498};

542/// @brief Thrown to a pending `Completion` when `Bridge::setExecuteDeadline`'s
543/// duration elapses before any reply arrives — a frame silently
544/// dropped by `QtWebSocketServerConfig::messagesPerSecond`, or a
545/// genuinely hung server, either way.
546///
547/// Distinct from `TimeoutError`: that type means the *server* explicitly
548/// replied that it hit `LimitPolicy::executeTimeout` while the action was
549/// still running. `ClientTimeoutError` means the client gave up waiting —
550/// no reply of any kind arrived, so whether the server ever received the
551/// request, is still processing it, or replied to a connection that had
552/// already dropped is unknown. See `docs/spec/core/completion.md`.
553struct ClientTimeoutError : std::runtime_error {
554 /// @brief Constructs the error with a canned diagnostic message.
555 ClientTimeoutError() : std::runtime_error{"execute timed out waiting for any reply"} {}
556};

558/// @brief In-process backend that executes model actions on a thread pool strand.

561/// without a global lock on the pool.

563public:
564 /// @brief Constructs the backend using @p workerPool to run model actions.
565 /// @param workerPool Executor (typically a `ThreadPoolExecutor`) for model work.

568 /// @brief Creates a model instance via @p factory and registers it.
569 ///
570 /// @p typeId is accepted for interface compatibility but not used — the
571 /// concrete type is captured by the factory closure. If the new holder's
572 /// `isBackendChangeAware()` returns `true`, @p mid is also recorded in
573 /// `_changeAware` so `notifyBackendChanged()` finds it without a
574 /// `dynamic_cast` sweep.
575 /// @param factory Callable that constructs the `IModelHolder`.
576 /// @return Newly assigned `ModelId`.
577 ::morph::exec::detail::ModelId registerModel(
578 const std::string& /*typeId*/,
579 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory) override {
580 ::morph::observe::detail::emitMetric(::morph::observe::Metric::registerCount, 1.0);
581 ::morph::exec::detail::ModelId mid{_nextId.fetch_add(1) + 1};
582 std::scoped_lock const lock{_regMtx};
583 auto holder = factory();
584 if (holder->isBackendChangeAware()) {
585 _changeAware.insert(mid);
586 }
587 _models[mid] = std::move(holder);
588 return mid;
589 }

591 /// @brief Registers or attaches to the shared instance holding @p primary.
592 ///
593 /// An empty @p primary bypasses the directory entirely and produces a
594 /// private instance, exactly as `registerModel` does.
595 /// @param typeId String type-id of the model — the directory's first key component.
596 /// @param factory Callable that constructs the `IModelHolder`; not called on an attach.
597 /// @param identity Entity key for the action log plus the directory primary key.
598 /// @return Id of the shared (or newly created) instance.
599 ::morph::exec::detail::ModelId registerModelShared(
600 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
601 detail::InstanceIdentity identity) override {
602 if (identity.primary.empty()) {
603 return registerModelWithContext(typeId, std::move(factory), identity.contextKey);
604 }
605 ::morph::observe::detail::emitMetric(::morph::observe::Metric::registerCount, 1.0);
606 DirectoryKey dirKey{typeId, std::string{identity.primary}};
607 std::scoped_lock const lock{_regMtx};
608 if (auto found = _directory.find(dirKey); found != _directory.end()) {
609 auto const foundMid = found->second;
610 auto flagsIter = _hydrationFlags.find(foundMid);
611 if (flagsIter != _hydrationFlags.end() && flagsIter->second->poisoned.load()) {
612 // Its first action already failed; it must not be handed to a
613 // new attacher. Evict it and fall through to the fresh-
614 // instance path below, exactly as if this had been a
615 // directory miss. Its own attachCount reference is untouched,
616 // so whoever created it still tears it down normally when
617 // they release it.
618 _directory.erase(found);
619 _sharedKeyOf.erase(foundMid);
620 } else {
621 _attachCount[foundMid] += 1;
622 return foundMid;
623 }
624 }
625 ::morph::exec::detail::ModelId const mid{_nextId.fetch_add(1) + 1};
626 auto holder = factory();
627 if (holder->isBackendChangeAware()) {
628 _changeAware.insert(mid);
629 }
630 _models[mid] = std::move(holder);
631 _directory.emplace(dirKey, mid);
632 _sharedKeyOf.emplace(mid, std::move(dirKey));
633 _attachCount[mid] = 1;
634 _hydrationFlags[mid] = std::make_shared<HydrationFlags>();
635 return mid;
636 }

638 /// @brief Enters an already-live, still-anonymous instance into the
639 /// directory under @p primary. Thread-safe.
640 /// @param mid Live instance to promote.
641 /// @param typeId Model type id — the directory's first key component.
642 /// @param primary Canonical string encoding of the key to file it under.
643 void assignPrimary(::morph::exec::detail::ModelId mid, const std::string& typeId,
644 std::string_view primary) override {
645 if (primary.empty()) {
646 return;
647 }
648 std::scoped_lock const lock{_regMtx};
649 if (!_models.contains(mid)) {
650 return;
651 }
652 DirectoryKey dirKey{typeId, std::string{primary}};
653 if (_directory.contains(dirKey)) {
654 return;
655 }
656 // Only a truly anonymous instance (no existing directory entry) can
657 // be promoted. An instance already filed under a *different* real key
658 // must not be silently re-filed onto this one -- instances never
659 // change key; re-pointing the handler is the supported way to move to
660 // a different entity, and it leaves the old key, and every other
661 // client still attached under it, untouched.
662 if (_sharedKeyOf.contains(mid)) {
663 return;
664 }
665 _directory.emplace(dirKey, mid);
666 _sharedKeyOf.emplace(mid, std::move(dirKey));
667 _attachCount.try_emplace(mid, 1);
668 }

670 /// @brief Lists the primary keys of live shared instances of @p typeId. Thread-safe.
671 /// @param typeId String type-id to enumerate.
672 /// @return Canonical key strings of the live shared instances, in unspecified order.
673 std::vector<std::string> listInstances(const std::string& typeId) override {
674 std::vector<std::string> keys;
675 std::scoped_lock const lock{_regMtx};
676 for (const auto& [dirKey, mid] : _directory) {
677 if (dirKey.first == typeId) {
678 keys.push_back(dirKey.second);
679 }
680 }
681 return keys;
682 }

684 /// @brief Removes the model with @p mid, or releases one attachment to it. Thread-safe.
685 ///
686 /// A private instance is erased outright. A shared instance has its attach
687 /// count decremented and is erased — and removed from the directory — only
688 /// when that count reaches zero, so releasing one handler never destroys an
689 /// instance another handler still holds.
690 /// @param mid Id returned by a prior `registerModel()`/`registerModelShared()` call.
691 void deregisterModel(::morph::exec::detail::ModelId mid) override {
692 ::morph::observe::detail::emitMetric(::morph::observe::Metric::deregisterCount, 1.0);
693 std::scoped_lock const lock{_regMtx};
694 if (auto refIter = _attachCount.find(mid); refIter != _attachCount.end()) {
695 refIter->second -= 1;
696 if (refIter->second > 0) {
697 return;
698 }
699 _attachCount.erase(refIter);
700 if (auto keyIter = _sharedKeyOf.find(mid); keyIter != _sharedKeyOf.end()) {
701 _directory.erase(keyIter->second);
702 _sharedKeyOf.erase(keyIter);
703 }
704 }
705 _models.erase(mid);
706 _changeAware.erase(mid);
707 _hydrationFlags.erase(mid);
708 }

710 /// @brief Schedules `onBackendChanged()` on each change-aware model's strand. Thread-safe.
711 ///
712 /// Only models recorded in `_changeAware` — maintained by `registerModel`/
713 /// `deregisterModel` from `IModelHolder::isBackendChangeAware()`, a
714 /// compile-time answer per model type — are visited; there is no
715 /// `dynamic_cast` and no scan of models that never opted in. Each such
716 /// model's `onBackendChanged()` (the `IModelHolder` base virtual) is
717 /// **posted onto that model's own strand** (the same per-`ModelId` serial
718 /// queue `execute` uses), rather than invoked inline on the caller's thread.
719 /// Two properties follow, and they are the whole point:
720 ///
721 /// - **Strand-serialised, lock-free model code.** `onBackendChanged()` runs
722 /// on the pool inside the model's strand, so it never overlaps an
723 /// `execute()` on the same model. A model reacting to a backend switch
724 /// (e.g. draining an offline queue and mutating counters) needs no locking
725 /// of its own state — exactly what `offline.md` promises.
726 /// - **Not under `Bridge::_mtx`.** `Bridge::switchBackend` calls this while
727 /// holding `_mtx`, but only the cheap `post()` runs there; the model body
728 /// runs later on a pool thread. A model that re-enters the bridge from
729 /// `onBackendChanged()` (`switchBackend`/`registerHandler`/`deregisterHandler`)
730 /// therefore acquires `_mtx` freshly on the strand thread instead of
731 /// deadlocking on a lock the switch caller still holds.
732 ///
733 /// A `shared_ptr` copy of each holder is captured into the posted task so a
734 /// concurrent `deregisterModel` cannot free the model out from under its
735 /// pending notification (mirrors `execute`'s holder capture).
736 void notifyBackendChanged() override {
737 std::vector<std::pair<::morph::exec::detail::ModelId, std::shared_ptr<::morph::model::detail::IModelHolder>>>
738 aware;
739 {
740 std::scoped_lock const lock{_regMtx};
741 aware.reserve(_changeAware.size());
742 for (auto modelId : _changeAware) {
743 auto iter = _models.find(modelId);
744 if (iter != _models.end()) {
745 aware.emplace_back(modelId, iter->second);
746 }
747 }
748 }
749 for (auto& [modelId, holder] : aware) {
750 _strand.post(modelId, [h = std::move(holder)]() mutable { h->onBackendChanged(); });
751 }
752 }

754 /// @brief Schedules `call.localOp` on the model's strand and returns a `Completion`.
755 ///
756 /// The completion resolves with the opaque result on the strand thread and
757 /// the callbacks are delivered via @p cbExec.
758 ///
759 /// @param mid Target model id.
760 /// @param call Bundled action; `localOp` is the only field used here.
761 /// @param cbExec Executor for delivering callbacks.
762 /// @return Completion that will carry the result or an exception.
763 ::morph::async::Completion<std::shared_ptr<void>> execute(::morph::exec::detail::ModelId mid,
764 detail::ActionCall call,
765 ::morph::exec::IExecutor* cbExec) override {
766 auto compState = std::make_shared<::morph::async::detail::CompletionState<std::shared_ptr<void>>>();
767 ::morph::async::Completion<std::shared_ptr<void>> comp{compState, cbExec};
768
769 std::shared_ptr<::morph::model::detail::IModelHolder> holder;
770 std::shared_ptr<HydrationFlags> hydration;
771 {
772 std::scoped_lock const lock{_regMtx};
773 auto iter = _models.find(mid);
774 if (iter != _models.end()) {
775 holder = iter->second;
776 }
777 if (auto flagsIter = _hydrationFlags.find(mid); flagsIter != _hydrationFlags.end()) {
778 hydration = flagsIter->second;
779 }
780 }
781 if (!holder) {
782 compState->setException(
783 std::make_exception_ptr(std::runtime_error("model not found: id=" + std::to_string(mid.v))));
784 return comp;
785 }
786 trackPending(compState);
787 auto localOp = std::move(call.localOp);
788 auto session = std::move(call.session);
789 auto modelTypeId = call.modelTypeId;
790 auto actionTypeId = call.actionTypeId;
791 // Captured by shared_ptr, never by raw `this`: see the Global
792 // Constraints note on `~StrandExecutor`'s member-destruction-order
793 // subtlety. A shared_ptr copy has its own lifetime, independent of
794 // LocalBackend's, so it stays valid even if the backend is torn down
795 // while this task is still queued or running. `hydration` follows the
796 // same rule and may be null (a private instance has no entry).
797 auto inFlightCounter = _inFlight;
798 auto const inFlightAfterInc = inFlightCounter->fetch_add(1, std::memory_order_relaxed) + 1;
799 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeInFlight,
800 static_cast<double>(inFlightAfterInc));
801 _strand.post(mid, [localOp = std::move(localOp), holder = std::move(holder), compState,
802 session = std::move(session), modelTypeId = std::move(modelTypeId),
803 actionTypeId = std::move(actionTypeId), inFlightCounter, hydration]() mutable {
804 auto const start = std::chrono::steady_clock::now();
805 auto const spanId = ::morph::observe::detail::beginSpan(session.requestId, modelTypeId, actionTypeId);
806 bool ok = false;
807 // Resolve `compState` only after every metric and `endSpan` below are
808 // recorded — nothing synchronizes a `.then()`/`.onError()` callback
809 // (delivered via `cbExec`, which may run inline/synchronously) with
810 // anything after `setValue`/`setException` returns, so resolving first
811 // would let the caller observe completion before these metrics are
812 // emitted. This is a real race, not just a theoretical one.
813 std::shared_ptr<void> value;
814 std::exception_ptr error;
815 try {
816 ::morph::session::detail::ScopedContext const scoped{session};
817 value = localOp(*holder);
818 ok = true;
819 } catch (...) {
820 error = std::current_exception();
821 }
822 ::morph::observe::detail::endSpan(spanId, ok);
823 auto const elapsedMs =
824 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - start).count();
825 std::array<std::pair<std::string_view, std::string_view>, 2> const tags{
826 {{"modelType", modelTypeId}, {"actionType", actionTypeId}}};
827 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeLatencyMs, elapsedMs, tags);
828 if (!ok) {
829 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeErrors, 1.0, tags);
830 }
831 auto const inFlightAfterDec = inFlightCounter->fetch_sub(1, std::memory_order_relaxed) - 1;
832 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeInFlight,
833 static_cast<double>(inFlightAfterDec));
834 // Only a freshly created shared instance carries a HydrationFlags
835 // token (a private instance, or one that was already live at
836 // attach time, has none). Its very first action's outcome decides
837 // whether it gets poisoned; a later action's failure is an
838 // ordinary error, not a hydration failure.
839 if (hydration && hydration->firstActionPending.exchange(false) && !ok) {
840 hydration->poisoned.store(true);
841 }
842 // Resolve last: the Completion is still settled exactly once, only
843 // its position relative to the now-recorded instrumentation moved.
844 if (ok) {
845 compState->setValue(std::move(value));
846 } else {
847 compState->setException(error);
848 }
849 });
850 return comp;
851 }

853 /// @brief Resolves every still-pending completion this backend produced with @p exc.
854 /// @param exc Exception delivered to every pending completion's error sink.
855 void cancelPending(const std::exception_ptr& exc) override {
856 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> snapshot;
857 {
858 std::scoped_lock const lock{_pendingMtx};
859 snapshot.swap(_pending);
860 }
861 for (auto& weak : snapshot) {
862 if (auto state = weak.lock()) {
863 state->setException(exc);
864 }
865 }
866 }

869 void trackPending(const std::shared_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>& state) {
870 std::scoped_lock const lock{_pendingMtx};
871 std::erase_if(_pending, [](const auto& weak) { return weak.expired(); });
872 _pending.emplace_back(state);
873 }

875 ::morph::exec::detail::StrandExecutor _strand;
876 std::mutex _regMtx;
877 std::unordered_map<::morph::exec::detail::ModelId, std::shared_ptr<::morph::model::detail::IModelHolder>,
878 ::morph::exec::detail::ModelIdHash>
879 _models;
880 // Ids of models whose holder answered `isBackendChangeAware() == true` at
881 // registration time. Maintained in lockstep with `_models` (inserted in
882 // `registerModel`, erased in `deregisterModel`, both under `_regMtx`) so
883 // `notifyBackendChanged()` never needs to inspect a model it doesn't have
884 // to. Always a subset of `_models`' keys.
885 std::unordered_set<::morph::exec::detail::ModelId, ::morph::exec::detail::ModelIdHash> _changeAware;
886 // Shared-instance directory: (typeId, primary) -> ModelId, plus the reverse
887 // lookup and per-instance attach count. All three are maintained under
888 // _regMtx alongside _models, so directory membership can never desync from
889 // instance existence — the same invariant RemoteServer's connection scopes
890 // maintain. Only instances registered with a non-empty primary appear here;
891 // a private instance has no entry in any of the three, which is exactly what
892 // makes deregisterModel's decrement path a no-op for it.
893 // Tracks, per freshly created shared instance, whether its very first
894 // action has settled yet and — if it has — whether that first action
895 // failed. Consulted lazily by registerModelShared's directory-hit branch,
896 // which evicts a "poisoned" instance (its first action failed, so per
897 // docs/spec/core/shared_instances.md's Failure modes it must not be left
898 // half-hydrated in the directory) and falls through to creating a fresh
899 // one, instead of handing the broken instance to a new attacher. Owned
900 // via shared_ptr and captured that way — never via raw `this` — into
901 // execute()'s strand task, which may still be running after LocalBackend
902 // itself is destroyed (see execute()'s existing capture-by-shared_ptr
903 // rationale).
904 struct HydrationFlags {
905 std::atomic<bool> firstActionPending{true};
906 std::atomic<bool> poisoned{false};
907 };
908
909 using DirectoryKey = std::pair<std::string, std::string>;
910 std::unordered_map<DirectoryKey, ::morph::exec::detail::ModelId, ::morph::model::detail::PairKeyHash> _directory;
911 std::unordered_map<::morph::exec::detail::ModelId, DirectoryKey, ::morph::exec::detail::ModelIdHash> _sharedKeyOf;
912 std::unordered_map<::morph::exec::detail::ModelId, std::size_t, ::morph::exec::detail::ModelIdHash> _attachCount;
913 std::unordered_map<::morph::exec::detail::ModelId, std::shared_ptr<HydrationFlags>, ::morph::exec::detail::ModelIdHash>
914 _hydrationFlags;
915 std::atomic<uint64_t> _nextId{0};
916 std::mutex _pendingMtx;
917 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> _pending;
918 // Concurrent in-flight executes, for the executeInFlight metric. A
919 // shared_ptr (not a plain atomic member) so strand tasks hold their own
920 // reference instead of capturing `this` — see execute()'s comment and the
921 // Global Constraints note on ~StrandExecutor's destruction order.
922 std::shared_ptr<std::atomic<std::size_t>> _inFlight = std::make_shared<std::atomic<std::size_t>>(0);
923};
