25/// @brief All the information needed to dispatch one action call through a backend.
26///
27/// Backends that execute locally use `localOp` directly. Remote backends
28/// serialize via `serializeAction` and deserialize replies via `deserializeResult`.
29struct ActionCall {
30 /// @brief String id of the target model type (from `ModelTraits`).
31 std::string modelTypeId;
32
33 /// @brief String id of the action type (from `ActionTraits`).
34 std::string actionTypeId;
35
36 /// @brief Serialises the action to JSON. Called only on the remote path.
37 std::function<std::string()> serializeAction;
38
39 /// @brief Deserialises a JSON reply into the opaque result `shared_ptr<void>`.
40 std::function<std::shared_ptr<void>(std::string_view)> deserializeResult;
41
42 /// @brief Executes the action directly against a model holder. Used on the local path.
43 std::function<std::shared_ptr<void>(::morph::model::detail::IModelHolder&)> localOp;
44
45 /// @brief Session context attached to this call.
46 ///
47 /// Local backends thread it through a thread-local before invoking `localOp`;
48 /// remote backends serialise it into the wire envelope.
49 ::morph::session::Context session;
50};

52/// @brief Abstract interface for execution backends (local, remote, …).
53///
54/// A backend owns model instances and dispatches actions against them.
55/// `Bridge` holds one active backend at a time and can swap it atomically
56/// via `Bridge::switchBackend()`.
57// NOLINTBEGIN(cppcoreguidelines-special-member-functions)
58struct IBackend {
59 virtual ~IBackend() = default;
60
61 /// @brief Registers a new model instance and returns its opaque id.
62 virtual ::morph::exec::detail::ModelId registerModel(
63 const std::string& typeId,
64 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory) = 0;
65
66 /// @brief Registers a new model instance, additionally passing @p contextKey —
67 /// the instance's stable identity (e.g. an account id) — through to
68 /// backends that can make use of it.
69 ///
70 /// Default implementation forwards to `registerModel()` and drops @p contextKey,
71 /// which is exactly correct for `LocalBackend`: the caller's own @p factory
72 /// closure already captures whatever identity it needs directly (see
73 /// `IModelHolder::attachActionLog`), so there is nothing for the backend to
74 /// forward. Backends whose model instances live behind a wire protocol
75 /// (`SimulatedRemoteBackend`) override this to carry @p contextKey across —
76 /// see `wire::Envelope::contextKey` and `RemoteServer::setLogProvider`.
77 /// @param typeId String type-id of the model to instantiate.
78 /// @param factory Callable that constructs the `IModelHolder` (local path only).
79 /// @param contextKey Stable identity of the new instance; empty if none.
80 /// @return Newly assigned `ModelId`.
81 virtual ::morph::exec::detail::ModelId registerModelWithContext(
82 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
83 std::string_view contextKey) {
84 (void)contextKey;
85 return registerModel(typeId, std::move(factory));
86 }
87
88 /// @brief Removes the model identified by @p mid from the backend.
89 virtual void deregisterModel(::morph::exec::detail::ModelId mid) = 0;
90
91 /// @brief Dispatches @p call against the model identified by @p mid.
92 virtual ::morph::async::Completion<std::shared_ptr<void>> execute(
93 ::morph::exec::detail::ModelId mid, ActionCall call, ::morph::exec::IExecutor* cbExec) = 0;
94
95 /// @brief Called by `Bridge::switchBackend()` after all handlers are re-registered.
96 virtual void notifyBackendChanged() = 0;
97
98 /// @brief Resolves every still-pending completion this backend produced with @p exc.
99 ///
100 /// Called by `Bridge::switchBackend()` on the outgoing backend after the swap,
101 /// and by `Bridge`'s destructor. After this call, any later `setValue` /
102 /// `setException` on those states is a no-op (the state is already ready), so
103 /// in-flight server replies cannot resurrect a cancelled completion.
104 virtual void cancelPending(const std::exception_ptr& exc) = 0;
105
106 /// @brief Installs a callback invoked when the backend reconnects to its peer.
107 ///
108 /// Used by backends that may lose and re-establish their transport (e.g.
109 /// `QtWebSocketBackend`). `Bridge` installs a handler that re-registers every
110 /// live `HandlerBinding` so model ids stay valid after the reconnect.
111 ///
112 /// Default implementation: store-and-ignore. Backends with no transport (e.g.
113 /// `LocalBackend`) never invoke it.
114 /// @param handler Callable invoked on the backend's transport thread after a
115 /// successful reconnect. Pass `nullptr` to clear.
116 virtual void setReconnectHandler(const std::function<void()>& handler) { (void)handler; }
117};

146/// @brief In-process backend that executes model actions on a thread pool strand.
147///
148/// Each model instance gets its own strand so actions are serialised per-model
149/// without a global lock on the pool.
150class LocalBackend : public detail::IBackend {
151public:
152 /// @brief Constructs the backend using @p workerPool to run model actions.
153 /// @param workerPool Executor (typically a `ThreadPoolExecutor`) for model work.
154 explicit LocalBackend(::morph::exec::IExecutor& workerPool) : _strand{workerPool} {}
155
156 /// @brief Creates a model instance via @p factory and registers it.
157 ///
158 /// @p typeId is accepted for interface compatibility but not used — the
159 /// concrete type is captured by the factory closure.
160 /// @param factory Callable that constructs the `IModelHolder`.
161 /// @return Newly assigned `ModelId`.
162 ::morph::exec::detail::ModelId registerModel(
163 const std::string& /*typeId*/,
164 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory) override {
165 ::morph::exec::detail::ModelId mid{_nextId.fetch_add(1) + 1};
166 std::scoped_lock const lock{_regMtx};
167 _models[mid] = factory();
168 return mid;
169 }
170
171 /// @brief Removes the model with @p mid. Thread-safe.
172 /// @param mid Id returned by a prior `registerModel()` call.
173 void deregisterModel(::morph::exec::detail::ModelId mid) override {
174 std::scoped_lock const lock{_regMtx};
175 _models.erase(mid);
176 }
177
178 /// @brief Notifies every live model that implements `IBackendChangedSink`. Thread-safe.
179 void notifyBackendChanged() override {
180 std::scoped_lock const lock{_regMtx};
181 for (auto& [modelId, holder] : _models) {
182 if (auto* sink = dynamic_cast<::morph::model::detail::IBackendChangedSink*>(holder.get())) {
183 sink->onBackendChanged();
184 }
185 }
186 }
187
188 /// @brief Schedules `call.localOp` on the model's strand and returns a `Completion`.
189 ///
190 /// The completion resolves with the opaque result on the strand thread and
191 /// the callbacks are delivered via @p cbExec.
192 ///
193 /// @param mid Target model id.
194 /// @param call Bundled action; `localOp` is the only field used here.
195 /// @param cbExec Executor for delivering callbacks.
196 /// @return Completion that will carry the result or an exception.
197 ::morph::async::Completion<std::shared_ptr<void>> execute(
198 ::morph::exec::detail::ModelId mid, detail::ActionCall call,
199 ::morph::exec::IExecutor* cbExec) override {
200 auto compState = std::make_shared<::morph::async::detail::CompletionState<std::shared_ptr<void>>>();
201 ::morph::async::Completion<std::shared_ptr<void>> comp{compState, cbExec};
202
203 std::shared_ptr<::morph::model::detail::IModelHolder> holder;
204 {
205 std::scoped_lock const lock{_regMtx};
206 auto iter = _models.find(mid);
207 if (iter != _models.end()) {
208 holder = iter->second;
209 }
210 }
211 if (!holder) {
212 compState->setException(
213 std::make_exception_ptr(std::runtime_error("model not found: id=" + std::to_string(mid.v))));
214 return comp;
215 }
216 trackPending(compState);
217 auto localOp = std::move(call.localOp);
218 auto session = std::move(call.session);
219 _strand.post(mid, [localOp = std::move(localOp), holder = std::move(holder), compState,
220 session = std::move(session)]() mutable {
221 try {
222 ::morph::session::detail::ScopedContext const scoped{session};
223 compState->setValue(localOp(*holder));
224 } catch (...) {
225 compState->setException(std::current_exception());
226 }
227 });
228 return comp;
229 }
230
231 /// @brief Resolves every still-pending completion this backend produced with @p exc.
232 /// @param exc Exception delivered to every pending completion's error sink.
233 void cancelPending(const std::exception_ptr& exc) override {
234 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> snapshot;
235 {
236 std::scoped_lock const lock{_pendingMtx};
237 snapshot.swap(_pending);
238 }
239 for (auto& weak : snapshot) {
240 if (auto state = weak.lock()) {
241 state->setException(exc);
242 }
243 }
244 }
245
246private:
247 void trackPending(const std::shared_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>& state) {
248 std::scoped_lock const lock{_pendingMtx};
249 std::erase_if(_pending, [](const auto& weak) { return weak.expired(); });
250 _pending.emplace_back(state);
251 }
252
253 ::morph::exec::detail::StrandExecutor _strand;
254 std::mutex _regMtx;
255 std::unordered_map<::morph::exec::detail::ModelId, std::shared_ptr<::morph::model::detail::IModelHolder>,
256 ::morph::exec::detail::ModelIdHash>
257 _models;
258 std::atomic<uint64_t> _nextId{0};
259 std::mutex _pendingMtx;
260 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> _pending;
261};
