29/// @brief All the information needed to dispatch one action call through a backend.
30///
31/// Backends that execute locally use `localOp` directly. Remote backends
32/// serialize via `serializeAction` and deserialize replies via `deserializeResult`.
33struct ActionCall {
34 /// @brief String id of the target model type (from `ModelTraits`).
35 std::string modelTypeId;
36
37 /// @brief String id of the action type (from `ActionTraits`).
38 std::string actionTypeId;
39
40 /// @brief Serialises the action to JSON. Called only on the remote path.
41 std::function<std::string()> serializeAction;
42
43 /// @brief Deserialises a JSON reply into the opaque result `shared_ptr<void>`.
44 std::function<std::shared_ptr<void>(std::string_view)> deserializeResult;
45
46 /// @brief Executes the action directly against a model holder. Used on the local path.
47 std::function<std::shared_ptr<void>(::morph::model::detail::IModelHolder&)> localOp;
48
49 /// @brief Session context attached to this call.
50 ///
51 /// Local backends thread it through a thread-local before invoking `localOp`;
52 /// remote backends serialise it into the wire envelope.
53 ::morph::session::Context session;
54};

56/// @brief Abstract interface for execution backends (local, remote, …).
57///
58/// A backend owns model instances and dispatches actions against them.
59/// `Bridge` holds one active backend at a time and can swap it atomically
60/// via `Bridge::switchBackend()`.
61// NOLINTBEGIN(cppcoreguidelines-special-member-functions)
62struct IBackend {
63 virtual ~IBackend() = default;
64
65 /// @brief Registers a new model instance and returns its opaque id.
66 virtual ::morph::exec::detail::ModelId registerModel(
67 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory) = 0;
68
69 /// @brief Registers a new model instance, additionally passing @p contextKey —
70 /// the instance's stable identity (e.g. an account id) — through to
71 /// backends that can make use of it.
72 ///
73 /// Default implementation forwards to `registerModel()` and drops @p contextKey,
74 /// which is exactly correct for `LocalBackend`: the caller's own @p factory
75 /// closure already captures whatever identity it needs directly (see
76 /// `IModelHolder::attachActionLog`), so there is nothing for the backend to
77 /// forward. Backends whose model instances live behind a wire protocol
78 /// (`SimulatedRemoteBackend`) override this to carry @p contextKey across —
79 /// see `wire::Envelope::contextKey` and `RemoteServer::setLogProvider`.
80 /// @param typeId String type-id of the model to instantiate.
81 /// @param factory Callable that constructs the `IModelHolder` (local path only).
82 /// @param contextKey Stable identity of the new instance; empty if none.
83 /// @return Newly assigned `ModelId`.
84 virtual ::morph::exec::detail::ModelId registerModelWithContext(
85 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
86 std::string_view contextKey) {
87 (void)contextKey;
88 return registerModel(typeId, std::move(factory));
89 }
90
91 /// @brief Removes the model identified by @p mid from the backend.
92 virtual void deregisterModel(::morph::exec::detail::ModelId mid) = 0;
93
94 /// @brief Dispatches @p call against the model identified by @p mid.
95 virtual ::morph::async::Completion<std::shared_ptr<void>> execute(::morph::exec::detail::ModelId mid,
96 ActionCall call,
97 ::morph::exec::IExecutor* cbExec) = 0;
98
99 /// @brief Called by `Bridge::switchBackend()` after all handlers are re-registered.
100 virtual void notifyBackendChanged() = 0;
101
102 /// @brief Resolves every still-pending completion this backend produced with @p exc.
103 ///
104 /// Called by `Bridge::switchBackend()` on the outgoing backend after the swap,
105 /// and by `Bridge`'s destructor. After this call, any later `setValue` /
106 /// `setException` on those states is a no-op (the state is already ready), so
107 /// in-flight server replies cannot resurrect a cancelled completion.
108 virtual void cancelPending(const std::exception_ptr& exc) = 0;
109
110 /// @brief Installs a callback invoked when the backend reconnects to its peer.
111 ///
112 /// Used by backends that may lose and re-establish their transport (e.g.
113 /// `QtWebSocketBackend`). `Bridge` installs a handler that re-registers every
114 /// live `HandlerBinding` so model ids stay valid after the reconnect.
115 ///
116 /// Default implementation: store-and-ignore. Backends with no transport (e.g.
117 /// `LocalBackend`) never invoke it.
118 /// @param handler Callable invoked on the backend's transport thread after a
119 /// successful reconnect. Pass `nullptr` to clear.
120 virtual void setReconnectHandler(const std::function<void()>& handler) { (void)handler; }
121};

165/// @brief In-process backend that executes model actions on a thread pool strand.
166///
167/// Each model instance gets its own strand so actions are serialised per-model
168/// without a global lock on the pool.
169class LocalBackend : public detail::IBackend {
170public:
171 /// @brief Constructs the backend using @p workerPool to run model actions.
172 /// @param workerPool Executor (typically a `ThreadPoolExecutor`) for model work.
173 explicit LocalBackend(::morph::exec::IExecutor& workerPool) : _strand{workerPool} {}
174
175 /// @brief Creates a model instance via @p factory and registers it.
176 ///
177 /// @p typeId is accepted for interface compatibility but not used — the
178 /// concrete type is captured by the factory closure. If the new holder's
179 /// `isBackendChangeAware()` returns `true`, @p mid is also recorded in
180 /// `_changeAware` so `notifyBackendChanged()` finds it without a
181 /// `dynamic_cast` sweep.
182 /// @param factory Callable that constructs the `IModelHolder`.
183 /// @return Newly assigned `ModelId`.
184 ::morph::exec::detail::ModelId registerModel(
185 const std::string& /*typeId*/,
186 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory) override {
187 ::morph::observe::detail::emitMetric(::morph::observe::Metric::registerCount, 1.0);
188 ::morph::exec::detail::ModelId mid{_nextId.fetch_add(1) + 1};
189 std::scoped_lock const lock{_regMtx};
190 auto holder = factory();
191 if (holder->isBackendChangeAware()) {
192 _changeAware.insert(mid);
193 }
194 _models[mid] = std::move(holder);
195 return mid;
196 }
197
198 /// @brief Removes the model with @p mid. Thread-safe.
199 /// @param mid Id returned by a prior `registerModel()` call.
200 void deregisterModel(::morph::exec::detail::ModelId mid) override {
201 ::morph::observe::detail::emitMetric(::morph::observe::Metric::deregisterCount, 1.0);
202 std::scoped_lock const lock{_regMtx};
203 _models.erase(mid);
204 _changeAware.erase(mid);
205 }
206
207 /// @brief Schedules `onBackendChanged()` on each change-aware model's strand. Thread-safe.
208 ///
209 /// Only models recorded in `_changeAware` — maintained by `registerModel`/
210 /// `deregisterModel` from `IModelHolder::isBackendChangeAware()`, a
211 /// compile-time answer per model type — are visited; there is no
212 /// `dynamic_cast` and no scan of models that never opted in. Each such
213 /// model's `onBackendChanged()` (the `IModelHolder` base virtual) is
214 /// **posted onto that model's own strand** (the same per-`ModelId` serial
215 /// queue `execute` uses), rather than invoked inline on the caller's thread.
216 /// Two properties follow, and they are the whole point:
217 ///
218 /// - **Strand-serialised, lock-free model code.** `onBackendChanged()` runs
219 /// on the pool inside the model's strand, so it never overlaps an
220 /// `execute()` on the same model. A model reacting to a backend switch
221 /// (e.g. draining an offline queue and mutating counters) needs no locking
222 /// of its own state — exactly what `offline.md` promises.
223 /// - **Not under `Bridge::_mtx`.** `Bridge::switchBackend` calls this while
224 /// holding `_mtx`, but only the cheap `post()` runs there; the model body
225 /// runs later on a pool thread. A model that re-enters the bridge from
226 /// `onBackendChanged()` (`switchBackend`/`registerHandler`/`deregisterHandler`)
227 /// therefore acquires `_mtx` freshly on the strand thread instead of
228 /// deadlocking on a lock the switch caller still holds.
229 ///
230 /// A `shared_ptr` copy of each holder is captured into the posted task so a
231 /// concurrent `deregisterModel` cannot free the model out from under its
232 /// pending notification (mirrors `execute`'s holder capture).
233 void notifyBackendChanged() override {
234 std::vector<std::pair<::morph::exec::detail::ModelId, std::shared_ptr<::morph::model::detail::IModelHolder>>>
235 aware;
236 {
237 std::scoped_lock const lock{_regMtx};
238 aware.reserve(_changeAware.size());
239 for (auto modelId : _changeAware) {
240 auto iter = _models.find(modelId);
241 if (iter != _models.end()) {
242 aware.emplace_back(modelId, iter->second);
243 }
244 }
245 }
246 for (auto& [modelId, holder] : aware) {
247 _strand.post(modelId, [h = std::move(holder)]() mutable { h->onBackendChanged(); });
248 }
249 }
250
251 /// @brief Schedules `call.localOp` on the model's strand and returns a `Completion`.
252 ///
253 /// The completion resolves with the opaque result on the strand thread and
254 /// the callbacks are delivered via @p cbExec.
255 ///
256 /// @param mid Target model id.
257 /// @param call Bundled action; `localOp` is the only field used here.
258 /// @param cbExec Executor for delivering callbacks.
259 /// @return Completion that will carry the result or an exception.
260 ::morph::async::Completion<std::shared_ptr<void>> execute(::morph::exec::detail::ModelId mid,
261 detail::ActionCall call,
262 ::morph::exec::IExecutor* cbExec) override {
263 auto compState = std::make_shared<::morph::async::detail::CompletionState<std::shared_ptr<void>>>();
264 ::morph::async::Completion<std::shared_ptr<void>> comp{compState, cbExec};
265
266 std::shared_ptr<::morph::model::detail::IModelHolder> holder;
267 {
268 std::scoped_lock const lock{_regMtx};
269 auto iter = _models.find(mid);
270 if (iter != _models.end()) {
271 holder = iter->second;
272 }
273 }
274 if (!holder) {
275 compState->setException(
276 std::make_exception_ptr(std::runtime_error("model not found: id=" + std::to_string(mid.v))));
277 return comp;
278 }
279 trackPending(compState);
280 auto localOp = std::move(call.localOp);
281 auto session = std::move(call.session);
282 auto modelTypeId = call.modelTypeId;
283 auto actionTypeId = call.actionTypeId;
284 // Captured by shared_ptr, never by raw `this`: see the Global
285 // Constraints note on `~StrandExecutor`'s member-destruction-order
286 // subtlety. A shared_ptr copy has its own lifetime, independent of
287 // LocalBackend's, so it stays valid even if the backend is torn down
288 // while this task is still queued or running.
289 auto inFlightCounter = _inFlight;
290 auto const inFlightAfterInc = inFlightCounter->fetch_add(1, std::memory_order_relaxed) + 1;
291 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeInFlight,
292 static_cast<double>(inFlightAfterInc));
293 _strand.post(mid, [localOp = std::move(localOp), holder = std::move(holder), compState,
294 session = std::move(session), modelTypeId = std::move(modelTypeId),
295 actionTypeId = std::move(actionTypeId), inFlightCounter]() mutable {
296 auto const start = std::chrono::steady_clock::now();
297 auto const spanId = ::morph::observe::detail::beginSpan(session.requestId, modelTypeId, actionTypeId);
298 bool ok = false;
299 // Resolve `compState` only after every metric and `endSpan` below are
300 // recorded — nothing synchronizes a `.then()`/`.onError()` callback
301 // (delivered via `cbExec`, which may run inline/synchronously) with
302 // anything after `setValue`/`setException` returns, so resolving first
303 // would let the caller observe completion before these metrics are
304 // emitted. This is a real race, not just a theoretical one.
305 std::shared_ptr<void> value;
306 std::exception_ptr error;
307 try {
308 ::morph::session::detail::ScopedContext const scoped{session};
309 value = localOp(*holder);
310 ok = true;
311 } catch (...) {
312 error = std::current_exception();
313 }
314 ::morph::observe::detail::endSpan(spanId, ok);
315 auto const elapsedMs =
316 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - start).count();
317 std::array<std::pair<std::string_view, std::string_view>, 2> const tags{
318 {{"modelType", modelTypeId}, {"actionType", actionTypeId}}};
319 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeLatencyMs, elapsedMs, tags);
320 if (!ok) {
321 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeErrors, 1.0, tags);
322 }
323 auto const inFlightAfterDec = inFlightCounter->fetch_sub(1, std::memory_order_relaxed) - 1;
324 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeInFlight,
325 static_cast<double>(inFlightAfterDec));
326 // Resolve last: the Completion is still settled exactly once, only
327 // its position relative to the now-recorded instrumentation moved.
328 if (ok) {
329 compState->setValue(std::move(value));
330 } else {
331 compState->setException(error);
332 }
333 });
334 return comp;
335 }
336
337 /// @brief Resolves every still-pending completion this backend produced with @p exc.
338 /// @param exc Exception delivered to every pending completion's error sink.
339 void cancelPending(const std::exception_ptr& exc) override {
340 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> snapshot;
341 {
342 std::scoped_lock const lock{_pendingMtx};
343 snapshot.swap(_pending);
344 }
345 for (auto& weak : snapshot) {
346 if (auto state = weak.lock()) {
347 state->setException(exc);
348 }
349 }
350 }
351
352private:
353 void trackPending(const std::shared_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>& state) {
354 std::scoped_lock const lock{_pendingMtx};
355 std::erase_if(_pending, [](const auto& weak) { return weak.expired(); });
356 _pending.emplace_back(state);
357 }
358
359 ::morph::exec::detail::StrandExecutor _strand;
360 std::mutex _regMtx;
361 std::unordered_map<::morph::exec::detail::ModelId, std::shared_ptr<::morph::model::detail::IModelHolder>,
362 ::morph::exec::detail::ModelIdHash>
363 _models;
364 // Ids of models whose holder answered `isBackendChangeAware() == true` at
365 // registration time. Maintained in lockstep with `_models` (inserted in
366 // `registerModel`, erased in `deregisterModel`, both under `_regMtx`) so
367 // `notifyBackendChanged()` never needs to inspect a model it doesn't have
368 // to. Always a subset of `_models`' keys.
369 std::unordered_set<::morph::exec::detail::ModelId, ::morph::exec::detail::ModelIdHash> _changeAware;
370 std::atomic<uint64_t> _nextId{0};
371 std::mutex _pendingMtx;
372 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> _pending;
373 // Concurrent in-flight executes, for the executeInFlight metric. A
374 // shared_ptr (not a plain atomic member) so strand tasks hold their own
375 // reference instead of capturing `this` — see execute()'s comment and the
376 // Global Constraints note on ~StrandExecutor's destruction order.
377 std::shared_ptr<std::atomic<std::size_t>> _inFlight = std::make_shared<std::atomic<std::size_t>>(0);
378};
