117/// @brief The workspace's tabs with no widgets: which screens are open, in which order, which is active, and how
118/// they open, close and are recorded (spec 6 §7, §15).
119///
120/// Opening a screen whose identity an open tab already has focuses that tab, a singleton screen has one tab, and a
121/// `reuseEmpty` screen opened with an identity replaces its tab with no identity and no unsaved changes. Closing runs
122/// each screen's close protocol (spec 6 §3), one tab at a time, and a walk over several tabs stops at the first veto.
123/// `next` and `previous` walk the most recently used order. The page each tab shows is the application's:
124/// `mountPage` makes it and mounts the screen there, `unmountPage` is told when it goes. Neither copyable nor movable;
125/// owned by the owner thread of its runtime.
126class WorkspaceModel {
127public:
128 /// @brief An empty workspace.
129 /// @param runtime The runtime the screens' reactive nodes belong to. Borrowed.
130 /// @param host Answers identities and routes. Borrowed; it outlives the model.
131 /// @param options The screens' catalog facts, the record, and the page callbacks.
132 /// @throws std::invalid_argument when `mountPage` is empty.
133 WorkspaceModel(reactive::Runtime& runtime MORPH_LIFETIMEBOUND, DocumentScreenHost& host MORPH_LIFETIMEBOUND,
134 WorkspaceModelOptions options)
135 : _runtime{&runtime},
136 _host{&host},
137 _options{std::move(options)},
138 _changes{runtime, 0},
139 _activeTab{runtime, std::nullopt},
140 _tabStates{runtime, [this] { return tabStates(); }},
141 _activeState{runtime, [this] { return _activeTab.get(); }} {
142 if (!_options.mountPage) {
143 throw std::invalid_argument{"morph::ui::WorkspaceModel: mountPage is empty; it makes each tab's page"};
144 }
145 }
146
147 /// @brief Unmounts every screen without asking, newest first.
148 ~WorkspaceModel() {
149 while (!_tabs.empty()) {
150 unmountLast();
151 }
152 }

159 /// @brief Opens a screen, or focuses the tab that already shows it; the tab opened or focused is the active one.
160 /// @param screenId The screen id.
161 /// @param params Its params.
162 /// @return The tab, or why the screen could not be mounted; a refused mount, or a `mountPage` that throws, opens
163 /// nothing.
164 std::expected<WorkspaceTabId, MountError> open(std::string_view screenId, ParamMap const& params) {
165 WorkspaceScreen const facts = factsOf(screenId);
166 std::optional<std::string> const identity = _host->identityOf(screenId, params);
167 for (auto const& tab : _tabs) {
168 if (tab->screenId == screenId && (facts.singleton || (identity && identityText(*tab) == *identity))) {
169 activateTab(tab->id, false);
170 return tab->id;
171 }
172 }
173 std::optional<WorkspaceTabId> replaced;
174 if (facts.reuseEmpty && identity) {
175 for (auto const& tab : _tabs) {
176 if (tab->screenId == screenId && tab->handle->identity().peek().isNull() &&
177 !tab->handle->dirty().peek()) {
178 replaced = tab->id;
179 break;
180 }
181 }
182 }
183 auto opened = mountTab(screenId, params);
184 if (!opened) {
185 return std::unexpected(std::move(opened.error()));
186 }
187 WorkspaceTabId const tabId = *opened;
188 if (replaced) {
189 moveTab(tabId, indexOf(*replaced));
190 removeTab(*replaced, true);
191 }
192 activateTab(tabId, false);
193 return tabId;
194 }

196 /// @brief Opens the start screens: with `resume`, every recorded route whose screen is known and available, in
197 /// order, and the fallback when none of them opened; otherwise the `start` screen. The first tab opened is
198 /// the active one.
199 /// @param how The start screen, the fallback and the record.
200 /// @return The number of tabs opened.
201 std::size_t start(WorkspaceStart const& how) {
202 std::size_t opened = 0;
203 std::optional<WorkspaceTabId> first;
204 auto const openOne = [&](std::string_view screenId, ParamMap const& params) {
205 if (auto const tab = open(screenId, params)) {
206 ++opened;
207 if (!first) {
208 first = *tab;
209 }
210 }
211 };
212 if (how.start == "resume") {
213 for (std::string const& route : how.recorded) {
214 std::optional<Route> const read = readRoute(route);
215 if (read && factsOf(read->screen).available) {
216 openOne(read->screen, read->params);
217 }
218 }
219 if (opened == 0 && !how.fallback.empty()) {
220 openOne(how.fallback, {});
221 }
222 } else {
223 openOne(how.start, {});
224 }
225 if (first && indexOf(*first) != kNone) {
226 activateTab(*first, false);
227 }
228 return opened;
229 }

244 /// @param tab The tab.
245 /// @param done Called once with whether it closed; may be empty.

249 /// @brief Closes every tab but one, one at a time, stopping at the first veto.
250 /// @param tab The tab kept.
251 /// @param done Called once with whether every other tab closed; may be empty.
252 /// @throws std::out_of_range for a tab the model does not hold.
253 void closeOthers(WorkspaceTabId tab, std::function<void(bool)> done) {
254 WorkspaceTabId const kept = held(tab);
255 std::vector<WorkspaceTabId> ids;
256 ids.reserve(_tabs.size());
257 for (auto const& open : _tabs) {
258 if (open->id != kept) {
259 ids.push_back(open->id);
260 }
261 }
262 closeEach(std::move(ids), std::move(done));
263 }

404 // A recorded route read back: a host screen's route is its id, a document screen's is read against its params.
405 [[nodiscard]] std::optional<Route> readRoute(std::string const& route) const {

416 // Asks for a new page at the end and mounts the screen in it; a refusal, or a throw, opens nothing.
417 std::expected<WorkspaceTabId, MountError> mountTab(std::string_view screenId, ParamMap const& params) {
418 WorkspaceTabId const tabId{++_lastId};
419 std::expected<std::unique_ptr<ScreenHandle>, MountError> handle;
420 try {
421 handle = _options.mountPage(tabId, screenId, params);
422 } catch (std::exception const& failure) {
423 handle = std::unexpected(MountError{.screenId = std::string{screenId}, .reason = failure.what()});
424 } catch (...) {
425 handle = std::unexpected(MountError{.screenId = std::string{screenId}, .reason = "mountPage threw"});
426 }
427 if (handle && *handle == nullptr) {
428 handle = std::unexpected(
429 MountError{.screenId = std::string{screenId}, .reason = "mountPage answered no handle"});
430 }
431 if (!handle) {
432 return std::unexpected(std::move(handle.error()));
433 }
434 auto tab = std::make_unique<Tab>();
435 tab->id = tabId;
436 tab->screenId = std::string{screenId};
437 if (!factsOf(screenId).host.empty()) {
438 tab->route = std::string{screenId};
439 } else if (auto route = _host->routeOf(screenId, params)) {
440 tab->route = std::move(*route);
441 }
442 tab->handle = std::move(*handle);
443 tab->scope = std::make_unique<reactive::Scope>(*_runtime);
444 ScreenHandle const* const screen = tab->handle.get();
445 tab->scope->effect([this, screen] {
446 static_cast<void>(screen->title().get());
447 static_cast<void>(screen->dirty().get());
448 static_cast<void>(screen->identity().get());
449 _runtime->untracked([this] { changed(); });
450 });
451 _tabs.push_back(std::move(tab));
452 changed();
453 recordChanged();
454 return tabId;
455 }
456
457 // The open tabs changed: the record is written now, or once the burst is over.
458 void recordChanged() {
459 if (!_options.onRecord) {
460 return;
461 }
462 if (_options.scheduler == nullptr) {
463 _options.onRecord(record());
464 return;
465 }
466 std::weak_ptr<bool> const alive = _alive;
467 _recordTimer = _options.scheduler->after(_options.recordDelay, [this, alive] {
468 if (!alive.expired()) {
469 flushRecord();
470 }
471 });
472 }

474 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): a tab, then the index it moves to.
475 void moveTab(WorkspaceTabId tabId, std::size_t target) {
476 std::size_t const from = indexOf(tabId);
477 if (from == kNone || from == target) {
478 return;
479 }
480 std::unique_ptr<Tab> tab = std::move(_tabs.at(from));
481 _tabs.erase(_tabs.begin() + static_cast<std::ptrdiff_t>(from));
482 _tabs.insert(_tabs.begin() + static_cast<std::ptrdiff_t>(target), std::move(tab));
483 changed();
484 }

486 // Destroys the last tab's handle, then tells the application its page can go.
487 void unmountLast() {
488 WorkspaceTabId const tabId = _tabs.back()->id;
489 _tabs.pop_back();
490 if (_options.unmountPage) {
491 _options.unmountPage(tabId);
492 }
493 }
494
495 void removeTab(WorkspaceTabId tabId, bool force) {
496 std::size_t const where = indexOf(tabId);
497 if (where == kNone) {
498 return;
499 }
500 if (force) {
501 _tabs.at(where)->handle->forceClose();
502 }
503 _tabs.erase(_tabs.begin() + static_cast<std::ptrdiff_t>(where));
504 if (_options.unmountPage) {
505 _options.unmountPage(tabId);
506 }
507 std::erase(_recent, tabId);
508 _walk.reset();
509 recordChanged();
510 if (_active == tabId) {
511 setActive(std::nullopt);
512 if (!_recent.empty()) {
513 activateTab(_recent.front(), false);
514 return;
515 }
516 }
517 changed();
518 }
519
520 void activateTab(WorkspaceTabId tabId, bool walking) {
521 if (!walking) {
522 _walk.reset();
523 }
524 setActive(tabId);
525 std::erase(_recent, tabId);
526 _recent.insert(_recent.begin(), tabId);
527 changed();
528 }
529
530 void walk(int step) {
531 if (_tabs.size() < 2 || !_active) {
532 return;
533 }
534 if (!_walk) {
535 _walk = Walk{.order = _recent, .position = 0};
536 }
537 auto const count = static_cast<std::ptrdiff_t>(_walk->order.size());
538 std::ptrdiff_t const next = (((static_cast<std::ptrdiff_t>(_walk->position) + step) % count) + count) % count;
539 _walk->position = static_cast<std::size_t>(next);
540 // A walk keeps its own order; the tab it shows moves to the front of the recent list.
541 activateTab(_walk->order.at(_walk->position), true);
542 }
543
544 // Closes the tabs one at a time, stopping at the first that stays open.
545 void closeEach(std::vector<WorkspaceTabId> ids, std::function<void(bool)> done) {
546 std::erase_if(ids, [this](WorkspaceTabId tabId) { return indexOf(tabId) == kNone; });
547 if (ids.empty()) {
548 if (done) {
549 done(true);
550 }
551 return;
552 }
553 WorkspaceTabId const tabId = ids.front();
554 ids.erase(ids.begin());
555 std::weak_ptr<bool> const alive = _alive;
556 _tabs.at(indexOf(tabId))
557 ->handle->requestClose(
558 [this, alive, tabId, rest = std::move(ids), done = std::move(done)](bool closed) mutable {
559 if (alive.expired()) {
560 return;
561 }
562 if (!closed) {
563 if (done) {
564 done(false);
565 }
566 return;
567 }
568 removeTab(tabId, false);
569 closeEach(std::move(rest), std::move(done));
570 });
571 }

573 void setActive(std::optional<WorkspaceTabId> tab) {
574 _active = tab;
575 _runtime->batch([this, tab] { _activeTab.set(tab); });
576 }

578 // Something a tab strip shows changed: `tabs()` follows, and the application is told.
579 void changed() {
580 _runtime->batch([this] { _changes.set(_changes.peek() + 1); });
581 if (_options.onChanged) {
582 _options.onChanged();
583 }
584 }
