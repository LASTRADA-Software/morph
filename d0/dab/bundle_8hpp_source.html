83/// @brief The content hash of @p bytes: `sha256:` and the SHA-256 digest in 64 lower-case hexadecimal digits.
84/// @param bytes The bytes.
85/// @return The hash.
86[[nodiscard]] inline std::string contentHash(std::string_view bytes) {
87 static constexpr std::string_view kHex = "0123456789abcdef";
88 std::string const raw = ::morph::session::detail::sha256(bytes);
89 std::string out{kHashPrefix};
90 out.reserve(kHashPrefix.size() + kHashDigits);
91 for (char const byte : raw) {
92 auto const value = static_cast<unsigned char>(byte);
93 out.push_back(kHex.at(static_cast<std::size_t>(value >> 4U)));
94 out.push_back(kHex.at(static_cast<std::size_t>(value & 0x0FU)));
95 }
96 return out;
97}

99/// @brief Whether @p text is a well-formed content hash.
100/// @param text The text.
101/// @return `true` for `sha256:` followed by exactly 64 lower-case hexadecimal digits.
102[[nodiscard]] inline bool isContentHash(std::string_view text) {
103 if (text.size() != kHashPrefix.size() + kHashDigits || !text.starts_with(kHashPrefix)) {
104 return false;
105 }
106 return std::ranges::all_of(text.substr(kHashPrefix.size()), [](char const digit) {
107 return (digit >= '0' && digit <= '9') || (digit >= 'a' && digit <= 'f');
108 });
109}

129/// @brief A bundle's manifest: every file the application serves, by its hash (spec 5 §12, "Content addressing").

138 /// @brief Finds a file.
139 /// @param kind The file kind.
140 /// @param fileId The file id.
141 /// @return The file, or null.
142 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): a manifest is keyed by kind, then id.

144 auto const found = std::ranges::find_if(
145 files, [kind, fileId](ManifestFile const& file) { return file.kind == kind && file.id == fileId; });
146 return found == files.end() ? nullptr : &*found;
147 }

206/// @brief The body of a `ui-catalog` reply: the catalog filtered for the principal, and the facts a client's
207/// load-time validation reads (`ui::Catalog`).
208struct CatalogReply {
209 /// @brief The entries the principal sees, in registration order.
210 std::vector<CatalogEntry> entries;
211 /// @brief The action ids the principal may run, which `{"can": …}` reads.
212 std::vector<std::string> permitted;
213 /// @brief The model facts by type id; absent when the server has none to give.
214 std::optional<std::map<std::string, ModelInfo, std::less<>>> models;
215 /// @brief The action facts by action id; absent when the server has none to give.
216 std::optional<std::map<std::string, ActionInfo, std::less<>>> actions;
217
218 /// @brief The facts as the lookups load-time validation takes. A lookup the reply carried no facts for is left
219 /// empty, so its rules are skipped.
220 /// @return The catalog.
221 [[nodiscard]] Catalog facts() const {
222 Catalog catalog;
223 if (models) {
224 catalog.model = [known = std::make_shared<std::map<std::string, ModelInfo, std::less<>> const>(*models)](
225 std::string_view typeId) -> std::optional<ModelInfo> {
226 auto const found = known->find(typeId);
227 if (found == known->end()) {
228 return std::nullopt;
229 }
230 return found->second;
231 };
232 }
233 if (actions) {
234 catalog.action = [known = std::make_shared<std::map<std::string, ActionInfo, std::less<>> const>(
235 *actions)](std::string_view actionId) -> std::optional<ActionInfo> {
236 auto const found = known->find(actionId);
237 if (found == known->end()) {
238 return std::nullopt;
239 }
240 return found->second;
241 };
242 }
243 return catalog;
244 }
245};

281/// @brief The must-understand key (`!name`) of one object, if it has one.
282/// @param json The value; anything but an object has none.
283/// @param where What the object is, for the reason.
284/// @return Why the message is refused, or nothing.
285[[nodiscard]] inline std::optional<std::string> mustUnderstand(Json const& json, std::string_view where) {
286 auto const* const object = std::get_if<Json::object_t>(&json.data);
287 if (object == nullptr) {
288 return std::nullopt;
289 }
290 for (auto const& [key, value] : *object) {
291 static_cast<void>(value);
292 if (key.starts_with('!')) {
293 return "the must-understand key `" + key + "` in " + std::string{where} + " is not understood";
294 }
295 }
296 return std::nullopt;
297}
298
299/// @brief The must-understand keys of a message: its top level, every element of its listed arrays and every value
300/// of its listed maps.
301/// @param root The parsed message.
302/// @param what What the message is.
303/// @param arrays The keys of arrays whose elements are objects this layer defines.
304/// @param maps The keys of maps whose values are objects this layer defines.
305/// @return Why the message is refused, or nothing.
306// NOLINTBEGIN(bugprone-easily-swappable-parameters): the arrays' keys, then the maps', as a message lists them.
307[[nodiscard]] inline std::optional<std::string> mustUnderstandIn(Json const& root, std::string_view what,
308 std::initializer_list<std::string_view> arrays,
309 std::initializer_list<std::string_view> maps) {
310 if (auto reason = mustUnderstand(root, what)) {
311 return reason;
312 }
313 auto const* const object = std::get_if<Json::object_t>(&root.data);
314 if (object == nullptr) {
315 return std::nullopt;
316 }
317 for (std::string_view const key : arrays) {
318 auto const* const found = object->find(key);
319 auto const* const items = found == object->end() ? nullptr : std::get_if<Json::array_t>(&found->second.data);
320 if (items == nullptr) {
321 continue;
322 }
323 for (Json const& item : *items) {
324 if (auto reason = mustUnderstand(item, std::string{what} + " " + std::string{key})) {
325 return reason;
326 }
327 }
328 }
329 for (std::string_view const key : maps) {
330 auto const* const found = object->find(key);
331 auto const* const items = found == object->end() ? nullptr : std::get_if<Json::object_t>(&found->second.data);
332 if (items == nullptr) {
333 continue;
334 }
335 for (auto const& [name, item] : *items) {
336 if (auto reason = mustUnderstand(item, std::string{what} + " " + std::string{key} + " " + name)) {
337 return reason;
338 }
339 }
340 }
341 return std::nullopt;
342}
343// NOLINTEND(bugprone-easily-swappable-parameters)

345/// @brief Reads a message: bounded depth, must-understand keys refused, unknown keys ignored.
346/// @tparam T The message type.
347/// @param text The JSON text.
348/// @param what What the message is, for the reason.
349/// @param arrays The keys of arrays whose elements are objects this layer defines.
350/// @param maps The keys of maps whose values are objects this layer defines.
351/// @return The message, or why it was refused.
352template <typename T>
353[[nodiscard]] std::expected<T, std::string> readMessage(std::string_view text, std::string_view what,
354 std::initializer_list<std::string_view> arrays,
355 std::initializer_list<std::string_view> maps) {
356 auto const tree = parseJson(text);
357 if (!tree) {
358 return std::unexpected(std::string{what} + ": " + tree.error());
359 }
360 if (auto reason = mustUnderstandIn(*tree, what, arrays, maps)) {
361 return std::unexpected(std::move(*reason));
362 }
363 T value{};
364 if (auto const error = glz::read<glz::opts{.error_on_unknown_keys = false}>(value, text)) {
365 return std::unexpected(std::string{what} + ": " + glz::format_error(error, text));
366 }
367 return value;
368}

370/// @brief Writes a message as JSON, escaping control bytes as `wire::encode` does.
371/// @tparam T The message type.
372/// @param value The message.
373/// @return The JSON text.
374template <typename T>
375[[nodiscard]] std::string writeMessage(T const& value) {
376 std::string out;
377 // A reflected struct of strings, numbers, maps and generic JSON has no write error to report.
378 static_cast<void>(glz::write<::morph::wire::detail::EscapingWriteOpts{}>(value, out));
379 return out;
380}

382/// @brief Checks a name's length.
383/// @param name The name.
384/// @param what What it names, for the reason.
385/// @return Why it is refused, or nothing.
386// NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the name, then what it names, as the reason reads.
387[[nodiscard]] inline std::optional<std::string> checkName(std::string_view name, std::string_view what) {
388 if (name.size() > kMaxNameBytes) {
389 return std::string{what} + " is longer than " + std::to_string(kMaxNameBytes) + " bytes";
390 }
391 return std::nullopt;

394/// @brief Checks a list of names: its length and each name's.
395/// @param names The names.
396/// @param limit The most names allowed.
397/// @param list What the list is, for the reason.
398/// @param item What one name is, for the reason.
399/// @return Why the list is refused, or nothing.
400// NOLINTBEGIN(bugprone-easily-swappable-parameters): the list's description, then one item's, as the reasons read.
401[[nodiscard]] inline std::optional<std::string> checkNames(std::vector<std::string> const& names, std::size_t limit,
402 std::string_view list, std::string_view item) {
403 if (names.size() > limit) {
404 return std::string{list} + " list more than " + std::to_string(limit) + " items";
405 }
406 for (std::string const& name : names) {
407 if (auto reason = checkName(name, item)) {
408 return reason;
409 }
410 }
411 return std::nullopt;
412}

414/// @brief Checks a map of host names to versions: its size and each name's length.
415/// @param names The map.
416/// @param list What the map holds, for the reason.
417/// @param item What one name is, for the reason.
418/// @return Why the map is refused, or nothing.
419[[nodiscard]] inline std::optional<std::string> checkHostNames(

422 return std::string{list} + " list more than " + std::to_string(kMaxListItems) + " items";

427 return reason;
428 }
429 }
430 return std::nullopt;

441/// @brief Reads a manifest a server sent.
442///
443/// Refused: text that is not JSON or nests deeper than `kMaxJsonDepth`, a must-understand key at the top or in a
444/// file, more than `kMaxBundleFiles` files, a name longer than `kMaxNameBytes`, a malformed hash, a size above
445/// `kMaxFileBytes`, or two files of one kind with one id. Unknown keys are ignored.
446/// @param text The manifest's text, already verified against its digest.
447/// @return The manifest, or why it was refused.
448[[nodiscard]] inline std::expected<Manifest, std::string> parseManifest(std::string_view text) {
449 auto manifest = detail::readMessage<Manifest>(text, "the manifest", {"files"}, {});
450 if (!manifest) {
451 return std::unexpected(std::move(manifest.error()));
452 }
453 if (manifest->files.size() > kMaxBundleFiles) {
454 return std::unexpected("the manifest lists more than " + std::to_string(kMaxBundleFiles) + " files");
455 }
456 if (auto reason = detail::checkName(manifest->application, "the application id")) {
457 return std::unexpected(std::move(*reason));
458 }
459 if (auto reason = detail::checkName(manifest->version, "the application version")) {
460 return std::unexpected(std::move(*reason));
461 }
462 std::set<std::pair<std::string_view, std::string_view>> seen;
463 for (ManifestFile const& file : manifest->files) {
464 if (auto reason = detail::checkName(file.id, "a file id")) {
465 return std::unexpected(std::move(*reason));
466 }
467 if (auto reason = detail::checkName(file.kind, "a file kind")) {
468 return std::unexpected(std::move(*reason));
469 }
470 if (!isContentHash(file.hash)) {
471 return std::unexpected("the manifest file `" + file.id + "` has a malformed hash");
472 }
473 if (file.size > kMaxFileBytes) {
474 return std::unexpected("the manifest file `" + file.id + "` is larger than " +
475 std::to_string(kMaxFileBytes) + " bytes");
476 }
477 if (!seen.emplace(file.kind, file.id).second) {
478 return std::unexpected("the manifest lists the " + file.kind + " `" + file.id + "` twice");
479 }
480 }
481 return manifest;
482}

512 return std::unexpected(std::move(*reason));

543/// @brief Reads a `ui-catalog` reply body.
544///
545/// Refused: text that is not JSON or nests too deep, a must-understand key at the top, in an entry or in a model's
546/// or action's facts, more than `kMaxBundleFiles` entries, more than `kMaxCatalogFacts` permitted actions, model
547/// facts or action facts, or an entry id longer than `kMaxNameBytes`.
548/// @param text The reply body.
549/// @return The reply, or why it was refused.
550[[nodiscard]] inline std::expected<CatalogReply, std::string> parseCatalogReply(std::string_view text) {
551 auto reply = detail::readMessage<CatalogReply>(text, "the catalog", {"entries"}, {"models", "actions"});
552 if (!reply) {
553 return std::unexpected(std::move(reply.error()));
554 }
555 if (reply->entries.size() > kMaxBundleFiles) {
556 return std::unexpected("the catalog lists more than " + std::to_string(kMaxBundleFiles) + " entries");
557 }
558 std::size_t const models = reply->models ? reply->models->size() : 0;
559 std::size_t const actions = reply->actions ? reply->actions->size() : 0;
560 if (std::ranges::max({reply->permitted.size(), models, actions}) > kMaxCatalogFacts) {
561 return std::unexpected("the catalog carries more than " + std::to_string(kMaxCatalogFacts) +
562 " facts of a kind");
563 }
564 for (CatalogEntry const& entry : reply->entries) {
565 if (auto reason = detail::checkName(entry.id, "a catalog entry id")) {
566 return std::unexpected(std::move(*reason));
567 }
568 }
569 return reply;
570}

572/// @brief Reads a `ui-fetch` request body.
573///
574/// Refused: a body longer than `kMaxRequestBytes`, text that is not JSON or nests too deep, a must-understand key,
575/// or a hash that is not a content hash.
576/// @param text The request body.
577/// @return The request, or why it was refused.
578[[nodiscard]] inline std::expected<FetchRequest, std::string> parseFetchRequest(std::string_view text) {
579 if (text.size() > kMaxRequestBytes) {
580 return std::unexpected("the ui-fetch request is larger than " + std::to_string(kMaxRequestBytes) + " bytes");
581 }
582 auto request = detail::readMessage<FetchRequest>(text, "the ui-fetch request", {}, {});
583 if (!request) {
584 return std::unexpected(std::move(request.error()));
585 }
586 if (!isContentHash(request->hash)) {
587 return std::unexpected(std::string{"the ui-fetch request's hash is not a content hash"});
588 }
589 return request;
590}
