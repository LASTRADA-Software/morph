11/// @file
12/// The `collection` and `wizard` builders that carry their forms documents (spec 5 §11): a view descriptor's `v-*`
13/// document and a wizard's `w-*` document go into the node's catalog facts, so the bundle serves them with the
14/// `ui-catalog` reply and a client builds the node without knowing the types. Separate from `author.hpp` so the
15/// builders that need no forms documents do not pull in the forms headers.
16
17namespace morph::ui::author {
18
19/// @brief A view descriptor: a `views::` kind and the registered query action it lists.
20template <typename V>
21concept ViewDescriptor = requires {
22 typename V::kind;
23 typename V::query;
24};
25
26/// @brief A `flows::Wizard`: a title and its steps.
27template <typename W>
28concept WizardDescriptor = requires {
29 typename W::steps;
30 W::title();
31};
32
33/// @brief The action of a wizard's first step, which names the wizard in a document.
34/// @tparam W The wizard.
35template <WizardDescriptor W>
36using FirstStepAction = std::tuple_element_t<0, typename W::steps>::action;
37
38/// @brief A `collection` node over a view descriptor: the node names the view's query action, and the view's
39/// document travels with the node's catalog facts.
40/// @tparam View The view descriptor.
41/// @tparam Model The model type of the alias; it must run the view's query.
42/// @param model The alias.
43/// @param presentation The keys that reach every form the view draws; its title replaces the view's heading.
44/// @return The node.
45template <ViewDescriptor View, typename Model>
46 requires Runs<Model, typename View::query>
47[[nodiscard]] Node collection(ModelHandle<Model> const& model, FormPresentation const& presentation = {}) {
48 using Query = View::query;
49 CatalogFacts facts;
50 facts.action<Model, Query>();
51 facts.views.emplace(std::string{::morph::model::ActionTraits<Query>::typeId()},
52 ::morph::views::viewSchemaJson<View>());
53 detail::NodeWriter node{"collection"};
54 node.text("model", model.alias()).text("action", std::string{::morph::model::ActionTraits<Query>::typeId()});
55 detail::present(node, presentation);
56 return node.facts(facts).done();
57}
58
59/// @brief A `wizard` node over a `flows::Wizard`: the node names the first step's action, and the wizard's document
60/// travels with the node's catalog facts.
61/// @tparam Wizard The wizard.
62/// @tparam Model The model type of the alias; it must run the first step.
63/// @param model The alias.
64/// @param presentation The keys that reach every step's form; its title replaces the view's heading.
65/// @return The node.
66template <WizardDescriptor Wizard, typename Model>
67 requires Runs<Model, FirstStepAction<Wizard>>
68[[nodiscard]] Node wizard(ModelHandle<Model> const& model, FormPresentation const& presentation = {}) {
69 using First = FirstStepAction<Wizard>;
70 CatalogFacts facts;
71 facts.action<Model, First>();
72 facts.wizards.emplace(std::string{::morph::model::ActionTraits<First>::typeId()},
73 ::morph::flows::wizardSchemaJson<Wizard>());
74 detail::NodeWriter node{"wizard"};
75 node.text("model", model.alias()).text("action", std::string{::morph::model::ActionTraits<First>::typeId()});
76 detail::present(node, presentation);
77 return node.facts(facts).done();
78}
