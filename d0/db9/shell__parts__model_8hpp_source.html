42/// @brief What the model asks of its client.
43struct ShellModelOptions {
44 /// @brief A module's label in the navigator, for a search heading with no title of its own.
45 std::function<std::string(std::string_view module)> moduleLabel;
46 /// @brief The title of a screen in the catalog, for the inbox button.
47 std::function<std::string(std::string_view screen)> screenTitle;
48 /// @brief Whether the catalog lists a screen. A part that names one the catalog does not list is left out, not
49 /// reported: the authoriser hides it from this person. Empty: every screen is taken to be listed.
50 std::function<bool(std::string_view screen)> screenListed;
51 /// @brief Whether the catalog lists a module. A part that names one the catalog does not list is left out, not
52 /// reported: the authoriser hides it from this person. Empty: every module is taken to be listed.
53 std::function<bool(std::string_view module)> moduleListed;
54 /// @brief Whether the catalog says an action's reply holds a registered record, which is what makes a read of it
55 /// refresh when the record changes. Empty: every action is taken to.
56 std::function<bool(std::string_view action)> replyIsRecord;
57 /// @brief The locales the bundle has catalogues for, for a menu that lists none of its own.
58 std::function<std::vector<std::string>()> catalogueLocales;
59 /// @brief How many open tabs hold unsaved changes.
60 std::function<std::size_t()> dirtyTabs;
61 /// @brief Asks once whether to discard the unsaved work of @p tabs tabs; the answer comes to @p done.
62 std::function<void(std::size_t tabs, std::function<void(bool)> done)> confirmDiscard;
63 /// @brief Closes the tabs with unsaved changes without asking.
64 std::function<void()> closeDirtyTabs;
65 /// @brief Receives each problem once.
66 std::function<void(std::string const&)> onProblem;
67};

183 // A module or screen the catalog does not list is hidden from this person by the authoriser, so the part that
184 // names it is left out without a word. An unknown name is the server's refusal at load, not the client's report.
