115/// @brief What the model asks of its client.
116struct ShellModelOptions {
117 /// @brief A module's label in the navigator, for a search heading with no title of its own.
118 std::function<std::string(std::string_view module)> moduleLabel;
119 /// @brief The title of a screen in the catalog, for the inbox button.
120 std::function<std::string(std::string_view screen)> screenTitle;
121 /// @brief Whether the catalog lists a screen. A part that names one the catalog does not list is left out, not
122 /// reported: the authoriser hides it from this person. Empty: every screen is taken to be listed.
123 std::function<bool(std::string_view screen)> screenListed;
124 /// @brief Whether the catalog lists a module. A part that names one the catalog does not list is left out, not
125 /// reported: the authoriser hides it from this person. Empty: every module is taken to be listed.
126 std::function<bool(std::string_view module)> moduleListed;
127 /// @brief Whether the catalog says an action's reply holds a registered record, which is what makes a read of it
128 /// refresh when the record changes. Empty: every action is taken to.
129 std::function<bool(std::string_view action)> replyIsRecord;
130 /// @brief The locales the bundle has catalogues for, for a menu that lists none of its own.
131 std::function<std::vector<std::string>()> catalogueLocales;
132 /// @brief How many open tabs hold unsaved changes.
133 std::function<std::size_t()> dirtyTabs;
134 /// @brief Asks once whether to discard the unsaved work of @p tabs tabs; the answer comes to @p done.
135 std::function<void(std::size_t tabs, std::function<void(bool)> done)> confirmDiscard;
136 /// @brief Closes the tabs with unsaved changes without asking.
137 std::function<void()> closeDirtyTabs;
138 /// @brief Receives each problem once.
139 std::function<void(std::string const&)> onProblem;
140};

142/// @brief The declared parts of a shell, read from the app root.
143///
144/// Neither copyable nor movable; used on the owner thread of its session. It borrows the session, the app root and the
145/// parts, and must go before any of them.
146class ShellPartsModel {
147public:
148 /// @brief The model; it installs the `setContext` handler and starts following the switcher.
149 /// @param session The session. Borrowed.
150 /// @param root The app root whose queries the parts are. Borrowed.
151 /// @param parts The parts. Borrowed.
152 /// @param options What it asks of its client.
153 ShellPartsModel(Session& session MORPH_LIFETIMEBOUND, AppRoot& root MORPH_LIFETIMEBOUND,
154 ShellParts const& parts MORPH_LIFETIMEBOUND, ShellModelOptions options)
155 : _session{&session},
156 _root{&root},
157 _parts{&parts},
158 _options{std::move(options)},
159 _chosen{session.runtime(), Value{}},
160 _answered{session.runtime(), !parts.context.has_value()},
161 _searchText{session.runtime(), std::string{}},
162 _optionList{session.runtime(), std::vector<Option>{}},
163 _connection{session.runtime(), std::nullopt} {
164 _session->setContextHandler([this](Value const& value) { return requestContext(value); });
165 checkDeclaration();
166 if (_parts->context) {
167 _follow = std::make_unique<reactive::Effect>(_session->runtime(), [this] { followOptions(); });
168 }
169 }
170
171 ShellPartsModel(ShellPartsModel const&) = delete;
172 ShellPartsModel& operator=(ShellPartsModel const&) = delete;
173 ShellPartsModel(ShellPartsModel&&) = delete;
174 ShellPartsModel& operator=(ShellPartsModel&&) = delete;
175
176 /// @brief Takes the handler and the scope off the session.
177 ~ShellPartsModel() {
178 _follow.reset();
179 _session->setContextHandler({});
180 _session->setContextOffered(false);
181 _session->setContext(Value{});
182 }

209 /// @brief One module's count. A tracked read.
210 struct ModuleState {
211 /// @brief The module.
212 std::string module;
213 /// @brief The count; empty while unanswered.
214 std::optional<long long> count;
215 /// @brief Whether the read failed.
216 bool failed = false;
217 };
218
219 /// @brief Every declared module's count. A tracked read.
220 /// @return The counts, in declaration order; a denied one, and one of a module the catalog does not list, left
221 /// out.
222 [[nodiscard]] std::vector<ModuleState> moduleCounts() const {
223 std::vector<ModuleState> out;
224 for (std::size_t index = 0; index < _parts->modules.size(); ++index) {
225 ModuleCount const& declared = _parts->modules.at(index);
226 AppRoot::QueryRead const read = _root->query(shell_names::count(index));
227 if (read.denied || moduleHidden(declared.module)) {
228 continue;
229 }
230 ModuleState state{.module = declared.module, .count = std::nullopt, .failed = read.failed};
231 if (read.failed) {
232 problem("shell count " + shell_names::count(index) + ": the read failed");
233 }
234 if (!read.failed && !read.pending) {
235 state.count = countOf(read.value, declared.countField, shell_names::count(index));
236 state.failed = !state.count.has_value() && !read.value.isNull();
237 }
238 out.push_back(std::move(state));
239 }
240 return out;
241 }

252 /// @brief The `setContext` command: changes the scope to @p value if it is one of the options, asking once first
253 /// when open tabs hold unsaved changes.
254 /// @param value The option's value.
255 /// @return Whether the request was accepted: the value is an option's. The switch itself may wait for the answer.
256 bool requestContext(Value const& value) {
257 std::vector<Option> const& options = _optionList.peek();
258 auto const found =
259 std::ranges::find_if(options, [&value](Option const& option) { return option.value == value; });
260 if (found == options.end()) {
261 return false;
262 }
263 if (value == _chosen.peek()) {
264 return true;
265 }
266 std::size_t const dirty = _options.dirtyTabs ? _options.dirtyTabs() : 0;
267 if (dirty == 0 || !_options.confirmDiscard) {
268 apply(value);
269 return true;
270 }
271 _options.confirmDiscard(dirty, [this, value, alive = std::weak_ptr<bool>{_alive}](bool discard) {
272 if (alive.expired() || !discard) {
273 return;
274 }
275 if (_options.closeDirtyTabs) {
276 _options.closeDirtyTabs();
277 }
278 apply(value);
279 });
280 return true;
281 }

389 // What the catalog can say about the declaration, reported once at the start: a count whose reply is not a
390 // registered record (it is then read on mount and on a context change only).
391 void checkDeclaration() const {
392 auto const record = [this](ModuleCount const& count, std::string const& part) {
393 if (_options.replyIsRecord && !_options.replyIsRecord(count.action)) {
394 problem("shell " + part + ": the reply of " + count.action +
395 " is not a registered record, so the count is read on mount and on a context change only");
396 }
397 };
398 for (std::size_t index = 0; index < _parts->modules.size(); ++index) {
399 if (!moduleHidden(_parts->modules.at(index).module)) {
400 record(_parts->modules.at(index), "count " + shell_names::count(index));
401 }
402 }
403 if (_parts->inbox && !screenHidden(_parts->inbox->screen)) {
404 for (std::size_t index = 0; index < _parts->inbox->sources.size(); ++index) {
405 record(_parts->inbox->sources.at(index), "inbox " + shell_names::inbox(index));
406 }
407 }
408 }

417 // A reply value as the text a widget shows: a string as itself, a typed value in the locale, a message reference
418 // through the catalogue.
419 [[nodiscard]] static std::string describe(Value const& value, FormatContext const& format) {
420 if (ValueRecord const* const record = value.asRecord()) {
421 auto const key = record->find("t");
422 if (key != record->end() && key->second.asString() != nullptr) {
423 std::string pattern = *key->second.asString();
424 bool translated = false;
425 if (format.translations) {
426 if (auto const text = format.translations(pattern, format.locale)) {
427 pattern = *text;
428 translated = true;
429 }
430 }
431 if (auto const fallback = record->find("fallback");
432 !translated && fallback != record->end() && fallback->second.asString() != nullptr) {
433 pattern = *fallback->second.asString();
434 }
435 if (auto const args = record->find("args"); args != record->end()) {
436 if (ValueRecord const* const named = args->second.asRecord()) {
437 for (auto const& [name, argument] : *named) {
438 std::string const token = "{$" + name + "}";
439 for (std::size_t at = pattern.find(token); at != std::string::npos;
440 at = pattern.find(token, at)) {
441 std::string const shown = detail::textOf(argument, format);
442 pattern.replace(at, token.size(), shown);
443 at += shown.size();
444 }
445 }
446 }
447 }
448 return pattern;
449 }
450 }
451 return detail::textOf(value, format);
452 }

454 [[nodiscard]] static std::string translate(std::string const& textOrKey, FormatContext const& format) {
455 if (!textOrKey.empty() && format.translations) {
456 if (auto const text = format.translations(textOrKey, format.locale)) {
457 return *text;
458 }
459 }
460 return textOrKey;
461 }

463 [[nodiscard]] static std::string memberText(Value const& record, std::string const& name,
464 FormatContext const& format) {
465 if (name.empty()) {
466 return {};
467 }
468 Value const* const found = record.field(name);
469 return found == nullptr ? std::string{} : describe(*found, format);

472 // A count is a JSON integer, or the wire form of a `morph::units::Quantity`, whose magnitude is a whole number: a
473 // `Quantity` is written as its exact value (`num`, `den`, `dp`), which a client reads as a decimal, and with a
474 // `unit` as a quantity. The unit is the application's and is not read. Anything else, and a negative number, is
475 // not a count.
476 [[nodiscard]] std::optional<long long> countOf(Value const& reply, std::string const& field,
477 std::string const& query) const {
478 Value const* const found = reply.field(field);
479 std::optional<std::int64_t> number;
480 if (found != nullptr) {
481 if (std::int64_t const* const plain = found->asInt()) {
482 number = *plain;
483 } else if (Decimal const* const exact = found->asDecimal();
484 exact != nullptr && exact->value.denominator == 1) {
485 number = exact->value.numerator;
486 } else if (Quantity const* const quantity = found->asQuantity();
487 quantity != nullptr && quantity->value.denominator == 1) {
488 number = quantity->value.numerator;
489 }
490 }
491 if (!number || *number < 0) {
492 if (!reply.isNull()) {
493 problem("shell count " + query + ": the reply has no non-negative integer `" + field + "`");
494 }
495 return std::nullopt;
496 }
497 return static_cast<long long>(*number);
498 }

500 // The switcher's options, from the last answer.
501 [[nodiscard]] std::vector<Option> parseOptions(AppRoot::QueryRead const& read, FormatContext const& format) {
502 std::vector<Option> out;
503 ContextSwitch const& declared = *_parts->context;
504 Value const* const list = read.value.field(declared.optionsField);
505 ValueList const* const items = list == nullptr ? nullptr : list->asList();
506 if (items == nullptr) {
507 if (!read.value.isNull()) {
508 problem("shell context: the reply has no list `" + declared.optionsField + "`");
509 }
510 return out;
511 }
512 for (Value const& item : *items) {
513 Value const* const value = item.field(declared.valueField);
514 if (value == nullptr || (value->asString() == nullptr && value->asInt() == nullptr)) {
515 problem("shell context: an option has no string or integer `" + declared.valueField + "`");
516 continue;
517 }
518 out.push_back(Option{.value = *value,
519 .shown = ShellOptionView{.label = memberText(item, declared.labelField, format),
520 .detail = memberText(item, declared.detailField, format),
521 .group = memberText(item, declared.groupField, format)}});
522 }
523 return out;
524 }

526 // Keeps the chosen scope among the options: the first option when it is not, none when there are none.
527 void followOptions() {
528 AppRoot::QueryRead const read = _root->query(shell_names::kContext);
529 FormatContext const& format = _session->formatting();
530 std::vector<Option> options = parseOptions(read, format);
531 bool const settled = (!read.pending && !read.value.isNull()) || read.failed || read.denied;
532 if (read.failed && !read.denied) {
533 problem("shell context: the scope switcher's read failed");
534 }
535 _session->runtime().untracked([&] {
536 _optionList.set(std::move(options));
537 std::vector<Option> const& held = _optionList.peek();
538 _session->setContextOffered(!held.empty());
539 if (held.empty()) {
540 if (!_chosen.peek().isNull()) {
541 apply(Value{});
542 }
543 } else {
544 bool const present =
545 std::ranges::any_of(held, [this](Option const& option) { return option.value == _chosen.peek(); });
546 if (!present) {
547 apply(held.front().value);
548 }
549 }
550 if (settled && !_answered.peek()) {
551 _answered.set(true);
552 }
553 });
554 }
555
556 void apply(Value const& value) {
557 _chosen.set(value);
558 _session->setContext(value);
559 }

561 [[nodiscard]] ShellContextView contextView(FormatContext const& format) const {
562 ShellContextView out;
563 if (!_parts->context) {
564 return out;
565 }
566 // A switcher the server refuses is the authoriser working: it is left out.
567 if (_root->query(shell_names::kContext).denied) {
568 return out;
569 }
570 Value const chosen = _chosen.get();
571 std::vector<Option> const& options = _optionList.get();
572 out.shown = true;
573 for (std::size_t index = 0; index < options.size(); ++index) {
574 out.options.push_back(options.at(index).shown);
575 if (options.at(index).value == chosen) {
576 out.chosen = index;
577 }
578 }
579 out.enabled = !out.options.empty();
580 static_cast<void>(format);
581 return out;
582 }

590 [[nodiscard]] std::vector<Target> sourceHits(std::size_t source) const {
591 std::vector<Target> out;
592 if (source >= _parts->search.size() || _searchText.peek().empty()) {
593 return out;
594 }
595 SearchSource const& declared = _parts->search.at(source);
596 AppRoot::QueryRead const read = _root->query(shell_names::search(source));
597 // The hits of a source the view does not draw are not the person's to open.
598 if (read.denied || read.failed || moduleHidden(declared.module)) {
599 return out;
600 }
601 Value const* const list = read.value.field(declared.resultsField);
602 ValueList const* const items = list == nullptr ? nullptr : list->asList();
603 if (items == nullptr) {
604 return out;
605 }
606 for (Value const& item : *items) {
607 if (!isHit(item) || out.size() >= kSearchHitsPerSource) {
608 continue;
609 }
610 Value const* const screen = item.field("screen");
611 Target target{.screen = *screen->asString(), .params = {}};
612 if (Value const* const params = item.field("params"); params != nullptr && params->asRecord() != nullptr) {
613 target.params = *params->asRecord();
614 }
615 out.push_back(std::move(target));
616 }
617 return out;

620 [[nodiscard]] ShellSearchView searchView(FormatContext const& format) const {
621 ShellSearchView out;
622 if (_parts->search.empty()) {
623 return out;
624 }
625 out.shown = true;
626 out.hint = translate(_parts->searchHint, format);
627 if (_searchText.get().empty()) {
628 return out;
629 }
630 for (std::size_t source = 0; source < _parts->search.size(); ++source) {
631 SearchSource const& declared = _parts->search.at(source);
632 AppRoot::QueryRead const read = _root->query(shell_names::search(source));
633 if (read.denied || moduleHidden(declared.module)) {
634 continue;
635 }
636 ShellSourceView shown;
637 shown.source = source;
638 std::string title = translate(declared.title, format);
639 if (title.empty() && _options.moduleLabel) {
640 title = _options.moduleLabel(declared.module);
641 }
642 shown.title = title.empty() ? declared.module : title;
643 shown.pending = read.pending;
644 shown.failed = read.failed;
645 std::string const query = shell_names::search(source);
646 if (read.failed) {
647 problem("shell search " + query + ": the read failed");
648 } else {
649 Value const* const list = read.value.field(declared.resultsField);
650 ValueList const* const items = list == nullptr ? nullptr : list->asList();
651 if (items == nullptr && !read.value.isNull()) {
652 problem("shell search " + query + ": the reply has no list `" + declared.resultsField + "`");
653 }
654 for (Value const& item : items == nullptr ? ValueList{} : *items) {
655 if (!isHit(item)) {
656 problem("shell search " + query +
657 ": a hit has no string `screen` or no `title` and is dropped");
658 continue;
659 }
660 if (shown.hits.size() < kSearchHitsPerSource) {
661 shown.hits.push_back(ShellHitView{.kind = memberText(item, "kind", format),
662 .title = describe(*item.field("title"), format),
663 .detail = memberText(item, "detail", format)});
664 }
665 }
666 }
667 out.sources.push_back(std::move(shown));
668 }
669 return out;
670 }

672 [[nodiscard]] ShellInboxView inboxView(FormatContext const& format) const {
673 ShellInboxView out;
674 if (!_parts->inbox || screenHidden(_parts->inbox->screen)) {
675 return out;
676 }
677 static_cast<void>(format);
678 long long sum = 0;
679 bool answered = false;
680 bool permitted = false;
681 for (std::size_t index = 0; index < _parts->inbox->sources.size(); ++index) {
682 ModuleCount const& declared = _parts->inbox->sources.at(index);

684 if (read.denied) {
685 continue;
686 }
687 permitted = true;

689 out.failed = true;

691 } else if (!read.pending) {
692 if (auto const count = countOf(read.value, declared.countField, shell_names::inbox(index))) {
693 sum += *count;
694 answered = true;
695 } else if (!read.value.isNull()) {
696 out.failed = true;
697 }
698 }
699 }
700 out.shown = permitted;
701 out.title = _options.screenTitle ? _options.screenTitle(_parts->inbox->screen) : _parts->inbox->screen;
702 if (answered) {
703 out.count = sum;
704 }
705 return out;

760 problem("shell principal: the reply has no `" + declared.nameField + "`");

771 .detail = memberText(role, "detail", format)});

773 out.roles.push_back(ShellRoleView{.label = describe(role, format), .detail = {}});
774 }

782 if (list.back().label == "morph.shell.locale." + tag) {
783 list.back().label = tag;
784 }
785 }
786 return list;
787 };
788 out.locales = choices(localeTags());
789 for (std::size_t index = 0; index < out.locales.size(); ++index) {
790 if (out.locales.at(index).tag == format.locale) {
791 out.chosenLocale = index;
792 }

795 std::string const chosenFormat = _session->formatLocale();
796 for (std::size_t index = 0; index < out.formats.size(); ++index) {
797 if (out.formats.at(index).tag == chosenFormat) {
798 out.chosenFormat = index;
799 }
