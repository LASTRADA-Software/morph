30/// @brief Replacements for the parts of a collection view a host draws from its own UI kit. Each one left empty
31/// keeps the built-in part.
32struct CollectionChrome {
33 /// @brief Replaces the title and the collection actions' buttons. Called once, with the session, whose model
34 /// gives the title, columns and actions and whose `fire` runs one.
35 std::function<ui::Node(CollectionSession&)> header;
36 /// @brief Replaces each row's cells and buttons: the rows are then a list of these nodes rather than a table.
37 /// Called once per row mounted, with the session and the row's signal (read it inside bindings).
38 std::function<ui::Node(CollectionSession&, reactive::Signal<CollectionRow> const&)> row;
39 /// @brief Replaces the confirm dialog: mounted while an action waits for confirmation, unmounted once it is
40 /// answered. Called with the session, whose `pendingConfirm`, `confirm` and `cancelConfirm` it uses.
41 std::function<ui::Node(CollectionSession&)> confirm;
42 /// @brief Replaces the editor dialog of a `collection` view; it is always mounted, and shows the editor while
43 /// `openRow()` is engaged. Called once with the session and the editor's form view, which it places.
44 /// A master-detail view has no dialog and does not call it.
45 std::function<ui::Node(CollectionSession&, ui::Node editor)> editor;
46};
47
48/// @brief How `collectionView` draws a collection and labels its chrome.
49struct CollectionViewOptions {
50 /// @brief The editor's form view options; its overrides reach the editor's fields.
51 FormViewOptions form{};
52 /// @brief The host's replacements for the view's own parts.
53 CollectionChrome chrome{};
54 /// @brief The caption of a row's Open button.
55 std::string openLabel = "Open";
56 /// @brief The confirm dialog's title.
57 std::string confirmTitle = "Confirm";
58 /// @brief The confirm dialog's question.
59 std::string confirmMessage = "Are you sure?";
60 /// @brief The confirm dialog's accept button.
61 std::string acceptLabel = "OK";
62 /// @brief The confirm dialog's cancel button.
63 std::string cancelLabel = "Cancel";
64};

68/// @brief The error line of a collection: the list's failure, else the last action's.
69/// @param session The session.
70/// @return `error: <message>`, or empty.
71[[nodiscard]] inline std::string collectionErrorText(CollectionSession const& session) {
72 std::exception_ptr error = session.listError();
73 if (error == nullptr) {
74 error = session.actionError();
75 }
76 return error == nullptr ? std::string{} : "error: " + reactive::errorMessage(error);
77}

79/// @brief The indices of a collection's actions with one scope, in document order.
80/// @param model The model.
81/// @param scope The scope.
82/// @return The indices into `model.actions()`.
83[[nodiscard]] inline std::vector<std::size_t> actionsOf(CollectionModel const& model, CollectionActionScope scope) {
84 std::vector<std::size_t> indices;
85 for (std::size_t i = 0; i < model.actions().size(); ++i) {
86 if (elementAt(model.actions(), i).scope == scope) {
87 indices.push_back(i);
88 }
89 }
90 return indices;
91}

93/// @brief The built-in header: the title, a busy indicator while the list loads, and the collection actions.
94/// @param session The session.
95/// @return The header.
96[[nodiscard]] inline ui::Node collectionHeader(CollectionSession& session) {
97 CollectionSession* const view = &session;
98 ui::Text title{.text = std::string{session.model().title()}, .role = ui::TextRole::Heading};
99 title.common.testId = "collection_title";
100 std::vector<ui::Node> children{
101 ui::text(std::move(title)),
102 ui::busy(ui::Busy{.active = [view] { return view->loading(); }, .label = std::string{}}),
103 };
104 for (std::size_t const index : actionsOf(session.model(), CollectionActionScope::Collection)) {
105 CollectionAction const& action = elementAt(session.model().actions(), index);
106 ui::Button button{.label = action.label, .onClick = [view, index] { view->fire(index); }};
107 button.common.enabled = [view] { return !view->actionPending(); };
108 button.common.testId = "collectionAction_" + action.action;
109 children.push_back(ui::button(std::move(button)));
110 }
111 return ui::row(ui::Row{.children = std::move(children), .gap = 4});
112}

114/// @brief A row's built-in buttons: Open (with a row action) and each row action.
115/// @param session The session.
116/// @param rowKey The row's key text.
117/// @param options The view options.
118/// @return The buttons, in a row.
119[[nodiscard]] inline ui::Node rowButtons(CollectionSession& session, std::string const& rowKey,
120 CollectionViewOptions const& options) {
121 CollectionSession* const view = &session;
122 std::vector<ui::Node> buttons;
123 if (session.model().rowAction().has_value()) {
124 ui::Button open{.label = options.openLabel, .onClick = [view, rowKey] { view->open(rowKey); }};
125 open.common.enabled = [view] { return view->editor() != nullptr; };
126 open.common.testId = "rowOpen_" + rowKey;
127 buttons.push_back(ui::button(std::move(open)));
128 }
129 for (std::size_t const index : actionsOf(session.model(), CollectionActionScope::Row)) {
130 CollectionAction const& action = elementAt(session.model().actions(), index);
131 ui::Button button{.label = action.label, .onClick = [view, index, rowKey] { view->fire(index, rowKey); }};
132 button.common.enabled = [view] { return !view->actionPending(); };
133 button.common.testId = "rowAction_" + action.action + "_" + rowKey;
134 buttons.push_back(ui::button(std::move(button)));
135 }
136 return ui::row(ui::Row{.children = std::move(buttons), .gap = 4});
137}

139/// @brief The rows table's activation and selection: activating a row opens it when the view has a row action; in a
140/// master-detail view a row is selected while the editor is open on it, and selecting one opens it.
141/// @param session The session.
142/// @return The options, with the table's test id.
143[[nodiscard]] inline ui::TableOptions rowsTableOptions(CollectionSession& session) {
144 CollectionSession* const view = &session;
145 ui::TableOptions table{};
146 if (session.model().rowAction().has_value()) {
147 table.onActivate = [view](ui::Key const& key) {
148 if (auto const* const text = std::get_if<std::string>(&key)) {
149 view->open(*text);
150 }
151 };
152 }
153 if (session.model().kind() == CollectionKind::MasterDetail) {
154 table.selectionMode = ui::SelectionMode::Single;
155 table.selection = [view] {
156 std::optional<std::string> const& open = view->openRow();
157 return open.has_value() ? std::vector<ui::Key>{ui::Key{*open}} : std::vector<ui::Key>{};
158 };
159 table.onSelectionChange = [view](std::vector<ui::Key> const& keys) {
160 auto const* const text = keys.empty() ? nullptr : std::get_if<std::string>(&keys.front());
161 if (text == nullptr) {
162 view->close();
163 } else {
164 view->open(*text);
165 }
166 };
167 }
168 table.common.testId = "collection_rows";
169 return table;
170}

172/// @brief The rows: a table of the visible columns' cells and the row buttons, or a list of the row chrome's nodes.
173/// In a master-detail view a row is selected while the editor is open on it, and selecting one opens it.
174/// @param session The session.
175/// @param options The view options, shared with the row views.
176/// @return The rows.
177[[nodiscard]] inline ui::Node collectionRows(CollectionSession& session,
178 std::shared_ptr<CollectionViewOptions const> const& options) {
179 CollectionSession* const view = &session;
180 auto const rows = [view] { return view->rows(); };
181 auto const keyOf = [](CollectionRow const& row) { return ui::Key{row.key}; };
182 if (options->chrome.row) {
183 ui::Node list = ui::forEach<CollectionRow>(
184 rows, keyOf,
185 [view, options](reactive::Signal<CollectionRow> const& slot) { return options->chrome.row(*view, slot); });
186 return list;
187 }
188 std::vector<ui::TableColumn> columns;
189 std::vector<std::string> fields;
190 for (CollectionColumn const& column : session.model().columns()) {
191 if (!column.hidden) {
192 columns.push_back(ui::TableColumn{.label = column.label});
193 fields.push_back(column.field);
194 }
195 }
196 bool const buttons =
197 session.model().rowAction().has_value() || !actionsOf(session.model(), CollectionActionScope::Row).empty();
198 if (buttons) {
199 columns.push_back(ui::TableColumn{.label = std::string{}});
200 }
201 ui::TableOptions table = rowsTableOptions(session);
202 return ui::table<CollectionRow>(
203 std::move(columns), rows, keyOf,
204 [view, options, fields, buttons](reactive::Signal<CollectionRow> const& slot) {
205 reactive::Signal<CollectionRow> const* const row = &slot;
206 std::string const key = slot.peek().key;
207 std::vector<ui::Node> cells;
208 for (std::size_t i = 0; i < fields.size(); ++i) {
209 ui::Text cell{.text = [row, i] {
210 std::vector<std::string> const& texts = row->get().cells;
211 return i < texts.size() ? texts.at(i) : std::string{};
212 }};
213 cell.common.testId = "cell_" + fields.at(i) + "_" + key;
214 cells.push_back(ui::text(std::move(cell)));
215 }
216 if (buttons) {
217 cells.push_back(rowButtons(*view, key, *options));
218 }
219 return cells;
220 },
221 std::move(table));
222}

224/// @brief The built-in confirm dialog: open while an action waits, its question and the accept and cancel buttons.
225/// @param session The session.
226/// @param options The view options.
227/// @return The dialog.
228[[nodiscard]] inline ui::Node confirmDialog(CollectionSession& session, CollectionViewOptions const& options) {
229 CollectionSession* const view = &session;
230 ui::Button accept{.label = options.acceptLabel, .onClick = [view] { view->confirm(); }};
231 accept.common.testId = "confirm_accept";
232 ui::Button cancel{.label = options.cancelLabel, .onClick = [view] { view->cancelConfirm(); }};
233 cancel.common.testId = "confirm_cancel";
234 ui::Text message{.text = options.confirmMessage};
235 message.common.testId = "confirm_message";
236 ui::Dialog dialog{
237 .open = [view] { return view->pendingConfirm().has_value(); },
238 .title = options.confirmTitle,
239 .child = ui::column(ui::Column{
240 .children = {ui::text(std::move(message)),
241 ui::row(ui::Row{.children = {ui::button(std::move(accept)), ui::button(std::move(cancel))},
242 .gap = 4})},
243 .gap = 4}),
244 .onDismiss = [view] { view->cancelConfirm(); },
245 };
246 dialog.common.testId = "confirmDialog";
247 return ui::dialog(std::move(dialog));
248}

250/// @brief The confirm chrome, mounted only while an action waits.
251/// @param session The session.
252/// @param chrome The host's confirm chrome.
253/// @return A switch showing it while `pendingConfirm()` is engaged.
254[[nodiscard]] inline ui::Node confirmChrome(CollectionSession& session,
255 std::function<ui::Node(CollectionSession&)> const& chrome) {
256 CollectionSession const* const view = &session;
257 return ui::switchOf(ui::Switch{
258 .selector = [view] { return ui::Key{std::int64_t{view->pendingConfirm().has_value() ? 1 : 0}}; },
259 .cases = {ui::SwitchCase{.key = ui::Key{std::int64_t{1}}, .node = chrome(session)}},
260 });
261}

265/// @brief Renders @p session: the header (title, busy indicator, collection actions), the error line, the rows as a
266/// table (visible columns' exact cells, then Open and the row actions), the confirm dialog, and the row
267/// editor (`formView` of `editor()`) as a dialog for a `collection`, or beside the table, shown while a row
268/// is open, for a `master-detail` view.
269///
270/// Test ids: `collection_title`, `collectionAction_<Action>`, `collection_error`, `collection_rows`,
271/// `cell_<field>_<rowKey>`, `rowOpen_<rowKey>`, `rowAction_<Action>_<rowKey>`, `confirmDialog`, `confirm_message`,
272/// `confirm_accept`, `confirm_cancel`, `editorDialog` and `detailForm`; the editor's fields have their form ids.
273///
274/// The nodes bind to the session by pointer: mount the result after the session and destroy the mount before it.
275/// @param session The session.
276/// @param options The editor's form options, the chrome and the labels.
277/// @return The view.
278[[nodiscard]] inline ui::Node collectionView(CollectionSession& session, CollectionViewOptions options = {}) {
279 auto const shared = std::make_shared<CollectionViewOptions const>(std::move(options));
280 CollectionSession* const view = &session;
281 std::vector<ui::Node> list;
282 list.push_back(shared->chrome.header ? shared->chrome.header(session) : detail::engine::collectionHeader(session));
283 ui::Text error{.text = [view] { return detail::engine::collectionErrorText(*view); }, .role = ui::TextRole::Error};
284 error.common.visible = [view] { return view->listError() != nullptr || view->actionError() != nullptr; };
285 error.common.testId = "collection_error";
286 list.push_back(ui::text(std::move(error)));
287 list.push_back(detail::engine::collectionRows(session, shared));
288 list.push_back(shared->chrome.confirm ? detail::engine::confirmChrome(session, shared->chrome.confirm)
289 : detail::engine::confirmDialog(session, *shared));
290 FormSession* const editor = session.editor();
291 bool const masterDetail = session.model().kind() == CollectionKind::MasterDetail;
292 if (editor != nullptr && !masterDetail) {
293 ui::Node form = formView(*editor, shared->form);
294 if (shared->chrome.editor) {
295 list.push_back(shared->chrome.editor(session, std::move(form)));
296 } else {
297 ui::Dialog dialog{
298 .open = [view] { return view->openRow().has_value(); },
299 // NOLINTNEXTLINE(bugprone-unchecked-optional-access): an editor exists only for a row action.
300 .title = session.model().rowAction().value().action,
301 .child = std::move(form),
302 .onDismiss = [view] { view->close(); },
303 };
304 dialog.common.testId = "editorDialog";
305 list.push_back(ui::dialog(std::move(dialog)));
306 }
307 }
308 ui::Node column = ui::column(ui::Column{.children = std::move(list), .gap = 4});
309 if (editor == nullptr || !masterDetail) {
310 return column;
311 }
312 ui::Column detail{.children = {formView(*editor, shared->form)}};
313 detail.common.visible = [view] { return view->openRow().has_value(); };
314 detail.common.testId = "detailForm";
315 return ui::row(ui::Row{.children = {std::move(column), ui::column(std::move(detail))}, .gap = 5});
316}
