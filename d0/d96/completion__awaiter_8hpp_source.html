21/// @file
22/// @brief The awaiter behind `Completion<T>::operator co_await`.
23///
24/// A fragment of `completion.hpp` rather than of `coroutine.hpp`, so a
25/// `Completion<T>` is awaitable wherever it is visible. Specified in
26/// `docs/spec/core/coroutines.md`.
27
28namespace morph::async::detail {
29
30template <typename T>
31struct CompletionState;
32
33/// @brief The awaiter `Completion<T>::operator co_await() &&` returns.
34///
35/// The await is one more `then`/`onError` pair on the completion, holding only
36/// a heap `Shared` -- never the coroutine frame -- and gated by a
37/// `CallbackToken`. With a stoppable token on the awaiting promise it also
38/// registers a stop callback. Exactly one of *the completion's handler* and
39/// *the stop callback* moves `Shared::outcome` off `Pending`, and only that one
40/// resumes the coroutine.
41///
42/// `await_suspend` touches only local copies of the shared pointers once it has
43/// published anything another thread could act on: from then on the coroutine
44/// may already have been resumed elsewhere, and the awaiter with its frame
45/// destroyed.
46/// @tparam T The completion's value type; copied out of the settled state.
47template <typename T>
48class CompletionAwaiter {
49 enum class Outcome : std::uint8_t { Pending, Settled, Cancelled };
50
51 struct Shared;
52
53 /// The stop callback's callable: forwards to `Shared::onStop`.
54 struct OnStop {
55 Shared* shared;
56 void operator()() const noexcept { shared->onStop(); }
57 };
58
59 struct Shared {
60 std::atomic<Outcome> outcome{Outcome::Pending};
61 /// Set once `await_suspend` has finished registering; before that a
62 /// stop only records itself and `await_suspend` answers it.
63 std::atomic<bool> armed{false};
64 /// Taken by whichever of `await_suspend` and `onStop` answers a stop.
65 std::atomic<bool> claimed{false};
66 std::optional<T> value;
67 std::exception_ptr error;
68 std::coroutine_handle<> continuation;
69 /// Where the coroutine was running when it suspended: resumed there.
70 /// Empty outside every executor's task.
71 ::core::async::ResumeTarget context;
72 ::morph::exec::IExecutor* fallback = nullptr;
73 CallbackScope scope;
74 std::optional<::core::async::StopCallback<OnStop>> stopCallback;
75
76 // seq_cst, as `armed` is: whoever decides and then reads `armed` must
77 // not miss the other side's store (a store-buffering pattern).
78 bool decide(Outcome outcome_) noexcept {
79 auto expected = Outcome::Pending;
80 return outcome.compare_exchange_strong(expected, outcome_, std::memory_order_seq_cst);
81 }
82
83 /// A settled completion, on the completion's executor: resumes in the
84 /// suspending context, or right here.
85 void resumeSettled() {
86 if (context) {
87 context.submit(::core::async::ParkedWork{.resume = continuation});
88 } else {
89 continuation.resume();
90 }
91 }
92
93 /// A stop, on whichever thread requested it: resumes through the
94 /// suspending context, or on the completion's executor. Not inline
95 /// here, because the requesting thread is not the coroutine's; the
96 /// context itself may resume inline (a Task handler's resumer whose
97 /// strands are closed does).
98 void resumeCancelled() {
99 if (context) {
100 context.submit(::core::async::ParkedWork{.resume = continuation});
101 } else {
102 fallback->post([handle = continuation] { handle.resume(); });
103 }
104 }
105
106 void onStop() noexcept {
107 if (!decide(Outcome::Cancelled)) {
108 return;
109 }
110 scope.requestStop();
111 if (armed.load(std::memory_order_seq_cst) && !claimed.exchange(true, std::memory_order_acq_rel)) {
112 try {
113 resumeCancelled();
114 } catch (...) { // NOLINT(bugprone-empty-catch): a stop callback must not throw
115 // A queue that cannot accept one more closure leaves nothing to
116 // resume through; the coroutine stays suspended until its
117 // owner destroys it.
118 }
119 }
120 }
121 };
122
123public:
124 /// @param state The completion's state; consumed. Null for an empty completion.
125 explicit CompletionAwaiter(std::shared_ptr<CompletionState<T>> state)
126 : _state{std::move(state)}, _shared{std::make_shared<Shared>()} {}
127
128 /// Withdraws the await if the frame is destroyed while suspended, so neither
129 /// a later settlement nor a later stop resumes a frame that is gone.
130 ~CompletionAwaiter() {
131 _shared->stopCallback.reset();
132 _shared->scope.requestStop();
133 }
134
135 CompletionAwaiter(const CompletionAwaiter&) = delete;
136 CompletionAwaiter& operator=(const CompletionAwaiter&) = delete;
137 CompletionAwaiter(CompletionAwaiter&&) = delete;
138 CompletionAwaiter& operator=(CompletionAwaiter&&) = delete;

140 /// @return Always false: every decision is `await_suspend`'s, so this stays
141 /// a constant (see core-cpp's `awaitReadyIsConstantFalse`).
142 [[nodiscard]] constexpr bool await_ready() const noexcept { return false; }
143
144 /// @tparam Promise The awaiting coroutine's promise type.
145 /// @param awaiting The coroutine performing the `co_await`.
146 /// @return False to continue at once (an unusable completion, or a stop
147 /// already requested); true once the await is attached.
148 template <typename Promise>
149 bool await_suspend(std::coroutine_handle<Promise> awaiting) {
150 auto shared = _shared;
151 auto state = _state;
152 if (state == nullptr || state->cbExec == nullptr) {
153 shared->error = std::make_exception_ptr(std::logic_error{
154 state == nullptr ? "co_await on an empty morph::async::Completion (default-constructed or moved-from)"
155 : "co_await on a morph::async::Completion with no callback executor to resume on"});
156 shared->outcome.store(Outcome::Settled);
157 return false;
158 }
159 shared->continuation = awaiting;
160 shared->context = ::core::async::ResumeTarget::current();
161 shared->fallback = state->cbExec;
162
163 if constexpr (::core::async::HasStopToken<Promise>) {
164 ::core::async::StopToken const token = awaiting.promise().stopToken();
165 if (token.stop_requested()) {
166 shared->outcome.store(Outcome::Cancelled);
167 return false;
168 }
169 if (token.stop_possible()) {
170 shared->stopCallback.emplace(token, OnStop{shared.get()});
171 shared->armed.store(true, std::memory_order_seq_cst);
172 if (shared->outcome.load(std::memory_order_seq_cst) == Outcome::Cancelled) {
173 // A stop landed while registering. Whoever claims answers
174 // it; if onStop claimed first it is resuming us already.
175 return shared->claimed.exchange(true, std::memory_order_acq_rel);
176 }
177 }
178 }
179
180 auto const token = shared->scope.token();
181 try {
182 state->attachThen([token, shared](const T& settled) {
183 if (token.active() && shared->decide(Outcome::Settled)) {
184 shared->value.emplace(settled);
185 shared->resumeSettled();
186 }
187 });
188 state->attachOnError([token, shared](std::exception_ptr error) {
189 if (token.active() && shared->decide(Outcome::Settled)) {
190 shared->error = std::move(error);
191 shared->resumeSettled();
192 }
193 });
194 } catch (...) {
195 // The throw resumes the coroutine, so nothing else may. If a stop
196 // has claimed the await it is resuming the coroutine already, and
197 // the frame is no longer this call's to throw into: that stop's
198 // OperationCancelled is what the coroutine sees.
199 if (shared->claimed.exchange(true, std::memory_order_acq_rel)) {
200 shared->scope.requestStop();
201 return true;
202 }
203 // Likewise if the handler attached before the throw has already
204 // settled the await: it resumed the coroutine.
205 if (!shared->decide(Outcome::Settled) && shared->outcome.load() == Outcome::Settled) {
206 shared->scope.requestStop();
207 return true;
208 }
209 // Still suspended, and decided: withdraw the stop callback (waiting
210 // out one that is running) and the handler attached before the throw.
211 shared->stopCallback.reset();
212 shared->scope.requestStop();
213 throw;
214 }
215 return true;
216 }
217
218 /// @return The settled value.
219 /// @throws core::async::OperationCancelled if a stop withdrew the await.
220 /// @throws Whatever the completion was rejected with.
221 /// @throws std::logic_error for an empty completion or one with no executor.
222 T await_resume() {
223 _shared->stopCallback.reset();
224 if (_shared->outcome.load() == Outcome::Cancelled) {
225 throw ::core::async::OperationCancelled{};
226 }
227 if (_shared->error) {
228 std::rethrow_exception(_shared->error);
229 }
230 return std::move(*_shared->value);
231 }
232
233private:
234 std::shared_ptr<CompletionState<T>> _state;
235 std::shared_ptr<Shared> _shared;
236};
