22/// @brief Server-side message handler that owns model instances and dispatches actions.
23///
24/// `RemoteServer` receives JSON envelopes (`morph::wire::Envelope`) from any
25/// transport (WebSocket, in-process simulation, …) and executes the corresponding
26/// model operations via an `ActionDispatcher`. Authorization is delegated to an
27/// `IAuthorizer` that defaults to allow-all.
28///
29/// @par Heap allocation requirement
30/// `RemoteServer` **must** be heap-allocated via `std::make_shared`. `handle()`
31/// captures `shared_from_this()` to prevent use-after-free when the worker pool
32/// outlives the server object.
33///
34/// @par Wire format
35/// All requests and replies are encoded as `morph::wire::Envelope` JSON. See
36/// `wire.hpp` for the field semantics. The `kind` field is the discriminator.
37class RemoteServer : public std::enable_shared_from_this<RemoteServer> {
38public:
39 /// @brief Constructs a server backed by @p workerPool with allow-all authorization.
40 ///
41 /// @param workerPool Pool used to process messages asynchronously.
42 /// @param dispatcher Action dispatcher; defaults to the process-level singleton.
43 /// @param registry Model factory registry; defaults to the process-level singleton.
44 explicit RemoteServer(
45 ::morph::exec::IExecutor& workerPool,
46 ::morph::model::detail::ActionDispatcher& dispatcher = ::morph::model::detail::defaultDispatcher(),
47 ::morph::model::detail::ModelRegistryFactory& registry = ::morph::model::detail::defaultRegistry())
48 : _pool{workerPool},
49 _strand{workerPool},
50 _dispatcher{dispatcher},
51 _registry{registry},
52 _authorizer{::morph::session::allowAllAuthorizer()} {}
53
54 /// @brief Constructs a server with a custom authorizer.
55 ///
56 /// @param workerPool Pool used to process messages asynchronously.
57 /// @param authorizer Authorizer consulted for every `execute` envelope.
58 /// @param dispatcher Action dispatcher; defaults to the process-level singleton.
59 /// @param registry Model factory registry; defaults to the process-level singleton.
60 RemoteServer(
61 ::morph::exec::IExecutor& workerPool, std::shared_ptr<::morph::session::IAuthorizer> authorizer,
62 ::morph::model::detail::ActionDispatcher& dispatcher = ::morph::model::detail::defaultDispatcher(),
63 ::morph::model::detail::ModelRegistryFactory& registry = ::morph::model::detail::defaultRegistry())
64 : _pool{workerPool},
65 _strand{workerPool},
66 _dispatcher{dispatcher},
67 _registry{registry},
68 _authorizer{std::move(authorizer)} {
69 if (!_authorizer) {
70 _authorizer = ::morph::session::allowAllAuthorizer();
71 }
72 }

74 /// @brief Asynchronously processes a JSON `Envelope` and calls @p reply with the response.
75 ///
76 /// The message is dispatched to the worker pool. @p reply is called exactly
77 /// once from the pool thread when processing completes.
78 ///
79 /// Thread-safe. Safe to call before the previous call's reply has been delivered.
80 ///
81 /// @param msg JSON-encoded `morph::wire::Envelope` (via `wire::encode`).
82 /// @param reply Callback invoked with the JSON-encoded reply envelope.
83 void handle(std::string msg, std::function<void(std::string)> reply) {
84 auto self = shared_from_this();
85 _pool.post(
86 [self, msg = std::move(msg), reply = std::move(reply)]() mutable { self->dispatchMessage(msg, reply); });
87 }
88
89 /// @brief Synchronously processes a JSON `Envelope` on the calling thread and returns the reply.
90 ///
91 /// Equivalent to `handle()` but never posts to the worker pool, so it is safe
92 /// to call from a thread that *is* the worker pool — for example, from a
93 /// `BridgeHandler` constructor invoked from inside an action handler.
94 ///
95 /// Only safe for control messages (`register`, `deregister`) — `execute`
96 /// messages still post to the strand and would return before the result is
97 /// produced. The implementation rejects `execute` to make this explicit.
98 ///
99 /// @param msg JSON-encoded `morph::wire::Envelope` (via `wire::encode`).
100 /// @return JSON-encoded reply envelope.
101 std::string handleInline(const std::string& msg) {
102 std::string reply;
103 std::function<void(std::string)> capture = [&reply](std::string out) { reply = std::move(out); };
104 dispatchMessage(msg, capture);
105 return reply;
106 }
107
108 /// @brief Callable that supplies the action log to attach to a newly
109 /// registered instance, given its model type and `contextKey`.
110 ///
111 /// Return `nullptr` to register the instance with no log attached (e.g. for
112 /// model types or context keys the host app doesn't want journaled).
113 using LogProvider =
114 std::function<std::shared_ptr<::morph::journal::IActionLog>(std::string_view modelType, std::string_view contextKey)>;
115
116 /// @brief Installs @p provider, consulted on every `register` envelope whose
117 /// `contextKey` is non-empty.
118 ///
119 /// This is what closes the gap `IModelHolder::attachActionLog` leaves open
120 /// for remote topologies: `RemoteServer` owns the actual model instances for
121 /// every remote/simulated-remote client, so it is the only place that can
122 /// attach a log to them. Pass `nullptr` to remove a previously installed
123 /// provider (new registrations get no log). Thread-safe.
124 /// @param provider Callable invoked synchronously while handling `register`.
125 void setLogProvider(LogProvider provider) {
126 std::scoped_lock const lock{_logProviderMtx};
127 _logProvider = std::move(provider);
128 }
129
130private:
131 void dispatchMessage(const std::string& msg, std::function<void(std::string)>& reply) {
132 ::morph::wire::Envelope env;
133 try {
134 env = ::morph::wire::decode(msg);
135 } catch (const std::exception& exc) {
136 reply(::morph::wire::encode(::morph::wire::makeErr(exc.what())));
137 return;
138 }
139 try {
140 if (env.kind == "register") {
141 if (env.typeId.empty()) {
142 throw std::runtime_error("register requires a typeId");
143 }
144 auto holder = _registry.create(env.typeId);
145 if (!env.contextKey.empty()) {
146 LogProvider provider;
147 {
148 std::scoped_lock const lock{_logProviderMtx};
149 provider = _logProvider;
150 }
151 if (provider) {
152 if (auto log = provider(env.typeId, env.contextKey)) {
153 holder->attachActionLog(std::move(log), env.contextKey);
154 }
155 }
156 }
157 ::morph::exec::detail::ModelId const mid{_nextId.fetch_add(1) + 1};
158 {
159 std::scoped_lock const lock{_regMtx};
160 _models[mid] = std::move(holder);
161 }
162 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId, {}, mid.v)));
163 } else if (env.kind == "deregister") {
164 {
165 std::scoped_lock const lock{_regMtx};
166 _models.erase(::morph::exec::detail::ModelId{env.modelId});
167 }
168 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId)));
169 } else if (env.kind == "execute") {
170 dispatchExecute(std::move(env), reply);
171 } else {
172 reply(::morph::wire::encode(
173 ::morph::wire::makeErr("unknown envelope kind: " + env.kind, env.callId)));
174 }
175 } catch (const std::exception& exc) {
176 reply(::morph::wire::encode(::morph::wire::makeErr(exc.what(), env.callId)));
177 }
178 }
179
180 void dispatchExecute(::morph::wire::Envelope env, std::function<void(std::string)> reply) {
181 if (!_authorizer->authorize(env.session, env.modelType, env.actionType)) {
182 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
183 return;
184 }
185 ::morph::exec::detail::ModelId const mid{env.modelId};
186 std::shared_ptr<::morph::model::detail::IModelHolder> holder;
187 {
188 std::scoped_lock const lock{_regMtx};
189 auto iter = _models.find(mid);
190 if (iter != _models.end()) {
191 holder = iter->second;
192 }
193 }
194 if (!holder) {
195 reply(::morph::wire::encode(::morph::wire::makeErr("model not found", env.callId)));
196 return;
197 }
198 _strand.post(mid, [&disp = _dispatcher, env = std::move(env), holder = std::move(holder),
199 reply = std::move(reply)]() mutable {
200 try {
201 ::morph::session::detail::ScopedContext const scoped{env.session};
202 auto result = disp.dispatch(env.modelType, env.actionType, *holder, env.body);
203 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId, std::move(result))));
204 } catch (const std::exception& exc) {
205 reply(::morph::wire::encode(::morph::wire::makeErr(exc.what(), env.callId)));
206 }
207 });
208 }
209
210 ::morph::exec::IExecutor& _pool;
211 ::morph::exec::detail::StrandExecutor _strand;
212 ::morph::model::detail::ActionDispatcher& _dispatcher;
213 ::morph::model::detail::ModelRegistryFactory& _registry;
214 std::shared_ptr<::morph::session::IAuthorizer> _authorizer;
215 std::mutex _regMtx;
216 std::unordered_map<::morph::exec::detail::ModelId, std::shared_ptr<::morph::model::detail::IModelHolder>,
217 ::morph::exec::detail::ModelIdHash>
218 _models;
219 std::atomic<uint64_t> _nextId{0};
220 std::mutex _logProviderMtx;
221 LogProvider _logProvider;
222};

224/// @brief `IBackend` adapter that routes all calls through a `RemoteServer` using a
225/// synchronous request-reply protocol.
226///
227/// Intended for testing and in-process simulation of remote execution. All
228/// `registerModel()`, `deregisterModel()`, and `execute()` calls are forwarded
229/// as protocol messages to the server; the calling thread blocks until the reply
230/// arrives via `std::promise`.
231class SimulatedRemoteBackend : public detail::IBackend {
232public:
233 /// @brief Constructs the backend targeting @p server.
234 /// @param server The `RemoteServer` instance to forward calls to.
235 explicit SimulatedRemoteBackend(RemoteServer& server) : _server{server} {}
236
237 /// @brief Registers the model type on the server and returns its assigned id.
238 ///
239 /// Processed inline on the calling thread (no pool round-trip), so it is safe
240 /// to call from any thread including a worker in the same pool that backs the
241 /// `RemoteServer`. The @p factory argument is ignored — model construction is
242 /// delegated to the server's `ModelRegistryFactory`.
243 ///
244 /// @param typeId String type-id sent in the `register` message.
245 /// @return `ModelId` assigned by the server.
246 /// @throws std::runtime_error if the server replies with an error.
247 ::morph::exec::detail::ModelId registerModel(
248 const std::string& typeId,
249 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> /*factory*/) override {
250 return registerModelWithContext(typeId, {}, {});
251 }
252
253 /// @brief Registers the model type on the server, carrying @p contextKey across
254 /// the wire so the server's `RemoteServer::LogProvider` (if configured)
255 /// can attach an action log to the instance it creates.
256 ///
257 /// @p factory is still ignored — model construction is delegated to the
258 /// server's `ModelRegistryFactory`, same as `registerModel()`.
259 /// @param typeId String type-id sent in the `register` message.
260 /// @param contextKey Stable identity of the new instance; empty if none.
261 /// @return `ModelId` assigned by the server.
262 /// @throws std::runtime_error if the server replies with an error.
263 ::morph::exec::detail::ModelId registerModelWithContext(
264 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> /*factory*/,
265 std::string_view contextKey) override {
266 auto reply = ::morph::wire::decode(_server.handleInline(
267 ::morph::wire::encode(::morph::wire::makeRegister(typeId, std::string{contextKey}))));
268 if (reply.kind == "ok") {
269 return ::morph::exec::detail::ModelId{reply.modelId};
270 }
271 throw std::runtime_error("register failed: " + reply.message);
272 }
273
274 /// @brief Deregisters the model on the server. Processed inline; safe from any thread.
275 /// @param mid Id of the model to deregister.
276 void deregisterModel(::morph::exec::detail::ModelId mid) override {
277 (void)_server.handleInline(::morph::wire::encode(::morph::wire::makeDeregister(mid.v)));
278 }
279
280 /// @brief Serialises the action, sends it to the server, and returns a `Completion`.
281 ///
282 /// The `Completion` resolves when the server's reply is received and
283 /// deserialized. Callbacks are posted via @p cbExec. The session attached to
284 /// the call (via `Bridge::setDefaultSession()` or the per-call API) is
285 /// serialised into the envelope.
286 ///
287 /// @param mid Target model id on the server.
288 /// @param call Bundled action; `serializeAction` and `deserializeResult` are used.
289 /// @param cbExec Executor for delivering the completion callbacks.
290 /// @return Completion that resolves with the deserialized result or an error.
291 ::morph::async::Completion<std::shared_ptr<void>> execute(::morph::exec::detail::ModelId mid,
292 detail::ActionCall call,
293 ::morph::exec::IExecutor* cbExec) override {
294 auto state = std::make_shared<::morph::async::detail::CompletionState<std::shared_ptr<void>>>();
295 ::morph::async::Completion<std::shared_ptr<void>> comp{state, cbExec};
296 trackPending(state);
297
298 ::morph::wire::Envelope env;
299 env.kind = "execute";
300 env.modelId = mid.v;
301 env.modelType = call.modelTypeId;
302 env.actionType = call.actionTypeId;
303 env.body = call.serializeAction();
304 env.session = std::move(call.session);
305 auto deser = std::move(call.deserializeResult);
306
307 _server.handle(::morph::wire::encode(env),
308 [state, deser = std::move(deser)](const std::string& replyJson) mutable {
309 try {
310 auto reply = ::morph::wire::decode(replyJson);
311 if (reply.kind == "ok") {
312 state->setValue(deser(reply.body));
313 } else {
314 throw std::runtime_error(
315 reply.message.empty() ? "malformed reply" : reply.message);
316 }
317 } catch (...) {
318 state->setException(std::current_exception());
319 }
320 });
321 return comp;
322 }
323
324 /// @brief No-op — models live in `RemoteServer`, not locally.
325 void notifyBackendChanged() override {}
326
327 /// @brief Resolves every still-pending completion this backend produced with @p exc.
328 /// @param exc Exception delivered to every pending completion's error sink.
329 void cancelPending(const std::exception_ptr& exc) override {
330 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> snapshot;
331 {
332 std::scoped_lock const lock{_pendingMtx};
333 snapshot.swap(_pending);
334 }
335 for (auto& weak : snapshot) {
336 if (auto state = weak.lock()) {
337 state->setException(exc);
338 }
339 }
340 }
341
342private:
343 void trackPending(const std::shared_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>& state) {
344 std::scoped_lock const lock{_pendingMtx};
345 std::erase_if(_pending, [](const auto& weak) { return weak.expired(); });
346 _pending.emplace_back(state);
347 }
348
349 RemoteServer& _server;
350 std::mutex _pendingMtx;
351 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> _pending;
352};
