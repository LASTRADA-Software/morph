60/// @brief Background scheduler that invokes a callback once after a delay, unless cancelled first.
61///
62/// `RemoteServer` is transport-agnostic and its `IExecutor` has no delayed-post
63/// primitive, so a single dedicated thread per instance tracks pending
64/// deadlines and fires callbacks when they elapse. Used to enforce
65/// `LimitPolicy::executeTimeout` — see `docs/spec/core/backend.md`.
66class TimeoutScheduler {
67public:
68 /// @brief Opaque identifier for one scheduled callback.
69 using Handle = std::uint64_t;
70
71 /// @brief Starts the background thread.
72 TimeoutScheduler() : _thread{[this] { run(); }} {}
73
74 /// @brief Stops the background thread and joins it.
75 ~TimeoutScheduler() {
76 {
77 std::scoped_lock const lock{_mtx};
78 _stop = true;
79 }
80 _cv.notify_all();
81 _thread.join();
82 }
83
84 TimeoutScheduler(const TimeoutScheduler&) = delete;
85 TimeoutScheduler& operator=(const TimeoutScheduler&) = delete;
86 TimeoutScheduler(TimeoutScheduler&&) = delete;
87 TimeoutScheduler& operator=(TimeoutScheduler&&) = delete;
88
89 /// @brief Schedules @p callback to run after @p delay on the scheduler's
90 /// background thread, unless cancelled first via `cancel()`.
91 /// @param delay Time to wait before firing.
92 /// @param callback Invoked on the scheduler thread if not cancelled in time.
93 /// Exceptions it throws are logged and swallowed.
94 /// @return Handle usable with `cancel()`.
95 Handle schedule(std::chrono::milliseconds delay, std::function<void()> callback) {
96 auto const deadline = std::chrono::steady_clock::now() + delay;
97 std::scoped_lock const lock{_mtx};
98 Handle const handle = ++_nextHandle;
99 auto iter = _entries.emplace(deadline, Entry{handle, std::move(callback)});
100 _index[handle] = iter;
101 _cv.notify_all();
102 return handle;
103 }
104
105 /// @brief Cancels a previously scheduled callback immediately.
106 ///
107 /// If @p handle has not fired yet, its entry (and anything its callback
108 /// captured) is erased right away — the caller does not have to wait for
109 /// the original deadline for that memory to be released. A no-op if
110 /// @p handle already fired or was already cancelled.
111 /// @param handle Handle returned by a prior `schedule()` call.
112 void cancel(Handle handle) {
113 std::scoped_lock const lock{_mtx};
114 auto found = _index.find(handle);
115 if (found == _index.end()) {
116 return;
117 }
118 _entries.erase(found->second);
119 _index.erase(found);
120 }
121
122private:
123 struct Entry {
124 Handle handle;
125 std::function<void()> callback;
126 };
127
128 void run() {
129 std::unique_lock lock{_mtx};
130 while (!_stop) {
131 if (_entries.empty()) {
132 _cv.wait(lock);
133 continue;
134 }
135 auto const nextDeadline = _entries.begin()->first;
136 _cv.wait_until(lock, nextDeadline);
137 if (_stop) {
138 break;
139 }
140 auto now = std::chrono::steady_clock::now();
141 while (!_entries.empty() && _entries.begin()->first <= now) {
142 auto iter = _entries.begin();
143 Entry entry = std::move(iter->second);
144 _index.erase(entry.handle);
145 _entries.erase(iter);
146 lock.unlock();
147 try {
148 entry.callback();
149 } catch (const std::exception& exc) {
150 ::morph::log::logError("[timeout-scheduler] callback threw: " + std::string{exc.what()});
151 } catch (...) {
152 ::morph::log::logError("[timeout-scheduler] callback threw unknown exception");
153 }
154 lock.lock();
155 now = std::chrono::steady_clock::now();
156 }
157 }
158 }
159
160 std::mutex _mtx;
161 std::condition_variable _cv;
162 std::multimap<std::chrono::steady_clock::time_point, Entry> _entries;
163 std::unordered_map<Handle, std::multimap<std::chrono::steady_clock::time_point, Entry>::iterator> _index;
164 Handle _nextHandle{0};
165 bool _stop{false};
166 std::thread _thread;
167};

169/// @brief Keyed 64-bit bijection that turns a monotonic counter into an
170/// unguessable, non-sequential id.
171///
172/// Implements a 4-round Feistel network over two 32-bit halves. A Feistel
173/// network is a bijection over its full domain for *any* round function —
174/// that is what guarantees `RemoteServer` never hands out the same id twice
175/// for two different counter values. What makes the permutation *opaque*
176/// rather than merely "scrambled" is that each round's mixing function folds
177/// in a secret round key drawn once, at construction, from
178/// `std::random_device`: an *unkeyed* public mixing function would be
179/// invertible by anyone reading the source, letting an attacker who observes
180/// one id recover the counter and predict the next; the secret per-round keys
181/// prevent that without needing the mixing function itself to be secret.
182///
183/// This is a self-contained reference construction (no external crypto
184/// dependency), in the same spirit as the hand-rolled HMAC-SHA256 in
185/// `session_auth.hpp`: adequate for the stated defence-in-depth goal (opaque
186/// ids are not the authorization boundary — `IAuthorizer::authorizeInstance`
187/// is), not a cryptographically-audited primitive.
188class OpaqueIdGenerator {
189public:
190 /// @brief Draws four independent 32-bit round keys from `std::random_device`.
191 OpaqueIdGenerator() {
192 std::random_device rd;
193 for (auto& key : _roundKeys) {
194 // std::random_device::result_type is unsigned int on this platform's
195 // standard library, so the cast below is a no-op here -- but the
196 // standard does not guarantee that, so it stays for portability to a
197 // standard library where result_type is wider than uint32_t.
198#if defined(__GNUC__) && !defined(__clang__)
199#pragma GCC diagnostic push
200#pragma GCC diagnostic ignored "-Wuseless-cast"
201#endif
202 key = static_cast<uint32_t>(rd());
203#if defined(__GNUC__) && !defined(__clang__)
204#pragma GCC diagnostic pop
205#endif
206 }
207 }
208
209 /// @brief Applies the keyed permutation to @p counter.
210 ///
211 /// Bijective over the full 64-bit domain: distinct @p counter values
212 /// always produce distinct results (the Feistel structure guarantees
213 /// this regardless of the round function), so a monotonically
214 /// increasing, non-repeating @p counter can never yield a collision.
215 /// @param counter Monotonic input, e.g. from an atomic counter.
216 /// @return A 64-bit value that is a bijective function of @p counter.
217 [[nodiscard]] uint64_t permute(uint64_t counter) const noexcept {
218 auto lo = static_cast<uint32_t>(counter & 0xffffffffULL);
219 auto hi = static_cast<uint32_t>(counter >> 32);
220 for (const uint32_t roundKey : _roundKeys) {
221 const uint32_t nextHi = lo;
222 lo = hi ^ mix(lo, roundKey);
223 hi = nextHi;
224 }
225 return (static_cast<uint64_t>(hi) << 32) | static_cast<uint64_t>(lo);
226 }
227
228private:
229 /// @brief Keyed avalanche mix (fmix32-style) used as the Feistel round function.
230 /// @param half Current 32-bit half being folded into the other half.
231 /// @param key This round's secret key.
232 /// @return A well-mixed 32-bit value depending non-linearly on both @p half and @p key.
233 [[nodiscard]] static uint32_t mix(uint32_t half, uint32_t key) noexcept {
234 uint32_t val = half ^ key;
235 val ^= val >> 16;
236 val *= 0x7feb352dU;
237 val ^= val >> 15;
238 val *= 0x846ca68bU;
239 val ^= val >> 16;
240 return val;
241 }
242
243 std::array<uint32_t, 4> _roundKeys{};
244};

317 /// @brief Asynchronously processes a JSON `Envelope` and calls @p reply with the response.
318 ///
319 /// The message is dispatched to the worker pool. @p reply is called exactly
320 /// once from the pool thread when processing completes.
321 ///
322 /// Thread-safe. Safe to call before the previous call's reply has been delivered.
323 ///
324 /// @param msg JSON-encoded `morph::wire::Envelope` (via `wire::encode`).
325 /// @param reply Callback invoked with the JSON-encoded reply envelope.
326 void handle(std::string msg, std::function<void(std::string)> reply) {
327 auto self = shared_from_this();
328 _pool.post(
329 [self, msg = std::move(msg), reply = std::move(reply)]() mutable { self->dispatchMessage(msg, reply); });
330 }

332 /// @brief Like `handle(msg, reply)`, but additionally attributes any
333 /// `register` decoded from @p msg to a connection scope.
334 ///
335 /// A `register` processed under a non-zero @p cid records the new model in
336 /// that connection's scope, so a later `closeConnection(cid)` reclaims it.
337 /// Passing `cid == 0` is exactly the unscoped, two-argument `handle()` —
338 /// nothing is recorded and nothing is ever cleaned up automatically.
339 ///
340 /// Thread-safe. Safe to call before the previous call's reply has been delivered.
341 ///
342 /// @param msg JSON-encoded `morph::wire::Envelope` (via `wire::encode`).
343 /// @param reply Callback invoked with the JSON-encoded reply envelope.
344 /// @param cid Connection scope to attribute a `register` in @p msg to;
345 /// `0` means unscoped.
346 void handle(std::string msg, std::function<void(std::string)> reply, ConnectionId cid) {
347 auto self = shared_from_this();
348 _pool.post([self, msg = std::move(msg), reply = std::move(reply), cid]() mutable {
349 self->dispatchMessage(msg, reply, cid);
350 });
351 }

353 /// @brief Synchronously processes a JSON `Envelope` on the calling thread and returns the reply.
354 ///
355 /// Equivalent to `handle()` but never posts to the worker pool, so it is safe
356 /// to call from a thread that *is* the worker pool — for example, from a
357 /// `BridgeHandler` constructor invoked from inside an action handler.
358 ///
359 /// Only safe for control messages (`register`, `deregister`). An `execute`
360 /// envelope posts to the strand and produces its reply asynchronously, after
361 /// this synchronous call has already returned and destroyed the local reply
362 /// buffer the deferred callback would write into. To keep that from becoming a
363 /// dangling write, `execute` is rejected up front with an `err` reply.
364 ///
365 /// @param msg JSON-encoded `morph::wire::Envelope` (via `wire::encode`).
366 /// @return JSON-encoded reply envelope.
367 std::string handleInline(const std::string& msg) {
368 try {
369 auto env = ::morph::wire::decode(msg);
370 if (env.kind == "execute") {
371 return ::morph::wire::encode(::morph::wire::makeErr(
372 "handleInline does not support execute (reply is asynchronous)", env.callId));
373 }
374 } catch (const std::exception&) { // NOLINT(bugprone-empty-catch)
375 // Malformed input: fall through so dispatchMessage produces the
376 // canonical decode-error reply (avoids duplicating that path here).
377 }
378 std::string reply;
379 std::function<void(std::string)> capture = [&reply](std::string out) noexcept { reply = std::move(out); };
380 dispatchMessage(msg, capture);
381 return reply;
382 }

384 /// @brief Opens a new connection scope and returns its id.
385 ///
386 /// Call once per accepted transport connection (e.g. from a WebSocket
387 /// server's "new connection" callback). The returned id is never `0`, so
388 /// it can always be distinguished from the reserved "unscoped" value. Pass
389 /// it to the scoped `handle(msg, reply, cid)` overload for every message
390 /// received on that connection, and to `closeConnection(cid)` once the
391 /// connection is gone.
392 ///
393 /// Thread-safe.
394 /// @return A fresh, non-zero `ConnectionId`.
395 [[nodiscard]] ConnectionId openConnection() {
396 ConnectionId const cid{_nextConnectionId.fetch_add(1) + 1};
397 std::scoped_lock const lock{_regMtx};
398 _connectionScopes.try_emplace(cid);
399 return cid;
400 }
401
402 /// @brief Reclaims every model still registered under @p cid, then drops the scope.
403 ///
404 /// Call once the transport observes the connection is gone (disconnect,
405 /// close, error). Erases every surviving model in `cid`'s scope from the
406 /// registry exactly as an explicit `deregister` would (so a later
407 /// `execute` against one of those ids replies `err "model not found"`),
408 /// then drops the scope itself.
409 ///
410 /// Idempotent: `cid == 0`, an unknown `cid`, or a `cid` already closed is a
411 /// no-op. Deliberately does **not** consult `IAuthorizer` — this is the
412 /// server's own housekeeping in reaction to a transport-level event, not
413 /// an action attributable to any caller (see docs/spec/core/backend.md).
414 ///
415 /// Safe to call while a model in the scope has an `execute` in flight: the
416 /// strand task already holds its own `shared_ptr` to the model holder, so
417 /// erasing the registry entry here only prevents *new* lookups (see
418 /// docs/spec/concurrency_and_lifetimes.md).
419 ///
420 /// Thread-safe.
421 /// @param cid Connection scope to close, as returned by `openConnection()`.
422 void closeConnection(ConnectionId cid) {
423 if (cid == 0) {
424 return;
425 }
426 std::scoped_lock const lock{_regMtx};
427 auto scopeIter = _connectionScopes.find(cid);
428 if (scopeIter == _connectionScopes.end()) {
429 return;
430 }
431 for (const auto& [mid, refs] : scopeIter->second) {
432 // Release exactly as many references as this connection held. A
433 // shared instance another connection is still attached to survives;
434 // a private one (count 1, no directory entry) is erased outright,
435 // which is byte-for-byte the previous behaviour.
436 for (std::size_t idx = 0; idx < refs; ++idx) {
437 releaseInstanceLocked(mid);
438 }
439 }
440 _connectionScopes.erase(scopeIter);
441 }

566 /// @brief Blocks until every in-flight `execute` has delivered its reply,

568 ///
569 /// "In-flight" is one counter (`_inFlightExecutes`), incremented when
570 /// `dispatchExecute` admits a call for dispatch (right before posting to
571 /// the model's strand) and decremented right before its reply is sent, on

573 /// firing first) — the same state `LimitPolicy::maxInFlightExecutes` (if

575 /// rather than double-counted. Waits on a condition variable signalled
576 /// when the counter reaches zero — not a busy poll. Safe to call from any
577 /// thread, independently of `beginShutdown()` (it only observes the

581 /// elapsed; `false` on timeout.
582 [[nodiscard]] bool drainedWithin(std::chrono::milliseconds deadline) {
583 std::unique_lock lock{_drainMtx};
584 return _drainCv.wait_for(lock, deadline,
585 [this] { return _inFlightExecutes.load(std::memory_order_acquire) == 0; });

592 /// @brief Releases one reference to @p mid, destroying it at zero. Caller holds `_regMtx`.
593 ///
594 /// A private instance has no `_attachCount` entry and is erased outright —
595 /// byte-for-byte the pre-sharing behaviour. A shared instance is erased, and
596 /// removed from the directory, only when its last attachment goes away, so
597 /// one client's `deregister` or dropped connection never tears an instance
598 /// out from under another client still using it.
599 /// @param mid Instance to release.
600 /// @return `true` if this call destroyed the instance.
601 bool releaseInstanceLocked(::morph::exec::detail::ModelId mid) {
602 if (auto refIter = _attachCount.find(mid); refIter != _attachCount.end()) {
603 refIter->second -= 1;
604 if (refIter->second > 0) {
605 return false;
606 }
607 _attachCount.erase(refIter);
608 if (auto keyIter = _sharedKeyOf.find(mid); keyIter != _sharedKeyOf.end()) {
609 _directory.erase(keyIter->second);
610 _sharedKeyOf.erase(keyIter);
611 }
612 }
613 _models.erase(mid);
614 _owners.erase(mid);
615 _firstActionPending.erase(mid);
616 _poisoned.erase(mid);
617 return true;
618 }

673 /// @brief Attaches to an already-live directory entry, if there is one. Caller holds `_regMtx`.
674 ///
675 /// Factored out because `acquireSharedInstance` checks the directory twice —
676 /// once on entry, and again under the insert lock after building a holder
677 /// outside it, in case a concurrent request for the same key won the race.
678 /// The second check is by nature almost never taken, so duplicating the body
679 /// would leave a block that is both untested and free to drift from the one
680 /// that is.
681 ///
682 /// @param dirKey Directory key being acquired.
683 /// @param env Decoded request, for `callId`.
684 /// @param reply Reply sink; invoked only when this returns `true`.
685 /// @param cid Connection scope, or `0` for unscoped.
686 /// @return `true` if an entry existed and @p reply was invoked; `false` to keep going.
687 bool attachExistingLocked(const DirectoryKey& dirKey, const ::morph::wire::Envelope& env,
688 const std::function<void(std::string)>& reply, ConnectionId cid) {
689 auto found = _directory.find(dirKey);
690 if (found == _directory.end()) {
691 return false;
692 }
693 auto const mid = found->second;
694 if (_poisoned.contains(mid)) {
695 // This instance's first action already failed; it must not be
696 // handed to a new attacher. Evict it from the directory -- its
697 // own eventual release still tears it down normally -- and report
698 // a miss so the caller falls through to creating a fresh
699 // instance.
700 _directory.erase(found);
701 _sharedKeyOf.erase(mid);
702 return false;
703 }
704 _attachCount[mid] += 1;
705 if (!noteScopeAttachLocked(mid, cid)) {
706 releaseInstanceLocked(mid);
707 reply(::morph::wire::encode(::morph::wire::makeErr("connection closed", env.callId)));
708 return true;
709 }
710 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId, {}, mid.v)));
711 return true;
712 }

714 /// @brief Acquires (or creates) the shared instance for `(typeId, primary)` and replies.
715 ///
716 /// The register-or-attach core shared by the `register` branch (when
717 /// `shared` is set) and by `attach`. A shared instance is recorded with an
718 /// **empty owner principal**: `IAuthorizer::authorizeInstance`'s documented
719 /// `ownerPrincipal == ctx.principal` policy would otherwise reject every
720 /// client but the one that created it, defeating cross-client sharing
721 /// outright. Gating access to a shared model is therefore `authorize`'s job
722 /// (per type and action) or the model's own — see docs/spec/security.md.
723 ///
724 /// @param env Decoded request; uses `typeId`, `primary`, `contextKey`, `callId`.
725 /// @param reply Reply sink; always invoked exactly once.
726 /// @param cid Connection scope, or `0` for unscoped.
727 /// @param releaseCurrent Instance to release once the target is
728 /// confirmed acquired or confirmed about to be
729 /// created (an `attach` re-point), or `ModelId{0}`.
730 /// A throwing *construction* (`_registry.create`)
731 /// never touches it. Once construction succeeds,
732 /// release happens in the same locked section as
733 /// the `maxLiveModels` admission check, so a sole
734 /// holder's release frees the slot the re-point
735 /// itself needs rather than losing it to the cap;
736 /// the only remaining post-release failure is the
737 /// connection's own scope having closed
738 /// concurrently, in which case no further request
739 /// on it will run anyway.
740 void acquireSharedInstance(const ::morph::wire::Envelope& env, const std::function<void(std::string)>& reply,
741 ConnectionId cid, ::morph::exec::detail::ModelId releaseCurrent) {
742 LimitPolicy limits;
743 {
744 std::scoped_lock const lock{_limitsMtx};
745 limits = _limits;
746 }
747 DirectoryKey dirKey{env.typeId, env.primary};
748 {
749 std::scoped_lock const lock{_regMtx};
750 if (attachExistingLocked(dirKey, env, reply, cid)) {
751 // Acquired (or re-confirmed) the target before touching
752 // `releaseCurrent` -- a same-key re-attach lands on the exact
753 // same mid attachExistingLocked just incremented, so
754 // releasing it here cancels out only the redundant reference
755 // that call just took, never the caller's sole hold on the
756 // instance it is "re-pointing" to itself. A genuinely
757 // different-key re-point releases the real old instance, same
758 // as before.
759 if (releaseCurrent.v != 0U) {
760 releaseScopedLocked(releaseCurrent, cid);
761 }
762 return;
763 }
764 }
765 // Directory miss. Construct outside the lock, exactly as the private
766 // register path does, then re-check under the insert lock: a concurrent
767 // request for the same key may have won the race while we built ours.
768 auto holder = _registry.create(env.typeId);
769 attachLogIfConfigured(*holder, env);
770 ::morph::exec::detail::ModelId const fresh{nextOpaqueId()};
771 {
772 std::scoped_lock const lock{_regMtx};
773 if (attachExistingLocked(dirKey, env, reply, cid)) {
774 if (releaseCurrent.v != 0U) {
775 releaseScopedLocked(releaseCurrent, cid);
776 }
777 return;
778 }
779 // Confirmed miss: release the old instance now, in the same
780 // locked section as the maxLiveModels admission check, so a sole
781 // holder's release frees exactly the slot this re-point needs
782 // rather than losing it to the cap in between -- the property
783 // the single `attach` wire request exists to provide. The only
784 // way admission can still fail after this is a concurrently
785 // closed connection scope (noteScopeAttachLocked below), which
786 // makes "stranding" moot: no further request on that connection
787 // will ever run anyway.
788 if (releaseCurrent.v != 0U) {
789 releaseScopedLocked(releaseCurrent, cid);
790 }
791 if (limits.maxLiveModels != 0 && _models.size() >= limits.maxLiveModels) {
792 reply(::morph::wire::encode(::morph::wire::makeErr("too many models", env.callId)));
793 return;
794 }
795 if (!noteScopeAttachLocked(fresh, cid)) {
796 reply(::morph::wire::encode(::morph::wire::makeErr("connection closed", env.callId)));
797 return;
798 }
799 _models[fresh] = std::move(holder);
800 _owners[fresh] = std::string{}; // shared instances are ownerless, by design
801 _directory.emplace(dirKey, fresh);
802 _sharedKeyOf.emplace(fresh, std::move(dirKey));
803 _attachCount[fresh] = 1;
804 _firstActionPending.insert(fresh);
805 }
806 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId, {}, fresh.v)));
807 }

810 ///
811 /// The existing holder of a key always wins: promoting onto a key another
812 /// instance already holds is a silent no-op rather than a displacement.
813 /// Symmetrically, an instance that already holds a *different* real key is
814 /// left exactly where it is — also a silent no-op — since instances never
815 /// change key (docs/spec/core/shared_instances.md); only a still-anonymous

821 return;

830 _directory.emplace(dirKey, mid);
831 _sharedKeyOf.emplace(mid, std::move(dirKey));
832 _attachCount.try_emplace(mid, 1);

853 // One flat switch over the wire's `kind` discriminator. Splitting it would
854 // scatter the authorization sequence each branch depends on across helpers,
855 // with no reader benefit.
856 // NOLINTNEXTLINE(readability-function-cognitive-complexity)
857 void dispatchMessage(const std::string& msg, std::function<void(std::string)>& reply, ConnectionId cid = 0) {
858 ::morph::wire::Envelope env;
859 try {
860 env = ::morph::wire::decode(msg);
861 } catch (const std::exception& exc) {
862 // Undecodable envelope: the one server-side record of a request
863 // that never dispatched at all. A client that swallows its own
864 // error (or is malformed precisely because it is confused) would
865 // otherwise leave no trace here. The payload prefix is the most
866 // useful field for diagnosing *why* the client sent something
867 // malformed -- and the most likely to carry application data, so
868 // it is capped at kLogPayloadPreviewBytes rather than logged in
869 // full. See docs/spec/core/backend.md, "Server-side observability".
870 constexpr std::size_t kLogPayloadPreviewBytes = 256;
871 std::string_view const preview =
872 std::string_view{msg}.substr(0, std::min(msg.size(), kLogPayloadPreviewBytes));
873 ::morph::log::logError("[dispatchMessage] undecodable envelope from connection {}: {} ({} bytes, "
874 "payload prefix: {}{})",
875 cid, exc.what(), msg.size(), preview,
876 msg.size() > kLogPayloadPreviewBytes ? "..." : "");
877 reply(::morph::wire::encode(::morph::wire::makeErr(exc.what())));
878 return;
879 }
880 // One line per successfully-decoded request -- the point every kind
881 // funnels through, so a client stuck mid-handshake (or one that never
882 // sent anything) is distinguishable from one whose requests are
883 // arriving normally. Deliberately omits the session principal (opt-in
884 // territory: personal data in many deployments) and the payload body
885 // (already covered, truncated, on the decode-failure path above;
886 // logging every successful body by default would be far higher volume
887 // and duplicate what dispatch already records via the action log for
888 // execute).
889 ::morph::log::logDebug("[dispatchMessage] connection {}: kind={} callId={} typeId={} modelId={} "
890 "modelType={} actionType={} bodyBytes={}",
891 cid, env.kind, env.callId, env.typeId, env.modelId, env.modelType, env.actionType,
892 env.body.size());
893 // Once shutdown has begun, new work is rejected fast — before any of
894 // the existing register/execute validation runs — while `deregister`
895 // (and any other kind) still flows through unchanged, so a client can
896 // still tear its models down cleanly during the drain window.
897 if ((env.kind == "register" || env.kind == "execute" || env.kind == "attach") &&
898 _shuttingDown.load(std::memory_order_acquire)) {
899 reply(::morph::wire::encode(::morph::wire::makeErr("server shutting down", env.callId)));
900 return;
901 }
902 try {
903 if (env.kind == "register") {
904 ::morph::observe::detail::emitMetric(::morph::observe::Metric::registerCount, 1.0);
905 if (env.typeId.empty()) {
906 throw std::runtime_error("register requires a typeId");
907 }
908 LimitPolicy limits;
909 {
910 std::scoped_lock const lock{_limitsMtx};
911 limits = _limits;
912 }
913 // Cheap early rejection, so a server already at its cap does not
914 // pay for authorize()/authenticate() and a model construction it
915 // is about to discard. Advisory only — the binding check is the
916 // re-test under the insert lock further below.
917 //
918 // Skipped for a *shared* register, which may well create nothing:
919 // if the key is already live it only takes another reference, and
920 // `maxLiveModels` caps live models, not attachments to them.
921 // Rejecting here would make a loaded server refuse the second
922 // client of an instance it is already hosting — exactly when
923 // sharing is worth the most. `acquireSharedInstance` re-tests the
924 // cap under the insert lock, where it can tell the two apart.
925 if (limits.maxLiveModels != 0 && (!env.shared || env.primary.empty())) {
926 std::scoped_lock const lock{_regMtx};
927 if (_models.size() >= limits.maxLiveModels) {
928 reply(::morph::wire::encode(::morph::wire::makeErr("too many models", env.callId)));
929 return;
930 }
931 }
932 // Authenticate the caller and make the verified identity
933 // authoritative, exactly as dispatchExecute does for execute: a
934 // verifying authorizer's returned principal overwrites
935 // env.session.principal; a non-authenticating authorizer
936 // (including allow-all) clears it, so the register decision
937 // below — and the owner recorded from it — never key on the
938 // client's unverified claim.
939 if (auto verified = _authorizer->authenticate(env.session)) {
940 env.session.principal = std::move(*verified);
941 } else {
942 env.session.principal.clear();
943 }
944 // Bound *who may create* an instance. The default hook allows
945 // all, so an unconfigured server registers any known type
946 // exactly as before; a deployer opts into gating registration
947 // by overriding authorizeRegister. No instance is constructed
948 // on denial.
949 if (!_authorizer->authorizeRegister(env.session, env.typeId)) {
950 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
951 return;
952 }
953 // A `shared` register naming a primary is a register-or-attach
954 // against the directory; everything below is the private path,
955 // unchanged.
956 if (env.shared && !env.primary.empty()) {
957 acquireSharedInstance(env, reply, cid, ::morph::exec::detail::ModelId{0});
958 return;
959 }
960 auto holder = _registry.create(env.typeId);
961 attachLogIfConfigured(*holder, env);
962 // Record the owner principal for per-instance authorization:
963 // env.session's principal is already the verified identity
964 // stamped above (empty if the authorizer does not
965 // authenticate), never the client's raw claim. This is what
966 // lets `authorizeInstance` later deny a different principal.
967 ::morph::exec::detail::ModelId const mid{nextOpaqueId()};
968 bool scopeAlreadyClosed = false;
969 bool overLiveModelCap = false;
970 {
971 std::scoped_lock const lock{_regMtx};
972 // Authoritative cap re-test, in the same critical section as
973 // the insert. The check near the top of this branch releases
974 // _regMtx before authorize(), authenticate() and
975 // _registry.create() run, so concurrent registers all
976 // observed the same under-cap size and every one of them
977 // proceeded to insert -- overshooting maxLiveModels by up to
978 // the worker pool's width. Only a check that cannot be
979 // separated from its insert actually bounds anything.
980 if (limits.maxLiveModels != 0 && _models.size() >= limits.maxLiveModels) {
981 overLiveModelCap = true;
982 }
983 // A non-zero cid attributes the new instance to that
984 // connection's scope, next to _models/_owners under the
985 // same lock so scope membership can never desync from
986 // instance existence. cid == 0 (the unscoped default)
987 // records nothing, matching today's behavior byte-for-byte.
988 //
989 // find(), never operator[]: the scope may already be gone.
990 // handle() *posts* this work to the pool, while
991 // closeConnection() runs synchronously on the transport's
992 // disconnect callback from another thread, so a client that
993 // registers and immediately drops its socket genuinely
994 // interleaves the two. operator[] would default-construct —
995 // resurrecting a scope closeConnection() had already
996 // erased — and nothing ever closes a scope twice, so this
997 // model and every later one on the dead cid would be
998 // unreclaimable: an unbounded leak that, with
999 // `maxLiveModels` set, wedges the server permanently at
1000 // `err "too many models"`.
1001 if (!overLiveModelCap && cid != 0) {
1002 auto scopeIter = _connectionScopes.find(cid);
1003 if (scopeIter == _connectionScopes.end()) {
1004 scopeAlreadyClosed = true;
1005 } else {
1006 scopeIter->second[mid] += 1;
1007 }
1008 }
1009 if (!overLiveModelCap && !scopeAlreadyClosed) {
1010 _models[mid] = std::move(holder);
1011 _owners[mid] = std::move(env.session.principal);
1012 }
1013 }
1014 if (overLiveModelCap) {
1015 // Lost the race for the last slot. `holder` goes out of
1016 // scope unregistered.
1017 reply(::morph::wire::encode(::morph::wire::makeErr("too many models", env.callId)));
1018 return;
1019 }
1020 if (scopeAlreadyClosed) {
1021 // The connection that asked for this instance is gone. Let
1022 // `holder` go out of scope unregistered rather than leak it,
1023 // and answer the (already dead) caller honestly.
1024 reply(::morph::wire::encode(::morph::wire::makeErr("connection closed", env.callId)));
1025 return;
1026 }
1027 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId, {}, mid.v)));
1028 } else if (env.kind == "attach") {
1029 if (env.typeId.empty()) {
1030 throw std::runtime_error("attach requires a typeId");
1031 }
1032 if (auto verified = _authorizer->authenticate(env.session)) {
1033 env.session.principal = std::move(*verified);
1034 } else {
1035 env.session.principal.clear();
1036 }
1037 if (!_authorizer->authorizeRegister(env.session, env.typeId)) {
1038 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
1039 return;
1040 }
1041 acquireSharedInstance(env, reply, cid, ::morph::exec::detail::ModelId{env.modelId});
1042 } else if (env.kind == "assign") {
1043 if (env.typeId.empty()) {
1044 throw std::runtime_error("assign requires a typeId");
1045 }
1046 // Mirrors "attach"'s gate: filing an instance into the shared
1047 // directory -- whether by creating it (register) or by
1048 // promoting one already live (assign) -- is bounds-checked
1049 // identically. Unlike "attach"/"register", assign never
1050 // constructs a model, but it still changes what a future
1051 // attacher of `primary` reaches, so it must not be reachable
1052 // by an unauthenticated or unauthorized caller either.
1053 if (auto verified = _authorizer->authenticate(env.session)) {
1054 env.session.principal = std::move(*verified);
1055 } else {
1056 env.session.principal.clear();
1057 }
1058 if (!_authorizer->authorizeRegister(env.session, env.typeId)) {
1059 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
1060 return;
1061 }
1062 {
1063 std::scoped_lock const lock{_regMtx};
1064 applyAssignLocked(env);
1065 }
1066 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId, {}, env.modelId)));
1067 } else if (env.kind == "instances") {
1068 if (env.typeId.empty()) {
1069 throw std::runtime_error("instances requires a typeId");
1070 }
1071 if (auto verified = _authorizer->authenticate(env.session)) {
1072 env.session.principal = std::move(*verified);
1073 } else {
1074 env.session.principal.clear();
1075 }
1076 // Enumeration is a read channel over the directory: gate it with
1077 // `authorize` for the model type (empty action id) so a deployer
1078 // can refuse listing without refusing use. It discloses the live
1079 // key set to anyone admitted — see docs/spec/security.md.
1080 if (!_authorizer->authorize(env.session, env.typeId, {})) {
1081 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
1082 return;
1083 }
1084 handleInstances(env, reply);
1085 } else if (env.kind == "deregister") {
1086 ::morph::observe::detail::emitMetric(::morph::observe::Metric::deregisterCount, 1.0);
1087 ::morph::exec::detail::ModelId const mid{env.modelId};
1088 // Per-instance authorization also gates deregister: consult the
1089 // hook with the recorded owner before destroying the instance.
1090 // The default hook allows all, so unconfigured behaviour is
1091 // unchanged; an ownership-enforcing authorizer can reject a
1092 // caller tearing down an instance it does not own.
1093 std::string owner;
1094 bool known = false;
1095 {
1096 std::scoped_lock const lock{_regMtx};
1097 auto iter = _owners.find(mid);
1098 if (iter != _owners.end()) {
1099 owner = iter->second;
1100 known = true;
1101 }
1102 }
1103 if (known && !_authorizer->authorizeInstance(env.session, {}, {}, mid.v, owner)) {
1104 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
1105 return;
1106 }
1107 {
1108 std::scoped_lock const lock{_regMtx};
1109 // Release exactly the reference *this* connection (`cid`,
1110 // the scope the deregister request itself carries) holds,
1111 // not whichever connection happened to attach the
1112 // instance last -- a shared instance may have several
1113 // owning connections at once, and crediting the release
1114 // to the wrong one either strands a reference nobody will
1115 // ever decrement, or lets one connection's deregister
1116 // silently consume another's hold.
1117 releaseScopedLocked(mid, cid);
1118 }
1119 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId)));

1121 dispatchExecute(std::move(env), reply);
1122 } else if (env.kind == "hello") {
1123 const std::uint32_t minV = _minVersion.load();
1124 const std::uint32_t maxV = _maxVersion.load();
1125 if (env.protocolVersion < minV || env.protocolVersion > maxV) {
1126 reply(::morph::wire::encode(::morph::wire::makeErr("protocol version unsupported", env.callId)));
1127 } else {
1128 std::string body;
1129 (void)glz::write_json(::morph::wire::ProtocolRange{.min = minV, .max = maxV}, body);
1130 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId, std::move(body))));
1131 }

1133 reply(::morph::wire::encode(::morph::wire::makeErr("unknown envelope kind: " + env.kind, env.callId)));
1134 }

1136 reply(::morph::wire::encode(::morph::wire::makeErr(exc.what(), env.callId)));

1140 // A single ordered gate sequence — limits, authorize, authenticate, lookup,
1141 // per-instance authorize — whose *order* is the security contract itself
1142 // (see docs/spec/security.md), so it is deliberately not broken up.
1143 // NOLINTNEXTLINE(readability-function-cognitive-complexity)
1144 void dispatchExecute(::morph::wire::Envelope env, std::function<void(std::string)> reply) {
1145 LimitPolicy limits;
1146 {
1147 std::scoped_lock const lock{_limitsMtx};
1148 limits = _limits;
1149 }
1150 // Cheap early shed, so an overloaded server does not pay for
1151 // authorize()/authenticate() on work it is about to refuse. Advisory
1152 // only — the binding check is the atomic reservation further below,
1153 // since this load and that increment are separated by authorization
1154 // and a registry lookup.
1155 if (limits.maxInFlightExecutes != 0 &&
1156 _inFlightExecutes.load(std::memory_order_relaxed) >= limits.maxInFlightExecutes) {
1157 reply(::morph::wire::encode(::morph::wire::makeErr("server busy", env.callId)));
1158 return;
1159 }
1160 if (!_authorizer->authorize(env.session, env.modelType, env.actionType)) {
1161 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
1162 return;
1163 }
1164 // Make the identity authoritative. A verifying authorizer returns the
1165 // principal it extracted from a valid token; we stamp it so model code
1166 // reading session::current()->principal can trust it. If authenticate()
1167 // returns nullopt the authorizer cannot vouch for the caller, so we CLEAR
1168 // the client-asserted principal rather than passing it through unverified.
1169 // This closes two holes: (1) the TOCTOU window where a token that passed
1170 // authorize() expires before authenticate() (worst case is now an empty
1171 // principal, never the attacker's claim), and (2) an authorize-only or
1172 // allow-all authorizer that never authenticates (the model never sees an
1173 // untrusted principal as authoritative). See docs/spec/security.md.
1174 if (auto verified = _authorizer->authenticate(env.session)) {
1175 env.session.principal = std::move(*verified);
1176 } else {
1177 env.session.principal.clear();
1178 }
1179 ::morph::exec::detail::ModelId const mid{env.modelId};
1180 std::shared_ptr<::morph::model::detail::IModelHolder> holder;
1181 std::string owner;
1182 bool known = false;
1183 {
1184 std::scoped_lock const lock{_regMtx};
1185 auto iter = _models.find(mid);
1186 if (iter != _models.end()) {
1187 holder = iter->second;
1188 if (auto ownerIter = _owners.find(mid); ownerIter != _owners.end()) {
1189 owner = ownerIter->second;
1190 known = true;
1191 }
1192 }
1193 }
1194 if (!holder) {
1195 reply(::morph::wire::encode(::morph::wire::makeErr("model not found", env.callId)));
1196 return;
1197 }
1198 // Per-instance (row-level) authorization. `authorize` above only saw the
1199 // model *type*; this consults the optional ownership hook with the target
1200 // instance id and its recorded owner. The default hook allows all, so
1201 // behaviour is unchanged unless an authorizer overrides it. env.session
1202 // now carries the verified principal (stamped just above), so an
1203 // ownership authorizer compares the recorded owner against it.
1204 if (known && !_authorizer->authorizeInstance(env.session, env.modelType, env.actionType, mid.v, owner)) {
1205 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
1206 return;
1207 }
1208 // Capture a strong self-reference so the server (and therefore
1209 // `_dispatcher`, which is a reference member) stays alive until this
1210 // strand task runs and delivers its reply. `handle()`'s task only holds
1211 // `self` until it enqueues onto the strand; without this capture the last
1212 // external shared_ptr could drop before the strand task executes, leaving
1213 // `_dispatcher` dangling (use-after-free) or the reply silently lost so a
1214 // client Completion hangs forever. See docs/spec/concurrency_and_lifetimes.md.
1215 // Concurrent in-flight executes: this is the same counter
1216 // LimitPolicy::maxInFlightExecutes checks above, reused (not duplicated)
1217 // as the executeInFlight metric's gauge, health()'s inFlight field, and
1218 // drainedWithin()'s drain-detection condition — one counter, four
1219 // consumers, never double-counted.
1220 //
1221 // Reserved with a compare-exchange rather than an unconditional
1222 // fetch_add, so `maxInFlightExecutes` is an actual bound. The advisory
1223 // check at the top of this function is a plain load, and authorize(),
1224 // authenticate() and the registry lookup all run between it and here —
1225 // long enough for every thread in the worker pool to observe the same
1226 // under-limit value and proceed, overshooting the cap by up to the
1227 // pool's width. That is exactly the burst the limit exists to prevent.
1228 // Reserving here, at the last point before the slot is genuinely taken,
1229 // needs no unwind on the early-return paths above.
1230 std::size_t inFlightAfterInc = 0;
1231 if (limits.maxInFlightExecutes != 0) {
1232 std::size_t current = _inFlightExecutes.load(std::memory_order_relaxed);
1233 for (;;) {
1234 if (current >= limits.maxInFlightExecutes) {
1235 reply(::morph::wire::encode(::morph::wire::makeErr("server busy", env.callId)));
1236 return;
1237 }
1238 // compare_exchange_weak refreshes `current` on failure, so a
1239 // losing thread re-tests the limit rather than forcing its way in.
1240 if (_inFlightExecutes.compare_exchange_weak(current, current + 1, std::memory_order_relaxed)) {
1241 inFlightAfterInc = current + 1;
1242 break;
1243 }
1244 }
1245 } else {
1246 inFlightAfterInc = _inFlightExecutes.fetch_add(1, std::memory_order_relaxed) + 1;
1247 }
1248 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeInFlight,
1249 static_cast<double>(inFlightAfterInc));
1250 auto self = shared_from_this();
1251 std::uint64_t const callId = env.callId;
1252
1253 // `finished` fires the caller's `reply` exactly once — whichever of the
1254 // timeout path or the strand path gets there first — and always
1255 // decrements the in-flight counter exactly once, regardless of which
1256 // path won. This preserves handle()'s reply-exactly-once contract even
1257 // though two independent paths can now race to resolve the same call.
1258 auto finished = std::make_shared<std::atomic_flag>();
1259 auto replySlot = std::make_shared<std::function<void(std::string)>>(std::move(reply));
1260 auto complete = [self, finished, replySlot](std::string msg) {
1261 if (!finished->test_and_set()) {
1262 auto const inFlightAfterDec = self->_inFlightExecutes.fetch_sub(1, std::memory_order_relaxed) - 1;
1263 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeInFlight,
1264 static_cast<double>(inFlightAfterDec));
1265 if (inFlightAfterDec == 0) {
1266 // Wakes any drainedWithin() waiter blocked on the in-flight
1267 // count reaching zero. Locking _drainMtx here (rather than
1268 // just notifying) closes the classic lost-wakeup window: it
1269 // guarantees this notify cannot land between a waiter's
1270 // predicate check and its wait_for() call.
1271 std::scoped_lock const drainLock{self->_drainMtx};
1272 self->_drainCv.notify_all();
1273 }
1274 (*replySlot)(std::move(msg));
1275 }
1276 };
1277
1278 detail::TimeoutScheduler::Handle timeoutHandle{};
1279 if (limits.executeTimeout.count() > 0) {
1280 std::scoped_lock const lock{_limitsMtx};
1281 if (_timeoutScheduler) {
1282 timeoutHandle = _timeoutScheduler->schedule(limits.executeTimeout, [complete, callId]() mutable {
1283 complete(::morph::wire::encode(::morph::wire::makeErr("timeout", callId)));
1284 });
1285 }
1286 }

1288 _strand.post(mid, [self, env = std::move(env), holder = std::move(holder), complete, timeoutHandle]() mutable {
1289 ::morph::exec::detail::ModelId const targetMid{env.modelId};
1290 auto const start = std::chrono::steady_clock::now();
1291 auto const spanId =
1292 ::morph::observe::detail::beginSpan(env.session.requestId, env.modelType, env.actionType);
1293 // Metrics and endSpan are recorded before `complete(...)` runs (below)
1294 // so a caller observing completion — via handle()'s reply or the
1295 // timeout path racing it — can never see the reply before this
1296 // dispatch's own instrumentation is recorded. This mirrors the
1297 // reply-exactly-once contract `complete` already provides: whichever
1298 // path wins the race, the metrics for *this* strand task are always
1299 // emitted here, exactly once, regardless of which path's reply the
1300 // caller actually receives.
1301 try {
1302 ::morph::session::detail::ScopedContext const scoped{env.session};
1303 // `dispatch` (registry.hpp, ActionDispatcher::registerAction's runner)
1304 // now throws morph::model::ValidationError when the decoded action
1305 // fails ActionValidator<Action>::ready(...), before Model::execute
1306 // runs. No special-casing is needed here: ValidationError derives
1307 // from std::runtime_error, so it is caught by the handler below and
1308 // turned into an ordinary `err` reply carrying its message and
1309 // callId, exactly like any other dispatch failure. See
1310 // docs/spec/core/registry.md.
1311 auto result = self->_dispatcher.dispatch(env.modelType, env.actionType, *holder, env.body);
1312 {
1313 std::scoped_lock const lock{self->_limitsMtx};
1314 if (self->_timeoutScheduler) {
1315 self->_timeoutScheduler->cancel(timeoutHandle);
1316 }
1317 }
1318 ::morph::observe::detail::endSpan(spanId, true);
1319 auto const elapsedMs =
1320 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - start).count();
1321 std::array<std::pair<std::string_view, std::string_view>, 2> const tags{
1322 {{"modelType", env.modelType}, {"actionType", env.actionType}}};
1323 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeLatencyMs, elapsedMs, tags);
1324 {
1325 std::scoped_lock const lock{self->_regMtx};
1326 self->_firstActionPending.erase(targetMid);
1327 }
1328 complete(::morph::wire::encode(::morph::wire::makeOk(env.callId, std::move(result))));
1329 } catch (const std::exception& exc) {
1330 {
1331 std::scoped_lock const lock{self->_limitsMtx};
1332 if (self->_timeoutScheduler) {
1333 self->_timeoutScheduler->cancel(timeoutHandle);
1334 }
1335 }
1336 ::morph::observe::detail::endSpan(spanId, false);
1337 auto const elapsedMs =
1338 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - start).count();
1339 std::array<std::pair<std::string_view, std::string_view>, 2> const tags{
1340 {{"modelType", env.modelType}, {"actionType", env.actionType}}};
1341 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeLatencyMs, elapsedMs, tags);
1342 ::morph::observe::detail::emitMetric(::morph::observe::Metric::executeErrors, 1.0, tags);
1343 {
1344 std::scoped_lock const lock{self->_regMtx};
1345 if (auto iter = self->_firstActionPending.find(targetMid);
1346 iter != self->_firstActionPending.end()) {
1347 self->_firstActionPending.erase(iter);
1348 self->_poisoned.insert(targetMid);
1349 }
1350 }
1351 complete(::morph::wire::encode(::morph::wire::makeErr(exc.what(), env.callId)));
1352 }
1353 });
1354 }

1357 ///
1358 /// Runs an internal monotonic counter through `detail::OpaqueIdGenerator`,
1359 /// so distinct calls never collide (the permutation is a bijection) but
1360 /// the returned values are not sequential. Skips the one counter value
1361 /// (if any) whose permutation is exactly `0` — `ModelId`'s reserved
1362 /// "unbound" sentinel (see `strand.hpp`) — which is possible in principle
1363 /// (the permutation is a bijection over the *entire* 64-bit domain, so
1364 /// exactly one input maps to `0`) but has probability 1-in-2^64 for a
1365 /// random key; guarded defensively rather than ever handed out.

1369 do {
1370 id = _idGen.permute(_nextId.fetch_add(1) + 1);
1371 } while (id == 0);
1372 return ::morph::exec::detail::ModelId{id};
1373 }
