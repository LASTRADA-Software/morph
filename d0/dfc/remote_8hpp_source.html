22/// @brief Server-side message handler that owns model instances and dispatches actions.
23///
24/// `RemoteServer` receives JSON envelopes (`morph::wire::Envelope`) from any
25/// transport (WebSocket, in-process simulation, …) and executes the corresponding
26/// model operations via an `ActionDispatcher`. Authorization is delegated to an
27/// `IAuthorizer` that defaults to allow-all.
28///
29/// @par Heap allocation requirement
30/// `RemoteServer` **must** be heap-allocated via `std::make_shared`. `handle()`
31/// captures `shared_from_this()` to prevent use-after-free when the worker pool
32/// outlives the server object.
33///
34/// @par Wire format
35/// All requests and replies are encoded as `morph::wire::Envelope` JSON. See
36/// `wire.hpp` for the field semantics. The `kind` field is the discriminator.
37class RemoteServer : public std::enable_shared_from_this<RemoteServer> {
38public:
39 /// @brief Constructs a server backed by @p workerPool with allow-all authorization.
40 ///
41 /// @param workerPool Pool used to process messages asynchronously.
42 /// @param dispatcher Action dispatcher; defaults to the process-level singleton.
43 /// @param registry Model factory registry; defaults to the process-level singleton.
44 explicit RemoteServer(
45 ::morph::exec::IExecutor& workerPool,
46 ::morph::model::detail::ActionDispatcher& dispatcher = ::morph::model::detail::defaultDispatcher(),
47 ::morph::model::detail::ModelRegistryFactory& registry = ::morph::model::detail::defaultRegistry())
48 : _pool{workerPool},
49 _strand{workerPool},
50 _dispatcher{dispatcher},
51 _registry{registry},
52 _authorizer{::morph::session::allowAllAuthorizer()} {}

74 /// @brief Asynchronously processes a JSON `Envelope` and calls @p reply with the response.
75 ///
76 /// The message is dispatched to the worker pool. @p reply is called exactly
77 /// once from the pool thread when processing completes.
78 ///
79 /// Thread-safe. Safe to call before the previous call's reply has been delivered.
80 ///
81 /// @param msg JSON-encoded `morph::wire::Envelope` (via `wire::encode`).
82 /// @param reply Callback invoked with the JSON-encoded reply envelope.
83 void handle(std::string msg, std::function<void(std::string)> reply) {
84 auto self = shared_from_this();
85 _pool.post(
86 [self, msg = std::move(msg), reply = std::move(reply)]() mutable { self->dispatchMessage(msg, reply); });
87 }

89 /// @brief Synchronously processes a JSON `Envelope` on the calling thread and returns the reply.
90 ///
91 /// Equivalent to `handle()` but never posts to the worker pool, so it is safe
92 /// to call from a thread that *is* the worker pool — for example, from a
93 /// `BridgeHandler` constructor invoked from inside an action handler.
94 ///
95 /// Only safe for control messages (`register`, `deregister`). An `execute`
96 /// envelope posts to the strand and produces its reply asynchronously, after
97 /// this synchronous call has already returned and destroyed the local reply
98 /// buffer the deferred callback would write into. To keep that from becoming a
99 /// dangling write, `execute` is rejected up front with an `err` reply.
100 ///
101 /// @param msg JSON-encoded `morph::wire::Envelope` (via `wire::encode`).
102 /// @return JSON-encoded reply envelope.
103 std::string handleInline(const std::string& msg) {
104 try {
105 auto env = ::morph::wire::decode(msg);
106 if (env.kind == "execute") {
107 return ::morph::wire::encode(::morph::wire::makeErr(
108 "handleInline does not support execute (reply is asynchronous)", env.callId));
109 }
110 } catch (const std::exception&) { // NOLINT(bugprone-empty-catch)
111 // Malformed input: fall through so dispatchMessage produces the
112 // canonical decode-error reply (avoids duplicating that path here).
113 }
114 std::string reply;
115 std::function<void(std::string)> capture = [&reply](std::string out) { reply = std::move(out); };
116 dispatchMessage(msg, capture);
117 return reply;
118 }

120 /// @brief Callable that supplies the action log to attach to a newly
121 /// registered instance, given its model type and `contextKey`.
122 ///
123 /// Return `nullptr` to register the instance with no log attached (e.g. for
124 /// model types or context keys the host app doesn't want journaled).
125 using LogProvider =
126 std::function<std::shared_ptr<::morph::journal::IActionLog>(std::string_view modelType, std::string_view contextKey)>;

128 /// @brief Installs @p provider, consulted on every `register` envelope whose
129 /// `contextKey` is non-empty.
130 ///
131 /// This is what closes the gap `IModelHolder::attachActionLog` leaves open
132 /// for remote topologies: `RemoteServer` owns the actual model instances for
133 /// every remote/simulated-remote client, so it is the only place that can
134 /// attach a log to them. Pass `nullptr` to remove a previously installed
135 /// provider (new registrations get no log). Thread-safe.
136 /// @param provider Callable invoked synchronously while handling `register`.
137 void setLogProvider(LogProvider provider) {
138 std::scoped_lock const lock{_logProviderMtx};
139 _logProvider = std::move(provider);
140 }

142private:
143 void dispatchMessage(const std::string& msg, std::function<void(std::string)>& reply) {
144 ::morph::wire::Envelope env;
145 try {
146 env = ::morph::wire::decode(msg);
147 } catch (const std::exception& exc) {
148 reply(::morph::wire::encode(::morph::wire::makeErr(exc.what())));
149 return;
150 }
151 try {
152 if (env.kind == "register") {
153 if (env.typeId.empty()) {
154 throw std::runtime_error("register requires a typeId");
155 }
156 auto holder = _registry.create(env.typeId);
157 if (!env.contextKey.empty()) {
158 LogProvider provider;
159 {
160 std::scoped_lock const lock{_logProviderMtx};
161 provider = _logProvider;
162 }
163 if (provider) {
164 if (auto log = provider(env.typeId, env.contextKey)) {
165 holder->attachActionLog(std::move(log), env.contextKey);
166 }
167 }
168 }
169 // Record the owner principal for per-instance authorization. We
170 // derive it from the *verified* identity of the register call
171 // (never the client's raw claim): a verifying authorizer returns
172 // the principal for the register envelope's session; an
173 // authorize-only / allow-all authorizer returns nullopt, so the
174 // instance is recorded as unowned (empty). This is what lets
175 // `authorizeInstance` later deny a different principal.
176 std::string owner;
177 if (auto verified = _authorizer->authenticate(env.session)) {
178 owner = std::move(*verified);
179 }
180 ::morph::exec::detail::ModelId const mid{_nextId.fetch_add(1) + 1};
181 {
182 std::scoped_lock const lock{_regMtx};
183 _models[mid] = std::move(holder);
184 _owners[mid] = std::move(owner);
185 }
186 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId, {}, mid.v)));
187 } else if (env.kind == "deregister") {
188 ::morph::exec::detail::ModelId const mid{env.modelId};
189 // Per-instance authorization also gates deregister: consult the
190 // hook with the recorded owner before destroying the instance.
191 // The default hook allows all, so unconfigured behaviour is
192 // unchanged; an ownership-enforcing authorizer can reject a
193 // caller tearing down an instance it does not own.
194 std::string owner;
195 bool known = false;
196 {
197 std::scoped_lock const lock{_regMtx};
198 auto iter = _owners.find(mid);
199 if (iter != _owners.end()) {
200 owner = iter->second;
201 known = true;
202 }
203 }
204 if (known && !_authorizer->authorizeInstance(env.session, {}, {}, mid.v, owner)) {
205 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
206 return;
207 }
208 {
209 std::scoped_lock const lock{_regMtx};
210 _models.erase(mid);
211 _owners.erase(mid);
212 }
213 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId)));
214 } else if (env.kind == "execute") {
215 dispatchExecute(std::move(env), reply);
216 } else {
217 reply(::morph::wire::encode(
218 ::morph::wire::makeErr("unknown envelope kind: " + env.kind, env.callId)));
219 }
220 } catch (const std::exception& exc) {
221 reply(::morph::wire::encode(::morph::wire::makeErr(exc.what(), env.callId)));
222 }
223 }

225 void dispatchExecute(::morph::wire::Envelope env, std::function<void(std::string)> reply) {
226 if (!_authorizer->authorize(env.session, env.modelType, env.actionType)) {
227 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
228 return;
229 }
230 // Make the identity authoritative. A verifying authorizer returns the
231 // principal it extracted from a valid token; we stamp it so model code
232 // reading session::current()->principal can trust it. If authenticate()
233 // returns nullopt the authorizer cannot vouch for the caller, so we CLEAR
234 // the client-asserted principal rather than passing it through unverified.
235 // This closes two holes: (1) the TOCTOU window where a token that passed
236 // authorize() expires before authenticate() (worst case is now an empty
237 // principal, never the attacker's claim), and (2) an authorize-only or
238 // allow-all authorizer that never authenticates (the model never sees an
239 // untrusted principal as authoritative). See docs/spec/security.md.
240 if (auto verified = _authorizer->authenticate(env.session)) {
241 env.session.principal = std::move(*verified);
242 } else {
243 env.session.principal.clear();
244 }
245 ::morph::exec::detail::ModelId const mid{env.modelId};
246 std::shared_ptr<::morph::model::detail::IModelHolder> holder;
247 std::string owner;
248 bool known = false;
249 {
250 std::scoped_lock const lock{_regMtx};
251 auto iter = _models.find(mid);
252 if (iter != _models.end()) {
253 holder = iter->second;
254 if (auto ownerIter = _owners.find(mid); ownerIter != _owners.end()) {
255 owner = ownerIter->second;
256 known = true;
257 }
258 }
259 }
260 if (!holder) {
261 reply(::morph::wire::encode(::morph::wire::makeErr("model not found", env.callId)));
262 return;
263 }
264 // Per-instance (row-level) authorization. `authorize` above only saw the
265 // model *type*; this consults the optional ownership hook with the target
266 // instance id and its recorded owner. The default hook allows all, so
267 // behaviour is unchanged unless an authorizer overrides it. env.session
268 // now carries the verified principal (stamped just above), so an
269 // ownership authorizer compares the recorded owner against it.
270 if (known && !_authorizer->authorizeInstance(env.session, env.modelType, env.actionType, mid.v, owner)) {
271 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
272 return;
273 }
274 // Capture a strong self-reference so the server (and therefore
275 // `_dispatcher`, which is a reference member) stays alive until this
276 // strand task runs and delivers its reply. `handle()`'s task only holds
277 // `self` until it enqueues onto the strand; without this capture the last
278 // external shared_ptr could drop before the strand task executes, leaving
279 // `_dispatcher` dangling (use-after-free) or the reply silently lost so a
280 // client Completion hangs forever. See docs/spec/concurrency_and_lifetimes.md.
281 auto self = shared_from_this();
282 _strand.post(mid, [self = std::move(self), env = std::move(env), holder = std::move(holder),
283 reply = std::move(reply)]() mutable {
284 try {
285 ::morph::session::detail::ScopedContext const scoped{env.session};
286 auto result = self->_dispatcher.dispatch(env.modelType, env.actionType, *holder, env.body);
287 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId, std::move(result))));
288 } catch (const std::exception& exc) {
289 reply(::morph::wire::encode(::morph::wire::makeErr(exc.what(), env.callId)));
290 }
291 });
292 }

294 ::morph::exec::IExecutor& _pool;
295 ::morph::exec::detail::StrandExecutor _strand;
296 ::morph::model::detail::ActionDispatcher& _dispatcher;
297 ::morph::model::detail::ModelRegistryFactory& _registry;
298 std::shared_ptr<::morph::session::IAuthorizer> _authorizer;
299 std::mutex _regMtx;
300 std::unordered_map<::morph::exec::detail::ModelId, std::shared_ptr<::morph::model::detail::IModelHolder>,
301 ::morph::exec::detail::ModelIdHash>
302 _models;
303 // Owner principal recorded per instance at register time, consulted by
304 // IAuthorizer::authorizeInstance on execute/deregister. Guarded by _regMtx
305 // (same lock as _models); empty string means "no recorded owner".
306 std::unordered_map<::morph::exec::detail::ModelId, std::string, ::morph::exec::detail::ModelIdHash> _owners;
307 std::atomic<uint64_t> _nextId{0};
308 std::mutex _logProviderMtx;
309 LogProvider _logProvider;

312/// @brief `IBackend` adapter that routes all calls through a `RemoteServer` as
313/// wire `Envelope` messages.
314///
315/// Intended for testing and in-process simulation of remote execution.
316/// `registerModel()` and `deregisterModel()` are processed inline on the calling
317/// thread via `RemoteServer::handleInline`. `execute()` is asynchronous: it sends
318/// the message through `RemoteServer::handle` and resolves the returned
319/// `Completion` when the reply arrives (there is no `std::promise` and no
320/// blocking wait).
321class SimulatedRemoteBackend : public detail::IBackend {
322public:
323 /// @brief Constructs the backend targeting @p server.
324 /// @param server The `RemoteServer` instance to forward calls to.
325 explicit SimulatedRemoteBackend(RemoteServer& server) : _server{server} {}
326
327 /// @brief Registers the model type on the server and returns its assigned id.
328 ///
329 /// Processed inline on the calling thread (no pool round-trip), so it is safe
330 /// to call from any thread including a worker in the same pool that backs the
331 /// `RemoteServer`. The @p factory argument is ignored — model construction is
332 /// delegated to the server's `ModelRegistryFactory`.
333 ///
334 /// @param typeId String type-id sent in the `register` message.
335 /// @return `ModelId` assigned by the server.
336 /// @throws std::runtime_error if the server replies with an error.
337 ::morph::exec::detail::ModelId registerModel(
338 const std::string& typeId,
339 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> /*factory*/) override {
340 return registerModelWithContext(typeId, {}, {});
341 }
342
343 /// @brief Registers the model type on the server, carrying @p contextKey across
344 /// the wire so the server's `RemoteServer::LogProvider` (if configured)
345 /// can attach an action log to the instance it creates.
346 ///
347 /// @p factory is still ignored — model construction is delegated to the
348 /// server's `ModelRegistryFactory`, same as `registerModel()`.
349 /// @param typeId String type-id sent in the `register` message.
350 /// @param contextKey Stable identity of the new instance; empty if none.
351 /// @return `ModelId` assigned by the server.
352 /// @throws std::runtime_error if the server replies with an error.
353 ::morph::exec::detail::ModelId registerModelWithContext(
354 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> /*factory*/,
355 std::string_view contextKey) override {
356 auto reply = ::morph::wire::decode(_server.handleInline(
357 ::morph::wire::encode(::morph::wire::makeRegister(typeId, std::string{contextKey}))));
358 if (reply.kind == "ok") {
359 return ::morph::exec::detail::ModelId{reply.modelId};
360 }
361 throw std::runtime_error("register failed: " + reply.message);
362 }

364 /// @brief Deregisters the model on the server. Processed inline; safe from any thread.
365 /// @param mid Id of the model to deregister.
366 void deregisterModel(::morph::exec::detail::ModelId mid) override {
367 (void)_server.handleInline(::morph::wire::encode(::morph::wire::makeDeregister(mid.v)));
368 }

370 /// @brief Serialises the action, sends it to the server, and returns a `Completion`.
371 ///
372 /// The `Completion` resolves when the server's reply is received and
373 /// deserialized. Callbacks are posted via @p cbExec. The session attached to
374 /// the call (via `Bridge::setDefaultSession()` or the per-call API) is
375 /// serialised into the envelope.
376 ///
377 /// @param mid Target model id on the server.
378 /// @param call Bundled action; `serializeAction` and `deserializeResult` are used.
379 /// @param cbExec Executor for delivering the completion callbacks.
380 /// @return Completion that resolves with the deserialized result or an error.
381 ::morph::async::Completion<std::shared_ptr<void>> execute(::morph::exec::detail::ModelId mid,
382 detail::ActionCall call,
383 ::morph::exec::IExecutor* cbExec) override {
384 auto state = std::make_shared<::morph::async::detail::CompletionState<std::shared_ptr<void>>>();
385 ::morph::async::Completion<std::shared_ptr<void>> comp{state, cbExec};
386 trackPending(state);
387
388 ::morph::wire::Envelope env;
389 env.kind = "execute";
390 env.modelId = mid.v;
391 env.modelType = call.modelTypeId;
392 env.actionType = call.actionTypeId;
393 env.body = call.serializeAction();
394 env.session = std::move(call.session);
395 auto deser = std::move(call.deserializeResult);
396
397 _server.handle(::morph::wire::encode(env),
398 [state, deser = std::move(deser)](const std::string& replyJson) mutable {
399 try {
400 auto reply = ::morph::wire::decode(replyJson);
401 if (reply.kind == "ok") {
402 state->setValue(deser(reply.body));
403 } else {
404 throw std::runtime_error(
405 reply.message.empty() ? "malformed reply" : reply.message);
406 }
407 } catch (...) {
408 state->setException(std::current_exception());
409 }
410 });
411 return comp;
412 }

414 /// @brief No-op — models live in `RemoteServer`, not locally.
415 void notifyBackendChanged() override {}

417 /// @brief Resolves every still-pending completion this backend produced with @p exc.
418 /// @param exc Exception delivered to every pending completion's error sink.
419 void cancelPending(const std::exception_ptr& exc) override {
420 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> snapshot;
421 {
422 std::scoped_lock const lock{_pendingMtx};
423 snapshot.swap(_pending);
424 }
425 for (auto& weak : snapshot) {
426 if (auto state = weak.lock()) {
427 state->setException(exc);
428 }
429 }
430 }
431
432private:
433 void trackPending(const std::shared_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>& state) {
434 std::scoped_lock const lock{_pendingMtx};
435 std::erase_if(_pending, [](const auto& weak) { return weak.expired(); });
436 _pending.emplace_back(state);
437 }
438
439 RemoteServer& _server;
440 std::mutex _pendingMtx;
441 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> _pending;
442};
