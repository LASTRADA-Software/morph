20/// @brief Server-side message handler that owns model instances and dispatches actions.
21///
22/// `RemoteServer` receives JSON envelopes (`morph::wire::Envelope`) from any
23/// transport (WebSocket, in-process simulation, …) and executes the corresponding
24/// model operations via an `ActionDispatcher`. Authorization is delegated to an
25/// `IAuthorizer` that defaults to allow-all.
26///
27/// @par Heap allocation requirement
28/// `RemoteServer` **must** be heap-allocated via `std::make_shared`. `handle()`
29/// captures `shared_from_this()` to prevent use-after-free when the worker pool
30/// outlives the server object.
31///
32/// @par Wire format
33/// All requests and replies are encoded as `morph::wire::Envelope` JSON. See
34/// `wire.hpp` for the field semantics. The `kind` field is the discriminator.
35class RemoteServer : public std::enable_shared_from_this<RemoteServer> {
36public:
37 /// @brief Constructs a server backed by @p workerPool with allow-all authorization.
38 ///
39 /// @param workerPool Pool used to process messages asynchronously.
40 /// @param dispatcher Action dispatcher; defaults to the process-level singleton.
41 /// @param registry Model factory registry; defaults to the process-level singleton.
42 explicit RemoteServer(
43 ::morph::exec::IExecutor& workerPool,
44 ::morph::model::detail::ActionDispatcher& dispatcher = ::morph::model::detail::defaultDispatcher(),
45 ::morph::model::detail::ModelRegistryFactory& registry = ::morph::model::detail::defaultRegistry())
46 : _pool{workerPool},
47 _strand{workerPool},
48 _dispatcher{dispatcher},
49 _registry{registry},
50 _authorizer{::morph::session::allowAllAuthorizer()} {}
51
52 /// @brief Constructs a server with a custom authorizer.
53 ///
54 /// @param workerPool Pool used to process messages asynchronously.
55 /// @param authorizer Authorizer consulted for every `execute` envelope.
56 /// @param dispatcher Action dispatcher; defaults to the process-level singleton.
57 /// @param registry Model factory registry; defaults to the process-level singleton.
58 RemoteServer(
59 ::morph::exec::IExecutor& workerPool, std::shared_ptr<::morph::session::IAuthorizer> authorizer,
60 ::morph::model::detail::ActionDispatcher& dispatcher = ::morph::model::detail::defaultDispatcher(),
61 ::morph::model::detail::ModelRegistryFactory& registry = ::morph::model::detail::defaultRegistry())
62 : _pool{workerPool},
63 _strand{workerPool},
64 _dispatcher{dispatcher},
65 _registry{registry},
66 _authorizer{std::move(authorizer)} {
67 if (!_authorizer) {
68 _authorizer = ::morph::session::allowAllAuthorizer();
69 }
70 }
71
72 /// @brief Asynchronously processes @p msg and calls @p reply with the response.
73 ///
74 /// The message is dispatched to the worker pool. @p reply is called exactly
75 /// once from the pool thread when processing completes.
76 ///
77 /// Thread-safe. Safe to call before the previous call's reply has been delivered.
78 ///
79 /// @param msg Pipe-delimited protocol message.
80 /// @param reply Callback invoked with the response string.
81 void handle(std::string msg, std::function<void(std::string)> reply) {
82 auto self = shared_from_this();
83 _pool.post(
84 [self, msg = std::move(msg), reply = std::move(reply)]() mutable { self->dispatchMessage(msg, reply); });
85 }
86
87 /// @brief Synchronously processes @p msg on the calling thread and returns the reply.
88 ///
89 /// Equivalent to `handle()` but never posts to the worker pool, so it is safe
90 /// to call from a thread that *is* the worker pool — for example, from a
91 /// `BridgeHandler` constructor invoked from inside an action handler.
92 ///
93 /// Only safe for control messages (`register`, `deregister`) — `execute`
94 /// messages still post to the strand and would return before the result is
95 /// produced. The implementation rejects `execute` to make this explicit.
96 ///
97 /// @param msg Pipe-delimited control message.
98 /// @return Reply string the async path would have produced.
99 std::string handleInline(const std::string& msg) {
100 std::string reply;
101 std::function<void(std::string)> capture = [&reply](std::string out) { reply = std::move(out); };
102 dispatchMessage(msg, capture);
103 return reply;
104 }
105
106private:
107 void dispatchMessage(const std::string& msg, std::function<void(std::string)>& reply) {
108 ::morph::wire::Envelope env;
109 try {
110 env = ::morph::wire::decode(msg);
111 } catch (const std::exception& exc) {
112 reply(::morph::wire::encode(::morph::wire::makeErr(exc.what())));
113 return;
114 }
115 try {
116 if (env.kind == "register") {
117 if (env.typeId.empty()) {
118 throw std::runtime_error("register requires a typeId");
119 }
120 auto holder = _registry.create(env.typeId);
121 ::morph::exec::detail::ModelId mid{_nextId.fetch_add(1) + 1};
122 {
123 std::scoped_lock lock{_regMtx};
124 _models[mid] = std::move(holder);
125 }
126 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId, {}, mid.v)));
127 } else if (env.kind == "deregister") {
128 {
129 std::scoped_lock lock{_regMtx};
130 _models.erase(::morph::exec::detail::ModelId{env.modelId});
131 }
132 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId)));
133 } else if (env.kind == "execute") {
134 dispatchExecute(std::move(env), reply);
135 } else {
136 reply(::morph::wire::encode(
137 ::morph::wire::makeErr("unknown envelope kind: " + env.kind, env.callId)));
138 }
139 } catch (const std::exception& exc) {
140 reply(::morph::wire::encode(::morph::wire::makeErr(exc.what(), env.callId)));
141 }
142 }
143
144 void dispatchExecute(::morph::wire::Envelope env, std::function<void(std::string)> reply) {
145 if (!_authorizer->authorize(env.session, env.modelType, env.actionType)) {
146 reply(::morph::wire::encode(::morph::wire::makeErr("unauthorized", env.callId)));
147 return;
148 }
149 ::morph::exec::detail::ModelId mid{env.modelId};
150 std::shared_ptr<::morph::model::detail::IModelHolder> holder;
151 {
152 std::scoped_lock lock{_regMtx};
153 auto iter = _models.find(mid);
154 if (iter != _models.end()) {
155 holder = iter->second;
156 }
157 }
158 if (!holder) {
159 reply(::morph::wire::encode(::morph::wire::makeErr("model not found", env.callId)));
160 return;
161 }
162 _strand.post(mid, [&disp = _dispatcher, env = std::move(env), holder = std::move(holder),
163 reply = std::move(reply)]() mutable {
164 try {
165 ::morph::session::detail::ScopedContext scoped{env.session};
166 auto result = disp.dispatch(env.modelType, env.actionType, *holder, env.body);
167 reply(::morph::wire::encode(::morph::wire::makeOk(env.callId, std::move(result))));
168 } catch (const std::exception& exc) {
169 reply(::morph::wire::encode(::morph::wire::makeErr(exc.what(), env.callId)));
170 }
171 });
172 }
173
174 ::morph::exec::IExecutor& _pool;
175 ::morph::exec::detail::StrandExecutor _strand;
176 ::morph::model::detail::ActionDispatcher& _dispatcher;
177 ::morph::model::detail::ModelRegistryFactory& _registry;
178 std::shared_ptr<::morph::session::IAuthorizer> _authorizer;
179 std::mutex _regMtx;
180 std::unordered_map<::morph::exec::detail::ModelId, std::shared_ptr<::morph::model::detail::IModelHolder>,
181 ::morph::exec::detail::ModelIdHash>
182 _models;
183 std::atomic<uint64_t> _nextId{0};
184};

186/// @brief `IBackend` adapter that routes all calls through a `RemoteServer` using a
187/// synchronous request-reply protocol.
188///
189/// Intended for testing and in-process simulation of remote execution. All
190/// `registerModel()`, `deregisterModel()`, and `execute()` calls are forwarded
191/// as protocol messages to the server; the calling thread blocks until the reply
192/// arrives via `std::promise`.
193class SimulatedRemoteBackend : public detail::IBackend {
194public:
195 /// @brief Constructs the backend targeting @p server.
196 /// @param server The `RemoteServer` instance to forward calls to.
197 explicit SimulatedRemoteBackend(RemoteServer& server) : _server{server} {}
198
199 /// @brief Registers the model type on the server and returns its assigned id.
200 ///
201 /// Processed inline on the calling thread (no pool round-trip), so it is safe
202 /// to call from any thread including a worker in the same pool that backs the
203 /// `RemoteServer`. The @p factory argument is ignored — model construction is
204 /// delegated to the server's `ModelRegistryFactory`.
205 ///
206 /// @param typeId String type-id sent in the `register` message.
207 /// @return `ModelId` assigned by the server.
208 /// @throws std::runtime_error if the server replies with an error.
209 ::morph::exec::detail::ModelId registerModel(
210 const std::string& typeId,
211 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()>) override {
212 auto reply = ::morph::wire::decode(
213 _server.handleInline(::morph::wire::encode(::morph::wire::makeRegister(typeId))));
214 if (reply.kind == "ok") {
215 return ::morph::exec::detail::ModelId{reply.modelId};
216 }
217 throw std::runtime_error("register failed: " + reply.message);
218 }
219
220 /// @brief Deregisters the model on the server. Processed inline; safe from any thread.
221 /// @param mid Id of the model to deregister.
222 void deregisterModel(::morph::exec::detail::ModelId mid) override {
223 (void)_server.handleInline(::morph::wire::encode(::morph::wire::makeDeregister(mid.v)));
224 }
225
226 /// @brief Serialises the action, sends it to the server, and returns a `Completion`.
227 ///
228 /// The `Completion` resolves when the server's reply is received and
229 /// deserialized. Callbacks are posted via @p cbExec. The session attached to
230 /// the call (via `Bridge::setDefaultSession()` or the per-call API) is
231 /// serialised into the envelope.
232 ///
233 /// @param mid Target model id on the server.
234 /// @param call Bundled action; `serializeAction` and `deserializeResult` are used.
235 /// @param cbExec Executor for delivering the completion callbacks.
236 /// @return Completion that resolves with the deserialized result or an error.
237 ::morph::async::Completion<std::shared_ptr<void>> execute(::morph::exec::detail::ModelId mid,
238 detail::ActionCall call,
239 ::morph::exec::IExecutor* cbExec) override {
240 auto state = std::make_shared<::morph::async::detail::CompletionState<std::shared_ptr<void>>>();
241 ::morph::async::Completion<std::shared_ptr<void>> comp{state, cbExec};
242 trackPending(state);
243
244 ::morph::wire::Envelope env;
245 env.kind = "execute";
246 env.modelId = mid.v;
247 env.modelType = call.modelTypeId;
248 env.actionType = call.actionTypeId;
249 env.body = call.serializeAction();
250 env.session = std::move(call.session);
251 auto deser = std::move(call.deserializeResult);
252
253 _server.handle(::morph::wire::encode(env),
254 [state, deser = std::move(deser)](const std::string& replyJson) mutable {
255 try {
256 auto reply = ::morph::wire::decode(replyJson);
257 if (reply.kind == "ok") {
258 state->setValue(deser(reply.body));
259 } else {
260 throw std::runtime_error(
261 reply.message.empty() ? "malformed reply" : reply.message);
262 }
263 } catch (...) {
264 state->setException(std::current_exception());
265 }
266 });
267 return comp;
268 }
269
270 /// @brief No-op — models live in `RemoteServer`, not locally.
271 void notifyBackendChanged() override {}
272
273 /// @brief Resolves every still-pending completion this backend produced with @p exc.
274 /// @param exc Exception delivered to every pending completion's error sink.
275 void cancelPending(const std::exception_ptr& exc) override {
276 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> snapshot;
277 {
278 std::scoped_lock lock{_pendingMtx};
279 snapshot.swap(_pending);
280 }
281 for (auto& weak : snapshot) {
282 if (auto state = weak.lock()) {
283 state->setException(exc);
284 }
285 }
286 }
287
288private:
289 void trackPending(const std::shared_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>& state) {
290 std::scoped_lock lock{_pendingMtx};
291 std::erase_if(_pending, [](const auto& weak) { return weak.expired(); });
292 _pending.emplace_back(state);
293 }
294
295 RemoteServer& _server;
296 std::mutex _pendingMtx;
297 std::vector<std::weak_ptr<::morph::async::detail::CompletionState<std::shared_ptr<void>>>> _pending;
298};
