18/// @brief Server-side message handler that owns model instances and dispatches actions.
19///
20/// `RemoteServer` receives pipe-delimited text messages (from a WebSocket, a
21/// simulated transport, or any other source) and executes the corresponding
22/// model operations via an `ActionDispatcher`.
23///
24/// @par Heap allocation requirement
25/// `RemoteServer` **must** be heap-allocated via `std::make_shared`. `handle()`
26/// captures `shared_from_this()` to prevent use-after-free when the worker pool
27/// outlives the server object.
28///
29/// @par Message protocol
30/// - `register|<typeId>` — creates a model instance; replies `ok|<modelId>`.
31/// - `deregister|<modelId>` — destroys the instance; replies `ok`.
32/// - `execute|<mid>|<modelTy>|<actionTy>|<body>` — dispatches an action; replies `ok|<result>`.
33/// - `execute|<callId>|<mid>|<modelTy>|<actionTy>|<body>` — same with a correlation id.
34/// - Any error replies with `err|<message>` or `err|<callId>|<message>`.
35class RemoteServer : public std::enable_shared_from_this<RemoteServer> {
36public:
37 /// @brief Constructs a server backed by @p workerPool.
38 ///
39 /// @param workerPool Pool used to process messages asynchronously.
40 /// @param dispatcher Action dispatcher; defaults to the process-level singleton.
41 /// @param registry Model factory registry; defaults to the process-level singleton.
42 explicit RemoteServer(
43 ::morph::exec::IExecutor& workerPool,
44 ::morph::model::detail::ActionDispatcher& dispatcher = ::morph::model::detail::defaultDispatcher(),
45 ::morph::model::detail::ModelRegistryFactory& registry = ::morph::model::detail::defaultRegistry())
46 : _pool{workerPool}, _strand{workerPool}, _dispatcher{dispatcher}, _registry{registry} {}
47
48 /// @brief Asynchronously processes @p msg and calls @p reply with the response.
49 ///
50 /// The message is dispatched to the worker pool. @p reply is called exactly
51 /// once from the pool thread when processing completes.
52 ///
53 /// Thread-safe. Safe to call before the previous call's reply has been delivered.
54 ///
55 /// @param msg Pipe-delimited protocol message.
56 /// @param reply Callback invoked with the response string.
57 void handle(std::string msg, std::function<void(std::string)> reply) {
58 auto self = shared_from_this();
59 _pool.post(
60 [self, msg = std::move(msg), reply = std::move(reply)]() mutable { self->dispatchMessage(msg, reply); });
61 }
62
63private:
64 void dispatchMessage(const std::string& msg, std::function<void(std::string)>& reply) {
65 // Split into at most 6 parts to support both the 5-part SimulatedRemote
66 // protocol and the 6-part Qt WebSocket protocol (which carries a callId).
67 auto parts = split(msg, '|', 6);
68 try {
69 if (parts[0] == "register") {
70 if (parts.size() < 2) {
71 throw std::runtime_error("register requires a typeId");
72 }
73 auto holder = _registry.create(parts[1]);
74 ::morph::exec::detail::ModelId mid{_nextId.fetch_add(1) + 1};
75 {
76 std::scoped_lock lock{_regMtx};
77 _models[mid] = std::move(holder);
78 }
79 reply("ok|" + std::to_string(mid.v));
80 } else if (parts[0] == "deregister") {
81 if (parts.size() < 2) {
82 throw std::runtime_error("deregister requires a modelId");
83 }
84 ::morph::exec::detail::ModelId mid{std::stoull(parts[1])};
85 std::scoped_lock lock{_regMtx};
86 _models.erase(mid);
87 reply("ok");
88 } else if (parts[0] == "execute") {
89 if (parts.size() == 6) {
90 // Qt WebSocket format: execute|callId|mid|modelTy|actionTy|body
91 std::string callId = parts[1];
92 ::morph::exec::detail::ModelId mid{std::stoull(parts[2])};
93 dispatchExecute(std::move(callId), mid, parts[3], parts[4], parts[5], reply);
94 } else if (parts.size() == 5) {
95 // SimulatedRemote format: execute|mid|modelTy|actionTy|body
96 ::morph::exec::detail::ModelId mid{std::stoull(parts[1])};
97 dispatchExecute("", mid, parts[2], parts[3], parts[4], reply);
98 } else {
99 throw std::runtime_error("execute requires 5 or 6 parts");
100 }
101 } else {
102 reply("err|bad msg type");
103 }
104 } catch (const std::exception& exc) {
105 reply(std::string{"err|"} + exc.what());
106 }
107 }
108
109 void dispatchExecute(std::string callId, ::morph::exec::detail::ModelId mid, const std::string& modelTy,
110 const std::string& actionTy, const std::string& body,
111 std::function<void(std::string)> reply) {
112 std::shared_ptr<::morph::model::detail::IModelHolder> holder;
113 {
114 std::scoped_lock lock{_regMtx};
115 auto iter = _models.find(mid);
116 if (iter != _models.end()) {
117 holder = iter->second;
118 }
119 }
120 if (!holder) {
121 if (callId.empty()) {
122 reply("err|model not found");
123 } else {
124 reply("err|" + callId + "|model not found");
125 }
126 return;
127 }
128 _strand.post(mid, [&disp = _dispatcher, callId = std::move(callId), modelTy, actionTy, body,
129 holder = std::move(holder), reply = std::move(reply)]() mutable {
130 try {
131 auto result = disp.dispatch(modelTy, actionTy, *holder, body);
132 if (callId.empty()) {
133 reply("ok|" + result);
134 } else {
135 reply("ok|" + callId + "|" + result);
136 }
137 } catch (const std::exception& exc) {
138 if (callId.empty()) {
139 reply(std::string{"err|"} + exc.what());
140 } else {
141 reply("err|" + callId + "|" + exc.what());
142 }
143 }
144 });
145 }

147 static std::vector<std::string> split(const std::string& src, char sep, int maxParts) {
148 std::vector<std::string> out;
149 std::size_t pos = 0;
150 while (static_cast<int>(out.size()) < maxParts - 1) {
151 auto found = src.find(sep, pos);
152 if (found == std::string::npos) {
153 break;
154 }
155 out.emplace_back(src.substr(pos, found - pos));
156 pos = found + 1;
157 }
158 out.emplace_back(src.substr(pos));
159 return out;
160 }
161
162 ::morph::exec::IExecutor& _pool;
163 ::morph::exec::detail::StrandExecutor _strand;
164 ::morph::model::detail::ActionDispatcher& _dispatcher;
165 ::morph::model::detail::ModelRegistryFactory& _registry;
166 std::mutex _regMtx;
167 std::unordered_map<::morph::exec::detail::ModelId, std::shared_ptr<::morph::model::detail::IModelHolder>,
168 ::morph::exec::detail::ModelIdHash>
169 _models;
170 std::atomic<uint64_t> _nextId{0};
171};

173/// @brief `IBackend` adapter that routes all calls through a `RemoteServer` using a
174/// synchronous request-reply protocol.
175///
176/// Intended for testing and in-process simulation of remote execution. All
177/// `registerModel()`, `deregisterModel()`, and `execute()` calls are forwarded
178/// as protocol messages to the server; the calling thread blocks until the reply
179/// arrives via `std::promise`.
180class SimulatedRemoteBackend : public detail::IBackend {
181public:
182 /// @brief Constructs the backend targeting @p server.
183 /// @param server The `RemoteServer` instance to forward calls to.
184 explicit SimulatedRemoteBackend(RemoteServer& server) : _server{server} {}
185
186 /// @brief Registers the model type on the server and returns its assigned id.
187 ///
188 /// Blocks until the server replies. The @p factory argument is ignored —
189 /// model construction is delegated to the server's `ModelRegistryFactory`.
190 ///
191 /// @param typeId String type-id sent in the `register` message.
192 /// @return `ModelId` assigned by the server.
193 /// @throws std::runtime_error if the server replies with an error.
194 ::morph::exec::detail::ModelId registerModel(
195 const std::string& typeId,
196 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()>) override {
197 std::promise<::morph::exec::detail::ModelId> prom;
198 auto fut = prom.get_future();
199 _server.handle("register|" + typeId, [&prom](const std::string& reply) {
200 if (reply.starts_with("ok|")) {
201 prom.set_value(::morph::exec::detail::ModelId{std::stoull(reply.substr(3))});
202 } else {
203 prom.set_exception(std::make_exception_ptr(std::runtime_error("register failed: " + reply)));
204 }
205 });
206 return fut.get();
207 }
208
209 /// @brief Deregisters the model on the server. Blocks until the reply arrives.
210 /// @param mid Id of the model to deregister.
211 void deregisterModel(::morph::exec::detail::ModelId mid) override {
212 std::promise<void> prom;
213 auto fut = prom.get_future();
214 _server.handle("deregister|" + std::to_string(mid.v), [&prom](const std::string&) { prom.set_value(); });
215 fut.get();
216 }
217
218 /// @brief Serialises the action, sends it to the server, and returns a `Completion`.
219 ///
220 /// The `Completion` resolves when the server's reply is received and
221 /// deserialized. Callbacks are posted via @p cbExec.
222 ///
223 /// @param mid Target model id on the server.
224 /// @param call Bundled action; `serializeAction` and `deserializeResult` are used.
225 /// @param cbExec Executor for delivering the completion callbacks.
226 /// @return Completion that resolves with the deserialized result or an error.
227 ::morph::async::Completion<std::shared_ptr<void>> execute(::morph::exec::detail::ModelId mid,
228 detail::ActionCall call,
229 ::morph::exec::IExecutor* cbExec) override {
230 auto state = std::make_shared<::morph::async::detail::CompletionState<std::shared_ptr<void>>>();
231 ::morph::async::Completion<std::shared_ptr<void>> comp{state, cbExec};
232
233 std::string body = call.serializeAction();
234 std::string msg =
235 "execute|" + std::to_string(mid.v) + "|" + call.modelTypeId + "|" + call.actionTypeId + "|" + body;
236 auto deser = std::move(call.deserializeResult);
237
238 _server.handle(std::move(msg), [state, deser = std::move(deser)](const std::string& reply) mutable {
239 try {
240 if (reply.starts_with("ok|")) {
241 state->setValue(deser(reply.substr(3)));
242 } else if (reply.starts_with("err|")) {
243 throw std::runtime_error(reply.substr(4));
244 } else {
245 throw std::runtime_error("malformed reply: " + reply);
246 }
247 } catch (...) {
248 state->setException(std::current_exception());
249 }
250 });
251 return comp;
252 }
253
254 /// @brief No-op — models live in `RemoteServer`, not locally.
255 void notifyBackendChanged() override {}
256
257private:
258 RemoteServer& _server;
259};
