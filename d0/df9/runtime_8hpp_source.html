21/// @brief How a `Runtime` bounds and reports its flushes.
22struct RuntimeOptions {
23 /// @brief How many times one Effect may run in a single flush before the flush is treated as a
24 /// write cycle, reported, and stopped.
25 std::size_t maxEffectRunsPerFlush = detail::kDefaultMaxEffectRunsPerFlush;
26 /// @brief How many flushes in a row are re-posted after an Effect threw. Each re-post runs what is still
27 /// queued, so a defect does not stall it until the next write; past the bound it waits for that
28 /// write, so an Effect that throws on every run cannot loop for ever. A write from outside a flush
29 /// starts a new count.
30 std::size_t maxThrowReposts = detail::kDefaultMaxThrowReposts;
31 /// @brief Called on the owner after every flush that processed a queued Effect; a frontend
32 /// schedules its redraw here. May be empty. An exception it throws is reported
33 /// (`detail::site::kAfterFlushThrew`) and dropped, so it never reaches the owner executor.
34 std::function<void()> afterFlush;
35};
36
37/// @brief The owner of a signal graph: every node takes one, and every operation runs on its owner.
38///
39/// Writes are batched. When the outermost batch ends with effects queued, the
40/// runtime posts **one** flush to the owner; no Effect ever runs inside `set()`.
41/// Destroying a Runtime while nodes made from it are alive is reported
42/// (`detail::site::kRuntimeOutlived`); the nodes stay valid and any flush does nothing.
43///
44/// Off the owner, `batch`, `widgetEvent` and `untracked` are reported (`detail::site::kOffOwner`) and
45/// refused: the body still runs, because the caller may need what it returns, but the runtime's state
46/// (batch depth, widget-event depth, tracking frame) is left alone, and each graph operation inside the
47/// body is checked, reported and refused on its own.
48class Runtime {
49public:
50 /// @param owner The executor the graph belongs to (the GUI or loop executor). It must be serial
51 /// (`IExecutor::isSerial`), queue what is posted and run it later, in order, as `Completion`
52 /// requires of its owner: one that runs a post inline would run Effects inside `set()`, and a pool
53 /// would run two flushes at once. Borrowed: it must outlive the Runtime and every node made from it.
54 /// @param options Flush bounds and after-flush hook.
55 /// @throws std::invalid_argument when @p owner is not serial.
56 explicit Runtime(exec::IExecutor& owner MORPH_LIFETIMEBOUND, RuntimeOptions options = {})
57 : _core{makeCore(owner)} {
58 _core->configure(detail::FlushBounds{.maxEffectRunsPerFlush = options.maxEffectRunsPerFlush,
59 .maxThrowReposts = options.maxThrowReposts},
60 std::move(options.afterFlush));
61 }
62
63 /// @brief Reports `detail::site::kRuntimeOutlived` when nodes made from it are still alive, then
64 /// detaches the core, so a flush still posted does nothing.
65 ~Runtime() {
66 if (_core->liveNodes() != 0) {
67 _core->report(detail::site::kRuntimeOutlived);
68 }
69 _core->detach();
70 }
71
72 Runtime(Runtime const&) = delete;
73 Runtime& operator=(Runtime const&) = delete;

77 /// @brief Runs @p fn as one batch: every write inside it is one flush.
78 /// @tparam F A callable taking no arguments.
79 /// @param fn The body. Writes made before it throws still flush.
80 /// @return Whatever @p fn returns.
81 template <typename F>
82 decltype(auto) batch(F&& fn) {
83 if (!_core->checkOwner()) {
84 return std::invoke(std::forward<F>(fn));
85 }
86 detail::BatchScope const scope{*_core};
87 return std::invoke(std::forward<F>(fn));
88 }
89
90 /// @brief Runs a widget callback as one batch. A flush that falls due while it runs, or that a nested
91 /// event loop inside it (a modal dialog) picks up, is deferred: the end of the outermost widget
92 /// event posts it. So a remount never destroys a widget whose native handler is on the stack, and a
93 /// nested loop does not spin on a flush it may not run. The callback is untracked, like
94 /// `untracked`: a widget callback is a transition, so one dispatched from inside an Effect's
95 /// run does not subscribe that Effect to what it reads.
96 /// @tparam F A callable taking no arguments.
97 /// @param fn The callback.
98 /// @return Whatever @p fn returns.
99 template <typename F>
100 decltype(auto) widgetEvent(F&& fn) {
101 if (!_core->checkOwner()) {
102 return std::invoke(std::forward<F>(fn));
103 }
104 detail::WidgetEventScope const event{*_core};
105 detail::TrackingFrame const untrackedFrame{*_core, nullptr};
106 return batch(std::forward<F>(fn));
107 }
108
109 /// @brief Runs @p fn with tracking off: reads inside it subscribe nothing.
110 /// @tparam F A callable taking no arguments.
111 /// @param fn The body.
112 /// @return Whatever @p fn returns.
113 template <typename F>
114 decltype(auto) untracked(F&& fn) const {
115 if (!_core->checkOwner()) {
116 return std::invoke(std::forward<F>(fn));
117 }
118 detail::TrackingFrame const frame{*_core, nullptr};
119 return std::invoke(std::forward<F>(fn));
120 }
121
122 /// @brief The owner executor.
123 /// @return The executor passed at construction.
124 [[nodiscard]] exec::IExecutor& owner() const noexcept { return _core->owner(); }
125
126 /// @brief Whether a flush is posted and has not run yet.
127 /// @return True between the first write that queued an Effect and the flush.
128 [[nodiscard]] bool isFlushRequested() const noexcept { return _core->isFlushRequested(); }
129
130 /// @brief The shared core every node holds.
131 /// @return The core.
132 [[nodiscard]] std::shared_ptr<detail::RuntimeCore> const& core() const noexcept { return _core; }
133
134private:
135 [[nodiscard]] static std::shared_ptr<detail::RuntimeCore> makeCore(exec::IExecutor& owner) {
136 if (!owner.isSerial()) {
137 throw std::invalid_argument{"morph::reactive::Runtime: the owner executor must be serial"};
138 }
139 return std::make_shared<detail::RuntimeCore>(owner);
140 }
141
142 std::shared_ptr<detail::RuntimeCore> _core;
143};
