23/// @brief A retained widget. A new widget is visible, enabled, content-sized, not draggable and not a drop target.
24///
25/// Its destructor detaches it from its parent and frees its native resources; no handler of the widget runs after
26/// the destructor returns. A handler may destroy its own widget: the backend keeps the handler alive until that call
27/// returns, for example by calling a copy. A backend may assume a handler, a drop's `accepts` predicate included,
28/// never throws: the mount catches and reports what one throws. Every setter takes UTF-8.
29///
30/// A setter called with the value the widget already shows changes nothing: a text input keeps its cursor, its
31/// selection and an input method's composition. Input widgets are controlled: a user's action changes what the
32/// widget shows, as the toolkit does, and reports the request through the handler; the mount then shows what the
33/// application's slot says, in a turn posted after the request's flush, so a refused request snaps back.
34class Widget {
35public:
36 Widget() = default;
37 virtual ~Widget() = default;
38 Widget(Widget const&) = delete;
39 Widget& operator=(Widget const&) = delete;
40 Widget(Widget&&) = delete;
41 Widget& operator=(Widget&&) = delete;
42
43 /// @brief Shows or hides the widget and everything inside it.
44 /// @param visible Whether it is shown.
45 virtual void setVisible(bool visible) = 0;
46
47 /// @brief Enables or disables input on the widget and everything inside it.
48 /// @param enabled Whether it accepts input.
49 virtual void setEnabled(bool enabled) = 0;
50
51 /// @brief Sets the size request.
52 /// @param hints The request in both dimensions.
53 virtual void setLayout(LayoutHints const& hints) = 0;
54
55 /// @brief Makes the widget draggable, carrying @p key, or not draggable.
56 /// @param key Engaged: the key a drop delivers. `nullopt`: not draggable.
57 virtual void setDragKey(std::optional<Key> const& key) = 0;
58
59 /// @brief Makes the widget a drop target.
60 /// @param accepts Whether a dragged key may be dropped here; the backend asks before highlighting.
61 /// @param onDrop Called with the key on a drop that `accepts` allowed.
62 virtual void setDropHandler(std::function<bool(Key const&)> accepts, std::function<void(Key)> onDrop) = 0;
63};

72 /// @param child One of this container's children.
73 /// @param index The new position; past the end means last.
74 virtual void moveChild(Widget& child, std::size_t index) = 0;

280/// @brief Keyed rows under column headers, with selection and activation by key.
281///
282/// Its children are the rows, each a horizontal stack whose children are the cells in column order. A hidden cell
283/// keeps its column: the cells after it stay under their own headers.
284///
285/// Selection is kept by key. The widget keeps the keys last requested by `setSelection` or selected by the user, and
286/// marks the current rows that have one of them: a key with no row marks nothing and stays requested, so a row that
287/// arrives with it later is marked, a row that goes is unmarked, and a reorder keeps the marks. Rows and selection
288/// may therefore arrive in either order. The selection mode limits what the user can select, not what `setSelection`
289/// marks.
290///
291/// A user's selection replaces the requested keys, and the widget shows it whether or not the application binds the
292/// selection. `onSelectionChange` then gets exactly the keys of the existing rows now selected, in any mode: a key
293/// still waiting for its row is dropped by a user's change, while `setSelection` keeps it waiting. It reports the
294/// user's selections only: no setter, and no row arriving or going, calls it.
295class TableWidget : public ContainerWidget {
296public:
297 /// @brief Sets the column headers and widths.
298 /// @param columns The columns, in order.
299 virtual void setColumns(std::vector<TableColumn> const& columns) = 0;
300
301 /// @brief Sets how many rows the user can select.
302 /// @param mode None, one, or any number.
303 virtual void setSelectionMode(SelectionMode mode) = 0;
304
305 /// @brief Tells the table which key a row stands for; called once per row, right after it is built. Marks the row
306 /// when its key is requested.
307 /// @param row One of this table's rows.
308 /// @param key The row's key.
309 virtual void setRowKey(Widget& row, Key const& key) = 0;
310
311 /// @brief Requests the rows with these keys, and marks those it has. Never calls the `onSelectionChange` handler.
312 /// @param keys The selected keys.
313 virtual void setSelection(std::vector<Key> const& keys) = 0;
314
315 /// @brief Sets what a user's selection change calls, with the keys of every row then selected.
316 /// @param onSelectionChange The handler; empty does nothing.
317 virtual void setOnSelectionChange(std::function<void(std::vector<Key>)> onSelectionChange) = 0;
318
319 /// @brief Sets what activating a row (Enter or a double click) calls, with its key.
320 /// @param onActivate The handler; empty does nothing.
321 virtual void setOnActivate(std::function<void(Key)> onActivate) = 0;
322};
