23/// @brief A retained widget. A new widget is visible, enabled, content-sized, not draggable and not a drop target.
24///
25/// Its destructor detaches it from its parent and frees its native resources; no handler of the widget runs after
26/// the destructor returns. A handler may destroy its own widget: the backend keeps the handler alive until that call
27/// returns, for example by calling a copy. A backend may assume a handler, a drop's `accepts` predicate included,
28/// never throws: the mount catches and reports what one throws. Every setter takes UTF-8.
29///
30/// A setter called with the value the widget already shows changes nothing: a text input keeps its cursor, its
31/// selection and an input method's composition. Input widgets are controlled: a user's action changes what the
32/// widget shows, as the toolkit does, and reports the request through the handler; the mount then shows what the
33/// application's slot says, in a turn posted after the request's flush, so a refused request snaps back.
34class Widget {
35public:
36 Widget() = default;
37 virtual ~Widget() = default;
38 Widget(Widget const&) = delete;
39 Widget& operator=(Widget const&) = delete;
40 Widget(Widget&&) = delete;
41 Widget& operator=(Widget&&) = delete;
42
43 /// @brief Shows or hides the widget and everything inside it.
44 /// @param visible Whether it is shown.
45 virtual void setVisible(bool visible) = 0;
46
47 /// @brief Enables or disables input on the widget and everything inside it.
48 /// @param enabled Whether it accepts input.
49 virtual void setEnabled(bool enabled) = 0;
50
51 /// @brief Sets the size request.
52 /// @param hints The request in both dimensions.
53 virtual void setLayout(LayoutHints const& hints) = 0;
54
55 /// @brief Makes the widget draggable, carrying @p key, or not draggable.
56 /// @param key Engaged: the key a drop delivers. `nullopt`: not draggable.
57 virtual void setDragKey(std::optional<Key> const& key) = 0;
58
59 /// @brief Makes the widget a drop target.
60 /// @param accepts Whether a dragged key may be dropped here; the backend asks before highlighting.
61 /// @param onDrop Called with the key on a drop that `accepts` allowed.
62 virtual void setDropHandler(std::function<bool(Key const&)> accepts, std::function<void(Key)> onDrop) = 0;
63
64 /// @brief Sets the name assistive technology reads for the widget.
65 /// @param name The accessible name; empty leaves the widget's own text as its name.
66 virtual void setAccessibleName(std::string_view name) = 0;
67
68 /// @brief Sets the role assistive technology announces for the widget.
69 /// @param role The role, such as `"heading"`; empty leaves the kind's own role.
70 virtual void setAccessibleRole(std::string_view role) = 0;
71
72 /// @brief Sets the stable identifier an automated test finds the widget by.
73 /// @param testId The identifier; empty means none.
74 virtual void setTestId(std::string_view testId) = 0;
75
76 /// @brief Sets the text shown while the pointer rests on the widget.
77 /// @param text The tooltip; empty shows none.
78 virtual void setTooltip(std::string_view text) = 0;
79
80 /// @brief Styles the widget and everything inside it with a named token set of the theme. May be called again:
81 /// each call restyles the widget and everything inside it.
82 /// @param surface The surface's name; empty is the theme's default.
83 virtual void setSurface(std::string_view surface) = 0;
84
85 /// @brief Declares the keyboard chords the widget handles while focus is inside it.
86 ///
87 /// A chord pressed while focus is inside several widgets that declare it goes to the innermost of them only.
88 /// @param chords The chords, each modifiers joined to a key name, such as `"Ctrl+S"`.
89 /// @param onChord Called with the chord that was pressed.
90 virtual void setKeys(std::vector<std::string> const& chords, std::function<void(std::string)> onChord) = 0;
91
92 /// @brief Moves keyboard focus to the widget.
93 virtual void focus() = 0;
94
95 /// @brief Calls @p onHover each time the pointer comes to rest over the widget. A backend without a pointer, or
96 /// one that does not report hovering, keeps the default, which never calls it: hovering is a hint, such as
97 /// a prefetch, never something a widget relies on.
98 /// @param onHover The handler; empty stops calling one.
99 // NOLINTNEXTLINE(performance-unnecessary-value-param): a sink every override moves from; the default drops it.
100 virtual void setOnHover(std::function<void()> onHover) { static_cast<void>(onHover); }
101};

493/// @brief Children laid out row-major in columns.
494class GridWidget : public ContainerWidget {
495public:
496 /// @brief Sets the column count; the columns share the width.
497 /// @param columns At least one.
498 virtual void setColumns(int columns) = 0;
499
500 /// @brief Sets the column tracks in place of a column count; the mount calls it once, before any child, and then
501 /// never calls `setColumns`.
502 ///
503 /// Fixed tracks take their amount, content tracks their widest cell that spans only them, and stretch tracks
504 /// share what is left by weight. A spanning cell covers its tracks and the gaps between them, and sizes none.
505 /// Across its track, a text cell's text and a label-width cell sit where the track's `align` says (the first
506 /// track's, for a spanning cell); a container cell fills the track.
507 /// @param tracks At least one.
508 virtual void setTracks(std::vector<GridTrack> const& tracks) = 0;
509
510 /// @brief Makes the column count follow the grid's width: the columns shown are
511 /// `max(1, min(columns, floor((W + g) / (m * u + g) + e)))`, W the width inside the grid, g the gap, m the
512 /// least width given here, u the unit and e 1e-6, so a width that holds exactly n columns, computed in
513 /// floating point, gives n. The natural width is unchanged (`columns` times the widest cell, a span divided
514 /// out); the natural height is that of the columns shown at the width the grid was last laid out at, so a
