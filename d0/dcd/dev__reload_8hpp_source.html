32/// @brief The loaded documents a host mounts from, by screen id.
33using DocumentMap = std::map<std::string, std::shared_ptr<Document const>, std::less<>>;
34
35/// @brief The ids of the documents (screens and the app document) whose files differ between two manifests: changed,
36/// added or removed. Themes, message catalogues and components are not documents; a document that names a
37/// changed component names it by its new hash, so it changes too.
38/// @param before The manifest the client had.
39/// @param after The manifest it has now.
40/// @return The ids, sorted and each once.
41[[nodiscard]] inline std::vector<std::string> changedDocuments(Manifest const& before, Manifest const& after) {
42 auto const documents = [](Manifest const& manifest) {
43 std::map<std::string, std::string, std::less<>> hashes;
44 for (ManifestFile const& file : manifest.files) {
45 if (file.kind == kScreenFile || file.kind == kAppFile) {
46 hashes.insert_or_assign(file.id, file.hash);
47 }
48 }
49 return hashes;
50 };
51 auto const old = documents(before);
52 auto const now = documents(after);
53 std::vector<std::string> changed;
54 for (auto const& [document, hash] : now) {
55 auto const found = old.find(document);
56 if (found == old.end() || found->second != hash) {
57 changed.push_back(document);
58 }
59 }
60 for (auto const& [document, hash] : old) {
61 static_cast<void>(hash);
62 if (!now.contains(document)) {
63 changed.push_back(document);
64 }
65 }
66 std::ranges::sort(changed);
67 return changed;
68}

89/// @brief Loads a changed screen's new text and checks it as a served screen document: its id may be served, it
90/// parses,
91/// and the document claims the id it is served under.
92/// @param screenId The id the screen is served under.
93/// @param text The screen's new document text.
94/// @param options How the document loads: the catalog facts.
95/// @return The loaded document, or why it is refused, naming @p screenId when the loader named no document.
96[[nodiscard]] inline std::expected<Document, Refusal> loadReloadedScreen(std::string const& screenId,
97 std::string const& text,
98 LoadOptions const& options) {
99 if (auto refusal = servedScreenRefusal(screenId)) {
100 return std::unexpected(std::move(*refusal));
101 }
102 auto loaded = loadDocumentText(text, options);
103 if (!loaded.has_value()) {
104 Refusal refusal = std::move(loaded.error());
105 if (refusal.document.empty()) {
106 refusal.document = screenId;
107 }
108 return std::unexpected(std::move(refusal));
109 }
110 if (auto refusal = servedDocumentRefusal(screenId, loaded->id)) {
111 return std::unexpected(std::move(*refusal));
112 }
113 return std::move(*loaded);
114}

116} // namespace detail

118/// @brief Loads the changed documents of a new bundle and remounts the mounted screens that show them.
119///
120/// Each changed screen document is loaded with @p options. A document that does not load is refused and keeps the old
121/// document, in @p documents and on every screen showing it. A document that loads is mounted, through
122/// `DocumentScreenHost::remount`, on every screen of @p mounted showing it, which keeps each state of the screen's own
123/// scope declared under the same name and type. A screen showing a draft (`DocumentScreenHost::mountDraft`) is left as
124/// it is. A remount the interpreter refuses keeps that screen on the old document, and keeps the old document in @p
125/// documents. A changed id that is the chat panel's, and a document that names another id than its screen's, are
126/// refused the same way. A changed id the bundle no longer holds, and the app document, which is the session's root
127/// scope and not a screen (`DocumentScreenHost::mountApp`), change nothing here.
128/// @param changed The ids of the changed documents (`changedDocuments`).
129/// @param bundle The new bundle's documents.
130/// @param options How the documents load: the catalog facts.
131/// @param documents The host's documents, updated in place.
132/// @param host The host that mounted every screen of @p mounted.
133/// @param mounted The mounted screens.
134/// @return The screens remounted, and the refusals.
135[[nodiscard]] inline ReloadOutcome reloadDocuments(std::span<std::string const> changed, Bundle const& bundle,
136 LoadOptions const& options, DocumentMap& documents,
137 DocumentScreenHost& host, std::span<MountedScreen const> mounted) {
138 ReloadOutcome outcome;
139 for (std::string const& screenId : changed) {
140 auto const text = bundle.documents.find(screenId);
141 if (screenId == "app" || text == bundle.documents.end()) {
142 continue;
143 }
144 auto loaded = detail::loadReloadedScreen(screenId, text->second, options);
145 if (!loaded.has_value()) {
146 outcome.refused.push_back(std::move(loaded.error()));
147 continue;
148 }
149 auto const document = std::make_shared<Document const>(std::move(*loaded));
150 bool kept = true;
151 for (MountedScreen const& screen : mounted) {
152 // A draft stays mounted: it is checked against the new bundle when the person previews or saves it, and
153 // discarding it mounts the document the map holds then.
154 if (screen.screenId != screenId || screen.handle == nullptr ||
155 DocumentScreenHost::draftedNow(*screen.handle)) {
156 continue;
157 }
158 if (auto remounted = host.remount(*screen.handle, document); !remounted) {
159 outcome.refused.push_back(
160 Refusal{.document = screenId, .path = {}, .reason = std::move(remounted.error().reason)});
161 kept = false;
162 continue;
163 }
164 if (std::ranges::find(outcome.remounted, screenId) == outcome.remounted.end()) {
165 outcome.remounted.push_back(screenId);
166 }
167 }
168 if (kept) {
169 documents.insert_or_assign(screenId, document);
170 }
171 }
172 return outcome;
173}
