5/// @file forms/sections.hpp
6/// @brief Unordered sections: N independently editable action drafts on one
7/// screen, each with its own readiness gate and no sequencing.
8///
9/// The sibling to `morph::flows::FlowSession`. A wizard has one active step and
10/// throws on a field belonging to any other; a section set has no active
11/// section at all -- every declared section is editable at every moment, and
12/// each dispatches on its own as soon as its draft validates.
13///
14/// Choose this when order carries no meaning (tabs, cards, a settings page) and
15/// `FlowSession` when it does. Like flows, this is additive metadata and
16/// dispatch bookkeeping over the ordinary `BridgeHandler<Model>::execute<A>()`
17/// path: no new wire format, no new execution mode.
18/// See docs/spec/forms/sections.md.

41/// @brief One section of a `SectionGroup`: a registered action, a display
42/// title, and zero or more `Bind` prefill declarations.
43/// @tparam Action Registered action type (`BRIDGE_REGISTER_ACTION`) this section fires.
44/// @tparam Title Human title for the section (tab label / card header).
45/// @tparam Binds Zero or more `Bind<Field, Path>` prefill declarations.
46template <typename Action, morph::forms::FixedString Title, typename... Binds>
47struct Section {
48 /// @brief The section's action type.
49 using action = Action;
50
51 /// @brief Tuple of this section's `Bind<...>` prefill declarations (possibly empty).
52 using binds = std::tuple<Binds...>;
53
54 /// @brief The section's display title.
55 /// @return The declared title.
56 [[nodiscard]] static constexpr std::string_view title() noexcept { return Title.view(); }
57};

59/// @brief An unordered set of `Section`s sharing one screen.
60/// @tparam Title Human title for the whole group.
61/// @tparam Sections One or more `Section<Action, Title, Binds...>` types.
62template <morph::forms::FixedString Title, typename... Sections>
63struct SectionGroup {
64 /// @brief Tuple of this group's `Section<...>` types.
65 using sections = std::tuple<Sections...>;
66
67 /// @brief The group's display title.
68 /// @return The declared title.
69 [[nodiscard]] static constexpr std::string_view title() noexcept { return Title.view(); }
70};

72/// @brief Traits specialisation mapping a `SectionGroup` type to its string type-id.
73///
74/// Specialise via `BRIDGE_REGISTER_SECTION_GROUP` rather than by hand. The
75/// default is a forward declaration — using it without a specialisation is an
76/// incomplete-type error. The schema needs a stable name for the group;
77/// deriving one from the C++ type would tie the wire format to a mangled name.
78/// @tparam G Concrete `SectionGroup<...>` type.
79template <typename G>
80struct SectionGroupTraits; // forward — specialise or use BRIDGE_REGISTER_SECTION_GROUP

82/// @brief Generates the `s-*` JSON document for section group @p G.
83///
84/// Emits `s-id`, `s-title`, and an `s-sections` array; each entry carries
85/// `action` (the section's registered action type-id), `title`, and — only when
86/// the section declares at least one `Bind` — a `prefill` object mapping field
87/// name to `"<ActionTypeId>.<field>"` path.
88///
89/// Deliberately emits no index or order field. A renderer arranges the sections
90/// itself, and a position in the document would suggest a sequence this type
91/// does not have.
92/// @tparam G Concrete `SectionGroup<Title, Sections...>` type.
93/// @return The group's JSON document. Empty string only if glaze's own JSON
94/// writer fails on the assembled DOM (schema generation never throws).
95// NOLINTBEGIN(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) -- glaze DOM requires operator[]
96template <typename G>
97[[nodiscard]] std::string sectionGroupSchemaJson() {
98 glz::generic_u64 dom{};
99 dom["s-id"] = std::string{SectionGroupTraits<G>::typeId()};
100 dom["s-title"] = std::string{G::title()};
101
102 glz::generic_u64::array_t sections{};
103 detail::forEachTupleElement<typename G::sections>([&]<typename SectionT, std::size_t I>() {
104 static_cast<void>(I);
105 glz::generic_u64 entry{};
106 entry["action"] = std::string{::morph::model::ActionTraits<typename SectionT::action>::typeId()};
107 entry["title"] = std::string{SectionT::title()};
108 if constexpr (std::tuple_size_v<typename SectionT::binds> != 0) {
109 detail::emitBindsInto<typename SectionT::binds>(entry["prefill"]);
110 }
111 sections.emplace_back(std::move(entry));
112 });
113 dom["s-sections"] = sections;
114
115 return glz::write_json(dom).value_or(std::string{});
116}
117// NOLINTEND(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)

119/// @brief Drives N independently editable action drafts on one screen.
120///
121/// Each section accumulates its own draft through `set<>` and dispatches
122/// through the handler as soon as `ActionValidator<A>::ready` accepts it.
123/// There is no active section and no sequence: a field belonging to any
124/// declared section may be set at any time, in any order. This is the whole
125/// difference from `morph::flows::FlowSession`, which throws on a field
126/// outside its current step.
127///
128/// A ready section re-fires on every subsequent `set<>`. There is no latch and
129/// no coalescing, matching `FlowSession`; a caller that wants one request per
130/// pause debounces on its own side, where it knows what a pause means.
131/// @tparam Model The model the handler is bound to.
132/// @tparam Sections One or more `Section<Action, Title, Binds...>` types.
133template <typename Model, typename... Sections>
134class SectionSet {
135 static_assert(sizeof...(Sections) > 0, "SectionSet: a group needs at least one section");
136 static_assert(::morph::forms::detail::AllDistinct<typename Sections::action...>::value,
137 "SectionSet: section action types must be pairwise distinct");
138
139public:
140 /// @brief Constructs a section set dispatching through @p handler.
141 /// @param handler Handler every section dispatches through. Must outlive
142 /// this `SectionSet`.
143 /// @param onError Optional callback invoked when a section's dispatch
144 /// fails. When absent, the error is logged via
145 /// `morph::log::logError`. Stored and invoked for this
146 /// object's whole lifetime, so anything the callable refers
147 /// to must outlive it.
148 explicit SectionSet(::morph::bridge::BridgeHandler<Model>& handler MORPH_LIFETIMEBOUND,
149 std::function<void(std::exception_ptr)> onError MORPH_LIFETIMEBOUND = nullptr)
150 : _handler{handler}, _onError{std::move(onError)} {}
151
152 SectionSet(const SectionSet&) = delete;
153 SectionSet& operator=(const SectionSet&) = delete;
154 SectionSet(SectionSet&&) = delete;
155 SectionSet& operator=(SectionSet&&) = delete;
156
157 /// @brief Stops every callback this set installed from being delivered.
158 ///
159 /// There is nothing to detach: a section's continuations are owned by the
160 /// in-flight dispatch, not held in a map this object could remove itself
161 /// from. `_callbacks` gates each one on a token it checks before touching
162 /// `this`, so a completion resolving after this object is gone finds the
163 /// token stopped and returns without dereferencing anything.
164 ///
165 /// `requestStop()` is called explicitly rather than left to the member's
166 /// own destruction, even though `_callbacks` is declared last: members are
167 /// destroyed only *after* the destructor body runs, so a body that later
168 /// grew a call pumping an event loop would otherwise deliver into a
169 /// half-dead object. The body does not do that today; stopping first keeps
170 /// it correct if one is ever added.
171 ///
172 /// How strong the gate is depends on which thread destroys this object,
173 /// exactly as `CallbackScope`'s "Boundary of the guarantee" describes.
174 /// Destroying it off the delivery thread is advisory only, and that caller
175 /// owns its own synchronisation.
176 ~SectionSet() { _callbacks.requestStop(); }
177
178 /// @brief Sets one field of its section's draft, dispatching that section
179 /// if the draft is now ready.
180 ///
181 /// Unlike `FlowSession::set<>` this imposes no ordering: the field's action
182 /// need only be one of the declared sections. Readiness is evaluated on a
183 /// copy taken under the lock, so the dispatch happens with the lock
184 /// released and a concurrent edit to another section cannot block on it.
185 /// @tparam FieldPtr Pointer-to-data-member of a declared section's action struct.
186 /// @param value New value for the field.
187 template <auto FieldPtr>
188 void set(::morph::bridge::detail::MemberPointerTraits<decltype(FieldPtr)>::ValueType value) {
189 using A = ::morph::bridge::detail::MemberPointerTraits<decltype(FieldPtr)>::ClassType;
190 static_assert((std::is_same_v<A, typename Sections::action> || ...),
191 "SectionSet::set<>: the field's action is not a section of this group");
192 A draft{};
193 {
194 std::scoped_lock const lock{_mtx};
195 std::get<A>(_drafts).*FieldPtr = std::move(value);
196 draft = std::get<A>(_drafts);
197 }
198 if (::morph::model::ActionValidator<A>::ready(draft)) {
199 fire<A>(std::move(draft));
200 }
201 }
202
203 /// @brief Clears one section's draft back to a default-constructed action.
204 ///
205 /// Touches no other section and dispatches nothing. Values already captured
206 /// from a previous successful dispatch stay in `resolved()`: they describe
207 /// what the model was told, which resetting an editor does not undo.
208 /// @tparam A The section's action type.
209 template <typename A>
210 void reset() {
211 static_assert((std::is_same_v<A, typename Sections::action> || ...),
212 "SectionSet::reset<>: not a section of this group");
213 std::scoped_lock const lock{_mtx};
214 std::get<A>(_drafts) = A{};
215 }
216
217 /// @brief Snapshots one section's current draft.
218 ///
219 /// A copy taken under the lock, not a reference into live state, so a
220 /// renderer can read one section while another thread edits a different one.
221 /// @tparam A The section's action type.
222 /// @return The draft as it stands.
223 template <typename A>
224 [[nodiscard]] A draft() const {
225 static_assert((std::is_same_v<A, typename Sections::action> || ...),
226 "SectionSet::draft<>: not a section of this group");
227 std::scoped_lock const lock{_mtx};
228 return std::get<A>(_drafts);
229 }
230
231 /// @brief Looks up a value captured from a section's submitted draft or result.
232 /// @param path `"<ActionTypeId>.<field>"`, matching a section's declared `Bind::path()`.
233 /// @return The field's JSON-encoded value, or `std::nullopt` if @p path was
234 /// never captured — the section never fired successfully, or has no
235 /// such field.
236 [[nodiscard]] std::optional<std::string> resolved(std::string_view path) const {
237 std::scoped_lock const lock{_mtx};
238 auto iter = _resolvedValues.find(std::string{path});
239 if (iter == _resolvedValues.end()) {
240 return std::nullopt;
241 }
242 return iter->second;
243 }
244
245private:
246 /// @brief Records section @p A's submitted draft and its result under
247 /// `"<ActionTypeId>.<field>"` keys.
248 ///
249 /// Draft fields go in first and result fields overwrite them on a name
250 /// collision: the result is what the model actually settled on, so it is
251 /// the value a later section should prefill from.
252 /// @tparam A Section action type.
253 /// @param result The dispatch's successful result.
254 template <typename A>
255 void captureResult(const ::morph::model::ActionTraits<A>::Result& result) {
256 std::scoped_lock const lock{_mtx};
257 auto const typeId = ::morph::model::ActionTraits<A>::typeId();
258 auto record = [&](const auto& value) {
259 ::morph::forms::detail::forEachNamedMember(
260 value, [&]<std::size_t I>(std::string_view name, const auto& member) {
261 static_cast<void>(I);
262 std::string json;
263 // The `!write_json(...)` guard's false arm (write failure)
264 // is not exercised by this file's own test suite: every
265 // section action's fields are plain, well-formed data
266 // glaze's JSON writer cannot fail on. The same
267 // "untestable line" flows.hpp's captureResult already
268 // carries, for the same reason.
269 if (!glz::write_json(member, json)) {
270 _resolvedValues[std::string{typeId} + "." + std::string{name}] = std::move(json);
271 }
272 });
273 };
274 record(std::get<A>(_drafts));
275 record(result);
276 }

278 /// @brief Reports a dispatch failure no `onError` callback was given for.
279 /// @param typeId The failing section's action type-id.
280 /// @param err The captured exception.
281 static void logUnhandledError(std::string_view typeId, const std::exception_ptr& err) {
282 try {
283 std::rethrow_exception(err);
284 } catch (const std::exception& exc) {
285 ::morph::log::logError(std::string{"[section:"} + std::string{typeId} +
286 "] unhandled exception: " + exc.what());
287 } catch (...) {
288 ::morph::log::logError(std::string{"[section:"} + std::string{typeId} + "] unhandled unknown exception");
289 }
290 }
291
292 /// @brief Dispatches section @p A's ready draft and routes its outcome.
293 ///
294 /// Both closures are gated on `_callbacks`, so neither touches anything on
295 /// `this` once the set has been destroyed — a completion can still resolve
296 /// after the set is gone.
297 ///
298 /// Nothing here is keyed to a "current" section, which is what makes
299 /// sections cheaper than flow steps: a late reply cannot be stale, because
300 /// there is no position for it to be stale relative to.
301 /// @tparam A Section action type.
302 /// @param draft The ready action to execute.
303 template <typename A>
304 void fire(A draft) {
305 _handler.execute(std::move(draft))
306 .then(_callbacks,
307 [this](const ::morph::model::ActionTraits<A>::Result& result) {
308 this->template captureResult<A>(result);
309 })
310 .onError(_callbacks, [this](const std::exception_ptr& err) {
311 if (_onError) {
312 _onError(err);
313 } else {
314 logUnhandledError(::morph::model::ActionTraits<A>::typeId(), err);
315 }
316 });
317 }
318
319 ::morph::bridge::BridgeHandler<Model>& _handler;
320 std::function<void(std::exception_ptr)> _onError;
321 // _handler/_onError are set once at construction and never reassigned.
322 // Everything below is touched both by the owning thread (set/reset/draft)
323 // and by a dispatch's continuation, which runs on whatever thread resolves
324 // the BridgeHandler completion -- see docs/spec/core/bridge.md's
325 // executor/callback model.
326 mutable std::mutex _mtx;
327 std::tuple<typename Sections::action...> _drafts{};
328 std::unordered_map<std::string, std::string> _resolvedValues;
329 // Declared last, so it is the first member destroyed: every gated callback
330 // is stopped before the state those callbacks touch goes away.
331 ::morph::async::CallbackScope _callbacks;
332};
