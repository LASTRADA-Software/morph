5/// @file quantity.hpp
6/// @brief Unit-tagged, optionally-empty exact values for morph actions.
7///
8/// `morph::units::Quantity<U>` wraps an *optional* `morph::math::Rational`
9/// and carries its physical unit as a template argument — a value of the
10/// application's own unit enum. The design decisions, in order:
11///
12/// - **One kind of empty.** The blank state ("not entered", "not measured")
13/// lives *inside* the quantity as `std::optional<Rational>`; action structs
14/// never wrap a `Quantity` in another `std::optional`. Whether a field may
15/// still be empty at submit time is field *metadata* (see
16/// `morph/forms.hpp`), not a second wrapper type.
17/// - **Units are types.** `Quantity<Unit::kg>` and `Quantity<Unit::m3>`
18/// cannot be mixed accidentally; `operator*` / `operator/` deduce the
19/// result unit at compile time from the *application-defined* `consteval`
20/// algebra on the unit enum (see below). Unsupported combinations do not
21/// compile.
22/// - **Units never travel.** On the morph JSON wire a quantity is just its
23/// nullable Rational payload; the unit appears only in generated JSON
24/// Schemas (as `ExtUnits`) and in C++ types. A client cannot send a
25/// mismatched unit.
26/// - **Precision is declared in the type; actual precision is runtime
27/// data.** Each field carries a *declared* decimal count as a template
28/// argument — defaulting from the unit's metadata, overridable per field
29/// (`Quantity<Unit::m3, 4>`) — which drives the schema's
30/// `x-decimalPlaces` and `fromDouble`. The value's *actual* precision is
31/// the runtime `DecimalPlaces` tag inside the Rational: it max-propagates
32/// through arithmetic and can be changed at run time
33/// (`withDecimalPlaces`, `atDeclaredPrecision`).
34/// - **Empty propagates.** Arithmetic on an empty quantity yields an empty
35/// quantity (spreadsheet/SQL-NULL semantics). Division by zero also yields
36/// empty — the framework cannot distinguish "no data" from "no result"
37/// without complicating every call site; validate before dividing if the
38/// distinction matters.
39///
40/// @par Defining a unit system (application side)
41/// @code{.cpp}
42/// enum class Unit : std::uint16_t { scalar, kg, m3, kg_per_m3 };
43///
44/// // Explicit _per_ / _times_ names; ids are wire/schema vocabulary:
45/// // append new enumerators, never renumber or rename existing ones.
46/// template <>
47/// struct morph::units::UnitTraits<Unit> {
48/// static constexpr morph::units::UnitMeta meta(Unit unit) noexcept {
49/// switch (unit) {
50/// case Unit::scalar: return {"scalar", "", 3};
51/// case Unit::kg: return {"kg", "kg", 3};
52/// case Unit::m3: return {"m3", "m³", 3};
53/// case Unit::kg_per_m3: return {"kg_per_m3", "kg/m³", 1};
54/// }
55/// return {"?", "?", 3};
56/// }
57/// };
58///
59/// // The unit algebra: consteval, so an unsupported combination is a
60/// // compile-time error at the call site that attempted it.
61/// consteval Unit operator/(Unit lhs, Unit rhs) {
62/// if (rhs == Unit::scalar) return lhs;
63/// if (lhs == rhs) return Unit::scalar;
64/// if (lhs == Unit::kg && rhs == Unit::m3) return Unit::kg_per_m3;
65/// throw "unsupported unit quotient";
66/// }
67///
68/// using Mass = morph::units::Quantity<Unit::kg>;
69/// using Volume = morph::units::Quantity<Unit::m3>;
70/// // Mass{...} / Volume{...} -> morph::units::Quantity<Unit::kg_per_m3>
71///
72/// // Optional: convertible entry units with exact ratios. Renderers offer a
73/// // unit selector and recalculate exactly; payloads stay canonical.
74/// // static constexpr std::span<const UnitAlternative<Unit>> alternatives(Unit);
75/// @endcode

91/// @brief Static description of one unit, supplied by `UnitTraits`.
92struct UnitMeta {
93 /// @brief Stable ascii identifier, e.g. `"kg_per_m3"`. Appears in JSON
94 /// Schemas (`ExtUnits.unitAscii`); part of the protocol vocabulary.
95 std::string_view id;

97 /// @brief Human display text, e.g. `"kg/m³"` (`ExtUnits.unitUnicode`).
98 std::string_view display;
99
100 /// @brief Default decimal places for parsing/formatting values of this
101 /// unit; surfaces in schemas as `x-decimalPlaces`.
102 std::uint32_t defaultDecimals{3};
103};

105/// @brief One convertible display/entry unit for a canonical unit.
106///
107/// The ratio is exact: `value_in_canonical = value_in_alternative * num/den`
108/// (e.g. grams -> kilograms is `{Unit::g, 1, 1000}`). Both components must
109/// be positive.
110/// @tparam E The application's unit enum type.
111template <typename E>
112struct UnitAlternative {
113 /// @brief The alternative unit (must have `UnitTraits` metadata).
114 E unit{};
115
116 /// @brief Numerator of the exact alternative-to-canonical ratio.
117 std::int64_t num{1};
118
119 /// @brief Denominator of the exact alternative-to-canonical ratio.
120 std::int64_t den{1};
121};
122
123/// @brief Customisation point: the application specialises this for its unit
124/// enum and returns a `UnitMeta` per enumerator.
125///
126/// Optionally, a specialisation may also provide
127/// `static constexpr std::span<const UnitAlternative<E>> alternatives(E)` —
128/// the convertible display/entry units per canonical unit. Renderers then
129/// offer a unit selector on such fields and recalculate entered values
130/// exactly on switch; the wire and the model always stay in the canonical
131/// unit.
132/// @tparam E The application's unit enum type.
133template <typename E>
134struct UnitTraits;
135
136/// @brief Concept: `UnitTraits<E>` also declares convertible alternatives.
137template <typename E>
138concept HasUnitAlternatives = requires(E unit) {
139 { UnitTraits<E>::alternatives(unit) } -> std::convertible_to<std::span<const UnitAlternative<E>>>;
140};
141
142/// @brief Concept: an enum with a `UnitTraits` specialisation.
143template <typename E>
144concept UnitEnum = std::is_enum_v<E> && requires(E unit) {
145 { UnitTraits<E>::meta(unit) } -> std::convertible_to<UnitMeta>;
146};

148/// @brief A unit-tagged, optionally-empty exact value with a *declared*
149/// precision in its type.
150///
151/// Two precisions exist by design and must not be conflated:
152///
153/// - **Declared precision** (`DeclaredDecimals`, a template argument): a
154/// property of the *field* — how many decimals this slot is specified to
155/// hold. It defaults from the unit's `UnitTraits` metadata and can be
156/// overridden per field (`Quantity<Unit::m3, 4>`). It feeds the generated
157/// schema (`x-decimalPlaces`), `fromDouble`, and `atDeclaredPrecision`.
158/// - **Actual precision**: a property of the *value* — the runtime
159/// `DecimalPlaces` tag inside the `Rational` payload. It propagates through
160/// arithmetic (max rule) and can be changed at run time
161/// (`withDecimalPlaces`).
162///
163/// Default state is *empty* — a form draft starts blank while the unit and
164/// the declared precision are already known from the type. Same-unit
165/// quantities convert freely across declared precisions (the value,
166/// including its runtime tag, carries over unchanged).
167///
168/// @tparam U An enumerator of an application unit enum
169/// satisfying `UnitEnum`.
170/// @tparam DeclaredDecimals Declared decimal places of the field; defaults
171/// to the unit's `UnitMeta::defaultDecimals`.
172template <auto U, std::uint32_t DeclaredDecimals = UnitTraits<decltype(U)>::meta(U).defaultDecimals>
173 requires UnitEnum<decltype(U)>
174struct Quantity {
175 static_assert(DeclaredDecimals >= 1 && DeclaredDecimals <= math::kMaxDecimalPlaces,
176 "declared decimals must be within [1, kMaxDecimalPlaces]");
177
178 /// @brief The payload; `std::nullopt` means "not entered / not measured".
179 std::optional<math::Rational> value;
180
181 /// @brief Constructs the empty state.
182 constexpr Quantity() noexcept = default;
183
184 /// @brief Engages with @p engaged (which keeps its own runtime precision).
185 /// @param engaged The exact value to hold.
186 constexpr Quantity(math::Rational engaged) noexcept : value{engaged} {}
187
188 /// @brief Adopts an optional payload as-is.
189 /// @param payload Engaged or empty payload.
190 constexpr Quantity(std::optional<math::Rational> payload) noexcept : value{payload} {}
191
192 /// @brief Converts from the same unit at another declared precision; the
193 /// value and its runtime precision tag carry over unchanged.
194 /// @param other Same-unit quantity with a different declared precision.
195 template <std::uint32_t OtherDecimals>
196 requires(OtherDecimals != DeclaredDecimals)
197 constexpr Quantity(const Quantity<U, OtherDecimals>& other) noexcept : value{other.value} {}
198
199 /// @brief The unit this quantity is denominated in (compile-time).
200 static constexpr auto unit = U;
201
202 /// @brief Declared decimal places of this field (compile-time).
203 static constexpr std::uint32_t declaredDecimals = DeclaredDecimals;
204
205 /// @brief The unit's static metadata (id, display text, default decimals).
206 /// @return The `UnitMeta` from the application's `UnitTraits` specialisation.
207 [[nodiscard]] static constexpr UnitMeta unitMeta() noexcept { return UnitTraits<decltype(U)>::meta(U); }
208
209 /// @brief The declared precision as a `DecimalPlaces` strong type.
210 /// @return `DecimalPlaces{declaredDecimals}`.
211 [[nodiscard]] static constexpr math::DecimalPlaces declaredPrecision() noexcept {
212 return math::DecimalPlaces{DeclaredDecimals};
213 }
214
215 /// @brief The convertible display/entry units declared for this field's
216 /// unit (empty when the unit system declares none).
217 /// @return Exact-ratio alternatives from `UnitTraits<E>::alternatives`.
218 [[nodiscard]] static constexpr std::span<const UnitAlternative<decltype(U)>> unitAlternatives() noexcept {
219 if constexpr (HasUnitAlternatives<decltype(U)>) {
220 return UnitTraits<decltype(U)>::alternatives(U);
221 } else {
222 return {};
223 }
224 }
225
226 /// @brief Converts a floating-point reading at the declared precision.
227 /// @param raw The value to convert.
228 /// @return The engaged quantity, or empty when @p raw is not finite or
229 /// does not fit (empty-propagation philosophy: no error channel).
230 [[nodiscard]] static Quantity fromDouble(double raw) noexcept {
231 if (auto converted = math::Rational::fromFloat(raw, declaredPrecision()); converted.has_value()) {
232 return Quantity{*converted};
233 }
234 return {};
235 }
236
237 /// @brief Whether a value has been entered/measured.
238 /// @return `true` if the payload is engaged.
239 [[nodiscard]] constexpr bool hasValue() const noexcept { return value.has_value(); }
240
241 /// @brief Unchecked access to the engaged value (UB when empty, exactly
242 /// like `std::optional` — the unchecked contract is the point).
243 /// @return The engaged exact value.
244 // NOLINTNEXTLINE(bugprone-unchecked-optional-access)
245 [[nodiscard]] constexpr const math::Rational& operator*() const noexcept { return *value; }
246
247 /// @brief Retags the value's *runtime* precision (the exact value itself
248 /// is unchanged — precision only affects rounding and formatting).
249 /// @param newPrecision Runtime precision; silently clamped to the valid range.
250 /// @return The retagged quantity, or empty if this is empty.
251 [[nodiscard]] constexpr Quantity withDecimalPlaces(math::DecimalPlaces newPrecision) const noexcept {
252 if (!value) {
253 return {};
254 }
255 auto adjusted = *value;
256 adjusted.decimalPlaces = math::DecimalPlaces{math::detail::clampWireDecimalPlaces(newPrecision.value)};
257 return Quantity{adjusted};
258 }
259
260 /// @brief Retags the value's runtime precision to the declared one —
261 /// e.g. before display, after arithmetic widened it.
262 /// @return The retagged quantity, or empty if this is empty.
263 [[nodiscard]] constexpr Quantity atDeclaredPrecision() const noexcept {
264 return withDecimalPlaces(declaredPrecision());
265 }
266};
267
268/// @brief Ordering across same-unit quantities of any declared precision;
269/// empty sorts before engaged, engaged values compare exactly.
270/// @param lhs Left operand.
271/// @param rhs Right operand (same unit, any declared precision).
272/// @return The ordering of the two payloads.
273template <auto U, std::uint32_t DecA, std::uint32_t DecB>
274[[nodiscard]] constexpr std::strong_ordering operator<=>(const Quantity<U, DecA>& lhs,
275 const Quantity<U, DecB>& rhs) noexcept {
276 return lhs.value <=> rhs.value;
277}
278
279/// @brief Equality across same-unit quantities of any declared precision.
280/// @param lhs Left operand.
281/// @param rhs Right operand (same unit, any declared precision).
282/// @return `true` when both are empty or both engaged with equal values.
283template <auto U, std::uint32_t DecA, std::uint32_t DecB>
284[[nodiscard]] constexpr bool operator==(const Quantity<U, DecA>& lhs, const Quantity<U, DecB>& rhs) noexcept {
285 return lhs.value == rhs.value;
286}

313/// @brief Same-unit sum across any declared precisions. Empty if either
314/// operand is empty.
315/// @param lhs Left operand.
316/// @param rhs Right operand.
317/// @return The exact sum, or empty.
318template <auto U, std::uint32_t DecA, std::uint32_t DecB>
319[[nodiscard]] constexpr Quantity<U> operator+(const Quantity<U, DecA>& lhs, const Quantity<U, DecB>& rhs) noexcept {
320 if (!lhs.value || !rhs.value) {
321 return {};
322 }
323 return Quantity<U>{*lhs.value + *rhs.value};
324}
325
326/// @brief Same-unit difference across any declared precisions. Empty if
327/// either operand is empty.
328/// @param lhs Left operand.
329/// @param rhs Right operand.
330/// @return The exact difference, or empty.
331template <auto U, std::uint32_t DecA, std::uint32_t DecB>
332[[nodiscard]] constexpr Quantity<U> operator-(const Quantity<U, DecA>& lhs, const Quantity<U, DecB>& rhs) noexcept {
333 if (!lhs.value || !rhs.value) {
334 return {};
335 }
336 return Quantity<U>{*lhs.value - *rhs.value};
337}
338
339/// @brief Negation. Empty stays empty; the declared precision is kept.
340/// @param operand Value to negate.
341/// @return The negated value, or empty.
342template <auto U, std::uint32_t Dec>
343[[nodiscard]] constexpr Quantity<U, Dec> operator-(const Quantity<U, Dec>& operand) noexcept {
344 if (!operand.value) {

350/// @brief Cross-unit product; the result unit is `A * B` per the application's
351/// consteval unit algebra. Does not participate in overload resolution
352/// when the algebra rejects the combination.
353/// @param lhs Left operand.
354/// @param rhs Right operand.
355/// @return The exact product in the deduced unit, or empty.
356template <auto A, std::uint32_t DecA, auto B, std::uint32_t DecB>
357 requires requires { typename std::type_identity_t<Quantity<A * B>>; }
358[[nodiscard]] constexpr Quantity<A * B> operator*(const Quantity<A, DecA>& lhs,
359 const Quantity<B, DecB>& rhs) noexcept {
360 if (!lhs.value || !rhs.value) {
361 return {};
362 }
363 return Quantity<A * B>{*lhs.value * *rhs.value};
364}
365
366/// @brief Cross-unit quotient; the result unit is `A / B` per the application's
367/// consteval unit algebra. Empty when either side is empty *or* the
368/// divisor is zero.
369/// @param lhs Dividend.
370/// @param rhs Divisor.
371/// @return The exact quotient in the deduced unit, or empty.
372template <auto A, std::uint32_t DecA, auto B, std::uint32_t DecB>
373 requires requires { typename std::type_identity_t<Quantity<A / B>>; }
374[[nodiscard]] constexpr Quantity<A / B> operator/(const Quantity<A, DecA>& lhs,
375 const Quantity<B, DecB>& rhs) noexcept {
376 if (!lhs.value || !rhs.value) {
377 return {};
378 }
379 // dividedBy is the single division-by-zero authority: it reports exactly
380 // when the divisor is zero, which maps to the empty result here.
381 auto quotient = lhs.value->dividedBy(*rhs.value);
382 if (!quotient) {
383 return {};
384 }
385 return Quantity<A / B>{*quotient};
386}

413/// @brief Divides by a dimensionless rational (unit and declared precision
414/// unchanged). Empty when the quantity is empty or the divisor is zero.
415/// @param lhs Quantity to divide.
416/// @param divisor Dimensionless divisor.
417/// @return The divided quantity, or empty.
418template <auto U, std::uint32_t Dec>
419[[nodiscard]] constexpr Quantity<U, Dec> operator/(const Quantity<U, Dec>& lhs,
420 const math::Rational& divisor) noexcept {
421 if (!lhs.value) {
422 return {};
423 }
424 auto quotient = lhs.value->dividedBy(divisor);
425 if (!quotient) {
426 return {};
427 }
