64/// @brief Whether `==` is usable for @p T: declared, and its element types usable too.
65///
66/// `std::equality_comparable` alone is not enough. The standard containers declare an unconstrained
67/// `operator==`, so it holds for a container whose element has no `==`, and comparing two of them then fails
68/// to compile. So a range is checked through its element type, and `std::optional`, `std::pair`, `std::tuple`
69/// and `std::variant` through theirs.
70/// @tparam T The value type, without cv-qualifiers.
71template <typename T>
72struct EqualityUsable : std::bool_constant<std::equality_comparable<T>> {};
73
74/// @brief A range is usable when it declares `==` and its element type is usable. A range whose element is
75/// itself (`std::filesystem::path`) is checked as a plain type.
76/// @tparam T The range type.
77template <typename T>
78 requires std::ranges::range<T const> && (!std::same_as<std::remove_cv_t<std::ranges::range_value_t<T const>>, T>)
79struct EqualityUsable<T>
80 : std::bool_constant<std::equality_comparable<T> &&
81 EqualityUsable<std::remove_cv_t<std::ranges::range_value_t<T const>>>::value> {};
82
83/// @brief An optional is usable when its value type is.
84/// @tparam T The value type.
85template <typename T>
86struct EqualityUsable<std::optional<T>> : EqualityUsable<std::remove_cv_t<T>> {};
87
88/// @brief A pair is usable when both its types are.
89/// @tparam First The first type.
90/// @tparam Second The second type.
91template <typename First, typename Second>
92struct EqualityUsable<std::pair<First, Second>> : std::bool_constant<EqualityUsable<std::remove_cv_t<First>>::value &&
93 EqualityUsable<std::remove_cv_t<Second>>::value> {
94};
95
96/// @brief A tuple is usable when every element type is.
97/// @tparam Ts The element types.
98template <typename... Ts>
99struct EqualityUsable<std::tuple<Ts...>> : std::bool_constant<(EqualityUsable<std::remove_cv_t<Ts>>::value && ...)> {};
100
101/// @brief A variant is usable when every alternative is.
102/// @tparam Ts The alternatives.
103template <typename... Ts>
104struct EqualityUsable<std::variant<Ts...>> : std::bool_constant<(EqualityUsable<std::remove_cv_t<Ts>>::value && ...)> {
105};

125/// @brief State shared by one `Runtime` and every node made from it.
126///
127/// Nodes hold it by `shared_ptr`, so a node that outlives its `Runtime` (a
128/// reported misuse) still has a valid core. The posted flush holds it weakly
129/// and checks `detach()`, so a flush after the `Runtime` is gone does nothing.
130class RuntimeCore : public std::enable_shared_from_this<RuntimeCore> {
131public:
132 /// @param owner The executor every operation must run on, and the one flushes are posted to. It must
133 /// queue what is posted and run it later, in order: an executor that runs a post inline would
134 /// run Effects inside `set()`. Borrowed: it must outlive the core.
135 explicit RuntimeCore(exec::IExecutor& owner MORPH_LIFETIMEBOUND) noexcept : _affinity{owner} {}
136
137 ~RuntimeCore() = default;
138 RuntimeCore(RuntimeCore const&) = delete;
139 RuntimeCore& operator=(RuntimeCore const&) = delete;
140 RuntimeCore(RuntimeCore&&) = delete;
141 RuntimeCore& operator=(RuntimeCore&&) = delete;
142
143 /// @brief Whether the calling thread is on the owner, without reporting anything.
144 /// @return What `checkOwner()` returns, minus the report.
145 [[nodiscard]] bool onOwner() const noexcept { return _affinity.here(); }
146
147 /// @brief Whether the calling thread is on the owner, reporting `site::kOffOwner` when it is not.
148 /// @return True on the owner: inside one of its tasks, or anywhere on the thread that built the Runtime
149 /// when that thread was running no executor's task at the time (a Qt slot, `main()`, a test
150 /// body). That second half is a per-thread check: on that thread a task of another executor
151 /// passes too, so running a different executor's tasks there is not detected.
152 [[nodiscard]] bool checkOwner() const noexcept {
153 if (_affinity.here()) {
154 return true;
155 }
156 report(site::kOffOwner);
157 return false;
158 }
159
160 /// @brief Reports a misuse: asserts in a debug build, or hands it to an installed owner probe.
161 /// @param where One of the `site::` constants.
162 void report(char const* where) const noexcept {
163 exec::detail::noteOwner(where, _affinity.owner().coreExecutor(), false);
164 }
165
166 /// @brief The owner executor.
167 /// @return The executor passed at construction.
168 [[nodiscard]] exec::IExecutor& owner() const noexcept { return _affinity.owner(); }
169
170 /// @brief Sets the flush bounds and the after-flush hook.
171 /// @param bounds How often one Effect may run in a flush, and how many flushes in a row are re-posted after
172 /// an Effect threw.
173 /// @param afterFlush Called after every flush that processed a queued Effect; may be empty.
174 void configure(FlushBounds bounds, std::function<void()> afterFlush) {
175 _bounds = bounds;
176 _afterFlush = std::move(afterFlush);
177 }
178
179 /// @brief Opens a batch; batches nest.
180 void beginBatch() noexcept { ++_batchDepth; }
181
182 /// @brief Closes a batch; closing the outermost one requests a flush when effects are queued.
183 ///
184 /// The outermost batch closing outside a flush is a write from outside the graph, which starts a new count
185 /// of the re-posts after a throwing Effect.
186 void endBatch() {
187 if (--_batchDepth == 0) {
188 if (!_flushing) {
189 _throwReposts = 0;
190 }
191 requestFlush();
192 }
193 }
194
195 /// @brief Queues an Effect that turned stale. Defined below `Node`.
196 /// @param effect The Effect node.
197 void enqueue(Node& effect);
198
199 /// @brief Posts one flush to the owner, unless one is already requested or running, nothing is
200 /// queued, or the Runtime is gone.
201 void requestFlush() {
202 if (_flushing || _isFlushRequested || _queue.empty() || _detached) {
203 return;
204 }
205 // Set before posting, so the flag is already down again if the flush runs inside post().
206 _isFlushRequested = true;
207 try {
208 _affinity.owner().post([weak = weak_from_this()] {
209 if (auto const core = weak.lock()) {
210 core->flush();
211 }
212 });
213 } catch (...) {
214 // Nothing was posted: the queue is kept, and the next write posts again.
215 _isFlushRequested = false;
216 throw;
217 }
218 }
219
220 /// @brief Runs every queued Effect that a source change really affects. Defined below `Node`.
221 void flush();
222
223 /// @brief Removes every reference the core holds to a node being destroyed. Defined below `Node`.
224 /// @param node The node.
225 void forget(Node& node) noexcept;
226
227 /// @brief The innermost tracking frame, or null outside every tracked run.
228 /// @return The frame reads register with.
229 [[nodiscard]] TrackingFrame* tracking() const noexcept { return _tracking; }
230
231 /// @brief Installs the innermost tracking frame.
232 /// @param frame The frame, or null.
233 void setTracking(TrackingFrame* frame) noexcept { _tracking = frame; }
234
235 /// @brief Whether a Computed is computing on this runtime.
236 /// @return True between `enterComputed()` and the matching `leaveComputed()`.
237 [[nodiscard]] bool isComputing() const noexcept { return _computing != 0; }
238
239 /// @brief Marks a Computed as computing.
240 void enterComputed() noexcept { ++_computing; }
241
242 /// @brief Marks a Computed as done computing.
243 void leaveComputed() noexcept { --_computing; }
244
245 /// @brief Marks the start of a widget callback.
246 void enterWidgetEvent() noexcept { ++_widgetEventDepth; }
247
248 /// @brief Marks the end of a widget callback.
249 void leaveWidgetEvent() noexcept { --_widgetEventDepth; }
250
251 /// @brief How deeply batches are nested on the owner.
252 /// @return Zero outside every batch.
253 [[nodiscard]] std::size_t batchDepth() const noexcept { return _batchDepth; }
254
255 /// @brief Whether a widget callback is running on the owner.
256 /// @return True between `enterWidgetEvent()` and the matching `leaveWidgetEvent()`.
257 [[nodiscard]] bool isInWidgetEvent() const noexcept { return _widgetEventDepth != 0; }
258
259 /// @brief Whether a flush is posted and has not run yet.
260 /// @return The coalescing flag.
261 [[nodiscard]] bool isFlushRequested() const noexcept { return _isFlushRequested; }
262
263 /// @brief Marks the Runtime as gone: flushes do nothing from now on.
264 void detach() noexcept { _detached = true; }
265
266 /// @brief Counts a node made from this runtime.
267 /// @return The node's creation order: greater than that of every node made from this runtime before it.
268 [[nodiscard]] std::uint64_t nodeCreated() noexcept {
269 ++_liveNodes;
270 return ++_createdNodes;
271 }
272
273 /// @brief Counts a node destroyed.
274 void nodeDestroyed() noexcept { --_liveNodes; }
275
276 /// @brief How many nodes made from this runtime are alive.
277 /// @return The live-node count.
278 [[nodiscard]] std::size_t liveNodes() const noexcept { return _liveNodes; }
279
280private:
281 void dropQueue() noexcept;
282
283 exec::detail::OwnerAffinity _affinity;
284 // A binary heap ordered by depth, then creation, the shallowest and oldest on top: see `flush()`.
285 std::vector<Node*> _queue;
286 std::function<void()> _afterFlush;
287 TrackingFrame* _tracking = nullptr;
288 FlushBounds _bounds;
289 // Flushes re-posted in a row after an Effect threw.
290 std::size_t _throwReposts = 0;
291 std::size_t _batchDepth = 0;
292 std::size_t _computing = 0;
293 std::size_t _widgetEventDepth = 0;
294 std::size_t _liveNodes = 0;
295 std::uint64_t _createdNodes = 0;
296 std::uint64_t _flushEpoch = 0;
297 bool _isFlushRequested = false;
298 bool _flushing = false;
299 bool _detached = false;
300};

302/// @brief Collects the sources one tracked run reads, and restores the previous frame when it ends.
303///
304/// A frame whose observer is null is an untracked region: reads inside it subscribe nothing.
305class TrackingFrame {
306public:
307 /// @param core The runtime core.
308 /// @param observer The node whose run this is, or null for an untracked region.
309 TrackingFrame(RuntimeCore& core, Node* observer) noexcept
310 : _core{&core}, _previous{core.tracking()}, _observer{observer} {
311 core.setTracking(this);
312 }
313
314 /// @brief Reinstalls the enclosing frame.
315 ~TrackingFrame() { _core->setTracking(_previous); }
316 TrackingFrame(TrackingFrame const&) = delete;
317 TrackingFrame& operator=(TrackingFrame const&) = delete;
318 TrackingFrame(TrackingFrame&&) = delete;
319 TrackingFrame& operator=(TrackingFrame&&) = delete;
320
321 /// @brief Records that the run read @p source; a no-op in an untracked region.
322 /// @param source The node read.
323 void add(Node& source) {
324 if (_observer != nullptr && std::ranges::find(_sources, &source) == _sources.end()) {
325 _sources.push_back(&source);
326 }
327 }
328
329 /// @brief Drops @p node, which is being destroyed, from the recorded sources. When @p node is this
330 /// frame's observer, the frame stops recording and remembers that its observer is gone.
331 /// @param node The node.
332 void forget(Node& node) noexcept {
333 std::erase(_sources, &node);
334 if (&node == _observer) {
335 _observer = nullptr;
336 _observerDestroyed = true;
337 }
338 }
339
340 /// @brief Whether the observer was destroyed during the run: a body may destroy its own node (a
341 /// remount inside an Effect), and the node's run must not touch it afterwards.
342 /// @return True once the observer's destructor has run.
343 [[nodiscard]] bool observerDestroyed() const noexcept { return _observerDestroyed; }
344
345 /// @brief Hands the recorded sources to the observer.
346 /// @return The sources, deduplicated, in first-read order.
347 [[nodiscard]] std::vector<Node*> takeSources() noexcept { return std::move(_sources); }
348
349 /// @brief The enclosing frame.
350 /// @return The frame that was innermost when this one was opened, or null.
351 [[nodiscard]] TrackingFrame* previous() const noexcept { return _previous; }
352
353private:
354 RuntimeCore* _core;
355 TrackingFrame* _previous;
356 Node* _observer;
357 std::vector<Node*> _sources;
358 bool _observerDestroyed = false;
359};

361/// @brief A graph node: a source (Signal), an observer (Effect), or both (Computed).
362///
363/// Non-copyable and non-movable: other nodes hold raw pointers to it, and its
364/// destructor unlinks it in both directions. A derived node supplies `recompute()`
365/// and `isEffect()` as private overrides; only `Node` calls them.
366///
367/// A node may be destroyed while it is being brought up to date (an Effect body
368/// that remounts the view owning it). The destructor tells the running pull and
369/// the running tracking frame, and neither touches the node again.
370class Node {
371public:
372 /// @brief Constructs a node. Off the owner this is reported (`site::kOffOwner`) and the node is
373 /// not counted towards the core's live nodes, whose counter belongs to the owner.
374 /// @param core The runtime core this node belongs to.
375 /// @param depth How deeply the scope owning this node is nested; queued Effects run shallowest first.
376 explicit Node(std::shared_ptr<RuntimeCore> core, std::size_t depth = 0)
377 : _core{std::move(core)}, _depth{depth}, _bornOnOwner{_core->checkOwner()} {
378 if (_bornOnOwner) {
379 _creationOrder = _core->nodeCreated();
380 }
381 }
382
383 /// @brief Unlinks this node from its sources and observers, drops every reference the core holds
384 /// to it, and tells a pull running on it that it is gone.
385 ///
386 /// Destroying a node off the owner is a misuse. It is reported (`site::kOffOwner`) but cannot be
387 /// refused: the destructor still edits the owner's graph, queue and counters, and what that does
388 /// beyond the report is undefined.
389 virtual ~Node() {
390 static_cast<void>(_core->checkOwner());
391 if (_destroyedWatch != nullptr) {
392 *_destroyedWatch = true;
393 }
394 for (Node* const source : _sources) {
395 std::erase(source->_observers, this);
396 }
397 for (Node* const observer : _observers) {
398 std::erase(observer->_sources, this);
399 }
400 _core->forget(*this);
401 if (_bornOnOwner) {
402 _core->nodeDestroyed();
403 }
404 }
405
406 Node(Node const&) = delete;
407 Node& operator=(Node const&) = delete;
408 Node(Node&&) = delete;
409 Node& operator=(Node&&) = delete;
410
411 /// @brief Push phase: raises this node to @p colour and marks its observers Check.
412 ///
413 /// An Effect is queued when it leaves Clean. A node already at least as stale stops the walk, because
414 /// its observers were marked when it turned stale, unless an observer has since ended Clean above it
415 /// (`exposeStaleSources()`); then the walk passes through it once.
416 /// @param colour Dirty for a direct observer of a change, Check for a transitive one.
417 // NOLINTNEXTLINE(misc-no-recursion) -- the recursion depth is the graph's depth
418 void markStale(Colour colour) {
419 if (_colour >= colour && !_staleAboveClean) {
420 return;
421 }
422 if (_colour < colour) {
423 if (_colour == Colour::Clean && isEffect()) {
424 // Queued before the colour changes: an enqueue that fails to allocate leaves the node
425 // Clean, so a later walk that reaches it queues it again.
426 _core->enqueue(*this);
427 }
428 _colour = colour;
429 }
430 // Cleared before the walk: two flagged nodes can observe each other (a cycle a refused self-read
431 // leaves linked), and a revisit must stop here. Restored if the walk throws, so the next write
432 // passes through again; a node this walk raised from Clean is not flagged, and an Effect behind
433 // it whose enqueue failed is not reached again.
434 bool const flagged = std::exchange(_staleAboveClean, false);
435 try {
436 for (Node* const observer : _observers) {
437 observer->markStale(Colour::Check);
438 }
439 } catch (...) {
440 _staleAboveClean = flagged;
441 throw;
442 }
443 }
444
445 /// @brief Pull phase: brings this node up to date, recomputing only if a source really changed.
446 ///
447 /// A pull that throws leaves the node Clean (so later changes still reach it) and rethrows. A pull
448 /// during which the node is destroyed returns without touching it.
449 /// @return False when the node was destroyed during the pull: the caller must not touch it either.
450 // GCC sees `_destroyedWatch` keep the address of the local `destroyed` on the path where the
451 // node is destroyed during the pull. On that path `this` is gone too and nothing reads the
452 // member again; on every other path `endPull` restores the enclosing watch before returning.
453#if defined(__GNUC__) && !defined(__clang__)
454#pragma GCC diagnostic push
455#pragma GCC diagnostic ignored "-Wdangling-pointer"
456#endif
457 // NOLINTNEXTLINE(misc-no-recursion) -- the recursion depth is the graph's depth
458 bool updateIfNecessary() {
459 bool destroyed = false;
460 bool* const enclosing = std::exchange(_destroyedWatch, &destroyed);
461 try {
462 pull(destroyed);
463 } catch (...) {
464 endPull(destroyed, enclosing);
465 throw;
466 }
467 endPull(destroyed, enclosing);
468 return !destroyed;
469 }
470#if defined(__GNUC__) && !defined(__clang__)
471#pragma GCC diagnostic pop
472#endif
473
474 /// @brief Counts one run of this Effect in the flush numbered @p epoch.
475 /// @param epoch The current flush's number.
476 /// @return How many times it has run in that flush, this one included.
477 [[nodiscard]] std::size_t countRun(std::uint64_t epoch) noexcept {
478 if (_epoch != epoch) {
479 _epoch = epoch;
480 _runs = 0;
481 }
482 return ++_runs;
483 }
484
485 /// @brief How deeply the scope owning this node is nested; the flush runs shallower Effects first.
486 /// @return Zero for a node no scope owns.
487 [[nodiscard]] std::size_t depth() const noexcept { return _depth; }
488
489 /// @brief Whether this node runs after @p other when both are queued: it is deeper, or as deep and newer.
490 /// @param other The other node.
491 /// @return The flush queue's ordering.
492 [[nodiscard]] bool runsAfter(Node const& other) const noexcept {
493 return std::pair{_depth, _creationOrder} > std::pair{other._depth, other._creationOrder};
494 }
495
496 /// @brief Where this node stands in its runtime's creation order; among Effects of one depth, the flush
497 /// runs older ones first.
498 /// @return Greater than that of every node made from the same runtime before it; zero for a node
499 /// constructed off the owner, which is never queued.
500 [[nodiscard]] std::uint64_t creationOrder() const noexcept { return _creationOrder; }
501
502 /// @brief Marks this node Clean without running it (a dropped queue entry).
503 void settle() noexcept {
504 _colour = Colour::Clean;
505 exposeStaleSources();
506 }

508protected:
509 /// @brief The runtime core.
510 /// @return The core this node belongs to.
511 [[nodiscard]] RuntimeCore& core() const noexcept { return *_core; }
512
513 /// @brief Whether this node was constructed on the owner; one that was not is refused its work.
514 /// @return The owner check the constructor made.
515 [[nodiscard]] bool bornOnOwner() const noexcept { return _bornOnOwner; }
516
517 /// @brief Registers this node as a source of the innermost tracked run, if any.
518 void trackRead() const {
519 if (TrackingFrame* const frame = _core->tracking(); frame != nullptr) {
520 // Reading is logically const; the dependency link is bookkeeping.
521 frame->add(const_cast<Node&>(*this)); // NOLINT(cppcoreguidelines-pro-type-const-cast)
522 }
523 }

525 /// @brief Marks this node Dirty, so the next pull recomputes it.
526 void forceDirty() noexcept { _colour = Colour::Dirty; }
527
528 /// @brief Whether a pull is bringing this node up to date right now. A read of the node from inside
529 /// that pull can only come from its own run or from a source's run that reads it: a cycle.
530 /// @return True between the start and the end of `updateIfNecessary()` on this node.
531 [[nodiscard]] bool isPulling() const noexcept { return _destroyedWatch != nullptr; }
532
533 /// @brief Marks every observer of this node @p colour.
534 /// @param colour The colour to raise them to.
535 void markObserversStale(Colour colour) {
536 for (Node* const observer : _observers) {
537 observer->markStale(colour);
538 }
539 }

541 /// @brief Replaces this node's sources with @p fresh, unlinking the ones no longer read.
542 ///
543 /// Strong guarantee: linking the new sources is the only step that allocates, and it runs first;
544 /// if it throws, the links it made are undone, so no source keeps a link this node does not list.
545 /// @param fresh The sources the latest run read.
546 void adoptSources(std::vector<Node*> fresh) {
547 std::size_t linked = 0;
548 try {
549 for (Node* const source : fresh) {
550 if (std::ranges::find(_sources, source) == _sources.end()) {
551 source->_observers.push_back(this);
552 }
553 ++linked;
554 }
555 } catch (...) {
556 for (Node* const source : std::span{fresh}.first(linked)) {
557 if (std::ranges::find(_sources, source) == _sources.end()) {
558 std::erase(source->_observers, this);
559 }
560 }
561 throw;
562 }
563 for (Node* const old : _sources) {
564 if (std::ranges::find(fresh, old) == fresh.end()) {
565 std::erase(old->_observers, this);
566 }
567 }
568 _sources = std::move(fresh);
569 }

571 /// @brief Adds @p fresh to this node's sources without dropping any: what a failed run keeps.
572 /// @param fresh The sources the failed run read before it threw.
573 void mergeSources(std::vector<Node*> fresh) {
574 std::vector<Node*> kept;
575 std::ranges::copy_if(_sources, std::back_inserter(kept),
576 [&fresh](Node const* old) { return std::ranges::find(fresh, old) == fresh.end(); });
577 fresh.insert(fresh.end(), kept.begin(), kept.end());
578 adoptSources(std::move(fresh));
579 }

582 /// @brief Brings the node's value up to date.
583 /// @return True when observers must treat the value as changed.
584 virtual bool recompute() = 0;

586 /// @brief Whether this node is an Effect, which the flush queue runs.
587 /// @return False unless overridden.
588 [[nodiscard]] virtual bool isEffect() const noexcept { return false; }

590 // Returns as soon as `destroyed` is set: from then on `this` is gone.
591 // NOLINTNEXTLINE(misc-no-recursion) -- the recursion depth is the graph's depth
592 void pull(bool const& destroyed) {
593 if (_colour == Colour::Check) {
594 for (std::size_t i = 0; i < _sources.size();) {
595 Node* const source = _sources.at(i);
596 source->updateIfNecessary();
597 if (destroyed) {
598 return;
599 }
600 if (_colour == Colour::Dirty) {
601 break;
602 }
603 // A source destroyed by its own pull has unlinked itself, and the next one now sits at i.
604 if (i < _sources.size() && _sources.at(i) == source) {
605 ++i;
606 }
607 }
608 }
609 if (_colour == Colour::Dirty) {
610 bool const changed = recompute();
611 if (destroyed) {
612 return;
613 }
614 if (changed) {
615 markObserversStale(Colour::Dirty);
616 }
617 }
618 }
619
620 void endPull(bool destroyed, bool* enclosing) noexcept {
621 if (destroyed) {
622 if (enclosing != nullptr) {
623 *enclosing = true;
624 }
625 return;
626 }
627 _destroyedWatch = enclosing;
628 _colour = Colour::Clean;
629 _staleAboveClean = false;
630 exposeStaleSources();
631 }
632
633 // This node is Clean while a source may still be stale: an Effect settled by a write cycle without
634 // running, an Effect whose run wrote a source of a Computed it had read (the write cannot mark the
635 // running Effect), or a run that threw before reading every source. A stale source stops the next
636 // write's walk, so this node would never hear of it. Each stale source is flagged, so that walk
637 // passes through it once, and its own sources are visited the same way; the visit stops at a Clean
638 // source, whose walk passes anyway, and at one already flagged, whose sources were visited when it
639 // was flagged. Colours are kept, so the next pull still recomputes exactly what changed.
640 // NOLINTNEXTLINE(misc-no-recursion) -- the recursion depth is the graph's depth
641 void exposeStaleSources() noexcept {
642 for (Node* const source : _sources) {
643 if (source->_colour != Colour::Clean && !source->_staleAboveClean) {
644 source->_staleAboveClean = true;
645 source->exposeStaleSources();
646 }
647 }
648 }
649
650 std::shared_ptr<RuntimeCore> _core;
651 std::vector<Node*> _sources;
652 std::vector<Node*> _observers;
653 // Points at the running pull's flag while one runs on this node.
654 bool* _destroyedWatch = nullptr;
655 std::uint64_t _epoch = 0;
656 std::uint64_t _creationOrder = 0;
657 std::size_t _depth;
658 std::size_t _runs = 0;
659 Colour _colour = Colour::Clean;
660 // Stale, with an observer that may be Clean: the next markStale walks through instead of stopping.
661 bool _staleAboveClean = false;
662 bool _bornOnOwner;
663};

665/// @brief The flush queue's heap order: the node that runs first is on top.
666/// @param lhs A queued node.
667/// @param rhs Another queued node.
668/// @return Whether @p lhs runs after @p rhs.
669inline bool runsLater(Node const* lhs, Node const* rhs) noexcept { return lhs->runsAfter(*rhs); }
670
671inline void RuntimeCore::enqueue(Node& effect) {
672 _queue.push_back(&effect);
673 std::ranges::push_heap(_queue, runsLater);
674}
675
676// Queued Effects run shallowest scope first, then oldest first. An owner's Effects therefore run before those
677// of the scopes it owns, and an Effect a run creates is newer than the run's own Effect: either way a parent
678// runs before every child it mounted, and unmounts a child before that child can see the state the parent
679// is leaving, whatever order the writes came in. Each run is pulled, so the order changes nothing a run
680// reads: only which Effects still exist to run.
681inline void RuntimeCore::flush() {
682 _isFlushRequested = false;
683 if (_detached || _queue.empty()) {
684 return;
685 }
686 if (_widgetEventDepth != 0) {
687 // A nested event loop inside a widget handler (a modal dialog) ran this flush. A remount must not
688 // destroy the widget whose handler is on the stack, so the queue waits: the widget event's batch is
689 // still open, and closing it, at the end of the outermost widget event, posts the flush again. Not
690 // re-posting here keeps the nested loop from spinning on it.
691 return;
692 }
693 _flushing = true;
694 ++_flushEpoch;
695 bool stopped = false;
696 while (!_queue.empty()) {
697 std::ranges::pop_heap(_queue, runsLater);
698 Node* const effect = _queue.back();
699 _queue.pop_back();
700 if (effect->countRun(_flushEpoch) > _bounds.maxEffectRunsPerFlush) {
701 report(site::kWriteCycle);
702 effect->settle();
703 dropQueue();
704 break;
705 }
706 try {
707 effect->updateIfNecessary();
708 } catch (...) {
709 report(site::kEffectThrew);
710 stopped = true;
711 break;
712 }
713 }
714 _flushing = false;
715 if (!stopped) {
716 _throwReposts = 0;
717 } else if (_throwReposts < _bounds.maxThrowReposts) {
718 // What is still queued runs in a flush of its own, so a defect does not stall it until the next write;
719 // a bounded number of times in a row, so an Effect that throws on every run cannot loop for ever. Past
720 // the bound the queue waits for the next write.
721 ++_throwReposts;
722 requestFlush();
723 }
724 if (_afterFlush) {
725 // The hook runs as a task of the owner executor, which has no use for the exception; the flush is
726 // already complete, so reporting it loses nothing.
727 try {
728 _afterFlush();
729 } catch (...) {
730 report(site::kAfterFlushThrew);
731 }
732 }
733}

735inline void RuntimeCore::forget(Node& node) noexcept {
736 if (std::erase(_queue, &node) != 0) {
737 std::ranges::make_heap(_queue, runsLater);
738 }
739 for (TrackingFrame* frame = _tracking; frame != nullptr; frame = frame->previous()) {
740 frame->forget(node);
741 }
742}

744inline void RuntimeCore::dropQueue() noexcept {
745 for (Node* const node : _queue) {
746 node->settle();
747 }
748 _queue.clear();
749}
