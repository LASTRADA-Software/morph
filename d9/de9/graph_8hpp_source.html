69/// @brief Whether `==` is usable for @p T: declared, and its element types usable too.
70///
71/// `std::equality_comparable` alone is not enough. The standard containers declare an unconstrained
72/// `operator==`, so it holds for a container whose element has no `==`, and comparing two of them then fails
73/// to compile. So a range is checked through its element type, and `std::optional`, `std::pair`, `std::tuple`
74/// and `std::variant` through theirs.
75/// @tparam T The value type, without cv-qualifiers.
76template <typename T>
77struct EqualityUsable : std::bool_constant<std::equality_comparable<T>> {};
78
79/// @brief A range is usable when it declares `==` and its element type is usable. A range whose element is
80/// itself (`std::filesystem::path`) is checked as a plain type.
81/// @tparam T The range type.
82template <typename T>
83 requires std::ranges::range<T const> && (!std::same_as<std::remove_cv_t<std::ranges::range_value_t<T const>>, T>)
84struct EqualityUsable<T>
85 : std::bool_constant<std::equality_comparable<T> &&
86 EqualityUsable<std::remove_cv_t<std::ranges::range_value_t<T const>>>::value> {};
87
88/// @brief An optional is usable when its value type is.
89/// @tparam T The value type.
90template <typename T>
91struct EqualityUsable<std::optional<T>> : EqualityUsable<std::remove_cv_t<T>> {};
92
93/// @brief A pair is usable when both its types are.
94/// @tparam First The first type.
95/// @tparam Second The second type.
96template <typename First, typename Second>
97struct EqualityUsable<std::pair<First, Second>> : std::bool_constant<EqualityUsable<std::remove_cv_t<First>>::value &&
98 EqualityUsable<std::remove_cv_t<Second>>::value> {
99};
100
101/// @brief A tuple is usable when every element type is.
102/// @tparam Ts The element types.
103template <typename... Ts>
104struct EqualityUsable<std::tuple<Ts...>> : std::bool_constant<(EqualityUsable<std::remove_cv_t<Ts>>::value && ...)> {};
105
106/// @brief A variant is usable when every alternative is.
107/// @tparam Ts The alternatives.
108template <typename... Ts>
109struct EqualityUsable<std::variant<Ts...>> : std::bool_constant<(EqualityUsable<std::remove_cv_t<Ts>>::value && ...)> {
110};

130/// @brief State shared by one `Runtime` and every node made from it.
131///
132/// Nodes hold it by `shared_ptr`, so a node that outlives its `Runtime` (a
133/// reported misuse) still has a valid core. The posted flush holds it weakly
134/// and checks `detach()`, so a flush after the `Runtime` is gone does nothing.
135class RuntimeCore : public std::enable_shared_from_this<RuntimeCore> {
136public:
137 /// @param owner The executor every operation must run on, and the one flushes are posted to. It must
138 /// queue what is posted and run it later, in order: an executor that runs a post inline would
139 /// run Effects inside `set()`. Borrowed: it must outlive the core.
140 explicit RuntimeCore(exec::IExecutor& owner MORPH_LIFETIMEBOUND) noexcept : _affinity{owner} {}
141
142 ~RuntimeCore() = default;
143 RuntimeCore(RuntimeCore const&) = delete;
144 RuntimeCore& operator=(RuntimeCore const&) = delete;
145 RuntimeCore(RuntimeCore&&) = delete;
146 RuntimeCore& operator=(RuntimeCore&&) = delete;
147
148 /// @brief Whether the calling thread is on the owner, without reporting anything.
149 /// @return What `checkOwner()` returns, minus the report.
150 [[nodiscard]] bool onOwner() const noexcept { return _affinity.here(); }
151
152 /// @brief Whether the calling thread is on the owner, reporting `site::kOffOwner` when it is not.
153 /// @return True on the owner: inside one of its tasks, or anywhere on the thread that built the Runtime
154 /// when that thread was running no executor's task at the time (a Qt slot, `main()`, a test
155 /// body). That second half is a per-thread check: on that thread a task of another executor
156 /// passes too, so running a different executor's tasks there is not detected.
157 [[nodiscard]] bool checkOwner() const noexcept {
158 if (_affinity.here()) {
159 return true;
160 }
161 report(site::kOffOwner);
162 return false;
163 }
164
165 /// @brief Reports a misuse: asserts in a debug build, or hands it to an installed owner probe.
166 /// @param where One of the `site::` constants.
167 void report(char const* where) const noexcept {
168 // Counted with atomics: a misuse off the owner is reported from the thread that made it.
169 std::uint64_t const index = _misuses.fetch_add(1, std::memory_order_relaxed);
170 _recentMisuses.at(index % kRecentMisuses).store(where, std::memory_order_relaxed);
171 exec::detail::noteOwner(where, _affinity.owner().coreExecutor(), false);
172 }
173
174 /// @brief How many misuses have been reported.
175 /// @return The count.
176 [[nodiscard]] std::uint64_t misuses() const noexcept { return _misuses.load(std::memory_order_relaxed); }
177
178 /// @brief The sites of the latest misuses, oldest first, at most `kRecentMisuses`.
179 /// @return The `site::` constants.
180 [[nodiscard]] std::vector<char const*> recentMisuses() const {
181 std::uint64_t const total = misuses();
182 std::uint64_t const kept = std::min<std::uint64_t>(total, kRecentMisuses);
183 std::vector<char const*> sites;
184 sites.reserve(static_cast<std::size_t>(kept));
185 for (std::uint64_t index = total - kept; index < total; ++index) {
186 if (char const* const where = _recentMisuses.at(index % kRecentMisuses).load(std::memory_order_relaxed)) {
187 sites.push_back(where);
188 }
189 }
190 return sites;
191 }

193 /// @brief The owner executor.
194 /// @return The executor passed at construction.
195 [[nodiscard]] exec::IExecutor& owner() const noexcept { return _affinity.owner(); }

197 /// @brief Sets the flush bounds and the after-flush hook.
198 /// @param bounds How often one Effect may run in a flush, and how many flushes in a row are re-posted after

200 /// @param afterFlush Called after every flush that processed a queued Effect; may be empty.
201 void configure(FlushBounds bounds, std::function<void()> afterFlush) {
202 _bounds = bounds;

237 /// @brief Closes a batch; closing the outermost one requests a flush when effects are queued.
238 ///
239 /// The outermost batch closing outside a flush is a write from outside the graph, which starts a new count
240 /// of the re-posts after a throwing Effect.
241 void endBatch() {
242 if (--_batchDepth == 0) {
243 if (!_flushing) {
244 _throwReposts = 0;
245 }
246 requestFlush();
247 }
248 }

250 /// @brief Queues an Effect that turned stale. Defined below `Node`.
251 /// @param effect The Effect node.
252 void enqueue(Node& effect);

254 /// @brief Posts one flush to the owner, unless one is already requested or running, nothing is
255 /// queued, or the Runtime is gone.
256 void requestFlush() {
257 if (_flushing || _isFlushRequested || _queue.empty() || _detached) {
258 return;
259 }
260 // Set before posting, so the flag is already down again if the flush runs inside post().
261 _isFlushRequested = true;
262 try {
263 _affinity.owner().post([weak = weak_from_this()] {
264 if (auto const core = weak.lock()) {
265 core->flush();
266 }
267 });
268 } catch (...) {
269 // Nothing was posted: the queue is kept, and the next write posts again.
270 _isFlushRequested = false;
271 throw;
272 }
273 }

307 /// @return Zero outside every batch.

309
310 /// @brief Whether a widget callback is running on the owner.
311 /// @return True between `enterWidgetEvent()` and the matching `leaveWidgetEvent()`.
312 [[nodiscard]] bool isInWidgetEvent() const noexcept { return _widgetEventDepth != 0; }
313
314 /// @brief Whether a flush is posted and has not run yet.
315 /// @return The coalescing flag.
316 [[nodiscard]] bool isFlushRequested() const noexcept { return _isFlushRequested; }
317
318 /// @brief Marks the Runtime as gone: flushes do nothing from now on.
319 void detach() noexcept { _detached = true; }

322 /// @return The node's creation order: greater than that of every node made from this runtime before it.
323 [[nodiscard]] std::uint64_t nodeCreated() noexcept {
324 ++_liveNodes;
325 return ++_createdNodes;
326 }

328 /// @brief Counts a node destroyed.
329 void nodeDestroyed() noexcept { --_liveNodes; }

331 /// @brief How many nodes made from this runtime are alive.
332 /// @return The live-node count.
333 [[nodiscard]] std::size_t liveNodes() const noexcept { return _liveNodes; }
334
335private:
336 void dropQueue() noexcept;
337
338 exec::detail::OwnerAffinity _affinity;
339 // A binary heap ordered by depth, then creation, the shallowest and oldest on top: see `flush()`.
340 std::vector<Node*> _queue;
341 std::function<void()> _afterFlush;
342 TrackingFrame* _tracking = nullptr;
343 FlushBounds _bounds;
344 // Flushes re-posted in a row after an Effect threw.
345 std::size_t _throwReposts = 0;
346 std::size_t _batchDepth = 0;
347 std::size_t _computing = 0;
348 std::size_t _widgetEventDepth = 0;
349 std::size_t _liveNodes = 0;
350 std::uint64_t _createdNodes = 0;
351 std::uint64_t _flushEpoch = 0;
352 std::uint64_t _batches = 0;
353 std::uint64_t _flushes = 0;
354 std::uint64_t _effectsRun = 0;
355 std::uint64_t _lastFlushEffects = 0;
356 std::uint64_t _maxFlushEffects = 0;
357 mutable std::atomic<std::uint64_t> _misuses{0};
358 mutable std::array<std::atomic<char const*>, kRecentMisuses> _recentMisuses{};
359 bool _isFlushRequested = false;
360 bool _flushing = false;
361 bool _detached = false;
362};

364/// @brief Collects the sources one tracked run reads, and restores the previous frame when it ends.
365///
366/// A frame whose observer is null is an untracked region: reads inside it subscribe nothing.
367class TrackingFrame {
368public:
369 /// @param core The runtime core.
370 /// @param observer The node whose run this is, or null for an untracked region.
371 TrackingFrame(RuntimeCore& core, Node* observer) noexcept
372 : _core{&core}, _previous{core.tracking()}, _observer{observer} {
373 core.setTracking(this);
374 }
375
376 /// @brief Reinstalls the enclosing frame.
377 ~TrackingFrame() { _core->setTracking(_previous); }
378 TrackingFrame(TrackingFrame const&) = delete;
379 TrackingFrame& operator=(TrackingFrame const&) = delete;
380 TrackingFrame(TrackingFrame&&) = delete;
381 TrackingFrame& operator=(TrackingFrame&&) = delete;
382
383 /// @brief Records that the run read @p source; a no-op in an untracked region.
384 /// @param source The node read.
385 void add(Node& source) {
386 if (_observer != nullptr && std::ranges::find(_sources, &source) == _sources.end()) {
387 _sources.push_back(&source);
388 }
389 }
390
391 /// @brief Drops @p node, which is being destroyed, from the recorded sources. When @p node is this
392 /// frame's observer, the frame stops recording and remembers that its observer is gone.
393 /// @param node The node.
394 void forget(Node& node) noexcept {
395 std::erase(_sources, &node);
396 if (&node == _observer) {
397 _observer = nullptr;
398 _observerDestroyed = true;
399 }
400 }
401
402 /// @brief Whether the observer was destroyed during the run: a body may destroy its own node (a
403 /// remount inside an Effect), and the node's run must not touch it afterwards.
404 /// @return True once the observer's destructor has run.
405 [[nodiscard]] bool observerDestroyed() const noexcept { return _observerDestroyed; }
406
407 /// @brief Hands the recorded sources to the observer.
408 /// @return The sources, deduplicated, in first-read order.
409 [[nodiscard]] std::vector<Node*> takeSources() noexcept { return std::move(_sources); }
410
411 /// @brief The enclosing frame.
412 /// @return The frame that was innermost when this one was opened, or null.
413 [[nodiscard]] TrackingFrame* previous() const noexcept { return _previous; }
414
415private:
416 RuntimeCore* _core;
417 TrackingFrame* _previous;
418 Node* _observer;
419 std::vector<Node*> _sources;
420 bool _observerDestroyed = false;
421};

423/// @brief A graph node: a source (Signal), an observer (Effect), or both (Computed).
424///
425/// Non-copyable and non-movable: other nodes hold raw pointers to it, and its
426/// destructor unlinks it in both directions. A derived node supplies `recompute()`
427/// and `isEffect()` as private overrides; only `Node` calls them.
428///
429/// A node may be destroyed while it is being brought up to date (an Effect body
430/// that remounts the view owning it). The destructor tells the running pull and
431/// the running tracking frame, and neither touches the node again.
432class Node {
433public:
434 /// @brief Constructs a node. Off the owner this is reported (`site::kOffOwner`) and the node is
435 /// not counted towards the core's live nodes, whose counter belongs to the owner.
436 /// @param core The runtime core this node belongs to.
437 /// @param depth How deeply the scope owning this node is nested; queued Effects run shallowest first.
438 explicit Node(std::shared_ptr<RuntimeCore> core, std::size_t depth = 0)
439 : _core{std::move(core)}, _depth{depth}, _bornOnOwner{_core->checkOwner()} {
440 if (_bornOnOwner) {
441 _creationOrder = _core->nodeCreated();
442 }
443 }
444
445 /// @brief Unlinks this node from its sources and observers, drops every reference the core holds
446 /// to it, and tells a pull running on it that it is gone.
447 ///
448 /// Destroying a node off the owner is a misuse. It is reported (`site::kOffOwner`) but cannot be
449 /// refused: the destructor still edits the owner's graph, queue and counters, and what that does
450 /// beyond the report is undefined.
451 virtual ~Node() {
452 static_cast<void>(_core->checkOwner());
453 if (_destroyedWatch != nullptr) {
454 *_destroyedWatch = true;
455 }
456 for (Node* const source : _sources) {
457 std::erase(source->_observers, this);
458 }
459 for (Node* const observer : _observers) {
460 std::erase(observer->_sources, this);
461 }
462 _core->forget(*this);
463 if (_bornOnOwner) {
464 _core->nodeDestroyed();
465 }
466 }
467
468 Node(Node const&) = delete;
469 Node& operator=(Node const&) = delete;
470 Node(Node&&) = delete;
471 Node& operator=(Node&&) = delete;
472
473 /// @brief Push phase: raises this node to @p colour and marks its observers Check.
474 ///
475 /// An Effect is queued when it leaves Clean. A node already at least as stale stops the walk, because
476 /// its observers were marked when it turned stale, unless an observer has since ended Clean above it
477 /// (`exposeStaleSources()`); then the walk passes through it once.
478 /// @param colour Dirty for a direct observer of a change, Check for a transitive one.
479 // NOLINTNEXTLINE(misc-no-recursion) -- the recursion depth is the graph's depth
480 void markStale(Colour colour) {
481 if (_colour >= colour && !_staleAboveClean) {
482 return;
483 }
484 if (_colour < colour) {
485 if (_colour == Colour::Clean && isEffect()) {
486 // Queued before the colour changes: an enqueue that fails to allocate leaves the node
487 // Clean, so a later walk that reaches it queues it again.
488 _core->enqueue(*this);
489 }
490 _colour = colour;
491 }
492 // Cleared before the walk: two flagged nodes can observe each other (a cycle a refused self-read
493 // leaves linked), and a revisit must stop here. Restored if the walk throws, so the next write
494 // passes through again; a node this walk raised from Clean is not flagged, and an Effect behind
495 // it whose enqueue failed is not reached again.
496 bool const flagged = std::exchange(_staleAboveClean, false);
497 try {
498 for (Node* const observer : _observers) {
499 observer->markStale(Colour::Check);
500 }
501 } catch (...) {
502 _staleAboveClean = flagged;
503 throw;
504 }
505 }
506
507 /// @brief Pull phase: brings this node up to date, recomputing only if a source really changed.
508 ///
509 /// A pull that throws leaves the node Clean (so later changes still reach it) and rethrows. A pull
510 /// during which the node is destroyed returns without touching it.
511 /// @return False when the node was destroyed during the pull: the caller must not touch it either.
512 // GCC sees `_destroyedWatch` keep the address of the local `destroyed` on the path where the
513 // node is destroyed during the pull. On that path `this` is gone too and nothing reads the
514 // member again; on every other path `endPull` restores the enclosing watch before returning.
515#if defined(__GNUC__) && !defined(__clang__)
516#pragma GCC diagnostic push
517#pragma GCC diagnostic ignored "-Wdangling-pointer"
518#endif
519 // NOLINTNEXTLINE(misc-no-recursion) -- the recursion depth is the graph's depth
520 bool updateIfNecessary() {
521 bool destroyed = false;
522 bool* const enclosing = std::exchange(_destroyedWatch, &destroyed);
523 // Once the pull has destroyed this node, `this` is gone: not even a member call may follow.
524 auto const finish = [&] {
525 if (destroyed) {
526 if (enclosing != nullptr) {
527 *enclosing = true;
528 }
529 return;
530 }
531 endPull(enclosing);
532 };
533 try {
534 pull(destroyed);
535 } catch (...) {
536 finish();
537 throw;
538 }
539 finish();
540 return !destroyed;
541 }
542#if defined(__GNUC__) && !defined(__clang__)
543#pragma GCC diagnostic pop
544#endif

613 /// @brief Replaces this node's sources with @p fresh, unlinking the ones no longer read.
614 ///
615 /// Strong guarantee: linking the new sources is the only step that allocates, and it runs first;
616 /// if it throws, the links it made are undone, so no source keeps a link this node does not list.
617 /// @param fresh The sources the latest run read.
618 void adoptSources(std::vector<Node*> fresh) {
619 std::size_t linked = 0;
620 try {
621 for (Node* const source : fresh) {
622 if (std::ranges::find(_sources, source) == _sources.end()) {
623 source->_observers.push_back(this);
624 }
625 ++linked;
626 }
627 } catch (...) {
628 for (Node* const source : std::span{fresh}.first(linked)) {
629 if (std::ranges::find(_sources, source) == _sources.end()) {
630 std::erase(source->_observers, this);
631 }
632 }
633 throw;
634 }
635 for (Node* const old : _sources) {
636 if (std::ranges::find(fresh, old) == fresh.end()) {
637 std::erase(old->_observers, this);
638 }
639 }
640 _sources = std::move(fresh);
641 }

643 /// @brief Adds @p fresh to this node's sources without dropping any: what a failed run keeps.
644 /// @param fresh The sources the failed run read before it threw.
645 void mergeSources(std::vector<Node*> fresh) {

647 std::ranges::copy_if(_sources, std::back_inserter(kept),
648 [&fresh](Node const* old) { return std::ranges::find(fresh, old) == fresh.end(); });
649 fresh.insert(fresh.end(), kept.begin(), kept.end());

654 /// @brief Brings the node's value up to date.
655 /// @return True when observers must treat the value as changed.
656 virtual bool recompute() = 0;

662 // Returns as soon as `destroyed` is set: from then on `this` is gone.
663 // NOLINTNEXTLINE(misc-no-recursion) -- the recursion depth is the graph's depth
664 void pull(bool const& destroyed) {
665 if (_colour == Colour::Check) {
666 for (std::size_t i = 0; i < _sources.size();) {
667 Node* const source = _sources.at(i);
668 source->updateIfNecessary();
669 if (destroyed) {
670 return;
671 }
672 if (_colour == Colour::Dirty) {
673 break;
674 }
675 // A source destroyed by its own pull has unlinked itself, and the next one now sits at i.
676 if (i < _sources.size() && _sources.at(i) == source) {
677 ++i;
678 }
679 }
680 }
681 if (_colour == Colour::Dirty) {
682 bool const changed = recompute();
683 if (destroyed) {
684 return;
685 }
686 if (changed) {
687 markObserversStale(Colour::Dirty);
688 }
689 }
690 }

692 void endPull(bool* enclosing) noexcept {
693 _destroyedWatch = enclosing;
694 _colour = Colour::Clean;
695 _staleAboveClean = false;
696 exposeStaleSources();
697 }
698
699 // This node is Clean while a source may still be stale: an Effect settled by a write cycle without
700 // running, an Effect whose run wrote a source of a Computed it had read (the write cannot mark the
701 // running Effect), or a run that threw before reading every source. A stale source stops the next
702 // write's walk, so this node would never hear of it. Each stale source is flagged, so that walk
703 // passes through it once, and its own sources are visited the same way; the visit stops at a Clean
704 // source, whose walk passes anyway, and at one already flagged, whose sources were visited when it
705 // was flagged. Colours are kept, so the next pull still recomputes exactly what changed.
706 // NOLINTNEXTLINE(misc-no-recursion) -- the recursion depth is the graph's depth
707 void exposeStaleSources() noexcept {
708 for (Node* const source : _sources) {
709 if (source->_colour != Colour::Clean && !source->_staleAboveClean) {
710 source->_staleAboveClean = true;
711 source->exposeStaleSources();
712 }
713 }
714 }

716 std::shared_ptr<RuntimeCore> _core;
717 std::vector<Node*> _sources;
718 std::vector<Node*> _observers;
719 // Points at the running pull's flag while one runs on this node.
720 bool* _destroyedWatch = nullptr;
721 std::uint64_t _epoch = 0;
722 std::uint64_t _creationOrder = 0;
723 std::size_t _depth;
724 std::size_t _runs = 0;
725 Colour _colour = Colour::Clean;
726 // Stale, with an observer that may be Clean: the next markStale walks through instead of stopping.
727 bool _staleAboveClean = false;
728 bool _bornOnOwner;
729};

731/// @brief The flush queue's heap order: the node that runs first is on top.
732/// @param lhs A queued node.
733/// @param rhs Another queued node.
734/// @return Whether @p lhs runs after @p rhs.
735inline bool runsLater(Node const* lhs, Node const* rhs) noexcept { return lhs->runsAfter(*rhs); }
736
737inline void RuntimeCore::enqueue(Node& effect) {
738 _queue.push_back(&effect);
739 std::ranges::push_heap(_queue, runsLater);
740}
741
742// Queued Effects run shallowest scope first, then oldest first. An owner's Effects therefore run before those
743// of the scopes it owns, and an Effect a run creates is newer than the run's own Effect: either way a parent
744// runs before every child it mounted, and unmounts a child before that child can see the state the parent
745// is leaving, whatever order the writes came in. Each run is pulled, so the order changes nothing a run
746// reads: only which Effects still exist to run.
747inline void RuntimeCore::flush() {
748 _isFlushRequested = false;
749 if (_detached || _queue.empty()) {
750 return;
751 }
752 if (_widgetEventDepth != 0) {
753 // A nested event loop inside a widget handler (a modal dialog) ran this flush. A remount must not
754 // destroy the widget whose handler is on the stack, so the queue waits: the widget event's batch is
755 // still open, and closing it, at the end of the outermost widget event, posts the flush again. Not
756 // re-posting here keeps the nested loop from spinning on it.
757 return;
758 }
759 _flushing = true;
760 ++_flushEpoch;
761 ++_flushes;
762 _lastFlushEffects = 0;
763 bool stopped = false;
764 while (!_queue.empty()) {
765 std::ranges::pop_heap(_queue, runsLater);
766 Node* const effect = _queue.back();
767 _queue.pop_back();
768 ++_effectsRun;
769 _maxFlushEffects = std::max(_maxFlushEffects, ++_lastFlushEffects);
770 if (effect->countRun(_flushEpoch) > _bounds.maxEffectRunsPerFlush) {
771 report(site::kWriteCycle);
772 effect->settle();
773 dropQueue();
774 break;
775 }
776 try {
777 effect->updateIfNecessary();
778 } catch (...) {
779 report(site::kEffectThrew);
780 stopped = true;
781 break;
782 }
783 }
784 _flushing = false;
785 if (!stopped) {
786 _throwReposts = 0;
787 } else if (_throwReposts < _bounds.maxThrowReposts) {
788 // What is still queued runs in a flush of its own, so a defect does not stall it until the next write;
789 // a bounded number of times in a row, so an Effect that throws on every run cannot loop for ever. Past
790 // the bound the queue waits for the next write.
791 ++_throwReposts;
792 requestFlush();
793 }
794 if (_afterFlush) {
795 // The hook runs as a task of the owner executor, which has no use for the exception; the flush is
796 // already complete, so reporting it loses nothing.
797 try {
798 _afterFlush();
799 } catch (...) {
800 report(site::kAfterFlushThrew);
801 }
802 }
803}

805inline void RuntimeCore::forget(Node& node) noexcept {
806 if (std::erase(_queue, &node) != 0) {
807 std::ranges::make_heap(_queue, runsLater);
808 }
809 for (TrackingFrame* frame = _tracking; frame != nullptr; frame = frame->previous()) {
810 frame->forget(node);
811 }
812}
