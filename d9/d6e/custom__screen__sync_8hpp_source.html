45/// @brief The custom documents a client has fetched, by id: what `DocumentScreenHostOptions::documents` answers for a
46/// `custom:` id.
47///
48/// The documents lookup is synchronous, so whatever opens a custom screen (a navigator entry, **Open as**, a restored
49/// tab) has its document fetched into the cache first. A list that shows a kept screen with another hash, or a whole
50/// list that no longer shows it, drops it, so the next open fetches the stored document again.
51class MORPH_API CustomDocuments {
52public:
53 /// @brief One fetched screen.
54 struct Entry {
55 /// @brief Its summary, as `get` answered it.
56 CustomScreenSummary summary;
57 /// @brief Its document's JSON text.
58 std::string text;
59 /// @brief Its document, loaded.
60 std::shared_ptr<Document const> document;
61 };
62
63 /// @brief Loads what `get` answered and keeps it under its id, replacing what was kept.
64 /// @param found The reply.
65 /// @param options How documents load: the catalog facts.
66 /// @return The loaded document, or why it does not load in this client, why the screen is broken, or that it names
67 /// another id; nothing is kept then: a broken screen is never kept.
68 std::expected<std::shared_ptr<Document const>, Refusal> put(CustomScreenFound found, LoadOptions const& options);
69
70 /// @brief The loaded document of a kept screen.
71 /// @param screenId The screen's id.
72 /// @return It, or null when the screen is not kept.
73 [[nodiscard]] std::shared_ptr<Document const> find(std::string_view screenId) const;
74
75 /// @brief A kept screen.
76 /// @param screenId The screen's id.
77 /// @return It, valid until the cache next changes; null when the screen is not kept.
78 [[nodiscard]] Entry const* entry(std::string_view screenId) const;
79
80 /// @brief Drops each kept screen @p listed shows with another hash and, unless the list was truncated, each it
81 /// does not show.
82 /// @param listed What `list` answered.
83 /// @return The ids dropped.
84 std::vector<std::string> refresh(CustomScreensListed const& listed);
85
86 /// @brief Drops a kept screen.
87 /// @param screenId The screen's id; one not kept changes nothing.
88 void forget(std::string_view screenId);
89
90 /// @brief Drops every kept screen: what a change of session does.
91 void clear() noexcept { _entries.clear(); }
92
93private:
94 std::map<std::string, Entry, std::less<>> _entries;
95};

97/// @brief Whether a server offers custom screens, as far as a client knows.
98enum class CustomScreensOffer : std::uint8_t {
99 Unknown, ///< Not asked yet, or asked under a session that has since changed.
100 Offered, ///< `list` answered.
101 Off, ///< The registration of `morph.customScreens`, or `list`, was refused.
102};

104/// @brief What an open or a fetch answers: the loaded document, or why there is none.
105using CustomDocumentDone = std::function<void(std::expected<std::shared_ptr<Document const>, std::string>)>;

107/// @brief The client of a server's custom screens model (`morph.customScreens`) for one application, through the
108/// interpreter's own `Dispatcher`, so it travels the same bridge, session and backend as the screens' models.
109///
110/// No call names a person: the server keys every screen by the verified session's principal. Every reply arrives on
111/// the runtime's owner. Replies still in flight when the sync is destroyed or `reset` are dropped.
112class MORPH_API CustomScreenSync {
113public:
114 /// @brief A sync for @p application.
115 /// @param dispatcher Where model calls go: the interpreter's own. Borrowed: it must outlive the sync.
116 /// @param application The application's id, the `app` of every call.
117 /// @param catalog The catalog facts a fetched document loads with, read at each load.
118 /// @param session The session whose watch set carries the record `morph.customScreens` once the server offers
119 /// custom screens, so a save from anywhere relists and every watch set the session sends,
120 /// `reconnected`'s included, names it; null for a source whose server sends no events, such as
121 /// an in-process one. Borrowed: it must outlive the sync.
122 CustomScreenSync(Dispatcher& dispatcher MORPH_LIFETIMEBOUND, std::string application,
123 std::function<Catalog()> catalog, Session* session = nullptr);
124 CustomScreenSync(CustomScreenSync const&) = delete;
125 CustomScreenSync& operator=(CustomScreenSync const&) = delete;
126 CustomScreenSync(CustomScreenSync&&) = delete;
127 CustomScreenSync& operator=(CustomScreenSync&&) = delete;
128 ~CustomScreenSync();
129
130 /// @brief Asks whether the server offers custom screens, by calling `list`.
131 ///
132 /// The list answering makes the offer `Offered`, keeps the list, tells `setOnListed`'s handler and has the session
133 /// watch the record. Any refusal makes it `Off`: a server that never registered the model refuses its registration
134 /// with `unknown model type: morph.customScreens`, which is how a client learns custom screens are not offered;
135 /// any other failure is logged at warn too, worded by its cause (refused, unreadable reply, or the transport),
136 /// since it is a misconfiguration or an outage rather than an absence.
137 /// @param done Called once with whether they are offered; may be empty.
138 void probe(std::function<void(bool offered)> done = {});
139
140 /// @brief What the last probe found.
141 /// @return The offer.
142 [[nodiscard]] CustomScreensOffer offer() const noexcept { return _offer; }
143
144 /// @brief Why the offer is `Off`.
145 /// @return The refusal's text; empty unless the offer is `Off`.
146 [[nodiscard]] std::string const& offRefusal() const noexcept { return _offRefusal; }
147
148 /// @brief Sets what is told each list the sync receives, from a probe or a relist.
149 /// @param onListed The handler: the list, and the ids of the kept documents that list dropped (changed on the
150 /// server, or gone); empty tells nothing.
151 void setOnListed(
152 std::function<void(CustomScreensListed const&, std::vector<std::string> const& dropped)> onListed) {
153 _onListed = std::move(onListed);
154 }
155
156 /// @brief The last list received.
157 /// @return It; empty before the first.
158 [[nodiscard]] CustomScreensListed const& listed() const noexcept { return _listed; }
159
160 /// @brief Calls `list` again, while the offer is `Offered`: the cache drops what the list shows changed, and
161 /// `setOnListed`'s handler is told. A refusal is logged at warn and keeps the last list.
162 ///
163 /// The replies of one sync's calls arrive in the order the calls were made, because every call goes through
164 /// the one private channel; a list never overtakes a later one.
165 void relist();
166
167 /// @brief The document of a custom screen: the kept one, or `fetch`ed.
168 /// @param screenId The screen's id.
169 /// @param done Called once with the document, or why there is none. With a kept document it is called before
170 /// `open` returns; otherwise later, on the runtime's owner.
171 void open(std::string const& screenId, CustomDocumentDone done);
172
173 /// @brief Calls `get` and keeps the document it answers, replacing a kept one.
174 /// @param screenId The screen's id.
175 /// @param done Called once with the document, or why there is none: the store's refusal as it gave it, why the
176 /// document does not load in this client, or that the screen is broken, with the checks' reason. A
177 /// broken screen is never kept.
178 void fetch(std::string const& screenId, CustomDocumentDone done);
179
180 /// @brief Calls `get` and keeps nothing: what a source view of a broken screen reads.
181 /// @param screenId The screen's id.
182 /// @param done Called once with the summary and the document, or the refusal.
183 void get(std::string const& screenId, std::function<void(std::expected<CustomScreenFound, std::string>)> done);
184
185 /// @brief Asks the server whether @p document would be saved as a custom screen based on @p basedOn.
186 /// @param basedOn The base screen's id.
187 /// @param document The document's JSON text.
188 /// @param done Called once with the first refusal, or none, or why the call failed.
189 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the `check` action's order.
190 void check(std::string basedOn, std::string document,
191 std::function<void(std::expected<std::optional<CheckRefusal>, std::string>)> done);
192
193 /// @brief Saves a custom screen; its `app` is set to the sync's. A save that is answered relists.
194 /// @param request The screen to overwrite, if any, its title, base and document.
195 /// @param done Called once with the screen's id, or the refusal as the store or the checks gave it.
196 void save(CustomScreensSave request, std::function<void(std::expected<std::string, std::string>)> done);
197
198 /// @brief Fetches the document of every custom screen @p routes name that is not kept, so a start that reopens
199 /// them can read the routes back.
200 /// @param routes The routes, a workspace's record or a history's.
201 /// @param done Called once every fetch has answered, with the ids no document came for, in the routes' order.
202 void prefetchRoutes(std::vector<std::string> const& routes, std::function<void(std::vector<std::string>)> done);
203
204 /// @brief The connection was made again: while offered, relists; while off for a cause other than the server
205 /// lacking the model (a refusal, an unreadable reply, an outage), probes again. The watch on the record is
206 /// the session's: its own `reconnected` sends it.
207 void reconnected();
208
209 /// @brief A server event: a `changes` naming the record `morph.customScreens` relists, and a `resync` relists
210 /// (the session sends the watch again itself). Anything else, or anything before the offer is `Offered`,
211 /// is ignored.
212 /// @param event The event.
213 void deliverEvent(wire::EventBody const& event);
214
215 /// @brief Forgets everything learnt under the session that is ending: the offer, the list, the cache and the
216 /// binding. Replies still in flight are dropped.
217 void reset();
218
219 /// @brief The fetched documents.
220 /// @return The cache.
221 [[nodiscard]] CustomDocuments& documents() noexcept { return _documents; }
222
223 /// @brief The fetched documents.
224 /// @return The cache.
225 [[nodiscard]] CustomDocuments const& documents() const noexcept { return _documents; }
226
227private:
228 // What turned the offer off; only an absent model is not reported.
229 enum class OffCause : std::uint8_t { Absent, Refused, Unreadable, Transport };
230
231 // Sends one action through the binding, made at the first call; @p onReply gets the reply's JSON, @p failed the
232 // reason and the failure's kind (`detail::defaultErrorKind`).
233 void call(std::string_view action, std::string const& body, std::function<void(std::string const&)> onReply,
234 std::function<void(std::string const&, std::string const&)> const& failed);
235 void off(OffCause cause, std::string reason);
236 void listedNow(CustomScreensListed listed);
237
238 Dispatcher* _dispatcher;
239 std::string _application;
240 std::function<Catalog()> _catalog;
241 Session* _session;
242 std::shared_ptr<ModelChannel> _channel;
243 CustomScreensOffer _offer = CustomScreensOffer::Unknown;
244 std::string _offRefusal;
245 OffCause _offCause = OffCause::Absent;
246 CustomScreensListed _listed;
247 std::function<void(CustomScreensListed const&, std::vector<std::string> const&)> _onListed;
248 CustomDocuments _documents;
249 // Last, so it stops every delivery before anything a reply would touch goes; `reset` replaces it.
250 std::unique_ptr<async::CallbackScope> _scope = std::make_unique<async::CallbackScope>();
251};
