26/// @brief Reconnect tuning for `SocketBackend`.
27struct SocketBackendConfig {
28 /// @brief Whether to attempt automatic reconnect after an unsolicited disconnect.
29 bool reconnectEnabled = true;
30 /// @brief Delay before the first reconnect attempt after a disconnect.
31 std::chrono::milliseconds initialReconnectDelay{500};
32 /// @brief Upper bound on the exponential backoff between reconnect attempts.
33 std::chrono::milliseconds maxReconnectDelay{30000};
34 /// @brief Multiplier applied to the delay after each failed attempt.
35 double backoffMultiplier = 2.0;
36 /// @brief Maximum time to wait for the initial TCP connect to complete.
37 std::chrono::milliseconds connectTimeout{5000};
38};

40/// @brief `IBackend` implementation that communicates with a `RemoteServer`
41/// over a raw-socket RFC 6455 WebSocket connection — no Qt, no GUI
42/// event loop required.
43///
44/// Mirrors `morph::qt::QtWebSocketBackend`'s observable behavior
45/// (`registerModel` synchronous, `deregisterModel` fire-and-forget, `execute`
46/// asynchronous and callId-multiplexed, `DisconnectedError`/reconnect
47/// semantics) but runs its own dedicated I/O thread and uses
48/// `std::condition_variable` instead of a nested Qt event loop. Unlike
49/// `QtWebSocketBackend`, `SocketBackend` may safely be driven from multiple
50/// threads concurrently (`registerModel`/`execute`/`deregisterModel` are all
51/// thread-safe) — there is no single "owning" event-loop thread.
52///
53/// @par TLS
54/// Not supported. `serverUrl` must be `ws://`; `wss://` throws from the
55/// constructor. See `docs/spec/core/backend.md`'s `morph::net` section.
56class SocketBackend : public ::morph::backend::detail::IBackend {
57public:
58 /// @brief Alias for the reconnect configuration struct.
59 using Config = SocketBackendConfig;
60
61 /// @brief Parses @p serverUrl and starts the background I/O thread, which
62 /// connects asynchronously.
63 /// @param serverUrl `ws://host:port[/path]` URL of the remote `RemoteServer`.
64 /// @param cfg Reconnect tuning. Default: enabled, 500ms initial / 30s cap, 2x backoff.
65 /// @throws std::runtime_error immediately (before starting the I/O thread)
66 /// if @p serverUrl is not a well-formed `ws://` URL (see `parseWsUrl`).
67 // Copy/move are implicitly deleted by the non-copyable/non-movable
68 // std::mutex/std::condition_variable/std::thread members below — no
69 // explicit `= delete` needed (matches the rest of the codebase's
70 // convention, e.g. `LocalBackend`).
71 explicit SocketBackend(std::string serverUrl, Config cfg = {})
72 : _url{::morph::net::detail::parseWsUrl(serverUrl)},
73 _cfg{cfg},
74 _currentReconnectDelay{cfg.initialReconnectDelay} {
75 _ioThread = std::thread{[this] { ioThreadMain(); }};
76 _handlerThread = std::thread{[this] { handlerThreadMain(); }};
77 }
78
79 /// @brief Shuts down the I/O thread and resolves any still-pending completions.
80 ///
81 /// @note May block up to `Config::connectTimeout` if destruction races an
82 /// in-flight (re)connect attempt — the TCP connect phase is bounded by
83 /// that timeout, but the handshake read that follows a successful TCP
84 /// connect has no separate timeout of its own in this reference
85 /// implementation. See `docs/spec/core/backend.md`'s `morph::net` section.
86 ~SocketBackend() override {
87 _shuttingDown.store(true);
88 {
89 std::scoped_lock lock{_socketMtx};
90 if (_socket.valid()) {
91 _socket.shutdownBoth();
92 }
93 }
94 _reconnectCv.notify_all();
95 if (_ioThread.joinable()) {
96 _ioThread.join();
97 }
98 // Joined after _ioThread, not before: a reconnect handler parked in
99 // sendSync is released by onDisconnected()'s _syncCv notify, which only
100 // runs as ioThreadMain unwinds. Waking _handlerCv first would not free
101 // it -- that wait is on _syncCv.
102 {
103 std::scoped_lock const lock{_handlerMtx};
104 _handlerPending = false;
105 }
106 _handlerCv.notify_all();
107 if (_handlerThread.joinable()) {
108 _handlerThread.join();
109 }
110 cancelPending(std::make_exception_ptr(::morph::backend::DisconnectedError{}));
111 }

113 /// @brief Blocks the calling thread until connected or @p timeout elapses.
114 /// @param timeout Maximum time to wait.
115 /// @return `true` if connected before the timeout, `false` otherwise.
116 bool waitForConnected(std::chrono::milliseconds timeout = std::chrono::milliseconds{5000}) {
117 std::unique_lock lock{_connectMtx};
118 _connectCv.wait_for(lock, timeout, [this] { return _connected.load() || _shuttingDown.load(); });
119 return _connected.load();
120 }

122 /// @brief Sends a `register` message and blocks until the reply arrives.
123 ///
124 /// Thread-safe, but only one synchronous control call (`registerModel`) may
125 /// be in flight at a time across the whole backend; a second call while one
126 /// is outstanding throws immediately rather than queuing. The factory
127 /// argument is ignored — model construction is delegated to the server.
128 /// @param typeId String type-id of the model to register.
129 /// @return `ModelId` assigned by the server.
130 /// @throws std::runtime_error if the server replies with an error, the
131 /// socket is not connected, or a synchronous call is already in flight.
132 ::morph::exec::detail::ModelId registerModel(
133 const std::string& typeId,
134 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> /*factory*/) override {
135 auto env = ::morph::wire::makeRegister(typeId);
136 env.session = currentSession();
137 std::string replyJson;
138 try {
139 replyJson = sendSync(::morph::wire::encode(env));
140 } catch (const std::exception& exc) {
141 throw std::runtime_error(std::string{"register failed: "} + exc.what());
142 }
143 auto reply = ::morph::wire::decode(replyJson);
144 if (reply.kind == "ok") {
145 return ::morph::exec::detail::ModelId{reply.modelId};
146 }
147 throw std::runtime_error("register failed: " + reply.message);
148 }
149
150 /// @brief Sends a shared (register-or-attach) `register` and blocks for the reply.
151 ///
152 /// An empty primary degrades to the private path. Same synchronous-call
153 /// constraint as `registerModel`.
154 /// @param typeId String type-id of the model.
155 /// @param factory Ignored — the server constructs via its own registry.
156 /// @param identity Entity key for the action log plus the directory primary key.
157 /// @return `ModelId` of the shared (or newly created) instance.
158 /// @throws std::runtime_error if the server replies with an error or the socket is down.
159 ::morph::exec::detail::ModelId registerModelShared(
160 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
161 ::morph::backend::detail::InstanceIdentity identity) override {
162 if (identity.primary.empty()) {
163 return registerModelWithContext(typeId, std::move(factory), identity.contextKey);
164 }
165 auto env = ::morph::wire::makeRegisterShared(typeId, std::string{identity.primary},
166 std::string{identity.contextKey});
167 env.session = currentSession();
168 return sendControlForId(env, "register");
169 }
170
171 /// @brief Sends an `attach` and blocks for the reply, re-pointing from @p current.
172 /// @param typeId String type-id of the model.
173 /// @param factory Ignored — the server constructs via its own registry.
174 /// @param identity Entity key for the action log plus the directory primary key.
175 /// @param current Instance currently held, or `ModelId{0}` if none.
176 /// @return `ModelId` of the instance now attached to.
177 /// @throws std::runtime_error if the server replies with an error or the socket is down.
178 ::morph::exec::detail::ModelId attachModel(
179 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
180 ::morph::backend::detail::InstanceIdentity identity, ::morph::exec::detail::ModelId current) override {
181 if (identity.primary.empty()) {
182 if (current.v != 0U) {
183 deregisterModel(current);
184 }
185 return registerModelWithContext(typeId, std::move(factory), identity.contextKey);
186 }
187 auto env = ::morph::wire::makeAttach(typeId, std::string{identity.primary}, current.v,
188 std::string{identity.contextKey});
189 env.session = currentSession();
190 return sendControlForId(env, "attach");
191 }

207 /// @brief Asks the server for the live shared primary keys of @p typeId.
208 /// @param typeId String type-id to enumerate.
209 /// @return Canonical key strings of the live shared instances.
210 /// @throws std::runtime_error if the server replies with an error or the socket is down.
211 std::vector<std::string> listInstances(const std::string& typeId) override {
212 std::string replyJson;
213 try {
214 auto env = ::morph::wire::makeInstances(typeId);
215 env.session = currentSession();
216 replyJson = sendSync(::morph::wire::encode(env));
217 } catch (const std::exception& exc) {
218 throw std::runtime_error(std::string{"instances failed: "} + exc.what());
219 }
220 auto reply = ::morph::wire::decode(replyJson);
221 if (reply.kind != "ok") {
222 throw std::runtime_error("instances failed: " + reply.message);
223 }
224 std::vector<std::string> keys;
225 if (auto errCode = glz::read_json(keys, reply.body)) {
226 throw std::runtime_error("instances decode failed: " + glz::format_error(errCode, reply.body));
227 }
228 return keys;
229 }

231 /// @brief Sends a `deregister` message fire-and-forget (does not wait for a reply).
232 /// @param mid Id of the model to remove on the server.
233 void deregisterModel(::morph::exec::detail::ModelId mid) override {
234 if (_connected.load()) {
235 try {
236 auto env = ::morph::wire::makeDeregister(mid.v);
237 env.session = currentSession();
238 sendFrame(::morph::net::detail::WsOpcode::kText, ::morph::wire::encode(env));
239 } catch (const std::exception&) {
240 // Fire-and-forget: same documented trade-off as
241 // QtWebSocketBackend — a failed send just leaks the model on
242 // the server (see docs/spec/core/backend.md's Limitations).
243 }
244 }

266 /// @brief Sends an `execute` message and returns a `Completion` resolved on reply.
267 /// @param mid Target model id on the server.
268 /// @param call Bundled action; `serializeAction` and `deserializeResult` are used.
269 /// @param cbExec Executor for delivering the completion callbacks.
270 /// @return Completion resolved asynchronously when the server's reply arrives,
271 /// or immediately with `DisconnectedError` if not connected.
272 ::morph::async::Completion<std::shared_ptr<void>> execute(::morph::exec::detail::ModelId mid,
273 ::morph::backend::detail::ActionCall call,
274 ::morph::exec::IExecutor* cbExec) override {
275 auto compState = std::make_shared<::morph::async::detail::CompletionState<std::shared_ptr<void>>>();
276 ::morph::async::Completion<std::shared_ptr<void>> comp{compState, cbExec};
277
278 if (!_connected.load()) {
279 compState->setException(std::make_exception_ptr(::morph::backend::DisconnectedError{}));
280 return comp;
281 }
282
283 std::uint64_t const callId = ++_nextCallId;
284 ::morph::wire::Envelope env;
285 env.kind = "execute";
286 env.callId = callId;
287 env.modelId = mid.v;
288 env.modelType = call.modelTypeId;
289 env.actionType = call.actionTypeId;
290 env.body = call.serializeAction();
291 env.session = std::move(call.session);
292
293 {
294 std::scoped_lock lock{_pendingMtx};
295 _pending[callId] = PendingExecute{compState, std::move(call.deserializeResult), cbExec};
296 }
297
298 try {
299 sendFrame(::morph::net::detail::WsOpcode::kText, ::morph::wire::encode(env));
300 } catch (const std::exception&) {
301 // The write raced a disconnect; the io thread's disconnect
302 // handler drains _pending (including this entry) via cancelPending.
303 }
304 return comp;
305 }

310 /// @brief Resolves every pending execute call's `Completion` with @p exc.
311 /// @param exc Exception delivered to every pending completion's error sink.
312 void cancelPending(const std::exception_ptr& exc) override {
313 std::unordered_map<std::uint64_t, PendingExecute> drained;
314 {
315 std::scoped_lock lock{_pendingMtx};
316 drained.swap(_pending);
317 }
318 for (auto& [callId, pending] : drained) {
319 (void)callId;
320 if (pending.state) {
321 pending.state->setException(exc);
322 }
323 }
324 }

355 void sendFrame(::morph::net::detail::WsOpcode opcode, std::string_view payload) {
356 std::scoped_lock lock{_socketMtx};
357 if (!_socket.valid()) {
358 throw std::runtime_error("SocketBackend::sendFrame: not connected");
359 }
360 std::string frame = ::morph::net::detail::encodeWsFrame(opcode, payload, /*mask=*/true);
361 _socket.sendAll(frame.data(), frame.size());
362 }

364 std::string sendSync(const std::string& payload) {
365 std::unique_lock lock{_syncMtx};
366 if (_syncInFlight) {
367 throw std::runtime_error("sendSync: a synchronous call is already in flight (reentrant use)");
368 }
369 if (!_connected.load()) {
370 throw std::runtime_error("disconnected");
371 }
372 _syncInFlight = true;
373 _syncReply.reset();
374 lock.unlock();
375
376 try {
377 sendFrame(::morph::net::detail::WsOpcode::kText, payload);
378 } catch (const std::exception&) {
379 std::scoped_lock relock{_syncMtx};
380 _syncInFlight = false;
381 throw std::runtime_error("disconnected");
382 }
383
384 std::unique_lock waitLock{_syncMtx};
385 _syncCv.wait(waitLock, [this] { return _syncReply.has_value() || !_connected.load(); });
386 bool const gotReply = _syncReply.has_value();
387 std::string result = gotReply ? std::move(*_syncReply) : std::string{};
388 _syncReply.reset();
389 _syncInFlight = false;
390 if (!gotReply) {
391 throw std::runtime_error("disconnected");
392 }
393 return result;
394 }

396 void onConnected() {
397 bool const isReconnect = _everConnected.exchange(true);
398 _connected.store(true);
399 _currentReconnectDelay = _cfg.initialReconnectDelay;
400 _connectCv.notify_all();
401 if (isReconnect) {
402 // Hand the callback to _handlerThread rather than running it here.
403 // onConnected() is called from ioThreadMain() immediately *before*
404 // readLoop() starts, and a reconnect handler is expected to
405 // re-register its models (Bridge::installReconnectHandler does
406 // exactly that), which goes through sendSync -> wait on _syncCv for
407 // a reply that only readLoop can ever deliver. Run inline, that
408 // wait blocks the one thread responsible for satisfying it: the
409 // transport deadlocks permanently, with no timeout to break it.
410 // Off-thread, the handler's sendSync overlaps readLoop as intended
411 // -- its request may even reach the socket before readLoop starts,
412 // which is harmless, since the reply simply waits in the kernel
413 // buffer.
414 std::scoped_lock const lock{_handlerMtx};
415 _handlerPending = true;
416 _handlerCv.notify_all();
417 }
418 }
419
420 /// Serializes reconnect-handler invocations off the I/O thread. Coalescing
421 /// via a flag (rather than queuing every request) is deliberate: if a second
422 /// reconnect lands while a handler is still running, re-running it once
423 /// afterwards is the correct catch-up, and it bounds concurrent handler runs
424 /// to one.
425 void handlerThreadMain() {
426 for (;;) {
427 std::function<void()> handler;
428 {
429 std::unique_lock lock{_handlerMtx};
430 _handlerCv.wait(lock, [this] { return _handlerPending || _shuttingDown.load(); });
431 if (_shuttingDown.load()) {
432 return;
433 }
434 _handlerPending = false;
435 }
436 {
437 std::scoped_lock const lock{_reconnectHandlerMtx};
438 handler = _reconnectHandler;
439 }
440 if (!handler) {
441 continue;
442 }
443 try {
444 handler();
445 } catch (const std::exception& exc) {
446 // A handler that throws (typically because the link dropped
447 // again mid-re-registration, surfacing as "disconnected") must
448 // not take this thread down: the next reconnect has to find it
449 // still waiting.
450 ::morph::log::logWarn(std::string{"[net::SocketBackend] reconnect handler threw: "} + exc.what());
451 } catch (...) {
452 ::morph::log::logWarn("[net::SocketBackend] reconnect handler threw a non-std exception");
453 }
454 }
455 }

457 void onDisconnected() {
458 _connected.store(false);
459 {
460 std::scoped_lock lock{_socketMtx};
461 _socket = ::morph::net::detail::TcpSocket{};
462 }
463 {
464 std::scoped_lock lock{_syncMtx};
465 _syncReply.reset();
466 }
467 _syncCv.notify_all();
468 _connectCv.notify_all();
469 cancelPending(std::make_exception_ptr(::morph::backend::DisconnectedError{}));
470 }

472 void dispatchIncomingEnvelope(const std::string& payload) {
473 ::morph::wire::Envelope env;
474 try {
475 env = ::morph::wire::decode(payload);
476 } catch (const std::exception&) {
477 {
478 std::scoped_lock const lock{_syncMtx};
479 if (_syncInFlight) {
480 // Hand the raw text to the parked caller so it can report
481 // something better than "disconnected".
482 _syncReply = payload;
483 _syncCv.notify_all();
484 return;
485 }
486 }
487 // No sync waiter, and the callId is unreadable, so this reply cannot
488 // be matched to the execute it belongs to. Dropping it silently left
489 // that execute's Completion unsettled forever. Every message here is
490 // required to be one envelope, so an undecodable one means the
491 // peer's framing is no longer trustworthy: fail the pending calls
492 // rather than wait on a stream that may never produce a matching
493 // reply. Mirrors QtWebSocketBackend::onTextMessage.
494 cancelPending(std::make_exception_ptr(
495 std::runtime_error("protocol error: server sent a message that is not a valid envelope")));
496 return;
497 }
498 if (env.callId != 0U) {
499 PendingExecute pending;
500 {
501 std::scoped_lock lock{_pendingMtx};
502 auto iter = _pending.find(env.callId);
503 if (iter == _pending.end()) {
504 return; // late/cancelled reply — dropped silently
505 }
506 pending = std::move(iter->second);
507 _pending.erase(iter);
508 }
509 if (env.kind == "ok") {
510 try {
511 pending.state->setValue(pending.deserialize(env.body));
512 } catch (...) {
513 pending.state->setException(std::current_exception());
514 }
515 } else {
516 pending.state->setException(std::make_exception_ptr(std::runtime_error(env.message)));
517 }
518 return;
519 }
520 std::scoped_lock lock{_syncMtx};
521 if (_syncInFlight) {
522 _syncReply = payload;
523 _syncCv.notify_all();
524 }
525 }

527 // Returns false when the connection should stop reading (peer sent
528 // Close, or a protocol error was detected).
529 bool drainFrames(::morph::net::detail::WsFrameReader& reader) {
530 using ::morph::net::detail::WsOpcode;
531 for (;;) {
532 std::optional<::morph::net::detail::WsFrame> frame;
533 try {
534 frame = reader.tryExtractFrame();
535 } catch (const std::exception&) {
536 return false;
537 }
538 if (!frame) {
539 return true;
540 }
541 if (frame->opcode == WsOpcode::kClose) {
542 try {
543 sendFrame(WsOpcode::kClose, "");
544 } catch (const std::exception&) {
545 }
546 return false;
547 }
548 if (frame->opcode == WsOpcode::kPing) {
549 try {
550 sendFrame(WsOpcode::kPong, frame->payload);
551 } catch (const std::exception&) {
552 }
553 continue;
554 }
555 if (frame->opcode == WsOpcode::kPong) {
556 continue;
557 }
558 if (frame->opcode == WsOpcode::kText) {
559 dispatchIncomingEnvelope(frame->payload);
560 }
561 }
562 }

564 void readLoop(const std::string& leftover) {
565 ::morph::net::detail::WsFrameReader reader;
566 reader.feed(leftover);
567 char buf[4096];
568 for (;;) {
569 if (!drainFrames(reader)) {
570 return;
571 }
572 std::size_t got = 0;
573 try {
574 got = _socket.recvSome(buf, sizeof(buf));
575 } catch (const std::exception&) {
576 return;
577 }
578 if (got == 0) {
579 return;
580 }
581 reader.feed(std::string_view{buf, got});
582 }
583 }

585 void ioThreadMain() {
586 while (!_shuttingDown.load()) {
587 bool connectedOk = false;
588 try {
589 auto socket = ::morph::net::detail::TcpSocket::connect(_url.host, _url.port, _cfg.connectTimeout);
590 std::string leftover = ::morph::net::detail::performClientHandshake(socket, _url);
591 {
592 std::scoped_lock lock{_socketMtx};
593 _socket = std::move(socket);
594 }
595 connectedOk = true;
596 onConnected();
597 readLoop(leftover);
598 } catch (const std::exception&) {
599 // Falls through to the disconnect/reconnect handling below.
600 }
601 onDisconnected();
602 if (_shuttingDown.load()) {
603 return;
604 }
605 bool const wasEverConnected = _everConnected.load();
606 if (!connectedOk && !wasEverConnected) {
607 // Never reached the server even once — fail fast, no retry
608 // (mirrors QtWebSocketBackend's "no reconnect for
609 // never-connected sockets").
610 return;
611 }
612 if (!_cfg.reconnectEnabled) {
613 return;
614 }
615 std::unique_lock lock{_reconnectMtx};
616 _reconnectCv.wait_for(lock, _currentReconnectDelay, [this] { return _shuttingDown.load(); });
617 auto const nextDelayMs = static_cast<std::chrono::milliseconds::rep>(
618 static_cast<double>(_currentReconnectDelay.count()) * _cfg.backoffMultiplier);
619 _currentReconnectDelay = std::min(std::chrono::milliseconds{nextDelayMs}, _cfg.maxReconnectDelay);
620 }
621 }
