39/// @brief Reconnect tuning for `SocketBackend`.
40struct SocketBackendConfig {
41 /// @brief Whether to attempt automatic reconnect after an unsolicited disconnect.
42 bool reconnectEnabled = true;
43 /// @brief Delay before the first reconnect attempt after a disconnect.
44 std::chrono::milliseconds initialReconnectDelay{500};
45 /// @brief Upper bound on the exponential backoff between reconnect attempts.
46 std::chrono::milliseconds maxReconnectDelay{30000};
47 /// @brief Multiplier applied to the delay after each failed attempt.
48 double backoffMultiplier = 2.0;
49 /// @brief Maximum time for the whole connect: name resolution and every
50 /// resolved address.
51 std::chrono::milliseconds connectTimeout{5000};
52 /// @brief Bound on writing one chunk (up to 64 KiB) of an outgoing frame.
53 ///
54 /// A loop timer armed around each chunk's write closes the connection when
55 /// it runs out, so a peer that stops reading ends in a disconnect (and the
56 /// usual reconnect) rather than in a queue of unsent frames that grows
57 /// forever. Generous on purpose: it bounds a write making no progress, not
58 /// a slow one. Zero disables it.
59 std::chrono::milliseconds sendTimeout{30000};
60 /// @brief Bound on the whole handshake response read that follows a
61 /// successful TCP connect, from its first byte to the header's end.
62 ///
63 /// A peer that accepts the TCP connection and then never completes the
64 /// WebSocket upgrade — a TCP load balancer connecting lazily to its
65 /// backend would — is dropped after this long, and counts as a failed
66 /// connect. Zero disables it.
67 std::chrono::milliseconds handshakeTimeout{10000};
68};

70/// @brief `IBackend` implementation that communicates with a `RemoteServer`
71/// over a raw-socket RFC 6455 WebSocket connection — no Qt, no GUI
72/// event loop required.
73///
74/// Mirrors `morph::qt::QtWebSocketBackend`'s observable behavior
75/// (`deregisterModel` fire-and-forget, `execute` and `bindModel` asynchronous
76/// and callId-multiplexed, `DisconnectedError`/reconnect semantics). Its
77/// socket, its pending calls, its session, its reconnect handler and its
78/// reconnect state machine live on an `exec::IoLoop`, which it shares with
79/// every other socket, timer and probe the application builds on that loop.
80///
81/// Cross-thread surface: `execute`, `bindModel`, `promoteModel`,
82/// `deregisterModel`, `cancelPending`, `setSession` and `setReconnectHandler`
83/// post to the loop and return, so calls from one thread take effect in the
84/// order they were made; `waitForConnected` posts a waiter and blocks the
85/// caller, never the loop; the synchronous control verbs (`registerModel`,
86/// `registerModelWithContext`, `assignPrimary`, `listInstances`) issue the same
87/// posted request and wait for its reply, never on the loop's own thread.
88/// Every field is touched only on the loop, except the `connected` flag.
89///
90/// @par TLS
91/// Not supported. `serverUrl` must be `ws://`; `wss://` throws from the
92/// constructor. See `docs/spec/core/backend.md`'s `morph::net` section.
93class SocketBackend : public ::morph::backend::detail::IBackend {
94public:
95 /// @brief Alias for the reconnect configuration struct.
96 using Config = SocketBackendConfig;
97
98 /// @brief Parses @p serverUrl and starts connecting on @p loop.
99 /// @param loop The application's I/O loop. Borrowed: it must outlive
100 /// this backend.
101 /// @param serverUrl `ws://host:port[/path]` URL of the remote `RemoteServer`.
102 /// @param cfg Reconnect tuning. Default: enabled, 500ms initial / 30s cap, 2x backoff.
103 /// @throws std::runtime_error immediately if @p serverUrl is not a
104 /// well-formed `ws://` URL (see `parseWsUrl`).
105 SocketBackend(::morph::exec::IoLoop& loop MORPH_LIFETIMEBOUND, std::string_view serverUrl, Config cfg = {})
106 : _url{::morph::net::detail::parseWsUrl(serverUrl)},
107 _loop{&loop},
108 _core{std::make_shared<Core>(loop, _url, cfg)} {
109 start();
110 }
111
112 /// @brief Parses @p serverUrl and starts connecting on a loop of its own,
113 /// for a caller with no `IoLoop` to share.
114 /// @param serverUrl `ws://host:port[/path]` URL of the remote `RemoteServer`.
115 /// @param cfg Reconnect tuning. Default: enabled, 500ms initial / 30s cap, 2x backoff.
116 /// @throws std::runtime_error immediately, before any loop is started, if
117 /// @p serverUrl is not a well-formed `ws://` URL (see `parseWsUrl`).
118 explicit SocketBackend(std::string_view serverUrl, Config cfg = {})
119 : _url{::morph::net::detail::parseWsUrl(serverUrl)},
120 _ownedLoop{std::make_unique<::morph::exec::IoLoop>()},
121 _loop{_ownedLoop.get()},
122 _core{std::make_shared<Core>(*_ownedLoop, _url, cfg)} {
123 start();
124 }

131 /// @brief Closes the connection and rejects every pending call with
132 /// `DisconnectedError`.
133 ///
134 /// The close runs on the loop: inline when this runs on the loop's own
135 /// thread (from a callback), posted and waited for otherwise. It never
136 /// waits for a connect in progress: a dial that completes afterwards finds
137 /// the backend closed and drops its socket.
138 // NOLINTNEXTLINE(bugprone-exception-escape): teardown has no recovery if handing the close to the loop fails, so terminating is the outcome.
139 ~SocketBackend() override {
140 _loop->runAndWait([core = _core] { core->close(); });
141 }

143 /// @brief Blocks the calling thread until connected or @p timeout elapses.
144 ///
145 /// Posts a waiter to the loop, which releases it when a connection
146 /// completes. On the loop's own thread it cannot wait for the loop, so it
147 /// answers at once.
148 ///
149 /// @warning The backend must outlive this call. See
150 /// `docs/spec/core/backend.md`'s "Lifetime & ownership" section.
151 /// @param timeout Maximum time to wait.
152 /// @return `true` if connected before the timeout, `false` otherwise.
153 bool waitForConnected(std::chrono::milliseconds timeout = std::chrono::milliseconds{5000}) {
154 if (_core->connected.load() || _loop->runningHere()) {
155 return _core->connected.load();
156 }
157 auto waiter = std::make_shared<std::promise<void>>();
158 std::future<void> const released = waiter->get_future();
159 _loop->post([core = _core, waiter] { core->addConnectWaiter(waiter); });
160 static_cast<void>(released.wait_for(timeout));
161 return _core->connected.load();
162 }

166 /// The same request `bindModel` posts, waited for on the calling thread —

172 /// @return `ModelId` assigned by the server.

175 ::morph::exec::detail::ModelId registerModel(
176 const std::string& typeId,

181 /// @brief Registers a private instance carrying @p contextKey and waits for the reply.
182 ///
183 /// The server constructs the holder itself, so `contextKey` is the only

185 /// `RemoteServer::attachLogIfConfigured` consults its `LogProvider` only

187 /// @param typeId String type-id of the model to register.
188 /// @param factory Ignored — the server constructs via its own registry.
189 /// @param contextKey Stable identity of the new instance; empty if none.
190 /// @return `ModelId` assigned by the server.
191 /// @throws std::runtime_error if the server replies with an error, the
192 /// socket is not connected, or this is the loop's own thread.
193 ::morph::exec::detail::ModelId registerModelWithContext(
194 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
195 std::string_view contextKey) override {
196 return awaitReply<::morph::exec::detail::ModelId>("register", [&](::morph::exec::IExecutor& waiter) {
197 return bindModel(::morph::backend::detail::BindRequest{.typeId = typeId,
198 .factory = std::move(factory),
199 .contextKey = std::string{contextKey},
200 .primary = {},
201 .current = {}},
202 waiter);
203 });
204 }

231 /// @brief Acquires a model instance without blocking the calling thread.
232 ///
233 /// Sends the control envelope @p request's shape names (see
234 /// `backend::detail::BindRequest`'s table) carrying a non-zero `callId`
235 /// drawn from the same counter `execute` uses, and settles the returned
236 /// `Completion` on the I/O loop when the matching reply arrives. No thread
237 /// is parked anywhere.
238 ///
239 /// An empty `primary` with a live `current` gives that instance up first,
240 /// and an empty `primary` binds a private instance.
241 ///
242 /// @param request Owning bind request; moved from.
243 /// @param cbExec Executor the continuation is delivered on. Borrowed: it
244 /// must outlive the returned `Completion`.
245 /// @return A `Completion` resolved with the bound `ModelId`; rejected with
246 /// `backend::DisconnectedError` if the socket is down or drops
247 /// before the reply, or with a `std::runtime_error` carrying the
248 /// server's own error message.
249 ::morph::async::Completion<::morph::exec::detail::ModelId> bindModel(::morph::backend::detail::BindRequest request,
250 ::morph::exec::IExecutor& cbExec) override {
251 if (request.primary.empty() && request.current.v != 0U) {
252 // An empty primary with a live instance: the instance being given up
253 // is released before the private bind that replaces it.
254 deregisterModel(request.current);
255 }
256 if (request.primary.empty()) {
257 return sendControlAsync(::morph::wire::makeRegister(request.typeId, request.contextKey), "register",
258 std::nullopt, cbExec);
259 }
260 if (request.current.v != 0U) {
261 return sendControlAsync(
262 ::morph::wire::makeAttach(request.typeId, request.primary, request.current.v, request.contextKey),
263 "attach", std::nullopt, cbExec);
264 }
265 return sendControlAsync(::morph::wire::makeRegisterShared(request.typeId, request.primary, request.contextKey),
266 "register", std::nullopt, cbExec);
267 }

269 /// @brief Files an already-live server-side instance under a key, without
270 /// blocking the calling thread.
271 ///
272 /// The promote counterpart of `bindModel`, on the same callId-multiplexed
273 /// path. Resolves with `request.mid` echoed back exactly as
274 /// `IBackend::promoteModel` documents — including for the two guards
275 /// `assignPrimary` applies locally (empty `primary`, zero `mid`), which
276 /// resolve without sending anything.
277 ///
278 /// @param request Owning promote request; moved from.
279 /// @param cbExec Executor the continuation is delivered on. Borrowed: it
280 /// must outlive the returned `Completion`.
281 /// @return A `Completion` resolved with `request.mid`, or rejected as
282 /// `bindModel` documents.
283 ::morph::async::Completion<::morph::exec::detail::ModelId> promoteModel(
284 ::morph::backend::detail::PromoteRequest request, ::morph::exec::IExecutor& cbExec) override {
285 if (request.primary.empty() || request.mid.v == 0U) {
286 auto state = std::make_shared<::morph::async::detail::CompletionState<::morph::exec::detail::ModelId>>();
287 ::morph::async::Completion<::morph::exec::detail::ModelId> comp{state, &cbExec};
288 state->setValue(request.mid);
289 return comp;
290 }
291 return sendControlAsync(::morph::wire::makeAssign(request.typeId, request.primary, request.mid.v), "assign",
292 request.mid, cbExec);
293 }

295 /// @brief Asks the server for the live shared primary keys of @p typeId
296 /// and waits for the reply, never on the loop's own thread.
297 /// @param typeId String type-id to enumerate.
298 /// @return Canonical key strings of the live shared instances.
299 /// @throws std::runtime_error if the server replies with an error, the
300 /// socket is down, or this is the loop's own thread.
301 std::vector<std::string> listInstances(const std::string& typeId) override {
302 auto body = awaitReply<std::string>("instances", [&](::morph::exec::IExecutor& waiter) {
303 auto state = std::make_shared<::morph::async::detail::CompletionState<std::string>>();
304 ::morph::async::Completion<std::string> comp{state, &waiter};
305 _loop->post([core = _core, env = ::morph::wire::makeInstances(typeId),
306 pending = PendingControl{
307 .state = nullptr, .what = "instances", .echo = std::nullopt, .body = state}]() mutable {
308 core->fileControl(std::move(env), std::move(pending));
309 });
310 return comp;
311 });
312 std::vector<std::string> keys;
313 if (auto errCode = glz::read_json(keys, body)) {
314 throw std::runtime_error("instances decode failed: " + glz::format_error(errCode, body));
315 }
316 return keys;
317 }

337 /// @brief Sends an `execute` message and returns a `Completion` resolved on reply.
338 ///
339 /// The action is serialised here, on the calling thread; the loop then
340 /// assigns the call id, files the pending record and writes the frame.
341 /// @param mid Target model id on the server.
342 /// @param call Bundled action; `serializeAction` and `deserializeResult` are used.
343 /// @param cbExec Executor for delivering the completion callbacks.
344 /// @return Completion resolved asynchronously when the server's reply arrives,
345 /// or with `DisconnectedError` if the backend is not connected.
346 ::morph::async::Completion<std::shared_ptr<void>> execute(::morph::exec::detail::ModelId mid,
347 ::morph::backend::detail::ActionCall call,
348 ::morph::exec::IExecutor* cbExec) override {
349 auto compState = std::make_shared<::morph::async::detail::CompletionState<std::shared_ptr<void>>>();
350 ::morph::async::Completion<std::shared_ptr<void>> comp{compState, cbExec};
351 if (!_core->connected.load()) {
352 // Answered here rather than on the loop, which would answer the
353 // same: the loop re-checks, so a connection lost after this read
354 // is caught there.
355 compState->setException(std::make_exception_ptr(::morph::backend::DisconnectedError{}));
356 return comp;
357 }
358
359 ::morph::wire::Envelope env;
360 env.kind = "execute";
361 env.modelId = mid.v;
362 env.modelType = call.modelTypeId;
363 env.actionType = call.actionTypeId;
364 env.body = call.serializeBody();
365 env.session = std::move(call.session);
366
367 _loop->post([core = _core, env,
368 pending = PendingExecute{.state = compState, .deserialize = call.deserializeResult}]() mutable {
369 core->fileExecute(std::move(env), std::move(pending));
370 });
371 return comp;
372 }

377 /// @brief Resolves every pending call's `Completion` with @p exc.
378 ///
379 /// Posted to the loop, so it covers every call issued before it — the loop
380 /// runs posts in order. Covers both in-flight tables: the `execute` calls
381 /// and the `bindModel`/`promoteModel` control calls. A bind left out of

386 _loop->post([core = _core, exc] {
387 ::morph::exec::detail::noteOwner("SocketBackend::cancelPending", core->loop.loop(),
388 core->loop.runningHere());
389 core->cancelAll(exc);
390 });
391 }

427 /// @brief Waits on the calling thread for the reply to the request @p ask
428 /// sends, settled on the loop.
429 ///
430 /// The answer's owner is an executor local to this call, which the
431 /// calling thread pumps while it waits: the attach and the delivery are
432 /// both tasks of it, so the waiting thread is the one place they run.
433 /// @tparam T The answer's type.
434 /// @tparam Ask Callable taking the waiter (`morph::exec::IExecutor&`) and
435 /// returning a `Completion<T>` delivered on it.
436 /// @param what Verb name for the error message.
437 /// @param ask Sends the request, naming the waiter as the answer's owner.
438 /// @return The settled value.
439 /// @throws std::runtime_error on the loop's own thread, when the loop has
440 /// stopped, on a disconnect, or with the server's refusal.
441 template <typename T, typename Ask>
442 T awaitReply(std::string_view what, Ask ask) {
443 if (_loop->runningHere()) {
444 throw std::runtime_error(std::string{what} +
445 " failed: a synchronous call cannot wait on the I/O loop's own thread");
446 }
447 ::morph::exec::MainThreadExecutor waiter;
448 auto completion = ask(static_cast<::morph::exec::IExecutor&>(waiter));
449 std::optional<T> value;
450 std::exception_ptr error;
451 bool done = false;
452 waiter.post([&completion, &value, &error, &done] {
453 completion
454 .then([&value, &done](const T& settled) {
455 value.emplace(settled);
456 done = true;
457 })
458 .onError([&error, &done](const std::exception_ptr& failure) {
459 error = failure;
460 done = true;
461 });
462 });
463 // The request was posted to the loop before this round trip, so a
464 // loop that runs it has filed the request too, and will answer it or
465 // reject it when it closes. A stopped loop drops both unrun.
466 bool alive = false;
467 _loop->runAndWait([&alive] { alive = true; });
468 if (!alive) {
469 throw std::runtime_error(std::string{what} + " failed: the I/O loop has stopped");
470 }
471 // runFor pumps until its deadline even once `done` is set, so it is
472 // only the wait between replies, kept short.
473 while (!done) {
474 if (!waiter.runOnce()) {
475 waiter.runFor(std::chrono::milliseconds{1});
476 }
477 }
478 if (error) {
479 try {
480 std::rethrow_exception(error);
481 } catch (const ::morph::backend::DisconnectedError&) {
482 throw std::runtime_error(std::string{what} + " failed: disconnected");
483 }
484 }
485 return std::move(*value);
486 }

493 /// @brief One in-flight control call issued through `bindModel`/`promoteModel`.
494 ///
495 /// Kept in its own table rather than beside the executes: an execute reply
496 /// settles a `Completion<shared_ptr<void>>` through a deserializer, a
497 /// control reply settles a `Completion<ModelId>` and has none. The two
498 /// tables share one `callId` counter (the execute table's), so an id is
499 /// never ambiguous between them.
500 struct PendingControl {
501 /// @brief State of the `Completion<ModelId>` this call settles, or
502 /// null for a call that settles `body` instead.
503 std::shared_ptr<::morph::async::detail::CompletionState<::morph::exec::detail::ModelId>> state;
504 /// @brief Verb name prefixing the server's error message:
505 /// `"<what> failed: ..."`.
506 std::string what;
507 /// @brief Id to resolve with, for `promoteModel`, which echoes its
508 /// request's `mid`. `nullopt` means "resolve with the reply's".
509 std::optional<::morph::exec::detail::ModelId> echo;
510 /// @brief State of a call answered with the reply's body
511 /// (`instances`), or null.
512 std::shared_ptr<::morph::async::detail::CompletionState<std::string>> body;
513
514 /// @brief Rejects whichever state this call settles.
515 /// @param failure The failure.
516 void reject(const std::exception_ptr& failure) const {
517 if (state) {
518 state->setException(failure);
519 }
520 if (body) {
521 body->setException(failure);
522 }
523 }
524 };

526 /// @brief Posts one control envelope to the loop's callId-multiplexed
527 /// path and returns the `Completion` its reply will settle.
528 /// @param env Envelope to send; its `callId` and `session` are filled in
529 /// on the loop.
530 /// @param what Verb name for the error message.
531 /// @param echo Id to resolve with, or `nullopt` to use the reply's `modelId`.
532 /// @param cbExec Executor the continuation is delivered on.
533 /// @return The `Completion` the reply — or a disconnect — settles.
534 ::morph::async::Completion<::morph::exec::detail::ModelId> sendControlAsync(
535 ::morph::wire::Envelope env, std::string_view what, std::optional<::morph::exec::detail::ModelId> echo,
536 ::morph::exec::IExecutor& cbExec) {
537 auto state = std::make_shared<::morph::async::detail::CompletionState<::morph::exec::detail::ModelId>>();
538 ::morph::async::Completion<::morph::exec::detail::ModelId> comp{state, &cbExec};
539 _loop->post([core = _core, env,
540 pending = PendingControl{
541 .state = state, .what = std::string{what}, .echo = echo, .body = nullptr}]() mutable {
542 core->fileControl(std::move(env), std::move(pending));
543 });
544 return comp;
545 }

547 /// @brief Settles one control call from its matched reply envelope.
548 /// @param pending Entry taken out of the control table.
549 /// @param reply Decoded reply carrying the same `callId`.
550 static void settleControl(const PendingControl& pending, const ::morph::wire::Envelope& reply) {
551 if (reply.kind != "ok") {

673 /// Settles every pending call, of both kinds, with @p exc. The tables
674 /// are emptied first, so a settle that re-enters the backend files
675 /// into empty tables.
676 void cancelAll(const std::exception_ptr& exc) {
677 auto drained = executes.drain();
678 auto drainedControl = controls.drain();
679 for (auto& entry : drained) {
680 if (entry.second.state) {
681 entry.second.state->setException(exc);
682 }
683 }
684 for (auto& entry : drainedControl) {
685 entry.second.reject(exc);
686 }

689 void onConnected() {
690 bool const isReconnect = everConnected;
691 everConnected = true;
692 connected.store(true);
693 reconnectDelay = cfg.initialReconnectDelay;
694 for (auto const& waiter : std::exchange(connectWaiters, {})) {
695 waiter->set_value();
696 }
697 if (isReconnect && reconnectHandler && reconnectExec != nullptr) {
698 // Posted, never run here: the handler re-registers through
699 // `bindModel`, whose replies this loop delivers.
700 reconnectExec->post(reconnectHandler);
701 }
702 }

704 void onDisconnected() {
705 connected.store(false);
706 if (conn) {
707 ::morph::net::detail::closeAfterFlush(conn);
708 conn.reset();
709 }
710 cancelAll(std::make_exception_ptr(::morph::backend::DisconnectedError{}));
711 }

713 /// Arms the backoff timer for the next attempt, and grows the delay.
714 void scheduleReconnect() {
715 note("SocketBackend::reconnect");
716 backoffTimer = loop.loop().addTimer(loop.loop().clock().now() + reconnectDelay, &Core::onBackoff, this);
717 auto const nextDelayMs = static_cast<std::chrono::milliseconds::rep>(
718 static_cast<double>(reconnectDelay.count()) * cfg.backoffMultiplier);
719 reconnectDelay = std::min(std::chrono::milliseconds{nextDelayMs}, cfg.maxReconnectDelay);
720 }

722 /// The backoff timer's callback. `close()` retires the timer, so the
723 /// core it points at is alive whenever it runs.
724 static void onBackoff(void* corePtr) {
725 auto& self = *static_cast<Core*>(corePtr);
726 self.backoffTimer = {};
727 self.startAttempt();

730 /// Ends everything: the connection, the backoff, the waiters, every
731 /// pending call. Idempotent. After it, no flow touches the backend.
732 void close() {
733 note("SocketBackend::close");
734 if (closed) {
735 return;
736 }
737 closed = true;
738 connected.store(false);
739 static_cast<void>(loop.loop().cancelTimer(backoffTimer));
740 if (conn) {
741 ::morph::net::detail::closeConnection(*conn);
742 conn.reset();
743 }
744 connectWaiters.clear();
745 reconnectHandler = nullptr;
746 cancelAll(std::make_exception_ptr(::morph::backend::DisconnectedError{}));
747 }

749 void dispatchIncomingEnvelope(const std::string& payload) {
750 MORPH_ZONE("SocketBackend::dispatchIncomingEnvelope");
751 ::morph::wire::Envelope env;
752 try {
753 env = ::morph::wire::decode(payload);
754 } catch (const std::exception&) {
755 // The callId is unreadable, so this reply cannot be matched to
756 // the call it belongs to. Every message
757 // here is required to be one envelope, so an undecodable one
758 // means the peer's framing is no longer trustworthy: fail the
759 // pending calls rather than wait on a stream that may never
760 // produce a matching reply. Mirrors
761 // QtWebSocketBackend::onTextMessage.
762 cancelAll(std::make_exception_ptr(
763 std::runtime_error("protocol error: server sent a message that is not a valid envelope")));
764 return;
765 }
766 note("SocketBackend::reply");
767 auto pending = executes.take(env.callId);
768 if (!pending) {
769 // Not an execute: it may be a control reply for a `bindModel`/
770 // `promoteModel` in flight, which shares this id space.
771 if (auto control = controls.take(env.callId)) {
772 settleControl(*control, env);
773 }
774 return; // otherwise a late/cancelled reply, dropped silently
775 }
776 // Triage shared with SimulatedRemoteBackend and QtWebSocketBackend
777 // (core/detail/reply_router.hpp), so the three cannot drift. The
778 // Timeout arm is the server's own `LimitPolicy::executeTimeout`
779 // reply, not a generic application error -- surfaced as the same
780 // TimeoutError type the other two backends give callers for this
781 // case.
782 switch (::morph::backend::detail::classifyExecuteReply(env)) {
783 case ::morph::backend::detail::ExecuteReplyKind::Value:
784 try {
785 if (pending->deserialize == nullptr) {
786 throw std::runtime_error{"ActionCall::deserializeResult is null"};
787 }
788 pending->state->setValue(pending->deserialize(env.body));
789 } catch (...) {
790 pending->state->setException(std::current_exception());
791 }
792 break;
793 case ::morph::backend::detail::ExecuteReplyKind::Timeout:
794 pending->state->setException(std::make_exception_ptr(::morph::backend::TimeoutError{}));
795 break;
796 case ::morph::backend::detail::ExecuteReplyKind::Error:
797 default:
798 // `default:` only because the project builds with
799 // -Wswitch-default; ExecuteReplyKind is a closed enum and
800 // all three enumerators are handled explicitly.
801 pending->state->setException(std::make_exception_ptr(std::runtime_error(env.message)));
802 break;
803 }
804 }

806 /// Handles every complete frame @p reader holds.
807 /// @return `false` when the connection should stop reading (peer sent
808 /// Close, a protocol error, or this backend closed meanwhile).
809 bool drainFrames(const std::shared_ptr<::morph::net::detail::LoopConnection>& connection,
810 ::morph::net::detail::WsFrameReader& reader) {
811 MORPH_ZONE("SocketBackend::drainFrames");
812 using ::morph::net::detail::WsOpcode;
813 for (;;) {
814 std::optional<::morph::net::detail::WsFrame> frame;
815 try {
816 frame = reader.tryExtractFrame();
817 } catch (const std::exception&) {
818 return false;
819 }
820 if (!frame) {
821 return true;
822 }
823 if (frame->opcode == WsOpcode::kClose) {
824 static_cast<void>(::morph::net::detail::enqueueFrame(
825 connection, ::morph::net::detail::encodeWsFrame(WsOpcode::kClose, frame->payload, true)));
826 return false;
827 }
828 if (frame->opcode == WsOpcode::kPing) {
829 static_cast<void>(::morph::net::detail::enqueueFrame(
830 connection, ::morph::net::detail::encodeWsFrame(WsOpcode::kPong, frame->payload, true)));
831 continue;
832 }
833 if (frame->opcode == WsOpcode::kText) {
834 dispatchIncomingEnvelope(frame->payload);
835 // A settle may have run user code that destroyed the
836 // backend, closing this core.
837 if (closed) {
838 return false;
839 }
840 }
841 }
842 }

844 /// One connection attempt, start to end: dial, handshake, read until
845 /// the connection ends, then decide whether to try again.
846 static ::core::async::Task<void> attemptFlow(std::shared_ptr<Core> self) {
847 bool reachedServer = false;
848 {
849 ::core::net::DialOptions options;
850 options.connectTimeout = self->cfg.connectTimeout;
851 auto dialed = co_await ::core::net::connect(&self->loop.loop(), self->url.host, self->url.port,
852 &::core::net::defaultAsyncResolver(), options);
853 if (self->closed) {
854 co_return;
855 }
856 if (dialed) {
857 auto connection = std::make_shared<::morph::net::detail::LoopConnection>(
858 self->loop.loop(), std::move(*dialed), self->cfg.sendTimeout);
859 self->conn = connection;
860 std::string const key = ::morph::net::detail::generateClientKey();
861 static_cast<void>(::morph::net::detail::enqueueFrame(
862 connection, ::morph::net::detail::buildClientHandshakeRequest(self->url, key)));
863 auto header =
864 co_await ::morph::net::detail::readHeaderBlockAsync(connection, self->cfg.handshakeTimeout);
865 if (self->closed) {
866 co_return;
867 }
868 bool verified = false;
869 if (header) {
870 try {
871 ::morph::net::detail::verifyServerHandshakeResponse(header->header, key);
872 verified = true;
873 } catch (const std::exception&) { // NOLINT(bugprone-empty-catch)
874 // A failed connect, like any other: see below.
875 }
876 }
877 if (verified) {
878 reachedServer = true;
879 self->onConnected();
880 // Client role: RFC 6455 §5.1 forbids a server from
881 // masking the frames it sends, so this reader must
882 // reject a masked one.
883 ::morph::net::detail::WsFrameReader reader{/*expectMasked=*/false};
884 reader.feed(header->leftover);
885 std::array<std::byte, 4096> buf{};
886 while (self->drainFrames(connection, reader) && !connection->closed) {
887 auto const got = co_await connection->socket->read(buf);
888 if (self->closed || !got || *got == 0) {
889 break;
890 }
891 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast): the socket yields bytes, the reader takes text; same object representation.
892 reader.feed(std::string_view{reinterpret_cast<char const*>(buf.data()), *got});
893 }
894 }
895 if (self->closed) {
896 co_return;
897 }
898 }
899 }
900 self->onDisconnected();
901 if (!reachedServer && !self->everConnected) {
902 // Never reached the server even once — fail fast, no retry
903 // (mirrors QtWebSocketBackend's "no reconnect for
904 // never-connected sockets").
905 co_return;
906 }
907 if (self->cfg.reconnectEnabled) {
908 self->scheduleReconnect();
909 }
910 }
