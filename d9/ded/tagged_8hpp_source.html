5/// @file util/tagged.hpp
6/// @brief `Tagged<T, "Name">` — an opaque, type-safe newtype for protocol
7/// scalars that serializes transparently as its underlying `T`.
8///
9/// A protocol scalar (a pagination cursor, an event id, a job id, a bearer
10/// token) is a `T` (usually `std::string` or an integer) that must not be
11/// interchangeable with a different scalar of the same underlying type —
12/// `Tagged<std::string, "UserId">` and `Tagged<std::string, "AccountId">`
13/// carry the same wire representation but are distinct C++ types, so passing
14/// one where the other is expected is a compile error rather than a silent
15/// mix-up caught (or not) at runtime.
16///
17/// `Tag` is a `morph::detail::FixedString` non-type template parameter (the
18/// same structural string type `morph::forms::Choice` and
19/// `morph::units::NamedQuantity` use), so the identity lives in the type
20/// itself — two `Tagged<T, "X">` written identically in different
21/// translation units are one and the same type, and no companion `enum` or
22/// registry is needed to keep tags distinct.
23///
24/// **Wire.** On the morph JSON wire a `Tagged<T, Tag>` is exactly its `T`
25/// payload — `glz::meta` maps the instance to `value` alone, so it reads and
26/// writes identically to a bare `T`, and `to_json_schema` delegates to `T`'s
27/// own schema (the tag never appears on the wire or in the generated
28/// schema).
29///
30/// **Forms palette.** `Tagged` is a *required* scalar, not an
31/// optionally-empty field like `Quantity`/`Choice`/`Timestamp`: it always
32/// holds a `T` (default-constructed, never `std::nullopt`). Its `hasValue()`
33/// therefore always returns `true`, satisfying
34/// `morph::forms::EmptyCapableField` so it composes with the palette (rule
35/// vocabulary, `allRequiredEngaged`, etc.) the same way any other required
36/// field does — nothing in the palette special-cases it away like a plain,
37/// non-empty-capable scalar member would be.

52/// @brief An opaque, type-safe newtype wrapping a protocol scalar `T`,
53/// distinguished at compile time by the string tag `Tag`.
54///
55/// @tparam T The underlying wire scalar (e.g. `std::string`, `std::int64_t`).
56/// @tparam Tag A `morph::detail::FixedString` NTTP naming this newtype (e.g.
57/// `"UserId"`); part of the C++ type, never part of the wire.
58template <typename T, morph::detail::FixedString Tag>
59struct Tagged {
60 /// @brief The underlying scalar value.
61 T value{};
62
63 /// @brief Constructs the default-valued state (`T{}`).
64 constexpr Tagged() noexcept(std::is_nothrow_default_constructible_v<T>) = default;
65
66 /// @brief Wraps @p wrapped.
67 /// @param wrapped The scalar value to tag.
68 constexpr explicit Tagged(T wrapped) noexcept(std::is_nothrow_move_constructible_v<T>)
69 : value{std::move(wrapped)} {}
70
71 /// @brief The compile-time tag naming this newtype.
72 /// @return The tag text, e.g. `"UserId"`.
73 [[nodiscard]] static constexpr std::string_view tag() noexcept { return Tag.view(); }
74
75 /// @brief Always `true`: `Tagged` is a required scalar, never empty.
76 /// @return `true`.
77 [[nodiscard]] constexpr bool hasValue() const noexcept { return true; }
78
79 /// @brief Read access to the wrapped value.
80 /// @return The underlying scalar.
81 [[nodiscard]] constexpr const T& get() const noexcept { return value; }
82
83 /// @brief Unchecked access to the wrapped value, mirroring `Quantity`'s
84 /// and `Choice`'s `operator*` for consistency across the
85 /// empty-capable-field family.
86 /// @return The underlying scalar.
87 [[nodiscard]] constexpr const T& operator*() const noexcept { return value; }
88
89 /// @brief Equality on the wrapped value.
90 /// @param other Tagged value to compare against (same `T`/`Tag`).
91 /// @return `true` when both wrap equal values.
92 [[nodiscard]] constexpr bool operator==(const Tagged& other) const noexcept(noexcept(value == other.value))
93 requires std::equality_comparable<T>
94 = default;
95
96 /// @brief Three-way comparison on the wrapped value, when `T` supports it.
97 /// @param other Tagged value to compare against (same `T`/`Tag`).
98 /// @return The ordering of the wrapped values.
99 [[nodiscard]] constexpr auto operator<=>(const Tagged& other) const noexcept(noexcept(value <=> other.value))
100 requires std::three_way_comparable<T>
101 = default;
102};

154/// @brief Stable shape tag for `Tagged<T, Tag>`, carrying both the tag text
155/// and the wrapped type's own shape.
156///
157/// `Tagged` is a transparent wrapper — its JSON simply *is* `T`'s — so hiding
158/// `T` behind the wrapper's name would lose real information: retyping
159/// `Tagged<std::string, "acct">` to `Tagged<std::int64_t, "acct">` genuinely
160/// changes the recorded JSON. `Inner` keeps that visible, and the tag text
161/// separates two wrappers that are structurally identical but semantically
162/// different (`"acct"` versus `"user"`), which is the swap nothing else in the
163/// system can see: the tag never travels on the wire.
164///
165/// `Tag.view()` is a `FixedString` NTTP spelled at the use site in these
166/// sources, so it is stable across compilers in a way `glz::name_v` is not.
167/// See `morph/core/payload_shape_tag.hpp`.
168/// @tparam T The wrapped scalar type.
169/// @tparam Tag The compile-time tag.
170template <typename T, morph::detail::FixedString Tag>
171struct morph::model::PayloadShapeTag<morph::util::Tagged<T, Tag>> {
172 /// @brief The wrapped type, rendered inside this tag.
173 using Inner = T;
174
175 /// @brief This type's stable shape name, e.g. `"tagged.acct"`.
176 /// @return The name; the referenced storage lives for the whole process.
177 static std::string_view name() {
178 static const std::string kName = "tagged." + std::string{Tag.view()};
179 return kName;
180 }
181};
