39/// @brief What names an action in a document.
40enum class ActionUseKind : std::uint8_t {

42 Mutation, ///< A mutation declaration.

44 Collection, ///< A `collection` node.

48/// @brief The name of an action use's kind.

56/// @brief One place a document names an action.
57struct ActionUse {
58 /// @brief The action id.
59 std::string actionId;
60 /// @brief The model type the alias it goes through binds; empty when neither the document nor the catalog says.
61 std::string modelType;
62 /// @brief What names it.
63 ActionUseKind kind = ActionUseKind::Query;
64 /// @brief The JSON Pointer of the declaration or node.
65 std::string pointer;
66};

70/// @brief The scopes around a node, innermost last: the aliases in its reach.
71using ScopeChain = std::vector<ScopeDecl const*>;
72
73/// @brief The model type an alias binds, from the innermost scope that declares it.
74/// @param scopes The scopes around the use.
75/// @param alias The alias.
76/// @return The type; empty when no scope declares the alias.
77[[nodiscard]] inline std::string typeOf(ScopeChain const& scopes, std::string_view alias) {
78 for (ScopeDecl const* const scope : scopes | std::views::reverse) {
79 auto const found = std::ranges::find(scope->models, alias, &ModelDecl::name);
80 if (found != scope->models.end()) {
81 return found->type;
82 }
83 }
84 return {};
85}

87/// @brief Records one use, its model type the alias's, else the catalog's for the action.
88/// @param uses Where it is recorded.
89/// @param scopes The scopes around the use.
90/// @param catalog The facts a borrowed alias's type is read from.
91/// @param alias The alias the use goes through.
92/// @param action The action id.
93/// @param kind What names it.
94/// @param pointer Its JSON Pointer.
95inline void addUse(std::vector<ActionUse>& uses, ScopeChain const& scopes, Catalog const& catalog,
96 std::string_view alias, std::string action, ActionUseKind kind, std::string pointer) {
97 std::string type = typeOf(scopes, alias);
98 // A borrowed alias has no type of its own in the document: the catalog says which model has the action.
99 if (type.empty() && catalog.action) {
100 if (std::optional<ActionInfo> const info = catalog.action(action)) {
101 type = info->modelType;
102 }
103 }
104 uses.push_back(ActionUse{
105 .actionId = std::move(action), .modelType = std::move(type), .kind = kind, .pointer = std::move(pointer)});
106}
107
108/// @brief Records the queries and mutations of the innermost scope.
109/// @param uses Where they are recorded.
110/// @param scopes The scopes, the one read last.
111/// @param catalog The facts a borrowed alias's type is read from.
112inline void scopeUses(std::vector<ActionUse>& uses, ScopeChain const& scopes, Catalog const& catalog) {
113 ScopeDecl const& scope = *scopes.back();
114 for (QueryDecl const& query : scope.queries) {
115 addUse(uses, scopes, catalog, query.model, query.action, ActionUseKind::Query, query.path);
116 }
117 for (MutationDecl const& mutation : scope.mutations) {
118 addUse(uses, scopes, catalog, mutation.model, mutation.action, ActionUseKind::Mutation, mutation.path);
119 }
120}
121
122/// @brief Records the uses of a node and everything under it, with the scope it creates.
123/// @param uses Where they are recorded.
124/// @param scopes The scopes around the node; restored on return.
125/// @param catalog The facts a borrowed alias's type is read from.
126/// @param node The node.
127// NOLINTNEXTLINE(misc-no-recursion): a node's children are nodes.
128inline void nodeUses(std::vector<ActionUse>& uses, ScopeChain& scopes, Catalog const& catalog, DocNode const& node) {
129 if (node.scope) {
130 scopes.push_back(&*node.scope);
131 scopeUses(uses, scopes, catalog);
132 }
133 static constexpr std::array<std::pair<std::string_view, ActionUseKind>, 3> kNamed{
134 {{"form", ActionUseKind::Form}, {"collection", ActionUseKind::Collection}, {"wizard", ActionUseKind::Wizard}}};
135 for (auto const& [kind, use] : kNamed) {
136 if (node.kind == kind) {
137 addUse(uses, scopes, catalog, node.name("model"), std::string{node.name("action")}, use, node.path);
138 }
139 }
140 for (auto const& [key, nodes] : node.children) {
141 static_cast<void>(key);
142 for (DocNodePtr const& child : nodes) {
143 nodeUses(uses, scopes, catalog, *child);
144 }
145 }
146 for (auto const& [key, parts] : node.parts) {
147 static_cast<void>(key);
148 for (DocNodePtr const& part : parts) {
149 nodeUses(uses, scopes, catalog, *part);
150 }
151 }
152 if (node.scope) {
153 scopes.pop_back();
154 }
155}

159/// @brief Every place a loaded document names an action: its queries and mutations at any scope, and its `form`,
160/// `collection` and `wizard` nodes, in document order.
161///
162/// A model type is the alias's, from the nearest enclosing scope that declares it. An alias that borrows another has
163/// no type in the document, and takes the catalog's `ActionInfo::modelType`. A `run` command names a mutation, not an
164/// action, so it is no use of its own.
165/// @param document The document: a screen, or the app document.
166/// @param catalog The facts a borrowed alias's model type is read from; empty leaves such a type empty.
167/// @return The uses.
168[[nodiscard]] inline std::vector<ActionUse> actionUses(Document const& document, Catalog const& catalog = {}) {
169 std::vector<ActionUse> uses;
170 detail::explorer::ScopeChain scopes{&document.scope};
171 detail::explorer::scopeUses(uses, scopes, catalog);
172 if (document.view) {
173 detail::explorer::nodeUses(uses, scopes, catalog, *document.view);
174 }
175 return uses;
176}

178/// @brief One field of an action, as the explorer lists it.
179struct ActionField {
180 /// @brief The wire name.
181 std::string name;
182 /// @brief The type: the field kind's name, or for an enum "one of: …", a quantity "quantity in <unit>" with its

194/// @brief Texts joined by a separator.
195/// @param parts The texts.
196/// @param separator What goes between two.
197/// @return The joined text.
198[[nodiscard]] inline std::string joined(std::vector<std::string> const& parts, std::string_view separator) {
199 std::string out;
200 for (std::string const& part : parts) {
201 out += out.empty() ? part : std::string{separator} + part;
202 }
203 return out;
204}

206/// @brief A field's type as the explorer lists it.
207/// @param field The field, as the forms engine reads it.
208/// @return The type.
209[[nodiscard]] inline std::string typeText(forms::FieldSpec const& field) {
210 using forms::FieldKind;
211 if (field.kind == FieldKind::Enum) {
212 std::vector<std::string> labels;
213 labels.reserve(field.enumOptions.size());
214 for (forms::ChoiceOption const& option : field.enumOptions) {
215 labels.push_back(option.label);
216 }
217 return "one of: " + joined(labels, ", ");
218 }
219 if (field.kind == FieldKind::Quantity) {
220 std::string const text = "quantity in " + (field.unit.empty() ? field.unitAscii : field.unit);
221 std::vector<std::string> others;
222 for (std::size_t index = 1; index < field.units.size(); ++index) {
223 others.push_back(field.units.at(index).display);
224 }
225 return others.empty() ? text : text + " (or " + joined(others, ", ") + ")";
226 }
227 if (field.kind == FieldKind::Decimal && field.decimalPlaces) {
228 return "decimal, " + std::to_string(*field.decimalPlaces) + " places";
229 }
230 if (field.kind == FieldKind::Choice && field.choice) {
231 return "choice from " + field.choice->optionsAction;
232 }
233 if (field.kind == FieldKind::Array) {
234 return "list of " + field.itemType;
235 }
236 if (field.kind == FieldKind::ObjectArray) {
237 return "list of objects";
238 }
239 return std::string{forms::fieldKindName(field.kind)};
240}

244/// @brief The fields of an action, in `x-order`, as the forms engine reads its schema.
245/// @param schemaJson The action's schema, as `forms::schemaJson<A>()` emits it.
246/// @return The rows; none for a schema without properties. Or why the schema cannot be read.
247[[nodiscard]] inline std::expected<std::vector<ActionField>, std::string> actionFields(std::string_view schemaJson) {
248 auto model = forms::FormModel::fromSchema("", schemaJson);
249 if (!model) {
250 std::string const& where = model.error().path;
251 return std::unexpected(where.empty() ? model.error().message : where + ": " + model.error().message);
252 }
253 std::vector<ActionField> fields;
254 for (forms::FieldSpec const& spec : model->fields()) {
255 ActionField field{.name = spec.name, .type = detail::explorer::typeText(spec), .required = spec.required};
256 if (auto const& minimum = spec.exactMinimum ? spec.exactMinimum : spec.minimum) {
257 field.rules.push_back("min " + *minimum);
258 }
259 if (auto const& maximum = spec.exactMaximum ? spec.exactMaximum : spec.maximum) {
260 field.rules.push_back("max " + *maximum);
261 }
262 if (spec.multipleOf) {
263 field.rules.push_back("multiple of " + *spec.multipleOf);
264 }
265 if (spec.readOnly) {
266 field.rules.emplace_back("read-only");
267 }
268 if (spec.hidden) {
269 field.rules.emplace_back("hidden");
270 }
271 for (forms::RuleExpr const& rule : model->rules()) {
272 if (rule.names(spec.name)) {
273 field.rules.emplace_back(rule.kindName());
274 }
275 }
276 fields.push_back(std::move(field));
277 }
278 return fields;
279}

295/// @brief The one-node document "Try it" mounts for an action.
296///
297/// A private instance of the model under the alias `target`, the states `reply` and `failure`, and a column holding
298/// the node over the action (test id `tryit`) and, for a form, two texts showing the states (`tryit_result`,
299/// `tryit_error`). A form submits from its button only, and its `onSuccess` and `onError` set the states; a
300/// collection and a wizard show their own results and errors.
301/// @param actionId The action id.
302/// @param modelType The model type the action belongs to.
303/// @param kind The node.
304/// @return The document's JSON text, with the id `tryItScreenId(actionId)`.
305// NOLINTNEXTLINE(bugprone-easily-swappable-parameters): an action id and a model type read alike, and neither reads.
306[[nodiscard]] inline std::string tryItDocument(std::string_view actionId, std::string_view modelType, TryItKind kind) {
307 auto const quoted = [](std::string_view text) { return glz::write_json(std::string{text}).value_or("\"\""); };
308 std::string const action = quoted(actionId);
309 std::string node;
310 if (kind == TryItKind::Form) {
311 node = R"({"kind":"form","testId":"tryit","model":"target","action":)" + action +
312 R"(,"submit":"explicit",
313 "onSuccess":[{"set":"reply","value":{"ref":"event"}},{"set":"failure","value":null}],
314 "onError":[{"set":"failure","value":{"ref":"event"}},{"set":"reply","value":null}]},
315 {"kind":"text","testId":"tryit_result","role":"success","text":{"ref":"reply"}},
316 {"kind":"text","testId":"tryit_error","role":"error","text":{"if":[
317 {"fn":"engaged","args":[{"ref":"failure"}]},
318 {"fn":"format","args":["Failed: {0}",{"ref":"failure"}]},""]}})";
319 } else {
320 node = std::string{R"({"kind":")"} + (kind == TryItKind::Collection ? "collection" : "wizard") +
321 R"(","testId":"tryit","model":"target","action":)" + action + "}";
322 }
323 return R"({"vocab":"ui/1","id":)" + quoted(tryItScreenId(actionId)) + R"(,"title":)" +
324 quoted("Try " + std::string{actionId}) + R"(,
325 "models":{"target":{"type":)" +
326 quoted(modelType) + R"(,"instance":"private"}},
327 "state":{"reply":{"init":null},"failure":{"init":null}},
328 "view":{"kind":"column","gap":3,"children":[)" +
329 node + "]}}";
330}

332/// @brief A model's `schemas` document: `{actionType: schema}`, or why the server would not describe the model.
333using ModelSchemas = std::expected<std::string, std::string>;
334
335/// @brief The schemas documents of the models, by model type.
336using SchemasByModel = std::map<std::string, ModelSchemas, std::less<>>;
337
338/// @brief The actions a model's `schemas` document lists, with each one's schema.
339/// @param document The document's JSON text.
340/// @return The schemas by action id, or why the document cannot be read.
341[[nodiscard]] inline std::expected<std::map<std::string, std::string, std::less<>>, std::string> schemasOf(
342 std::string_view document) {
343 auto parsed = parseJson(document);
344 if (!parsed) {
345 return std::unexpected(std::move(parsed.error()));
346 }
347 auto const* const actions = std::get_if<Json::object_t>(&parsed->data);
348 if (actions == nullptr) {
349 return std::unexpected(std::string{"the schemas document is not an object"});
350 }
351 std::map<std::string, std::string, std::less<>> schemas;
352 for (auto const& [action, schema] : *actions) {
353 schemas.emplace(action, glz::write_json(schema).value_or(std::string{}));
354 }
355 return schemas;
356}

358/// @brief The catalog a "Try it" document loads with: @p catalog, whose `action` lookup also answers an action
359/// only a
360/// model's `schemas` document lists (load-time validation refuses an action the catalog does not know, and
361/// the catalog knows only those a served document names), and whose `schema` lookup also answers from the
362/// documents.
363/// @param catalog The client's catalog.
364/// @param schemas The models' `schemas` documents.
365/// @return The catalog.
366[[nodiscard]] inline Catalog tryItCatalog(Catalog catalog, SchemasByModel const& schemas) {
367 auto const known = std::make_shared<std::map<std::string, std::pair<std::string, std::string>, std::less<>>>();
368 for (auto const& [type, document] : schemas) {
369 if (!document) {
370 continue;
371 }
372 if (auto read = schemasOf(*document)) {
373 for (auto& [action, schema] : *read) {
374 known->emplace(action, std::pair{type, std::move(schema)});
375 }
376 }
377 }
378 catalog.action = [inner = std::move(catalog.action),
379 known](std::string_view actionId) -> std::optional<ActionInfo> {
380 if (inner) {
381 if (std::optional<ActionInfo> info = inner(actionId)) {
382 return info;
383 }
384 }
385 auto const found = known->find(actionId);
386 if (found == known->end()) {
387 return std::nullopt;
388 }
389 ActionInfo info;
390 info.modelType = found->second.first;
391 return info;
392 };
393 catalog.schema = [inner = std::move(catalog.schema),
394 known](std::string_view actionId) -> std::optional<std::string> {
395 if (inner) {
396 if (std::optional<std::string> schema = inner(actionId)) {
397 return schema;
398 }
399 }
400 auto const found = known->find(actionId);
401 if (found == known->end()) {
402 return std::nullopt;
403 }
404 return found->second.second;
405 };
406 return catalog;
407}

429/// @brief The app explorer, as a host screen (spec 6 §15): a splitter with the list — a search, the catalog's
430/// screens,
431/// and every model the catalog names with every action of it — and the chosen action's sections: its facts,
432/// its fields, the places that use it, and "Try it", the widget the framework builds for it, mounted live.
433///
434/// Built from backend widgets, so every frontend shows it. Its title is "Explorer"; it is never dirty, and closes
435/// when asked. Neither copyable nor movable; owned by the owner thread of its backend.
436///
437/// Test ids: `explorer`, `explorer_search`, `explorer_screen_<id>`, `explorer_model_<type>`,
438/// `explorer_action_<id>`, `explorer_title`, `explorer_facts`, `explorer_fields`, `explorer_uses`,
439/// `explorer_open_<n>`, `explorer_tryit`, `explorer_tryit_status`, `explorer_schema`, and inside "Try it" the
440/// generated document's own (`tryit`, …).
441class Explorer final : public ScreenHandle {
442public:
443 /// @brief The explorer, appended to @p page.
444 /// @param runtime The reactive runtime its title is computed in. Borrowed.
445 /// @param backend The backend its widgets are made with. Borrowed.
446 /// @param page The container it is appended to; it outlives the explorer.
447 /// @param options What it shows and reports.
448 Explorer(reactive::Runtime& runtime MORPH_LIFETIMEBOUND, IViewBackend& backend MORPH_LIFETIMEBOUND,
449 ContainerWidget& page MORPH_LIFETIMEBOUND, ExplorerOptions options)
450 : _backend{&backend},
451 _options{std::move(options)},
452 _scope{runtime},
453 _title{_scope.computed([] { return std::string{"Explorer"}; })},
454 _dirty{_scope.computed([] { return false; })},
455 _identity{_scope.computed([] { return Value{}; })},
456 _root{backend.createSplitter(&page, Axis::Horizontal)},
457 _listScroll{backend.createScroll(_root.get(), Axis::Vertical)},
458 _list{backend.createStack(_listScroll.get(), Axis::Vertical)},
459 _detailScroll{backend.createScroll(_root.get(), Axis::Vertical)},
460 _detail{backend.createStack(_detailScroll.get(), Axis::Vertical)} {
461 if (!_options.facts.action && !_options.facts.model) {
462 _options.facts = _options.catalog.facts();
463 }
464 _permitted.insert(_options.catalog.permitted.begin(), _options.catalog.permitted.end());
465 for (auto const& [screenId, document] : _options.documents) {
466 for (ActionUse& use : actionUses(*document, _options.facts)) {
467 std::string action = use.actionId;
468 _uses[std::move(action)].emplace_back(screenId, std::move(use));
469 }
470 }
471 _root->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
472 _root->setTestId("explorer");
473 _root->setSizes({kListWidth, kDetailWidth});
474 _listScroll->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
475 _detailScroll->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
476 _list->setGap(2);
477 _list->setLayout({.width = Sizing::stretch(), .height = {}});
478 _detail->setGap(4);
479 _detail->setLayout({.width = Sizing::stretch(), .height = {}});
480 _detail->setTestId("explorer_detail");
481 listAll();
482 showNothingChosen();
483 }

668 // Shows what a search matches: a screen by its title, an action by its id, a model by its type or any of its
669 // actions; a model with a matching action opens.
670 void filter(std::string const& wanted) {
671 std::string const needle = lower(wanted);
672 auto const matches = [&needle](std::string_view text) {
673 return needle.empty() || lower(text).contains(needle);
674 };
675 for (ScreenItem& item : _screens) {
676 item.button->setVisible(matches(item.title));
677 }
678 for (Model& model : _models) {
679 bool const typeMatches = matches(model.type);
680 bool any = false;
681 for (ActionEntry& action : model.actions) {
682 bool const shown = typeMatches || matches(action.id);
683 action.button->setVisible(shown);
684 any = any || shown;
685 }
686 model.box->setVisible(typeMatches || any);
687 if (!needle.empty() && any) {
688 model.box->setOpen(true);
689 }
690 }

755 // The catalog's facts of an action, as label and value cells; none when the catalog has none for it.
756 [[nodiscard]] std::vector<std::string> catalogFacts(std::string const& actionId) const {
757 if (!_options.catalog.actions || !_options.catalog.actions->contains(actionId)) {
758 return {};
759 }
760 ActionInfo const& info = _options.catalog.actions->at(actionId);
761 auto const listed = [](std::vector<std::string> const& names) {
762 return names.empty() ? std::string{"—"} : detail::explorer::joined(names, ", ");
763 };
764 std::vector<std::string> cells{"sends a credential",
765 info.credential ? "yes" : "no",
766 "server-calculated",
767 listed(info.calculatedFields),
768 "secret",
769 listed(info.secretFields)};
770 if (info.listFields) {
771 cells.insert(cells.end(), {"list fields", listed(*info.listFields)});
772 }
773 cells.insert(cells.end(), {"effects", effectsText(info)});
774 if (!info.records.empty()) {
775 cells.insert(cells.end(), {"records", recordsText(info)});
776 }
777 return cells;
778 }

780 // What an action declares it changes (spec 8 §5): why a screen did or did not update after it ran.
781 [[nodiscard]] static std::string effectsText(ActionInfo const& info) {
782 if (!info.effects) {
783 return "reads";
784 }
785 std::vector<std::string> effects;
786 for (ActionEffect const& effect : *info.effects) {
787 if (effect.kind == "changesInstance") {
788 effects.emplace_back("changes its instance");
789 } else if (effect.record) {
790 effects.push_back(effect.kind + " " + *effect.record);
791 } else {
792 effects.push_back(effect.kind);
793 }
794 }
795 return detail::explorer::joined(effects, ", ");
796 }
797
798 // The records an action's result holds, each with its key field and where the result holds it.
799 [[nodiscard]] static std::string recordsText(ActionInfo const& info) {
800 std::vector<std::string> records;
801 for (ActionRecordInfo const& record : info.records) {
802 std::vector<std::string> paths;
803 paths.reserve(record.paths.size());
804 for (std::string const& path : record.paths) {
805 paths.push_back(path.empty() ? std::string{"the result"} : path);
806 }
807 std::string key{" keyless"};
808 if (auto const& field = record.keyField) {
809 key = " by " + *field;
810 }
811 records.push_back(record.record + key + " at " + detail::explorer::joined(paths, ", "));
812 }
813 return detail::explorer::joined(records, "; ");
814 }

816 // The payload fingerprint and shape a schema carries, as label and value cells.
817 [[nodiscard]] static std::vector<std::string> payloadFacts(std::string const& schema) {
818 std::vector<std::string> cells;
819 auto parsed = parseJson(schema);
820 auto const* const object = parsed ? std::get_if<Json::object_t>(&parsed->data) : nullptr;
821 if (object == nullptr) {
822 return cells;
823 }
824 for (auto const& [key, value] : *object) {
825 if (key != "x-payloadFingerprint" && key != "x-payloadShape") {
826 continue;
827 }
828 std::string const* const plain = std::get_if<std::string>(&value.data);
829 cells.emplace_back(key == "x-payloadFingerprint" ? "payload fingerprint" : "payload shape");
830 cells.push_back(plain != nullptr ? *plain : glz::write_json(value).value_or(std::string{}));
831 }
832 return cells;
833 }
