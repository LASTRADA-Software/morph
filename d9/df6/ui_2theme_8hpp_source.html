101/// @brief A colour, 8 bits per channel.
102struct Rgba {
103 /// @brief Red.
104 std::uint8_t r = 0;
105 /// @brief Green.
106 std::uint8_t g = 0;
107 /// @brief Blue.
108 std::uint8_t b = 0;
109 /// @brief Alpha; 255 is opaque.
110 std::uint8_t a = 255;
111
112 /// @brief Field-wise equality.
113 /// @return Whether every channel is equal.
114 bool operator==(Rgba const&) const = default;
115
116 /// @brief The colour as `#rrggbb`, or `#rrggbbaa` when it is not opaque.
117 /// @return The text.
118 [[nodiscard]] std::string hex() const {
119 static constexpr std::string_view kDigits = "0123456789abcdef";
120 std::string out = "#";
121 auto const put = [&out](std::uint8_t const channel) {
122 out += kDigits.at(static_cast<std::size_t>(channel >> 4U));
123 out += kDigits.at(static_cast<std::size_t>(channel & 0xFU));
124 };
125 put(r);
126 put(g);
127 put(b);
128 if (a != 255) {
129 put(a);
130 }
131 return out;
132 }
133};

153/// @return It, or nothing when it is not a number.
154[[nodiscard]] inline std::optional<double> numberOf(Json const& json) {
155 if (auto const* const value = std::get_if<double>(&json.data)) {
156 return *value;

176/// @brief How many px a `rem` is: the default root font size, which the renderers do not change.
177inline constexpr double kPixelsPerRem = 16.0;

179/// @brief A DTCG `dimension` value in px.
180/// @param json The value: an object of a number `value` and the unit `px` or `rem`.
181/// @return Its length in px, or nothing when it is not a dimension.
182[[nodiscard]] inline std::optional<double> pixelsOf(Json const& json) {
183 auto const* const object = std::get_if<Json::object_t>(&json.data);
184 if (object == nullptr) {
185 return std::nullopt;
186 }
187 Json const* const value = memberOf(*object, "value");
188 Json const* const unit = memberOf(*object, "unit");
189 auto const length = value != nullptr ? numberOf(*value) : std::nullopt;
190 auto const* const unitText = unit != nullptr ? std::get_if<std::string>(&unit->data) : nullptr;
191 if (!length || unitText == nullptr) {
192 return std::nullopt;
193 }
194 if (*unitText == "px") {
195 return length;
196 }
197 if (*unitText == "rem") {
198 return *length * kPixelsPerRem;
199 }
200 return std::nullopt;
201}

203/// @brief Reads `#rgb`, `#rrggbb` or `#rrggbbaa`.
204/// @param text The text.
205/// @return The colour, or nothing.
206[[nodiscard]] inline std::optional<Rgba> hexColor(std::string_view text) {
207 if (!text.starts_with('#')) {
208 return std::nullopt;
209 }
210 text.remove_prefix(1);
211 auto const digit = [](char const symbol) -> int {
212 if (symbol >= '0' && symbol <= '9') {
213 return symbol - '0';
214 }
215 if (symbol >= 'a' && symbol <= 'f') {
216 return symbol - 'a' + 10;
217 }
218 if (symbol >= 'A' && symbol <= 'F') {
219 return symbol - 'A' + 10;
220 }
221 return -1;
222 };
223 if (!std::ranges::all_of(text, [&digit](char const symbol) { return digit(symbol) >= 0; })) {
224 return std::nullopt;
225 }
226 auto const pair = [&digit, text](std::size_t const offset) {
227 return static_cast<std::uint8_t>((digit(text.at(offset)) * 16) + digit(text.at(offset + 1)));
228 };
229 if (text.size() == 3) {
230 auto const single = [&digit, text](std::size_t const offset) {
231 return static_cast<std::uint8_t>(digit(text.at(offset)) * 17);
232 };
233 return Rgba{.r = single(0), .g = single(1), .b = single(2), .a = 255};
234 }
235 if (text.size() == 6 || text.size() == 8) {
236 return Rgba{.r = pair(0), .g = pair(2), .b = pair(4), .a = text.size() == 8 ? pair(6) : std::uint8_t{255}};
237 }
238 return std::nullopt;
239}

241/// @brief A channel from 0 to 1, as 8 bits.
242/// @param value The channel.
243/// @return It, or nothing outside 0 to 1.
244[[nodiscard]] inline std::optional<std::uint8_t> channelOf(double const value) {
245 if (value < 0.0 || value > 1.0) {
246 return std::nullopt;
247 }
248 return static_cast<std::uint8_t>(std::lround(value * 255.0));

251/// @brief Whether a string is an alias, `{a.b}`.
252/// @param text The text.
253/// @return The path it names, or nothing.
254[[nodiscard]] inline std::optional<std::string_view> aliasOf(std::string_view text) {
255 if (text.size() < 3 || !text.starts_with('{') || !text.ends_with('}')) {
256 return std::nullopt;
257 }
258 return text.substr(1, text.size() - 2);
259}

261/// @brief Whether a value holds an alias anywhere.
262/// @param json The value.
263/// @return Whether it does.
264// NOLINTNEXTLINE(misc-no-recursion): a composite value is walked member by member, bounded by kMaxJsonDepth.
265[[nodiscard]] inline bool holdsAlias(Json const& json) {
266 if (auto const* const text = std::get_if<std::string>(&json.data)) {
267 return aliasOf(*text).has_value();
268 }
269 if (auto const* const array = std::get_if<Json::array_t>(&json.data)) {
270 for (Json const& item : *array) {
271 if (holdsAlias(item)) {
272 return true;
273 }
274 }
275 }
276 if (auto const* const object = std::get_if<Json::object_t>(&json.data)) {
277 for (auto const& [name, member] : *object) {
278 static_cast<void>(name);
279 if (holdsAlias(member)) {
280 return true;
281 }
282 }
283 }
284 return false;
285}

306/// @brief Reads a colour value: `#rgb`, `#rrggbb`, `#rrggbbaa`, or the DTCG 2025.10 object, by its `hex` or by its
307/// `srgb` `components` (0 to 1) and `alpha`.
308/// @param json The value.
309/// @return The colour, or nothing when it is none of those.
310[[nodiscard]] inline std::optional<Rgba> colorOf(Json const& json) {
311 if (auto const* const text = std::get_if<std::string>(&json.data)) {
312 return detail::hexColor(*text);
313 }
314 auto const* const object = std::get_if<Json::object_t>(&json.data);
315 if (object == nullptr) {
316 return std::nullopt;
317 }
318 std::optional<double> alpha = 1.0;
319 if (Json const* const given = detail::memberOf(*object, "alpha")) {
320 alpha = detail::numberOf(*given);
321 }
322 auto const opacity = alpha ? detail::channelOf(*alpha) : std::nullopt;
323 if (!opacity) {
324 return std::nullopt;
325 }
326 if (Json const* const hex = detail::memberOf(*object, "hex")) {
327 auto const* const text = std::get_if<std::string>(&hex->data);
328 auto color = text != nullptr ? detail::hexColor(*text) : std::nullopt;
329 if (color) {
330 color->a = *opacity;
331 }
332 return color;
333 }
334 Json const* const space = detail::memberOf(*object, "colorSpace");
335 Json const* const components = detail::memberOf(*object, "components");
336 auto const* const spaceName = space != nullptr ? std::get_if<std::string>(&space->data) : nullptr;
337 auto const* const list = components != nullptr ? std::get_if<Json::array_t>(&components->data) : nullptr;
338 if (spaceName == nullptr || *spaceName != "srgb" || list == nullptr || list->size() != 3) {
339 return std::nullopt;
340 }
341 std::array<std::uint8_t, 3> channels{};
342 for (std::size_t index = 0; index < 3; ++index) {
343 auto const number = detail::numberOf(list->at(index));
344 auto const channel = number ? detail::channelOf(*number) : std::nullopt;
345 if (!channel) {
346 return std::nullopt;
347 }
348 channels.at(index) = *channel;
349 }
350 return Rgba{.r = channels.at(0), .g = channels.at(1), .b = channels.at(2), .a = *opacity};
351}

355/// @brief Checks a `dimension` or a `duration`: an object of a number `value` and a known `unit`.
356/// @param type `dimension` or `duration`.
357/// @param value The value.
358/// @return Why it is not one, or nothing.
359[[nodiscard]] inline std::optional<std::string> checkMeasure(std::string_view type, Json const& value) {
360 bool const dimension = type == "dimension";
361 auto const* const object = std::get_if<Json::object_t>(&value.data);
362 Json const* const number = object != nullptr ? memberOf(*object, "value") : nullptr;
363 Json const* const unit = object != nullptr ? memberOf(*object, "unit") : nullptr;
364 auto const* const unitText = unit != nullptr ? std::get_if<std::string>(&unit->data) : nullptr;
365 std::array<std::string_view, 2> const units =
366 dimension ? std::array<std::string_view, 2>{"px", "rem"} : std::array<std::string_view, 2>{"ms", "s"};
367 bool const known = unitText != nullptr && std::ranges::find(units, *unitText) != units.end();
368 if (number != nullptr && numberOf(*number) && known) {
369 return std::nullopt;
370 }
371 return dimension ? std::string{"a dimension is an object of a number value and the unit px or rem"}
372 : std::string{"a duration is an object of a number value and the unit ms or s"};
373}

403/// @brief Checks a `shadow`: an object of a `color` and the dimensions `offsetX`, `offsetY`, `blur` and `spread`, or a
404/// non-empty list of them.
405/// @param value The value.
406/// @return Why it is not one, or nothing.
407[[nodiscard]] inline std::optional<std::string> checkShadow(Json const& value) {
408 auto const one = [](Json const& layer) {
409 auto const* const object = std::get_if<Json::object_t>(&layer.data);
410 if (object == nullptr) {
411 return false;
412 }
413 Json const* const color = memberOf(*object, "color");
414 if (color == nullptr || !colorOf(*color)) {
415 return false;
416 }
417 return std::ranges::all_of(std::array<std::string_view, 4>{"offsetX", "offsetY", "blur", "spread"},
418 [object](std::string_view name) {
419 Json const* const length = memberOf(*object, name);
420 return length != nullptr && !checkMeasure("dimension", *length);
421 });
422 };
423 if (auto const* const layers = std::get_if<Json::array_t>(&value.data)) {
424 if (!layers->empty() && std::ranges::all_of(*layers, one)) {
425 return std::nullopt;
426 }
427 } else if (one(value)) {
428 return std::nullopt;
429 }
430 return std::string{"a shadow is an object of a colour and the dimensions offsetX, offsetY, blur and spread"};
431}

434 if (type == "shadow") {
435 return checkShadow(value);
436 }
437 if (type == "color") {

466/// @brief The `$type` an object declares, or the one it inherits.
467/// @param object The group's or token's members.
468/// @param where Its path with a trailing `.`, for the refusal.
469/// @param inherited The nearest enclosing `$type`.
470/// @return The type, or why the declaration is refused.
471// NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the path, then the type passed down, as readGroup takes them.
472[[nodiscard]] inline std::expected<std::string, std::string> declaredType(Json::object_t const& object,
473 std::string_view where,
474 std::string const& inherited) {
475 Json const* const declared = memberOf(object, "$type");
476 if (declared == nullptr) {
477 return inherited;
478 }
479 auto const* const text = std::get_if<std::string>(&declared->data);
480 if (text == nullptr) {
481 return std::unexpected(joined({"`", where, "$type` is a string"}));
482 }
483 return *text;

486/// @brief Reads one token into @p out.
487/// @param path The token's path.
488/// @param object The token's members.
489/// @param value Its `$value`.
490/// @param type The type it inherits.
491/// @param out Where it goes.
492/// @return Why the file is refused, or nothing.
493[[nodiscard]] inline std::optional<std::string> readToken(std::string const& path, Json::object_t const& object,
494 Json const& value, std::string const& type, TokenSet& out) {
495 auto tokenType = declaredType(object, path + ".", type);
496 if (!tokenType) {
497 return std::move(tokenType.error());
498 }
499 bool const alias = holdsAlias(value);
500 if (tokenType->empty()) {
501 if (alias && std::holds_alternative<std::string>(value.data)) {
502 out.insert_or_assign(path, Token{.type = {}, .value = value});
503 return std::nullopt;
504 }
505 return joined({"the token `", path, "` has no `$type`"});
506 }
507 if (std::ranges::find(kKnownTokenTypes, *tokenType) == kKnownTokenTypes.end()) {
508 return std::nullopt;
509 }
510 if (!alias) {
511 if (auto reason = checkTokenValue(*tokenType, value)) {
512 return joined({"the token `", path, "`: ", *reason});
513 }
514 }
515 out.insert_or_assign(path, Token{.type = std::move(*tokenType), .value = value});
516 return std::nullopt;
517}

519/// @brief Refuses a member name a token file may not have.
520/// @param prefix The group's path, with a trailing `.`.
521/// @param name The member's name.
522/// @return Why it is refused, or nothing.
523[[nodiscard]] inline std::optional<std::string> checkMemberName(std::string const& prefix, std::string const& name) {
524 if (name.starts_with('!')) {
525 return joined({"the must-understand key `", prefix, name, "` is not understood"});
526 }
527 if (name.find_first_of(".{}") != std::string::npos) {
528 return joined({"the token or group name `", prefix, name, "` has `.`, `{` or `}`"});
529 }
530 return std::nullopt;

533/// @brief Reads the tokens of one group into @p out.
534/// @param group The group's members.
535/// @param prefix The group's path, with a trailing `.`; empty at the top.
536/// @param inherited The nearest enclosing `$type`.
537/// @param out Where the tokens go.
538/// @return Why the file is refused, or nothing.
539// NOLINTBEGIN(misc-no-recursion,bugprone-easily-swappable-parameters): groups nest, bounded by kMaxJsonDepth; the
540// group's path, then the type it passes down.
541[[nodiscard]] inline std::optional<std::string> readGroup(Json::object_t const& group, std::string const& prefix,
542 std::string const& inherited, TokenSet& out) {
543 // NOLINTEND(misc-no-recursion,bugprone-easily-swappable-parameters)
544 auto const type = declaredType(group, prefix, inherited);
545 if (!type) {
546 return type.error();
547 }
548 if (memberOf(group, "$extends") != nullptr) {
549 return joined({"`", prefix, "$extends`: group inheritance is not supported"});
550 }
551 for (auto const& [name, member] : group) {
552 if (name.starts_with('$')) {
553 continue;
554 }
555 if (auto reason = checkMemberName(prefix, name)) {
556 return reason;
557 }
558 auto const* const object = std::get_if<Json::object_t>(&member.data);
559 if (object == nullptr) {
560 return joined({"`", prefix, name, "` is a token or a group, an object"});
561 }
562 std::string const path = prefix + name;
563 Json const* const value = memberOf(*object, "$value");
564 auto reason = value == nullptr ? readGroup(*object, path + ".", *type, out)
565 : readToken(path, *object, *value, *type, out);
566 if (reason) {
567 return reason;
568 }
569 if (out.size() > kMaxThemeTokens) {
570 return "a token file holds more than " + std::to_string(kMaxThemeTokens) + " tokens";
571 }
572 }
573 return std::nullopt;
574}

576/// @brief Resolves tokens' aliases, depth first, each token once.
577class AliasResolver {
578public:
579 /// @brief A resolver over @p merged.
580 /// @param merged The layered tokens.
581 explicit AliasResolver(TokenSet const& merged) : _merged{&merged} {}
582
583 /// @brief Resolves every token.
584 /// @return The resolved set, or why it is refused.
585 [[nodiscard]] std::expected<TokenSet, std::string> all() {
586 for (auto const& [path, token] : *_merged) {
587 static_cast<void>(token);
588 if (auto reason = resolve(path)) {
589 return std::unexpected(std::move(*reason));
590 }
591 }
592 return std::move(_done);
593 }
594
595private:
596 /// Resolves one token into `_done`.
597 // NOLINTNEXTLINE(misc-no-recursion): an alias resolves its target first, bounded by kMaxAliasChain.
598 [[nodiscard]] std::optional<std::string> resolve(std::string const& path) {
599 if (_done.contains(path)) {
600 return std::nullopt;
601 }
602 if (std::ranges::find(_chain, path) != _chain.end()) {
603 std::string cycle;
604 for (std::string const& step : _chain) {
605 cycle += step + " -> ";
606 }
607 return "an alias cycle: " + cycle + path;
608 }
609 if (_chain.size() >= kMaxAliasChain) {
610 return "an alias chain longer than " + std::to_string(kMaxAliasChain) + " at `" + path + "`";
611 }
612 auto const found = _merged->find(path);
613 if (found == _merged->end()) {
614 // Only an alias names a path the set may lack, and the aliasing token is on the chain.
615 return "`" + _chain.back() + "` refers to `" + path + "`, which no token file of the variant has";
616 }
617 _chain.push_back(path);
618 Token token = found->second;
619 if (auto const* const text = std::get_if<std::string>(&token.value.data)) {
620 if (auto const target = aliasOf(*text)) {
621 std::string const targetPath{*target};
622 if (auto reason = resolve(targetPath)) {
623 return reason;
624 }
625 Token const& resolved = _done.at(targetPath);
626 if (!token.type.empty() && token.type != resolved.type) {
627 return "the " + token.type + " token `" + path + "` aliases the " + resolved.type + " token `" +
628 targetPath + "`";
629 }
630 token = Token{.type = resolved.type, .value = resolved.value};
631 }
632 } else if (auto reason = replaceAliases(token.value)) {
633 return reason;
634 }
635 if (auto reason = checkTokenValue(token.type, token.value)) {
636 return "the token `" + path + "`: " + *reason;
637 }
638 _chain.pop_back();
639 _done.insert_or_assign(path, std::move(token));
640 return std::nullopt;
641 }
642
643 /// Replaces each alias inside a composite value with its target's resolved value.
644 // NOLINTNEXTLINE(misc-no-recursion): a composite value is walked member by member, bounded by kMaxJsonDepth.
645 [[nodiscard]] std::optional<std::string> replaceAliases(Json& value) {
646 if (auto const* const text = std::get_if<std::string>(&value.data)) {
647 if (auto const target = aliasOf(*text)) {
648 std::string const targetPath{*target};
649 if (auto reason = resolve(targetPath)) {
650 return reason;
651 }
652 value = _done.at(targetPath).value;
653 }
654 return std::nullopt;
655 }
656 if (auto* const array = std::get_if<Json::array_t>(&value.data)) {
657 for (Json& item : *array) {
658 if (auto reason = replaceAliases(item)) {
659 return reason;
660 }
661 }
662 } else if (auto* const object = std::get_if<Json::object_t>(&value.data)) {
663 for (auto& [name, member] : *object) {
664 static_cast<void>(name);
665 if (auto reason = replaceAliases(member)) {
666 return reason;
667 }
668 }
669 }
670 return std::nullopt;
671 }
672
673 TokenSet const* _merged;
674 TokenSet _done;
675 std::vector<std::string> _chain;
676};

749/// @brief Parses a DTCG 2025.10 token file.
750///
751/// Groups nest; a token is an object with `$value`, and its `$type` is its own or the nearest group's. Refused: text
752/// that is not a JSON object or nests deeper than `kMaxJsonDepth`, a must-understand (`!`) name, a name with `.`, `{`
753/// or `}`, a member that is not an object, a token with no type that is not a plain alias, a value that does not
754/// have its checked type's shape, `$extends`, and more than `kMaxThemeTokens` tokens. A token of a `$type` the client
755/// does not know is dropped.
756/// @param text The file.
757/// @return The tokens by path, aliases unresolved, or why the file is refused.
758[[nodiscard]] inline std::expected<TokenSet, std::string> parseTokens(std::string_view text) {
759 auto const json = parseJson(text);
760 if (!json) {
761 return std::unexpected(json.error());
762 }
763 auto const* const object = std::get_if<Json::object_t>(&json->data);
764 if (object == nullptr) {
765 return std::unexpected(std::string{"a token file is an object of groups and tokens"});
766 }
767 TokenSet tokens;
768 if (auto reason = detail::readGroup(*object, {}, {}, tokens)) {
769 return std::unexpected(std::move(*reason));
770 }
771 return tokens;
772}

774/// @brief Layers token files, later ones winning per path, and resolves every alias.
775///
776/// Refused: an alias cycle, an alias to a token no layer has, a chain of more than `kMaxAliasChain` aliases, an
777/// alias whose declared type differs from its target's, and a resolved value that does not have its type's shape.
778/// @param layers The files, `base` first.

784 merged.insert_or_assign(path, token);
785 }
786 }
787 return detail::AliasResolver{merged}.all();

790/// @brief Parses a theme index: `{"vocab": "ui-theme/1", "base": id, "dark": id, …, "density": {...},
791/// "surfaces": {...}, "icons": {...}}`.
792///
793/// Refused: not a JSON object, a `vocab` other than `ui-theme/1`, no `base`, a must-understand (`!`) key, a variant
794/// that is not a file id, a `density` or `surfaces` that is not a map of name to file id, and more than
795/// `kMaxThemeVariants` variants, densities or surfaces, and an `icons` that is not an object of icon name
796/// (`isIconName`) to file id or holds more than `kMaxThemeIcons` names.
797/// @param text The index.
798/// @return The index, or why it is refused.
799[[nodiscard]] inline std::expected<ThemeIndex, std::string> parseThemeIndex(std::string_view text) {
800 auto const json = parseJson(text);
801 if (!json) {
802 return std::unexpected(json.error());
803 }
804 auto const* const object = std::get_if<Json::object_t>(&json->data);
805 if (object == nullptr) {
806 return std::unexpected(std::string{"a theme index is an object"});
807 }
808 if (auto reason = detail::mustUnderstand(*json, "the theme index")) {
809 return std::unexpected(std::move(*reason));
810 }
811 Json const* const vocab = detail::memberOf(*object, "vocab");
812 auto const* const vocabText = vocab != nullptr ? std::get_if<std::string>(&vocab->data) : nullptr;
813 if (vocabText == nullptr || *vocabText != kThemeVocabulary) {
814 return std::unexpected(std::string{"a theme index's vocab is ui-theme/1"});
815 }
816 ThemeIndex index;
817 for (auto const& [key, member] : *object) {
818 if (key == "vocab" || key == "density" || key == "surfaces" || key == "icons") {
819 continue;
820 }
821 auto const* const fileId = std::get_if<std::string>(&member.data);
822 if (fileId == nullptr || fileId->empty()) {
823 return std::unexpected("the variant `" + key + "` names a token file");
824 }
825 if (key == "base") {
826 index.base = *fileId;
827 } else {
828 index.variants.insert_or_assign(key, *fileId);
829 }
830 }
831 if (index.base.empty()) {
832 return std::unexpected(std::string{"a theme index names its `base` token file"});
833 }
834 if (auto reason = detail::readThemeMap(*object, "density", index.densities)) {
835 return std::unexpected(std::move(*reason));
836 }
837 if (auto reason = detail::readThemeMap(*object, "surfaces", index.surfaces)) {
838 return std::unexpected(std::move(*reason));
839 }
840 if (auto reason = detail::readIcons(*object, index.icons)) {
841 return std::unexpected(std::move(*reason));
842 }
843 if (index.variants.size() > kMaxThemeVariants || index.densities.size() > kMaxThemeVariants ||
844 index.surfaces.size() > kMaxThemeVariants) {
845 return std::unexpected("a theme index lists more than " + std::to_string(kMaxThemeVariants) +
846 " variants, densities or surfaces");
847 }
848 return index;
849}

851/// @brief Finds a token file of the bundle by id.
852using TokenFileLookup = std::function<TokenSet const*(std::string_view fileId)>;

854/// @brief Resolves one combination of a theme: `base`, then the variant, the density and the surface, each optional.
855/// @param index The theme index.
856/// @param files The bundle's token files.
857/// @param variant A variant name, or empty for `base` alone.
858/// @param density A density name, or empty for none.
859/// @param surface A surface name, or empty for none.
860/// @return The resolved tokens, or why they are refused: a name the index does not list, a file the bundle lacks, or
861/// `resolveTokens`' refusals.
862// NOLINTBEGIN(bugprone-easily-swappable-parameters): variant, density, surface: the order they layer in.
863[[nodiscard]] inline std::expected<TokenSet, std::string> resolveTheme(ThemeIndex const& index,
864 TokenFileLookup const& files,
865 std::string_view variant,
866 std::string_view density,
867 std::string_view surface) {
868 // NOLINTEND(bugprone-easily-swappable-parameters)
869 std::vector<TokenSet const*> layers;
870 auto const layer = [&](std::string const& fileId) -> std::optional<std::string> {
871 TokenSet const* const tokens = files(fileId);
872 if (tokens == nullptr) {
873 return "the token file `" + fileId + "` is not in the bundle";
874 }
875 layers.push_back(tokens);
876 return std::nullopt;
877 };
878 auto const named = [&](std::map<std::string, std::string, std::less<>> const& from, std::string_view name,
879 std::string_view what) -> std::optional<std::string> {
880 if (name.empty()) {
881 return std::nullopt;
882 }
883 auto const found = from.find(name);
884 if (found == from.end()) {
885 return "the theme has no " + std::string{what} + " `" + std::string{name} + "`";
886 }
887 return layer(found->second);
888 };
889 std::optional<std::string> reason = layer(index.base);
890 if (!reason) {
891 reason = named(index.variants, variant, "variant");
892 }
893 if (!reason) {
894 reason = named(index.densities, density, "density");
895 }
896 if (!reason) {
897 reason = named(index.surfaces, surface, "surface");
898 }
899 if (reason) {
900 return std::unexpected(std::move(*reason));
901 }
902 return resolveTokens(layers);
903}

905/// @brief Checks that every combination of a theme resolves: each variant (or none), density (or none) and surface
906/// (or none) over `base`.
907/// @param index The theme index.
908/// @param files The bundle's token files.
909/// @return Why the theme is refused, naming the combination, or nothing.
910[[nodiscard]] inline std::optional<std::string> checkTheme(ThemeIndex const& index, TokenFileLookup const& files) {
911 auto const namesOf = [](std::map<std::string, std::string, std::less<>> const& from) {
912 std::vector<std::string> names{std::string{}};
913 for (auto const& [name, fileId] : from) {
914 static_cast<void>(fileId);
915 names.push_back(name);
916 }
917 return names;
918 };
919 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): variant, density, surface: the order they layer in.
920 auto const label = [](std::string const& variant, std::string const& density, std::string const& surface) {
921 std::string out = "base";
922 if (!variant.empty()) {
923 out += " + " + variant;
924 }
925 if (!density.empty()) {
926 out += " + density " + density;
927 }
928 if (!surface.empty()) {
929 out += " + surface " + surface;
930 }
931 return out;
932 };
933 std::vector<std::string> const surfaces = namesOf(index.surfaces);
934 for (std::string const& variant : namesOf(index.variants)) {
935 for (std::string const& density : namesOf(index.densities)) {
936 for (std::string const& surface : surfaces) {
937 auto const resolved = resolveTheme(index, files, variant, density, surface);
938 if (!resolved) {
939 return label(variant, density, surface) + ": " + resolved.error();
940 }
941 }
942 }
943 }
944 return std::nullopt;
945}

1067/// @brief A checked `shadow` value as a `ThemeShadow`; of a list of layers, the first.
1068/// @param json The value, checked by `checkShadow`.
1069/// @return It, or nothing when it is not a shadow.
1070[[nodiscard]] inline std::optional<ThemeShadow> shadowOf(Json const& json) {
1071 Json const* layer = &json;
1072 if (auto const* const layers = std::get_if<Json::array_t>(&json.data)) {
1073 if (layers->empty()) {
1074 return std::nullopt;
1075 }
1076 layer = &layers->front();
1077 }
1078 auto const* const object = std::get_if<Json::object_t>(&layer->data);
1079 if (object == nullptr) {
1080 return std::nullopt;
1081 }
1082 Json const* const color = memberOf(*object, "color");
1083 auto const rgba = color != nullptr ? colorOf(*color) : std::nullopt;
