98/// @brief The index of a theme: which token files make up each variant.
99struct ThemeIndex {
100 /// @brief The file every variant layers over.
101 std::string base;
102 /// @brief Named variants, such as `light` and `dark`, each one file.
103 std::map<std::string, std::string, std::less<>> variants;
104 /// @brief Densities, such as `comfortable` and `compact`, each one file.
105 std::map<std::string, std::string, std::less<>> densities;
106 /// @brief Named surfaces, such as `sidebar`, each one file that overrides the variant for a subtree.
107 std::map<std::string, std::string, std::less<>> surfaces;
108 /// @brief The icon set: icon name to the bundle's `icon` file that draws it; empty for none.
109 std::map<std::string, std::string, std::less<>> icons;
110};

197/// @brief Parses a DTCG 2025.10 token file.
198///
199/// Groups nest; a token is an object with `$value`, and its `$type` is its own or the nearest group's. Refused: text
200/// that is not a JSON object or nests deeper than `kMaxJsonDepth`, a must-understand (`!`) name, a name with `.`, `{`
201/// or `}`, a member that is not an object, a token with no type that is not a plain alias, a value that does not
202/// have its checked type's shape, `$extends`, and more than `kMaxThemeTokens` tokens, and a token that holds members
203/// of its own: write it as a group whose `$root` member is the token. A group's `$root` is the token at the group's
204/// own path, and an alias may name it either as `{a.b}` or as `{a.b.$root}`. A token of a `$type` the client does not
205/// know is kept unchecked with its type, which `unreadThemeTokens` names.
206/// @param text The file.
207/// @return The tokens by path, aliases unresolved, or why the file is refused.
208[[nodiscard]] MORPH_API std::expected<TokenSet, std::string> parseTokens(std::string_view text);
209
210/// @brief Layers token files, later ones winning per path, and resolves every alias.
211///
212/// Refused: an alias cycle, an alias to a token no layer has, a chain of more than `kMaxAliasChain` aliases, an
213/// alias whose declared type differs from its target's, and a resolved value that does not have its type's shape.
214/// @param layers The files, `base` first.
215/// @return The resolved tokens, or why the combination is refused.
216[[nodiscard]] MORPH_API std::expected<TokenSet, std::string> resolveTokens(std::vector<TokenSet const*> const& layers);
217
218/// @brief Parses a theme index: `{"vocab": "ui-theme/1", "base": id, "dark": id, …, "density": {...},
219/// "surfaces": {...}, "icons": {...}}`.
220///
221/// Refused: not a JSON object, a `vocab` other than `ui-theme/1`, no `base`, a must-understand (`!`) key, a variant
222/// that is not a file id, a `density` or `surfaces` that is not a map of name to file id, and more than
223/// `kMaxThemeVariants` variants, densities or surfaces, and an `icons` that is not an object of icon name
224/// (`isIconName`) to file id or holds more than `kMaxThemeIcons` names.
225/// @param text The index.
226/// @return The index, or why it is refused.
227[[nodiscard]] MORPH_API std::expected<ThemeIndex, std::string> parseThemeIndex(std::string_view text);

302/// @brief What a `ui-theme/1` token is read as.
303enum class ThemeTokenKind : std::uint8_t {
304 Color, ///< A `color` token.
305 Number, ///< A `number` token.
306 Dimension, ///< A `dimension` token, read in px (a `rem` is 16 px).
307 Shadow, ///< A `shadow` token: a colour and four px lengths; of a list of layers, the first.
308 FontFamily, ///< A `fontFamily` token: a family name, or a list of them in order of preference.
309 FontWeight, ///< A `fontWeight` token, read as its numeric weight, 1 to 1000, in `ThemeValues::numbers`.
310 Stroke, ///< A `strokeStyle` token: `solid` or `dashed`, in `ThemeValues::strokes`.
311 Keyword, ///< A `keyword` token: one of the names the spec lists, in `ThemeValues::keywords`.
312};
