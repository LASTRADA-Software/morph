106/// @brief A colour, 8 bits per channel.
107struct Rgba {
108 /// @brief Red.
109 std::uint8_t r = 0;
110 /// @brief Green.
111 std::uint8_t g = 0;
112 /// @brief Blue.
113 std::uint8_t b = 0;
114 /// @brief Alpha; 255 is opaque.
115 std::uint8_t a = 255;
116
117 /// @brief Field-wise equality.
118 /// @return Whether every channel is equal.
119 bool operator==(Rgba const&) const = default;
120
121 /// @brief The colour as `#rrggbb`, or `#rrggbbaa` when it is not opaque.
122 /// @return The text.
123 [[nodiscard]] std::string hex() const {
124 static constexpr std::string_view kDigits = "0123456789abcdef";
125 std::string out = "#";
126 auto const put = [&out](std::uint8_t const channel) {
127 out += kDigits.at(static_cast<std::size_t>(channel >> 4U));
128 out += kDigits.at(static_cast<std::size_t>(channel & 0xFU));
129 };
130 put(r);
131 put(g);
132 put(b);
133 if (a != 255) {
134 put(a);
135 }
136 return out;
137 }
138};

181/// @brief How many px a `rem` is: the default root font size, which the renderers do not change.
182inline constexpr double kPixelsPerRem = 16.0;
183
184/// @brief A DTCG `dimension` value in px.
185/// @param json The value: an object of a number `value` and the unit `px` or `rem`.
186/// @return Its length in px, or nothing when it is not a dimension.
187[[nodiscard]] inline std::optional<double> pixelsOf(Json const& json) {
188 auto const* const object = std::get_if<Json::object_t>(&json.data);
189 if (object == nullptr) {
190 return std::nullopt;
191 }
192 Json const* const value = memberOf(*object, "value");
193 Json const* const unit = memberOf(*object, "unit");
194 auto const length = value != nullptr ? numberOf(*value) : std::nullopt;
195 auto const* const unitText = unit != nullptr ? std::get_if<std::string>(&unit->data) : nullptr;
196 if (!length || unitText == nullptr) {
197 return std::nullopt;
198 }
199 if (*unitText == "px") {
200 return length;
201 }
202 if (*unitText == "rem") {
203 return *length * kPixelsPerRem;
204 }
205 return std::nullopt;
206}

208/// @brief Reads `#rgb`, `#rrggbb` or `#rrggbbaa`.
209/// @param text The text.
210/// @return The colour, or nothing.
211[[nodiscard]] inline std::optional<Rgba> hexColor(std::string_view text) {
212 if (!text.starts_with('#')) {
213 return std::nullopt;
214 }
215 text.remove_prefix(1);
216 auto const digit = [](char const symbol) -> int {
217 if (symbol >= '0' && symbol <= '9') {
218 return symbol - '0';
219 }
220 if (symbol >= 'a' && symbol <= 'f') {
221 return symbol - 'a' + 10;
222 }
223 if (symbol >= 'A' && symbol <= 'F') {
224 return symbol - 'A' + 10;
225 }
226 return -1;
227 };
228 if (!std::ranges::all_of(text, [&digit](char const symbol) { return digit(symbol) >= 0; })) {
229 return std::nullopt;
230 }
231 auto const pair = [&digit, text](std::size_t const offset) {
232 return static_cast<std::uint8_t>((digit(text.at(offset)) * 16) + digit(text.at(offset + 1)));
233 };
234 if (text.size() == 3) {
235 auto const single = [&digit, text](std::size_t const offset) {
236 return static_cast<std::uint8_t>(digit(text.at(offset)) * 17);
237 };
238 return Rgba{.r = single(0), .g = single(1), .b = single(2), .a = 255};
239 }
240 if (text.size() == 6 || text.size() == 8) {
241 return Rgba{.r = pair(0), .g = pair(2), .b = pair(4), .a = text.size() == 8 ? pair(6) : std::uint8_t{255}};
242 }
243 return std::nullopt;
244}

246/// @brief A channel from 0 to 1, as 8 bits.
247/// @param value The channel.
248/// @return It, or nothing outside 0 to 1.
249[[nodiscard]] inline std::optional<std::uint8_t> channelOf(double const value) {
250 if (value < 0.0 || value > 1.0) {
251 return std::nullopt;
252 }
253 return static_cast<std::uint8_t>(std::lround(value * 255.0));

259/// @brief Whether a string is an alias, `{a.b}`.
260/// @param text The text.
261/// @return The path it names, or nothing. A trailing `.$root` is dropped: `{color.ok.$root}` names `color.ok`, the
262/// group's own token.
263[[nodiscard]] inline std::optional<std::string_view> aliasOf(std::string_view text) {
264 if (text.size() < 3 || !text.starts_with('{') || !text.ends_with('}')) {
265 return std::nullopt;
266 }
267 std::string_view path = text.substr(1, text.size() - 2);
268 constexpr std::string_view suffix = ".$root";
269 if (path.size() > suffix.size() && path.ends_with(suffix)) {
270 path.remove_suffix(suffix.size());
271 }
272 return path;
273}

275/// @brief Whether a value holds an alias anywhere.
276/// @param json The value.
277/// @return Whether it does.
278// NOLINTNEXTLINE(misc-no-recursion): a composite value is walked member by member, bounded by kMaxJsonDepth.
279[[nodiscard]] inline bool holdsAlias(Json const& json) {
280 if (auto const* const text = std::get_if<std::string>(&json.data)) {
281 return aliasOf(*text).has_value();
282 }
283 if (auto const* const array = std::get_if<Json::array_t>(&json.data)) {
284 for (Json const& item : *array) {
285 if (holdsAlias(item)) {
286 return true;
287 }
288 }
289 }
290 if (auto const* const object = std::get_if<Json::object_t>(&json.data)) {
291 for (auto const& [name, member] : *object) {
292 static_cast<void>(name);
293 if (holdsAlias(member)) {
294 return true;
295 }
296 }
297 }
298 return false;
299}

320/// @brief Reads a colour value: `#rgb`, `#rrggbb`, `#rrggbbaa`, or the DTCG 2025.10 object, by its `hex` or by its
321/// `srgb` `components` (0 to 1) and `alpha`.
322/// @param json The value.
323/// @return The colour, or nothing when it is none of those.
324[[nodiscard]] inline std::optional<Rgba> colorOf(Json const& json) {
325 if (auto const* const text = std::get_if<std::string>(&json.data)) {
326 return detail::hexColor(*text);
327 }
328 auto const* const object = std::get_if<Json::object_t>(&json.data);
329 if (object == nullptr) {
330 return std::nullopt;
331 }
332 std::optional<double> alpha = 1.0;
333 if (Json const* const given = detail::memberOf(*object, "alpha")) {
334 alpha = detail::numberOf(*given);
335 }
336 auto const opacity = alpha ? detail::channelOf(*alpha) : std::nullopt;
337 if (!opacity) {
338 return std::nullopt;
339 }
340 if (Json const* const hex = detail::memberOf(*object, "hex")) {
341 auto const* const text = std::get_if<std::string>(&hex->data);
342 auto color = text != nullptr ? detail::hexColor(*text) : std::nullopt;
343 if (color) {
344 color->a = *opacity;
345 }
346 return color;
347 }
348 Json const* const space = detail::memberOf(*object, "colorSpace");
349 Json const* const components = detail::memberOf(*object, "components");
350 auto const* const spaceName = space != nullptr ? std::get_if<std::string>(&space->data) : nullptr;
351 auto const* const list = components != nullptr ? std::get_if<Json::array_t>(&components->data) : nullptr;
352 if (spaceName == nullptr || *spaceName != "srgb" || list == nullptr || list->size() != 3) {
353 return std::nullopt;
354 }
355 std::array<std::uint8_t, 3> channels{};
356 for (std::size_t index = 0; index < 3; ++index) {
357 auto const number = detail::numberOf(list->at(index));
358 auto const channel = number ? detail::channelOf(*number) : std::nullopt;
359 if (!channel) {
360 return std::nullopt;
361 }
362 channels.at(index) = *channel;
363 }
364 return Rgba{.r = channels.at(0), .g = channels.at(1), .b = channels.at(2), .a = *opacity};
365}

369/// @brief Checks a `dimension` or a `duration`: an object of a number `value` and a known `unit`.
370/// @param type `dimension` or `duration`.
371/// @param value The value.
372/// @return Why it is not one, or nothing.
373[[nodiscard]] inline std::optional<std::string> checkMeasure(std::string_view type, Json const& value) {
374 bool const dimension = type == "dimension";
375 auto const* const object = std::get_if<Json::object_t>(&value.data);
376 Json const* const number = object != nullptr ? memberOf(*object, "value") : nullptr;
377 Json const* const unit = object != nullptr ? memberOf(*object, "unit") : nullptr;
378 auto const* const unitText = unit != nullptr ? std::get_if<std::string>(&unit->data) : nullptr;
379 std::array<std::string_view, 2> const units =
380 dimension ? std::array<std::string_view, 2>{"px", "rem"} : std::array<std::string_view, 2>{"ms", "s"};
381 bool const known = unitText != nullptr && std::ranges::find(units, *unitText) != units.end();
382 if (number != nullptr && numberOf(*number) && known) {
383 return std::nullopt;
384 }
385 return dimension ? std::string{"a dimension is an object of a number value and the unit px or rem"}
386 : std::string{"a duration is an object of a number value and the unit ms or s"};
387}

422/// @brief Checks a `shadow`: an object of a `color` and the dimensions `offsetX`, `offsetY`, `blur` and `spread`, or a
423/// non-empty list of them.
424/// @param value The value.
425/// @return Why it is not one, or nothing.
426[[nodiscard]] inline std::optional<std::string> checkShadow(Json const& value) {
427 auto const one = [](Json const& layer) {
428 auto const* const object = std::get_if<Json::object_t>(&layer.data);
429 if (object == nullptr) {
430 return false;
431 }
432 Json const* const color = memberOf(*object, "color");
433 if (color == nullptr || !colorOf(*color)) {
434 return false;
435 }
436 return std::ranges::all_of(std::array<std::string_view, 4>{"offsetX", "offsetY", "blur", "spread"},
437 [object](std::string_view name) {
438 Json const* const length = memberOf(*object, name);
439 return length != nullptr && !checkMeasure("dimension", *length);
440 });
441 };
442 if (auto const* const layers = std::get_if<Json::array_t>(&value.data)) {
443 if (!layers->empty() && std::ranges::all_of(*layers, one)) {
444 return std::nullopt;
445 }
446 } else if (one(value)) {
447 return std::nullopt;
448 }
449 return std::string{"a shadow is an object of a colour and the dimensions offsetX, offsetY, blur and spread"};
450}

504// NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the path, then the type passed down, as readGroup takes them.
505[[nodiscard]] inline std::expected<std::string, std::string> declaredType(Json::object_t const& object,
506 std::string_view where,

519/// @brief Reads one token into @p out.
520/// @param path The token's path.
521/// @param object The token's members.
522/// @param value Its `$value`.
523/// @param type The type it inherits.
524/// @param out Where it goes.
525/// @return Why the file is refused, or nothing.
526[[nodiscard]] inline std::optional<std::string> readToken(std::string const& path, Json::object_t const& object,
527 Json const& value, std::string const& type, TokenSet& out) {
528 for (auto const& [name, member] : object) {
529 static_cast<void>(member);
530 if (!name.starts_with('$')) {
531 return joined({"the token `", path, "` holds `", name, "`: a token holds no tokens; write `", path,
532 "` as a group whose `$root` is the token"});
533 }
534 }
535 auto tokenType = declaredType(object, path + ".", type);
536 if (!tokenType) {
537 return std::move(tokenType.error());
538 }
539 bool const alias = holdsAlias(value);
540 if (tokenType->empty()) {
541 if (alias && std::holds_alternative<std::string>(value.data)) {
542 out.insert_or_assign(path, Token{.type = {}, .value = value});
543 return std::nullopt;
544 }
545 return joined({"the token `", path, "` has no `$type`"});
546 }
547 if (std::ranges::find(kKnownTokenTypes, *tokenType) == kKnownTokenTypes.end()) {
548 // Kept unchecked with its type, so the report of tokens nothing reads can name it.
549 out.insert_or_assign(path, Token{.type = std::move(*tokenType), .value = value});
550 return std::nullopt;
551 }
552 if (!alias) {
553 if (auto reason = checkTokenValue(*tokenType, value)) {
554 return joined({"the token `", path, "`: ", *reason});
555 }
556 }
557 out.insert_or_assign(path, Token{.type = std::move(*tokenType), .value = value});
558 return std::nullopt;
559}

579/// @param out Where the token goes.
580/// @return Why the file is refused, or nothing.
581// NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the group's path, then its inherited type; swapped, the `$root` token is read at the wrong path.
582[[nodiscard]] inline std::optional<std::string> readRoot(Json const& member, std::string const& prefix,
583 std::string const& inherited, TokenSet& out) {
584 if (prefix.empty()) {
585 return std::string{"`$root` names a group's own token, and the top of a file is no group"};
586 }
587 std::string const path = prefix.substr(0, prefix.size() - 1);
588 auto const* const object = std::get_if<Json::object_t>(&member.data);
589 Json const* const value = object != nullptr ? memberOf(*object, "$value") : nullptr;
590 if (value == nullptr) {
591 return joined({"`", prefix, "$root` is a token, an object with a `$value`"});
592 }
593 return readToken(path, *object, *value, inherited, out);
594}

596/// @brief Reads the tokens of one group into @p out.
597/// @param group The group's members.
598/// @param prefix The group's path, with a trailing `.`; empty at the top.
599/// @param inherited The nearest enclosing `$type`.
600/// @param out Where the tokens go.
601/// @return Why the file is refused, or nothing.
602// NOLINTBEGIN(misc-no-recursion,bugprone-easily-swappable-parameters): groups nest, bounded by kMaxJsonDepth; the
603// group's path, then the type it passes down.
604[[nodiscard]] inline std::optional<std::string> readGroup(Json::object_t const& group, std::string const& prefix,
605 std::string const& inherited, TokenSet& out) {
606 // NOLINTEND(misc-no-recursion,bugprone-easily-swappable-parameters)
607 auto const type = declaredType(group, prefix, inherited);
608 if (!type) {
609 return type.error();
610 }
611 if (memberOf(group, "$extends") != nullptr) {
612 return joined({"`", prefix, "$extends`: group inheritance is not supported"});
613 }
614 for (auto const& [name, member] : group) {
615 if (name == kRootMember) {
616 if (auto reason = readRoot(member, prefix, *type, out)) {
617 return reason;
618 }
619 continue;
620 }
621 if (name.starts_with('$')) {
622 continue;
623 }
624 if (auto reason = checkMemberName(prefix, name)) {
625 return reason;
626 }
627 auto const* const object = std::get_if<Json::object_t>(&member.data);
628 if (object == nullptr) {
629 return joined({"`", prefix, name, "` is a token or a group, an object"});
630 }
631 std::string const path = prefix + name;
632 Json const* const value = memberOf(*object, "$value");
633 auto reason = value == nullptr ? readGroup(*object, path + ".", *type, out)
634 : readToken(path, *object, *value, *type, out);
635 if (reason) {
636 return reason;
637 }
638 if (out.size() > kMaxThemeTokens) {
639 return "a token file holds more than " + std::to_string(kMaxThemeTokens) + " tokens";
640 }
641 }
642 return std::nullopt;
643}

645/// @brief Resolves tokens' aliases, depth first, each token once.
646class AliasResolver {
647public:
648 /// @brief A resolver over @p merged.
649 /// @param merged The layered tokens.
650 explicit AliasResolver(TokenSet const& merged) : _merged{&merged} {}
651
652 /// @brief Resolves every token.
653 /// @return The resolved set, or why it is refused.
654 [[nodiscard]] std::expected<TokenSet, std::string> all() {
655 for (auto const& [path, token] : *_merged) {
656 static_cast<void>(token);
657 if (auto reason = resolve(path)) {
658 return std::unexpected(std::move(*reason));
659 }
660 }
661 return std::move(_done);
662 }
663
664private:
665 /// Resolves one token into `_done`.
666 // NOLINTNEXTLINE(misc-no-recursion): an alias resolves its target first, bounded by kMaxAliasChain.
667 [[nodiscard]] std::optional<std::string> resolve(std::string const& path) {
668 if (_done.contains(path)) {
669 return std::nullopt;
670 }
671 if (std::ranges::find(_chain, path) != _chain.end()) {
672 std::string cycle;
673 for (std::string const& step : _chain) {
674 cycle += step + " -> ";
675 }
676 return "an alias cycle: " + cycle + path;
677 }
678 if (_chain.size() >= kMaxAliasChain) {
679 return "an alias chain longer than " + std::to_string(kMaxAliasChain) + " at `" + path + "`";
680 }
681 auto const found = _merged->find(path);
682 if (found == _merged->end()) {
683 // Only an alias names a path the set may lack, and the aliasing token is on the chain.
684 return "`" + _chain.back() + "` refers to `" + path + "`, which no token file of the variant has";
685 }
686 _chain.push_back(path);
687 Token token = found->second;
688 if (auto const* const text = std::get_if<std::string>(&token.value.data)) {
689 if (auto const target = aliasOf(*text)) {
690 std::string const targetPath{*target};
691 if (auto reason = resolve(targetPath)) {
692 return reason;
693 }
694 Token const& resolved = _done.at(targetPath);
695 if (!token.type.empty() && token.type != resolved.type) {
696 return "the " + token.type + " token `" + path + "` aliases the " + resolved.type + " token `" +
697 targetPath + "`";
698 }
699 token = Token{.type = resolved.type, .value = resolved.value};
700 }
701 } else if (auto reason = replaceAliases(token.value)) {
702 return reason;
703 }
704 if (auto reason = checkTokenValue(token.type, token.value)) {
705 return "the token `" + path + "`: " + *reason;
706 }
707 _chain.pop_back();
708 _done.insert_or_assign(path, std::move(token));
709 return std::nullopt;
710 }
711
712 /// Replaces each alias inside a composite value with its target's resolved value.
713 // NOLINTNEXTLINE(misc-no-recursion): a composite value is walked member by member, bounded by kMaxJsonDepth.
714 [[nodiscard]] std::optional<std::string> replaceAliases(Json& value) {
715 if (auto const* const text = std::get_if<std::string>(&value.data)) {
716 if (auto const target = aliasOf(*text)) {
717 std::string const targetPath{*target};
718 if (auto reason = resolve(targetPath)) {
719 return reason;
720 }
721 value = _done.at(targetPath).value;
722 }
723 return std::nullopt;
724 }
725 if (auto* const array = std::get_if<Json::array_t>(&value.data)) {
726 for (Json& item : *array) {
727 if (auto reason = replaceAliases(item)) {
728 return reason;
729 }
730 }
731 } else if (auto* const object = std::get_if<Json::object_t>(&value.data)) {
732 for (auto& [name, member] : *object) {
733 static_cast<void>(name);
734 if (auto reason = replaceAliases(member)) {
735 return reason;
736 }
737 }
738 }
739 return std::nullopt;
740 }
741
742 TokenSet const* _merged;
743 TokenSet _done;
744 std::vector<std::string> _chain;
745};

818/// @brief Parses a DTCG 2025.10 token file.
819///
820/// Groups nest; a token is an object with `$value`, and its `$type` is its own or the nearest group's. Refused: text
821/// that is not a JSON object or nests deeper than `kMaxJsonDepth`, a must-understand (`!`) name, a name with `.`, `{`
822/// or `}`, a member that is not an object, a token with no type that is not a plain alias, a value that does not
823/// have its checked type's shape, `$extends`, and more than `kMaxThemeTokens` tokens, and a token that holds members
824/// of its own: write it as a group whose `$root` member is the token. A group's `$root` is the token at the group's
825/// own path, and an alias may name it either as `{a.b}` or as `{a.b.$root}`. A token of a `$type` the client does not
826/// know is kept unchecked with its type, which `unreadThemeTokens` names.
827/// @param text The file.
828/// @return The tokens by path, aliases unresolved, or why the file is refused.
829[[nodiscard]] inline std::expected<TokenSet, std::string> parseTokens(std::string_view text) {
830 auto const json = parseJson(text);
831 if (!json) {
832 return std::unexpected(json.error());
833 }
834 auto const* const object = std::get_if<Json::object_t>(&json->data);
835 if (object == nullptr) {
836 return std::unexpected(std::string{"a token file is an object of groups and tokens"});
837 }
838 TokenSet tokens;
839 if (auto reason = detail::readGroup(*object, {}, {}, tokens)) {
840 return std::unexpected(std::move(*reason));
841 }
842 return tokens;
843}

845/// @brief Layers token files, later ones winning per path, and resolves every alias.
846///
847/// Refused: an alias cycle, an alias to a token no layer has, a chain of more than `kMaxAliasChain` aliases, an
848/// alias whose declared type differs from its target's, and a resolved value that does not have its type's shape.
849/// @param layers The files, `base` first.
850/// @return The resolved tokens, or why the combination is refused.
851[[nodiscard]] inline std::expected<TokenSet, std::string> resolveTokens(std::vector<TokenSet const*> const& layers) {
852 TokenSet merged;
853 for (TokenSet const* const layer : layers) {
854 for (auto const& [path, token] : *layer) {
855 merged.insert_or_assign(path, token);
856 }
857 }
858 return detail::AliasResolver{merged}.all();
859}

861/// @brief Parses a theme index: `{"vocab": "ui-theme/1", "base": id, "dark": id, …, "density": {...},
862/// "surfaces": {...}, "icons": {...}}`.
863///
864/// Refused: not a JSON object, a `vocab` other than `ui-theme/1`, no `base`, a must-understand (`!`) key, a variant
865/// that is not a file id, a `density` or `surfaces` that is not a map of name to file id, and more than
866/// `kMaxThemeVariants` variants, densities or surfaces, and an `icons` that is not an object of icon name
867/// (`isIconName`) to file id or holds more than `kMaxThemeIcons` names.
868/// @param text The index.
869/// @return The index, or why it is refused.
870[[nodiscard]] inline std::expected<ThemeIndex, std::string> parseThemeIndex(std::string_view text) {
871 auto const json = parseJson(text);
872 if (!json) {
873 return std::unexpected(json.error());
874 }
875 auto const* const object = std::get_if<Json::object_t>(&json->data);
876 if (object == nullptr) {
877 return std::unexpected(std::string{"a theme index is an object"});
878 }
879 if (auto reason = detail::mustUnderstand(*json, "the theme index")) {
880 return std::unexpected(std::move(*reason));
881 }
882 Json const* const vocab = detail::memberOf(*object, "vocab");
883 auto const* const vocabText = vocab != nullptr ? std::get_if<std::string>(&vocab->data) : nullptr;
884 if (vocabText == nullptr || *vocabText != kThemeVocabulary) {
885 return std::unexpected(std::string{"a theme index's vocab is ui-theme/1"});
886 }
887 ThemeIndex index;
888 for (auto const& [key, member] : *object) {
889 if (key == "vocab" || key == "density" || key == "surfaces" || key == "icons") {
890 continue;
891 }
892 auto const* const fileId = std::get_if<std::string>(&member.data);
893 if (fileId == nullptr || fileId->empty()) {
894 return std::unexpected("the variant `" + key + "` names a token file");
895 }
896 if (key == "base") {
897 index.base = *fileId;
898 } else {
899 index.variants.insert_or_assign(key, *fileId);
900 }
901 }
902 if (index.base.empty()) {
903 return std::unexpected(std::string{"a theme index names its `base` token file"});
904 }
905 if (auto reason = detail::readThemeMap(*object, "density", index.densities)) {
906 return std::unexpected(std::move(*reason));
907 }
908 if (auto reason = detail::readThemeMap(*object, "surfaces", index.surfaces)) {
909 return std::unexpected(std::move(*reason));
910 }
911 if (auto reason = detail::readIcons(*object, index.icons)) {
912 return std::unexpected(std::move(*reason));
913 }
914 if (index.variants.size() > kMaxThemeVariants || index.densities.size() > kMaxThemeVariants ||
915 index.surfaces.size() > kMaxThemeVariants) {
916 return std::unexpected("a theme index lists more than " + std::to_string(kMaxThemeVariants) +
917 " variants, densities or surfaces");
918 }
919 return index;
920}

925/// @brief Resolves one combination of a theme: `base`, then the variant, the density and the surface, each optional.
926/// @param index The theme index.
927/// @param files The bundle's token files.
928/// @param variant A variant name, or empty for `base` alone.
929/// @param density A density name, or empty for none.
930/// @param surface A surface name, or empty for none.
931/// @return The resolved tokens, or why they are refused: a name the index does not list, a file the bundle lacks, or
932/// `resolveTokens`' refusals.
933// NOLINTBEGIN(bugprone-easily-swappable-parameters): variant, density, surface: the order they layer in.
934[[nodiscard]] inline std::expected<TokenSet, std::string> resolveTheme(ThemeIndex const& index,
935 TokenFileLookup const& files,
936 std::string_view variant,
937 std::string_view density,
938 std::string_view surface) {
939 // NOLINTEND(bugprone-easily-swappable-parameters)
940 std::vector<TokenSet const*> layers;
941 auto const layer = [&](std::string const& fileId) -> std::optional<std::string> {
942 TokenSet const* const tokens = files(fileId);
943 if (tokens == nullptr) {
944 return "the token file `" + fileId + "` is not in the bundle";
945 }
946 layers.push_back(tokens);
947 return std::nullopt;
948 };
949 auto const named = [&](std::map<std::string, std::string, std::less<>> const& from, std::string_view name,
950 std::string_view what) -> std::optional<std::string> {
951 if (name.empty()) {
952 return std::nullopt;
953 }
954 auto const found = from.find(name);
955 if (found == from.end()) {
956 return "the theme has no " + std::string{what} + " `" + std::string{name} + "`";
957 }
958 return layer(found->second);
959 };
960 std::optional<std::string> reason = layer(index.base);
961 if (!reason) {
962 reason = named(index.variants, variant, "variant");
963 }
964 if (!reason) {
965 reason = named(index.densities, density, "density");
966 }
967 if (!reason) {
968 reason = named(index.surfaces, surface, "surface");
969 }
970 if (reason) {
971 return std::unexpected(std::move(*reason));
972 }
973 return resolveTokens(layers);
974}

1012/// @brief Why a resolved theme names a keyword that is not drawn.
1013/// @param resolved The resolved tokens.
1014/// @return The reason, such as `shape.iconOnly: only default and square are drawn`, or nothing when every keyword
1015/// the theme gives is one its spec lists.
1016[[nodiscard]] inline std::optional<std::string> keywordProblem(TokenSet const& resolved) {
1017 for (KeywordSpec const& spec : kKeywords) {
1018 auto const found = resolved.find(spec.path);
1019 if (found == resolved.end() || found->second.type != "keyword") {
1020 continue;
1021 }
1022 auto const* const named = std::get_if<std::string>(&found->second.value.data);
1023 if (named == nullptr || !isKeywordName(spec.path, *named)) {
1024 return std::string{spec.path} + ": only " + std::string{spec.names.front()} + " and " +
1025 std::string{spec.names.back()} + " are drawn";
1026 }
1027 }
1028 return std::nullopt;

1032/// @param resolved The resolved tokens.

1417/// @brief A checked `shadow` value as a `ThemeShadow`; of a list of layers, the first.
1418/// @param json The value, checked by `checkShadow`.
1419/// @return It, or nothing when it is not a shadow.
1420[[nodiscard]] inline std::optional<ThemeShadow> shadowOf(Json const& json) {
1421 Json const* layer = &json;
1422 if (auto const* const layers = std::get_if<Json::array_t>(&json.data)) {
1423 if (layers->empty()) {
1424 return std::nullopt;
1425 }
1426 layer = &layers->front();
1427 }
1428 auto const* const object = std::get_if<Json::object_t>(&layer->data);
1429 if (object == nullptr) {
1430 return std::nullopt;
1431 }
1432 Json const* const color = memberOf(*object, "color");
1433 auto const rgba = color != nullptr ? colorOf(*color) : std::nullopt;
1434 auto const length = [object](std::string_view name) -> std::optional<double> {
1435 Json const* const member = memberOf(*object, name);
1436 return member != nullptr ? pixelsOf(*member) : std::nullopt;
1437 };
1438 auto const x = length("offsetX");
1439 auto const y = length("offsetY");
1440 auto const blur = length("blur");
1441 auto const spread = length("spread");
1442 if (!rgba || !x || !y || !blur || !spread) {
1443 return std::nullopt;
1444 }
1445 return ThemeShadow{.color = *rgba, .offsetX = *x, .offsetY = *y, .blur = *blur, .spread = *spread};
1446}

1521/// @brief Reads one token of the set into @p values: the theme's value when it has the set's type, else the default.
1522/// @param values Receives the value.
1523/// @param spec The token of the set.
1524/// @param token The theme's token at the spec's path, or `nullptr`.
1525inline void fillValue(ThemeValues& values, ThemeTokenSpec const& spec, Token const* token) {
1526 std::string const path{spec.path};
1527 switch (spec.kind) {
1528 case ThemeTokenKind::Color: {
1529 auto const color = typedValue(token, "color", [](Json const& value) { return colorOf(value); });
1530 // A colour with no default follows another colour and is left out until the theme sets it.
1531 if (!color && spec.color.empty()) {
1532 break;
1533 }

1655/// @param fileIds The ids of the theme's files.
1656/// @param files The bundle's token files.
1657/// @return The paths.

1659 TokenFileLookup const& files) {

1661 for (std::string const& fileId : fileIds) {

1664 defined[path].push_back(&token);
1665 }

1669 std::vector<std::string> pending;
1670 auto const reach = [&read, &pending](std::string_view path) {
1671 if (read.emplace(path).second) {

1676 reach(spec.path);

1747/// @brief @p top drawn over the opaque colour @p under.
1748/// @param top The colour, which may be translucent.
1749/// @param under The colour beneath it.
1750/// @return The opaque colour that results.
1751[[nodiscard]] inline Rgba over(Rgba top, Rgba under) {
1752 auto const mix = [&top](std::uint8_t above, std::uint8_t below) {
1753 return static_cast<std::uint8_t>(std::lround(((above * top.a) + (below * (255.0 - top.a))) / 255.0));
1754 };
1755 return Rgba{.r = mix(top.r, under.r), .g = mix(top.g, under.g), .b = mix(top.b, under.b), .a = 255};
1756}
1757
1758/// @brief The WCAG 2 relative luminance of an opaque colour.
1759/// @param color The colour.
1760/// @return It, from 0 to 1.
1761[[nodiscard]] inline double luminance(Rgba color) {
1762 auto const linear = [](std::uint8_t channel) {
1763 double const value = channel / 255.0;
1764 return value <= 0.03928 ? value / 12.92 : std::pow((value + 0.055) / 1.055, 2.4);
1765 };
1766 return (0.2126 * linear(color.r)) + (0.7152 * linear(color.g)) + (0.0722 * linear(color.b));
1767}
1768
1769/// @brief The WCAG 2 contrast ratio of two opaque colours.
1770/// @param first One colour.
1771/// @param second The other.
1772/// @return The ratio, 1 to 21.
1773[[nodiscard]] inline double contrastRatio(Rgba first, Rgba second) {
1774 double const high = std::max(luminance(first), luminance(second));
1775 double const low = std::min(luminance(first), luminance(second));
1776 return (high + 0.05) / (low + 0.05);
1777}

1781/// @brief The fills text lies on, and the text colours that must stay readable on them.
1782///
1783/// `color.text` must hold 4.5:1 and `color.muted` 3:1 on `color.hover`, `color.selection`, `color.track` and
1784/// `color.controlSurface`. A theme that sets `color.selectionText` must hold it 4.5:1 on `color.selection`. Muted gets
1785/// 3:1 because the framework's own defaults hold that and not 4.5:1; the check catches a fill that makes text
1786/// disappear, not one that merely restyles it. A translucent fill is composited over `color.surface` first.
1787/// @param values The theme's values.
1788/// @return One sentence per text colour and fill that fall short, such as `color.hover: color.text is 1.50:1 on it,
1789/// below 4.5:1`; empty when every pair holds.
1790[[nodiscard]] inline std::vector<std::string> themeContrastProblems(ThemeValues const& values) {
1791 struct TextRule {
1792 std::string_view path;
1793 double minimum;
1794 std::string_view label;
1795 };
1796 static constexpr std::array<TextRule, 2> kTexts = {{{.path = "color.text", .minimum = 4.5, .label = "4.5"},
1797 {.path = "color.muted", .minimum = 3.0, .label = "3"}}};
1798 static constexpr std::array<std::string_view, 4> kFills = {"color.hover", "color.selection", "color.track",
1799 "color.controlSurface"};
1800 std::vector<std::string> problems;
1801 auto const colorAt = [&values](std::string_view path) -> std::optional<Rgba> {
1802 auto const found = values.colors.find(path);
1803 return found == values.colors.end() ? std::nullopt : std::optional<Rgba>{found->second};
1804 };
1805 auto const surface = colorAt("color.surface").value_or(Rgba{});
1806 auto const check = [&](std::string_view fillPath, TextRule const& text) {
1807 auto const fill = colorAt(fillPath);
1808 auto const color = colorAt(text.path);
1809 if (!fill || !color) {
1810 return;
1811 }
1812 Rgba const under = detail::over(*fill, detail::over(surface, Rgba{}));
1813 double const ratio = detail::contrastRatio(detail::over(*color, under), under);
1814 if (ratio < text.minimum) {
1815 problems.push_back(
1816 std::format("{}: {} is {:.2f}:1 on it, below {}:1", fillPath, text.path, ratio, text.label));
1817 }
1818 };
1819 for (std::string_view const fillPath : kFills) {
1820 for (TextRule const& text : kTexts) {
1821 check(fillPath, text);
1822 }
1823 }
1824 check("color.selection", {.path = "color.selectionText", .minimum = 4.5, .label = "4.5"});
1825 return problems;
1826}
