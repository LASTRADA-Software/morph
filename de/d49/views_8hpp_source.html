49/// @brief One action a view screen can fire — the row opener (`v-rowAction`)
50/// or an entry of `v-actions`.
51///
52/// Always built by `describeAction<Action>(...)`, never constructed by hand,
53/// so `actionTypeId` always reflects a real, registered
54/// `ActionTraits<Action>::typeId()` rather than a hand-typed, unchecked
55/// string (contrast `morph::forms::Choice`'s `OptionsAction` NTTP, which
56/// *is* an unchecked string — see docs/spec/forms/choice.md, "Limitations").
57struct ActionDescriptor {
58 /// @brief Registered type id of the target action.
59 std::string_view actionTypeId;
60 /// @brief Button label. Empty means "use the action type id as-is".
61 std::string_view label{};
62 /// @brief Row button vs. collection-wide button.
63 ActionScope scope{ActionScope::Row};
64 /// @brief Field-name mapping applied to prefill the target action's
65 /// draft from the activating row. Empty for an action that needs
66 /// no row context (e.g. a collection-scope "create").
67 std::span<const BindEntry> bind{};
68 /// @brief Whether a renderer must confirm before firing this action.
69 bool confirm{false};
70};

72/// @brief Builds an `ActionDescriptor` for @p Action, resolving its wire type
73/// id from the real, registered `ActionTraits<Action>` rather than a
74/// hand-typed string.
75/// @tparam Action Registered action type — `BRIDGE_REGISTER_ACTION` for
76/// @p Action must already have run earlier in this
77/// translation unit, or `ActionTraits<Action>` is incomplete
78/// and this fails to compile.
79/// @param label Button label. Defaults to empty ("use the action type id").
80/// @param scope Row button vs. collection-wide button. Defaults to `Row`.
81/// @param bind Field-name mapping prefilling the action's draft from the
82/// activating row. Defaults to empty (no prefill).
83/// @param confirm Whether a renderer must confirm before firing. Defaults to
84/// `false`.
85/// @return The populated descriptor.
86template <typename Action>
87[[nodiscard]] consteval ActionDescriptor describeAction(std::string_view label = {},
88 ActionScope scope = ActionScope::Row,
89 std::span<const BindEntry> bind = {}, bool confirm = false) {
90 return ActionDescriptor{
91 .actionTypeId = ::morph::model::ActionTraits<Action>::typeId(),
92 .label = label,
93 .scope = scope,
94 .bind = bind,
95 .confirm = confirm,
96 };
97}

178/// @brief Builds one `v-columns` entry, copying `x-decimalPlaces` /
179/// `ExtUnits` off @p rowDom's property node (and, via its `$ref`, the
180/// resolved `$def`) when @p field names a `Quantity` property.
181/// @param rowDom The row type's own `morph::forms::schemaJson<Row>()`,
182/// parsed into a DOM.
183/// @param field Wire field name on the row type.
184/// @param label Column header.
185/// @param hidden Whether the column is present in the row model but not
186/// displayed.
187/// @return The `v-columns` entry.
188[[nodiscard]] inline glz::generic_u64 buildColumnEntry(glz::generic_u64 const& rowDom, std::string const& field,
189 std::string const& label, bool hidden) {
190 glz::generic_u64 entry{};
191 entry["field"] = field;
192 entry["label"] = label;
193 if (hidden) {
194 entry["v-hidden"] = true;
195 }
196 auto const& propsObj = rowDom["properties"].get_object();
197 auto iter = propsObj.find(field);
198 if (iter == propsObj.end()) {
199 return entry; // declared/derived field not on the row type: bare column, no crash
200 }
201 auto const& prop = iter->second;
202 if (prop.contains("x-decimalPlaces")) {
203 entry["x-decimalPlaces"] = prop["x-decimalPlaces"];
204 }
205 // A Quantity field's ExtUnits sits directly on the property node when its
206 // Quantity type occurs only once in Row (glaze inlines a single-use
207 // struct schema in place); glaze only promotes the schema to a `$defs`
208 // entry (referenced back via `$ref`) when the *same* Quantity type
209 // occurs on more than one property (see forms.md's "CFSharedDefFields"
210 // fixture) — deriveColumns must read whichever shape schemaJson<Row>()
211 // actually produced for this particular row type.
212 if (prop.contains("ExtUnits")) {
213 entry["ExtUnits"] = prop["ExtUnits"];
214 } else if (prop.contains("$ref") && rowDom.contains("$defs")) {
215 auto const ref = prop["$ref"].get_string();
216 auto const defName = ref.substr(ref.find_last_of('/') + 1);
217 auto const& defs = rowDom["$defs"].get_object();
218 auto defIter = defs.find(defName);
219 if (defIter != defs.end() && defIter->second.contains("ExtUnits")) {
220 entry["ExtUnits"] = defIter->second["ExtUnits"];
221 }
222 }
223 return entry;
224}

226/// @brief Derives (or, when `V::columns` is declared, overrides) the
227/// `v-columns` array for row type @p Row, reusing
228/// `morph::forms::schemaJson<Row>()` as the single source of each
229/// field's declaration order, `x-decimalPlaces`, and `ExtUnits` — the
230/// same schema a standalone form for `Row` would carry.
231/// @tparam V View descriptor (only `HasColumnOverrides<V>` is consulted).
232/// @tparam Row The query action's row element type (a plain, reflectable,
233/// default-constructible aggregate).
234/// @return The `v-columns` array, JSON-encoded.
235template <typename V, typename Row>
236[[nodiscard]] std::string deriveColumns() {
237 glz::generic_u64 rowDom{};
238 // The `glz::read_json` half of this condition guards against
239 // `schemaJson<Row>()` returning text that isn't valid JSON at all. That
240 // can only happen when `glz::write_json_schema<Row>()` itself fails
241 // (forms.md, "Total schema failure yields an empty string") -- glaze's
242 // own schema writer failing for a real, compilable, default-constructible
243 // reflectable aggregate, not something reachable through any Row a
244 // caller can actually declare -- so this half is untestable dead code,
245 // kept only because it is cheaper to leave in than to prove absent at
246 // every call site. `!rowDom.contains("properties")` is the reachable
247 // half: it is what actually fires for a Row with no reflected members
248 // (see VtEmptyRow in test_views.cpp), since `mergeSchemaExtras` only
249 // ever writes `dom["properties"]` from inside the per-member visitor,
250 // which for zero members never runs.
251 if (glz::read_json(rowDom, ::morph::forms::schemaJson<Row>()) || !rowDom.contains("properties")) {
252 return "[]";
253 }
254 glz::generic_u64::array_t columns{};
255 if constexpr (HasColumnOverrides<V>) {
256 for (auto const& colOverride : V::columns) {
257 auto const label =
258 colOverride.label.empty() ? std::string{colOverride.field} : std::string{colOverride.label};
259 columns.emplace_back(buildColumnEntry(rowDom, std::string{colOverride.field}, label, colOverride.hidden));
260 }
261 } else {
262 auto const& propsObj = rowDom["properties"].get_object();
263 std::vector<std::pair<std::string, std::uint64_t>> ordered;
264 ordered.reserve(propsObj.size());
265 for (auto const& [name, prop] : propsObj) {
266 auto const order = prop.contains("x-order") ? prop["x-order"].template get<std::uint64_t>() : 0;
267 ordered.emplace_back(name, order);
268 }
269 std::ranges::sort(ordered, [](auto const& a, auto const& b) { return a.second < b.second; });
270 for (auto const& [name, order] : ordered) {
271 static_cast<void>(order);
272 columns.emplace_back(buildColumnEntry(rowDom, name, name, false));
273 }
274 }
275 return glz::write_json(columns).value_or("[]");
276}

278/// @brief Builds one `v-rowAction` / `v-actions` entry from an
279/// `ActionDescriptor`.
280/// @param descriptor The descriptor to serialise.
281/// @param includeButtonFields `true` for `v-actions` entries (adds `label`,
282/// `scope`, and `confirm` when set); `false` for
283/// `v-rowAction` (only `action` and `bind`).
284/// @return The JSON node.
285[[nodiscard]] inline glz::generic_u64 buildActionNode(ActionDescriptor const& descriptor, bool includeButtonFields) {
286 glz::generic_u64 node{};
287 node["action"] = std::string{descriptor.actionTypeId};
288 if (includeButtonFields) {
289 node["label"] = std::string{descriptor.label.empty() ? descriptor.actionTypeId : descriptor.label};
290 node["scope"] = std::string{descriptor.scope == ActionScope::Row ? "row" : "collection"};
291 if (descriptor.confirm) {
292 node["confirm"] = true;
293 }
294 }
295 if (!descriptor.bind.empty()) {
296 glz::generic_u64 bindNode{};
297 for (auto const& entry : descriptor.bind) {
298 bindNode[std::string{entry.actionField}] = std::string{entry.rowField};
299 }
300 node["bind"] = bindNode;
301 }
302 return node;
303}

305/// @brief Builds the complete `viewSchemaJson<V>()` document.
306/// @tparam V View descriptor.
307/// @return The view-schema JSON, or an empty string on internal DOM failure
308/// (schema generation never throws — see docs/spec/forms/forms.md).
309template <typename V>
310[[nodiscard]] std::string buildViewSchema() {
311 using Query = typename V::query;
312 using Result = typename ::morph::model::ActionTraits<Query>::Result;
313
314 glz::generic_u64 dom{};
315 dom["v-kind"] = std::string{viewKindName<typename V::kind>};
316 dom["v-query"] = std::string{::morph::model::ActionTraits<Query>::typeId()};
317 if constexpr (HasViewTitle<V>) {
318 dom["v-title"] = std::string{V::title};
319 } else {
320 dom["v-title"] = std::string{::morph::model::ActionTraits<Query>::typeId()};
321 }
322 if constexpr (HasRowKey<V>) {
323 dom["v-rowKey"] = std::string{V::rowKey};
324 } else {
325 dom["v-rowKey"] = std::string{"id"};
326 }
327
328 // The row element type is the query result itself when it is a
329 // std::vector<...>, else its first array-valued member, in declaration
330 // order — the same two shapes Choice's optionRows reads (DynamicForm.qml).
331 std::string columnsJson = "[]";
332 if constexpr (isStdVector<Result>) {
333 using Row = typename std::remove_cvref_t<Result>::value_type;
334 columnsJson = deriveColumns<V, Row>();
335 } else {
336 Result probe{};
337 bool found = false;
338 ::morph::forms::detail::forEachNamedMember(probe,
339 [&]<std::size_t I>(std::string_view name, const auto& member) {
340 static_cast<void>(name);
341 static_cast<void>(I);
342 using Member = std::remove_cvref_t<decltype(member)>;
343 if constexpr (isStdVector<Member>) {
344 if (!found) {
345 using Row = typename Member::value_type;
346 columnsJson = deriveColumns<V, Row>();
347 found = true;
348 }
349 }
350 });
351 }
352 glz::generic_u64 columnsDom{};
353 if (!glz::read_json(columnsDom, columnsJson)) {
354 dom["v-columns"] = columnsDom;
355 }
356
357 if constexpr (HasRowActionDescriptor<V>) {
358 dom["v-rowAction"] = buildActionNode(V::rowAction, false);
359 }
360 if constexpr (HasViewActions<V>) {
361 glz::generic_u64::array_t actionsArray{};
362 for (auto const& descriptor : V::actions) {
363 actionsArray.emplace_back(buildActionNode(descriptor, true));
364 }
365 dom["v-actions"] = actionsArray;
366 }
367
368 return glz::write_json(dom).value_or(std::string{});
369}

373/// @brief Generates the view-schema JSON document for view descriptor @p V.
374///
375/// A **new, separate top-level document** from `morph::forms::schemaJson<A>()`
376/// — never merged into any action schema. Computed once per type and cached,
377/// exactly like `schemaJson<A>()`.
378/// @tparam V View descriptor: `using kind = CollectionView` (or
379/// `MasterDetailView`); `using query = <registered query action>`;
380/// optional `static constexpr std::string_view title`, `rowKey`,
381/// `std::array<ColumnOverride, N> columns`,
382/// `ActionDescriptor rowAction`, and
383/// `std::array<ActionDescriptor, N> actions`.
384/// @return The view-schema JSON.
385template <typename V>
386[[nodiscard]] std::string viewSchemaJson() {
387 static const std::string cached = detail::buildViewSchema<V>();
388 return cached;
389}

391/// @brief Traits specialisation that maps a view descriptor type to its
392/// string type-id. Specialise via `BRIDGE_REGISTER_VIEW`.
393/// @tparam V Concrete view descriptor type.
394template <typename V>
395struct ViewTraits;
396
397/// @brief Process-level registry mapping a view's string type-id to its
398/// `viewSchemaJson<V>()` provider, so a controller can enumerate every
399/// registered view by name — parallels `ActionExecuteRegistry` for
400/// actions (docs/spec/core/bridge.md), but a view registers no
401/// executor, only a schema provider.
402class ViewRegistry {
403public:
404 /// @brief Registers @p V's schema provider under @p viewId. A second
405 /// registration for the same @p viewId silently replaces the
406 /// first (last-write-wins, the same policy `ActionDispatcher`
407 /// and `ModelRegistryFactory` use — see docs/spec/core/registry.md).
408 /// @tparam V Concrete view descriptor type.
409 /// @param viewId String type-id (`ViewTraits<V>::typeId()`).
410 template <typename V>
411 void registerView(std::string_view viewId) {
412 _providers.insert_or_assign(std::string{viewId}, [] { return viewSchemaJson<V>(); });
413 }
414
415 /// @brief Returns the cached `viewSchemaJson<V>()` for @p viewId.
416 /// @param viewId String type-id previously passed to `registerView`.
417 /// @return The view-schema JSON.
418 [[nodiscard]] std::string schemaJson(std::string_view viewId) const {
419 auto iter = _providers.find(std::string{viewId});
420 if (iter == _providers.end()) {
421 throw std::runtime_error("unknown view: " + std::string{viewId});
422 }
423 return iter->second();
424 }
425
426 /// @brief Returns every registered view id, in unspecified order.
427 /// @return The registered view ids.
428 [[nodiscard]] std::vector<std::string> viewIds() const {
429 std::vector<std::string> ids;
430 ids.reserve(_providers.size());
431 for (auto const& entry : _providers) {
432 ids.push_back(entry.first);
433 }
434 return ids;
435 }
436
437 /// @brief Returns the process-level singleton registry.
438 /// @return Reference to the singleton.
439 static ViewRegistry& instance() {
440 static ViewRegistry inst;
441 return inst;
442 }
443
444private:
445 std::unordered_map<std::string, std::function<std::string()>> _providers;
446};
