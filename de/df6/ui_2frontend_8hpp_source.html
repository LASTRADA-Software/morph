107/// @brief Why a source could not be opened.
108struct ConnectError {
109 /// @brief What went wrong.
110 enum class Kind : std::uint8_t {
111 Unreachable, ///< The server could not be reached, or the connection broke during the handshake.
112 NoUiService, ///< The peer answered, but serves no UI (`unknown envelope kind: ui-hello`).
113 Unsupported, ///< The bundle needs a vocabulary this client does not speak; the message names it.

119 /// @brief What to tell the user, UTF-8.

133/// @brief Where the screens and the models are: a server (remote) or the application linked in-process (local).
134///
135/// `main` builds one from its arguments and hands it to `Frontend::run`. Every call is made on the context's owner
136/// executor, and every callback it is given must be called there too.
137class AppSource {
138public:
139 AppSource() = default;
140 virtual ~AppSource() = default;
141 AppSource(AppSource const&) = delete;
142 AppSource& operator=(AppSource const&) = delete;
143 AppSource(AppSource&&) = delete;
144 AppSource& operator=(AppSource&&) = delete;
145
146 /// @brief Connects (or, local, sets up the models), signs the user in natively if the application needs it, and
147 /// delivers the bundle.
148 /// @param context The running frontend's context; its executor is every bridge's callback executor.
149 /// @param ready Called once, on the owner, now or later, with the bundle or why there is none. A source that is
150 /// closed first never calls it.
151 virtual void open(AppContext& context, std::function<void(std::expected<Bundle, ConnectError>)> ready) = 0;
152
153 /// @brief Replaces the dispatch target under the mounted screens: `Bridge::switchBackend`, posted to the owner.
154 /// @param backend One of the application's backends.
155 /// @param done Called once, on the owner, when the switch committed, or with why it did not; a switch that
156 /// fails leaves the old backend in place.
157 virtual void switchBackend(Backend backend, std::function<void(std::expected<void, std::string>)> done) = 0;
158
159 /// @brief Releases everything `open` made (connections, bridges, handlers) and drops a `ready` not yet
160 /// delivered. Called by `runApp` once the screens are gone and before the frontend's runtime is; the
161 /// source may be opened again afterwards.
162 virtual void close() = 0;
163
164 /// @brief Adds a listener called on the owner after each successful connect to the server, the first included
165 /// (spec 8 §7, "Reconnect"). A shell registers `Session::reconnected` here, so the session re-sends its
166 /// watch set and re-reads what it shows after an outage. The source owns the backend's single connect
167 /// hook, so anything else that wants connects subscribes here too.
168 ///
169 /// The default does nothing: a local source has no connection to lose.
170 /// @param listener Called on the owner after each connect; it must stay callable while the source is open.
171 virtual void onConnected(std::function<void()> const& listener) { static_cast<void>(listener); }
172};

210/// @brief Why no frontend was selected.
211struct FrontendError {
212 /// @brief What went wrong.
213 enum class Kind : std::uint8_t {
214 MissingName, ///< A `--ui` without a name.
215 UnknownName, ///< `--ui` or `MORPH_UI` names a frontend this binary was not built with.
216 NoneUsable, ///< Nothing was named, and no built frontend is usable here.
217 NotMade, ///< The chosen option has no `make`, or its `make` returned null.
218 };
219 /// @brief What went wrong.
220 Kind kind = Kind::NoneUsable;
221 /// @brief The name asked for, or the option's name for `NotMade`; empty otherwise.
222 std::string requested;
223 /// @brief The names of the frontends this binary was built with, in order.
224 std::vector<std::string> built;
225
226 /// @brief A message for the user, naming the built frontends.
227 /// @return The message.
228 [[nodiscard]] std::string message() const {
229 std::string names;
230 for (std::string const& name : built) {
231 names += names.empty() ? name : ", " + name;
232 }
233 if (names.empty()) {
234 names = "none";
235 }
236 switch (kind) {
237 case Kind::MissingName:
238 return "morph::ui: --ui needs a frontend name (built: " + names + ")";
239 case Kind::UnknownName:
240 return "morph::ui: no frontend named '" + requested + "' is built (built: " + names + ")";
241 case Kind::NotMade:
242 return "morph::ui: frontend '" + requested + "' could not be made";
243 case Kind::NoneUsable:
244 default:
245 return "morph::ui: no built frontend is usable here (built: " + names + ")";
246 }
247 }
248};

250/// @brief Reads an environment variable: its value, or `nullopt` when it is unset.
251using EnvironmentReader = std::function<std::optional<std::string>(std::string_view name)>;
252
253/// @brief The process environment, read through `std::getenv`.
254/// @return A reader over the process environment.
255[[nodiscard]] inline EnvironmentReader processEnvironment() {
256 return [](std::string_view name) -> std::optional<std::string> {
257 std::string const key{name};
258 // getenv races only with a concurrent setenv; frontend selection runs in main before any thread exists.
259 // MSVC deprecates getenv (C4996) in favour of _dupenv_s; a consumer built with /WX must not fail on it.
260 // clang-cl reports the same deprecation as -Wdeprecated-declarations, which the MSVC pragma does not reach.
261 // The MSVC pragma stays on the line just before the call, the only line it covers.
262#if defined(__clang__) && defined(_WIN32)
263#pragma clang diagnostic push
264#pragma clang diagnostic ignored "-Wdeprecated-declarations"
265#endif
266#ifdef _MSC_VER
267#pragma warning(suppress : 4996)
268#endif
269 char const* const value = std::getenv(key.c_str()); // NOLINT(concurrency-mt-unsafe)
270#if defined(__clang__) && defined(_WIN32)
271#pragma clang diagnostic pop
272#endif
273 if (value == nullptr) {
274 return std::nullopt;
275 }
276 return std::string{value};
277 };

282/// @brief The command line's choice of frontend, and the arguments that made it.
283struct CommandLineChoice {
284 /// @brief The last name given, or none.
285 std::optional<std::string> name;
286 /// @brief Whether a `--ui` was given without a name after it, or the last name given is empty.
287 bool missingName = false;
288 /// @brief The positions in the arguments of every `--ui=<name>`, `--ui` and the name after it, ascending.
289 std::vector<std::size_t> used;
290};
291
292/// @brief Reads `--ui=<name>` and `--ui <name>` from the arguments, up to a `--`.
293///
294/// The last one wins. An argument starting with `--` is never taken as the name after `--ui`: it is the next option,
295/// and the `--ui` before it has no name.
296/// @param args The arguments; a program name among them is never a `--ui` flag.
297/// @return The choice.
298[[nodiscard]] inline CommandLineChoice readCommandLine(std::span<std::string const> args) {
299 CommandLineChoice choice;
300 bool nameFollows = false;
301 for (std::size_t index = 0; std::string const& raw : args) {
302 std::string_view const arg = raw;
303 if (nameFollows) {
304 nameFollows = false;
305 if (arg.starts_with("--")) {
306 choice.missingName = true;
307 break;
308 }
309 choice.name = std::string{arg};
310 choice.used.push_back(index);
311 } else if (arg == "--") {
312 break;
313 } else if (arg == "--ui") {
314 nameFollows = true;
315 choice.used.push_back(index);
316 } else if (arg.starts_with("--ui=")) {
317 choice.name = std::string{arg.substr(std::string_view{"--ui="}.size())};
318 choice.used.push_back(index);
319 }
320 ++index;
321 }
322 choice.missingName = choice.missingName || nameFollows || (choice.name.has_value() && choice.name->empty());
323 return choice;
324}

328/// @brief Picks the frontend to run: the last `--ui=<name>` (or `--ui <name>`) before a `--`, else a non-empty
329/// `MORPH_UI`, else the first option in the given order whose `usable()` holds.
330///
331/// A frontend named on the command line or in `MORPH_UI` is used even when its `usable()` is false: the user asked
332/// for it. On success every `--ui` argument (with the name after a bare `--ui`) is removed from @p args, and the rest
333/// keep their order, for the caller to pass on; on an error @p args is left as it was.
334/// @param built The frontends this binary was built with, in order of preference.
335/// @param args The command-line arguments, with or without the program name first.
336/// @param env Reads `MORPH_UI`; the process environment by default. An empty reader reads nothing.
337/// @return The frontend, made; or why there is none.
338[[nodiscard]] inline std::expected<std::unique_ptr<Frontend>, FrontendError> selectFrontend(
339 std::span<FrontendOption const> built, std::vector<std::string>& args,
340 EnvironmentReader const& env = processEnvironment()) {
341 std::vector<std::string> names;
342 names.reserve(built.size());
343 for (FrontendOption const& option : built) {
344 names.push_back(option.name);
345 }
346 auto const failure = [&names](FrontendError::Kind kind, std::string requested) {
347 return std::unexpected{FrontendError{.kind = kind, .requested = std::move(requested), .built = names}};
348 };
349 detail::CommandLineChoice const choice = detail::readCommandLine(args);
350 if (choice.missingName) {
351 return failure(FrontendError::Kind::MissingName, {});
352 }
353 std::optional<std::string> requested = choice.name;
354 if (!requested.has_value() && env) {
355 if (std::optional<std::string> fromEnvironment = env("MORPH_UI");
356 fromEnvironment.has_value() && !fromEnvironment->empty()) {
357 requested = std::move(fromEnvironment);
358 }
359 }
360 FrontendOption const* chosen = nullptr;
361 if (requested.has_value()) {
362 auto const found = std::ranges::find(built, *requested, &FrontendOption::name);
363 if (found == built.end()) {
364 return failure(FrontendError::Kind::UnknownName, *requested);
365 }
366 chosen = &*found;
367 } else {
368 auto const found = std::ranges::find_if(
369 built, [](FrontendOption const& candidate) { return !candidate.usable || candidate.usable(); });
370 if (found == built.end()) {
371 return failure(FrontendError::Kind::NoneUsable, {});
372 }
373 chosen = &*found;
374 }
375 std::unique_ptr<Frontend> frontend = chosen->make ? chosen->make() : nullptr;
376 if (frontend == nullptr) {
377 return failure(FrontendError::Kind::NotMade, chosen->name);
378 }
379 for (std::size_t const index : choice.used | std::views::reverse) {
380 args.erase(args.begin() + static_cast<std::ptrdiff_t>(index));
381 }
382 return frontend;
383}

403/// @brief Mounts the shell for what the source delivered. Called once, on the owner; it may return null to mount
404/// nothing (after calling `AppContext::quit`, for example).
405using ShellFactory =
406 std::function<std::unique_ptr<MountedShell>(AppContext& context, std::expected<Bundle, ConnectError> opened)>;
407
408/// @brief Runs one source inside a frontend: opens it, mounts the shell when it is ready, runs the loop, and tears
409/// down in order.
410///
411/// The order is fixed here so that every frontend keeps it. When the loop returns, or throws, the shell is destroyed
412/// first, while the source's connections that its screens use are still open, and the source is closed next, while
413/// the runtime and the executor its handlers were made with are still alive; the frontend destroys those after this
414/// returns.
415///
416/// The source's `ready` may come during `open` (a local source) or later from the loop (a remote one). A second
417/// `ready` is ignored, and one that comes after the run has ended does nothing. A @p mountShell that throws quits the
418/// context with exit code 1, and its exception leaves this call once everything is torn down.
419/// @param context The frontend's context; its runtime must outlive this call.
420/// @param source The source to open; closed before this returns.
421/// @param mountShell Mounts the shell for what the source delivered.
422/// @param loop Runs the frontend's event loop and returns the exit code once `AppContext::quit` ends it.
423/// @return What @p loop returns.
424/// @throws std::invalid_argument when @p mountShell or @p loop is empty; whatever `open`, @p mountShell, @p loop or
425/// `close` throws, the first of them.
426[[nodiscard]] inline int runApp(AppContext& context, AppSource& source, ShellFactory const& mountShell,
427 std::function<int()> const& loop) {
428 if (!mountShell || !loop) {
429 throw std::invalid_argument{"morph::ui::runApp: the shell factory and the loop must not be empty"};
430 }
431 struct Run {
432 std::unique_ptr<MountedShell> shell;
433 bool opened = false;
434 std::exception_ptr mountFailure;
435 };
436 auto const run = std::make_shared<Run>();
437 std::exception_ptr failure;
438 int exitCode = 0;
439 try {
440 source.open(context,
441 [weak = std::weak_ptr{run}, &context, &mountShell](std::expected<Bundle, ConnectError> opened) {
442 std::shared_ptr<Run> const live = weak.lock();
443 if (live == nullptr || live->opened) {
444 return;
445 }
446 live->opened = true;
447 try {
448 live->shell = mountShell(context, std::move(opened));
449 } catch (...) {
450 live->mountFailure = std::current_exception();
451 context.quit(1);
452 }
453 });
454 exitCode = loop();
455 } catch (...) {
456 failure = std::current_exception();
457 }
458 // The shell goes first, and the run counts as opened from here on, so a `ready` that arrives while the source
459 // closes mounts nothing; one that arrives after this returns finds the run gone.
460 std::exception_ptr const mountFailure = run->mountFailure;
461 run->shell.reset();
462 run->opened = true;
463 try {
464 source.close();
465 } catch (...) {
466 if (!failure) {
467 failure = std::current_exception();
468 }
469 }
470 if (!failure) {
471 failure = mountFailure;
472 }
473 if (failure) {
474 std::rethrow_exception(failure);
475 }
476 return exitCode;
477}
