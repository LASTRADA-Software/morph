24/// @brief Mounts the application's shell for what the source delivered, on the window's widgets.

26/// `ui::ShellFactory` with the backend the shell's views mount on, since `ui::AppContext` names no renderer. Called

31/// @brief What a `qt_quick::Frontend` runs with.
32struct FrontendConfig {
33 /// @brief Mounts the shell when the source is ready; it must not be empty.
34 ShellFactory mountShell;
35 /// @brief The restyle map and the custom components the backend uses; null uses none. Borrowed: it outlives
36 /// the run.
37 Components* components = nullptr;
38 /// @brief The window's title.
39 std::string title = "morph";
40 /// @brief Called with the window's width class (`widthClassOf`) when the window first has a width and each time
41 /// the class changes; empty calls nothing. The shell sets it as `{"env": "widthClass"}`.
42 std::function<void(std::string const& widthClass)> onWidthClass;
43 /// @brief Development mode (spec 6 §21): with `On`, Ctrl+Shift+I in the window opens the inspector over the screen
44 /// the shell shows (`ui::MountedShell::inspectFocused`). With `Off`, the key does nothing.
45 ui::DevelopmentTools devTools = ui::DevelopmentTools::Off;
46};

54/// @brief Runs a morph application in a Qt Quick window, on the calling thread.
55///
56/// `run` makes, in this order, the `QGuiApplication` (unless the process has one already), a `qt::QtExecutor` (the
57/// reactive runtime's owner and every bridge handler's callback executor), the runtime, a `QtScheduler`, the QML
58/// engine, the window and the backend, then runs the source through `ui::runApp`: it opens the source, mounts the
59/// shell through `FrontendConfig::mountShell` when the source is ready, and runs the event loop until
60/// `AppContext::quit()` or the window closes. Teardown is the reverse: the screens, then the source's connections,
61/// then the runtime, then Qt. The style is Qt Quick Controls' `Basic`; on the `offscreen` platform the window
62/// renders in software.
63class Frontend final : public ui::Frontend {
64public:
65 /// @brief A frontend that runs with the process's command line.
66 /// @param argc The argument count; `QGuiApplication` keeps a reference to it.
67 /// @param argv The arguments.
68 /// @param config The shell factory, the components and the window title.
69 Frontend(int& argc MORPH_LIFETIMEBOUND, char** argv, FrontendConfig config);
70
71 /// @brief The name `--ui=` and `MORPH_UI` select this frontend by.
72 /// @return `"qt"`.
73 [[nodiscard]] std::string_view name() const override;
74
75 /// @brief Runs the application the source opens until it quits or its window closes.
76 /// @param source Opened once the runtime exists, and closed before it is destroyed.
77 /// @return The exit code given to `AppContext::quit()`; 0 when the window was closed.
78 /// @throws std::invalid_argument when `FrontendConfig::mountShell` is empty; whatever the source or the shell
79 /// factory throws.
80 int run(ui::AppSource& source) override;
81
82private:
83 int* _argc;
84 char** _argv;
85 FrontendConfig _config;
86};
