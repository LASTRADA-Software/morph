63/// @brief Keeps the route of the screen a shell shows in a history, and opens the route of an entry the user moves
64/// to (spec 3 §10).
65///
66/// The shell opens `initial()` first, when it can, and calls `shown` after it mounts each screen. The first `shown`
67/// replaces the entry the history started on, so opening the first screen does not add a step back to an empty page;
68/// each later call adds an entry, unless the route is already the current entry's. Moving back or forward calls
69/// `open`, which shows the route's screen, or the start screen when the route cannot be opened, before it returns.
70/// The first screen it shows replaces the entry instead of adding one, so a refused route is not left in the history;
71/// a screen shown after that, such as a redirect, adds an entry as it would anywhere else.
72class RouteHistory {
73public:
74 /// @brief A history over @p history, which it keeps and follows from now on.
75 /// @param history The history. Borrowed: it outlives this.
76 /// @param open Shows the screen of a route the user moved to, calling `shown` before it returns. It may destroy
77 /// this.
78 RouteHistory(IHistory& history, std::function<void(std::string const& route)> open)
79 : _history{&history}, _open{std::move(open)}, _initial{history.current()} {
80 _history->onPop([this, alive = std::weak_ptr<bool const>{_alive}](std::string const& route) {
81 if (alive.expired()) {
82 return;
83 }
84 bool const outer = std::exchange(_replaceNext, true);
85 auto const restore = [&] {
86 if (!alive.expired()) {
87 _replaceNext = outer;
88 }
89 };
90 try {
91 _open(route);
92 } catch (...) {
93 restore();
94 throw;
95 }
96 restore();
97 });
98 }
99
100 RouteHistory(RouteHistory const&) = delete;
101 RouteHistory& operator=(RouteHistory const&) = delete;
102 RouteHistory(RouteHistory&&) = delete;
103 RouteHistory& operator=(RouteHistory&&) = delete;
104
105 // The history keeps its callback, which does nothing once this is gone: clearing it here could clear the
106 // callback of a newer route history over the same history.
107 ~RouteHistory() = default;
108
109 /// @brief The route the history held when this was made: a reloaded page's, or a shared link's.
110 /// @return It, or empty for none.
111 [[nodiscard]] std::string const& initial() const noexcept { return _initial; }
112
113 /// @brief The shell now shows a screen whose route is @p route.
114 /// @param route The route; empty, for a screen with none, leaves the history as it is, so after a move back or
115 /// forward the next screen with a route replaces the moved-to entry.
116 void shown(std::string const& route) {
117 if (route.empty()) {
118 return;
119 }
120 bool const replace = std::exchange(_replaceNext, false);
121 if (route == _history->current()) {
122 return;
123 }
124 if (replace) {
125 _history->replace(route);
126 } else {
127 _history->push(route);
128 }
129 }
130
131private:
132 IHistory* _history;
133 std::function<void(std::string const&)> _open;
134 std::string _initial;
135 bool _replaceNext = true;
136 // Expires with this, so a pop whose `open` destroyed it does not touch it again.
137 std::shared_ptr<bool const> _alive = std::make_shared<bool const>(true);
138};
