20/// @file
21/// @brief `morph::ui::ShellPartsBinding`: joins an application's declared shell parts, read by a `ShellPartsModel`,
22/// to the `ShellFrame` that draws them, and carries the frame's choices back.
23///
24/// Specified in `docs/spec/ui/shell_parts.md`.
25
26namespace morph::ui {

28/// @brief What a binding asks of its client.
29struct ShellBindingOptions {
30 /// @brief What the model asks; `confirmDiscard` is the binding's own (a dialog of the frame) and is ignored here.
31 ShellModelOptions model;
32 /// @brief Opens a screen as a tab: a chosen hit, or the inbox.
33 std::function<void(ShellPartsModel::Target const&)> open;
34 /// @brief Opens several hits as tabs, in order, and leaves the first one the active tab (Shift Enter). Empty:
35 /// Shift Enter opens nothing.
36 std::function<void(std::vector<ShellPartsModel::Target> const&)> openAll;
37 /// @brief The person's menu changed the language, after the session took it.
38 std::function<void()> localeChosen;
39 /// @brief Called each time the switcher's readiness changes, and once at the start.
40 std::function<void(bool ready)> contextReady;
41};

43/// @brief The declared parts of a shell, read and drawn.
44///
45/// It makes the model, the effects that follow it and the frame's handlers, and takes them away in its destructor.
46/// Neither copyable nor movable; it keeps its own copy of the parts, and borrows the session, the app root and the
47/// frame, and must go before any of them.
48class ShellPartsBinding {
49public:
50 /// @brief The binding.
51 /// @param session The session. Borrowed.
52 /// @param root The app root whose queries the parts are. Borrowed.
53 /// @param parts The parts; copied.
54 /// @param frame The frame that draws them. Borrowed.
55 /// @param options What it asks of its client.
56 ShellPartsBinding(Session& session MORPH_LIFETIMEBOUND, AppRoot& root MORPH_LIFETIMEBOUND,
57 ShellParts const& parts MORPH_LIFETIMEBOUND, ShellFrame& frame MORPH_LIFETIMEBOUND,
58 ShellBindingOptions options)
59 : _session{&session}, _frame{&frame}, _parts{parts}, _options{std::move(options)} {
60 ShellModelOptions modelOptions = _options.model;
61 modelOptions.confirmDiscard = [this](std::size_t tabs, std::function<void(bool)> done) {
62 ask(tabs, std::move(done));
63 };
64 _model = std::make_unique<ShellPartsModel>(session, root, _parts, std::move(modelOptions));
65 ShellPartsModel* const model = _model.get();
66 ShellPartsHandlers handlers;
67 handlers.onContext = [model](std::size_t index) { model->chooseContext(index); };
68 handlers.onSearchText = [model](std::string const& text) { model->setSearchText(text); };
69 handlers.onSearchOpen = [this, model](std::size_t source, std::size_t hit) {
70 if (auto const target = model->hit(source, hit); target && _options.open) {
71 _options.open(*target);
72 }
73 };
74 handlers.onSearchOpenAll = [this, model] {
75 if (_options.openAll) {
76 _options.openAll(model->allHits());
77 }
78 };
79 handlers.onInbox = [this, model] {
80 std::string const screen = model->inboxScreen();
81 if (!screen.empty() && _options.open) {
82 _options.open(ShellPartsModel::Target{.screen = screen, .params = {}});
83 }
84 };
85 handlers.onLocale = [this, model](std::size_t index) {
86 model->chooseLocale(index);
87 if (_options.localeChosen) {
88 _options.localeChosen();
89 }
90 };
91 handlers.onFormat = [model](std::size_t index) { model->chooseFormat(index); };
92 handlers.onHelp = [this, model] {
93 if (auto const target = model->helpScreen()) {
94 if (_options.open) {
95 _options.open(*target);
96 }
97 } else {
98 static_cast<void>(model->runHelp());
99 }
100 };
101 _frame->setPartsHandlers(std::move(handlers));
102 if (!_parts.modules.empty()) {
103 _frame->navigator().setModuleGroups(true);
104 }
105 reactive::Runtime& runtime = session.runtime();
106 _view = std::make_unique<reactive::Effect>(runtime, [this, &runtime] {
107 ShellPartsView view = _model->view();
108 runtime.untracked([this, &view] { _frame->setParts(std::move(view)); });
109 });
110 _counts = std::make_unique<reactive::Effect>(runtime, [this, &runtime] {
111 auto counts = _model->moduleCounts();
112 ShellLabels const labels = shellLabels(_session->formatting());
113 runtime.untracked([this, &counts, &labels] { showCounts(counts, labels); });
114 });
115 _ready = std::make_unique<reactive::Effect>(runtime, [this, &runtime] {
116 bool const ready = _model->contextReady();
117 runtime.untracked([this, ready] {
118 if (_options.contextReady) {
119 _options.contextReady(ready);
120 }
121 });
122 });
123 }
124
125 ShellPartsBinding(ShellPartsBinding const&) = delete;
126 ShellPartsBinding& operator=(ShellPartsBinding const&) = delete;
127 ShellPartsBinding(ShellPartsBinding&&) = delete;
128 ShellPartsBinding& operator=(ShellPartsBinding&&) = delete;
129
130 /// @brief Stops following and takes the frame's handlers away.
131 ~ShellPartsBinding() {
132 _ready.reset();
133 _counts.reset();
134 _view.reset();
135 _frame->setPartsHandlers({});
136 _frame->setParts(ShellPartsView{});
137 if (!_parts.modules.empty()) {
138 Navigator& navigator = _frame->navigator();
139 for (ModuleCount const& declared : _parts.modules) {
140 navigator.setModuleBadge(declared.module, std::nullopt);
141 }
142 navigator.setModuleGroups(false);
143 }
144 _model.reset();
145 }
146
147 /// @brief The model.
148 /// @return It.
149 [[nodiscard]] ShellPartsModel& model() noexcept { return *_model; }
150
151 /// @brief Tells the frame the client's connection state, which its foot draws.
152 /// @param state The state.
153 void setConnection(ConnectionState state) { _model->setConnection(state); }
154
155 /// @brief Whether the switcher has answered, failed, or none is declared. A tracked read.
156 /// @return Whether the start screen may mount.
157 [[nodiscard]] bool contextReady() const { return _model->contextReady(); }
158
159private:
160 // Asks once whether to discard the unsaved work of @p tabs tabs.
161 void ask(std::size_t tabs, std::function<void(bool)> done) {
162 FormatContext const& format = _session->formattingNow();
163 std::string message = shellText("morph.shell.contextDirty", format);
164 if (auto const at = message.find("{n}"); at != std::string::npos) {
165 message.replace(at, 3, std::to_string(tabs));
166 }
167 _frame->confirm(message, shellText("morph.shell.contextDiscard", format),
168 shellText("morph.shell.contextCancel", format), std::move(done));
169 }
170
171 // Puts each module's count on its navigator row, and the words its accessible name carries.
172 void showCounts(std::vector<ShellPartsModel::ModuleState> const& counts, ShellLabels const& labels) {
173 Navigator& navigator = _frame->navigator();
174 FormatContext const& format = _session->formattingNow();
175 navigator.setModuleTexts(shellText("morph.shell.moduleClosed", format),
176 shellText("morph.shell.moduleOpen", format), labels.countFailed);
177 std::set<std::string, std::less<>> shown;
178 for (ShellPartsModel::ModuleState const& state : counts) {
179 shown.insert(state.module);
180 std::optional<NavigatorBadge> badge;
181 if (state.count && *state.count > 0) {
182 badge = NavigatorBadge{.text = shellCountText(*state.count), .tone = Tone::Neutral};
183 }
184 // The model leaves out every module the catalog does not list, so a module the navigator then does not
185 // list means the two disagree about the catalog: reported, once per module.
186 if (!navigator.setModuleBadge(state.module, badge, state.failed) && _options.model.onProblem &&
187 _unlisted.insert(state.module).second) {
188 _options.model.onProblem("the shell's module count names " + state.module +
189 ", which the catalog lists and the navigator does not");
190 }
191 }
192 for (ModuleCount const& declared : _parts.modules) {
193 if (!shown.contains(declared.module)) {
194 navigator.setModuleBadge(declared.module, std::nullopt);
195 }
196 }
197 }
198
199 Session* _session;
200 ShellFrame* _frame;
201 ShellParts _parts;
202 ShellBindingOptions _options;
203 std::unique_ptr<ShellPartsModel> _model;
204 std::set<std::string, std::less<>> _unlisted;
205 std::unique_ptr<reactive::Effect> _view;
206 std::unique_ptr<reactive::Effect> _counts;
207 std::unique_ptr<reactive::Effect> _ready;
208};
