23/// @brief Three-way answer to "may this callback still be delivered?".
24///
25/// Liveness and stop are *different* states and are kept distinguishable:
26/// "the owner is gone" and "the owner is still here but cancelled this" have
27/// the same effect on delivery but very different meanings for diagnostics,
28/// logging and tests. `CallbackToken::status()` reports which one applies.
29enum class CallbackStatus : std::uint8_t {
30 /// The owning `CallbackScope` still exists and has not been stopped —
31 /// callbacks gated on this token run.
32 Active,
33 /// The owning `CallbackScope` still exists but `requestStop()` was called
34 /// on this generation — the owner is alive and no longer interested.
35 Stopped,
36 /// The owning `CallbackScope` was destroyed, or `reset()` retired the
37 /// generation this token was issued from.
38 Expired,
39};
40
41/// @brief Weak, copyable observer of a `CallbackScope` — the half a callback captures.
42///
43/// A token keeps nothing alive. It answers whether the scope that issued it is
44/// still both present and interested, and it can wrap an arbitrary callable so
45/// that the callable no-ops once the answer turns negative.
46///
47/// A default-constructed token is permanently `CallbackStatus::Expired`:
48/// gating is fail-closed, so a token that was never bound to a scope suppresses
49/// rather than admits. That trades a silently-dropped callback for never
50/// dereferencing a freed receiver, which is the trade this whole type exists to
51/// make.
52///
53/// @par Thread safety
54/// Every member is safe to call from any thread, concurrently with the
55/// scope's `requestStop()` / `reset()` / destruction. What the *result* is worth
56/// across threads is a separate question — see `CallbackScope`'s
57/// "Boundary of the guarantee".
58class CallbackToken {
59public:
60 /// @brief Constructs an unbound token, permanently `CallbackStatus::Expired`.
61 CallbackToken() = default;
62
63 /// @brief Reports whether the issuing scope is alive, stopped, or gone.
64 /// @return `Active`, `Stopped` or `Expired` — see `CallbackStatus`.
65 [[nodiscard]] CallbackStatus status() const noexcept {
66 auto const state = _state.lock();
67 if (state == nullptr) {
68 return CallbackStatus::Expired;
69 }
70 return state->stopped.load(std::memory_order_acquire) ? CallbackStatus::Stopped : CallbackStatus::Active;
71 }
72
73 /// @brief Whether a callback gated on this token may run right now.
74 ///
75 /// Advisory across threads: true here means "was active at the moment of
76 /// the check". See `CallbackScope`'s "Boundary of the guarantee".
77 /// @return `true` iff `status() == CallbackStatus::Active`.
78 [[nodiscard]] bool active() const noexcept { return status() == CallbackStatus::Active; }
79
80 /// @brief Whether the issuing scope generation is gone (destroyed or `reset()`).
81 ///
82 /// Cheaper than `status()` — no strong reference is taken — and unlike
83 /// `active()` it does not conflate "gone" with "stopped".
84 /// @return `true` iff the issuing generation no longer exists.
85 [[nodiscard]] bool expired() const noexcept { return _state.expired(); }
86
87 /// @brief Wraps @p fn so it runs only while this token is `Active`.
88 ///
89 /// The returned callable forwards every argument to @p fn and returns
90 /// nothing. When the token is not `Active` at invocation time it does
91 /// nothing at all — @p fn is not called. The wrapper holds a *copy of the
92 /// token*, never a strong reference, so wrapping a callback cannot keep the
93 /// receiver's scope alive.
94 ///
95 /// A suppressed callback is destroyed, not leaked: the wrapper (and with it
96 /// @p fn and its captures) is released wherever the refused task is
97 /// discarded — normally on the delivery executor's thread.
98 ///
99 /// @tparam F Callable type to wrap. Must return `void` for every argument
100 /// list it is invoked with; a value-returning callable has no
101 /// defensible answer for the suppressed case and is rejected at
102 /// compile time.
103 /// @param fn Callable to gate. Moved into the returned wrapper.
104 /// @return A callable with @p fn's argument list and a `void` return.
105 template <typename F>
106 [[nodiscard]] auto guard(F&& fn) const {
107 return [token = *this, fn = std::forward<F>(fn)](auto&&... args) mutable -> void {
108 static_assert(std::is_void_v<std::invoke_result_t<F&, decltype(args)...>>,
109 "CallbackToken::guard() only wraps void-returning callables: there is no correct value to "
110 "return when delivery is suppressed.");
111 if (token.status() != CallbackStatus::Active) {
112 return;
113 }
114 std::invoke(fn, std::forward<decltype(args)>(args)...);
115 };
116 }
117
118private:
119 friend class CallbackScope;
120
121 explicit CallbackToken(std::weak_ptr<detail::CallbackScopeState> state) noexcept : _state{std::move(state)} {}
122
123 std::weak_ptr<detail::CallbackScopeState> _state;
124};

126/// @brief Owned lifetime-and-stop gate for asynchronous callbacks — a data member, not a base class.
127///
128/// A receiver that attaches callbacks to a `Completion<T>`, a `BridgeHandler`
129/// subscription, a timer tick or any other posted closure holds one of these as
130/// an ordinary member and hands out `CallbackToken`s. A gated callback runs only
131/// while the scope is **both alive and not stopped**.
132///
133/// Composition is the point. Requiring every consumer to derive from a framework
134/// base class constrains hierarchies that are already `QObject`s, already have a
135/// base, or are aggregates — for what is an implementation detail of one screen.
136/// A member composes with all of them, and code that never opts in is unaffected.
137///
138/// @code
139/// class BoardPresenter {
140/// void load() {
141/// _handler.execute(GetBoard{}).then(_callbacks, [this](GetBoardResult r) { render(r); });
142/// }
143/// void onUserNavigatedAway() { _callbacks.requestStop(); } // alive, but no longer interested
144/// void onNewQuery() { _callbacks.reset(); } // supersede: old replies dead, new ones live
145///
146/// morph::async::CallbackScope _callbacks; // declared LAST — see below
147/// };
148/// @endcode
149///
150/// @par Three verbs, two states
151/// - `requestStop()` — "I no longer care". The owner is still here; pending and
152/// future callbacks on this generation are refused. Idempotent.
153/// - `reset()` — "supersede". Every token issued so far goes permanently dead
154/// and the scope becomes deliverable again under a fresh generation, so
155/// "a new query cancels the old one" needs no heap-reallocated scope.
156/// - destruction — the same refusal as `requestStop()`, plus the owner is gone.
157///
158/// Destruction and `requestStop()` are indistinguishable *to a callback*; they
159/// differ only in whether the owner still exists, which `CallbackToken::status()`
160/// reports as `Expired` versus `Stopped`.
161///
162/// @par Declared last, destroyed first
163/// Put the member **after** (below) everything its callbacks touch. Members are
164/// destroyed in reverse declaration order, so a last-declared scope is the first
165/// thing to die and every gated callback is already refused before the fields it
166/// would have touched are torn down. This replaces the per-class "token must
167/// stay last-declared" convention that used to be re-derived and re-documented
168/// in every class that hand-rolled the pattern.
169///
170/// @par Teardown that pumps
171/// Members are destroyed *after* the destructor body runs. A destructor body
172/// that can pump a nested event loop (a `sendSync`-style blocking call) can
173/// therefore still deliver into a half-destroyed receiver. Such a destructor
174/// must call `requestStop()` as its first statement.
175///
176/// @par Boundary of the guarantee
177/// - **Executor-affine use gets the full guarantee.** When the scope is
178/// destroyed / stopped / reset on the *same* thread the delivery executor runs
179/// on — the normal case, where receiver and callbacks both live on the GUI
180/// thread — check-then-run is atomic with respect to those operations, and a
181/// gated callback never touches a dead or stopped receiver.
182/// - **Cross-thread stop is advisory.** A scope torn down on a different thread
183/// from the delivery executor can turn inactive between the token's check and
184/// the callback body. That is exactly the boundary of the hand-rolled
185/// `weak_ptr` idiom this type replaces — locking the token pins the *token*,
186/// never the receiver — and external synchronisation there remains the
187/// caller's job.
188/// - **Deliberately no block-until-drained.** Neither `requestStop()` nor the
189/// destructor waits for an in-flight callback to finish, `QObject::disconnect`
190/// style. A GUI-thread destructor blocking on a pool-thread callback that is
191/// itself blocked posting back to the GUI executor is a deadlock by
192/// construction.
193///
194/// @par Thread safety
195/// All members are safe to call concurrently from any thread. `requestStop()`
196/// and `stopRequested()` are lock-free atomic operations; `reset()` publishes a
197/// fresh generation and retires the previous one (stopping it first, so a token
198/// holder that raced the swap and pinned the old state still observes refusal).
199///
200/// Identity, not a value: neither copyable nor movable. A moved-from scope would
201/// have to either strand or silently retarget tokens already captured in flight;
202/// `reset()` covers the one case (regeneration) that motivates a move at all.
203class CallbackScope {
204public:
205 /// @brief Constructs a live, un-stopped scope.
206 CallbackScope() : _state{std::make_shared<detail::CallbackScopeState>()} {}
207
208 /// @brief Refuses every outstanding token, then releases the state.
209 ///
210 /// Equivalent, from a pending callback's point of view, to `requestStop()`.
211 /// Does not wait for in-flight callbacks — see "Boundary of the guarantee".
212 ~CallbackScope() { requestStop(); }
213
214 CallbackScope(const CallbackScope&) = delete;
215 CallbackScope& operator=(const CallbackScope&) = delete;
216 CallbackScope(CallbackScope&&) = delete;
217 CallbackScope& operator=(CallbackScope&&) = delete;
218
219 /// @brief Marks this generation stopped: gated callbacks stop being delivered.
220 ///
221 /// Idempotent and safe from any thread. The owner remains alive, so tokens
222 /// report `CallbackStatus::Stopped` rather than `Expired`. Undone only by
223 /// `reset()`, which starts a new generation.
224 void requestStop() noexcept {
225 if (_state != nullptr) {
226 _state->stopped.store(true, std::memory_order_release);
227 }
228 }
229
230 /// @brief Retires every token issued so far and starts a fresh, live generation.
231 ///
232 /// The supersede verb: an old in-flight reply can no longer be delivered
233 /// (its token reports `Expired`), while callbacks attached *after* this call
234 /// are deliverable again even if `requestStop()` had been called before it.
235 /// The outgoing generation is stopped before it is released, so a token
236 /// holder that pinned it while racing this call still observes refusal.
237 void reset() {
238 auto fresh = std::make_shared<detail::CallbackScopeState>();
239 requestStop();
240 _state = std::move(fresh);
241 }
242
243 /// @brief Whether this generation has been stopped.
244 /// @return `true` after `requestStop()`, until the next `reset()`.
245 [[nodiscard]] bool stopRequested() const noexcept {
246 return _state != nullptr && _state->stopped.load(std::memory_order_acquire);
247 }
248
249 /// @brief Issues a weak token for the current generation.
250 /// @return A `CallbackToken` observing this scope; keeps nothing alive.
251 [[nodiscard]] CallbackToken token() const noexcept { return CallbackToken{_state}; }
252
253 /// @brief Wraps @p fn so it runs only while this scope is alive and un-stopped.
254 ///
255 /// Forwards to `CallbackToken::guard()` on a token for the *current*
256 /// generation, so a later `reset()` retires the wrapper. This is the
257 /// general-purpose form for the callbacks that are not `Completion`
258 /// attachments — timer ticks, `IExecutor::post` closures, poller dispatch,
259 /// event sinks.
260 ///
261 /// @tparam F Callable type to wrap; must return `void` (see `CallbackToken::guard()`).
262 /// @param fn Callable to gate. Moved into the returned wrapper.
263 /// @return A callable with @p fn's argument list and a `void` return.
264 template <typename F>
265 [[nodiscard]] auto guard(F&& fn) const {
266 return token().guard(std::forward<F>(fn));
267 }
268
269private:
270 std::shared_ptr<detail::CallbackScopeState> _state;
271};
