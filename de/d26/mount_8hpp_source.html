83/// @brief Which entries of a permutation lie on one longest increasing run.
84///
85/// @p sequence holds, in current order, each widget's target position; the widgets on the run are already in
86/// order relative to each other and stay where they are, and every other one is moved once.
87/// @param sequence A permutation of `0 .. n-1`.
88/// @return Indexed by target position: whether that widget stays.
89[[nodiscard]] inline std::vector<bool> stableTargets(std::vector<std::size_t> const& sequence) {
90 constexpr auto kNone = static_cast<std::size_t>(-1);
91 std::vector<std::size_t> tails; // tails.at(n - 1): where the smallest tail of a run of length n sits
92 std::vector<std::size_t> previous(sequence.size(), kNone);
93 for (std::size_t i = 0; i < sequence.size(); ++i) {
94 auto const place = std::ranges::lower_bound(tails, sequence.at(i), {},
95 [&sequence](std::size_t entry) { return sequence.at(entry); });
96 auto const length = static_cast<std::size_t>(place - tails.begin());
97 if (length > 0) {
98 previous.at(i) = tails.at(length - 1);
99 }
100 if (length == tails.size()) {
101 tails.push_back(i);
102 } else {
103 tails.at(length) = i;
104 }
105 }
106 std::vector<bool> stable(sequence.size(), false);
107 for (std::size_t cursor = tails.empty() ? kNone : tails.back(); cursor != kNone; cursor = previous.at(cursor)) {
108 stable.at(sequence.at(cursor)) = true;
109 }
110 return stable;
111}

113/// @brief Brings a container's children from their current order into the target order with `moveChild`,
114/// moving only the children outside a longest run already in order.
115/// @param container The container.
116/// @param current Its children, in their current order. Kept equal to the container's order after every move, so
117/// it is still that order when a move throws.
118/// @param target The same children, in the order wanted.
119inline void reorderChildren(ContainerWidget& container, std::vector<Widget*>& current,
120 std::vector<Widget*> const& target) {
121 std::unordered_map<Widget const*, std::size_t> targetOf;
122 for (std::size_t i = 0; i < target.size(); ++i) {
123 targetOf.emplace(target.at(i), i);
124 }
125 std::vector<std::size_t> sequence;
126 sequence.reserve(current.size());
127 for (Widget const* const child : current) {
128 sequence.push_back(targetOf.at(child));
129 }
130 std::vector<bool> const stable = stableTargets(sequence);
131 auto const indexOf = [&current](Widget const* child) {
132 return static_cast<std::size_t>(std::ranges::find(current, child) - current.begin());
133 };
134 // From the back: each moved child goes directly before its successor in the target order, which is already
135 // in place, so one move per unstable child suffices.
136 for (std::size_t position = target.size(); position-- > 0;) {
137 if (stable.at(position)) {
138 continue;
139 }
140 Widget* const moving = target.at(position);
141 std::size_t const from = indexOf(moving);
142 std::size_t destination = current.size() - 1;
143 if (position + 1 < target.size()) {
144 std::size_t const before = indexOf(target.at(position + 1));
145 destination = from < before ? before - 1 : before;
146 }
147 if (destination == from) {
148 continue;
149 }
150 container.moveChild(*moving, destination);
151 current.erase(current.begin() + static_cast<std::ptrdiff_t>(from));
152 current.insert(current.begin() + static_cast<std::ptrdiff_t>(destination), moving);
153 }
154}

180 /// @brief Mounts @p node under @p parent; @p scope owns its widgets and bindings.
181 ///
182 /// A node's widget is adopted first, then its bindings, then its children, so destroying the scope destroys
183 /// bindings before their widget and children before their parent.
184 /// @param scope Owns everything mounted.
185 /// @param node The node; null mounts nothing.
186 /// @param parent The container to append to, or null for a root.
187 /// @return The node's widget, or null for a null node.
188 Widget* mount(reactive::Scope& scope, Node const& node, ContainerWidget* parent) {
189 if (!node) {
190 return nullptr;
191 }
192 return std::visit(
193 [&](auto const& spec) -> Widget* {
194 if constexpr (requires { spec.common.layout.alignSelf; }) {
195 checkPlacement(spec.common, parent);
196 }
197 return &this->mountKind(scope, spec, parent);
198 },
199 node->kind);
200 }

212 /// @brief Moves keyboard focus to the mounted widget whose `focusId` is @p focusId.
213 ///
214 /// Of several such widgets (rows of a list), the one mounted first is taken. A backend that cannot move focus
215 /// leaves it where it was; the request was still made.
216 /// @param focusId The name.
217 /// @return Whether a mounted widget has that name.
218 bool focus(std::string_view focusId) {
219 // lower_bound, not find: of equal names, find may answer any; lower_bound answers the first inserted.
220 auto const found = _focusable.lower_bound(focusId);
221 if (found == _focusable.end() || found->first != focusId) {
222 return false;
223 }
224 try {
225 found->second->focus();
226 } catch (...) { // NOLINT(bugprone-empty-catch)
227 // Focus is a request: a backend that cannot move it leaves it where it was.
228 }
229 return true;
230 }

265 // One mount: the root, or content, a page or a row a binding mounts later. The outermost pass focuses the first
266 // autofocus widget mounted during it, in document order, after everything in it has mounted; a pass nested in
267 // another (content a binding mounts while the root mounts) leaves that to the outer one. A pass that throws gives
268 // back the candidate it found, because its caller destroys what it mounted.
269 template <typename F>
270 auto mountPass(F const& body) -> std::invoke_result_t<F const&> {
271 Widget* const before = _autofocus;
272 ++_passDepth;
273 try {
274 auto result = body();
275 --_passDepth;
276 if (_passDepth == 0) {
277 if (Widget* const target = std::exchange(_autofocus, nullptr); target != nullptr) {
278 try {
279 target->focus();
280 } catch (...) { // NOLINT(bugprone-empty-catch)
281 // Focus is a request: a backend that cannot move it leaves it where it was.
282 }
283 }
284 }
285 return result;
286 } catch (...) {
287 _autofocus = before;
288 --_passDepth;
289 throw;
290 }
291 }

293 // A constant is applied once; a binding becomes an equality-gated Computed plus an Effect, made in that order
294 // after the widget, so the Effect dies first. Returns the Computed, or null for a constant.
295 template <typename T, typename Apply>
296 reactive::Computed<T>* bind(reactive::Scope& scope, Prop<T> const& prop, Apply apply) {
297 if (!prop.isBound()) {
298 apply(prop.constant());
299 return nullptr;
300 }
301 auto& value = scope.make<reactive::Computed<T>>(*_rt, prop.binding());
302 scope.effect([&value, apply = std::move(apply)] { apply(value.get()); });
303 return &value;
304 }

306 // Runs a widget callback, and returns whether it completed.
307 // - Inside widgetEvent: one batch, and a flush that would start meanwhile is re-posted, so a remount never
308 // destroys a widget whose native handler is on the stack.
309 // - Untracked: a backend may call a handler from inside a setter, and so from inside the binding Effect that
310 // called it; that Effect must not subscribe to what the handler reads.
311 // - An exception is reported and stops here: the backend that called the handler need not be exception-safe.
312 // Writes made before it still flush when the batch closes.
313 template <typename F>
314 static bool runCallback(reactive::Runtime& runtime, F const& body) {
315 return runtime.widgetEvent([&] {
316 return runtime.untracked([&] {
317 try {
318 body();
319 return true;
320 } catch (...) {
321 runtime.core()->report(site::kCallbackThrew);
322 return false;
323 }
324 });
325 });
326 }

328 template <typename... Args>
329 [[nodiscard]] std::function<void(Args...)> event(std::function<void(Args...)> const& handler) const {
330 if (!handler) {
331 return {};
332 }
333 // The closure owns a copy of the handler and frees it when it dies; the analyzer loses the copy's heap block
334 // in libc++'s std::function and reports it as leaked.
335 // NOLINTNEXTLINE(clang-analyzer-cplusplus.NewDeleteLeaks)
336 return [runtime = _rt, handler](Args... args) {
337 static_cast<void>(runCallback(*runtime, [&] { handler(std::move(args)...); }));
338 };

341 /// @brief A controlled input's re-assertion: what the widget shows as far as the mount knows, what its slot
342 /// says now, and how to show a value.
343 /// @tparam T The value type.
344 template <typename T>
345 struct Reassertion {
346 /// @brief What the widget shows: the last value the mount set, or the last value the user asked for since.
347 /// Owned by the scope, made before this, so it outlives it.
348 std::optional<T>* shown;
349 /// @brief What the slot says now.
350 std::function<T()> read;
351 /// @brief Shows a value, and records it in `*shown`.
352 std::function<void(T const&)> apply;
353 };
354
355 // What a controlled input's event calls after the application's handler, with the value the user asked for.
356 //
357 // The widget shows the request already: it is recorded as shown. Then one turn is posted to the owner; it reads
358 // the slot untracked and shows it when the widget shows anything else. The turn is posted after the event's
359 // widgetEvent closed, so it runs after the flush the event caused, whose bindings record what they set: an
360 // accepted request costs no setter call, a refused one snaps back, and a transformed one, which the binding has
361 // shown already, costs nothing more. The check is owned by @p scope and held weakly by the event and the turn,
362 // so neither does anything once the content is gone.
363 template <typename T, typename Read, typename Apply>
364 [[nodiscard]] std::function<void(T const&)> reasserter(reactive::Scope& scope, std::optional<T>& shown, Read read,
365 Apply apply) {
366 auto const check = std::make_shared<Reassertion<T>>(
367 Reassertion<T>{.shown = &shown, .read = std::move(read), .apply = std::move(apply)});
368 scope.make<std::shared_ptr<Reassertion<T>>>(check);
369 return [runtime = _rt, weak = std::weak_ptr{check}](T const& requested) {
370 if (auto const live = weak.lock()) {
371 *live->shown = requested;
372 } else {
373 return;
374 }
375 // Called from a widget handler, which must not throw: a post that fails leaves the widget showing the
376 // user's value until the slot next changes.
377 try {
378 runtime->owner().post([runtime, weak] {
379 if (auto const live = weak.lock()) {
380 runtime->untracked([&live] { showSlot(*live); });
381 }
382 });
383 } catch (...) { // NOLINT(bugprone-empty-catch): see above
384 }
385 };
386 }
387
388 // Shows the slot's value if the widget shows anything else. A slot that throws is its binding's failure, which
389 // the binding's Effect reports; the widget keeps what the user did.
390 template <typename T>
391 static void showSlot(Reassertion<T> const& check) noexcept {
392 try {
393 T const current = check.read();
394 if (!(check.shown->has_value() && **check.shown == current)) {
395 check.apply(current);
396 }
397 } catch (...) { // NOLINT(bugprone-empty-catch): reported by the binding's Effect, see above
398 }
399 }

401 // A controlled input's value: bound like any property, recording what it shows, and not sending what it already
402 // shows; a slot is also re-asserted after each user event (see `reasserter`). Returns what the event calls, or
403 // nothing for a constant, which is not a slot: the widget keeps what the user did to it.
404 template <typename T, typename Apply>
405 [[nodiscard]] std::function<void(T const&)> controlled(reactive::Scope& scope, Prop<T> const& prop, Apply apply) {
406 auto& shown = scope.make<std::optional<T>>();
407 // A value the widget already shows, the user's accepted edit above all, is not sent again.
408 auto show = [&shown, apply = std::move(apply)](T const& value) {
409 if (shown.has_value() && *shown == value) {
410 return;
411 }
412 apply(value);
413 shown = value;
414 };
415 reactive::Computed<T> const* const value = bind(scope, prop, show);
416 if (value == nullptr) {
417 return {};
418 }
419 return reasserter<T>(scope, shown, [value] { return value->get(); }, std::move(show));
420 }

422 // An input's event: the application's handler, run as `event` runs it, then the re-assertion of the request.
423 // Installed whenever either exists, so a slot without a handler is read-only rather than editable.
424 template <typename T, typename Arg>
425 [[nodiscard]] std::function<void(Arg)> inputEvent(std::function<void(Arg)> const& handler,
426 std::function<void(T const&)> reassert) const {
427 if (!reassert) {
428 return event(handler);
429 }
430 return [inner = event(handler), reassert = std::move(reassert)](Arg arg) {
431 T const requested(arg);
432 if (inner) {
433 inner(std::move(arg));
434 }
435 reassert(requested);
436 };
437 }

439 // The loader's rules for bounds, for a view tree built in C++: no negative bound, no minimum above its maximum,
440 // and no fixed amount outside its bounds, which would leave it no room.
441 static void checkLayout(Common const& common) {
442 LayoutHints const& hints = common.layout;
443 auto const refuse = [&common](std::string const& what) {
444 throw std::logic_error{"morph::ui::mount: " + what + " on the node" +
445 (common.testId.empty() ? std::string{} : " `" + common.testId + "`")};
446 };
447 auto const check = [&refuse](Sizing sizing, std::optional<int> minimum, std::optional<int> maximum,
448 std::string const& axis) {
449 if (minimum.value_or(0) < 0 || maximum.value_or(0) < 0) {
450 refuse("a negative " + axis + " bound");
451 }
452 if (minimum && maximum && *minimum > *maximum) {
453 refuse("a minimum " + axis + " above its maximum");
454 }
455 if (sizing.kind == Sizing::Kind::Fixed &&
456 (sizing.amount < minimum.value_or(sizing.amount) || sizing.amount > maximum.value_or(sizing.amount))) {
457 refuse("a fixed " + axis + " outside its bounds");
458 }
459 };
460 check(hints.width, hints.minWidth, hints.maxWidth, "width");
461 check(hints.height, hints.minHeight, hints.maxHeight, "height");
462 }

477 if (common.visible.isBound() || !common.visible.constant()) {
478 bind(scope, common.visible, [&widget](bool shown) { widget.setVisible(shown); });
479 }
480 if (common.enabled.isBound() || !common.enabled.constant()) {
481 bind(scope, common.enabled, [&widget](bool enabled) { widget.setEnabled(enabled); });
482 }
483 if (common.layout != LayoutHints{}) {
484 checkLayout(common);
485 widget.setLayout(common.layout);
486 }
487 if (common.dragKey.isBound() || common.dragKey.constant().has_value()) {
488 bind(scope, common.dragKey, [&widget](std::optional<Key> const& key) { widget.setDragKey(key); });
489 }
490 if (common.onDrop) {
491 applyDrop(widget, common);
492 }
493 if (common.a11y.name.isBound() || !common.a11y.name.constant().empty()) {
494 bind(scope, common.a11y.name, [&widget](std::string const& name) { widget.setAccessibleName(name); });

496 if (!common.a11y.role.empty()) {
497 widget.setAccessibleRole(common.a11y.role);
498 }
499 if (!common.testId.empty()) {
500 widget.setTestId(common.testId);
501 }
502 if (common.tooltip.isBound() || !common.tooltip.constant().empty()) {
503 bind(scope, common.tooltip, [&widget](std::string const& text) { widget.setTooltip(text); });
504 }
505 if (!common.surface.empty()) {
506 widget.setSurface(common.surface);
507 }
508 if (!common.keys.empty()) {
509 applyKeys(widget, common.keys);

512 _autofocus = &widget;
513 }
514 if (!common.focusId.empty()) {
515 scope.make<Focusable>(_focusable, _focusable.emplace(common.focusId, &widget));

833 // reset() tears the old case down, newest first, before the new one is built. The Computed behind a bound
834 // selector is equality-gated, so an unchanged key never gets here. A case that fails to mount leaves nothing for

853 // Every page is owned by `pageScope`, a child scope made before the binding, so pages are destroyed newest first,
854 // after the binding that shows them and before the tab bar. A scope fixes that order; a vector's element
855 // destruction order differs between standard libraries.
856 //
857 // A bound index is not equality-gated: the Effect reads the binding itself, not a Computed. After a failed page,
858 // the application's onSelect writes the index back inside this Effect's own run; a Computed would keep the failed
859 // index and find the user's next pick of that tab unchanged, while the Effect's dependency on the binding's own
860 // signals sees it as the change it is. The cost is that showPage also runs when something the binding reads
861 // changes but the index does not: for a page that is mounted it only re-sends the highlight, and for a page that
862 // failed it tries the mount again, once per run.
863 Widget& mountKind(reactive::Scope& scope, Tabs const& spec, ContainerWidget* parent) {
864 TabsWidget& widget = scope.adopt(_backend->createTabs(parent));
865 applyCommon(scope, widget, spec.common);
866 std::vector<std::string> labels;
867 labels.reserve(spec.tabs.size());
868 for (Tab const& tab : spec.tabs) {
869 labels.push_back(tab.label);
870 }
871 widget.setTabs(labels);
872 auto& pageScope = scope.child();
873 auto& pages = scope.make<Pages>();
874 pages.pageSlots.resize(spec.tabs.size(), nullptr);
875 // A bound index is controlled: after a pick, the bar highlights the page that is shown, so a pick the
876 // document refuses moves the highlight back. A constant index is not a slot, and the bar keeps the pick.
877 std::function<void(std::size_t const&)> reassert;
878 if (spec.selected.isBound()) {
879 reassert = reasserter<std::size_t>(
880 scope, pages.highlighted,
881 [&pages] { return pages.shown.value_or(std::numeric_limits<std::size_t>::max()); },
882 [&widget, &pages](std::size_t index) { highlight(widget, pages, index); });
883 }
884 widget.setOnSelect(inputEvent(spec.onSelect, reassert));
885 pages.onSelect = inputEvent(spec.onSelect, std::move(reassert));
886 auto show = [this, &widget, &pageScope, &pages, pageSpecs = spec.tabs, overlay = _overlay](std::size_t index) {
887 underOverlay(overlay,
888 [&] { _rt->untracked([&] { showPage(widget, pageScope, pages, pageSpecs, index); }); });
889 };
890 if (!spec.selected.isBound()) {
891 show(spec.selected.constant());
892 } else {
893 scope.effect([binding = spec.selected.binding(), show = std::move(show)] { show(binding()); });
894 }
895 return widget;
896 }

898 // A page is mounted the first time its tab is selected and only hidden afterwards, so its widgets keep their
899 // state (focus, scroll position, half-typed text) across tab switches. A new page is complete before anything
900 // else changes: one that fails to mount leaves the previous page shown and selected, and the bar's highlight,
901 // which the user's pick may already have moved, goes back to it. The next change of the index mounts again.
902 void showPage(TabsWidget& widget, reactive::Scope& pageScope, Pages& pages, std::vector<Tab> const& pageSpecs,
903 std::size_t index) {
904 bool const exists = index < pageSpecs.size();
905 bool const isNew = exists && pages.pageSlots.at(index) == nullptr;
906 if (isNew) {
907 try {
908 mountPage(widget, pageScope, pages, pageSpecs.at(index).node, index);
909 } catch (...) {
910 if (pages.shown.has_value()) {
911 restoreSelection(widget, pages, *pages.shown);
912 }
913 throw;
914 }
915 }
916 if (pages.shown != index) {
917 if (pages.shown.has_value() && *pages.shown < pages.pageSlots.size()) {
918 if (SlotWidget* const previous = pages.pageSlots.at(*pages.shown); previous != nullptr) {
919 previous->setVisible(false);
920 }
921 }
922 if (exists && !isNew) {
923 pages.pageSlots.at(index)->setVisible(true);
924 }
925 }
926 pages.shown = index;
927 highlight(widget, pages, index);
928 }

936 // After a page failed to mount: puts the bar's highlight back on the page still shown, and tells the application
937 // through `onSelect`, so its index matches the widget and selecting the failed tab again is a change that mounts
938 // again.
939 //
940 // onSelect runs last, from a copy: it is a widget callback, which may destroy the Mounted and with it `pages`,
941 // the widget and the Effect that called this. Nothing after it touches any of them; the caller only rethrows.
942 //
943 // Nothing here may replace the failure being reported: the callback's wrapper reports what the callback throws,
944 // and anything else thrown here is dropped.
945 static void restoreSelection(TabsWidget& widget, Pages& pages, std::size_t shown) noexcept {
946 try {
947 highlight(widget, pages, shown);
948 } catch (...) { // NOLINT(bugprone-empty-catch)
949 // The page's failure is the one reported; the bar keeps whatever highlight it had.
950 }
951 try {
952 std::function<void(std::size_t)> const onSelect = pages.onSelect;
953 if (onSelect) {
954 onSelect(shown);
955 }
956 } catch (...) { // NOLINT(bugprone-empty-catch)
957 // Only copying the wrapper can throw here; the application then keeps the failed index.
958 }
959 }

994 // A Dialog or a Drawer: the title, the dismissal, and content that exists only while open.
995 template <typename Spec, typename OverlayWidget>
996 Widget& mountOverlay(reactive::Scope& scope, Spec const& spec, OverlayWidget& widget) {
997 bind(scope, spec.title, [&widget](std::string const& title) { widget.setTitle(title); });
998 auto& content = scope.make<std::unique_ptr<reactive::Scope>>();
999 auto& shownOpen = scope.make<std::optional<bool>>();
1000 // A bound `open` is controlled: a dismissal the document refuses opens the dialog again, over the content it
1001 // still holds. Content that failed to mount is not there to show, so the dialog stays closed for it.
1002 std::function<void()> onDismiss = event(spec.onDismiss);
1003 if (spec.open.isBound()) {
1004 auto reassert =
1005 reasserter<bool>(scope, shownOpen, spec.open.binding(), [&widget, &content, &shownOpen](bool open) {
1006 if (!open || content != nullptr) {
1007 widget.setOpen(open);
1008 shownOpen = open;
1009 }
1010 });
1011 onDismiss = [inner = std::move(onDismiss), reassert = std::move(reassert)] {
1012 if (inner) {
1013 inner();
1014 }
1015 reassert(false);
1016 };
1017 }
1018 widget.setOnDismiss(onDismiss);
1019 auto show = [this, &widget, &content, &shownOpen, depth = scope.depth() + 1, child = spec.child](
1020 bool open, std::function<void()> const& dismiss) {
1021 if (!open) {
1022 content.reset();
1023 } else if (content == nullptr) {
1024 try {
1025 content = underOverlay(&widget, [&] { return mountContent(depth, child, widget); });
1026 } catch (...) {
1027 dismissFailed(dismiss);
1028 throw;
1029 }
1030 }
1031 widget.setOpen(open);
1032 shownOpen = open;
1033 };
1034 if (!spec.open.isBound()) {
1035 // A constant has no application state to bring back in line: content that fails throws out of the mount.
1036 show(spec.open.constant(), {});
1037 } else {
1038 scope.effect([binding = spec.open.binding(), show = std::move(show), onDismiss = std::move(onDismiss)] {
1039 show(binding(), onDismiss);
1040 });
1041 }
1042 return widget;
1043 }

1045 // After a Dialog's content failed to mount: tells the application through `onDismiss` that the dialog is closed.
1046 // It runs last, from a copy: it is a widget callback, which may destroy the Mounted and with it the dialog and
1047 // the Effect that called this; the caller only rethrows after it. Nothing thrown here may replace the failure
1048 // being reported.
1049 static void dismissFailed(std::function<void()> const& onDismiss) noexcept {
1050 try {
1051 std::function<void()> const dismiss = onDismiss;
1052 if (dismiss) {
1053 dismiss();
1054 }
1055 } catch (...) { // NOLINT(bugprone-empty-catch)
1056 // Only copying the wrapper can throw here; the application then keeps `open` true.
1057 }
1058 }

1095 // first, the rows (whose slots point into the session's snapshot) go next, and the model outlives its session.

1145 // Moves the widgets into the snapshot's order, and `rows` with them. A move that throws is recorded in `failure`
1146 // when it is the first failure, and leaves `rows` mirroring the container as it stands.
1147 static void reorder(std::vector<KeyedRow>& rows, ContainerWidget& container, std::vector<Key> const& keys,
1148 std::vector<std::size_t> const& order, std::exception_ptr& failure) {
1149 std::unordered_map<Key, std::size_t> position; // key -> index in `rows`
1150 std::unordered_map<Widget const*, std::size_t> rowOf;
1151 std::vector<Widget*> present;
1152 for (std::size_t i = 0; i < rows.size(); ++i) {
1153 KeyedRow const& row = rows.at(i);
1154 position.emplace(row.key, i);
1155 if (row.widget != nullptr) {
1156 rowOf.emplace(row.widget, i);
1157 present.push_back(row.widget);
1158 }
1159 }
1160 std::vector<Widget*> target;
1161 target.reserve(present.size());
1162 for (std::size_t const index : order) {
1163 if (auto const found = position.find(keys.at(index)); found != position.end()) {
1164 if (KeyedRow const& row = rows.at(found->second); row.widget != nullptr) {
1165 target.push_back(row.widget);
1166 }
1167 }
1168 }
1169 std::vector<KeyedRow> ordered;
1170 ordered.reserve(rows.size());
1171 try {
1172 reorderChildren(container, present, target);
1173 } catch (...) {
1174 if (!failure) {
1175 failure = std::current_exception();
1176 }
1177 }
1178 // `present` is the container's order now: the snapshot's, unless a move threw. Rows without a widget have
1179 // no place in the container and go last.
1180 for (Widget const* const widget : present) {
1181 ordered.push_back(std::move(rows.at(rowOf.at(widget))));
1182 }
1183 for (KeyedRow& row : rows) {
1184 if (row.widget == nullptr) {
1185 ordered.push_back(std::move(row));
1186 }
1187 }
1188 rows = std::move(ordered);
1189 }

1191 // Brings `rows` to the snapshot `keys`. `rows` mirrors the container's order on entry and on exit, also when
1192 // something throws, so a failure costs this run's remaining work and never the bookkeeping of the next one.
1193 // A row that fails to mount leaves nothing behind and does not stop the other rows; the first failure is
1194 // rethrown at the end, for the Effect to report, and the next run mounts the row again.
1195 void reconcile(std::size_t depth, ForEachSession& session, std::vector<KeyedRow>& rows, ContainerWidget& container,
1196 std::vector<Key> const& keys, RowHook const& onRow) {
1197 std::vector<std::size_t> const order = firstOccurrences(keys);
1198 std::unordered_map<Key, std::size_t> wanted; // key -> snapshot index
1199 for (std::size_t const index : order) {
1200 wanted.emplace(keys.at(index), index);
1201 }
1202
1203 // Unmount the gone keys, last first as KeyedRows does, then update the kept ones in place, which keeps their
1204 // widgets.
1205 for (KeyedRow& row : std::views::reverse(rows)) {
1206 if (!wanted.contains(row.key)) {
1207 row.scope.reset();
1208 }
1209 }
1210 std::erase_if(rows, [](KeyedRow const& row) { return row.scope == nullptr; });
1211 std::unordered_set<Key> kept;
1212 for (KeyedRow const& row : rows) {
1213 kept.insert(row.key);
1214 row.slot->assign(wanted.at(row.key));
1215 }
1216
1217 // Mount the new keys; each widget is appended to the container, and its row to `rows`.
1218 std::exception_ptr failure;
1219 rows.reserve(order.size());
1220 for (std::size_t const index : order) {
1221 if (kept.contains(keys.at(index))) {
1222 continue;
1223 }
1224 try {
1225 rows.push_back(mountRow(depth, session, container, keys.at(index), index, onRow));
1226 } catch (...) {
1227 if (!failure) {
1228 failure = std::current_exception();
1229 }
1230 }
1231 }
1232
1233 reorder(rows, container, keys, order, failure);
1234 if (failure) {
1235 std::rethrow_exception(failure);
1236 }
1237 }

1435/// @brief A view tree mounted on a backend: its widgets, built once, and the bindings that keep them current.
1436///
1437/// Destroying it destroys every binding before the widget it drives and every child before its parent. The
1438/// runtime, the backend and every signal a binding reads must outlive it; the owner of the runtime constructs and
1439/// destroys it. Non-copyable and non-movable: bindings point into it.
1440///
1441/// Every widget callback, a drop's `accepts` predicate included, runs as a `reactive::Runtime::widgetEvent` and
1442/// untracked, so a binding whose setter makes the backend call a handler does not subscribe to what the handler
1443/// reads. An exception a callback throws is reported (`detail::site::kCallbackThrew`) and goes no further: the
1444/// backend's call returns normally, a predicate that threw refuses the drop, and what the callback wrote before
1445/// throwing still flushes.
1446///
1447/// An input whose value is a slot is controlled: a text input's text, a checkbox's state, a select's key, a slider's,
1448/// date-time input's or file picker's value, a collapsible panel's state, a Tabs index, a Dialog's `open` and a
1449/// Table's selection. After a user's request, and after the flush it caused, a posted turn shows the slot's value if
1450/// the widget shows anything else: a refused request snaps back, an accepted one costs no setter call, and a
1451/// transformed one shows the request for that one turn. That holds with no handler too, which makes a slot without one
1452/// read-only. A constant is not a slot: the widget keeps what the user did to it.
1453///
1454/// A mount that throws goes no further than the content it was building, and leaves none of it behind. A node mounted
1455/// directly, a constant Switch selector, Tabs index or Dialog `open` included, throws out of the constructor. Content
1456/// a binding mounts, which a bound selector, index or `open` does in an Effect even during the constructor, is
1457/// reported (`reactive::detail::site::kEffectThrew`) instead: a Switch shows nothing for that key, having torn the old
1458/// case down first; a Tabs keeps the previous page shown and selected; a Dialog stays closed. A Tabs then calls its
1459/// `onSelect` with the page still shown, and a Dialog its `onDismiss`, each as a widget callback and as the last thing
1460/// the mount does, so the application's state follows the widget and its next pick of the failed tab, or opening of
1461/// the dialog, mounts again. A Switch selector is equality-gated, so its next change of key mounts again. A Tabs
1462/// index and a Dialog `open` are not: while one still names content that failed — no `onSelect` or `onDismiss` moved
1463/// it — every run of its binding, a change to any signal the binding reads included, tries that mount again, once
1464/// per run, reporting each failure and calling `onSelect` or `onDismiss` again.
1465///
1466/// A ForEach or a Table mounts its rows in an Effect too, one row at a time: a row that fails is reported, leaves
1467/// nothing behind and is left out, the other rows are mounted and ordered all the same, and the next change of the
1468/// rows mounts it again. A key repeated in one snapshot is reported (`detail::site::kDuplicateKey`), and only its
1469/// first row is shown.
1470///
1471/// Bindings run in owner order. The tree's own bindings are at the depth the mount was given; the content of a Switch
1472/// case, a Tabs page or a Dialog, and every ForEach or Table row, is in a scope one level deeper than the binding that
1473/// mounted it. So an Effect that removes content runs before the content's own bindings in the same flush, and a
1474/// binding never evaluates against state its owner is about to take away. A mount placed under an outer scope (a
1475/// screen's, a row's) takes that scope's depth plus one, so the outer scope's Effects run first too.
1476///
1477/// A mount must run to completion: nothing it calls, a backend factory or setter or a binding's evaluation, may
1478/// destroy this object, its runtime or its backend. Widget callbacks are events, not part of a mount, and may.
