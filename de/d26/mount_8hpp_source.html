80/// @brief Which entries of a permutation lie on one longest increasing run.
81///
82/// @p sequence holds, in current order, each widget's target position; the widgets on the run are already in
83/// order relative to each other and stay where they are, and every other one is moved once.
84/// @param sequence A permutation of `0 .. n-1`.
85/// @return Indexed by target position: whether that widget stays.
86[[nodiscard]] inline std::vector<bool> stableTargets(std::vector<std::size_t> const& sequence) {
87 constexpr auto kNone = static_cast<std::size_t>(-1);
88 std::vector<std::size_t> tails; // tails.at(n - 1): where the smallest tail of a run of length n sits
89 std::vector<std::size_t> previous(sequence.size(), kNone);
90 for (std::size_t i = 0; i < sequence.size(); ++i) {
91 auto const place = std::ranges::lower_bound(tails, sequence.at(i), {},
92 [&sequence](std::size_t entry) { return sequence.at(entry); });
93 auto const length = static_cast<std::size_t>(place - tails.begin());
94 if (length > 0) {
95 previous.at(i) = tails.at(length - 1);
96 }
97 if (length == tails.size()) {
98 tails.push_back(i);
99 } else {
100 tails.at(length) = i;
101 }
102 }
103 std::vector<bool> stable(sequence.size(), false);
104 for (std::size_t cursor = tails.empty() ? kNone : tails.back(); cursor != kNone; cursor = previous.at(cursor)) {
105 stable.at(sequence.at(cursor)) = true;
106 }
107 return stable;
108}

110/// @brief Brings a container's children from their current order into the target order with `moveChild`,
111/// moving only the children outside a longest run already in order.
112/// @param container The container.
113/// @param current Its children, in their current order. Kept equal to the container's order after every move, so
114/// it is still that order when a move throws.
115/// @param target The same children, in the order wanted.
116inline void reorderChildren(ContainerWidget& container, std::vector<Widget*>& current,
117 std::vector<Widget*> const& target) {
118 std::unordered_map<Widget const*, std::size_t> targetOf;
119 for (std::size_t i = 0; i < target.size(); ++i) {
120 targetOf.emplace(target.at(i), i);
121 }
122 std::vector<std::size_t> sequence;
123 sequence.reserve(current.size());
124 for (Widget const* const child : current) {
125 sequence.push_back(targetOf.at(child));
126 }
127 std::vector<bool> const stable = stableTargets(sequence);
128 auto const indexOf = [&current](Widget const* child) {
129 return static_cast<std::size_t>(std::ranges::find(current, child) - current.begin());
130 };
131 // From the back: each moved child goes directly before its successor in the target order, which is already
132 // in place, so one move per unstable child suffices.
133 for (std::size_t position = target.size(); position-- > 0;) {
134 if (stable.at(position)) {
135 continue;
136 }
137 Widget* const moving = target.at(position);
138 std::size_t const from = indexOf(moving);
139 std::size_t destination = current.size() - 1;
140 if (position + 1 < target.size()) {
141 std::size_t const before = indexOf(target.at(position + 1));
142 destination = from < before ? before - 1 : before;
143 }
144 if (destination == from) {
145 continue;
146 }
147 container.moveChild(*moving, destination);
148 current.erase(current.begin() + static_cast<std::ptrdiff_t>(from));
149 current.insert(current.begin() + static_cast<std::ptrdiff_t>(destination), moving);
150 }
151}

177 /// @brief Mounts @p node under @p parent; @p scope owns its widgets and bindings.
178 ///
179 /// A node's widget is adopted first, then its bindings, then its children, so destroying the scope destroys
180 /// bindings before their widget and children before their parent.
181 /// @param scope Owns everything mounted.
182 /// @param node The node; null mounts nothing.
183 /// @param parent The container to append to, or null for a root.
184 /// @return The node's widget, or null for a null node.
185 Widget* mount(reactive::Scope& scope, Node const& node, ContainerWidget* parent) {
186 if (!node) {
187 return nullptr;
188 }
189 return std::visit([&](auto const& spec) -> Widget* { return &this->mountKind(scope, spec, parent); },
190 node->kind);
191 }

193private:
194 // A constant is applied once; a binding becomes an equality-gated Computed plus an Effect, made in that order
195 // after the widget, so the Effect dies first. Returns the Computed, or null for a constant.
196 template <typename T, typename Apply>
197 reactive::Computed<T>* bind(reactive::Scope& scope, Prop<T> const& prop, Apply apply) {
198 if (!prop.isBound()) {
199 apply(prop.constant());
200 return nullptr;
201 }
202 auto& value = scope.make<reactive::Computed<T>>(*_rt, prop.binding());
203 scope.effect([&value, apply = std::move(apply)] { apply(value.get()); });
204 return &value;
205 }

207 // Runs a widget callback, and returns whether it completed.
208 // - Inside widgetEvent: one batch, and a flush that would start meanwhile is re-posted, so a remount never
209 // destroys a widget whose native handler is on the stack.
210 // - Untracked: a backend may call a handler from inside a setter, and so from inside the binding Effect that
211 // called it; that Effect must not subscribe to what the handler reads.
212 // - An exception is reported and stops here: the backend that called the handler need not be exception-safe.
213 // Writes made before it still flush when the batch closes.
214 template <typename F>
215 static bool runCallback(reactive::Runtime& runtime, F const& body) {
216 return runtime.widgetEvent([&] {
217 return runtime.untracked([&] {
218 try {
219 body();
220 return true;
221 } catch (...) {
222 runtime.core()->report(site::kCallbackThrew);
223 return false;
224 }
225 });
226 });
227 }
228
229 template <typename... Args>
230 [[nodiscard]] std::function<void(Args...)> event(std::function<void(Args...)> const& handler) const {
231 if (!handler) {
232 return {};
233 }
234 // The closure owns a copy of the handler and frees it when it dies; the analyzer loses the copy's heap block
235 // in libc++'s std::function and reports it as leaked.
236 // NOLINTNEXTLINE(clang-analyzer-cplusplus.NewDeleteLeaks)
237 return [runtime = _rt, handler](Args... args) {
238 static_cast<void>(runCallback(*runtime, [&] { handler(std::move(args)...); }));
239 };
240 }

242 /// @brief A controlled input's re-assertion: what the widget shows as far as the mount knows, what its slot
243 /// says now, and how to show a value.
244 /// @tparam T The value type.
245 template <typename T>
246 struct Reassertion {
247 /// @brief What the widget shows: the last value the mount set, or the last value the user asked for since.
248 /// Owned by the scope, made before this, so it outlives it.
249 std::optional<T>* shown;
250 /// @brief What the slot says now.
251 std::function<T()> read;
252 /// @brief Shows a value, and records it in `*shown`.
253 std::function<void(T const&)> apply;
254 };
255
256 // What a controlled input's event calls after the application's handler, with the value the user asked for.
257 //
258 // The widget shows the request already: it is recorded as shown. Then one turn is posted to the owner; it reads
259 // the slot untracked and shows it when the widget shows anything else. The turn is posted after the event's
260 // widgetEvent closed, so it runs after the flush the event caused, whose bindings record what they set: an
261 // accepted request costs no setter call, a refused one snaps back, and a transformed one, which the binding has
262 // shown already, costs nothing more. The check is owned by @p scope and held weakly by the event and the turn,
263 // so neither does anything once the content is gone.
264 template <typename T, typename Read, typename Apply>
265 [[nodiscard]] std::function<void(T const&)> reasserter(reactive::Scope& scope, std::optional<T>& shown, Read read,
266 Apply apply) {
267 auto const check = std::make_shared<Reassertion<T>>(
268 Reassertion<T>{.shown = &shown, .read = std::move(read), .apply = std::move(apply)});
269 scope.make<std::shared_ptr<Reassertion<T>>>(check);
270 return [runtime = _rt, weak = std::weak_ptr{check}](T const& requested) {
271 if (auto const live = weak.lock()) {
272 *live->shown = requested;
273 } else {
274 return;
275 }
276 // Called from a widget handler, which must not throw: a post that fails leaves the widget showing the
277 // user's value until the slot next changes.
278 try {
279 runtime->owner().post([runtime, weak] {
280 if (auto const live = weak.lock()) {
281 runtime->untracked([&live] { showSlot(*live); });
282 }
283 });
284 } catch (...) { // NOLINT(bugprone-empty-catch): see above
285 }
286 };
287 }
288
289 // Shows the slot's value if the widget shows anything else. A slot that throws is its binding's failure, which
290 // the binding's Effect reports; the widget keeps what the user did.
291 template <typename T>
292 static void showSlot(Reassertion<T> const& check) noexcept {
293 try {
294 T const current = check.read();
295 if (!(check.shown->has_value() && **check.shown == current)) {
296 check.apply(current);
297 }
298 } catch (...) { // NOLINT(bugprone-empty-catch): reported by the binding's Effect, see above
299 }
300 }

302 // A controlled input's value: bound like any property, recording what it shows, and not sending what it already
303 // shows; a slot is also re-asserted after each user event (see `reasserter`). Returns what the event calls, or
304 // nothing for a constant, which is not a slot: the widget keeps what the user did to it.
305 template <typename T, typename Apply>
306 [[nodiscard]] std::function<void(T const&)> controlled(reactive::Scope& scope, Prop<T> const& prop, Apply apply) {
307 auto& shown = scope.make<std::optional<T>>();
308 // A value the widget already shows, the user's accepted edit above all, is not sent again.
309 auto show = [&shown, apply = std::move(apply)](T const& value) {
310 if (shown.has_value() && *shown == value) {
311 return;
312 }
313 apply(value);
314 shown = value;
315 };
316 reactive::Computed<T> const* const value = bind(scope, prop, show);
317 if (value == nullptr) {
318 return {};
319 }
320 return reasserter<T>(scope, shown, [value] { return value->get(); }, std::move(show));
321 }

323 // An input's event: the application's handler, run as `event` runs it, then the re-assertion of the request.
324 // Installed whenever either exists, so a slot without a handler is read-only rather than editable.
325 template <typename T, typename Arg>
326 [[nodiscard]] std::function<void(Arg)> inputEvent(std::function<void(Arg)> const& handler,
327 std::function<void(T const&)> reassert) const {
328 if (!reassert) {
329 return event(handler);
330 }
331 return [inner = event(handler), reassert = std::move(reassert)](Arg arg) {
332 T const requested(arg);
333 if (inner) {
334 inner(std::move(arg));
335 }
336 reassert(requested);
337 };
338 }

340 void applyCommon(reactive::Scope& scope, Widget& widget, Common const& common) {
341 if (common.visible.isBound() || !common.visible.constant()) {
342 bind(scope, common.visible, [&widget](bool shown) { widget.setVisible(shown); });
343 }
344 if (common.enabled.isBound() || !common.enabled.constant()) {
345 bind(scope, common.enabled, [&widget](bool enabled) { widget.setEnabled(enabled); });
346 }
347 if (common.layout != LayoutHints{}) {
348 widget.setLayout(common.layout);
349 }
350 if (common.dragKey.isBound() || common.dragKey.constant().has_value()) {
351 bind(scope, common.dragKey, [&widget](std::optional<Key> const& key) { widget.setDragKey(key); });
352 }
353 if (common.onDrop) {
354 // No predicate accepts every key; a predicate that throws refuses the drop.
355 std::function<bool(Key const&)> accepts = [](Key const&) { return true; };
356 if (common.accepts) {
357 accepts = [runtime = _rt, predicate = common.accepts](Key const& key) {
358 bool accepted = false;
359 static_cast<void>(runCallback(*runtime, [&] { accepted = predicate(key); }));
360 return accepted;
361 };
362 }
363 widget.setDropHandler(std::move(accepts), event(common.onDrop));
364 }
365 }

509 // Content a binding mounts (a Switch case, Dialog content), all or nothing: it is built in a scope of its own,
510 // which the caller takes only once the mount completed, so content that throws partway leaves nothing behind.
511 // Built untracked, so the binding's Effect depends on its own prop alone. The scope is one level deeper than the
512 // one that owns the binding, so the owner's Effects run before the content's.
513 [[nodiscard]] std::unique_ptr<reactive::Scope> mountContent(std::size_t depth, Node const& node,
514 ContainerWidget& host) {
515 auto content = std::make_unique<reactive::Scope>(*_rt, depth);
516 _rt->untracked([&] { static_cast<void>(mount(*content, node, &host)); });
517 return content;
518 }

520 // reset() tears the old case down, newest first, before the new one is built. The Computed behind a bound
521 // selector is equality-gated, so an unchanged key never gets here. A case that fails to mount leaves nothing for
522 // its key; the next change of the key mounts again.
523 Widget& mountKind(reactive::Scope& scope, Switch const& spec, ContainerWidget* parent) {
524 SlotWidget& widget = scope.adopt(_backend->createSlot(parent));
525 applyCommon(scope, widget, spec.common);
526 auto& content = scope.make<std::unique_ptr<reactive::Scope>>();
527 bind(scope, spec.selector,
528 [this, &widget, &content, depth = scope.depth() + 1, cases = spec.cases,
529 fallback = spec.fallback](Key const& key) {
530 content.reset();
531 auto const chosen = std::ranges::find(cases, key, &SwitchCase::key);
532 Node const& node = chosen == cases.end() ? fallback : chosen->node;
533 if (node) {
534 content = mountContent(depth, node, widget);
535 }
536 });
537 return widget;
538 }

540 // Every page is owned by `pageScope`, a child scope made before the binding, so pages are destroyed newest first,
541 // after the binding that shows them and before the tab bar. A scope fixes that order; a vector's element
542 // destruction order differs between standard libraries.
543 //
544 // A bound index is not equality-gated: the Effect reads the binding itself, not a Computed. After a failed page,
545 // the application's onSelect writes the index back inside this Effect's own run; a Computed would keep the failed
546 // index and find the user's next pick of that tab unchanged, while the Effect's dependency on the binding's own
547 // signals sees it as the change it is. The cost is that showPage also runs when something the binding reads
548 // changes but the index does not: for a page that is mounted it only re-sends the highlight, and for a page that
549 // failed it tries the mount again, once per run.
550 Widget& mountKind(reactive::Scope& scope, Tabs const& spec, ContainerWidget* parent) {
551 TabsWidget& widget = scope.adopt(_backend->createTabs(parent));
552 applyCommon(scope, widget, spec.common);
553 std::vector<std::string> labels;
554 labels.reserve(spec.tabs.size());
555 for (Tab const& tab : spec.tabs) {
556 labels.push_back(tab.label);
557 }
558 widget.setTabs(labels);
559 auto& pageScope = scope.child();
560 auto& pages = scope.make<Pages>();
561 pages.slots.resize(spec.tabs.size(), nullptr);

584 // A page is mounted the first time its tab is selected and only hidden afterwards, so its widgets keep their
585 // state (focus, scroll position, half-typed text) across tab switches. A new page is complete before anything
586 // else changes: one that fails to mount leaves the previous page shown and selected, and the bar's highlight,
587 // which the user's pick may already have moved, goes back to it. The next change of the index mounts again.
588 void showPage(TabsWidget& widget, reactive::Scope& pageScope, Pages& pages, std::vector<Tab> const& pageSpecs,
589 std::size_t index) {
590 bool const exists = index < pageSpecs.size();
591 bool const isNew = exists && pages.slots.at(index) == nullptr;
592 if (isNew) {
593 try {
594 mountPage(widget, pageScope, pages, pageSpecs.at(index).node, index);
595 } catch (...) {
596 if (pages.shown.has_value()) {
597 restoreSelection(widget, pages, *pages.shown);
598 }
599 throw;
600 }
601 }
602 if (pages.shown != index) {
603 if (pages.shown.has_value() && *pages.shown < pages.slots.size()) {
604 if (SlotWidget* const previous = pages.slots.at(*pages.shown); previous != nullptr) {
605 previous->setVisible(false);
606 }
607 }
608 if (exists && !isNew) {
609 pages.slots.at(index)->setVisible(true);
610 }
611 }
612 pages.shown = index;
613 highlight(widget, pages, index);
614 }

623 // through `onSelect`, so its index matches the widget and selecting the failed tab again is a change that mounts
624 // again.

626 // onSelect runs last, from a copy: it is a widget callback, which may destroy the Mounted and with it `pages`,

639 if (onSelect) {

642 } catch (...) { // NOLINT(bugprone-empty-catch)
643 // Only copying the wrapper can throw here; the application then keeps the failed index.
644 }

657 // The content is built before the overlay opens and torn down before it closes. Content that fails to mount
658 // under a bound `open` leaves the dialog closed and calls `onDismiss`, so the application's `open` follows the
659 // widget and opening it again is a change that mounts again.
660 //
661 // A bound `open` is not equality-gated, for the reason a Tabs index is not: onDismiss writes `open` inside this
662 // Effect's own run, which a Computed would miss, and then find the next opening unchanged. An Effect that runs
663 // with `open` unchanged re-sends `setOpen`, and mounts content only if none is mounted: once per run while `open`
664 // stays true over content that failed.
665 Widget& mountKind(reactive::Scope& scope, Dialog const& spec, ContainerWidget* parent) {
666 DialogWidget& widget = scope.adopt(_backend->createDialog(parent));
667 applyCommon(scope, widget, spec.common);
668 bind(scope, spec.title, [&widget](std::string const& title) { widget.setTitle(title); });
669 auto& content = scope.make<std::unique_ptr<reactive::Scope>>();
670 auto& shownOpen = scope.make<std::optional<bool>>();
671 // A bound `open` is controlled: a dismissal the document refuses opens the dialog again, over the content it
672 // still holds. Content that failed to mount is not there to show, so the dialog stays closed for it.
673 std::function<void()> onDismiss = event(spec.onDismiss);
674 if (spec.open.isBound()) {
675 auto reassert =
676 reasserter<bool>(scope, shownOpen, spec.open.binding(), [&widget, &content, &shownOpen](bool open) {
677 if (!open || content != nullptr) {
678 widget.setOpen(open);
679 shownOpen = open;
680 }
681 });
682 onDismiss = [inner = std::move(onDismiss), reassert = std::move(reassert)] {
683 if (inner) {
684 inner();
685 }
686 reassert(false);
687 };
688 }
689 widget.setOnDismiss(onDismiss);
690 auto show = [this, &widget, &content, &shownOpen, depth = scope.depth() + 1, child = spec.child](
691 bool open, std::function<void()> const& dismiss) {
692 if (!open) {
693 content.reset();
694 } else if (content == nullptr) {
695 try {
696 content = mountContent(depth, child, widget);
697 } catch (...) {
698 dismissFailed(dismiss);
699 throw;
700 }
701 }
702 widget.setOpen(open);
703 shownOpen = open;
704 };
705 if (!spec.open.isBound()) {
706 // A constant has no application state to bring back in line: content that fails throws out of the mount.
707 show(spec.open.constant(), {});
708 } else {
709 scope.effect([binding = spec.open.binding(), show = std::move(show), onDismiss = std::move(onDismiss)] {
710 show(binding(), onDismiss);
711 });
712 }
713 return widget;
714 }

716 // After a Dialog's content failed to mount: tells the application through `onDismiss` that the dialog is closed.

724 dismiss();
725 }
726 } catch (...) { // NOLINT(bugprone-empty-catch)
727 // Only copying the wrapper can throw here; the application then keeps `open` true.
728 }

756 // The model, then the session, then the rows, then the Effect: destroyed in reverse, so the Effect stops
757 // first, the rows (whose slots point into the session's snapshot) go next, and the model outlives its session.
758 void mountRows(reactive::Scope& scope, std::shared_ptr<ForEachModel const> const& model,
759 ContainerWidget& container, RowHook onRow) {
760 if (!model) {
761 return;
762 }
763 scope.make<std::shared_ptr<ForEachModel const>>(model);
764 ForEachSession& session = scope.adopt(model->open(*_rt));
765 auto& rows = scope.make<KeyedRows>();
766 scope.effect([this, &session, &rows, &container, depth = scope.depth() + 1, onRow = std::move(onRow)] {
767 std::vector<Key> const keys = session.pull(); // tracked: the rows are this Effect's only source
768 // Untracked: what a row view reads while it is built, or what a kept row's update touches, must not
769 // become a dependency of the whole list.
770 _rt->untracked([&] { reconcile(depth, session, rows.list, container, keys, onRow); });
771 });

774 // The snapshot index of each key's first occurrence, in order; every later duplicate is reported.
775 [[nodiscard]] std::vector<std::size_t> firstOccurrences(std::vector<Key> const& keys) {
776 std::unordered_set<Key> seen;
777 std::vector<std::size_t> order;
778 order.reserve(keys.size());
779 for (std::size_t index = 0; index < keys.size(); ++index) {
780 if (seen.insert(keys.at(index)).second) {
781 order.push_back(index);
782 } else {
783 _rt->core()->report(site::kDuplicateKey);
784 }
785 }
786 return order;
787 }

789 // All or nothing: the row is built in a scope of its own, one level deeper than the ForEach's, which the caller
790 // takes only once the mount completed.
791 [[nodiscard]] KeyedRow mountRow(std::size_t depth, ForEachSession& session, ContainerWidget& container,
792 Key const& key, std::size_t index, RowHook const& onRow) {
793 KeyedRow row{.key = key, .scope = std::make_unique<reactive::Scope>(*_rt, depth)};
794 row.slot = &row.scope->adopt(session.makeRow(index));
795 row.widget = mount(*row.scope, row.slot->view(), &container);
796 if (row.widget != nullptr && onRow) {
797 onRow(*row.widget, key);
798 }
799 return row;

802 // Moves the widgets into the snapshot's order, and `rows` with them. A move that throws is recorded in `failure`
803 // when it is the first failure, and leaves `rows` mirroring the container as it stands.
804 static void reorder(std::vector<KeyedRow>& rows, ContainerWidget& container, std::vector<Key> const& keys,
805 std::vector<std::size_t> const& order, std::exception_ptr& failure) {
806 std::unordered_map<Key, std::size_t> position; // key -> index in `rows`
807 std::unordered_map<Widget const*, std::size_t> rowOf;
808 std::vector<Widget*> present;
809 for (std::size_t i = 0; i < rows.size(); ++i) {
810 KeyedRow const& row = rows.at(i);
811 position.emplace(row.key, i);
812 if (row.widget != nullptr) {
813 rowOf.emplace(row.widget, i);
814 present.push_back(row.widget);
815 }
816 }
817 std::vector<Widget*> target;
818 target.reserve(present.size());
819 for (std::size_t const index : order) {
820 if (auto const found = position.find(keys.at(index)); found != position.end()) {
821 if (KeyedRow const& row = rows.at(found->second); row.widget != nullptr) {
822 target.push_back(row.widget);
823 }
824 }
825 }
826 std::vector<KeyedRow> ordered;
827 ordered.reserve(rows.size());
828 try {
829 reorderChildren(container, present, target);
830 } catch (...) {
831 if (!failure) {
832 failure = std::current_exception();
833 }
834 }
835 // `present` is the container's order now: the snapshot's, unless a move threw. Rows without a widget have
836 // no place in the container and go last.
837 for (Widget const* const widget : present) {
838 ordered.push_back(std::move(rows.at(rowOf.at(widget))));
839 }
840 for (KeyedRow& row : rows) {
841 if (row.widget == nullptr) {
842 ordered.push_back(std::move(row));
843 }
844 }
845 rows = std::move(ordered);
846 }
847
848 // Brings `rows` to the snapshot `keys`. `rows` mirrors the container's order on entry and on exit, also when
849 // something throws, so a failure costs this run's remaining work and never the bookkeeping of the next one.
850 // A row that fails to mount leaves nothing behind and does not stop the other rows; the first failure is
851 // rethrown at the end, for the Effect to report, and the next run mounts the row again.
852 void reconcile(std::size_t depth, ForEachSession& session, std::vector<KeyedRow>& rows, ContainerWidget& container,
853 std::vector<Key> const& keys, RowHook const& onRow) {
854 std::vector<std::size_t> const order = firstOccurrences(keys);
855 std::unordered_map<Key, std::size_t> wanted; // key -> snapshot index
856 for (std::size_t const index : order) {
857 wanted.emplace(keys.at(index), index);
858 }
859
860 // Unmount the gone keys, last first as KeyedRows does, then update the kept ones in place, which keeps their
861 // widgets.
862 for (KeyedRow& row : std::views::reverse(rows)) {
863 if (!wanted.contains(row.key)) {
864 row.scope.reset();
865 }
866 }
867 std::erase_if(rows, [](KeyedRow const& row) { return row.scope == nullptr; });
868 std::unordered_set<Key> kept;
869 for (KeyedRow const& row : rows) {
870 kept.insert(row.key);
871 row.slot->assign(wanted.at(row.key));
872 }
873
874 // Mount the new keys; each widget is appended to the container, and its row to `rows`.
875 std::exception_ptr failure;
876 rows.reserve(order.size());
877 for (std::size_t const index : order) {
878 if (kept.contains(keys.at(index))) {
879 continue;
880 }
881 try {
882 rows.push_back(mountRow(depth, session, container, keys.at(index), index, onRow));
883 } catch (...) {
884 if (!failure) {
885 failure = std::current_exception();
886 }
887 }
888 }
889
890 reorder(rows, container, keys, order, failure);
891 if (failure) {
892 std::rethrow_exception(failure);
893 }
894 }

939/// @brief A view tree mounted on a backend: its widgets, built once, and the bindings that keep them current.
940///
941/// Destroying it destroys every binding before the widget it drives and every child before its parent. The
942/// runtime, the backend and every signal a binding reads must outlive it; the owner of the runtime constructs and
943/// destroys it. Non-copyable and non-movable: bindings point into it.
944///
945/// Every widget callback, a drop's `accepts` predicate included, runs as a `reactive::Runtime::widgetEvent` and
946/// untracked, so a binding whose setter makes the backend call a handler does not subscribe to what the handler
947/// reads. An exception a callback throws is reported (`detail::site::kCallbackThrew`) and goes no further: the
948/// backend's call returns normally, a predicate that threw refuses the drop, and what the callback wrote before
949/// throwing still flushes.
950///
951/// An input whose value is a slot is controlled: a text input's text, a checkbox's state, a select's key, a slider's,
952/// date-time input's or file picker's value, a collapsible panel's state, a Tabs index, a Dialog's `open` and a
953/// Table's selection. After a user's request, and after the flush it caused, a posted turn shows the slot's value if
954/// the widget shows anything else: a refused request snaps back, an accepted one costs no setter call, and a
955/// transformed one shows the request for that one turn. That holds with no handler too, which makes a slot without one
956/// read-only. A constant is not a slot: the widget keeps what the user did to it.
957///
958/// A mount that throws goes no further than the content it was building, and leaves none of it behind. A node mounted
959/// directly, a constant Switch selector, Tabs index or Dialog `open` included, throws out of the constructor. Content
960/// a binding mounts, which a bound selector, index or `open` does in an Effect even during the constructor, is
961/// reported (`reactive::detail::site::kEffectThrew`) instead: a Switch shows nothing for that key, having torn the old
962/// case down first; a Tabs keeps the previous page shown and selected; a Dialog stays closed. A Tabs then calls its
963/// `onSelect` with the page still shown, and a Dialog its `onDismiss`, each as a widget callback and as the last thing
964/// the mount does, so the application's state follows the widget and its next pick of the failed tab, or opening of
965/// the dialog, mounts again. A Switch selector is equality-gated, so its next change of key mounts again. A Tabs
966/// index and a Dialog `open` are not: while one still names content that failed — no `onSelect` or `onDismiss` moved
967/// it — every run of its binding, a change to any signal the binding reads included, tries that mount again, once
968/// per run, reporting each failure and calling `onSelect` or `onDismiss` again.
969///
970/// A ForEach or a Table mounts its rows in an Effect too, one row at a time: a row that fails is reported, leaves
971/// nothing behind and is left out, the other rows are mounted and ordered all the same, and the next change of the
972/// rows mounts it again. A key repeated in one snapshot is reported (`detail::site::kDuplicateKey`), and only its
973/// first row is shown.
974///
975/// Bindings run in owner order. The tree's own bindings are at the depth the mount was given; the content of a Switch
976/// case, a Tabs page or a Dialog, and every ForEach or Table row, is in a scope one level deeper than the binding that
977/// mounted it. So an Effect that removes content runs before the content's own bindings in the same flush, and a
978/// binding never evaluates against state its owner is about to take away. A mount placed under an outer scope (a
979/// screen's, a row's) takes that scope's depth plus one, so the outer scope's Effects run first too.
980///
981/// A mount must run to completion: nothing it calls, a backend factory or setter or a binding's evaluation, may
982/// destroy this object, its runtime or its backend. Widget callbacks are events, not part of a mount, and may.
983class Mounted {
984public:
985 /// @param runtime The runtime the bindings are made in. Borrowed.
986 /// @param backend Makes the widgets. Borrowed.
987 /// @param root The tree; kept alive as long as the mount.
988 /// @param parent The container the root widget is appended to, or null for a backend root.
989 /// @param depth The scope depth of the tree's own bindings (`reactive::Scope::depth`): zero for a mount no other
990 /// scope owns, one more than the owner's for a mount placed under a scope.
991 /// @throws std::invalid_argument when @p root is null.
992 Mounted(reactive::Runtime& runtime MORPH_LIFETIMEBOUND, IViewBackend& backend MORPH_LIFETIMEBOUND, Node root,
993 ContainerWidget* parent = nullptr, std::size_t depth = 0)
994 : _root{nonNull(std::move(root))},
995 _mounter{runtime, backend},
996 _scope{runtime, depth},
997 // Untracked, so mounting from inside an Effect does not subscribe that Effect to anything read here.
998 _widget{runtime.untracked([this, parent] { return _mounter.mount(_scope, _root, parent); })} {}
999
1000 ~Mounted() = default;
1001 Mounted(Mounted const&) = delete;
1002 Mounted& operator=(Mounted const&) = delete;
1003 Mounted(Mounted&&) = delete;
1004 Mounted& operator=(Mounted&&) = delete;
1005
1006 /// @brief The root node's widget.
1007 /// @return The widget; valid as long as this object.
1008 [[nodiscard]] Widget& root() const noexcept { return *_widget; }
1009
1010private:
1011 [[nodiscard]] static Node nonNull(Node root) {
1012 if (!root) {
1013 throw std::invalid_argument{"morph::ui::Mounted: the root node is null"};
1014 }
1015 return root;
1016 }
1017
1018 Node _root;
1019 detail::Mounter _mounter;
1020 reactive::Scope _scope; // after _mounter: destroyed first, while the mounter its bindings call is alive
1021 Widget* _widget;
1022};
