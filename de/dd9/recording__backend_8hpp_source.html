45/// @brief Escapes text with a backslash: a line break becomes backslash and `n`, a carriage return backslash and `r`,
46/// and a backslash or a character @p isSpecial selects is preceded by a backslash.
47/// @tparam Special A callable taking a `char` and returning whether to escape it.
48/// @param text The text.
49/// @param isSpecial Selects the characters to escape besides the backslash and the two line-ending characters.
50/// @return The escaped text.
51template <typename Special>
52[[nodiscard]] std::string escape(std::string_view text, Special const& isSpecial) {
53 std::string escaped;
54 escaped.reserve(text.size());
55 for (char const character : text) {
56 if (character == '\n') {
57 escaped += "\\n";
58 } else if (character == '\r') {
59 escaped += "\\r";
60 } else {
61 if (character == '\\' || isSpecial(character)) {
62 escaped += '\\';
63 }
64 escaped += character;
65 }
66 }
67 return escaped;
68}

86/// @brief Formats a menu's entries: `[a,b]`, each entry its label followed by the attributes that differ from the
87/// default, as `;icon=x`, `;keys=x`, `;checked=true` or `;checked=false`, and `;enabled=false`, then its
88/// submenu in brackets. A flat menu of plain entries reads `[Open,Quit]`.
89/// @param entries The entries.
90/// @return Their text; `=`, `,`, `[`, `]` and `;` in a label or attribute are escaped.
91// NOLINTNEXTLINE(misc-no-recursion): a submenu is formatted as a menu one level down.
92[[nodiscard]] inline std::string formatMenu(std::vector<MenuEntry> const& entries) {
93 auto const text = [](std::string_view value) {
94 return escape(value, [](char character) { return std::string_view{"=,[];"}.contains(character); });
95 };
96 std::string formatted = "[";
97 for (MenuEntry const& entry : entries) {
98 formatted += formatted.size() == 1 ? "" : ",";
99 formatted += text(entry.label);
100 if (!entry.icon.empty()) {
101 formatted += ";icon=" + text(entry.icon);
102 }
103 if (!entry.keys.empty()) {
104 formatted += ";keys=" + text(entry.keys);
105 }
106 if (entry.checked.has_value()) {
107 formatted += std::string{";checked="} + (*entry.checked ? "true" : "false");
108 }
109 if (!entry.enabled) {
110 formatted += ";enabled=false";
111 }
112 if (!entry.items.empty()) {
113 formatted += formatMenu(entry.items);
114 }
115 }
116 return formatted + "]";
117}

119/// @brief Formats a key: an integer in decimal, a string in double quotes, so `7` and `"7"` differ.
120/// @param key The key.
121/// @return Its text. A string key is escaped as `formatItem` escapes, and a double quote in it is preceded by a
122/// backslash.
123[[nodiscard]] inline std::string formatKey(Key const& key) {
124 if (auto const* const number = std::get_if<std::int64_t>(&key); number != nullptr) {
125 return std::to_string(*number);
126 }
127 return '"' +
128 escape(std::get<std::string>(key),
129 [](char character) { return std::string_view{"=,[]\""}.contains(character); }) +
130 '"';
131}

511/// @brief The records, the roots and the operation log that every fake widget of one backend writes to.
512///
513/// Held by `shared_ptr` from the backend and from every fake, so a fake that outlives its backend still has
514/// somewhere to record its destruction.
515class RecordStore {
516public:
517 /// @brief Records a new widget appended to @p parent and logs its creation.
518 /// @param kind The kind shown in the log and the dump.
519 /// @param parent The container it is appended to, or null for a root.
520 /// @param widget The fake widget.
521 /// @param initial Properties fixed at creation; recorded without a log line.
522 /// @return The new widget's id.
523 int create(std::string kind, ContainerWidget const* parent, Widget& widget, PropList initial) {
524 int const widgetId = ++_nextId;
525 int const parentId = parent == nullptr ? 0 : idOf(*parent);
526 Record record{.id = widgetId, .kind = std::move(kind), .parent = parentId, .widget = &widget};
527 for (auto& [name, value] : initial) {
528 record.props.insert_or_assign(std::move(name), std::move(value));
529 }
530 _log.push_back("create " + label(record) + " in " +
531 (parentId == 0 ? std::string{"root"} : label(at(parentId))));
532 siblingsOf(parentId).push_back(widgetId);
533 _ids.insert_or_assign(&widget, widgetId);
534 _records.insert_or_assign(widgetId, std::move(record));
535 return widgetId;
536 }

556 /// @brief Moves a widget among its siblings, as `ContainerWidget::moveChild` defines it, and logs it.
557 /// @param widgetId The widget.
558 /// @param index The new position; past the end means last.
559 // An id and a position: an `int` and a `std::size_t`, which convert into each other.
560 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
561 void move(int widgetId, std::size_t index) {
562 Record const& record = at(widgetId);
563 std::vector<int>& siblings = siblingsOf(record.parent);
564 std::erase(siblings, widgetId);
565 auto const position = static_cast<std::ptrdiff_t>(std::min(index, siblings.size()));
566 siblings.insert(siblings.begin() + position, widgetId);
567 _log.push_back("move " + label(record) + " to " + std::to_string(index));
568 }

570 /// @brief Forgets a destroyed widget and logs it. Its remaining children become roots.
571 /// @param widgetId The widget; an unknown id does nothing.
572 void destroy(int widgetId) {
573 auto const found = _records.find(widgetId);
574 if (found == _records.end()) {
575 return;
576 }
577 Record const& record = found->second;
578 int const parentId = record.parent;
579 _log.push_back("destroy " + label(record));
580 std::erase(siblingsOf(parentId), widgetId);
581 for (int const child : record.children) {
582 if (auto const orphan = _records.find(child); orphan != _records.end()) {
583 orphan->second.parent = 0;
584 _roots.push_back(child);
585 }
586 }
587 if (_focused == widgetId) {
588 _focused = 0;
589 }
590 _ids.erase(record.widget);
591 _records.erase(found);
592 if (auto const parent = _records.find(parentId); parent != _records.end()) {
593 if (auto const notify = parent->second.onChildDestroyed; notify) {
594 notify(widgetId);
595 }
596 }
597 }

675 // A widget tree is dumped by descending it. The id and the depth are an `int` and a `std::size_t`, which convert
676 // into each other.
677 // NOLINTNEXTLINE(misc-no-recursion,bugprone-easily-swappable-parameters)

691 }
692 }

2207 // A user reaches a widget that is shown and enabled, inside containers that are all shown, enabled and not
2208 // collapsed. A collapsed panel's own collapse control stays reachable, so only an outer panel's state counts. A
2209 // dialog shows nothing, itself included, until it is opened.
2210 [[nodiscard]] bool reachable(detail::Record const& record) const {
2211 if (flagIs(record, "visible", false) || flagIs(record, "enabled", false) || isClosedDialog(record)) {
2212 return false;
2213 }
2214 for (int innerId = record.id, outerId = record.parent; outerId != 0;) {
2215 detail::Record const& outer = _store->at(outerId);
2216 if (flagIs(outer, "visible", false) || flagIs(outer, "enabled", false) ||
2217 flagIs(outer, "collapsed", true) || isClosedDialog(outer) || closedAround(outer, innerId)) {
2218 return false;
