43/// @brief Escapes text with a backslash: a line break becomes backslash and `n`, a carriage return backslash and `r`,
44/// and a backslash or a character @p isSpecial selects is preceded by a backslash.
45/// @tparam Special A callable taking a `char` and returning whether to escape it.
46/// @param text The text.
47/// @param isSpecial Selects the characters to escape besides the backslash and the two line-ending characters.
48/// @return The escaped text.
49template <typename Special>
50[[nodiscard]] std::string escape(std::string_view text, Special const& isSpecial) {
51 std::string escaped;
52 escaped.reserve(text.size());
53 for (char const character : text) {
54 if (character == '\n') {
55 escaped += "\\n";
56 } else if (character == '\r') {
57 escaped += "\\r";
58 } else {
59 if (character == '\\' || isSpecial(character)) {
60 escaped += '\\';
61 }
62 escaped += character;
63 }
64 }
65 return escaped;
66}

68/// @brief Formats free text as a property value. Line breaks and `=` are escaped, so the value stays on its own line
69/// and no part of it reads as another `name=value`.
70/// @param text The text, UTF-8.
71/// @return The escaped text.
72[[nodiscard]] inline std::string formatText(std::string_view text) {
73 return escape(text, [](char character) { return character == '='; });
74}

76/// @brief Formats free text as an element of a `formatList` list. Also escapes `,`, `[` and `]`, so the text reads as
77/// one element.
78/// @param text The text, UTF-8.
79/// @return The escaped text.
80[[nodiscard]] inline std::string formatItem(std::string_view text) {
81 return escape(text, [](char character) { return std::string_view{"=,[]"}.contains(character); });
82}

84/// @brief Formats a key: an integer in decimal, a string in double quotes, so `7` and `"7"` differ.
85/// @param key The key.
86/// @return Its text. A string key is escaped as `formatItem` escapes, and a double quote in it is preceded by a
87/// backslash.
88[[nodiscard]] inline std::string formatKey(Key const& key) {
89 if (auto const* const number = std::get_if<std::int64_t>(&key); number != nullptr) {
90 return std::to_string(*number);
91 }
92 return '"' +
93 escape(std::get<std::string>(key),
94 [](char character) { return std::string_view{"=,[]\""}.contains(character); }) +
95 '"';
96}

301/// @brief One fake widget's state.
302struct Record {
303 /// @brief The widget's id: its creation number, from 1.
304 int id = 0;
305 /// @brief The kind shown in the log and the dump.
306 std::string kind;
307 /// @brief The parent's id; 0 for a root.
308 int parent = 0;
309 /// @brief The children's ids, in order.
310 std::vector<int> children;
311 /// @brief Every property set so far, formatted, in name order.
312 std::map<std::string, std::string, std::less<>> props;
313 /// @brief The handlers set so far.
314 Callbacks callbacks;
315 /// @brief The drag key, as `setDragKey` last set it.
316 std::optional<Key> dragKey;
317 /// @brief The fake widget object.
318 Widget* widget = nullptr;
319 /// @brief Called with a child's id once that child is destroyed; a fake table re-marks its selection with it.
320 std::function<void(int)> onChildDestroyed;
321};

323/// @brief The records, the roots and the operation log that every fake widget of one backend writes to.
324///
325/// Held by `shared_ptr` from the backend and from every fake, so a fake that outlives its backend still has
326/// somewhere to record its destruction.
327class RecordStore {
328public:
329 /// @brief Records a new widget appended to @p parent and logs its creation.
330 /// @param kind The kind shown in the log and the dump.
331 /// @param parent The container it is appended to, or null for a root.
332 /// @param widget The fake widget.
333 /// @param initial Properties fixed at creation; recorded without a log line.
334 /// @return The new widget's id.
335 int create(std::string kind, ContainerWidget const* parent, Widget& widget, PropList initial) {
336 int const widgetId = ++_nextId;
337 int const parentId = parent == nullptr ? 0 : idOf(*parent);
338 Record record{.id = widgetId, .kind = std::move(kind), .parent = parentId, .widget = &widget};
339 for (auto& [name, value] : initial) {
340 record.props.insert_or_assign(std::move(name), std::move(value));
341 }
342 _log.push_back("create " + label(record) + " in " +
343 (parentId == 0 ? std::string{"root"} : label(at(parentId))));
344 siblingsOf(parentId).push_back(widgetId);
345 _ids.insert_or_assign(&widget, widgetId);
346 _records.insert_or_assign(widgetId, std::move(record));
347 return widgetId;
348 }
349
350 /// @brief Sets a property through a widget setter, and logs it.
351 /// @param widgetId The widget.
352 /// @param name The property.
353 /// @param value Its formatted value.
354 void set(int widgetId, std::string_view name, std::string value) {
355 Record& record = at(widgetId);
356 _log.push_back("set " + label(record) + " " + std::string{name} + "=" + value);
357 record.props.insert_or_assign(std::string{name}, std::move(value));
358 }
359
360 /// @brief Sets a property the user changed through the widget itself; no setter ran, so nothing is logged.
361 /// @param widgetId The widget.
362 /// @param name The property.
363 /// @param value Its formatted value.
364 void setByUser(int widgetId, std::string_view name, std::string value) {
365 at(widgetId).props.insert_or_assign(std::string{name}, std::move(value));
366 }
367
368 /// @brief Moves a widget among its siblings, as `ContainerWidget::moveChild` defines it, and logs it.
369 /// @param widgetId The widget.
370 /// @param index The new position; past the end means last.
371 // An id and a position: an `int` and a `std::size_t`, which convert into each other.
372 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
373 void move(int widgetId, std::size_t index) {
374 Record const& record = at(widgetId);
375 std::vector<int>& siblings = siblingsOf(record.parent);
376 std::erase(siblings, widgetId);
377 auto const position = static_cast<std::ptrdiff_t>(std::min(index, siblings.size()));
378 siblings.insert(siblings.begin() + position, widgetId);
379 _log.push_back("move " + label(record) + " to " + std::to_string(index));
380 }
381
382 /// @brief Forgets a destroyed widget and logs it. Its remaining children become roots.
383 /// @param widgetId The widget; an unknown id does nothing.
384 void destroy(int widgetId) {
385 auto const found = _records.find(widgetId);
386 if (found == _records.end()) {
387 return;
388 }
389 Record const& record = found->second;
390 int const parentId = record.parent;
391 _log.push_back("destroy " + label(record));
392 std::erase(siblingsOf(parentId), widgetId);
393 for (int const child : record.children) {
394 if (auto const orphan = _records.find(child); orphan != _records.end()) {
395 orphan->second.parent = 0;
396 _roots.push_back(child);
397 }
398 }
399 _ids.erase(record.widget);
400 _records.erase(found);
401 if (auto const parent = _records.find(parentId); parent != _records.end()) {
402 if (auto const notify = parent->second.onChildDestroyed; notify) {
403 notify(widgetId);
404 }
405 }
406 }
407
408 /// @brief A widget's record.
409 /// @param widgetId The widget.
410 /// @return The record.
411 /// @throws std::out_of_range for an id that names no live widget.
412 [[nodiscard]] Record& at(int widgetId) {
413 auto const found = _records.find(widgetId);
414 if (found == _records.end()) {
415 throw std::out_of_range{"RecordingBackend: no widget #" + std::to_string(widgetId)};
416 }
417 return found->second;
418 }
419
420 /// @brief A widget's record.
421 /// @param widgetId The widget.
422 /// @return The record.
423 /// @throws std::out_of_range for an id that names no live widget.
424 [[nodiscard]] Record const& at(int widgetId) const {
425 auto const found = _records.find(widgetId);
426 if (found == _records.end()) {
427 throw std::out_of_range{"RecordingBackend: no widget #" + std::to_string(widgetId)};
428 }
429 return found->second;
430 }
431
432 /// @brief The id of a live widget made by this store's backend.
433 /// @param widget The widget.
434 /// @return Its id.
435 /// @throws std::out_of_range when @p widget is not one of them.
436 [[nodiscard]] int idOf(Widget const& widget) const {
437 auto const found = _ids.find(&widget);
438 if (found == _ids.end()) {
439 throw std::out_of_range{"RecordingBackend: not a live widget of this backend"};
440 }
441 return found->second;
442 }
443
444 /// @brief Every live widget's record.
445 /// @return The records, by id.
446 [[nodiscard]] std::map<int, Record> const& records() const noexcept { return _records; }
447
448 /// @brief The operation log.
449 /// @return One line per operation, oldest first.
450 [[nodiscard]] std::vector<std::string> const& log() const noexcept { return _log; }
451
452 /// @brief Empties the operation log.
453 void clearLog() noexcept { _log.clear(); }
454
455 /// @brief The golden tree.
456 /// @return One line per live widget, children indented two spaces per depth.
457 [[nodiscard]] std::string dump() const {
458 std::string text;
459 for (int const root : _roots) {
460 dumpInto(text, root, 0);
461 }
462 return text;
463 }
464
465private:
466 [[nodiscard]] static std::string label(Record const& record) {
467 return record.kind + "#" + std::to_string(record.id);
468 }
469
470 std::vector<int>& siblingsOf(int parentId) { return parentId == 0 ? _roots : at(parentId).children; }
471
472 // A widget tree is dumped by descending it. The id and the depth are an `int` and a `std::size_t`, which convert
473 // into each other.
474 // NOLINTNEXTLINE(misc-no-recursion,bugprone-easily-swappable-parameters)
475 void dumpInto(std::string& text, int widgetId, std::size_t depth) const {
476 Record const& record = at(widgetId);
477 text.append(depth * 2, ' ');
478 text += label(record);
479 for (auto const& [name, value] : record.props) {
480 text += ' ';
481 text += name;
482 text += '=';
483 text += value;
484 }
485 text += '\n';
486 for (int const child : record.children) {
487 dumpInto(text, child, depth + 1);
488 }
489 }
490
491 std::map<int, Record> _records;
492 std::unordered_map<Widget const*, int> _ids;
493 std::vector<int> _roots;
494 std::vector<std::string> _log;
495 int _nextId = 0;
496};

498/// @brief The `Widget` half of every fake: records itself on construction, records `Common`'s setters, and records
499/// its destruction.
500/// @tparam Interface The widget interface the fake implements.
501template <typename Interface>
502class FakeLeaf : public Interface {
503public:
504 /// @brief Records the new widget, appended to @p parent, and logs its creation.
505 /// @param store Where the widget records itself.
506 /// @param kind The kind shown in the log and the dump.
507 /// @param parent The container it is appended to, or null for a root.
508 /// @param initial Properties fixed at creation.
509 FakeLeaf(std::shared_ptr<RecordStore> store, std::string kind, ContainerWidget const* parent,
510 PropList initial = {})
511 : _store{std::move(store)}, _id{_store->create(std::move(kind), parent, *this, std::move(initial))} {}
512
513 /// @brief Forgets the widget and logs its destruction; its remaining children become roots.
514 ~FakeLeaf() override {
515 try {
516 _store->destroy(_id);
517 } catch (...) { // NOLINT(bugprone-empty-catch)
518 // Logging the destruction failed to allocate; the record stays, and a test reads it as a leak.
519 }
520 }
521
522 FakeLeaf(FakeLeaf const&) = delete;
523 FakeLeaf& operator=(FakeLeaf const&) = delete;
524 FakeLeaf(FakeLeaf&&) = delete;
525 FakeLeaf& operator=(FakeLeaf&&) = delete;
526
527 /// @brief Records property `visible`.
528 /// @param visible Whether it is shown.
529 void setVisible(bool visible) override { set("visible", formatBool(visible)); }
530
531 /// @brief Records property `enabled`.
532 /// @param enabled Whether it accepts input.
533 void setEnabled(bool enabled) override { set("enabled", formatBool(enabled)); }
534
535 /// @brief Records property `layout`, as `formatLayout` gives it.
536 /// @param hints The request in both dimensions.
537 void setLayout(LayoutHints const& hints) override { set("layout", formatLayout(hints)); }
538
539 /// @brief Records property `dragKey` and keeps the key for `RecordingBackend::drag`.
540 /// @param key Engaged: the key a drop delivers. `nullopt`: not draggable.
541 void setDragKey(std::optional<Key> const& key) override {
542 _store->at(_id).dragKey = key;
543 set("dragKey", formatOptionalKey(key));
544 }
545
546 /// @brief Records property `drop=handler` and keeps both handlers for `RecordingBackend::drag`.
547 /// @param accepts Whether a dragged key may be dropped here.
548 /// @param onDrop Called with the key on a drop that `accepts` allowed.
549 void setDropHandler(std::function<bool(Key const&)> accepts, std::function<void(Key)> onDrop) override {
550 Callbacks& handlers = callbacks();
551 handlers.accepts = std::move(accepts);
552 handlers.drop = std::move(onDrop);
553 set("drop", "handler");
554 }
555
556protected:
557 /// @brief Records a property set through a setter.
558 /// @param name The property.
559 /// @param value Its formatted value.
560 void set(std::string_view name, std::string value) { _store->set(_id, name, std::move(value)); }
561
562 /// @brief This widget's handlers.
563 /// @return The handlers, for a setter to store into.
564 [[nodiscard]] Callbacks& callbacks() { return _store->at(_id).callbacks; }
565
566 /// @brief The store this widget records into.
567 /// @return The store.
568 [[nodiscard]] RecordStore& store() const noexcept { return *_store; }
569
570 /// @brief This widget's id.
571 /// @return The id.
572 [[nodiscard]] int id() const noexcept { return _id; }
573
574private:
575 std::shared_ptr<RecordStore> _store;
576 int _id;
577};

1391 // A user reaches a widget that is shown and enabled, inside containers that are all shown, enabled and not
1392 // collapsed. A collapsed panel's own collapse control stays reachable, so only an outer panel's state counts. A
1393 // dialog shows nothing, itself included, until it is opened.
1394 [[nodiscard]] bool reachable(detail::Record const& record) const {
1395 if (flagIs(record, "visible", false) || flagIs(record, "enabled", false) || isClosedDialog(record)) {
1396 return false;
1397 }
1398 for (int outerId = record.parent; outerId != 0;) {
1399 detail::Record const& outer = _store->at(outerId);
1400 if (flagIs(outer, "visible", false) || flagIs(outer, "enabled", false) ||
1401 flagIs(outer, "collapsed", true) || isClosedDialog(outer)) {
1402 return false;
1403 }
1404 outerId = outer.parent;
1405 }
1406 return true;
1407 }

1410 return record.kind == "Dialog" && !flagIs(record, "open", true);
