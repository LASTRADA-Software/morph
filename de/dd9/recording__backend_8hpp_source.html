45/// @brief Escapes text with a backslash: a line break becomes backslash and `n`, a carriage return backslash and `r`,
46/// and a backslash or a character @p isSpecial selects is preceded by a backslash.
47/// @tparam Special A callable taking a `char` and returning whether to escape it.
48/// @param text The text.
49/// @param isSpecial Selects the characters to escape besides the backslash and the two line-ending characters.
50/// @return The escaped text.
51template <typename Special>
52[[nodiscard]] std::string escape(std::string_view text, Special const& isSpecial) {
53 std::string escaped;
54 escaped.reserve(text.size());
55 for (char const character : text) {
56 if (character == '\n') {
57 escaped += "\\n";
58 } else if (character == '\r') {
59 escaped += "\\r";
60 } else {
61 if (character == '\\' || isSpecial(character)) {
62 escaped += '\\';
63 }
64 escaped += character;
65 }
66 }
67 return escaped;
68}

70/// @brief Formats free text as a property value. Line breaks and `=` are escaped, so the value stays on its own line

86/// @brief Formats a menu's entries: `[a,b]`, each entry its label followed by the attributes that differ from the
87/// default, as `;icon=x`, `;keys=x`, `;checked=true` or `;checked=false`, and `;enabled=false`, then its
88/// submenu in brackets. A flat menu of plain entries reads `[Open,Quit]`.
89/// @param entries The entries.
90/// @return Their text; `=`, `,`, `[`, `]` and `;` in a label or attribute are escaped.
91// NOLINTNEXTLINE(misc-no-recursion): a submenu is formatted as a menu one level down.
92[[nodiscard]] inline std::string formatMenu(std::vector<MenuEntry> const& entries) {
93 auto const text = [](std::string_view value) {
94 return escape(value, [](char character) { return std::string_view{"=,[];"}.contains(character); });
95 };
96 std::string formatted = "[";
97 for (MenuEntry const& entry : entries) {
98 formatted += formatted.size() == 1 ? "" : ",";
99 formatted += text(entry.label);
100 if (!entry.icon.empty()) {
101 formatted += ";icon=" + text(entry.icon);
102 }
103 if (!entry.keys.empty()) {
104 formatted += ";keys=" + text(entry.keys);
105 }
106 if (entry.checked.has_value()) {
107 formatted += std::string{";checked="} + (*entry.checked ? "true" : "false");
108 }
109 if (!entry.enabled) {
110 formatted += ";enabled=false";
111 }
112 if (!entry.items.empty()) {
113 formatted += formatMenu(entry.items);
114 }
115 }
116 return formatted + "]";
117}

119/// @brief Formats a key: an integer in decimal, a string in double quotes, so `7` and `"7"` differ.
120/// @param key The key.
121/// @return Its text. A string key is escaped as `formatItem` escapes, and a double quote in it is preceded by a
122/// backslash.
123[[nodiscard]] inline std::string formatKey(Key const& key) {
124 if (auto const* const number = std::get_if<std::int64_t>(&key); number != nullptr) {
125 return std::to_string(*number);
126 }
127 return '"' +
128 escape(std::get<std::string>(key),
129 [](char character) { return std::string_view{"=,[]\""}.contains(character); }) +
130 '"';
131}

176/// @brief Formats a list as `[a,b,c]`.
177/// @tparam T The element type.
178/// @tparam F A callable turning a `T const&` into a `std::string`.
179/// @param items The elements.
180/// @param each Formats one element.
181/// @return The list's text.
182template <typename T, typename F>
183[[nodiscard]] std::string formatList(std::vector<T> const& items, F const& each) {
184 std::string text = "[";

188 text += ',';

191 text += each(item);
192 }
193 return text + "]";

410/// @brief One fake widget's state.
411struct Record {
412 /// @brief The widget's id: its creation number, from 1.
413 int id = 0;
414 /// @brief The kind shown in the log and the dump.
415 std::string kind;
416 /// @brief The parent's id; 0 for a root.
417 int parent = 0;
418 /// @brief The children's ids, in order.
419 std::vector<int> children;
420 /// @brief Every property set so far, formatted, in name order.
421 std::map<std::string, std::string, std::less<>> props;
422 /// @brief The handlers set so far.
423 Callbacks callbacks;
424 /// @brief The drag key, as `setDragKey` last set it.
425 std::optional<Key> dragKey;
426 /// @brief The chords, as `setKeys` last set them.
427 std::vector<std::string> chords;
428 /// @brief A menu's entries, as `setItems` last set them.
429 std::vector<MenuEntry> menu;
430 /// @brief The fake widget object.
431 Widget* widget = nullptr;
432 /// @brief Called with a child's id once that child is destroyed; a fake table re-marks its selection with it.
433 std::function<void(int)> onChildDestroyed;
434};

436/// @brief The records, the roots and the operation log that every fake widget of one backend writes to.
437///
438/// Held by `shared_ptr` from the backend and from every fake, so a fake that outlives its backend still has
439/// somewhere to record its destruction.
440class RecordStore {
441public:
442 /// @brief Records a new widget appended to @p parent and logs its creation.
443 /// @param kind The kind shown in the log and the dump.
444 /// @param parent The container it is appended to, or null for a root.
445 /// @param widget The fake widget.
446 /// @param initial Properties fixed at creation; recorded without a log line.
447 /// @return The new widget's id.
448 int create(std::string kind, ContainerWidget const* parent, Widget& widget, PropList initial) {
449 int const widgetId = ++_nextId;
450 int const parentId = parent == nullptr ? 0 : idOf(*parent);
451 Record record{.id = widgetId, .kind = std::move(kind), .parent = parentId, .widget = &widget};
452 for (auto& [name, value] : initial) {
453 record.props.insert_or_assign(std::move(name), std::move(value));
454 }
455 _log.push_back("create " + label(record) + " in " +
456 (parentId == 0 ? std::string{"root"} : label(at(parentId))));
457 siblingsOf(parentId).push_back(widgetId);
458 _ids.insert_or_assign(&widget, widgetId);
459 _records.insert_or_assign(widgetId, std::move(record));
460 return widgetId;
461 }
462
463 /// @brief Sets a property through a widget setter, and logs it.
464 /// @param widgetId The widget.
465 /// @param name The property.
466 /// @param value Its formatted value.
467 void set(int widgetId, std::string_view name, std::string value) {
468 Record& record = at(widgetId);
469 _log.push_back("set " + label(record) + " " + std::string{name} + "=" + value);
470 record.props.insert_or_assign(std::string{name}, std::move(value));
471 }
472
473 /// @brief Sets a property the user changed through the widget itself; no setter ran, so nothing is logged.
474 /// @param widgetId The widget.
475 /// @param name The property.
476 /// @param value Its formatted value.
477 void setByUser(int widgetId, std::string_view name, std::string value) {
478 at(widgetId).props.insert_or_assign(std::string{name}, std::move(value));
479 }
480
481 /// @brief Moves a widget among its siblings, as `ContainerWidget::moveChild` defines it, and logs it.
482 /// @param widgetId The widget.
483 /// @param index The new position; past the end means last.
484 // An id and a position: an `int` and a `std::size_t`, which convert into each other.
485 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
486 void move(int widgetId, std::size_t index) {
487 Record const& record = at(widgetId);
488 std::vector<int>& siblings = siblingsOf(record.parent);
489 std::erase(siblings, widgetId);
490 auto const position = static_cast<std::ptrdiff_t>(std::min(index, siblings.size()));
491 siblings.insert(siblings.begin() + position, widgetId);
492 _log.push_back("move " + label(record) + " to " + std::to_string(index));
493 }
494
495 /// @brief Forgets a destroyed widget and logs it. Its remaining children become roots.
496 /// @param widgetId The widget; an unknown id does nothing.
497 void destroy(int widgetId) {
498 auto const found = _records.find(widgetId);
499 if (found == _records.end()) {
500 return;
501 }
502 Record const& record = found->second;
503 int const parentId = record.parent;
504 _log.push_back("destroy " + label(record));
505 std::erase(siblingsOf(parentId), widgetId);
506 for (int const child : record.children) {
507 if (auto const orphan = _records.find(child); orphan != _records.end()) {
508 orphan->second.parent = 0;
509 _roots.push_back(child);
510 }
511 }
512 if (_focused == widgetId) {
513 _focused = 0;
514 }
515 _ids.erase(record.widget);
516 _records.erase(found);
517 if (auto const parent = _records.find(parentId); parent != _records.end()) {
518 if (auto const notify = parent->second.onChildDestroyed; notify) {
519 notify(widgetId);
520 }
521 }
522 }

583 /// @brief The golden tree.
584 /// @return One line per live widget, children indented two spaces per depth.
585 [[nodiscard]] std::string dump() const {
586 std::string text;
587 for (int const root : _roots) {

593private:
594 [[nodiscard]] static std::string label(Record const& record) {
595 return record.kind + "#" + std::to_string(record.id);
596 }
597
598 std::vector<int>& siblingsOf(int parentId) { return parentId == 0 ? _roots : at(parentId).children; }
599
600 // A widget tree is dumped by descending it. The id and the depth are an `int` and a `std::size_t`, which convert
601 // into each other.
602 // NOLINTNEXTLINE(misc-no-recursion,bugprone-easily-swappable-parameters)
603 void dumpInto(std::string& text, int widgetId, std::size_t depth) const {
604 Record const& record = at(widgetId);
605 text.append(depth * 2, ' ');
606 text += label(record);
607 for (auto const& [name, value] : record.props) {
608 text += ' ';
609 text += name;
610 text += '=';
611 text += value;
612 }
613 text += '\n';
614 for (int const child : record.children) {
615 dumpInto(text, child, depth + 1);
616 }
617 }

620 std::unordered_map<Widget const*, int> _ids;
621 std::vector<int> _roots;
622 std::vector<std::string> _log;
623 int _nextId = 0;
624 int _focused = 0;
625};

627/// @brief The `Widget` half of every fake: records itself on construction, records `Common`'s setters, and records
628/// its destruction.
629/// @tparam Interface The widget interface the fake implements.
630template <typename Interface>
631class FakeLeaf : public Interface {
632public:
633 /// @brief Records the new widget, appended to @p parent, and logs its creation.
634 /// @param store Where the widget records itself.
635 /// @param kind The kind shown in the log and the dump.
636 /// @param parent The container it is appended to, or null for a root.
637 /// @param initial Properties fixed at creation.
638 FakeLeaf(std::shared_ptr<RecordStore> store, std::string kind, ContainerWidget const* parent,
639 PropList initial = {})
640 : _store{std::move(store)}, _id{_store->create(std::move(kind), parent, *this, std::move(initial))} {}
641
642 /// @brief Forgets the widget and logs its destruction; its remaining children become roots.
643 ~FakeLeaf() override {
644 try {
645 _store->destroy(_id);
646 } catch (...) { // NOLINT(bugprone-empty-catch)
647 // Logging the destruction failed to allocate; the record stays, and a test reads it as a leak.
648 }
649 }

1909 // A user reaches a widget that is shown and enabled, inside containers that are all shown, enabled and not
1910 // collapsed. A collapsed panel's own collapse control stays reachable, so only an outer panel's state counts. A
1911 // dialog shows nothing, itself included, until it is opened.
1912 [[nodiscard]] bool reachable(detail::Record const& record) const {
1913 if (flagIs(record, "visible", false) || flagIs(record, "enabled", false) || isClosedDialog(record)) {
1914 return false;
1915 }
1916 for (int innerId = record.id, outerId = record.parent; outerId != 0;) {
1917 detail::Record const& outer = _store->at(outerId);
1918 if (flagIs(outer, "visible", false) || flagIs(outer, "enabled", false) ||
1919 flagIs(outer, "collapsed", true) || isClosedDialog(outer) || closedAround(outer, innerId)) {
1920 return false;
1921 }
1922 innerId = outerId;
1923 outerId = outer.parent;
1924 }
1925 return true;
1926 }
1927
1928 // A dialog or drawer shows nothing until it is opened.
1929 [[nodiscard]] static bool isClosedDialog(detail::Record const& record) {
1930 return (record.kind == "Dialog" || record.kind == "Drawer") && !flagIs(record, "open", true);
1931 }

1933 // A closed collapsible section hides its content, the child on the way in, unless that child is its header.
1934 [[nodiscard]] static bool closedAround(detail::Record const& outer, int childId) {
1935 if (outer.kind != "Collapsible" || !flagIs(outer, "open", false)) {
1936 return false;
1937 }
1938 auto const header = outer.props.find("header");
1939 return header == outer.props.end() || header->second != std::to_string(childId);
1940 }
1941
1942 [[nodiscard]] static bool flagIs(detail::Record const& record, std::string_view name, bool value) {
1943 auto const found = record.props.find(name);
1944 return found != record.props.end() && found->second == detail::formatBool(value);
1945 }

1947 // The copy keeps the handler alive while it runs, even when it destroys the widget it was stored on.
1948 template <typename... Args, typename... Values>
1949 static void invoke(std::function<void(Args...)> const& handler, Values&&... values) {
1950 auto const copy = handler;
1951 if (copy) {
1952 copy(std::forward<Values>(values)...);
1953 }
