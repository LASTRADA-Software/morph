40/// @brief Resumes coroutines through a `morph::exec::IExecutor`, as the
41/// current executor of each resumption.
42class ExecutorResumer final : public ::core::async::IExecutor, public std::enable_shared_from_this<ExecutorResumer> {
43public:
44 /// @param executor Where every resumption is posted; must outlive them all.
45 explicit ExecutorResumer(::morph::exec::IExecutor& executor MORPH_LIFETIMEBOUND) : _executor{executor} {}
46
47 using ::core::async::IExecutor::submit;
48
49 /// @brief Posts @p handle's resumption to the executor.
50 /// @param handle The coroutine to resume; borrowed.
51 void submit(std::coroutine_handle<> handle) override {
52 _executor.post([self = shared_from_this(), handle] {
53 std::shared_ptr<void> const keep = self;
54 ::core::async::ExecutorScope const scope{*self, &keep, nullptr};
55 handle.resume();
56 });
57 }
58
59 /// @brief Posts @p work's resumption to the executor.
60 /// @param work The coroutine to resume. The spawned task's driver owns its
61 /// frame, so the abandon claim is not taken.
62 void submit(::core::async::ParkedWork work) override { submit(work.resume); }
63
64private:
65 ::morph::exec::IExecutor& _executor;
66};

68/// @brief Suspends and resumes through @p target: how a spawned task's first
69/// step reaches its executor.
70struct HopTo {
71 ::core::async::IExecutor* target;
72
73 // Called through the awaiter by the compiler, so not static.
74 // NOLINTNEXTLINE(readability-convert-member-functions-to-static)
75 [[nodiscard]] constexpr bool await_ready() const noexcept { return false; }
76 void await_suspend(std::coroutine_handle<> awaiting) const { target->submit(awaiting); }
77 void await_resume() const noexcept {}

80/// @brief The detached coroutine `spawn` starts: hops onto the executor, runs
81/// the task there and logs what it lets escape.
82struct SpawnedTask {
83 // The compiler calls every member of a promise through the object, so none
84 // is made static.
85 // NOLINTBEGIN(readability-convert-member-functions-to-static)
86 struct promise_type {
87 [[nodiscard]] SpawnedTask get_return_object() const noexcept { return {}; }
88 [[nodiscard]] std::suspend_never initial_suspend() const noexcept { return {}; }
89 [[nodiscard]] std::suspend_never final_suspend() const noexcept { return {}; }
90 void return_void() const noexcept {}
91 // Only the hop can throw past the body's catch: an executor that cannot
92 // accept one closure, before anything has run.
93 [[noreturn]] void unhandled_exception() const noexcept { std::terminate(); }
94 };
95 // NOLINTEND(readability-convert-member-functions-to-static)
96};
97
98/// @brief The body of `spawn`.
99/// @param resumer The executor adapter; owned here for the task's lifetime.
100/// @param task The task to run.
101/// @return Nothing to hold: the coroutine frees itself when it finishes.
102inline SpawnedTask runSpawned(std::shared_ptr<ExecutorResumer> resumer, ::core::async::Task<void> task) {
103 co_await HopTo{resumer.get()};
104 try {
105 co_await std::move(task);
106 } catch (const std::exception& exc) {
107 ::morph::log::logError("[spawn] task threw: {}", exc.what());
108 } catch (...) {
109 ::morph::log::logError("[spawn] task threw unknown exception");
110 }
111}

115/// @brief Starts @p task detached, with every resumption -- its first step
116/// included -- posted to @p executor.
117///
118/// The entry point from code that is not itself a coroutine: with a
119/// `QtExecutor`, a GUI flow written as a coroutine runs every step on the GUI
120/// thread. Awaits inside @p task resume on @p executor, whichever thread the
121/// awaited operation completed on. An exception @p task lets escape is logged
122/// through `morph::log` and swallowed. Its stop token is never stopped.
123/// @param executor Where the task runs; must outlive it.
124/// @param task The task to run.
125inline void spawn(::morph::exec::IExecutor& executor, ::core::async::Task<void> task) {
126 detail::runSpawned(std::make_shared<detail::ExecutorResumer>(executor), std::move(task));
127}

129/// @brief The awaiter `delay()` returns: suspends until a `TimeoutScheduler`
130/// entry fires, or until a stop withdraws it.
131///
132/// Resumes on the executor the awaiting coroutine was running on (core-cpp's
133/// current executor), or on the scheduler's thread if there was none. With a
134/// stoppable token on the awaiting promise, a stop cancels the scheduler entry
135/// -- releasing its capture at once -- and resumes the coroutine with
136/// `core::async::OperationCancelled` on the same executor, or, with none,
137/// inline on the thread that requested the stop. Exactly one of the timer and the stop resumes it.
138class DelayAwaiter {
139 enum class Outcome : std::uint8_t { Pending, Fired, Cancelled };
140
141 struct Shared;
142
143 struct OnStop {
144 Shared* shared;
145 void operator()() const noexcept { shared->onStop(); }
146 };
147
148 struct Shared : std::enable_shared_from_this<Shared> {
149 std::atomic<Outcome> outcome{Outcome::Pending};
150 std::atomic<bool> armed{false};
151 std::atomic<bool> claimed{false};
152 std::atomic<detail::TimeoutScheduler::Handle> timer{0};
153 std::coroutine_handle<> continuation;
154 ::core::async::ResumeTarget context;
155 detail::TimeoutScheduler* scheduler = nullptr;
156 std::optional<::core::async::StopCallback<OnStop>> stopCallback;
157
158 // seq_cst, as `armed` is: whoever decides and then reads `armed` must
159 // not miss the other side's store (a store-buffering pattern).
160 bool decide(Outcome outcome_) noexcept {
161 auto expected = Outcome::Pending;
162 return outcome.compare_exchange_strong(expected, outcome_, std::memory_order_seq_cst);
163 }
164
165 void resume() const {
166 if (context) {
167 context.submit(::core::async::ParkedWork{.resume = continuation});
168 } else {
169 continuation.resume();
170 }
171 }
172
173 void onStop() noexcept {
174 if (!decide(Outcome::Cancelled)) {
175 return;
176 }
177 try {
178 // The coroutine may be resumed below and free the awaiter;
179 // this keeps the state alive until the callback has returned.
180 auto const keep = shared_from_this();
181 scheduler->cancel(timer.load());
182 if (armed.load(std::memory_order_seq_cst) && !claimed.exchange(true, std::memory_order_acq_rel)) {
183 resume();
184 }
185 } catch (...) { // NOLINT(bugprone-empty-catch): a stop callback must not throw
186 // No executor to resume through: the coroutine stays
187 // suspended until its owner destroys it.
188 }
189 }
190 };
191
192public:
193 /// @param scheduler The scheduler whose entry times the wait; must outlive it.
194 /// @param duration How long to wait at least.
195 DelayAwaiter(detail::TimeoutScheduler& scheduler MORPH_LIFETIMEBOUND, std::chrono::milliseconds duration)
196 : _shared{std::make_shared<Shared>()}, _duration{duration} {
197 _shared->scheduler = &scheduler;
198 }
199
200 /// @brief Withdraws the wait if the frame is destroyed while suspended.
201 ~DelayAwaiter() {
202 _shared->stopCallback.reset();
203 if (_shared->decide(Outcome::Cancelled)) {
204 _shared->scheduler->cancel(_shared->timer.load());
205 }
206 }
207
208 DelayAwaiter(const DelayAwaiter&) = delete;
209 DelayAwaiter& operator=(const DelayAwaiter&) = delete;
210 DelayAwaiter(DelayAwaiter&&) = delete;
211 DelayAwaiter& operator=(DelayAwaiter&&) = delete;
212
213 /// @return Always false: every decision is `await_suspend`'s.
214 // Called through the awaiter by the compiler, so not static.
215 // NOLINTNEXTLINE(readability-convert-member-functions-to-static)
216 [[nodiscard]] constexpr bool await_ready() const noexcept { return false; }
217
218 /// @tparam Promise The awaiting coroutine's promise type.
219 /// @param awaiting The coroutine performing the `co_await`.
220 /// @return False to continue at once, when a stop was already requested;
221 /// true once the timer is armed.
222 template <typename Promise>
223 bool await_suspend(std::coroutine_handle<Promise> awaiting) {
224 // Copied before `armed` is published: from then on a stop may resume
225 // the coroutine and free this awaiter with its frame.
226 auto shared = _shared;
227 auto const duration = _duration;
228 shared->continuation = awaiting;
229 shared->context = ::core::async::ResumeTarget::current();
230 if constexpr (::core::async::HasStopToken<Promise>) {
231 ::core::async::StopToken const token = awaiting.promise().stopToken();
232 if (token.stop_requested()) {
233 shared->outcome.store(Outcome::Cancelled);
234 return false;
235 }
236 if (token.stop_possible()) {
237 shared->stopCallback.emplace(token, OnStop{shared.get()});
238 }
239 }
240 shared->armed.store(true, std::memory_order_seq_cst);
241 if (shared->outcome.load(std::memory_order_seq_cst) == Outcome::Cancelled) {
242 return shared->claimed.exchange(true, std::memory_order_acq_rel);
243 }
244 auto const handle = shared->scheduler->schedule(duration, [shared] {
245 if (shared->decide(Outcome::Fired)) {
246 shared->resume();
247 }
248 });
249 shared->timer.store(handle);
250 // A stop that won between arming and here found no timer to cancel.
251 if (shared->outcome.load() == Outcome::Cancelled) {
252 shared->scheduler->cancel(handle);
253 }
254 return true;
255 }
256
257 /// @throws core::async::OperationCancelled if a stop withdrew the wait.
258 void await_resume() {
259 _shared->stopCallback.reset();
260 if (_shared->outcome.load() == Outcome::Cancelled) {
261 throw ::core::async::OperationCancelled{};
262 }
263 }
264
265private:
266 std::shared_ptr<Shared> _shared;
267 std::chrono::milliseconds _duration;
268};

270/// @brief Suspends the awaiting coroutine for at least @p duration.
271///
272/// `co_await morph::async::delay(scheduler, 50ms)`. Stop-aware; see
273/// `DelayAwaiter`.
274/// @param scheduler The scheduler whose entry times the wait; must outlive it.
275/// @param duration How long to wait at least.

277[[nodiscard]] inline DelayAwaiter delay(detail::TimeoutScheduler& scheduler MORPH_LIFETIMEBOUND,
