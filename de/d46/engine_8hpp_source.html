5/// @file table/engine.hpp
6/// @brief The table engine: sort, filter and view mapping, on and off the owner.
7///
8/// An `Engine` owns a table's view: the sort chain, the filter, and the
9/// mapping between source rows and view rows. It is owner-affine — every
10/// member is called on the owner thread — and toolkit-free: a renderer applies
11/// the `ViewChange` it emits as list-model operations.
12///
13/// **Execution.** Every recomputation is a job over an immutable snapshot of
14/// the rows, the key columns it may reuse, and the compiled sort and filter.
15/// A job never touches the engine. A small table (below
16/// `EngineOptions::smallTable` rows), or an engine with no owner executor,
17/// computes in one step inside the call. Otherwise the job runs on the worker
18/// executor, or, with no worker, in frame-budgeted steps posted to the owner;
19/// `pending()` is true until its result is applied on the owner, and the
20/// previous view stays visible meanwhile. One job runs at a time: a request
21/// that makes the running job's result useless (a new sort, filter or
22/// structural change) stops it and its result is dropped; updated rows wait
23/// for it and are repaired after.
24///
25/// **Changes.** Updated rows re-key only themselves and are merged back into
26/// the order, once per owner turn however many updates arrived. Inserted,
27/// removed and reset rows recompute the view. Work a dropped job had taken
28/// (updated rows, rows owed a `Changed`) is kept for the next one. With
29/// `ReorderPolicy::Deferred` an updated row keeps its place until the view has
30/// been quiet for the settle interval, or until its edit ends.
31///
32/// **Callbacks.** Every handler the engine calls (`onViewChange`, `onPending`,
33/// `onError`, the progress sink, the settle scheduler) may call the engine
34/// again or destroy it: the engine calls out last, or checks that it is still
35/// alive before it goes on.
36///
37/// See `docs/spec/table/engine.md`.

121/// @brief Column order and visibility, kept apart from the engine's view.
122///
123/// At least one column stays visible.
124class ColumnLayout {
125public:
126 /// @brief A layout of @p ids, all visible, in that order.
127 /// @param ids Column ids, the document's default order.
128 explicit ColumnLayout(std::vector<std::string> ids) : _defaults{std::move(ids)} { reset(); }
129
130 /// @brief The columns in display order.
131 /// @return Column ids.
132 [[nodiscard]] std::span<std::string const> order() const noexcept { return _order; }
133
134 /// @brief The visible columns in display order.
135 /// @return Column ids.
136 [[nodiscard]] std::vector<std::string> visible() const {
137 std::vector<std::string> out;
138 for (auto const& columnId : _order) {
139 if (!_hidden.contains(columnId)) {
140 out.push_back(columnId);
141 }
142 }
143 return out;
144 }
145
146 /// @brief Shows or hides a column.
147 /// @param columnId Column columnId.
148 /// @param show `true` to show.
149 /// @return `false` when the column is unknown, or hiding it would leave none visible.
150 bool setVisible(std::string const& columnId, bool show) {
151 if (std::ranges::find(_order, columnId) == _order.end()) {
152 return false;
153 }
154 if (show) {
155 _hidden.erase(columnId);
156 return true;
157 }
158 if (!_hidden.contains(columnId) && visible().size() == 1) {
159 return false;
160 }
161 _hidden.insert(columnId);
162 return true;
163 }
164
165 /// @brief Moves a column to a display position.
166 /// @param columnId Column columnId.
167 /// @param position Position in `order()`, clamped to the end.
168 /// @return `false` when the column is unknown.
169 bool move(std::string const& columnId, std::size_t position) {
170 auto const found = std::ranges::find(_order, columnId);
171 if (found == _order.end()) {
172 return false;
173 }
174 auto moved = std::move(*found);
175 _order.erase(found);
176 _order.insert(_order.begin() + static_cast<std::ptrdiff_t>(std::min(position, _order.size())),
177 std::move(moved));
178 return true;
179 }
180
181 /// @brief Restores the default order with every column visible.
182 void reset() {
183 _order = _defaults;
184 _hidden.clear();
185 }
186
187 /// @brief Adopts a new column list: kept columns keep their place and
188 /// visibility, new ones are appended visible, vanished ones dropped.
189 /// @param ids The new columns, as the new default order.
190 void rebuild(std::vector<std::string> ids) {
191 std::vector<std::string> order;
192 for (auto const& columnId : _order) {
193 if (std::ranges::find(ids, columnId) != ids.end()) {
194 order.push_back(columnId);
195 }
196 }
197 for (auto const& columnId : ids) {
198 if (std::ranges::find(order, columnId) == order.end()) {
199 order.push_back(columnId);
200 }
201 }
202 std::erase_if(_hidden,
203 [&](std::string const& hiddenId) { return std::ranges::find(ids, hiddenId) == ids.end(); });
204 _defaults = std::move(ids);
205 _order = std::move(order);
206 if (!_order.empty() && visible().empty()) {
207 _hidden.erase(_order.front());
208 }
209 }
210
211private:
212 std::vector<std::string> _defaults;
213 std::vector<std::string> _order;
214 std::unordered_set<std::string> _hidden;
215};

277/// @brief One recomputation of a view, runnable in steps.
278///
279/// Phases: build missing keys (or patch the keys of dirty rows), then either
280/// repair the previous order, or filter and sort, then diff against the
281/// previous view. `step` returns at a deadline or a stop request and resumes
282/// where it stopped.
283class ViewJob {
284public:
285 /// @brief Prepares a job.
286 /// @param input Everything it reads.
287 explicit ViewJob(JobInput input) : _in{std::move(input)} {
288 _result.keys = _in.keys;
289 _result.keys.resize(_in.columns.size());
290 }
291
292 /// @brief Runs until done, the deadline, or a stop request.
293 ///
294 /// An exception from an injected service (a collator, date parser or
295 /// comparator) ends the job: it is done, with `failure()` set.
296 /// @param deadline When to yield.
297 /// @param stop A stop request ends the step.
298 /// @return `true` when the job is done.
299 bool step(Deadline deadline, ::core::async::StopToken const& stop) {
300 try {
301 return run(deadline, stop);
302 } catch (std::exception const& error) {
303 _failure = error.what();
304 } catch (...) {
305 _failure = "unknown exception";
306 }
307 _phase = Phase::Done;
308 return true;
309 }
310
311 /// @brief Sets a receiver of (phases done, total) after each phase ends,
312 /// on the thread that runs the step.
313 /// @param listener The receiver.
314 void onPhase(std::function<void(std::size_t, std::size_t)> listener) { _onPhase = std::move(listener); }
315
316 /// @brief Phases done and total, for a progress sink.
317 /// @return (done, total).
318 [[nodiscard]] std::pair<std::size_t, std::size_t> progress() const noexcept {
319 return {static_cast<std::size_t>(_phase), static_cast<std::size_t>(Phase::Done)};
320 }
321
322 /// @brief Whether every phase has run.
323 /// @return `true` when done.
324 [[nodiscard]] bool done() const noexcept { return _phase == Phase::Done; }
325
326 /// @brief Why the job failed: the message of the exception that ended it.
327 /// @return The message, or nothing when the job did not fail.
328 [[nodiscard]] std::optional<std::string> const& failure() const noexcept { return _failure; }
329
330 /// @brief What the job reads.
331 /// @return The input.
332 [[nodiscard]] JobInput const& input() const noexcept { return _in; }
333
334 /// @brief Moves the result out. Valid once `done()`; the input stays readable.
335 /// @return The result.
336 [[nodiscard]] JobResult takeResult() { return std::move(_result); }
337
338private:
339 enum class Phase : std::uint8_t { Keys, Order, Diff, Done };
340
341 bool run(Deadline deadline, ::core::async::StopToken const& stop) {
342 while (_phase != Phase::Done) {
343 if (stop.stop_requested()) {
344 return false;
345 }
346 bool finished = false;
347 switch (_phase) {
348 case Phase::Keys:
349 finished = buildKeys(deadline, stop);
350 break;
351 case Phase::Order:
352 finished = order(deadline, stop);
353 break;
354 case Phase::Diff:
355 diff();
356 finished = true;
357 break;
358 case Phase::Done:
359 default:
360 finished = true;
361 break;
362 }
363 if (!finished) {
364 return false;
365 }
366 advance();
367 if (_onPhase) {
368 auto const [doneUnits, total] = progress();
369 _onPhase(doneUnits, total);
370 }
371 if (_phase != Phase::Done && detail::shouldYield(deadline, stop)) {
372 return false;
373 }
374 }
375 return true;
376 }
377
378 void advance() {
379 switch (_phase) {
380 case Phase::Keys:
381 _phase = Phase::Order;
382 break;
383 case Phase::Order:
384 _phase = Phase::Diff;
385 break;
386 case Phase::Diff:
387 case Phase::Done:
388 default:
389 _phase = Phase::Done;
390 break;
391 }
392 }

394 // Builds every needed column that has no keys, and patches the dirty rows
395 // of every column that has them (on a copy: the old keys may still be
396 // read by the owner or another job).
397 bool buildKeys(Deadline deadline, ::core::async::StopToken const& stop) {
398 MORPH_ZONE("table.keyBuild");
399 auto const rows = _in.snapshot->rowCount();
400 if (!_patched && !_in.dirty.empty()) {
401 for (std::size_t column = 0; column < _result.keys.size(); ++column) {
402 if (_result.keys[column] == nullptr) {
403 continue;
404 }
405 auto copy = std::make_shared<KeyColumn>(*_result.keys[column]);
406 for (auto const row : _in.dirty) {
407 copy->set(row, _in.snapshot->cell(row, column), _in.columns[column], _in.services);
408 }
409 _result.keys[column] = std::move(copy);
410 }
411 _patched = true;
412 }
413 while (_neededAt < _in.needed.size()) {
414 auto const column = _in.needed[_neededAt];
415 if (_result.keys[column] != nullptr && _building == nullptr) {
416 ++_neededAt;
417 continue;
418 }
419 if (_building == nullptr) {
420 _building = std::make_shared<KeyColumn>(KeyColumn::forColumn(_in.columns[column], _in.services, rows));
421 _rowAt = 0;
422 }
423 while (_rowAt < rows) {
424 auto const end = std::min(_rowAt + detail::kCheckEvery, rows);
425 morph::table::buildKeys(*_in.snapshot, column, _in.columns[column], _in.services, _rowAt, end,
426 *_building);
427 _rowAt = end;
428 if (_rowAt < rows && detail::shouldYield(deadline, stop)) {
429 return false;
430 }
431 }
432 _result.keys[column] = std::move(_building);
433 _building = nullptr;
434 ++_result.keyBuilds;
435 ++_neededAt;
436 }
437 return true;
438 }

466 std::vector<KeyColumn const*> out;
467 out.reserve(_result.keys.size());
468 for (auto const& keys : _result.keys) {
469 out.push_back(keys.get());

471 return out;

474 [[nodiscard]] bool passesFilter(std::uint32_t row) const {
475 if (_in.filter == nullptr) {
476 return true;
477 }
478 for (auto const& column : _in.filter->columns()) {
479 if (!CompiledFilter::passes(column, *_result.keys[column.column], row)) {
480 return false;
481 }
482 }
483 for (auto const& group : _in.filter->groups()) {
484 bool const any = std::ranges::any_of(group.columns, [&](CompiledFilter::Column const& column) {
485 return CompiledFilter::passes(column, *_result.keys[column.column], row);
486 });
487 if (!any) {
488 return false;
489 }
490 }
491 return true;
492 }

494 // Merges the dirty rows back into an order the clean rows still satisfy.
495 [[nodiscard]] std::vector<std::uint32_t> repairOrder(std::span<std::uint32_t const> previous,
496 std::vector<std::uint32_t> dirtyRows,
497 RowOrder const& orderBy) const {
498 std::vector<bool> isDirty(_in.snapshot->rowCount(), false);
499 for (auto const row : _in.dirty) {
500 isDirty[row] = true;
501 }
502 std::vector<std::uint32_t> clean;
503 clean.reserve(previous.size());
504 for (auto const row : previous) {
505 if (!isDirty[row]) {
506 clean.push_back(row);
507 }
508 }
509 auto const sortedDirty = sortRows(std::move(dirtyRows), orderBy);
510 std::vector<std::uint32_t> out(clean.size() + sortedDirty.size());
511 std::ranges::merge(clean, sortedDirty, out.begin(), orderBy);
512 return out;
513 }
514
515 // Merges the re-keyed dirty rows into the previous order (and the cached
516 // full order), without sorting the rest.
517 void repair() {
518 MORPH_ZONE("table.change");
519 auto const orderBy = rowOrder();
520 std::vector<std::uint32_t> passing;
521 for (auto const row : _in.dirty) {
522 if (passesFilter(row)) {
523 passing.push_back(row);
524 }
525 }
526 _result.view = repairOrder(_in.previousTrueView, std::move(passing), orderBy);
527 if (_in.cachedSorted != nullptr) {
528 _result.sorted =
529 std::make_shared<std::vector<std::uint32_t> const>(repairOrder(*_in.cachedSorted, _in.dirty, orderBy));
530 }
531 _result.repaired = true;
532 }
533
534 // Chooses the work: filter the cached sorted order (with the dirty rows
535 // merged back into it); or filter, then sort the survivors; or sort every
536 // row and cache the order.
537 void plan() {
538 auto const rows = _in.snapshot->rowCount();
539 auto const filtering = _in.filter != nullptr && !_in.filter->empty();
540 _started = true;
541 if (_in.cachedSorted != nullptr) {
542 if (_in.dirty.empty()) {
543 _base = *_in.cachedSorted;
544 _result.sorted = _in.cachedSorted;
545 } else {
546 _base = repairOrder(*_in.cachedSorted, _in.dirty, rowOrder());
547 _result.sorted = std::make_shared<std::vector<std::uint32_t> const>(_base);
548 }
549 _sorted = true;
550 } else {
551 _base.resize(rows);
552 std::ranges::iota(_base, 0U);
553 _sorted = _in.sort.empty();
554 if (!_sorted && !filtering) {
555 _sort.emplace(std::move(_base), rowOrder());
556 _result.fullSort = true;
557 }
558 }
559 if (filtering && (_sorted || _sort == std::nullopt)) {
560 _filterRun.emplace(*_in.filter, keyPointers(), std::move(_base));
561 }
562 }
563
564 bool order(Deadline deadline, ::core::async::StopToken const& stop) {
565 if (_in.repair) {
566 repair();
567 return true;
568 }
569 if (!_started) {
570 plan();
571 }
572 if (_filterRun) {
573 MORPH_ZONE("table.filter");
574 MORPH_ZONE_TEXT(describeFilter());
575 if (!_filterRun->run(deadline, stop)) {
576 return false;
577 }
578 _base = std::move(*_filterRun).take();
579 _filterRun.reset();
580 if (!_sorted) {
581 // Filter first, then sort only the survivors.
582 _sort.emplace(std::move(_base), rowOrder());
583 _result.subsetSort = true;
584 }
585 }
586 if (_sort) {
587 MORPH_ZONE("table.sort");
588 MORPH_ZONE_TEXT(describeSort());
589 if (!_sort->run(deadline, stop)) {
590 return false;
591 }
592 _base = std::move(*_sort).take();
593 _sort.reset();
594 _sorted = true;
595 if (_result.fullSort) {
596 _result.sorted = std::make_shared<std::vector<std::uint32_t> const>(_base);
597 }
598 }
599 _result.view = std::move(_base);
600 return true;
601 }
602
603 void diff() {
604 MORPH_ZONE("table.apply");
605 std::vector<std::uint32_t> changed = _in.dirty;
606 if (_in.indicesStable) {
607 if (_in.allChanged) {
608 changed = _result.view;
609 }
610 _result.change =
611 diffViews(_in.previousView, _result.view, changed, _in.snapshot->rowCount(), _in.resetThreshold);
612 return;
613 }
614 // Rows of the previous view map to new rows by key; vanished rows get
615 // identities past the new snapshot's rows.
616 auto const rows = _in.snapshot->rowCount();
617 std::unordered_map<RowId, std::uint32_t> byKey;
618 byKey.reserve(rows);
619 for (std::size_t row = 0; row < rows; ++row) {
620 byKey.emplace(_in.snapshot->rowId(row), static_cast<std::uint32_t>(row));
621 }
622 std::vector<std::uint32_t> previous;
623 previous.reserve(_in.previousView.size());
624 auto next = static_cast<std::uint32_t>(rows);
625 for (auto const row : _in.previousView) {
626 auto const found = byKey.find(_in.previousSnapshot->rowId(row));
627 previous.push_back(found != byKey.end() ? found->second : next++);
628 }
629 if (_in.allChanged) {
630 changed = _result.view;
631 }
632 _result.change = diffViews(previous, _result.view, changed, next, _in.resetThreshold);
633 }
634
635 JobInput _in;
636 JobResult _result;
637 Phase _phase = Phase::Keys;
638 // key build
639 bool _patched = false;
640 std::size_t _neededAt = 0;
641 std::size_t _rowAt = 0;
642 std::shared_ptr<KeyColumn> _building;
643 // order
644 bool _started = false;
645 bool _sorted = false;
646 std::vector<std::uint32_t> _base;
647 std::optional<FilterRun> _filterRun;
648 std::optional<ChunkedMergeSort> _sort;
649 // outcome
650 std::optional<std::string> _failure;
651 std::function<void(std::size_t, std::size_t)> _onPhase;
652};

656/// @brief Sorts, filters and maps the view of one table.
657///
658/// Owner-affine; see the file comment for how it executes. Holds a shared
659/// reference to its source and subscribes to its changes for its lifetime.
660class Engine final : private ChangeListener {
661public:

873 // the job's return whether a later change made its result useless.

886 // ── changes ─────────────────────────────────────────────────────────────
887
888 void rowsChanged(RowChange const& change) override {
889 switch (change.kind) {
890 case ChangeKind::Updated:
891 for (auto const row : change.rows) {
892 _notifiedRows.push_back(static_cast<std::uint32_t>(row));
893 }
894 break;
895 case ChangeKind::Reset:
896 _notifiedReset = true;
897 _notifiedStructural = true;
898 break;
899 case ChangeKind::Inserted:
900 case ChangeKind::Removed:
901 default:
902 _notifiedStructural = true;
903 break;
904 }
905 _notified = true;
906 ++_changeSeq;
907 if (_options.owner == nullptr) {
908 flushChanges();
909 return;
910 }
911 if (!_flushScheduled) {
912 _flushScheduled = true;
913 _options.owner->post(_alive.guard([this] { flushChanges(); }));
914 }
915 }
916
917 // Takes in every change notified since the last flush, then asks for the
918 // work it makes: one repair for any number of updates, or a recompute
919 // after a structural change.
920 void flushChanges() {
921 MORPH_ZONE("table.snapshot");
922 _flushScheduled = false;
923 if (!absorb()) {
924 return;
925 }
926 // An error comes back only from a job that ran inside this call, which
927 // has then called nothing out.
928 if (auto const done = schedule(); !done) {
929 reportError(done.error());
930 }
931 }
932
933 // Folds the notified changes into the pending work. Updated rows join the
934 // dirty rows; a structural change drops the keys built for the old row
935 // indices and owes every row a `Changed`.
936 bool absorb() {
937 if (!_notified) {
938 return false;
939 }
940 _notified = false;
941 _absorbedSeq = _changeSeq;
942 _snapshot = _source->snapshot();
943 if (_notifiedReset) {
944 adoptColumns();
945 }
946 if (_notifiedStructural) {
947 ++_epoch;
948 _keys.assign(_columns.size(), nullptr);
949 _keysSnapshot = nullptr;
950 _sortCache = {};
951 _dirty.clear();
952 _obligeAll = true;
953 } else {
954 _dirty.insert(_dirty.end(), _notifiedRows.begin(), _notifiedRows.end());
955 }
956 _notifiedRows.clear();
957 _notifiedStructural = false;
958 _notifiedReset = false;
959 return true;
960 }

962 // Re-reads the columns after a reset; a changed column set keeps the sort
963 // keys and filter entries whose columns remain.
964 void adoptColumns() {
965 std::vector<ColumnInfo> columns(_source->columns().begin(), _source->columns().end());
966 if (columns == _columns) {
967 return;
968 }
969 _columns = std::move(columns);
970 ++_columnsVersion;
971 fitToColumns();
972 }
973
974 void fitToColumns() {
975 std::erase_if(_sort, [&](SortKey const& key) { return !detail::findColumn(_columns, key.column); });
976 std::erase_if(_filterSpec.columns,
977 [&](auto const& entry) { return !detail::findColumn(_columns, entry.first).has_value(); });
978 for (auto& group : _filterSpec.groups) {
979 std::erase_if(group.columns,
980 [&](std::string const& columnId) { return !detail::findColumn(_columns, columnId); });
981 }
982 std::erase_if(_filterSpec.groups, [](GroupFilter const& group) { return group.columns.empty(); });
983 auto compiled = compile(_filterSpec);
984 if (compiled) {
985 _filter = std::make_shared<CompiledFilter const>(std::move(*compiled));
986 } else {
987 // A remaining entry no longer compiles for its column's new kind.
988 _filterSpec = {};
989 _filter = nullptr;
990 }
991 }

993 // compileFilter, with an exception from the collator or date parser as an error.
994 [[nodiscard]] std::expected<CompiledFilter, TableError> compile(FilterSpec const& spec) const {
995 try {
996 return compileFilter(spec, _columns, _options.services);
997 } catch (std::exception const& error) {
998 return std::unexpected(serviceFailed(error.what()));
999 } catch (...) {
1000 return std::unexpected(serviceFailed("unknown exception"));
1001 }

1005 return TableError{.code = TableErrorCode::ServiceFailed, .column = {}, .message = "a service failed: " + what};

1008 // ── scheduling ──────────────────────────────────────────────────────────
1009
1010 // Whether the shown view is out of date with the requested state.
1011 [[nodiscard]] bool hasWork() const {
1012 return !_dirty.empty() || _epoch != _viewEpoch || _sort != _appliedSort || _filter != _appliedFilter ||
1013 _columnsVersion != _viewColumnsVersion;
1014 }
1015
1016 // A repair merges the dirty rows into the shown order: only the rows'
1017 // contents changed since that order was computed. (A structural change
1018 // advances the epoch; every row is owed a `Changed` exactly while the
1019 // epoch is ahead of the view's.)
1020 [[nodiscard]] bool canRepair() const {
1021 return !_dirty.empty() && _epoch == _viewEpoch && _sort == _appliedSort && _filter == _appliedFilter &&
1022 _columnsVersion == _viewColumnsVersion;
1023 }
1024
1025 // Whether a change since @p input was taken makes its result useless.
1026 [[nodiscard]] bool supersedes(detail::JobInput const& input, Ticket const& ticket) const {
1027 return _sort != input.sort || _filter != input.filter || _epoch != ticket.epoch ||
1028 _columnsVersion != ticket.columnsVersion;
1029 }
1030
1031 // Starts the work the requested state needs. While a job runs, a request
1032 // that makes its result useless stops it; either way the work follows it.
1033 std::expected<void, TableError> schedule() {
1034 if (_running) {
1035 if (supersedes(*_runningInput, _runningTicket)) {
1036 _jobStop.request_stop();
1037 }
1038 return {};
1039 }
1040 if (!hasWork()) {
1041 return {};
1042 }
1043 auto const kind = canRepair() ? Request::Repair : Request::Full;
1044 if (_options.owner == nullptr || _snapshot->rowCount() < _options.smallTable) {
1045 return runNow(kind);
1046 }
1047 start(kind);
1048 return {};
1049 }
1050
1051 [[nodiscard]] detail::JobInput makeInput(Request kind, Ticket& ticket) {
1052 detail::JobInput input;
1053 input.snapshot = _snapshot;
1054 input.columns = _columns;
1055 input.services = _options.services.forTask();
1056 input.keys = _keys;
1057 input.keys.resize(_columns.size());
1058 for (auto const& key : _sort) {
1059 if (auto const index = detail::findColumn(_columns, key.column)) {
1060 input.needed.push_back(*index);
1061 }
1062 }
1063 if (_filter != nullptr) {
1064 for (auto const column : _filter->columnsUsed()) {
1065 input.needed.push_back(column);
1066 }
1067 }
1068 std::ranges::sort(input.needed);
1069 auto const [first, last] = std::ranges::unique(input.needed);
1070 input.needed.erase(first, last);
1071 input.sort = _sort;
1072 input.filter = _filter;
1073 // The cached order is of the rows the keys were built from (install
1074 // keeps no other); the job merges the dirty rows back into it.
1075 if (!_sort.empty() && _sortCache.order != nullptr && _sortCache.chain == _sort) {
1076 input.cachedSorted = _sortCache.order;
1077 }
1078 input.previousView = _view;
1079 input.previousSnapshot = _viewSnapshot;
1080 input.repair = kind == Request::Repair;
1081 input.indicesStable = _epoch == _viewEpoch;
1082 input.dirty = std::move(_dirty);
1083 _dirty.clear();
1084 std::ranges::sort(input.dirty);
1085 auto const [dupFirst, dupLast] = std::ranges::unique(input.dirty);
1086 input.dirty.erase(dupFirst, dupLast);
1087 if (input.repair) {
1088 input.previousTrueView = _trueView;
1089 }
1090 input.allChanged = _obligeAll;
1091 _obligeAll = false;
1092 input.resetThreshold = _options.resetThreshold;
1093 ticket = Ticket{.epoch = _epoch, .changeSeq = _absorbedSeq, .columnsVersion = _columnsVersion};
1094 return input;
1095 }

1097 // A job that was dropped or failed hands back the work it took.
1098 void giveBack(detail::JobInput const& input, Ticket const& ticket) {
1099 if (ticket.epoch == _epoch) {
1100 _dirty.insert(_dirty.end(), input.dirty.begin(), input.dirty.end());
1101 }
1102 _obligeAll = _obligeAll || input.allChanged;
1103 }

1105 // A failed job's sort and filter go back to those of the view shown.
1106 void revert() {
1107 _sort = _appliedSort;
1108 _filterSpec = _appliedFilterSpec;
1109 _filter = _appliedFilter;
1110 if (_columnsVersion != _viewColumnsVersion) {
1111 fitToColumns();
1112 }
1113 }

1115 std::expected<void, TableError> runNow(Request kind) {
1116 auto const began = std::chrono::steady_clock::now();
1117 ++_stats.jobs;
1118 Ticket ticket;
1119 detail::ViewJob job{makeInput(kind, ticket)};
1120 static_cast<void>(job.step(kNoDeadline, ::core::async::StopToken{}));
1121 if (job.failure()) {
1122 recordOwnerStep(began);
1123 giveBack(job.input(), ticket);
1124 revert();
1125 return std::unexpected(serviceFailed(*job.failure()));
1126 }
1127 auto const applied = install(job.input(), job.takeResult(), ticket);
1128 recordOwnerStep(began);
1129 deliver(applied);
1130 return {};
1131 }

1133 void start(Request kind) {
1134 _running = true;
1135 ++_stats.jobs;
1136 _jobStop = ::core::async::StopSource{};
1137 auto const job = std::make_shared<detail::ViewJob>(makeInput(kind, _runningTicket));
1138 _runningInput = &job->input();
1139 auto* const owner = _options.owner;
1140 if (_options.worker == nullptr) {
1141 owner->post(_alive.guard([this, job] { ownerStep(job); }));
1142 } else {
1143 // The worker task never names the engine: it holds the job and
1144 // callbacks gated on the engine's lifetime, which run on the owner.
1145 if (auto const progress = _options.services.progress) {
1146 auto report =
1147 _alive.guard([progress](std::size_t done, std::size_t total) { progress->progress(done, total); });
1148 job->onPhase([owner, report = std::move(report)](std::size_t done, std::size_t total) {
1149 owner->post([report, done, total] mutable { report(done, total); });
1150 });
1151 }
1152 auto finished = _alive.guard([this, job] { finish(job); });
1153 _options.worker->post([job, stop = _jobStop.get_token(), owner, finished = std::move(finished)] mutable {
1154 static_cast<void>(job->step(kNoDeadline, stop));
1155 owner->post(std::move(finished));
1156 });
1157 }
1158 // Last: the handler may make a request or destroy the engine.
1159 setPending(true);
1160 }
1161
1162 void ownerStep(std::shared_ptr<detail::ViewJob> const& job) {
1163 auto const began = std::chrono::steady_clock::now();
1164 auto const stop = _jobStop.get_token();
1165 bool const done = job->step(began + _options.frameBudget, stop);
1166 recordOwnerStep(began);
1167 if (auto const progress = _options.services.progress) {
1168 auto const alive = _alive.token();
1169 auto const [doneUnits, total] = job->progress();
1170 progress->progress(doneUnits, total);
1171 if (alive.expired()) {
1172 return;
1173 }
1174 }
1175 if (!done && !stop.stop_requested()) {
1176 _options.owner->post(_alive.guard([this, job] { ownerStep(job); }));
1177 return;
1178 }
1179 finish(job);
1180 }
1181
1182 void finish(std::shared_ptr<detail::ViewJob> const& job) {
1183 _running = false;
1184 _runningInput = nullptr;
1185 auto const& input = job->input();
1186 if (!job->done() || supersedes(input, _runningTicket)) {
1187 ++_stats.dropped;
1188 giveBack(input, _runningTicket);
1189 afterJob(std::nullopt);
1190 return;
1191 }
1192 if (job->failure()) {
1193 giveBack(input, _runningTicket);
1194 revert();
1195 afterJob(serviceFailed(*job->failure()));
1196 return;
1197 }
1198 auto const began = std::chrono::steady_clock::now();
1199 auto const applied = install(input, job->takeResult(), _runningTicket);
1200 recordOwnerStep(began);
1201 if (!deliver(applied)) {
1202 return;
1203 }
1204 afterJob(std::nullopt);
1205 }

1207 // After a job returns: starts the work that remains (not after a failure,
1208 // which would only fail again), lays out holds released meanwhile, ends
1209 // `pending()`, and reports the failure. A handler may have started a job
1210 // already; then `pending()` stays true.
1211 void afterJob(std::optional<TableError> error) {
1212 auto const alive = _alive.token();
1213 if (!error && !_running && hasWork()) {
1214 auto const done = schedule();
1215 if (alive.expired()) {
1216 return;
1217 }
1218 if (!done) {
1219 error = done.error();
1220 }
1221 }
1222 if (_relayoutPending && !_running) {
1223 relayout();
1224 if (alive.expired()) {
1225 return;
1226 }
1227 }
1228 if (!_running) {
1229 setPending(false);
1230 if (alive.expired()) {
1231 return;
1232 }
1233 }
1234 if (error) {
1235 reportError(*error);
1236 }
1237 }

1241 // Makes a finished job's result the shown view. Calls nothing out.
1242 [[nodiscard]] Applied install(detail::JobInput const& input, detail::JobResult result, Ticket const& ticket) {
1243 _stats.keyBuilds += result.keyBuilds;
1244 _stats.fullSorts += result.fullSort ? 1 : 0;
1245 _stats.subsetSorts += result.subsetSort ? 1 : 0;
1246 _stats.repairs += result.repaired ? 1 : 0;
1247 _keys = std::move(result.keys);
1248 _keysSnapshot = input.snapshot;
1249 if (result.sorted != nullptr) {
1250 _sortCache = SortCache{.chain = input.sort, .snapshot = input.snapshot, .order = std::move(result.sorted)};
1251 } else if (_sortCache.snapshot != input.snapshot) {
1252 // An order of older rows: a repair it missed would merge into it
1253 // as if it were current.
1254 _sortCache = {};
1255 }
1256 MORPH_PLOT("table.sourceRows", input.snapshot->rowCount());
1257 MORPH_PLOT("table.viewRows", result.view.size());
1258 MORPH_PLOT("table.keyCacheColumns",
1259 std::ranges::count_if(_keys, [](auto const& keys) { return keys != nullptr; }));
1260 bool const columnsChanged = ticket.columnsVersion != _viewColumnsVersion;
1261 if (columnsChanged) {
1262 _viewColumns = input.columns;
1263 _viewColumnsVersion = ticket.columnsVersion;
1264 }
1265 _viewEpoch = ticket.epoch;
1266 _appliedSort = input.sort;
1267 _appliedFilter = input.filter;
1268 _appliedFilterSpec = _filterSpec;
1269 _trueView = std::move(result.view);
1270 _viewSnapshot = input.snapshot;
1271 _relayoutPending = false;
1272 _viewIndexValid = false;
1273 _keyIndexValid = false;
1274 Applied out{.change = std::move(result.change), .armSettle = false};
1275 if (_options.reorder == ReorderPolicy::Deferred) {
1276 // An ended edit's token stands for the notifications up to its own
1277 // count: one the shown view had already taken in has no use.
1278 std::erase_if(_releaseOnRepair, [&](auto const& entry) { return entry.second <= _viewChangeSeq; });
1279 if (input.repair) {
1280 holdUpdatedRows(input);
1281 } else {
1282 // A sort, filter or structural change releases every held row.
1283 _holds.clear();
1284 }
1285 std::erase_if(_releaseOnRepair, [&](auto const& entry) { return entry.second <= ticket.changeSeq; });
1286 }
1287 _viewChangeSeq = ticket.changeSeq;
1288 if (!_holds.empty()) {
1289 auto displayed = holdLayout(input.previousView);
1290 out.change = diffViews(input.previousView, displayed, input.dirty, input.snapshot->rowCount(),
1291 _options.resetThreshold);
1292 _view = std::move(displayed);
1293 out.armSettle = true;
1294 } else {
1295 _view = _trueView;
1296 }
