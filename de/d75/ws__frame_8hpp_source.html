17/// @brief RFC 6455 §5.2 frame opcodes this reference implementation understands.
18enum class WsOpcode : std::uint8_t {
19 kContinuation = 0x0,
20 kText = 0x1,
21 kBinary = 0x2,
22 kClose = 0x8,
23 kPing = 0x9,
24 kPong = 0xA,
25};

27/// @brief One decoded (unfragmented) WebSocket frame.
28struct WsFrame {
29 /// @brief The frame's opcode.
30 WsOpcode opcode{WsOpcode::kText};
31 /// @brief The frame's (already-unmasked, if it was masked) payload bytes.
32 std::string payload;

36inline std::array<std::uint8_t, 4> randomMaskKey() {
37 static thread_local std::mt19937 gen{std::random_device{}()};
38 std::uniform_int_distribution<int> dist{0, 255};
39 std::array<std::uint8_t, 4> key{};
40 for (auto& b : key) {
41 b = static_cast<std::uint8_t>(dist(gen));
42 }
43 return key;
44}

47/// @brief Encodes one complete (unfragmented, `FIN=1`) WebSocket frame.
48/// @param opcode Frame opcode.
49/// @param payload Payload bytes.
50/// @param mask `true` to mask the frame (required for client-to-server
51/// frames per RFC 6455 §5.1); `false` for server-to-client frames.
52/// @return The wire bytes of the frame, ready to send on the socket.
53inline std::string encodeWsFrame(WsOpcode opcode, std::string_view payload, bool mask) {
54 std::string out;
55 out.push_back(static_cast<char>(0x80u | static_cast<std::uint8_t>(opcode))); // FIN=1, opcode
56 std::uint64_t const len = payload.size();
57 std::uint8_t const maskBit = mask ? 0x80u : 0x00u;
58 if (len <= 125) {
59 out.push_back(static_cast<char>(maskBit | static_cast<std::uint8_t>(len)));
60 } else if (len <= 0xFFFFu) {
61 out.push_back(static_cast<char>(maskBit | 126u));
62 out.push_back(static_cast<char>((len >> 8) & 0xFFu));
63 out.push_back(static_cast<char>(len & 0xFFu));
64 } else {
65 out.push_back(static_cast<char>(maskBit | 127u));
66 for (int shift = 56; shift >= 0; shift -= 8) {
67 out.push_back(static_cast<char>((len >> shift) & 0xFFu));
68 }
69 }
70 if (mask) {
71 auto key = ws_frame_impl::randomMaskKey();
72 for (auto b : key) {
73 out.push_back(static_cast<char>(b));
74 }
75 for (std::size_t i = 0; i < payload.size(); ++i) {
76 out.push_back(static_cast<char>(static_cast<std::uint8_t>(payload[i]) ^ key[i % key.size()]));
77 }
78 } else {
79 out.append(payload);
80 }
81 return out;
82}

84/// @brief Streaming WebSocket message decoder over a byte buffer fed incrementally.
85///
86/// `encodeWsFrame` never fragments what *this* implementation sends — the
87/// 64-bit extended-length field covers up to `wire::kMaxEnvelopeBytes`, so one
88/// `wire::Envelope` always goes out as a single frame. Incoming traffic is a
89/// different matter: a peer fragments according to *its* outgoing frame size,
90/// and Qt's `QWebSocket` defaults that to 512 KiB. `tryExtractFrame` therefore
91/// reassembles continuation frames and hands back only completed messages.
92class WsFrameReader {
93public:
94 /// @brief Appends newly received bytes to the internal buffer.
95 /// @param data Bytes read from the socket.
96 void feed(std::string_view data) { _buf.append(data); }
97
98 /// @brief Attempts to extract one complete *message* from the buffered bytes.
99 ///
100 /// Reassembles fragmented messages (RFC 6455 §5.4): a data message may
101 /// arrive as an unfinished text/binary frame followed by continuation
102 /// frames, with the last carrying FIN. Fragments are accumulated and only
103 /// the completed message is returned, carrying the opcode of its first
104 /// frame. This is not an exotic case — a peer fragments whenever a message
105 /// exceeds its outgoing frame size, and Qt's `QWebSocket` defaults that to
106 /// 512 KiB, so rejecting fragments broke interop with the transport this
107 /// project ships for every payload past that size.
108 ///
109 /// Control frames (close/ping/pong) may be interleaved between the
110 /// fragments of a data message and are returned as they arrive, without
111 /// disturbing the reassembly in progress. Per the RFC they may not
112 /// themselves be fragmented.
113 ///
114 /// @return The completed message if enough bytes are buffered, `std::nullopt`
115 /// otherwise.
116 /// @throws std::runtime_error on a protocol violation (a continuation with
117 /// no message in progress, a new data frame interrupting one, a
118 /// fragmented control frame) or a payload — single frame or
119 /// reassembled total — larger than `wire::kMaxEnvelopeBytes`.
120 std::optional<WsFrame> tryExtractFrame() {
121 for (;;) {
122 auto frame = tryExtractRawFrame();
123 if (!frame) {
124 return std::nullopt;
125 }
126 bool const isControl = (static_cast<std::uint8_t>(frame->opcode) & 0x08U) != 0;
127 if (isControl) {
128 if (!frame->fin) {
129 throw std::runtime_error("WsFrameReader: control frames must not be fragmented");
130 }
131 return WsFrame{.opcode = frame->opcode, .payload = std::move(frame->payload)};
132 }
133
134 if (frame->opcode == WsOpcode::kContinuation) {
135 if (!_assembling) {
136 throw std::runtime_error("WsFrameReader: continuation frame with no message in progress");
137 }
138 appendFragment(frame->payload);
139 if (!frame->fin) {
140 continue; // more fragments to come; keep draining the buffer
141 }
142 WsFrame completed{.opcode = _assemblyOpcode, .payload = std::move(_assembly)};
143 resetAssembly();
144 return completed;
145 }
146
147 // A data frame (text/binary) starts a message.
148 if (_assembling) {
149 throw std::runtime_error("WsFrameReader: new data frame while a fragmented message is in progress");
150 }
151 if (frame->fin) {
152 return WsFrame{.opcode = frame->opcode,
153 .payload = std::move(frame->payload)}; // the common, unfragmented case
154 }
155 _assembling = true;
156 _assemblyOpcode = frame->opcode;
157 _assembly.clear();
158 appendFragment(frame->payload);
159 }
160 }
161
162private:
163 /// @brief One frame exactly as it appeared on the wire, before any
164 /// fragmentation handling.
165 struct RawFrame {
166 bool fin;
167 WsOpcode opcode;
168 std::string payload;
169 };
170
171 void resetAssembly() {
172 _assembling = false;
173 _assembly.clear();
174 _assembly.shrink_to_fit();
175 }
176
177 void appendFragment(const std::string& payload) {
178 // Bound the reassembled total, not just each frame: without this, a
179 // peer could stream unlimited 1-byte continuations and grow this buffer
180 // without ever tripping the per-frame cap.
181 if (_assembly.size() + payload.size() > ::morph::wire::kMaxEnvelopeBytes) {
182 resetAssembly();
183 throw std::runtime_error("WsFrameReader: reassembled message exceeds kMaxEnvelopeBytes");
184 }
185 _assembly += payload;
186 }
187
188 std::optional<RawFrame> tryExtractRawFrame() {
189 if (_buf.size() < 2) {
190 return std::nullopt;
191 }
192 auto const byte0 = static_cast<std::uint8_t>(_buf[0]);
193 auto const byte1 = static_cast<std::uint8_t>(_buf[1]);
194 bool const fin = (byte0 & 0x80u) != 0;
195 auto const opcode = static_cast<WsOpcode>(static_cast<std::uint8_t>(byte0 & 0x0Fu));
196 bool const masked = (byte1 & 0x80u) != 0;
197 std::uint64_t payloadLen = byte1 & 0x7Fu;
198 std::size_t headerLen = 2;
199 if (payloadLen == 126) {
200 if (_buf.size() < 4) {
201 return std::nullopt;
202 }
203 payloadLen = (static_cast<std::uint64_t>(static_cast<std::uint8_t>(_buf[2])) << 8) |
204 static_cast<std::uint64_t>(static_cast<std::uint8_t>(_buf[3]));
205 headerLen = 4;
206 } else if (payloadLen == 127) {
207 if (_buf.size() < 10) {
208 return std::nullopt;
209 }
210 payloadLen = 0;
211 for (std::size_t i = 0; i < 8; ++i) {
212 payloadLen = (payloadLen << 8) | static_cast<std::uint8_t>(_buf[2 + i]);
213 }
214 headerLen = 10;
215 }
216 if (payloadLen > ::morph::wire::kMaxEnvelopeBytes) {
217 throw std::runtime_error("WsFrameReader: frame payload exceeds kMaxEnvelopeBytes");
218 }
219 std::size_t const maskLen = masked ? 4 : 0;
220 std::size_t const totalLen = headerLen + maskLen + static_cast<std::size_t>(payloadLen);
221 if (_buf.size() < totalLen) {
222 return std::nullopt;
223 }
224 std::string payload = _buf.substr(headerLen + maskLen, static_cast<std::size_t>(payloadLen));
225 if (masked) {
226 std::array<std::uint8_t, 4> key{};
227 for (std::size_t i = 0; i < 4; ++i) {
228 key[i] = static_cast<std::uint8_t>(_buf[headerLen + i]);
229 }
230 for (std::size_t i = 0; i < payload.size(); ++i) {
231 payload[i] = static_cast<char>(static_cast<std::uint8_t>(payload[i]) ^ key[i % key.size()]);
232 }
233 }
234 _buf.erase(0, totalLen);
235 return RawFrame{.fin = fin, .opcode = opcode, .payload = std::move(payload)};
236 }
237
238 std::string _buf;
239 // Reassembly state for a fragmented data message: `_assembly` accumulates
240 // the payload and `_assemblyOpcode` remembers the opcode of the first
241 // frame, since continuations carry opcode 0.
242 bool _assembling{false};
243 WsOpcode _assemblyOpcode{WsOpcode::kText};
244 std::string _assembly;
245};
