21/// @brief Opaque identifier for a model instance inside a backend.
22///
23/// The value 0 is reserved and means "not bound". All non-zero values are
24/// assigned by the backend and are stable for the lifetime of the model.
25struct ModelId {
26 /// @brief Raw numeric id. Zero means unbound.
27 uint64_t v{0};
28
29 /// @brief Three-way comparison — enables `==`, `!=`, `<`, `<=`, `>`, `>=`.
30 auto operator<=>(const ModelId&) const = default;
31};
32
33/// @brief Hash functor so `ModelId` can be used as an `unordered_map` key.
34struct ModelIdHash {
35 /// @brief Returns the hash of @p mid.
36 std::size_t operator()(ModelId mid) const noexcept { return std::hash<uint64_t>{}(mid.v); }
37};

39/// @brief Per-key serialising executor built on top of an arbitrary `IExecutor`.
40///
41/// Tasks posted with the same `ModelId` key are always executed in FIFO order
42/// with no overlap, even when the underlying executor is a thread pool. Tasks
43/// with different keys may run concurrently.
44///
45/// This removes the need for per-model mutexes: the model's `execute()` method
46/// is always called from exactly one task at a time.
47// NOLINTNEXTLINE(cppcoreguidelines-special-member-functions)
48class StrandExecutor {
49public:
50 /// @brief Constructs the strand executor wrapping @p base.
51 /// @param base Underlying executor that actually runs the tasks. Borrowed,
52 /// not owned: it must outlive this `StrandExecutor` *and* keep
53 /// running tasks until the destructor's wait has completed —
54 /// destroying it first deadlocks (see
55 /// `docs/spec/concurrency_and_lifetimes.md`, "Destruction
56 /// ordering").
57 explicit StrandExecutor(IExecutor& base MORPH_LIFETIMEBOUND) : _base{&base} {}
58
59 /// @brief Blocks until all in-flight tasks have completed, then destroys the executor.
60 ///
61 /// Waits for all in-flight lambdas to complete before destroying the map.
62 /// Without this, a pool thread running scheduleNext can access _strands
63 /// after it has been destroyed (TSan: data race on destructor vs erase).
64 ~StrandExecutor() {
65 std::unique_lock lock{_mapMtx};
66 _cv.wait(lock, [this] { return _inFlight == 0; });
67 }
68
69 /// @brief Posts @p task to the strand associated with @p key.
70 ///
71 /// The task is guaranteed to run after all previously posted tasks for the
72 /// same key have completed. Tasks for different keys may interleave freely.
73 /// Thread-safe.
74 /// @param key Model identifier that selects the strand.
75 /// @param task Callable to execute.
76 void post(ModelId key, std::function<void()> task) {
77 std::shared_ptr<Strand> strand;
78 bool schedule = false;
79 {
80 // Hold _mapMtx across the whole slot-lookup + push + re-arm
81 // decision, and take strand->mtx *while still holding _mapMtx*. The
82 // drain-and-erase step in scheduleNext makes its "keep-running vs.
83 // erase" decision under the same {_mapMtx, strand->mtx} pair, so the
84 // two are mutually exclusive.
85 //
86 // Doing the lookup and the re-arm under separate locks once opened a
87 // window: a post() could capture a strand, release _mapMtx, then
88 // re-arm it under strand->mtx *after* a concurrent drain had already
89 // erased it from the map — orphaning a live strand and letting a
90 // later post(key) create a second strand for the same key that ran
91 // concurrently. Because the map lookup and the re-arm now share
92 // _mapMtx with the erase, the strand we push into is always the map's
93 // current entry for this key: a strand that becomes `running` here is
94 // guaranteed to still be the map entry, and it cannot be erased
95 // out from under us (the erase needs _mapMtx too, and never fires
96 // while pending is non-empty).
97 //
98 // Lock order is _mapMtx → strand->mtx, matching scheduleNext —
99 // which acquires the two sequentially rather than with one
100 // scoped_lock over both, precisely to keep this order (see its own
101 // comment). A freshly created strand's mtx
102 // is uncontended; an existing strand's mtx can only be held
103 // elsewhere under the same _mapMtx-first order, so no deadlock.
104 std::scoped_lock const mapLock{_mapMtx};
105 auto slotIter = _strands.find(key);
106 if (slotIter == _strands.end()) {
107 slotIter = installStrand(key);
108 }
109 strand = slotIter->second;
110 std::scoped_lock const strandLock{strand->mtx};
111 strand->pending.push(std::move(task));
112 if (!strand->running) {
113 strand->running = true;
114 schedule = true;
115 // Account for the strand lambda we are about to dispatch *while
116 // still holding _mapMtx*, before releasing it. If we deferred the
117 // ++_inFlight into scheduleNext (which re-takes _mapMtx), a window
118 // opened between releasing _mapMtx here and re-acquiring it there
119 // in which ~StrandExecutor could observe _inFlight == 0, destroy
120 // _strands, and let scheduleNext touch freed state. Incrementing
121 // under this lock makes "decided to schedule" and "counted as
122 // in-flight" atomic, so the destructor never sees a zero it should
123 // not.
124 ++_inFlight;
125 }
126 }
127 if (schedule) {
128 scheduleNext(std::move(strand), key);
129 }
130 }
131
132private:
133 /// @brief FIFO of tasks queued on one strand, with the head task held inline.
134 ///
135 /// Behaviourally a `std::queue<std::function<void()>>` restricted to the
136 /// three operations the strand uses, and used under exactly the same
137 /// discipline: every call happens with the owning `Strand::mtx` held, so
138 /// this type does no locking of its own.
139 ///
140 /// It exists because of what the *container* cost, not what the strand
141 /// did with it. The drain-and-erase step in `scheduleNext` destroys the
142 /// whole `Strand` as soon as the queue empties, so a workload that
143 /// dispatches one action at a time against a model builds a fresh queue on
144 /// every call and puts exactly one task in it. libstdc++'s `std::deque`
145 /// allocates its node map *and* a first 512-byte buffer in its default
146 /// constructor, which comes to 576 bytes of the 760 a deque-only strand
147 /// costs per local dispatch. Holding the head task in the strand makes
148 /// that case allocation-free; the overflow deque is constructed only when
149 /// a second task is genuinely queued behind a running one, after which the
150 /// cost is the deque's as before.
151 ///
152 /// This changes no lifetime or locking rule: the erase still happens when
153 /// `empty()` becomes true, still under the `{_mapMtx, strand->mtx}` pair.
154 class PendingQueue {
155 public:
156 /// @brief Reports whether the queue holds no task.
157 /// @return `true` when nothing is queued.
158 [[nodiscard]] bool empty() const noexcept { return !_hasHead; }
159
160 /// @brief Appends @p task to the back of the queue.
161 /// @param task Callable to queue. An *empty* `std::function` is queued
162 /// and later dispatched like any other: occupancy is
163 /// tracked by a separate flag rather than by testing the
164 /// callable, so this type never silently drops one.
165 void push(std::function<void()>&& task) {
166 if (!_hasHead) {
167 _head = std::move(task);
168 _hasHead = true;
169 return;
170 }
171 if (!_overflow) {
172 _overflow = std::make_unique<std::deque<std::function<void()>>>();
173 }
174 _overflow->push_back(std::move(task));
175 }
176
177 /// @brief Removes the task at the front of the queue and returns it.
178 /// @return The front task.
179 /// @pre `!empty()`.
180 std::function<void()> pop() {
181 std::function<void()> task = std::move(_head);
182 if (_overflow && !_overflow->empty()) {
183 _head = std::move(_overflow->front());
184 _overflow->pop_front();
185 } else {
186 // A moved-from std::function is valid but unspecified; clear it
187 // explicitly so the slot holds no captured state while idle.
188 _head = nullptr;
189 _hasHead = false;
190 }
191 return task;
192 }
193
194 private:
195 std::function<void()> _head;
196 std::unique_ptr<std::deque<std::function<void()>>> _overflow;
197 bool _hasHead = false;
198 };

207 /// @brief The `ModelId` → strand map. Named so the recycled node type can be.
208 using StrandMap = std::unordered_map<ModelId, std::shared_ptr<Strand>, ModelIdHash>;
209
210 /// @brief Returns an iterator to the strand for @p key, creating the entry.
211 ///
212 /// **Precondition:** the caller holds `_mapMtx` and has already
213 /// established that `key` has no entry.
214 ///
215 /// This is a pure allocation optimisation and changes no lifetime or
216 /// locking rule. The drain step in `scheduleNext` removes the whole map
217 /// entry as soon as the queue empties, so a workload that dispatches one
218 /// action at a time against a model would otherwise pay for a fresh map
219 /// node *and* a fresh `make_shared<Strand>` on every call — 2 allocations
220 /// and 152 bytes on top of the queue's own share.
221 /// Rather than keep the slot alive across the drain (which would need a
222 /// deregistration hook and would trade this churn for a per-model entry
223 /// nothing reclaims), the drain `extract`s the node instead of erasing it
224 /// and parks it in `_spare`, and this re-keys and re-inserts that one
225 /// node. The entry still leaves the map at the same point under the same
226 /// locks, so the map is bounded exactly as before; `_spare` holds at most
227 /// one node and is freed with the executor.
228 ///
229 /// Reusing the parked node's `Strand` as well is guarded by sole
230 /// ownership. `use_count() == 1` means the recycled node holds the only
231 /// reference, so nothing else can reach the object and reusing it is
232 /// indistinguishable from constructing a new one. That is the whole
233 /// argument, and it is deliberately not "the previous owner makes no
234 /// further access": a strand lambda that is still finishing does hold a
235 /// reference, and when it does, this constructs a fresh `Strand` exactly
236 /// as before and recycles only the node.
237 ///
238 /// The parked strand needs no reset. It is only ever extracted from a
239 /// strand observed `!running` with an empty `pending` under
240 /// `{_mapMtx, strand->mtx}`, which is the state a fresh one is in. A
241 /// runtime re-check of that here would be an arm nothing can take, so it
242 /// is written down rather than branched on.
243 /// @param key Model identifier to install a strand for.
244 /// @return Iterator to the entry for @p key.
245 StrandMap::iterator installStrand(ModelId key) {
246 if (!_spare) {
247 auto const iter = _strands.emplace(key, std::make_shared<Strand>()).first;
248 iter->second->base = _base;
249 return iter;
250 }
251 _spare.key() = key;
252 auto& reused = _spare.mapped();
253 if (reused.use_count() != 1) {
254 reused = std::make_shared<Strand>();
255 }
256 reused->base = _base;
257 // `insert` consumes the node. Were the precondition ever violated it
258 // would instead hand the node back inside the returned object, which
259 // frees it — the same fate the old `erase` gave it — and `position`
260 // would name the existing entry, so the caller is right either way
261 // and there is nothing to branch on.
262 return _strands.insert(std::move(_spare)).position;
263 }

265 /// @brief Dispatches one strand lambda onto the base executor.
266 ///
267 /// **Precondition:** the caller must have already incremented `_inFlight`
268 /// (under `_mapMtx`) to account for this dispatch. `post()` does so in the
269 /// same critical section that decides to schedule, and the re-entrant call
270 /// below does so under the `_mapMtx` it already holds. Keeping the increment
271 /// with the *decision* (rather than here) closes the window where
272 /// `~StrandExecutor` could observe `_inFlight == 0` between the decision and
273 /// this dispatch and destroy `_strands` out from under us.
274 void scheduleNext(std::shared_ptr<Strand> strand, ModelId key) {
275 // Read `base` out before the capture list moves `strand` into the
276 // lambda: `strand->base` and the lambda's construction are
277 // unsequenced within one call expression, so reading through the
278 // moved-from pointer would be a real hazard rather than a stylistic
279 // one. The lambda's capture is non-const (hence `mutable` and the
280 // by-value parameter) so the drain below can release it early.
281 IExecutor* const base = strand->base;
282 base->post([this, strand = std::move(strand), key]() mutable {
283 std::function<void()> task;
284 {
285 std::scoped_lock const lock{strand->mtx};
286 task = strand->pending.pop();
287 }
288 try {
289 task();
290 } catch (const std::exception& exc) {
291 // The strand is where Model::execute() actually runs; a throw
292 // here must not stall the strand or vanish — log and continue so
293 // the next queued task for this model still runs.
294 ::morph::log::logError("[strand] task threw: " + std::string{exc.what()});
295 } catch (...) {
296 ::morph::log::logError("[strand] task threw unknown exception");
297 }
298 // Decide "keep running vs. drain-and-erase" atomically across the
299 // map slot and the strand's pending queue. Doing it in two steps
300 // (flip running under strand->mtx, then erase under _mapMtx) opened
301 // a window where another post() could re-arm this strand after we
302 // unlocked strand->mtx but before we erased the map entry. The
303 // subsequent erase then orphaned a live strand: a later post(key)
304 // would create a *new* strand for the same key, and the two
305 // strands could run model tasks concurrently → data race.
306 bool more = false;
307 {
308 // Same lock order as post(): _mapMtx first, then strand->mtx.
309 // Acquiring them sequentially (rather than via a single
310 // scoped_lock over both, whose std::lock back-off can grab them
311 // in address order) keeps a single, consistent ordering across
312 // every site that holds both, so there is no lock-ordering
313 // deadlock. This is the point where "drain-and-erase" is decided
314 // atomically w.r.t. a concurrent post(): a post() re-arming this
315 // strand and this block erasing it cannot interleave, because
316 // both hold _mapMtx across the whole decision.
317 std::scoped_lock const mapLock{_mapMtx};
318 std::scoped_lock const strandLock{strand->mtx};
319 more = !strand->pending.empty();
320 if (!more) {
321 strand->running = false;
322 auto iter = _strands.find(key);
323 if (iter != _strands.end() && iter->second == strand) {
324 // `extract`, not `erase`: same removal, same moment,
325 // same locks — the entry leaves the map here exactly as
326 // before, and every reason the erase had to happen
327 // under {_mapMtx, strand->mtx} still applies unchanged.
328 // The difference is only that the detached node's
329 // memory is parked for `installStrand` to re-key
330 // instead of being returned to the allocator. Any node
331 // already parked is freed by this assignment, so at
332 // most one is ever held. Freeing it runs no user code
333 // under these locks: a parked strand was parked
334 // because its pending queue was empty, so there is no
335 // captured task left to destroy.
336 _spare = _strands.extract(iter);
337 }
338 } else {
339 // Account for the re-armed dispatch *before* releasing
340 // _mapMtx (scheduleNext's precondition). Because this run's
341 // own decrement below has not happened yet, _inFlight is
342 // briefly 2 here and never dips to 0 across the handoff, so
343 // ~StrandExecutor cannot slip in and destroy _strands between
344 // the two runs.
345 ++_inFlight;
346 }
347 }
348 if (more) {
349 scheduleNext(strand, key);
350 } else {
351 // Drop this run's co-ownership here rather than leaving it to
352 // the lambda's destruction a few lines below. Nothing after
353 // this point touches the strand, and releasing it early is
354 // what lets `installStrand` see `use_count() == 1` on the
355 // node just parked in `_spare`: the next post() for this key
356 // is typically already blocked on _mapMtx when the block above
357 // releases it, so a reference held until the lambda dies would
358 // usually still be there when that post looks. This only
359 // affects *whether the object is recycled*, never whether the
360 // recycling is safe — a post that looks too early simply sees
361 // two owners and constructs a fresh Strand.
362 //
363 // Safe to be the last owner here: no lock is held (the block
364 // above released both), so this never destroys a mutex it is
365 // standing on. If the extract above did run, `_spare` owns the
366 // strand and this merely decrements.
367 strand.reset();
368 }
369 // Decrement after all map access is done; wake destructor if it is waiting.
370 {
371 std::scoped_lock const lock{_mapMtx};
372 if (--_inFlight == 0) {
373 // Inside the `== 0` branch, so a handoff does not signal at
374 // all: the re-arm above has already incremented for the
375 // next dispatch, so the count does not reach zero until the
376 // strand is quiescent. `~StrandExecutor` is the only waiter
377 // on this variable, so `notify_all` wakes at most one
378 // thread and is equivalent to `notify_one` here -- there is
379 // no herd to wake, and no predicate but `_inFlight == 0`
380 // for a wakeup to land on and be lost.
381 _cv.notify_all();
382 }
383 }
384 });
385 }
386
387 IExecutor* _base;
388 std::mutex _mapMtx;
389 std::condition_variable _cv;
390 int _inFlight{0};
391 StrandMap _strands;
392 /// @brief The one detached map node kept for reuse. Guarded by `_mapMtx`.
393 ///
394 /// Declared after `_strands` so it is destroyed first: the node owns
395 /// storage obtained from the map's allocator, and returning it before the
396 /// container goes away keeps that ordering obvious even though the default
397 /// allocator is stateless.
398 StrandMap::node_type _spare;
399};
