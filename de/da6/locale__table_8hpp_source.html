17/// @brief Where a locale puts a currency symbol relative to the amount.
18enum class SymbolPlacement : std::uint8_t {
19 Before, ///< `€13,196.40`.
20 After, ///< `13.196,40 €`.

23/// @brief The symbol a locale writes for one currency.
24struct CurrencySymbol {

31/// @brief How a locale writes money.
32struct CurrencyFacts {
33 /// @brief Where the symbol stands.
34 SymbolPlacement placement = SymbolPlacement::After;
35 /// @brief The symbols the locale writes. Empty: every currency is written by its code. Must outlive the facts.

53/// @brief Whether `money` writes a currency by its symbol or by its code.

95/// @brief The facts of a locale: the exact tag, then each shorter one (`de-AT`, then `de`).
96/// @param tag The BCP 47 tag.
97/// @return The row, or null when no prefix of @p tag has one.
98[[nodiscard]] inline LocaleFacts const* findLocaleFacts(std::string_view tag) noexcept {
99 while (!tag.empty()) {
100 for (LocaleFacts const& row : kLocaleTable) {
101 if (row.tag == tag) {
102 return &row;
103 }
104 }
105 std::size_t const cut = tag.rfind('-');
106 if (cut == std::string_view::npos) {
107 break;
108 }
109 tag = tag.substr(0, cut);
110 }
111 return nullptr;
112}

114/// @brief The facts of a locale, `en`'s where it has none.
115/// @param tag The BCP 47 tag.
116/// @return The row `findLocaleFacts` finds, else `en`'s.
117[[nodiscard]] inline LocaleFacts const& localeFacts(std::string_view tag) noexcept {
118 LocaleFacts const* const found = findLocaleFacts(tag);
119 return found != nullptr ? *found : kLocaleTable.front();
120}

122/// @brief Writes an amount with its currency.
123/// @param number The amount as `formatCanonicalNumber` wrote it, sign included.
124/// @param currency The ISO 4217 code.
125/// @param facts The locale's currency facts.
126/// @param numeric The locale's separators and sign; its sign is the one @p number starts with.
127/// @param form The symbol or the code.
128/// @return `number code` for the code form, with no symbol facts or for a currency @p facts lists no symbol for;
129/// otherwise the symbol placed by `facts.placement`, the sign first, a no-break space between a symbol that
130/// ends in an ASCII letter and the digits (and always between the digits and a trailing symbol).
131[[nodiscard]] inline std::string placeCurrency(std::string_view number, std::string_view currency,
132 CurrencyFacts const& facts, render::NumericLocale const& numeric,
133 MoneyForm form) {
134 auto const listed = [&]() -> CurrencySymbol const* {
135 for (CurrencySymbol const& entry : facts.symbols) {
136 if (entry.code == currency) {
137 return &entry;
138 }
139 }
140 return nullptr;
141 }();
142 if (form == MoneyForm::Code || listed == nullptr) {
143 return std::string{number} + " " + std::string{currency};
144 }
145 std::string_view const sign = numeric.negativeSign.empty() ? std::string_view{"-"} : numeric.negativeSign;
146 std::string_view digits = number;
147 std::string_view shownSign;
148 if (digits.starts_with(sign)) {
149 shownSign = sign;
150 digits.remove_prefix(sign.size());
151 }
152 std::string out{shownSign};
153 if (facts.placement == SymbolPlacement::Before) {
154 out += listed->symbol;
155 char const last = listed->symbol.empty() ? ' ' : listed->symbol.back();
156 if ((last >= 'A' && last <= 'Z') || (last >= 'a' && last <= 'z')) {
157 out += " ";
158 }
159 out += digits;
160 } else {
161 out += digits;
162 out += " ";
163 out += listed->symbol;
164 }
165 return out;
166}
