5/// @file rational.hpp
6/// @brief Exact rational arithmetic for morph model/action values.
7///
8/// `morph::math::Rational` is a small, value-semantic, trivially-copyable
9/// struct representing the rational number `numerator/denominator` with
10/// `std::int64_t` components. It carries a runtime decimal-precision tag,
11/// `decimalPlaces`, as a strong type (`DecimalPlaces`). Arithmetic is
12/// *exact* — sums, differences, products, and quotients are reduced to
13/// canonical form with no floating-point rounding error. The precision tag
14/// affects only decimal scaling (`Rational::fromFloat`) and rounding
15/// (`Rational::toDouble`, formatting); it never changes a stored value.
16///
17/// Adapted from LASTRADA `JPMath/Rational.hpp`, with the `boxed` strong-type
18/// dependency replaced by a self-contained `DecimalPlaces` and a Glaze wire
19/// codec added so the type round-trips through the morph JSON wire with its
20/// invariants restored on read.
21///
22/// The precision is supplied at construction. There is intentionally no
23/// default — every call site states the precision it intends, e.g.
24/// `Rational{1, 3, DecimalPlaces{9}}`. Precision is capped at
25/// `kMaxDecimalPlaces` (18, the largest `k` for which `10^k` fits in
26/// `int64_t`); out-of-range values assert in debug and clamp into
27/// `[1, kMaxDecimalPlaces]` in release.
28///
29/// @par Cross-precision arithmetic
30/// Binary arithmetic propagates `std::max` of the two operands'
31/// `decimalPlaces` — the wider precision wins. Comparison (`<=>`, `==`) is
32/// purely value-based on the canonical `(numerator, denominator)` pair and
33/// ignores `decimalPlaces` entirely.
34///
35/// @par Invariants
36/// Every public operation that produces a `Rational` restores:
37/// - `denominator > 0` (strictly positive — never zero, never negative)
38/// - `gcd(|numerator|, denominator) == 1`
39/// - canonical zero is `0/1`
40/// - `1 <= decimalPlaces.value <= kMaxDecimalPlaces`
41///
42/// All sign lives in the numerator.
43///
44/// @par Error handling
45/// The struct never throws. Operations that may fail (zero divisor,
46/// non-finite floating-point input, overflow during decimal scaling) return
47/// `std::expected<Rational, RationalError>`. `operator/` and `reciprocal`
48/// return `expected`; `operator+`, `operator-`, `operator*` on plain
49/// `Rational` pairs cannot fail and return a bare `Rational`.
50///
51/// @par Mixed-type expressions
52/// Whenever an arithmetic expression contains an
53/// `std::expected<Rational, RationalError>` sub-expression or a
54/// floating-point operand, the whole expression evaluates to
55/// `std::expected<Rational, RationalError>`. The float operand is lifted via
56/// `Rational::fromFloat` — its precision is taken from the Rational
57/// operand's `getDecimalPlaces()`. Errors short-circuit left to right.
58///
59/// @code{.cpp}
60/// using morph::math::DecimalPlaces;
61/// using morph::math::Rational;
62/// auto const a = Rational{7, 2, DecimalPlaces{9}};
63/// auto const b = Rational{2, DecimalPlaces{9}};
64/// auto const c = Rational{1, 2, DecimalPlaces{9}};
65/// auto result = a / b + c * 3.5;
66/// // decltype(result) == std::expected<Rational, RationalError>
67/// @endcode
68///
69/// @par Formatting
70/// `std::format` support is split by the supplied spec:
71/// - empty spec `"{}"` -> exact rational form (`"n/d"`, or `"n"` when integer)
72/// - non-empty spec `"{:.3f}"` -> delegated to `std::formatter<double>` on `toDouble()`
73///
74/// @par Wire format
75/// Over the morph JSON wire a `Rational` travels as the object
76/// `{"num":617,"den":50,"dp":2}`. Reading goes through the canonicalising
77/// constructor, so a non-canonical payload (`1234/100`) or a hostile one
78/// (`den == 0`, out-of-range `dp`) always lands as a valid, reduced value.
79///
80/// @note Negating a `Rational` built from `INT64_MIN` overflows; avoid that
81/// extreme value in code (the wire codec clamps it away for untrusted
82/// input).
83/// @note Comparison is exact over the full int64 range (128-bit cross
84/// products). Arithmetic, however, has the usual fixed-width envelope:
85/// `+`, `-`, `*` overflow int64 when reduced cross terms exceed ~2^63
86/// (e.g. sums over large coprime denominators). Keep operands within
87/// the decimal-scaled ranges the precision tags imply.

105/// @brief Strong type for a decimal-precision count.
106///
107/// Prevents the precision from being confused with a numerator or a
108/// denominator at a call site: `Rational{1, 3, DecimalPlaces{9}}`.
109struct DecimalPlaces {
110 /// @brief Raw digit count. Rational's invariant keeps it in [1, kMaxDecimalPlaces].
111 std::uint32_t value{};
112
113 constexpr DecimalPlaces() noexcept = default;
114
115 /// @brief Explicit so a bare integer never silently becomes a precision.
116 /// @param raw The digit count to box.
117 constexpr explicit DecimalPlaces(std::uint32_t raw) noexcept : value{raw} {}
118
119 /// @brief Value ordering/equality on the raw digit count.
120 /// @param other Precision to compare against.
121 /// @return The ordering of the two raw counts.
122 [[nodiscard]] constexpr auto operator<=>(const DecimalPlaces& other) const noexcept = default;
123};

126///

133
134 /// @brief Explicit so the type must be spelled out.
135 /// @param raw The integer to box as a numerator.
136 constexpr explicit Numerator(std::int64_t raw) noexcept : value{raw} {}
137};

139/// @brief Strong type for a Rational denominator.
140///
141/// Prevents numerator/denominator argument swapping at construction sites.
142struct Denominator {
143 /// @brief Raw signed integer value; must never be zero after canonicalisation.
144 std::int64_t value{};
145
146 constexpr Denominator() noexcept = default;
147
148 /// @brief Explicit so the type must be spelled out.
149 /// @param raw The integer to box as a denominator.
150 constexpr explicit Denominator(std::int64_t raw) noexcept : value{raw} {}
151};

153/// @brief Largest decimal precision the type supports: `10^18` is the
154/// greatest power of ten that fits in `std::int64_t`.
155inline constexpr std::uint32_t kMaxDecimalPlaces = 18;
156
157/// @brief Error states reachable through `Rational` operations.
158enum class RationalError : std::uint8_t {
159 DivisionByZero, ///< Divisor numerator is zero (operator/, reciprocal, From).
160 NotFinite, ///< Floating-point input was NaN or +/-Inf (fromFloat only).
161 Overflow, ///< Scaled magnitude exceeds int64_t range (fromFloat only).
162};

166/// @brief Clamps a raw precision into `[1, kMaxDecimalPlaces]` silently — for
167/// untrusted wire input.
168[[nodiscard]] constexpr std::uint32_t clampWireDecimalPlaces(std::uint32_t rawDecimalPlaces) noexcept {
169 if (rawDecimalPlaces < 1) {
170 return 1;
171 }
172 if (rawDecimalPlaces > kMaxDecimalPlaces) {
173 return kMaxDecimalPlaces;
174 }
175 return rawDecimalPlaces;
176}
177
178/// @brief Clamps like `clampWireDecimalPlaces` but asserts in debug — for
179/// call sites that state a precision in code, where out-of-range is a bug.
180[[nodiscard]] constexpr std::uint32_t clampDecimalPlaces(std::uint32_t rawDecimalPlaces) noexcept {
181 assert(rawDecimalPlaces >= 1 && rawDecimalPlaces <= kMaxDecimalPlaces);
182 return clampWireDecimalPlaces(rawDecimalPlaces);
183}

185/// @brief A 64x64 -> 128-bit unsigned product, comparable high-word-first.
186struct U128 {
187 std::uint64_t hi{};
188 std::uint64_t lo{};
189 [[nodiscard]] constexpr auto operator<=>(const U128&) const noexcept = default;
190};
191
192/// @brief Multiplies two u64 exactly into 128 bits (for exact fraction comparison).
193[[nodiscard]] constexpr U128 mulU64(std::uint64_t lhs, std::uint64_t rhs) noexcept {
194#ifdef __SIZEOF_INT128__
195 auto const product = static_cast<unsigned __int128>(lhs) * rhs;
196 return U128{.hi = static_cast<std::uint64_t>(product >> 64), .lo = static_cast<std::uint64_t>(product)};
197#else
198 // Portable 32-bit limb multiplication (MSVC has no __int128).
199 constexpr std::uint64_t mask = 0xffffffffULL;
200 auto const lhsLow = lhs & mask;
201 auto const lhsHigh = lhs >> 32;
202 auto const rhsLow = rhs & mask;
203 auto const rhsHigh = rhs >> 32;
204 auto const lowLow = lhsLow * rhsLow;
205 auto const lowHigh = lhsLow * rhsHigh;
206 auto const highLow = lhsHigh * rhsLow;
207 auto const highHigh = lhsHigh * rhsHigh;
208 auto const mid = (lowLow >> 32) + (lowHigh & mask) + (highLow & mask);
209 return U128{.hi = highHigh + (lowHigh >> 32) + (highLow >> 32) + (mid >> 32),
210 .lo = (mid << 32) | (lowLow & mask)};
211#endif
212}

214/// @brief Magnitude of a signed 64-bit value as u64; well-defined for INT64_MIN.
215[[nodiscard]] constexpr std::uint64_t absU64(std::int64_t value) noexcept {
216 return value < 0 ? 0ULL - static_cast<std::uint64_t>(value) : static_cast<std::uint64_t>(value);
217}

221/// @brief Exact rational number with a runtime decimal-precision tag.
222struct Rational {
223 /// @brief Signed numerator. Carries the sign of the rational value.
224 std::int64_t numerator{0};
225
226 /// @brief Strictly positive denominator. Never zero, never negative.
227 std::int64_t denominator{1};
228
229 /// @brief Decimal-precision tag. Invariant: 1 <= value <= kMaxDecimalPlaces.
230 DecimalPlaces decimalPlaces{1};
231
232 /// @brief Default-constructs the canonical zero (0/1) at precision 1.
233 constexpr Rational() noexcept = default;
234
235 /// @brief Constructs from a whole integer at the given precision.
236 /// @param whole The integer value; stored as `whole/1`.
237 /// @param wantedPrecision Decimal precision; clamped to [1, kMaxDecimalPlaces].
238 constexpr Rational(std::int64_t whole, DecimalPlaces wantedPrecision) noexcept
239 : numerator{whole}, decimalPlaces{detail::clampDecimalPlaces(wantedPrecision.value)} {}
240
241 /// @brief Constructs from explicit numerator/denominator, then canonicalises.
242 ///
243 /// A negative @p wantedDenominator flips the sign of @p wantedNumerator,
244 /// the pair is reduced by `gcd`, and a denominator of `0` is clamped to
245 /// `1`. Use `from` for explicit zero-denominator detection.
246 ///
247 /// @param wantedNumerator Signed numerator.
248 /// @param wantedDenominator Denominator. May be negative or zero on input.
249 /// @param wantedPrecision Decimal precision; clamped to [1, kMaxDecimalPlaces].
250 constexpr Rational(Numerator wantedNumerator, Denominator wantedDenominator,
251 DecimalPlaces wantedPrecision) noexcept
252 : numerator{wantedNumerator.value},
253 denominator{wantedDenominator.value},
254 decimalPlaces{detail::clampDecimalPlaces(wantedPrecision.value)} {
255 canonicalise();
256 }
257
258 /// @brief Validating factory.
259 /// @param wantedNumerator Signed numerator.
260 /// @param wantedDenominator Denominator; `0` is rejected instead of clamped.
261 /// @param wantedPrecision Decimal precision; clamped to [1, kMaxDecimalPlaces].
262 /// @return The canonical rational, or `DivisionByZero` if @p wantedDenominator is 0.
263 [[nodiscard]] static constexpr std::expected<Rational, RationalError> from(
264 Numerator wantedNumerator, Denominator wantedDenominator, DecimalPlaces wantedPrecision) noexcept {
265 if (wantedDenominator.value == 0) {
266 return std::unexpected(RationalError::DivisionByZero);
267 }
268 return Rational{wantedNumerator, wantedDenominator, wantedPrecision};
269 }
270
271 /// @brief Converts a `double` to a Rational scaled to @p wantedPrecision.
272 /// @param value Source value.
273 /// @param wantedPrecision Decimal precision to scale to.
274 /// @return A canonical Rational, or an error (NotFinite / Overflow).
275 /// @note `noexcept` but not `constexpr` — uses `std::llround` / `std::isfinite`.
276 [[nodiscard]] static std::expected<Rational, RationalError> fromFloat(double value,
277 DecimalPlaces wantedPrecision) noexcept;
278
279 /// @brief Converts a `float` to a Rational scaled to @p wantedPrecision.
280 /// @param value Source value.
281 /// @param wantedPrecision Decimal precision to scale to.
282 /// @return A canonical Rational, or an error (NotFinite / Overflow).
283 [[nodiscard]] static std::expected<Rational, RationalError> fromFloat(float value,
284 DecimalPlaces wantedPrecision) noexcept;
285
286 /// @brief Converts a `long double` to a Rational scaled to @p wantedPrecision.
287 /// @param value Source value.
288 /// @param wantedPrecision Decimal precision to scale to.
289 /// @return A canonical Rational, or an error (NotFinite / Overflow).
290 [[nodiscard]] static std::expected<Rational, RationalError> fromFloat(long double value,
291 DecimalPlaces wantedPrecision) noexcept;
292
293 /// @brief Canonical zero (`0/1`) at the given precision.
294 /// @param wantedPrecision Decimal precision of the returned value.
295 /// @return `0/1` tagged with @p wantedPrecision.
296 [[nodiscard]] static constexpr Rational zero(DecimalPlaces wantedPrecision) noexcept {
297 return Rational{Numerator{0}, Denominator{1}, wantedPrecision};
298 }
299
300 /// @brief Canonical one (`1/1`) at the given precision.
301 /// @param wantedPrecision Decimal precision of the returned value.
302 /// @return `1/1` tagged with @p wantedPrecision.
303 [[nodiscard]] static constexpr Rational one(DecimalPlaces wantedPrecision) noexcept {
304 return Rational{Numerator{1}, Denominator{1}, wantedPrecision};
305 }
306
307 /// @brief The value's current decimal precision.
308 /// @return The `decimalPlaces` tag.
309 [[nodiscard]] constexpr DecimalPlaces getDecimalPlaces() const noexcept { return decimalPlaces; }
310
311 /// @brief Whether the value equals `0/1`.
312 /// @return `true` if the numerator is zero.
313 [[nodiscard]] constexpr bool isZero() const noexcept { return numerator == 0; }
314
315 /// @brief Whether the value is an integer.
316 /// @return `true` if the denominator is 1.
317 [[nodiscard]] constexpr bool isInteger() const noexcept { return denominator == 1; }
318
319 /// @brief Whether the value is strictly less than zero.
320 /// @return `true` if the numerator is negative.
321 [[nodiscard]] constexpr bool isNegative() const noexcept { return numerator < 0; }
322
323 /// @brief Converts to `double`, rounded to this value's `decimalPlaces`.
324 /// @return The rounded floating-point reading.
325 [[nodiscard]] double toDouble() const noexcept { return toDouble(decimalPlaces.value); }
326
327 /// @brief Converts to `double`, rounded to @p requestedDecimalPlaces.
328 /// @param requestedDecimalPlaces Number of decimal digits to keep. Values
329 /// `> 18` fall back to unrounded conversion.
330 /// @return The rounded floating-point reading.
331 [[nodiscard]] double toDouble(std::uint32_t requestedDecimalPlaces) const noexcept;
332
333 /// @brief Negates. @note Negating a Rational built from `INT64_MIN` overflows.
334 /// @return The value with the numerator's sign flipped.
335 [[nodiscard]] constexpr Rational operator-() const noexcept {
336 return Rational{Numerator{-numerator}, Denominator{denominator}, decimalPlaces};
337 }
338
339 /// @brief Multiplicative inverse.
340 /// @return `denominator/numerator`, or `unexpected(DivisionByZero)` if zero.
341 [[nodiscard]] constexpr std::expected<Rational, RationalError> reciprocal() const noexcept {
342 if (numerator == 0) {
343 return std::unexpected(RationalError::DivisionByZero);
344 }
345 if (numerator < 0) {
346 return Rational{Numerator{-denominator}, Denominator{-numerator}, decimalPlaces};
347 }
348 return Rational{Numerator{denominator}, Denominator{numerator}, decimalPlaces};
349 }
350
351 /// @brief Three-way comparison. Value-only: ignores `decimalPlaces`.
352 ///
353 /// Exact for the full int64 range: cross-products are computed in 128
354 /// bits, so the ordering is always consistent with `operator==`.
355 /// @param other Value to compare against.
356 /// @return The ordering of the two exact values.
357 [[nodiscard]] constexpr std::strong_ordering operator<=>(const Rational& other) const noexcept {
358 auto const leftSign = static_cast<int>(numerator > 0) - static_cast<int>(numerator < 0);
359 auto const rightSign = static_cast<int>(other.numerator > 0) - static_cast<int>(other.numerator < 0);
360 if (leftSign != rightSign) {
361 return leftSign <=> rightSign;
362 }
363 if (leftSign == 0) {
364 return std::strong_ordering::equal;
365 }
366
367 auto const leftProduct = detail::mulU64(detail::absU64(numerator),
368 static_cast<std::uint64_t>(other.denominator));
369 auto const rightProduct = detail::mulU64(detail::absU64(other.numerator),
370 static_cast<std::uint64_t>(denominator));
371 auto const magnitude = leftProduct <=> rightProduct;
372 if (leftSign > 0) {
373 return magnitude;
374 }
375 if (std::is_lt(magnitude)) {
376 return std::strong_ordering::greater;
377 }
378 if (std::is_gt(magnitude)) {
379 return std::strong_ordering::less;
380 }
381 return std::strong_ordering::equal;
382 }
383
384 /// @brief Equality. Value-only: ignores `decimalPlaces`.
385 /// @param other Value to compare against.
386 /// @return `true` when the canonical pairs are identical.
387 [[nodiscard]] constexpr bool operator==(const Rational& other) const noexcept {
388 return numerator == other.numerator && denominator == other.denominator;
389 }
390
391 /// @brief In-place addition. Result precision becomes `max` of the two.
392 /// Uses reduce-before-multiply to extend the safe int64 range (Knuth 4.5.1).
393 /// @param rhs Value to add.
394 /// @return `*this`.
395 constexpr Rational& operator+=(const Rational& rhs) noexcept {
396 auto const denominatorGcd = std::gcd(denominator, rhs.denominator);
397 auto const rightDenominatorScaled = rhs.denominator / denominatorGcd;
398 auto const leftDenominatorScaled = denominator / denominatorGcd;
399 numerator = (numerator * rightDenominatorScaled) + (rhs.numerator * leftDenominatorScaled);
400 denominator = denominator * rightDenominatorScaled;
401 widenPrecisionTo(rhs.decimalPlaces);
402 canonicalise();
403 return *this;
404 }
405
406 /// @brief In-place subtraction. Result precision becomes `max` of the two.
407 /// @param rhs Value to subtract.
408 /// @return `*this`.
409 constexpr Rational& operator-=(const Rational& rhs) noexcept {
410 auto const denominatorGcd = std::gcd(denominator, rhs.denominator);
411 auto const rightDenominatorScaled = rhs.denominator / denominatorGcd;
412 auto const leftDenominatorScaled = denominator / denominatorGcd;
413 numerator = (numerator * rightDenominatorScaled) - (rhs.numerator * leftDenominatorScaled);
414 denominator = denominator * rightDenominatorScaled;
415 widenPrecisionTo(rhs.decimalPlaces);
416 canonicalise();
417 return *this;
418 }
419
420 /// @brief In-place multiplication. Result precision becomes `max` of the two.
421 /// Cross-cancels common factors before multiplying.
422 /// @param rhs Value to multiply by.
423 /// @return `*this`.
424 constexpr Rational& operator*=(const Rational& rhs) noexcept {
425 auto const absoluteLeftNumerator = numerator < 0 ? -numerator : numerator;
426 auto const absoluteRightNumerator = rhs.numerator < 0 ? -rhs.numerator : rhs.numerator;
427 auto const crossDivisorOne = std::gcd(absoluteLeftNumerator, rhs.denominator);
428 auto const crossDivisorTwo = std::gcd(absoluteRightNumerator, denominator);
429 auto const reducedLeftNumerator = numerator / crossDivisorOne;
430 auto const reducedRightNumerator = rhs.numerator / crossDivisorTwo;
431 auto const reducedLeftDenominator = denominator / crossDivisorTwo;
432 auto const reducedRightDenominator = rhs.denominator / crossDivisorOne;
433 numerator = reducedLeftNumerator * reducedRightNumerator;
434 denominator = reducedLeftDenominator * reducedRightDenominator;
435 widenPrecisionTo(rhs.decimalPlaces);
436 canonicalise();
437 return *this;
438 }
439
440 /// @brief Non-throwing division. Result precision becomes `max` of the two.
441 /// @param rhs Divisor.
442 /// @return `*this / rhs`, or `unexpected(DivisionByZero)` if @p rhs is zero.
443 [[nodiscard]] constexpr std::expected<Rational, RationalError> dividedBy(const Rational& rhs) const noexcept {
444 if (rhs.numerator == 0) {
445 return std::unexpected(RationalError::DivisionByZero);
446 }
447 auto leftCopy = *this;
448 auto const reciprocalNumerator = rhs.numerator > 0 ? rhs.denominator : -rhs.denominator;
449 auto const reciprocalDenominator = rhs.numerator > 0 ? rhs.numerator : -rhs.numerator;
450 leftCopy *= Rational{Numerator{reciprocalNumerator}, Denominator{reciprocalDenominator}, rhs.decimalPlaces};
451 return leftCopy;
452 }

454 /// @brief Flat JSON representation used by the Glaze codec below.
455 ///
456 /// Field names are the wire contract: `{"num":..,"den":..,"dp":..}`.
457 struct Wire {
458 std::int64_t num{0}; ///< Signed numerator as sent/received.
459 std::int64_t den{1}; ///< Denominator as sent/received; may be non-canonical.
460 std::uint32_t dp{1}; ///< Decimal-precision tag as sent/received.
461 };
462
463 /// @brief Wire-codec entry (Glaze read side): rebuilds through the
464 /// canonicalising constructor, silently clamping hostile input
465 /// (`den == 0`, out-of-range `dp`, `INT64_MIN` components whose
466 /// negation would overflow) instead of asserting.
467 /// @param wire Raw values decoded from JSON.
468void setWire(Wire wire) noexcept {
469 constexpr auto int64Min = std::numeric_limits<std::int64_t>::min();
470 constexpr auto negatableMin = -std::numeric_limits<std::int64_t>::max();
471 *this = Rational{Numerator{wire.num == int64Min ? negatableMin : wire.num},
472 Denominator{wire.den == int64Min ? negatableMin : wire.den},
473 DecimalPlaces{detail::clampWireDecimalPlaces(wire.dp)}};
474 }
475
476 /// @brief Wire-codec exit (Glaze write side).
477 /// @return The canonical members, ready for JSON encoding.
478 [[nodiscard]] Wire getWire() const noexcept { return Wire{.num = numerator, .den = denominator, .dp = decimalPlaces.value}; }
479
480private:
481 /// @brief Restores the canonical-form invariants in place (denominator > 0,
482 /// gcd reduced, zero denominator clamped to 1). Leaves `decimalPlaces`
483 /// untouched (it is not a value property).
484 constexpr void canonicalise() noexcept {
485 if (denominator == 0) {
486 denominator = 1;
487 return;
488 }
489 if (denominator < 0) {
490 numerator = -numerator;
491 denominator = -denominator;
492 }
493 auto const absoluteNumerator = numerator < 0 ? -numerator : numerator;
494 auto const divisor = std::gcd(absoluteNumerator, denominator);
495 if (divisor > 1) {
496 numerator /= divisor;
497 denominator /= divisor;
498 }
499 }
500
501 /// @brief Sets `decimalPlaces` to `max(this, other)`.
502 constexpr void widenPrecisionTo(DecimalPlaces other) noexcept {
503 if (other.value > decimalPlaces.value) {
504 decimalPlaces = other;
505 }
506 }
507};

517/// @brief Absolute value of a rational. Precision is preserved.
518/// @param value Value to take the absolute value of.
519/// @return The non-negative value with the same magnitude.
520[[nodiscard]] constexpr Rational abs(const Rational& value) noexcept {
521 return value.numerator < 0 ? Rational{Numerator{-value.numerator}, Denominator{value.denominator}, value.decimalPlaces}
522 : value;
523}

525/// @brief Rounds toward positive infinity.
526/// @param value Value to round.
527/// @return The smallest integer `>= value`.
528[[nodiscard]] constexpr std::int64_t ceil(const Rational& value) noexcept {
529 auto const quotient = value.numerator / value.denominator;
530 auto const remainder = value.numerator % value.denominator;
531 if (remainder != 0 && value.numerator > 0) {
532 return quotient + 1;
533 }
534 return quotient;
535}
536
537/// @brief Rounds toward negative infinity.
538/// @param value Value to round.
539/// @return The largest integer `<= value`.
540[[nodiscard]] constexpr std::int64_t floor(const Rational& value) noexcept {
541 auto const quotient = value.numerator / value.denominator;
542 auto const remainder = value.numerator % value.denominator;
543 if (remainder != 0 && value.numerator < 0) {
544 return quotient - 1;
545 }
546 return quotient;
547}
548
549/// @brief Truncates toward zero.
550/// @param value Value to truncate.
551/// @return The integer part of the value.
552[[nodiscard]] constexpr std::int64_t trunc(const Rational& value) noexcept {
553 return value.numerator / value.denominator;
554}

611/// @brief Concept: matches `std::expected<Rational, RationalError>`.

647/// @brief The precision to lift a float operand to: the Rational-family
648/// operand's precision. Picks whichever side carries one (at least one
649/// always does).
650template <typename Left, typename Right>
651[[nodiscard]] constexpr DecimalPlaces liftPrecision(const Left& lhs, const Right& rhs) noexcept {
652 if constexpr (IsRational<Left>) {
653 return lhs.decimalPlaces;
654 } else if constexpr (IsExpectedRational<Left>) {
655 return lhs ? lhs->decimalPlaces : DecimalPlaces{1};
656 } else if constexpr (IsRational<Right>) {
657 return rhs.decimalPlaces;
658 } else if constexpr (IsExpectedRational<Right>) {
659 return rhs ? rhs->decimalPlaces : DecimalPlaces{1};
660 } else {
661 return DecimalPlaces{1};
662 }
663}

757/// @brief Computes `10^decimalPlaces` as `std::int64_t`. Returns `0` on overflow.
758[[nodiscard]] constexpr std::int64_t powerOfTen(std::uint32_t decimalPlaces) noexcept {
759 if (decimalPlaces > 18) {
760 return 0;
761 }
762 auto result = std::int64_t{1};
763 for (std::uint32_t digit = 0; digit < decimalPlaces; ++digit) {
764 result *= 10;
765 }
766 return result;
767}
768
769/// @brief Shared implementation of the three `Rational::fromFloat` overloads.
770template <std::floating_point Float>
771[[nodiscard]] inline ExpectedRational fromFloatImpl(Float value, DecimalPlaces wantedPrecision) noexcept {
772 if (!std::isfinite(value)) {
773 return std::unexpected(RationalError::NotFinite);
774 }
775
776 auto const rawPrecision = clampDecimalPlaces(wantedPrecision.value);
777 // rawPrecision is clamped to <= kMaxDecimalPlaces, so powerOfTen cannot
778 // return its overflow sentinel here.
779 auto const scale = powerOfTen(rawPrecision);
780
781 auto const scaled = static_cast<long double>(value) * static_cast<long double>(scale);
782 // Bound against 2^63 (representable in every long double format), minus
783 // the half-ulp llround adds: values in [2^63 - 0.5, 2^63) round *up* to
784 // 2^63 and would overflow int64. Casting INT64_MAX instead would itself
785 // round up to 2^63 where long double == double. The negative bound is
786 // asymmetric because INT64_MIN == -2^63 is a valid result and llround
787 // maps (-2^63 - 0.5, -2^63] onto it, hence `<` against -2^63 exactly.
788 constexpr auto twoPow63 = 0x1p63L;
789 if (scaled >= twoPow63 - 0.5L || scaled < -twoPow63) {
790 return std::unexpected(RationalError::Overflow);
791 }
792
793 auto const scaledNumerator = static_cast<std::int64_t>(std::llround(scaled));
794 return Rational{Numerator{scaledNumerator}, Denominator{scale}, DecimalPlaces{rawPrecision}};
795}

815 auto const raw = static_cast<double>(numerator) / static_cast<double>(denominator);
