13/// @brief Configuration for `NetworkMonitor`.
14///
15/// Declared outside `NetworkMonitor` so that its default member initialisers
16/// are fully parsed before any constructor default argument that names
17/// `Config{}` is evaluated. Compilers (clang, GCC) reject `Config cfg = Config{}`
18/// in a constructor default argument when `Config` is an incomplete nested type.
19struct NetworkMonitorConfig {
20 /// @brief Time between probe calls.
21 std::chrono::milliseconds probeInterval = std::chrono::seconds{5};
22
23 /// @brief Consecutive failures required before going offline.
24 int failureThreshold = 3;
25
26 /// @brief Consecutive successes required before going online.
27 int onlineThreshold = 1;
28};

30/// @brief Background connectivity monitor that fires callbacks on state changes.
31///
32/// A dedicated thread calls the user-supplied probe function at regular intervals.
33/// The monitor starts in the *online* state and transitions to *offline* only after
34/// `failureThreshold` consecutive failures. It returns to *online* after
35/// `onlineThreshold` consecutive successes.
36///
37/// The monitor is non-copyable and non-movable. Destroy it to stop monitoring.
38class NetworkMonitor {
39public:
40 /// @brief Returns `true` when the network is reachable.
41 using ProbeFunction = std::function<bool()>;
42
43 /// @brief Called on connectivity state change.
44 using Callback = std::function<void()>;
45
46 /// @brief Alias for the configuration struct.
47 using Config = NetworkMonitorConfig;
48
49 /// @brief Starts the monitor and launches the probe thread.
50 ///
51 /// The monitor starts in the online state. The first `failureThreshold`
52 /// consecutive probe failures must occur before `onOffline` is invoked.
53 /// Callers that know the initial state is offline should use a probe that
54 /// starts returning `false` immediately.
55 ///
56 /// @param probe Callable that tests connectivity. Must not throw (exceptions are swallowed).
57 /// @param onOffline Called on the probe thread when the monitor goes offline.
58 /// @param onOnline Called on the probe thread when the monitor comes back online.
59 /// @param cfg Tuning parameters (interval, thresholds).
60 NetworkMonitor(ProbeFunction probe, Callback onOffline, Callback onOnline, Config cfg = Config{})
61 : _probe{std::move(probe)},
62 _onOffline{std::move(onOffline)},
63 _onOnline{std::move(onOnline)},
64 _cfg{cfg},
65 _online{true},
66 _stopped{false},
67 _thread{[this] { run(); }} {}
68
69 /// @brief Stops the monitor and waits for the probe thread to exit.
70 ///
71 /// Calls `stop()` internally. The destructor spin-waits on `_runExited` to
72 /// handle the case where `stop()` was called from within a probe callback
73 /// (which would deadlock a `join()`).
74 ~NetworkMonitor() {
75 stop();
76 while (!_runExited.load()) {
77 std::this_thread::yield();
78 }
79 }
80
81 NetworkMonitor(const NetworkMonitor&) = delete;
82 NetworkMonitor& operator=(const NetworkMonitor&) = delete;
83 NetworkMonitor(NetworkMonitor&&) = delete;
84 NetworkMonitor& operator=(NetworkMonitor&&) = delete;
85
86 /// @brief Returns `true` if the monitor currently considers the network reachable.
87 ///
88 /// Reads an atomic flag — safe to call from any thread at any time.
89 ///
90 /// @return Current online state.
91 [[nodiscard]] bool isOnline() const noexcept { return _online.load(); }
92
93 /// @brief Signals the probe thread to stop and, if possible, joins it.
94 ///
95 /// Idempotent — subsequent calls are no-ops. If called from inside the probe
96 /// thread (e.g. from a probe callback), the thread is detached instead of
97 /// joined to avoid a deadlock; the destructor then spin-waits for it to exit.
98 ///
99 /// Thread-safe.
100 void stop() {
101 {
102 std::scoped_lock const lock{_mtx};
103 if (_stopped) {
104 return;
105 }
106 _stopped = true;
107 }
108 _cv.notify_all();
109 if (_thread.joinable()) {
110 if (_thread.get_id() == std::this_thread::get_id()) {
111 _thread.detach();
112 } else {
113 _thread.join();
114 }
115 }
116 }
117
118private:
119 static bool safeProbe(const ProbeFunction& probe) noexcept {
120 try {
121 return probe ? probe() : false;
122 } catch (...) {
123 return false;
124 }
125 }
126
127 void handleProbeResult(bool probeOk, int& consecutiveFailures, int& consecutiveSuccesses) {
128 if (!probeOk) {
129 consecutiveSuccesses = 0;
130 ++consecutiveFailures;
131 if (_online && consecutiveFailures >= _cfg.failureThreshold) {
132 _online.store(false);
133 if (_onOffline) {
134 _onOffline();
135 }
136 }
137 } else {
138 consecutiveFailures = 0;
139 ++consecutiveSuccesses;
140 if (!_online && consecutiveSuccesses >= _cfg.onlineThreshold) {
141 _online.store(true);
142 if (_onOnline) {
143 _onOnline();
144 }
145 }
146 }
147 }
148
149 void run() {
150 int consecutiveFailures = 0;
151 int consecutiveSuccesses = 0;
152
153 while (true) {
154 {
155 std::unique_lock lock{_mtx};
156 _cv.wait_for(lock, _cfg.probeInterval, [this] { return _stopped.load(); });
157 if (_stopped) {
158 break;
159 }
160 }
161 handleProbeResult(safeProbe(_probe), consecutiveFailures, consecutiveSuccesses);
162 }
163 _runExited.store(true);
164 }
165
166 ProbeFunction _probe;
167 Callback _onOffline;
168 Callback _onOnline;
169 Config _cfg;
170 std::atomic<bool> _online;
171 std::atomic<bool> _stopped;
172 std::atomic<bool> _runExited{false};
173 std::mutex _mtx;
174 std::condition_variable _cv;
175 std::thread _thread;
176};
