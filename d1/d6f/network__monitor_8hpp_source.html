15/// @brief Configuration for `NetworkMonitor`.
16///
17/// Declared outside `NetworkMonitor` so that its default member initialisers
18/// are fully parsed before any constructor default argument that names
19/// `Config{}` is evaluated. Compilers (clang, GCC) reject `Config cfg = Config{}`
20/// in a constructor default argument when `Config` is an incomplete nested type.
21struct NetworkMonitorConfig {
22 /// @brief Time between probe calls.
23 std::chrono::milliseconds probeInterval = std::chrono::seconds{5};
24
25 /// @brief Consecutive failures required before going offline.
26 int failureThreshold = 3;
27
28 /// @brief Consecutive successes required before going online.
29 int onlineThreshold = 1;
30};

32/// @brief Background connectivity monitor that fires callbacks on state changes.
33///
34/// A dedicated thread calls the user-supplied probe function at regular intervals.
35/// The monitor starts in the *online* state and transitions to *offline* only after
36/// `failureThreshold` consecutive failures. It returns to *online* after
37/// `onlineThreshold` consecutive successes.
38///
39/// The monitor is non-copyable and non-movable. Destroy it to stop monitoring.
40class NetworkMonitor {
41public:
42 /// @brief Returns `true` when the network is reachable.
43 using ProbeFunction = std::function<bool()>;
44
45 /// @brief Called on connectivity state change.
46 using Callback = std::function<void()>;
47
48 /// @brief Alias for the configuration struct.
49 using Config = NetworkMonitorConfig;
50
51 /// @brief Starts the monitor and launches the probe thread.
52 ///
53 /// The monitor starts in the online state. The first `failureThreshold`
54 /// consecutive probe failures must occur before `onOffline` is invoked.
55 /// Callers that know the initial state is offline should use a probe that
56 /// starts returning `false` immediately.
57 ///
58 /// @param probe Callable that tests connectivity. Must not throw
59 /// (exceptions are swallowed). Stored and invoked on the
60 /// probe thread for this monitor's whole lifetime, so
61 /// anything the callable refers to must outlive the monitor.
62 /// @param onOffline Called on the probe thread when the monitor goes offline.
63 /// Retained on the same terms as @p probe.
64 /// @param onOnline Called on the probe thread when the monitor comes back
65 /// online. Retained on the same terms as @p probe.
66 /// @param cfg Tuning parameters (interval, thresholds).
67 NetworkMonitor(ProbeFunction probe MORPH_LIFETIMEBOUND, Callback onOffline MORPH_LIFETIMEBOUND,
68 Callback onOnline MORPH_LIFETIMEBOUND, Config cfg = Config{})
69 : _probe{std::move(probe)},
70 _onOffline{std::move(onOffline)},
71 _onOnline{std::move(onOnline)},
72 _cfg{cfg},
73 _online{true},
74 _stopped{false},
75 _thread{[this] { run(); }} {}
76
77 /// @brief Stops the monitor and waits for the probe thread to exit.
78 ///
79 /// Calls `stop()` internally. The destructor spin-waits on `_runExited` to
80 /// handle the case where `stop()` was called from within a probe callback
81 /// (which would deadlock a `join()`).
82 ///
83 /// The destructor itself must not run on the probe thread: this spin-wait
84 /// blocks until `run()` sets `_runExited`, but `run()` cannot reach that
85 /// store while it is paused lower on the same thread's call stack waiting
86 /// for a callback (and this destructor, called from within it) to return.
87 /// A callback that drops the owning object's last reference and thereby
88 /// triggers this destructor synchronously, on the probe thread, deadlocks
89 /// here by construction — not a supported lifetime. The supported,
90 /// documented seam for a self-thread teardown request is calling `stop()`
91 /// itself from the callback (see its doc comment and `_thread.get_id() ==
92 /// std::this_thread::get_id()` below): that only detaches and returns,
93 /// letting `run()` continue and exit on its own; the actual destructor
94 /// runs later, from a different thread, once the owner is done reacting
95 /// to the callback.
96 ~NetworkMonitor() {
97 stop();
98 while (!_runExited.load()) {
99 std::this_thread::yield();
100 }
101 }
102
103 NetworkMonitor(const NetworkMonitor&) = delete;
104 NetworkMonitor& operator=(const NetworkMonitor&) = delete;
105 NetworkMonitor(NetworkMonitor&&) = delete;
106 NetworkMonitor& operator=(NetworkMonitor&&) = delete;

108 /// @brief Returns `true` if the monitor currently considers the network reachable.
109 ///
110 /// Reads an atomic flag — safe to call from any thread at any time.
111 ///
112 /// @return Current online state.
113 [[nodiscard]] bool isOnline() const noexcept { return _online.load(); }
114
115 /// @brief Signals the probe thread to stop and, if possible, joins it.
116 ///
117 /// Idempotent — subsequent calls are no-ops. If called from inside the probe
118 /// thread (e.g. from a probe callback), the thread is detached instead of
119 /// joined to avoid a deadlock; the destructor then spin-waits for it to exit.
120 ///
121 /// Thread-safe.
122 void stop() {
123 {
124 std::scoped_lock const lock{_mtx};
125 if (_stopped) {
126 return;
127 }
128 _stopped = true;
129 }
130 _cv.notify_all();
131 if (_thread.joinable()) {
132 if (_thread.get_id() == std::this_thread::get_id()) {
133 _thread.detach();
134 } else {
135 _thread.join();
136 }
137 }
138 }
139
140private:
141 static bool safeProbe(const ProbeFunction& probe) noexcept {
142 try {
143 return probe ? probe() : false;
144 } catch (...) {
145 return false;
146 }
147 }
148
149 void handleProbeResult(bool probeOk, int& consecutiveFailures, int& consecutiveSuccesses) {
150 if (!probeOk) {
151 consecutiveSuccesses = 0;
152 ++consecutiveFailures;
153 if (_online && consecutiveFailures >= _cfg.failureThreshold) {
154 _online.store(false);
155 if (_onOffline) {
156 _onOffline();
157 }
158 }
159 } else {
160 consecutiveFailures = 0;
161 ++consecutiveSuccesses;
162 if (!_online && consecutiveSuccesses >= _cfg.onlineThreshold) {
163 _online.store(true);
164 if (_onOnline) {
165 _onOnline();
166 }
167 }
168 }
169 }
170
171 void run() {
172 int consecutiveFailures = 0;
173 int consecutiveSuccesses = 0;
174
175 while (true) {
176 {
177 std::unique_lock lock{_mtx};
178 _cv.wait_for(lock, _cfg.probeInterval, [this] { return _stopped.load(); });
179 if (_stopped) {
180 break;
181 }
182 }
183 handleProbeResult(safeProbe(_probe), consecutiveFailures, consecutiveSuccesses);
184 }
185 _runExited.store(true);
186 }
187
188 ProbeFunction _probe;
189 Callback _onOffline;
190 Callback _onOnline;
191 Config _cfg;
192 std::atomic<bool> _online;
193 std::atomic<bool> _stopped;
194 std::atomic<bool> _runExited{false};
195 std::mutex _mtx;
196 std::condition_variable _cv;
197 std::thread _thread;
198};
