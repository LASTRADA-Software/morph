5/// @file forms/choice.hpp
6/// @brief Fields whose value is picked from a list served by another action.
7///
8/// A `morph::forms::Choice<T, "ListPayees">` member declares, in the type:
9/// *this field is not free input — its options are the result of executing
10/// the named action.* Renderers turn it into a combo box: they execute the
11/// options action (over the same wire the form actions use), read the result
12/// rows, and offer `labelField` as the display text while submitting
13/// `valueField` as the payload.
14///
15/// @code{.cpp}
16/// struct SampleInfo { std::int64_t id = 0; std::string name; };
17/// struct SampleList { std::vector<SampleInfo> samples; };
18/// struct ListSamples {}; // registered like any other action (a pure query)
19///
20/// struct RecordMeasurement {
21/// morph::forms::Choice<std::int64_t, "ListSamples"> sampleId; // combo box
22/// ...
23/// };
24/// @endcode
25///
26/// In the generated schema the declaration surfaces as
27/// `x-optionsAction` / `x-optionValue` / `x-optionLabel` on the property
28/// (added by `morph::forms::schemaJson`), so a client knows *which action to
29/// call* and *which result fields to use* without hardcoding anything.
30///
31/// A `Choice` can also declare a `DependsOn` pack naming sibling fields whose
32/// current values parameterise the options action — a cascading picklist
33/// (e.g. the list of cities depends on the selected country). The options
34/// action then receives `{name: value, ...}` instead of an empty body, and
35/// the schema additionally carries `x-optionsDependsOn`. See
36/// `optionsDependsOn()` below; a `Choice` with no `DependsOn` is unaffected.
37///
38/// On the wire a `Choice` is its nullable underlying value (`T`); the
39/// options metadata never travels with payloads. Like `Quantity` and
40/// `Timestamp`, the blank state ("nothing selected") lives inside, and a
41/// non-optional `Choice` member is *required* by the `morph::forms` rules.

55/// @brief A structural, NTTP-capable compile-time string (for naming the
56/// options action and its value/label result fields in the type).
57///
58/// An alias for the shared `morph::detail::FixedString` — the same underlying
59/// type the units layer's `NamedQuantity` uses — so there is one definition, not
60/// two look-alikes. Kept as a public `morph::forms::` name for source
61/// compatibility.
62/// @tparam N Storage size including the terminating null.
63template <std::size_t N>
64using FixedString = ::morph::detail::FixedString<N>;
65
66/// @brief An optionally-empty value selected from options served by another
67/// action.
68///
69/// @tparam T Underlying value type submitted on the wire (e.g.
70/// `std::int64_t` for ids, `std::string` for codes).
71/// @tparam OptionsAction Type id of the registered action whose result
72/// provides the options. Executed with an empty body
73/// when `DependsOn` is empty (the common case);
74/// otherwise with a body built from the current
75/// values of the named sibling fields.
76/// @tparam ValueField Field of each result row submitted as the value.
77/// @tparam LabelField Field of each result row shown to the user.
78/// @tparam DependsOn Wire (JSON) field names of sibling fields in the
79/// same action whose current values parameterise the
80/// options action — a cascading picklist. Empty by
81/// default, which makes the options action
82/// independent (today's behavior, unchanged).
83template <typename T, FixedString OptionsAction, FixedString ValueField = "id", FixedString LabelField = "name",
84 FixedString... DependsOn>
85struct Choice {
86 /// @brief The payload; `std::nullopt` means "nothing selected".
87 std::optional<T> value;
88
89 /// @brief Constructs the empty state.
90 constexpr Choice() noexcept = default;
91
92 /// @brief Engages with @p selected.
93 /// @param selected The selected value.
94 constexpr Choice(T selected) noexcept(std::is_nothrow_move_constructible_v<T>) : value{std::move(selected)} {}
95
96 /// @brief Adopts an optional payload as-is.
97 /// @param payload Engaged or empty payload.
98 constexpr Choice(std::optional<T> payload) noexcept(std::is_nothrow_move_constructible_v<T>)
99 : value{std::move(payload)} {}
100
101 /// @brief Type id of the action that serves the options.
102 /// @return The action name declared in the field's type.
103 [[nodiscard]] static constexpr std::string_view optionsAction() noexcept { return OptionsAction.view(); }
104
105 /// @brief Result-row field submitted as the value.
106 /// @return The field name declared in the field's type.
107 [[nodiscard]] static constexpr std::string_view valueField() noexcept { return ValueField.view(); }
108
109 /// @brief Result-row field displayed to the user.
110 /// @return The field name declared in the field's type.
111 [[nodiscard]] static constexpr std::string_view labelField() noexcept { return LabelField.view(); }
112
113 /// @brief Wire field names of sibling fields whose current values
114 /// parameterise the options action (a cascading picklist).
115 /// @return The declared `DependsOn` names, in declaration order; empty
116 /// for an independent `Choice` (the default).
117 [[nodiscard]] static constexpr std::array<std::string_view, sizeof...(DependsOn)> optionsDependsOn() noexcept {
118 return {DependsOn.view()...};
119 }
120
121 /// @brief Whether a value has been selected.
122 /// @return `true` if the payload is engaged.
123 [[nodiscard]] constexpr bool hasValue() const noexcept { return value.has_value(); }
124
125 /// @brief Unchecked access to the selected value (UB when empty, exactly
126 /// like `std::optional` — the unchecked contract is the point).
127 /// @return The selected value — a reference into this `Choice`, valid only
128 /// for as long as it is.
129 // NOLINTNEXTLINE(bugprone-unchecked-optional-access)
130 [[nodiscard]] constexpr const T& operator*() const noexcept MORPH_LIFETIMEBOUND { return *value; }
131
132 /// @brief Equality on the payload; empty equals only empty.
133 /// @param other Choice to compare against.
134 /// @return `true` when both are empty or both hold equal values.
135 [[nodiscard]] constexpr bool operator==(const Choice& other) const noexcept = default;
136};
