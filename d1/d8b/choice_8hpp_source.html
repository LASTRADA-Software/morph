5/// @file choice.hpp
6/// @brief Fields whose value is picked from a list served by another action.
7///
8/// A `morph::forms::Choice<T, "ListPayees">` member declares, in the type:
9/// *this field is not free input — its options are the result of executing
10/// the named action.* Renderers turn it into a combo box: they execute the
11/// options action (over the same wire the form actions use), read the result
12/// rows, and offer `labelField` as the display text while submitting
13/// `valueField` as the payload.
14///
15/// @code{.cpp}
16/// struct SampleInfo { std::int64_t id = 0; std::string name; };
17/// struct SampleList { std::vector<SampleInfo> samples; };
18/// struct ListSamples {}; // registered like any other action (a pure query)
19///
20/// struct RecordMeasurement {
21/// morph::forms::Choice<std::int64_t, "ListSamples"> sampleId; // combo box
22/// ...
23/// };
24/// @endcode
25///
26/// In the generated schema the declaration surfaces as
27/// `x-optionsAction` / `x-optionValue` / `x-optionLabel` on the property
28/// (added by `morph::forms::schemaJson`), so a client knows *which action to
29/// call* and *which result fields to use* without hardcoding anything.
30///
31/// On the wire a `Choice` is its nullable underlying value (`T`); the
32/// options metadata never travels with payloads. Like `Quantity` and
33/// `Timestamp`, the blank state ("nothing selected") lives inside, and a
34/// non-optional `Choice` member is *required* by the `morph::forms` rules.

46/// @brief A structural, NTTP-capable compile-time string (for naming the
47/// options action and its value/label result fields in the type).
48/// @tparam N Storage size including the terminating null.
49template <std::size_t N>
50struct FixedString {
51 /// @brief Character storage (null-terminated).
52 std::array<char, N> data{};
53
54 /// @brief Captures a string literal.
55 /// @param literal The literal to copy, e.g. `"ListSamples"`.
56 // NOLINTNEXTLINE(modernize-avoid-c-arrays) — string literals ARE C arrays.
57 consteval FixedString(const char (&literal)[N]) noexcept {
58 for (std::size_t index = 0; index < N; ++index) {
59 data[index] = literal[index];
60 }
61 }
62
63 /// @brief A view of the string (without the terminating null).
64 /// @return The string contents.
65 [[nodiscard]] constexpr std::string_view view() const noexcept { return {data.data(), N - 1}; }
66};

68/// @brief An optionally-empty value selected from options served by another
69/// action.
70///
71/// @tparam T Underlying value type submitted on the wire (e.g.
72/// `std::int64_t` for ids, `std::string` for codes).
73/// @tparam OptionsAction Type id of the registered action whose result
74/// provides the options (executed with an empty body).
75/// @tparam ValueField Field of each result row submitted as the value.
76/// @tparam LabelField Field of each result row shown to the user.
77template <typename T, FixedString OptionsAction, FixedString ValueField = "id", FixedString LabelField = "name">
78struct Choice {
79 /// @brief The payload; `std::nullopt` means "nothing selected".
80 std::optional<T> value;
81
82 /// @brief Constructs the empty state.
83 constexpr Choice() noexcept = default;
84
85 /// @brief Engages with @p selected.
86 /// @param selected The selected value.
87 constexpr Choice(T selected) noexcept(std::is_nothrow_move_constructible_v<T>) : value{std::move(selected)} {}
88
89 /// @brief Adopts an optional payload as-is.
90 /// @param payload Engaged or empty payload.
91 constexpr Choice(std::optional<T> payload) noexcept(std::is_nothrow_move_constructible_v<T>)
92 : value{std::move(payload)} {}
93
94 /// @brief Type id of the action that serves the options.
95 /// @return The action name declared in the field's type.
96 [[nodiscard]] static constexpr std::string_view optionsAction() noexcept { return OptionsAction.view(); }
97
98 /// @brief Result-row field submitted as the value.
99 /// @return The field name declared in the field's type.
100 [[nodiscard]] static constexpr std::string_view valueField() noexcept { return ValueField.view(); }
101
102 /// @brief Result-row field displayed to the user.
103 /// @return The field name declared in the field's type.
104 [[nodiscard]] static constexpr std::string_view labelField() noexcept { return LabelField.view(); }
105
106 /// @brief Whether a value has been selected.
107 /// @return `true` if the payload is engaged.
108 [[nodiscard]] constexpr bool hasValue() const noexcept { return value.has_value(); }
109
110 /// @brief Unchecked access to the selected value (UB when empty, exactly
111 /// like `std::optional` — the unchecked contract is the point).
112 /// @return The selected value.
113 // NOLINTNEXTLINE(bugprone-unchecked-optional-access)
114 [[nodiscard]] constexpr const T& operator*() const noexcept { return *value; }
115
116 /// @brief Equality on the payload; empty equals only empty.
117 /// @param other Choice to compare against.
118 /// @return `true` when both are empty or both hold equal values.
119 [[nodiscard]] constexpr bool operator==(const Choice& other) const noexcept = default;
120};
