17/// @file
18/// @brief Per-model execute-ordering gate, extracted out of `RemoteServer`.
19///
20/// `RemoteServer`'s per-model execute-ordering gate (`ExecuteGate` before this
21/// extraction, `takeExecuteTicket`/`awaitExecuteTurn`/`releaseExecuteTicket`,
22/// and `ExecuteTicketGuard` — all `private`) is a standalone concurrency
23/// primitive that was trapped inside the server. Its contract — hand out
24/// monotonic tickets per `ModelId`; block a ticket until its predecessor
25/// releases; tolerate releases arriving out of order; erase the gate when
26/// drained; tolerate a gate already erased — touches no wire format, no
27/// authorizer, no model, no executor, no strand, and is expressible without
28/// `RemoteServer` at all.
29///
30/// This is a faithful, behavior-preserving port of the logic that used to
31/// live directly on `RemoteServer`: same fields, same locking, same
32/// out-of-order-release handling (`releasedOutOfOrder`, closing issue #449 --
33/// already fixed upstream before this extraction; see that field's own doc
34/// comment below for the full history). Nothing here is new logic.
35namespace morph::backend::detail {
36
37/// @brief Hands out monotonic per-`ModelId` tickets and lets callers block
38/// until their ticket's turn comes, tolerating releases that arrive
39/// out of ticket order.
40///
41/// Keyed by `ModelId`, not held forever: a model with no outstanding tickets
42/// has no entry in the internal map at all (erased once its last ticket is
43/// released), so this never grows unbounded across the gate's lifetime the
44/// way a per-model map with no cleanup would.
45class ExecuteOrderGate {
46public:
47 /// @brief Hands out the next ticket for @p mid, in call order.
48 ///
49 /// Intended to be called synchronously, on whatever thread received the
50 /// request, before anything asynchronous happens -- the ticket numbers
51 /// two calls receive for the same `mid` are therefore always in the
52 /// order this was called in.
53 /// @param mid The model the upcoming turn is for.
54 /// @return This call's ticket number.
55 [[nodiscard]] std::uint64_t take(::morph::exec::detail::ModelId mid) {
56 std::scoped_lock const lock{_mtx};
57 auto& gate = _gates[mid];
58 if (!gate) {
59 gate = std::make_shared<Gate>();
60 }
61 return gate->nextTicket++;
62 }
63
64 /// @brief Blocks until @p ticket is next in line for @p mid, then returns.
65 /// @param mid The model @p ticket was taken for.
66 /// @param ticket This call's ticket, from `take`.
67 void awaitTurn(::morph::exec::detail::ModelId mid, std::uint64_t ticket) {
68 std::unique_lock lock{_mtx};
69 auto iter = _gates.find(mid);
70 if (iter == _gates.end()) {
71 return; // Nothing left to wait for -- every ticket for mid already released.
72 }
73 auto gate = iter->second; // Keep it alive even if release() erases the map entry mid-wait.
74 gate->cv.wait(lock, [&gate, ticket] { return gate->nextToRun == ticket; });
75 }
76
77 /// @brief Releases @p ticket for @p mid, letting the next ticket (if any) proceed.
78 ///
79 /// Must be called exactly once per ticket taken. A ticket that is taken
80 /// but never released would permanently stall every later ticket for the
81 /// same `mid`, since `awaitTurn` waits with no deadline. (`RemoteServer`
82 /// enforces the "exactly once" half of that contract structurally, via
83 /// `ExecuteTicketGuard` below -- see its own doc comment.)
84 /// @param mid The model @p ticket was taken for.
85 /// @param ticket The ticket to release.
86 void release(::morph::exec::detail::ModelId mid, std::uint64_t ticket) {
87 std::scoped_lock const lock{_mtx};
88 auto iter = _gates.find(mid);
89 if (iter == _gates.end()) {
90 // A supported call, not an impossibility: the file-level contract
91 // above lists "tolerate a gate already erased" as a first-class
92 // element, and tests/test_execute_order_gate.cpp names the case
93 // ("releasing a ticket for a model with no gate entry is a
94 // harmless no-op").
95 return;
96 }
97 auto& gate = *iter->second;
98 // `nextToRun` is the lowest ticket that has *not* released yet, which
99 // is exactly what `awaitTurn`'s `nextToRun == ticket` predicate needs
100 // it to be. Advancing it to `ticket + 1` unconditionally was only
101 // correct if tickets released in order, and callers deliberately do
102 // not guarantee that: a caller may need to release a ticket
103 // immediately, without ever waiting for its turn, precisely so that
104 // ticket cannot hold up the live work behind it (`RemoteServer`'s
105 // rejection paths -- model not found, unauthorized, over limit, a
106 // shutdown gate -- all do exactly this). A later ticket releasing
107 // first therefore used to push `nextToRun` straight past an earlier
108 // ticket's number, whose waiter then had a predicate that could never
109 // become true again -- a caller parked forever (issue #449, the third
110 // occurrence of the stranded-ticket class #348 and #351 closed from
111 // the other end by making the release itself unmissable).
112 //
113 // So an out-of-order release is recorded rather than applied, and
114 // `nextToRun` walks forward only over a contiguous run of released
115 // tickets. `ticket < nextToRun` cannot happen: every ticket is
116 // released exactly once, and `nextToRun` only moves past tickets that
117 // have already released.
118 if (ticket == gate.nextToRun) {
119 ++gate.nextToRun;
120 while (!gate.releasedOutOfOrder.empty() && *gate.releasedOutOfOrder.begin() == gate.nextToRun) {
121 gate.releasedOutOfOrder.erase(gate.releasedOutOfOrder.begin());
122 ++gate.nextToRun;
123 }
124 } else {
125 gate.releasedOutOfOrder.insert(ticket);
126 }
127 if (gate.nextToRun == gate.nextTicket) {
128 // No ticket is currently waiting and none can arrive for a ticket
129 // number already handed out -- safe to drop the entry so a model
130 // with no in-flight tickets leaves no trace in this map. Reaching
131 // `nextTicket` this way means every ticket handed out has
132 // released, so `releasedOutOfOrder` is necessarily empty here.
133 _gates.erase(iter);
134 } else {
135 gate.cv.notify_all();
136 }
137 }
138
139 /// @brief Number of models with at least one outstanding (unreleased) ticket.
140 ///
141 /// Test-only observability, mirroring `PendingCallTable::size()` --
142 /// stale the moment the lock is released, so it must not drive a
143 /// check-then-act decision.
144 /// @return Current map size.
145 [[nodiscard]] std::size_t gateCount() const {
146 std::scoped_lock const lock{_mtx};
147 return _gates.size();
148 }
149
150private:
151 struct Gate {
152 std::uint64_t nextTicket = 0;
153 std::uint64_t nextToRun = 0;
154 // Tickets released *before* their turn came, held aside until every
155 // ticket ahead of them has released too. Not an optimisation: it is
156 // what makes `nextToRun` mean "the lowest ticket not yet released"
157 // rather than "one past whichever ticket released last". See
158 // `release` (issue #449). Ordered, because the release loop consumes
159 // it from the front; small by construction (it holds at most the
160 // tickets in flight for one model, minus one).
161 std::set<std::uint64_t> releasedOutOfOrder;
162 std::condition_variable cv;
163 };
164
165 mutable std::mutex _mtx;
166 std::unordered_map<::morph::exec::detail::ModelId, std::shared_ptr<Gate>, ::morph::exec::detail::ModelIdHash>
167 _gates;
168};

170/// @brief Owns a taken `ExecuteOrderGate` ticket and releases it on every exit
171/// path, unless ownership was explicitly handed on.
172///
173/// A ticket handed out by `ExecuteOrderGate::take` must be released exactly
174/// once by whatever path took it: an unreleased ticket permanently stalls
175/// every later ticket for the same model, because `awaitTurn` is a `cv.wait`
176/// with no deadline. In `RemoteServer` (the sole caller today) that rule used
177/// to be a per-call-site convention, and the convention was missed twice: by
178/// a shutdown gate that returns before the one place that released a ticket
179/// (issue #348), and by every exception that unwinds past a dispatch
180/// function's early returns (issue #351). This holder makes the rule
181/// structural instead of remembered: the ticket is owned from the moment it
182/// is taken until the guard dies, so every exit path -- `return`, `throw`,
183/// and any branch a later change adds -- releases it. Two members opt out
184/// deliberately:
185///
186/// - `release()`, for the paths that want the ticket freed *before* some
187/// later, possibly-slow step, so that step never holds a later same-model
188/// ticket up; and
189/// - `disarm()`, for the one place ownership is genuinely handed on to
190/// another owner (e.g. a task posted to a different executor).
191///
192/// Neither copyable nor movable: it is only ever a local of the frame that
193/// owns the ticket.
194class ExecuteTicketGuard {
195public:
196 /// @brief Adopts @p ticket on behalf of @p gate.
197 /// @param gate Gate @p ticket belongs to. Borrowed: a guard is always a
198 /// local of a frame that does not outlive the gate.
199 /// @param ticket This call's ticket from `ExecuteOrderGate::take`, or
200 /// `std::nullopt` if it took none, in which case the guard
201 /// is inert.
202 ExecuteTicketGuard(ExecuteOrderGate& gate MORPH_LIFETIMEBOUND,
203 std::optional<std::pair<::morph::exec::detail::ModelId, std::uint64_t>> ticket) noexcept
204 : _gate{gate}, _ticket{std::move(ticket)} {}
205
206 ExecuteTicketGuard(const ExecuteTicketGuard&) = delete;
207 ExecuteTicketGuard(ExecuteTicketGuard&&) = delete;
208 ExecuteTicketGuard& operator=(const ExecuteTicketGuard&) = delete;
209 ExecuteTicketGuard& operator=(ExecuteTicketGuard&&) = delete;
210
211 /// @brief Releases the ticket if it is still held -- the backstop that
212 /// covers every path which did not release explicitly.
213 ~ExecuteTicketGuard() { release(); }
214
215 /// @brief Releases the held ticket now, letting the next ticket for the
216 /// same model proceed. Idempotent: a no-op if no ticket was taken,
217 /// or if it has already been released or disarmed.
218 void release() {
219 if (!_ticket) {
220 return;
221 }
222 auto const held = *_ticket;
223 // Cleared before the release, not after: this guard owes nothing
224 // further from here on, so the destructor cannot release twice even
225 // if `ExecuteOrderGate::release` itself exits by exception.
226 _ticket.reset();
227 _gate.release(held.first, held.second);
228 }
229
230 /// @brief Gives up ownership *without* releasing, for the one path that
231 /// hands the ticket on to a new owner.
232 void disarm() noexcept { _ticket.reset(); }
233
234 /// @brief Blocks until the held ticket's turn comes (see
235 /// `ExecuteOrderGate::awaitTurn`). A no-op if no ticket is held.
236 void awaitTurn() {
237 if (_ticket) {
238 _gate.awaitTurn(_ticket->first, _ticket->second);
239 }
240 }
241
242private:
243 ExecuteOrderGate& _gate;
244 std::optional<std::pair<::morph::exec::detail::ModelId, std::uint64_t>> _ticket;
245};
