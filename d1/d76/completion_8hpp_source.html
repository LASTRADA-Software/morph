22template <typename T>
23struct CompletionState {
24 std::mutex mtx;
25 std::optional<T> value;
26 std::exception_ptr error;
27 bool ready = false;
28 // Every handler attached while the state is not yet ready is kept (not
29 // overwritten): a second/third attach composes with earlier ones instead of
30 // silently discarding them. Dispatch invokes all of them, in attachment
31 // order, from a single posted closure. See docs/spec/core/completion.md,
32 // "Failure modes" / fan-out.
33 std::vector<std::function<void(T)>> onOk;
34 std::vector<std::function<void(std::exception_ptr)>> onErr;
35 bool onErrAttached = false;
36 ::morph::exec::IExecutor* cbExec = nullptr;
37
38 void setValue(T val) {
39 std::function<void()> callback;
40 {
41 std::scoped_lock const lock{mtx};
42 if (ready) {
43 return;
44 }
45 value = std::move(val);
46 ready = true;
47 if (!onOk.empty()) {
48 auto savedFns = std::move(onOk);
49 auto savedVal = std::move(*value);
50 callback = [savedFns = std::move(savedFns), savedVal = std::move(savedVal)]() mutable {
51 // Every handler but the last sees a copy (the value is only
52 // moved into the final invocation), so an earlier handler
53 // cannot leave the value moved-from for a later one. Each
54 // handler is isolated in its own try/catch so one throwing
55 // handler cannot prevent its siblings from running --
56 // fan-out means every attached handler gets its turn,
57 // independent of whether an earlier one misbehaves. An
58 // escaping exception here would otherwise unwind the whole
59 // posted closure and silently skip every handler after the
60 // one that threw.
61 for (std::size_t i = 0; i + 1 < savedFns.size(); ++i) {
62 try {
63 savedFns[i](savedVal);
64 } catch (...) {
65 ::morph::log::logError("[completion] then handler threw; continuing with next handler");
66 }
67 }
68 try {
69 savedFns.back()(std::move(savedVal));
70 } catch (...) {
71 ::morph::log::logError("[completion] then handler threw; continuing with next handler");
72 }
73 };
74 }
75 }
76 if (callback != nullptr && cbExec != nullptr) {
77 cbExec->post(std::move(callback));
78 }
79 }
80 void setException(const std::exception_ptr& exc) {
81 std::function<void()> callback;
82 {
83 std::scoped_lock const lock{mtx};
84 if (ready) {
85 return;
86 }
87 error = exc;
88 ready = true;
89 if (!onErr.empty()) {
90 auto savedFns = std::move(onErr);
91 auto savedErr = error;
92 callback = [savedFns = std::move(savedFns), savedErr]() mutable {
93 // Isolate each handler so one throwing onError handler
94 // cannot suppress its siblings -- see the matching comment
95 // in setValue's callback above.
96 for (auto& fn : savedFns) {
97 try {
98 fn(savedErr);
99 } catch (...) {
100 ::morph::log::logError("[completion] onError handler threw; continuing with next handler");
101 }
102 }
103 };
104 // Only mark the error handled (suppressing the destructor's orphan
105 // log) if we actually have an executor to deliver on. With a null
106 // executor the callback below is never posted, so the error must
107 // still reach the orphan logger rather than vanish silently.
108 onErrAttached = (cbExec != nullptr);
109 }
110 }
111 if (callback != nullptr && cbExec != nullptr) {
112 cbExec->post(std::move(callback));
113 }
114 }
115 void attachThen(std::function<void(T)> handler) {
116 std::function<void()> fireNow;
117 {
118 std::scoped_lock const lock{mtx};
119 if (ready && value) {
120 auto savedVal = *value;
121 fireNow = [handler = std::move(handler), savedVal]() mutable { handler(std::move(savedVal)); };
122 } else if (!ready) {
123 onOk.push_back(std::move(handler));
124 }
125 }
126 if (fireNow != nullptr && cbExec != nullptr) {
127 cbExec->post(std::move(fireNow));
128 }
129 }
130 void attachOnError(std::function<void(std::exception_ptr)> handler) {
131 std::function<void()> fireNow;
132 {
133 std::scoped_lock const lock{mtx};
134 // See setException: only suppress orphan logging when an executor
135 // exists to actually deliver the handler; a null executor otherwise
136 // drops the error and silences the orphan logger both at once.
137 onErrAttached = (cbExec != nullptr);
138 if (ready && error) {
139 auto savedErr = error;
140 fireNow = [handler = std::move(handler), savedErr]() mutable { handler(savedErr); };
141 } else if (!ready) {
142 onErr.push_back(std::move(handler));
143 }
144 }
145 if (fireNow != nullptr && cbExec != nullptr) {
146 cbExec->post(std::move(fireNow));
147 }
148 }
149 ~CompletionState() {
150 if (!ready || !error || onErrAttached) {
151 return;
152 }
153 // No local guard around the logging calls: `morph::log`'s helpers are
154 // `noexcept` (docs/spec/core/logger.md, "Failure modes"), which is
155 // what makes them safe to call from a destructor at all. The variadic
156 // overload is deliberate — it formats *inside* that guarantee, where
157 // building the message by concatenation out here would allocate
158 // outside it and could still escape.
159 try {
160 std::rethrow_exception(error);
161 } catch (const std::exception& exc) {
162 ::morph::log::logError("[orphan] unhandled exception: {}", exc.what());
163 } catch (...) {
164 ::morph::log::logError("[orphan] unhandled unknown exception");
165 }
166 }
167};

173/// @brief Move-only handle representing the eventual result of an asynchronous operation.
174///
175/// Callbacks are posted to the `IExecutor` supplied at construction time, so
176/// they always run on the intended thread (e.g. the GUI thread).
177///
178/// @par Thread safety
179/// `then()` and `onError()` may be called from any thread. The registered
180/// callbacks are invoked via the executor, never directly from the producing thread.
181///
182/// @par Orphan detection
183/// If a `Completion` is destroyed before an `onError()` handler is attached and
184/// the operation has already failed, the exception is logged as an orphan error.
185///
186/// @par Lifetime and stop gating
187/// Because delivery always goes through an executor, the receiver can be
188/// destroyed — or simply lose interest — before the handler runs. Pass a
189/// `morph::async::CallbackScope` (or one of its `CallbackToken`s) as the first
190/// argument to `then()` / `onError()` and the handler is refused unless the
191/// scope is still alive and un-stopped at delivery time. `thenDetached()` /
192/// `onErrorDetached()` are the same ungated attachments as `then(fn)` /
193/// `onError(fn)`, spelled so a deliberately unmanaged callback says so.
194/// See `callback_scope.hpp` and docs/spec/core/callback_scope.md.
195///
196/// @tparam T Type of the success value.
197template <typename T>
198// NOLINTNEXTLINE(cppcoreguidelines-special-member-functions)
199class Completion {
200public:
201 /// @brief Constructs an empty (no-op) completion.
202 Completion() = default;
203
204 /// @brief Constructs a completion backed by @p statePtr, delivering callbacks via @p execPtr.
205 /// @param statePtr Shared state produced by the backend.
206 /// @param execPtr Executor on which callbacks are posted. If `nullptr`, callbacks are
207 /// never delivered — they are silently dropped (see `setValue`/`setException`,
208 /// which post only when `cbExec != nullptr`).
209 Completion(std::shared_ptr<detail::CompletionState<T>> statePtr, ::morph::exec::IExecutor* execPtr)
210 : _state{std::move(statePtr)} {
211 if (_state != nullptr) {
212 _state->cbExec = execPtr;
213 }
214 }
215
216 /// @brief Move constructor — transfers ownership of the underlying state.
217 Completion(Completion&&) noexcept = default;
218 /// @brief Move assignment — transfers ownership of the underlying state.
219 /// @return `*this`.
220 Completion& operator=(Completion&&) noexcept = default;
221 Completion(const Completion&) = delete;
222 Completion& operator=(const Completion&) = delete;
223
224 /// @brief Registers a success callback.
225 ///
226 /// @p handler is posted to the executor with the result value when the
227 /// operation completes successfully. If the operation has already completed,
228 /// the callback is posted immediately.
229 ///
230 /// @param handler Callable receiving the result by value.
231 /// @return `*this` for chaining.
232 Completion& then(std::function<void(T)> handler) {
233 if (_state != nullptr) {
234 _state->attachThen(std::move(handler));
235 }
236 return *this;
237 }
238
239 /// @brief Registers an error callback.
240 ///
241 /// @p handler is posted to the executor with the `std::exception_ptr` when
242 /// the operation fails. If the operation has already failed, the callback
243 /// is posted immediately. Attaching this handler suppresses orphan logging.
244 ///
245 /// @param handler Callable receiving the exception pointer.
246 /// @return `*this` for chaining.
247 Completion& onError(std::function<void(std::exception_ptr)> handler) {
248 if (_state != nullptr) {
249 _state->attachOnError(std::move(handler));
250 }
251 return *this;
252 }
253
254 /// @brief Registers a success callback gated on @p scope's lifetime and stop state.
255 ///
256 /// Identical to `then(handler)` except that @p handler is wrapped in a
257 /// `CallbackToken` gate at attach time: when the result is delivered, the
258 /// handler runs only if @p scope is still alive and has not been stopped or
259 /// `reset()`. Nothing else changes — the gate lives inside the stored
260 /// handler, so fan-out, the attach-after-ready fire-now path and executor
261 /// marshalling all behave exactly as they do for the ungated form.
262 ///
263 /// The scope is observed weakly; attaching does **not** extend its lifetime.
264 ///
265 /// @param scope Receiver-owned gate. Only a token for its *current*
266 /// generation is captured, so a later `reset()` retires this
267 /// attachment.
268 /// @param handler Callable receiving the result by value.
269 /// @return `*this` for chaining.
270 Completion& then(const CallbackScope& scope, std::function<void(T)> handler) {
271 return then(scope.token(), std::move(handler));
272 }
273
274 /// @brief Registers a success callback gated on an already-issued @p token.
275 ///
276 /// The token-taking form of `then(const CallbackScope&, handler)`, for
277 /// callers that hold a token rather than the scope itself (a helper that
278 /// was handed one, or code that captured a token before dispatching).
279 ///
280 /// @param token Gate observing some receiver's `CallbackScope`. A
281 /// default-constructed token suppresses unconditionally.
282 /// @param handler Callable receiving the result by value.
283 /// @return `*this` for chaining.
284 Completion& then(CallbackToken token, std::function<void(T)> handler) {
285 return then(std::function<void(T)>{token.guard(std::move(handler))});
286 }
287
288 /// @brief Registers an error callback gated on @p scope's lifetime and stop state.
289 ///
290 /// Identical to `onError(handler)` except that @p handler is wrapped in a
291 /// `CallbackToken` gate at attach time.
292 ///
293 /// Orphan logging is unaffected: attaching a gated handler suppresses it
294 /// exactly as the ungated form does, and an error whose delivery the scope
295 /// then refuses still counts as **handled**. Suppression is a deliberate act
296 /// by the receiver, not a dropped error nobody asked about.
297 ///
298 /// @param scope Receiver-owned gate; observed weakly.
299 /// @param handler Callable receiving the exception pointer.
300 /// @return `*this` for chaining.
301 Completion& onError(const CallbackScope& scope, std::function<void(std::exception_ptr)> handler) {
302 return onError(scope.token(), std::move(handler));
303 }
304
305 /// @brief Registers an error callback gated on an already-issued @p token.
306 ///
307 /// The token-taking form of `onError(const CallbackScope&, handler)`.
308 ///
309 /// @param token Gate observing some receiver's `CallbackScope`. A
310 /// default-constructed token suppresses unconditionally.
311 /// @param handler Callable receiving the exception pointer.
312 /// @return `*this` for chaining.
313 Completion& onError(CallbackToken token, std::function<void(std::exception_ptr)> handler) {
314 return onError(std::function<void(std::exception_ptr)>{token.guard(std::move(handler))});
315 }
316
317 /// @brief Registers a success callback whose lifetime is deliberately unmanaged.
318 ///
319 /// Exactly `then(handler)`, spelled so that "no scope gates this" is a
320 /// statement the author made on purpose and a reviewer can grep for. Use it
321 /// when the handler genuinely owns everything it touches — it captures only
322 /// values, or a `shared_ptr` it keeps alive itself.
323 ///
324 /// @param handler Callable receiving the result by value.
325 /// @return `*this` for chaining.
326 Completion& thenDetached(std::function<void(T)> handler) { return then(std::move(handler)); }
327
328 /// @brief Registers an error callback whose lifetime is deliberately unmanaged.
329 ///
330 /// The `onError` counterpart of `thenDetached()`.
331 ///
332 /// @param handler Callable receiving the exception pointer.
333 /// @return `*this` for chaining.
334 Completion& onErrorDetached(std::function<void(std::exception_ptr)> handler) {
335 return onError(std::move(handler));
336 }

338 /// @brief Returns the underlying shared state (for advanced / internal use).
339 /// @return Shared pointer to the completion state, or `nullptr` for empty completions.
340 [[nodiscard]] std::shared_ptr<detail::CompletionState<T>> state() const { return _state; }

342 /// @brief Producer-side handle paired with a `Completion<T>` by `makeSettleable()`.
343 ///
344 /// `Promise<T>` is the public settling counterpart to `Completion<T>`: it
345 /// exposes exactly `resolve()`/`reject()` against the same shared state a
346 /// paired `Completion<T>` observes via `then()`/`onError()`, without ever
347 /// naming `morph::async::detail::CompletionState<T>`. Intended for test code
348 /// that needs to construct a `Completion<T>` it can settle on demand — e.g.
349 /// standing in for a `Bridge`/`IBackend` round trip — instead of reaching
350 /// into `detail::CompletionState<T>` directly (see docs/spec/core/completion.md,
351 /// "Settleable promise seam").
352 ///
353 /// Move-only, mirroring `Completion<T>`: exactly one producer settles a
354 /// given operation.
355 class Promise {
356 public:
357 /// @brief Move constructor — transfers ownership of the shared state.
358 Promise(Promise&&) noexcept = default;
359 /// @brief Move assignment — transfers ownership of the shared state.
360 /// @return `*this`.
361 Promise& operator=(Promise&&) noexcept = default;
362 Promise(const Promise&) = delete;
363 Promise& operator=(const Promise&) = delete;
364 ~Promise() = default;
365
366 /// @brief Resolves the paired `Completion<T>` with @p val.
367 ///
368 /// No-op if the state is already settled (first result wins — see
369 /// `detail::CompletionState<T>::setValue`), or if this `Promise` was
370 /// moved from (mirrors `Completion<T>::then()`'s null-state no-op).
371 /// Safe to call from any thread.
372 /// @param val Success value delivered to every attached `then()` handler.
373 void resolve(T val) {
374 if (_state != nullptr) {
375 _state->setValue(std::move(val));
376 }
377 }
378
379 /// @brief Rejects the paired `Completion<T>` with @p exc.
380 ///
381 /// No-op if the state is already settled (first result wins — see
382 /// `detail::CompletionState<T>::setException`), or if this `Promise` was
383 /// moved from (mirrors `Completion<T>::onError()`'s null-state no-op).
384 /// Safe to call from any thread.
385 /// @param exc Error delivered to every attached `onError()` handler, or
386 /// logged as an orphan if none is ever attached.
387 void reject(std::exception_ptr exc) {
388 if (_state != nullptr) {
389 _state->setException(exc);
390 }
391 }
392
393 private:
394 friend class Completion<T>;
395 explicit Promise(std::shared_ptr<detail::CompletionState<T>> state) : _state{std::move(state)} {}
396
397 std::shared_ptr<detail::CompletionState<T>> _state;
398 };
399
400 /// @brief Constructs a `Completion<T>`/`Promise<T>` pair sharing one settleable state.
401 ///
402 /// The public "settleable promise" seam (issue #55): lets a caller — typically
403 /// test code — construct a `Completion<T>` it can resolve or reject on demand,
404 /// without a full `Bridge`/`IBackend` round trip and without reaching into
405 /// `morph::async::detail::CompletionState<T>`. Everything `Completion(state,
406 /// executor)` already provided by hand is available through this factory
407 /// instead: the returned `Completion<T>` is exactly what `then()`/`onError()`
408 /// observe; the returned `Promise` is exactly what settles it.
409 /// @param execPtr Executor callbacks are posted on; `nullptr` for a
410 /// write-only completion (see the two-argument constructor).
411 /// @return A `{Completion<T>, Promise}` pair sharing one `CompletionState<T>`.
412 [[nodiscard]] static std::pair<Completion<T>, Promise> makeSettleable(::morph::exec::IExecutor* execPtr) {
413 auto state = std::make_shared<detail::CompletionState<T>>();
414 Completion<T> completion{state, execPtr};
415 Promise promise{state};
416 return {std::move(completion), std::move(promise)};
417 }
418
419private:
420 std::shared_ptr<detail::CompletionState<T>> _state;
421};
