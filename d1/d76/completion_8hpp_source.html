25//
26// `enable_shared_from_this` is load-bearing, not decoration: the dispatch
27// closures below capture a `shared_ptr` to this state and read the settled
28// value *in place* instead of carrying a copy of it. That is what makes the
29// copy budget independent of how many handlers are attached, and what lets `T`
30// be move-only -- the closure stores a refcounted handle, so it stays
31// copy-constructible and `IExecutor::post`'s `std::function<void()>` is
32// untouched. Every `CompletionState<T>` in the tree is created by
33// `std::make_shared` (`Completion<T>::makeSettleable`, `Bridge`, the backends),
34// which is the precondition `shared_from_this()` needs.
35//
36// Reading `value` outside `mtx` from those closures is safe because `value` is
37// write-once: both `setValue` and `setException` return early when `ready`, and
38// nothing else ever assigns it. The store happens under the lock before the
39// closure is handed to the executor, and the executor's own queue provides the
40// happens-before edge to the thread that runs it.
41template <typename T>
42struct CompletionState : std::enable_shared_from_this<CompletionState<T>> {
43 static_assert(std::move_constructible<T>,
44 "morph::async::Completion<T> requires only that T be move-constructible. Copyability is a "
45 "per-handler obligation: a handler taking `const T&` imposes nothing, a handler taking `T` by "
46 "value requires T to be copyable and is diagnosed where that handler is written.");
47
48 std::mutex mtx;
49 std::optional<T> value;
50 std::exception_ptr error;
51 bool ready = false;
52 // Every handler attached while the state is not yet ready is kept (not
53 // overwritten): a second/third attach composes with earlier ones instead of
54 // silently discarding them. Dispatch invokes all of them, in attachment
55 // order, from a single posted closure. See docs/spec/core/completion.md,
56 // "Failure modes" / fan-out.
57 //
58 // Erased as `void(const T&)`, not `void(T)`. One erased type accepts every
59 // handler spelling a caller already writes -- `[](const T&)`, `[](T)`, a
60 // `std::function<void(T)>` object -- because each is invocable with
61 // `const T&`. A handler that wants its own value gets exactly one copy, at
62 // its own parameter binding, where the reader of that call site can see it;
63 // a handler that only observes pays nothing. Erasing as `void(T)` charged
64 // every handler a copy whether or not it wanted one (morph#553).
65 std::vector<std::function<void(const T&)>> onOk;
66 std::vector<std::function<void(std::exception_ptr)>> onErr;
67 bool onErrAttached = false;
68 ::morph::exec::IExecutor* cbExec = nullptr;
69
70 void setValue(T val) {
71 std::function<void()> callback;
72 {
73 std::scoped_lock const lock{mtx};
74 if (ready) {
75 return;
76 }
77 // Store first, drain `onOk` last. `value` is this state's own
78 // store and is never moved out of: a `then()` attached *after*
79 // this point (attachThen's `ready && value` branch) reads it
80 // again, and moving out of it left it engaged but moved-from so a
81 // later attacher silently observed a husk (morph#520). The value
82 // is observed, never consumed -- structurally, now that no
83 // dispatch path can take it.
84 //
85 // The ordering is what gives this block the strong exception
86 // guarantee for a `T` whose *move* constructor can throw: nothing
87 // has changed when the store below runs, and `optional::emplace`
88 // leaves the optional disengaged if the construction throws, so an
89 // escape leaves `onOk` holding every handler and the state unready
90 // rather than a state that already looks settled with its handlers
91 // already lost. Draining first (as an earlier revision did) meant
92 // a throwing store unwound with `savedFns` -- a local -- carrying
93 // every handler to its destructor: permanently unsettled, no
94 // handlers, and silent, because the destructor's orphan logger
95 // only fires when `error` is set. `std::move` on a vector is
96 // noexcept, so the ordering costs nothing.
97 //
98 // `emplace`, not `value = std::move(val)`: assigning through
99 // `std::optional` requires `T` to be move-*assignable* as well as
100 // move-constructible, which would quietly make the `static_assert`
101 // above a lie for a `T` with a deleted assignment operator.
102 // `value` is guaranteed disengaged here -- it is written only on
103 // this line, and the `ready` guard above makes this line run once.
104 value.emplace(std::move(val));
105 ready = true;
106 if (!onOk.empty()) {
107 auto savedFns = std::move(onOk);
108 callback = [self = this->shared_from_this(), savedFns = std::move(savedFns)]() {
109 // Every handler reads the one stored value in place; none
110 // can move out of it, so no handler can leave a husk for
111 // its siblings or for a later attacher, and the count of
112 // handlers costs nothing in copies. Each handler is
113 // isolated in its own try/catch so one throwing handler
114 // cannot prevent its siblings from running -- fan-out means
115 // every attached handler gets its turn, independent of
116 // whether an earlier one misbehaves. An escaping exception
117 // here would otherwise unwind the whole posted closure and
118 // silently skip every handler after the one that threw.
119 for (const auto& fn : savedFns) {
120 try {
121 fn(*self->value);
122 } catch (...) {
123 ::morph::log::logError("[completion] then handler threw; continuing with next handler");
124 }
125 }
126 };
127 }
128 }
129 if (callback != nullptr && cbExec != nullptr) {
130 cbExec->post(std::move(callback));
131 }
132 }
133 void setException(const std::exception_ptr& exc) {
134 std::function<void()> callback;
135 {
136 std::scoped_lock const lock{mtx};
137 if (ready) {
138 return;
139 }
140 // A null `exc` must never reach `error`. Storing one would set
141 // `ready` with `error` still falsy — a state no attach can act on,
142 // because `attachOnError` tests `ready && error` and `attachThen`
143 // tests `ready && value`, so both branches fall through and the
144 // handler is neither fired nor queued. The completion would then
145 // be dead in both directions for every later caller, and silently:
146 // `attachOnError` sets `onErrAttached` on entry, so even the
147 // destructor's orphan logger is suppressed. It would also hand any
148 // *already*-attached handler a null `exception_ptr`, which is UB to
149 // `std::rethrow_exception` — the idiomatic handler body, this
150 // file's own orphan logger included.
151 //
152 // Substituting here rather than at any one producer is deliberate:
153 // `reject(nullptr)` is reachable through the public `Promise<T>`
154 // seam, and around ten sites in the tree forward an
155 // `exception_ptr` straight through (`.onError([state](auto e) {
156 // state->setException(e); })`) without inspecting it, so a guard
157 // at one producer would leave every other one able to reintroduce
158 // the same wedge. See issue #347.
159 error = exc ? exc
160 : std::make_exception_ptr(
161 std::runtime_error{"completion rejected with no exception (null exception_ptr)"});
162 ready = true;
163 if (!onErr.empty()) {
164 auto savedFns = std::move(onErr);
165 auto savedErr = error;
166 callback = [savedFns = std::move(savedFns), savedErr]() mutable {
167 // Isolate each handler so one throwing onError handler
168 // cannot suppress its siblings -- see the matching comment
169 // in setValue's callback above.
170 for (auto& fn : savedFns) {
171 try {
172 fn(savedErr);
173 } catch (...) {
174 ::morph::log::logError("[completion] onError handler threw; continuing with next handler");
175 }
176 }
177 };
178 // Only mark the error handled (suppressing the destructor's orphan
179 // log) if we actually have an executor to deliver on. With a null
180 // executor the callback below is never posted, so the error must
181 // still reach the orphan logger rather than vanish silently.
182 onErrAttached = (cbExec != nullptr);
183 }
184 }
185 if (callback != nullptr && cbExec != nullptr) {
186 cbExec->post(std::move(callback));
187 }
188 }
189 void attachThen(std::function<void(const T&)> handler) {
190 std::function<void()> fireNow;
191 {
192 std::scoped_lock const lock{mtx};
193 if (ready && value) {
194 // Keep the state alive and read `value` in place rather than
195 // snapshotting it. The old shape copied `*value` into
196 // `savedVal` and then captured `savedVal` *by copy* before
197 // moving it into the handler -- two copies where the handler
198 // asked for at most one, and the reason a late attacher cost
199 // 2 copies rather than 1 (morph#553).
200 fireNow = [self = this->shared_from_this(), handler = std::move(handler)]() { handler(*self->value); };
201 } else if (!ready) {
202 onOk.push_back(std::move(handler));
203 }
204 }
205 if (fireNow != nullptr && cbExec != nullptr) {
206 cbExec->post(std::move(fireNow));
207 }
208 }
209 void attachOnError(std::function<void(std::exception_ptr)> handler) {
210 std::function<void()> fireNow;
211 {
212 std::scoped_lock const lock{mtx};
213 // See setException: only suppress orphan logging when an executor
214 // exists to actually deliver the handler; a null executor otherwise
215 // drops the error and silences the orphan logger both at once.
216 onErrAttached = (cbExec != nullptr);
217 if (ready && error) {
218 auto savedErr = error;
219 fireNow = [handler = std::move(handler), savedErr]() mutable { handler(savedErr); };
220 } else if (!ready) {
221 onErr.push_back(std::move(handler));
222 }
223 }
224 if (fireNow != nullptr && cbExec != nullptr) {
225 cbExec->post(std::move(fireNow));
226 }
227 }
228 ~CompletionState() {
229 if (!ready || !error || onErrAttached) {
230 return;
231 }
232 // No local guard around the logging calls: `morph::log`'s helpers are
233 // `noexcept` (docs/spec/core/logger.md, "Failure modes"), which is
234 // what makes them safe to call from a destructor at all. The variadic
235 // overload is deliberate — it formats *inside* that guarantee, where
236 // building the message by concatenation out here would allocate
237 // outside it and could still escape.
238 try {
239 std::rethrow_exception(error);
240 } catch (const std::exception& exc) {
241 ::morph::log::logError("[orphan] unhandled exception: {}", exc.what());
242 } catch (...) {
243 ::morph::log::logError("[orphan] unhandled unknown exception");
244 }
245 }
246};

250/// @brief A place a backend settles one dispatch, without naming its type.
251///
252/// The narrow half of `CompletionState<std::shared_ptr<void>>`: the two calls a
253/// backend actually makes on the completion it produced. Splitting them out is
254/// what lets `Bridge` hand the backend a sink that *is* the caller's typed
255/// completion state, instead of a second, erased completion whose only job is
256/// to be forwarded into the first.
257///
258/// That forwarding cost six heap allocations per dispatch — the erased state,
259/// the `.then` and `.onError` closures, their two handler vectors, and one of
260/// the two posted settle tasks — of the 14.06 a local round trip took. Through
261/// a sink it is one (morph#572, Part B).
262///
263/// @par Contract for implementers
264/// - **Settle once.** `settleValue` and `settleException` are mutually
265/// exclusive and each may be called at most once *in effect*; an
266/// implementation must tolerate extra calls and ignore them, because
267/// `IBackend::cancelPending` settles a sink that a reply may be racing to
268/// settle at the same moment. `CompletionState` is already first-result-wins,
269/// but anything an implementation does *besides* forwarding — decrementing a
270/// counter, cancelling a timer — needs its own latch.
271/// - **Do not let an exception escape** where one can be avoided. A sink is
272/// settled from a strand task or a transport thread whose only handler is the
273/// one the backend wrote around the call. The methods are deliberately *not*
274/// `noexcept`: `CompletionState::setValue` is not either (it builds a
275/// `std::function` for the posted callback), and promising more here would
276/// turn a `bad_alloc` into a `std::terminate` that today it is not.
277class ISettleSink {
278public:
279 ISettleSink() = default;
280 ISettleSink(const ISettleSink&) = delete;
281 ISettleSink(ISettleSink&&) = delete;
282 ISettleSink& operator=(const ISettleSink&) = delete;
283 ISettleSink& operator=(ISettleSink&&) = delete;
284 virtual ~ISettleSink() = default;
285
286 /// @brief Settles the dispatch with its opaque result.
287 /// @param value The result, as the backend produced it.
288 virtual void settleValue(std::shared_ptr<void> value) = 0;
289
290 /// @brief Settles the dispatch with a failure.
291 /// @param exc The exception to deliver.
292 virtual void settleException(const std::exception_ptr& exc) = 0;
293};
294
295/// @brief The `ISettleSink` that simply is a `CompletionState<std::shared_ptr<void>>`.
296///
297/// The adapter for every caller that wants the old shape: `IBackend::execute`
298/// returns a `Completion<std::shared_ptr<void>>`, so a backend implementing
299/// `executeInto` can serve `execute` by wrapping its own state in one of these.
300/// Nothing but forwarding happens here, so it needs no settle-once latch of its
301/// own — `CompletionState` already has one.
302class CompletionSettleSink final : public ISettleSink {
303public:
304 /// @brief Binds the sink to @p state.
305 /// @param state The completion state to settle. Must not be null.
306 explicit CompletionSettleSink(std::shared_ptr<CompletionState<std::shared_ptr<void>>> state)
307 : _state{std::move(state)} {}
308
309 /// @brief Forwards to `CompletionState::setValue`.
310 /// @param value The result to store.
311 void settleValue(std::shared_ptr<void> value) override { _state->setValue(std::move(value)); }
312
313 /// @brief Forwards to `CompletionState::setException`.
314 /// @param exc The exception to store.
315 void settleException(const std::exception_ptr& exc) override { _state->setException(exc); }
316
317private:
318 std::shared_ptr<CompletionState<std::shared_ptr<void>>> _state;
319};

323/// @brief Move-only handle representing the eventual result of an asynchronous operation.
324///
325/// Callbacks are posted to the `IExecutor` supplied at construction time, so
326/// they always run on the intended thread (e.g. the GUI thread).
327///
328/// @par Thread safety
329/// `then()` and `onError()` may be called from any thread. The registered
330/// callbacks are invoked via the executor, never directly from the producing thread.
331///
332/// @par Orphan detection
333/// If a `Completion` is destroyed before an `onError()` handler is attached and
334/// the operation has already failed, the exception is logged as an orphan error.
335///
336/// @par Lifetime and stop gating
337/// Because delivery always goes through an executor, the receiver can be
338/// destroyed — or simply lose interest — before the handler runs. Pass a
339/// `morph::async::CallbackScope` (or one of its `CallbackToken`s) as the first
340/// argument to `then()` / `onError()` and the handler is refused unless the
341/// scope is still alive and un-stopped at delivery time. `thenDetached()` /
342/// `onErrorDetached()` are the same ungated attachments as `then(fn)` /
343/// `onError(fn)`, spelled so a deliberately unmanaged callback says so.
344/// See `callback_scope.hpp` and docs/spec/core/callback_scope.md.
345///
346/// @par Value-handling contract
347/// `T` need only be **move-constructible**; that is the whole type requirement.
348/// Copyability is a *per-handler* obligation, reported where the handler is
349/// written rather than imposed on the whole instantiation:
350/// - a handler taking `const T&` costs **zero** copies;
351/// - a handler taking `T` by value costs **exactly one**, at its own parameter
352/// binding, and requires `T` to be copyable.
353///
354/// The budget does not depend on how many handlers are attached, nor on whether
355/// they attached before or after the completion settled. The value is
356/// **observed, never consumed**: no handler can move out of the stored value, so
357/// a `then()` attached after settling still sees the genuine result. A handler
358/// that wants to consume takes `T` by value and moves out of its own copy.
359/// See docs/spec/core/completion.md, "Value-handling contract".
360///
361/// @tparam T Type of the success value. Must be move-constructible.
362template <typename T>
363// NOLINTNEXTLINE(cppcoreguidelines-special-member-functions)
364class Completion {
365public:
366 /// @brief Constructs an empty (no-op) completion.
367 Completion() = default;
368
369 /// @brief Constructs a completion backed by @p statePtr, delivering callbacks via @p execPtr.
370 /// @param statePtr Shared state produced by the backend.
371 /// @param execPtr Executor on which callbacks are posted. If `nullptr`, callbacks are
372 /// never delivered — they are silently dropped (see `setValue`/`setException`,
373 /// which post only when `cbExec != nullptr`).
374 Completion(std::shared_ptr<detail::CompletionState<T>> statePtr, ::morph::exec::IExecutor* execPtr)
375 : _state{std::move(statePtr)} {
376 if (_state != nullptr) {
377 _state->cbExec = execPtr;
378 }
379 }
380
381 /// @brief Move constructor — transfers ownership of the underlying state.
382 Completion(Completion&&) noexcept = default;
383 /// @brief Move assignment — transfers ownership of the underlying state.
384 /// @return `*this`.
385 Completion& operator=(Completion&&) noexcept = default;
386 Completion(const Completion&) = delete;
387 Completion& operator=(const Completion&) = delete;
388
389 /// @brief Registers a success callback.
390 ///
391 /// @p handler is posted to the executor with the result value when the
392 /// operation completes successfully. If the operation has already completed,
393 /// the callback is posted immediately.
394 ///
395 /// @param handler Callable receiving the result. Take `const T&` to observe it for free;
396 /// take `T` by value to get your own copy, at the cost of exactly one copy.
397 /// @return `*this` for chaining — a reference into this `Completion`, valid
398 /// only for as long as it is.
399 Completion& then(std::function<void(const T&)> handler) MORPH_LIFETIMEBOUND {
400 if (_state != nullptr) {
401 _state->attachThen(std::move(handler));
402 }
403 return *this;
404 }
405
406 /// @brief Registers an error callback.
407 ///
408 /// @p handler is posted to the executor with the `std::exception_ptr` when
409 /// the operation fails. If the operation has already failed, the callback
410 /// is posted immediately. Attaching this handler suppresses orphan logging.
411 ///
412 /// @param handler Callable receiving the exception pointer.
413 /// @return `*this` for chaining — a reference into this `Completion`, valid
414 /// only for as long as it is.
415 Completion& onError(std::function<void(std::exception_ptr)> handler) MORPH_LIFETIMEBOUND {
416 if (_state != nullptr) {
417 _state->attachOnError(std::move(handler));
418 }
419 return *this;
420 }
421
422 /// @brief Registers a success callback gated on @p scope's lifetime and stop state.
423 ///
424 /// Identical to `then(handler)` except that @p handler is wrapped in a
425 /// `CallbackToken` gate at attach time: when the result is delivered, the
426 /// handler runs only if @p scope is still alive and has not been stopped or
427 /// `reset()`. Nothing else changes — the gate lives inside the stored
428 /// handler, so fan-out, the attach-after-ready fire-now path and executor
429 /// marshalling all behave exactly as they do for the ungated form.
430 ///
431 /// The scope is observed weakly; attaching does **not** extend its lifetime.
432 ///
433 /// @param scope Receiver-owned gate. Only a token for its *current*
434 /// generation is captured, so a later `reset()` retires this
435 /// attachment.
436 /// @param handler Callable receiving the result. Take `const T&` to observe it for free;
437 /// take `T` by value to get your own copy, at the cost of exactly one copy.
438 /// @return `*this` for chaining — a reference into this `Completion`, valid
439 /// only for as long as it is.
440 Completion& then(const CallbackScope& scope, std::function<void(const T&)> handler) MORPH_LIFETIMEBOUND {
441 return then(scope.token(), std::move(handler));
442 }
443
444 /// @brief Registers a success callback gated on an already-issued @p token.
445 ///
446 /// The token-taking form of `then(const CallbackScope&, handler)`, for
447 /// callers that hold a token rather than the scope itself (a helper that
448 /// was handed one, or code that captured a token before dispatching).
449 ///
450 /// @param token Gate observing some receiver's `CallbackScope`. A
451 /// default-constructed token suppresses unconditionally.
452 /// @param handler Callable receiving the result. Take `const T&` to observe it for free;
453 /// take `T` by value to get your own copy, at the cost of exactly one copy.
454 /// @return `*this` for chaining — a reference into this `Completion`, valid
455 /// only for as long as it is.
456 Completion& then(CallbackToken token, std::function<void(const T&)> handler) MORPH_LIFETIMEBOUND {
457 return then(std::function<void(const T&)>{token.guard(std::move(handler))});
458 }
459
460 /// @brief Registers an error callback gated on @p scope's lifetime and stop state.
461 ///
462 /// Identical to `onError(handler)` except that @p handler is wrapped in a
463 /// `CallbackToken` gate at attach time.
464 ///
465 /// Orphan logging is unaffected: attaching a gated handler suppresses it
466 /// exactly as the ungated form does, and an error whose delivery the scope
467 /// then refuses still counts as **handled**. Suppression is a deliberate act
468 /// by the receiver, not a dropped error nobody asked about.
469 ///
470 /// @param scope Receiver-owned gate; observed weakly.
471 /// @param handler Callable receiving the exception pointer.
472 /// @return `*this` for chaining — a reference into this `Completion`, valid
473 /// only for as long as it is.
474 Completion& onError(const CallbackScope& scope,
475 std::function<void(std::exception_ptr)> handler) MORPH_LIFETIMEBOUND {
476 return onError(scope.token(), std::move(handler));
477 }
478
479 /// @brief Registers an error callback gated on an already-issued @p token.
480 ///
481 /// The token-taking form of `onError(const CallbackScope&, handler)`.
482 ///
483 /// @param token Gate observing some receiver's `CallbackScope`. A
484 /// default-constructed token suppresses unconditionally.
485 /// @param handler Callable receiving the exception pointer.
486 /// @return `*this` for chaining — a reference into this `Completion`, valid
487 /// only for as long as it is.
488 Completion& onError(CallbackToken token, std::function<void(std::exception_ptr)> handler) MORPH_LIFETIMEBOUND {
489 return onError(std::function<void(std::exception_ptr)>{token.guard(std::move(handler))});
490 }

492 /// @brief Registers a success callback whose lifetime is deliberately unmanaged.
493 ///
494 /// Exactly `then(handler)`, spelled so that "no scope gates this" is a
495 /// statement the author made on purpose and a reviewer can grep for. Use it
496 /// when the handler genuinely owns everything it touches — it captures only
497 /// values, or a `shared_ptr` it keeps alive itself.
498 ///
499 /// @param handler Callable receiving the result. Take `const T&` to observe it for free;
500 /// take `T` by value to get your own copy, at the cost of exactly one copy.
501 /// @return `*this` for chaining — a reference into this `Completion`, valid
502 /// only for as long as it is.
503 Completion& thenDetached(std::function<void(const T&)> handler) MORPH_LIFETIMEBOUND {
504 return then(std::move(handler));
505 }

507 /// @brief Registers an error callback whose lifetime is deliberately unmanaged.
508 ///
509 /// The `onError` counterpart of `thenDetached()`.
510 ///
511 /// @param handler Callable receiving the exception pointer.
512 /// @return `*this` for chaining — a reference into this `Completion`, valid
513 /// only for as long as it is.
514 Completion& onErrorDetached(std::function<void(std::exception_ptr)> handler) MORPH_LIFETIMEBOUND {
515 return onError(std::move(handler));
516 }

520 [[nodiscard]] std::shared_ptr<detail::CompletionState<T>> state() const { return _state; }

522 /// @brief Producer-side handle paired with a `Completion<T>` by `makeSettleable()`.
523 ///
524 /// `Promise<T>` is the public settling counterpart to `Completion<T>`: it
525 /// exposes exactly `resolve()`/`reject()` against the same shared state a
526 /// paired `Completion<T>` observes via `then()`/`onError()`, without ever
527 /// naming `morph::async::detail::CompletionState<T>`. Intended for test code
528 /// that needs to construct a `Completion<T>` it can settle on demand — e.g.
529 /// standing in for a `Bridge`/`IBackend` round trip — instead of reaching
530 /// into `detail::CompletionState<T>` directly (see docs/spec/core/completion.md,
531 /// "Settleable promise seam").
532 ///
533 /// Move-only, mirroring `Completion<T>`: exactly one producer settles a
534 /// given operation.
535 class Promise {
536 public:
537 /// @brief Move constructor — transfers ownership of the shared state.
538 Promise(Promise&&) noexcept = default;
539 /// @brief Move assignment — transfers ownership of the shared state.
540 /// @return `*this`.
541 Promise& operator=(Promise&&) noexcept = default;
542 Promise(const Promise&) = delete;
543 Promise& operator=(const Promise&) = delete;
544 ~Promise() = default;
545
546 /// @brief Resolves the paired `Completion<T>` with @p val.
547 ///
548 /// No-op if the state is already settled (first result wins — see
549 /// `detail::CompletionState<T>::setValue`), or if this `Promise` was
550 /// moved from (mirrors `Completion<T>::then()`'s null-state no-op).
551 /// Safe to call from any thread.
552 /// @param val Success value delivered to every attached `then()` handler.
553 void resolve(T val) {
554 if (_state != nullptr) {
555 _state->setValue(std::move(val));
556 }
557 }
558
559 /// @brief Rejects the paired `Completion<T>` with @p exc.
560 ///
561 /// No-op if the state is already settled (first result wins — see
562 /// `detail::CompletionState<T>::setException`), or if this `Promise` was
563 /// moved from (mirrors `Completion<T>::onError()`'s null-state no-op).
564 /// Safe to call from any thread.
565 /// @param exc Error delivered to every attached `onError()` handler, or

580 /// @brief Constructs a `Completion<T>`/`Promise<T>` pair sharing one settleable state.
581 ///
582 /// The public "settleable promise" seam (issue #55): lets a caller — typically
583 /// test code — construct a `Completion<T>` it can resolve or reject on demand,
584 /// without a full `Bridge`/`IBackend` round trip and without reaching into
585 /// `morph::async::detail::CompletionState<T>`. Everything `Completion(state,

588 /// observe; the returned `Promise` is exactly what settles it.

591 /// @return A `{Completion<T>, Promise}` pair sharing one `CompletionState<T>`.

593 auto state = std::make_shared<detail::CompletionState<T>>();
594 Completion<T> completion{state, execPtr};
595 Promise promise{state};
596 return {std::move(completion), std::move(promise)};
597 }
