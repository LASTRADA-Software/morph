18/// @brief Outcome of a single `onOnline()` attempt sequence.
19enum class ReconnectOutcome : std::uint8_t {
20 Reconnected, ///< Backend reopened, made active, context bound, queue replay invoked.
21 GaveUp, ///< Exhausted maxAttempts without a successful reconnect; stayed offline.
22 Aborted, ///< shouldContinue() returned false before any reconnect (e.g. went offline again).
23};
24
25namespace detail {
26/// @brief Tag value for the `reconnectOutcome` metric's "outcome" dimension.
27constexpr std::string_view reconnectOutcomeName(ReconnectOutcome outcome) noexcept {
28 switch (outcome) {
29 case ReconnectOutcome::Reconnected:
30 return "Reconnected";
31 case ReconnectOutcome::GaveUp:
32 return "GaveUp";
33 case ReconnectOutcome::Aborted:
34 return "Aborted";
35 default:
36 // Unreachable through any real code path: ReconnectOutcome is a
37 // closed enum and every enumerator is already handled explicitly
38 // above. This arm only exists to satisfy the compiler that the
39 // function returns on every enum value, including one
40 // manufactured by an out-of-range `static_cast` -- mirrors
41 // ruleKindName's identical default: arm in forms.hpp.
42 return "?";
43 }
44}
45} // namespace detail

47/// @brief Tuning parameters for `ReconnectCoordinator`.
48///
49/// Declared outside the class so its default member initialisers are fully
50/// parsed before any constructor default argument names `Config{}`. (A nested
51/// incomplete type breaks constructor default arguments on clang/GCC — see the
52/// same note on `NetworkMonitorConfig`.)
53struct ReconnectCoordinatorConfig {
54 /// @brief Max reconnect attempts per `onOnline()` call before giving up.
55 int maxAttempts = 10;
56
57 /// @brief Delay between failed reconnect attempts.
58 std::chrono::milliseconds retryDelay = std::chrono::seconds{2};
59};

61/// @brief Sequences reconnect → activate → bind → replay when connectivity returns.
62///
63/// All side effects are injected via `Deps`. The coordinator contains only the
64/// retry loop, the ordering guarantees, and the abort checks. It performs no I/O
65/// and owns no thread; `onOnline()` / `onOffline()` run synchronously on the
66/// calling thread.
67///
68/// @par Ordering guarantee (the reason this class exists)
69/// Within a successful `onOnline()`, the steps run in this strict order:
70/// 1. `tryReconnect()` returns true (backend usable)
71/// 2. `activatePrimary()` (make primary the active backend)
72/// 3. `bindContext()` (rebind per-connection / per-session state)
73/// 4. `replay()` (drain + replay the offline queue)
74/// Step 4 MUST NOT run before step 3 completes, and step 3 MUST NOT run before
75/// step 2. Implementations and tests should treat this as an invariant.
76///
77/// @par Thread safety
78/// `onOnline()` and `onOffline()` are mutually serialised by an internal mutex,
79/// mirroring `SyncWorker::run()`. Calling them concurrently is safe; the second
80/// caller blocks. They are intended to be posted onto a worker executor by the
81/// host (see the wiring note in the design doc), not called on the probe thread.
82// NOLINTNEXTLINE(cppcoreguidelines-special-member-functions)
83class ReconnectCoordinator {
84public:
85 /// @brief Alias for the configuration struct.
86 using Config = ReconnectCoordinatorConfig;
87
88 /// @brief Injected, host-supplied side effects. None may be null.
89 struct Deps {
90 /// @brief Attempt to (re)open the primary backend.
91 /// @return true once the backend is genuinely usable (not merely TCP-open).
92 /// Must not throw; throwing is treated as a failed attempt.
93 std::function<bool()> tryReconnect;
94
95 /// @brief Make the freshly-reconnected primary the active backend.
96 /// Called exactly once per successful `onOnline()`, after `tryReconnect()`
97 /// succeeds and before `bindContext()`.
98 std::function<void()> activatePrimary;
99
100 /// @brief Switch the active backend to the local/offline one.
101 /// Called by `onOffline()`. Followed by `bindContext()` so context tracks
102 /// the now-active backend.
103 std::function<void()> activateLocal;
104
105 /// @brief Rebind per-connection / per-session context to the active backend.
106 /// Called after every activate* step. Must not throw.
107 std::function<void()> bindContext;
108
109 /// @brief Replay the offline queue against the now-active primary.
110 /// Typically wraps `SyncWorker::run()`. Called last in `onOnline()`.
111 std::function<void()> replay;
112
113 /// @brief Returns false to abort the current `onOnline()` sequence early
114 /// (e.g. the monitor reports the backend went offline again mid-retry).
115 /// Polled before each reconnect attempt and once more before replay.
116 std::function<bool()> shouldContinue;
117
118 /// @brief Sleep for the given duration between failed attempts.
119 /// Injected so tests can substitute a no-op / counter. Hosts wire this to
120 /// `std::this_thread::sleep_for`.
121 std::function<void(std::chrono::milliseconds)> sleep;
122 };
123
124 /// @brief Constructs a coordinator with injected dependencies and tuning.
125 /// @param deps Side-effect callbacks (all required, none null).
126 /// @param cfg Retry tuning.
127 ///
128 /// A null `Deps` member is logged via `morph::log::logError` in all builds;
129 /// construction still succeeds. Invoking the coordinator with any null member
130 /// is undefined behaviour.
131 explicit ReconnectCoordinator(Deps deps, Config cfg = Config{}) : _deps{std::move(deps)}, _cfg{cfg} {
132 assertDepsNonNull(_deps);
133 }
134
135 ReconnectCoordinator(const ReconnectCoordinator&) = delete;
136 ReconnectCoordinator& operator=(const ReconnectCoordinator&) = delete;
137 ReconnectCoordinator(ReconnectCoordinator&&) = delete;
138 ReconnectCoordinator& operator=(ReconnectCoordinator&&) = delete;
139
140 /// @brief Run the reconnect → activate → bind → replay sequence.
141 ///
142 /// Synchronous; runs entirely on the calling thread. Serialised against
143 /// `onOffline()` by an internal mutex.
144 ///
145 /// @return How the sequence ended (see `ReconnectOutcome`).
146 ReconnectOutcome onOnline() {
147 std::scoped_lock const lock{_mtx};
148
149 for (int attempt = 1; attempt <= _cfg.maxAttempts; ++attempt) {
150 if (!callShouldContinue()) {
151 return emitOutcome(ReconnectOutcome::Aborted);
152 }
153 ::morph::observe::detail::emitMetric(::morph::observe::Metric::reconnectAttempts, 1.0);
154 if (callTryReconnect()) {
155 _deps.activatePrimary();
156 _deps.bindContext();
157 // Re-check before replay so we never replay into a backend that
158 // just went away. We are still Reconnected either way; this only
159 // controls whether replay runs.
160 if (callShouldContinue()) {
161 _deps.replay();
162 }
163 return emitOutcome(ReconnectOutcome::Reconnected);
164 }
165 // No sleep after the final attempt — it would just waste retryDelay
166 // before giving up.
167 if (attempt < _cfg.maxAttempts) {
168 _deps.sleep(_cfg.retryDelay);
169 }
170 }
171
172 ::morph::log::logWarn("[reconnect_coordinator] gave up after " + std::to_string(_cfg.maxAttempts) +
173 " attempts, staying offline");
174 return emitOutcome(ReconnectOutcome::GaveUp);
175 }
176
177 /// @brief Switch to the local backend and rebind context.
178 ///
179 /// Idempotent at the policy level: safe to call when already local. Serialised
180 /// against `onOnline()` by an internal mutex.
181 void onOffline() {
182 std::scoped_lock const lock{_mtx};
183 _deps.activateLocal();
184 _deps.bindContext();
185 }
186
187private:
188 /// @brief Logs each null `Deps` member at error level (in all builds; does
189 /// not throw — construction still succeeds).
190 static void assertDepsNonNull(const Deps& deps) {
191 auto check = [](const char* name, bool present) {
192 if (!present) {
193 ::morph::log::logError(std::string{"[reconnect_coordinator] null Deps member: "} + name);
194 }
195 };
196 check("tryReconnect", static_cast<bool>(deps.tryReconnect));
197 check("activatePrimary", static_cast<bool>(deps.activatePrimary));
198 check("activateLocal", static_cast<bool>(deps.activateLocal));
199 check("bindContext", static_cast<bool>(deps.bindContext));
200 check("replay", static_cast<bool>(deps.replay));
201 check("shouldContinue", static_cast<bool>(deps.shouldContinue));
202 check("sleep", static_cast<bool>(deps.sleep));
203 }
204
205 /// @brief Calls `tryReconnect`, treating a thrown exception as a failed attempt.
206 [[nodiscard]] bool callTryReconnect() const noexcept {
207 try {
208 return _deps.tryReconnect();
209 } catch (...) {
210 return false;
211 }
212 }
213
214 /// @brief Calls `shouldContinue`, treating a throw as "do not continue".
215 [[nodiscard]] bool callShouldContinue() const noexcept {
216 try {
217 return _deps.shouldContinue();
218 } catch (...) {
219 return false;
220 }
221 }
222
223 /// @brief Emits the `reconnectOutcome` counter tagged by @p outcome, then
224 /// returns it — lets each `onOnline()` return site emit-and-return
225 /// in one expression.
226 static ReconnectOutcome emitOutcome(ReconnectOutcome outcome) noexcept {
227 std::array<std::pair<std::string_view, std::string_view>, 1> const tags{
228 {{"outcome", detail::reconnectOutcomeName(outcome)}}};
229 ::morph::observe::detail::emitMetric(::morph::observe::Metric::reconnectOutcome, 1.0, tags);
230 return outcome;
231 }
232
233 Deps _deps;
234 Config _cfg;
235 std::mutex _mtx; ///< Serialises onOnline()/onOffline().
236};
