44/// @brief Hash functor for `std::pair<std::string, std::string>` keys used by registries.
45/// Shared between `ActionDispatcher` (server-side dispatch) and `ActionExecuteRegistry`
46/// (client-side generic-execute) since their key types are structurally identical.
47struct PairKeyHash {
48 /// @brief Combines the hashes of `key.first` and `key.second`.
49 /// @param key The pair to hash.
50 /// @return The combined hash value.
51 [[nodiscard]] std::size_t operator()(const std::pair<std::string, std::string>& key) const noexcept {
52 std::size_t seed = std::hash<std::string>{}(key.first);
53 // NOLINTNEXTLINE(readability-magic-numbers, cppcoreguidelines-avoid-magic-numbers)
54 seed ^= std::hash<std::string>{}(key.second) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
55 return seed;
56 }
57};

59/// @brief Concept satisfied by actions that expose a `bool validate() const` member.
60///
61/// Drives the default `ActionValidator::ready(...)` path: if the action supplies
62/// a `validate()` method, it is called automatically; otherwise readiness defaults
63/// to `true` (matching the existing one-shot semantics).
64template <typename A>
65concept HasValidate = requires(const A& act) {
66 { act.validate() } -> std::convertible_to<bool>;
67};

71/// @brief Per-action validator used by `BridgeHandler::set<...>` to decide
72/// whether the in-progress draft is ready to execute.
73///
74/// Resolution order (highest-priority first):
75/// 1. **Explicit specialisation** of `ActionValidator<Action>` — typically via
76/// `BRIDGE_REGISTER_VALIDATOR(Action, fn)`. Wins over everything.
77/// 2. **`bool validate() const`** member on `Action` — picked up automatically.
78/// Co-locates the predicate with the data it inspects; preferred when you
79/// own the action type.
80/// 3. **Default** — returns `true`, matching one-shot semantics (the first
81/// `set<>` lands and the action fires).
82///
83/// Validation is intentionally a property of the **action**, not the model:
84/// different actions on the same model have different readiness requirements,
85/// and keeping the predicate next to the action keeps the GUI side oblivious
86/// to model internals.
87///
88/// @code
89/// struct FormAction {
90/// double a;
91/// double b;
92/// double c;
93/// bool validate() const { return a != 0.0 && b != 0.0 && c != 0.0; }
94/// };
95/// // No macro needed; ActionValidator<FormAction>::ready calls .validate() automatically.
96/// @endcode
97///
98/// @tparam Action Concrete action type.
99template <typename Action>
100struct ActionValidator {
101 /// @brief Returns `true` if the action is in a state that should be executed.
102 ///
103 /// Auto-detects a `bool validate() const` member on @p action via the
104 /// `morph::model::detail::HasValidate` concept. Falls back to `true`.
105 ///
106 /// @param action Draft action whose readiness is being checked.
107 /// @return `true` if the action should fire, `false` to keep collecting fields.
108 static constexpr bool ready(const Action& action) {
109 if constexpr (detail::HasValidate<Action>) {
110 return action.validate();
111 } else {
112 (void)action;
113 return true;
114 }
115 }
116};

118/// @brief Thrown when a decoded action fails `ActionValidator<Action>::ready(...)`
119/// on a server-side or local execution path, before `Model::execute` runs.
120///
121/// Raised by `morph::model::detail::ActionDispatcher::registerAction`'s runner
122/// (the server dispatch path used by `RemoteServer`, every remote and Qt
123/// WebSocket topology) and by `Bridge::executeVia`'s `localOp` (the in-process
124/// path used by `LocalBackend`) — the two execution sites an in-progress action
125/// reaches without first passing through the reactive `set<>` gate
126/// (`BridgeHandler::tryFireImpl`) or the type-erased `executeJson` gate
127/// (`ActionExecuteRegistry::registerAction`), both of which already enforce
128/// `ready()` themselves. Deriving from `std::runtime_error` means existing
129/// `catch (const std::exception&)` handling — e.g. `RemoteServer::dispatchExecute`'s
130/// strand catch, which turns any thrown exception into an `err` reply — keeps
131/// working unchanged; callers that care can `catch`/`dynamic_cast` the specific
132/// type instead.
133struct ValidationError : std::runtime_error {
134 /// @brief Constructs the error with a message of the form
135 /// `"action failed validation: <modelType>/<actionType>"`.
136 /// @param modelType `ModelTraits<Model>::typeId()` of the target model.
137 /// @param actionType `ActionTraits<Action>::typeId()` of the rejected action.
138 ValidationError(std::string_view modelType, std::string_view actionType)
139 : std::runtime_error("action failed validation: " + std::string{modelType} + "/" + std::string{actionType}) {}
140};

142/// @brief Whether an action's executions are recorded to an attached action log.
143///
144/// A strong type instead of a bare `bool` so registration call sites read as
145/// intent (`Loggable::No`) rather than an unexplained `false`.
146enum class Loggable : std::uint8_t { No, Yes };

148/// @brief Per-action policy deciding how repeated executions are checkpointed
149/// into a durable action log.
150///
151/// Deliberately minimal for now — only `coalesce` exists, and it has no
152/// registration macro yet. Specialise directly for the rare action where only
153/// the latest occurrence should survive a checkpoint (e.g. a form-field edit
154/// fired repeatedly via `BridgeHandler::set<...>`); every other action defaults
155/// to `false`, meaning every execution is treated as a distinct, permanent fact
156/// (the right default for anything resembling a business event).
157/// @tparam Action Concrete action type.
158template <typename Action>
159struct ActionLogPolicy {
160 /// @brief If `true`, a checkpoint keeps only the most recent entry per
161 /// `(modelType, entityKey, actionType)`. If `false` (default), every
162 /// entry survives.
163 static constexpr bool coalesce = false;
164};

168/// @brief Concept satisfied by `ActionTraits<A>` specialisations that expose a
169/// `Loggable loggable` static member.
170///
171/// Lets `actionLoggable()` default to `Loggable::Yes` for hand-written
172/// `ActionTraits` specialisations (as used in tests) that predate this member,
173/// instead of requiring every existing specialisation to be updated.
174template <typename A>
175concept HasLoggableFlag = requires {
176 { ActionTraits<A>::loggable } -> std::convertible_to<Loggable>;
177};

179/// @brief Returns `ActionTraits<A>::loggable` if present, otherwise `Loggable::Yes`.
180///
181/// Logging every action by default (opt out via the macro's 4th argument) means
182/// new actions are captured automatically; only actions that are known to be
183/// pure queries need to opt out explicitly.
184template <typename A>
185constexpr Loggable actionLoggable() {
186 if constexpr (HasLoggableFlag<A>) {
187 return ActionTraits<A>::loggable;
188 } else {
189 return Loggable::Yes;
190 }
191}

193/// @brief Concept satisfied by `ActionTraits<A>` specialisations that expose a
194/// `payloadSchema()` static member.
195///
196/// Both `BRIDGE_REGISTER_ACTION` and `BRIDGE_REGISTER_ACTION_FOR_CLIENT`
197/// generate one; a hand-written `ActionTraits` does not have to.
198template <typename A>
199concept HasPayloadSchema = requires {
200 { ActionTraits<A>::payloadSchema() } -> std::convertible_to<std::string_view>;
201};
202
203/// @brief Returns `ActionTraits<A>::payloadSchema()` if the specialisation
204/// provides one, otherwise the empty string.
205///
206/// **The fingerprint is a property of the codec, not of the action type.** The
207/// macro-generated codecs read and write through Glaze's reflection, so their
208/// JSON shape *is* the struct's reflected shape and
209/// `morph::model::payloadFingerprint` describes it exactly. A hand-written
210/// `ActionTraits` may map its struct to entirely different JSON -- it may not
211/// even be a reflectable type (`test_client_execute_deadline.cpp` registers an
212/// anonymous-namespace struct, which Glaze's traditional reflection cannot name
213/// at all) -- so deriving a "shape" from the struct would describe something
214/// that never reaches the journal.
215///
216/// Empty means *unstamped*: entries for such an action are recorded without a
217/// fingerprint and replay treats them exactly as every build before the
218/// fingerprint existed did (see `journal::UnstampedPayloadPolicy`). A
219/// hand-written specialisation that wants the guarantee opts in by defining its
220/// own `payloadSchema()`.
221///
222/// @tparam A Action type.
223/// @return The fingerprint, or `""` when the specialisation provides none.
224template <typename A>
225inline std::string actionPayloadSchema() {
226 if constexpr (HasPayloadSchema<A>) {
227 return std::string{ActionTraits<A>::payloadSchema()};
228 } else {
229 return {};
230 }
231}

238/// @brief Write options for an action's/result's own JSON body: identical to
239/// `morph::wire::detail::EscapingWriteOpts`, applied one layer down.
240///
241/// The envelope codec already escapes ASCII control bytes (see
242/// `docs/spec/core/wire.md`, "Control bytes in string fields"); the action and
243/// result bodies it carries are serialized here, separately, and need exactly
244/// the same treatment for exactly the same two reasons. With the option off,
245/// a raw `0x00`–`0x1F` in any caller-supplied string field of an action makes
246/// the body invalid JSON that the peer's own reader rejects — and, when the
247/// same string also contains an escaped character, glaze's chunked fast path
248/// silently rewrites such a byte as two `0x00`s, destroying the payload in a
249/// way that still decodes. Action bodies are pure caller data (a paste's
250/// content, a message, a filename), so this is if anything more exposed than
251/// the envelope was.
252///
253/// Deliberately duplicated rather than reused from `morph::wire`: the action
254/// codec belongs to the model layer and must not acquire a dependency on the
255/// transport layer's header just to share a four-line option struct.
256///
257/// Applies to writing only — glaze's reader already accepts `\\uXXXX`.
258struct EscapingWriteOpts : glz::opts {
259 /// @brief Emit control bytes as `\\uXXXX` rather than raw.
260 // NOLINTNEXTLINE(readability-identifier-naming) — the name is glaze's, not ours; the option is matched by name.
261 bool escape_control_characters = true;
262};

264/// @brief Everything a peer that is **not** linked against an action's C++ needs
265/// in order to build, and to sanity-check, a payload for it.
266///
267/// Assembled once per action type (see `actionDescription`) and served verbatim
268/// by `RemoteServer`'s `"schemas"` envelope kind. Two consumers, one document:
269/// a generic client renders `schema` as a form, and `RemoteServer`'s optional
270/// `PayloadCompleteness::RequireDeclaredFields` gate reads `required` — so the
271/// rule the server enforces is by construction the rule it published.
272struct ActionDescription {
273 /// @brief `morph::forms::schemaJson<A>()` with two extra top-level keys
274 /// merged in: `x-payloadFingerprint` and `x-payloadShape`.
275 ///
276 /// Empty only when schema generation produced nothing usable at all.
277 std::string schema;
278
279 /// @brief Wire names of every member the schema's `required` array lists.
280 ///
281 /// Derived from the served document rather than recomputed, so a field the
282 /// gate insists on is exactly a field the schema told the client about.
283 std::vector<std::string> required;
284};
285
286/// @brief Builds `A`'s `ActionDescription`.
287///
288/// The merge is done over a DOM rather than by splicing text: the fingerprint
289/// and shape are ordinary JSON string values, and glaze's writer is the only
290/// thing here that knows how to escape them. On a DOM read failure the raw
291/// `morph::forms::schemaJson<A>()` text is served unannotated and `required` is
292/// left empty — a description facility degrades, it does not throw.
293///
294/// @tparam A Action type.
295/// @return The assembled description.
296template <typename A>
297[[nodiscard]] inline ActionDescription buildActionDescription() {
298 ActionDescription desc;
299 desc.schema = ::morph::forms::schemaJson<A>();
300 // u64 number mode for the same reason mergeSchemaExtras uses it: the
301 // schema carries int64/uint64 bounds a double-only DOM would round on the
302 // way back out.
303 glz::generic_u64 dom{};
304 if (glz::read_json(dom, desc.schema) || !dom.is_object()) {
305 return desc;
306 }
307 if (dom.contains("required") && dom["required"].is_array()) {
308 for (const auto& name : dom["required"].get_array()) {
309 if (name.is_string()) {
310 desc.required.push_back(name.template get<std::string>());
311 }
312 }
313 }
314 dom["x-payloadFingerprint"] = payloadFingerprint<A>();
315 dom["x-payloadShape"] = payloadShapeString<A>();
316 std::string merged;
317 if (!glz::write_json(dom, merged)) {
318 desc.schema = std::move(merged);
319 }
320 return desc;
321}

323/// @brief `A`'s `ActionDescription`, computed once per type per process.
324///
325/// A function-local static, exactly like `morph::forms::schemaJson<A>()`'s own
326/// cache: `RemoteServer` may answer `"schemas"` on any pool thread, and the
327/// maps `ActionDispatcher` fills at static-init time are read-only afterwards,
328/// so the only mutable state involved has to be the one C++ already guarantees
329/// is initialised exactly once. A throw from `buildActionDescription` (an
330/// action whose `formRules` are self-contradicting — see
331/// `forms::UnsatisfiableFormError`) leaves the static uninitialised and a later
332/// call retries, rather than serving a half-built document.
333///
334/// @tparam A Action type.
335/// @return Reference to the process-lifetime description for `A`.
336template <typename A>
337[[nodiscard]] inline const ActionDescription& actionDescription() {
338 static const ActionDescription kDesc = buildActionDescription<A>();
339 return kDesc;
340}

346/// @brief Registry that maps (modelId, actionId) pairs to type-erased runner functions.
347///
348/// Used by `RemoteServer` to dispatch incoming JSON requests without knowing the
349/// concrete model or action types at the call site.
350class ActionDispatcher {
351public:
352 /// @brief Type-erased action runner: deserialises, executes, and serialises the result.
353 using Runner = std::function<std::string(IModelHolder&, std::string_view)>;
354
355 /// @brief Registers a runner for `(Model, Action)` under the given string ids.
356 ///
357 /// This is the single execution site used by `RemoteServer` (every remote and
358 /// Qt WebSocket topology) — `Model::execute()` runs here, on whichever process
359 /// actually owns @p holder. Before `Model::execute` runs, the runner reconciles
360 /// any `Quantity` fields to their declared precision
361 /// (`morph::forms::reconcileDeclaredPrecision`), overwrites any declared
362 /// computed fields from their inputs (`morph::forms::recomputeAll`,
363 /// `forms.hpp`), and enforces `ActionValidator<Action>::ready(action)`,
364 /// throwing `ValidationError` when it returns `false` — the same checks the
365 /// client bridge dispatch path (`ActionExecuteRegistry::registerAction`,
366 /// `bridge.hpp`) already performs. If a `journal::IActionLog` is attached to
367 /// @p holder (via `IModelHolder::attachActionLog`) and `Action` is loggable
368 /// (the default), the executed action is recorded automatically -- on a
369 /// successful `Model::execute` with `outcome = Outcome::Succeeded` and the
370 /// JSON result, and equally on a thrown `std::exception` (a validation
371 /// failure, a rejected write) with `outcome = Outcome::Failed` and
372 /// `error = what()`, `result` empty. Either way the exception (if any)
373 /// propagates unchanged after the entry is recorded.
374 ///
375 /// Every recorded entry is stamped with `payloadFingerprint<Action>()` in
376 /// `LogEntry::schema`, and the same fingerprint is filed under
377 /// `(modelId, actionId)` for `schemaFor()` to hand back to the type-erased
378 /// replay path. That pair is what lets `journal::replay()` notice that the
379 /// shape which wrote an entry is not the shape it is about to decode it
380 /// with -- see `docs/spec/journal/journal.md`, "Payload schema fingerprint".
381 /// @throws ValidationError if the decoded action fails `ActionValidator<Action>::ready`.
382 template <typename Model, typename Action>
383 void registerAction(std::string_view modelId, std::string_view actionId) {
384 Key const key{std::string{modelId}, std::string{actionId}};
385 _runners[key] = [](IModelHolder& holder, std::string_view payloadJson) {
386 auto action = ActionTraits<Action>::fromJson(payloadJson);
387 // Retag any Quantity fields to their declared precision so a
388 // hand-built wire payload matches the schema's advertised
389 // x-decimalPlaces, exactly as the client bridge dispatch path
390 // (ActionExecuteRegistry::registerAction, bridge.hpp) already does.
391 // No-op for actions with no Quantity members. See
392 // docs/spec/forms/forms.md.
393 ::morph::forms::reconcileDeclaredPrecision(action);
394 // Pre-decode wire validation seam: reject a Quantity field whose
395 // engaged value falls outside its unit's declared bounds
396 // (UnitTraits<E>::bounds), before the action's own validate()
397 // (a business-rule check) ever runs. No-op for actions with no
398 // Quantity members, or whose units declare no bounds(). See
399 // docs/spec/forms/forms.md, "Pre-decode wire validation".
400 ::morph::forms::enforceQuantityBounds(action);
401 // Overwrite any computed fields from their declared inputs. This is
402 // the true server-side execution site for every remote and Qt
403 // WebSocket topology (RemoteServer -> ActionDispatcher::dispatch) --
404 // the one path a hand-built wire envelope reaches directly,
405 // bypassing every client-side gate. A tampered computed value on
406 // the wire is discarded here, before the validator check and
407 // Model::execute run. No-op for actions with no computedFields. See
408 // docs/spec/forms/forms.md.
409 ::morph::forms::recomputeAll(action);
410 // Enforce the action's validator on the server dispatch path — the
411 // one path an untrusted remote client can drive directly with a
412 // hand-built envelope, bypassing the client-side gates
413 // (BridgeHandler::set<>'s tryFireImpl and
414 // ActionExecuteRegistry::registerAction). ActionValidator<Action>::ready
415 // auto-detects a `bool validate() const` member and defaults to
416 // `true` for actions with no validator, so this is a no-op for
417 // unvalidated actions (zero behavior change) and a hard gate for
418 // validated ones. The exception propagates out of this lambda to
419 // ActionDispatcher::dispatch's caller (RemoteServer::dispatchExecute's
420 // strand catch turns it into an `err` reply).
421 if (!ActionValidator<Action>::ready(action)) {
422 throw ValidationError{ModelTraits<Model>::typeId(), ActionTraits<Action>::typeId()};
423 }
424 auto& model = holder.template into<Model>();
425 // Both the success and failure paths below record a journal entry
426 // (when a log is attached and Action is loggable) so a rejected or
427 // throwing execution -- a validation failure, a lost connection, a
428 // rejected write -- still leaves an audit trail, not silence. The
429 // exception is rethrown unchanged either way; only the outcome
430 // shape differs. See docs/spec/journal/journal.md, "Outcome".
431 try {
432 auto result = model.execute(action);
433 auto resultJson = ActionTraits<Action>::resultToJson(result);
434 if constexpr (detail::actionLoggable<Action>() == Loggable::Yes) {
435 if (holder.hasActionLog()) {
436 // entityKey/principal/timestampMs are filled in by recordIfAttached.
437 holder.recordIfAttached(::morph::journal::LogEntry{
438 .seq = 0,
439 .modelType = std::string{ModelTraits<Model>::typeId()},
440 .entityKey = {},
441 .actionType = std::string{ActionTraits<Action>::typeId()},
442 .payload = std::string{payloadJson},
443 .schema = detail::actionPayloadSchema<Action>(),
444 .result = resultJson,
445 .outcome = ::morph::journal::Outcome::Succeeded,
446 .error = {},
447 .principal = {},
448 .timestampMs = 0,
449 });
450 }
451 }
452 return resultJson;
453 } catch (const std::exception& exc [[maybe_unused]]) {
454 if constexpr (detail::actionLoggable<Action>() == Loggable::Yes) {
455 if (holder.hasActionLog()) {
456 holder.recordIfAttached(::morph::journal::LogEntry{
457 .seq = 0,
458 .modelType = std::string{ModelTraits<Model>::typeId()},
459 .entityKey = {},
460 .actionType = std::string{ActionTraits<Action>::typeId()},
461 .payload = std::string{payloadJson},
462 .schema = detail::actionPayloadSchema<Action>(),
463 .result = {},
464 .outcome = ::morph::journal::Outcome::Failed,
465 .error = exc.what(),
466 .principal = {},
467 .timestampMs = 0,
468 });
469 }
470 }
471 throw;
472 }
473 };
474 _coalesce[key] = ActionLogPolicy<Action>::coalesce;
475 _schema[key] = detail::actionPayloadSchema<Action>();
476 // Deliberately a thunk, not the description itself: registration runs
477 // at static-init time, where `buildActionDescription`'s throw path (an
478 // action with self-contradicting `formRules`) would abort the process
479 // before `main` rather than surface as an `err` reply. The thunk defers
480 // the whole computation to the first caller who asks.
481 _describe[key] = []() -> const ActionDescription& { return actionDescription<Action>(); };
482 }

484 /// @brief Dispatches an action against @p holder and returns the JSON-encoded result.
485 std::string dispatch(std::string_view modelId, std::string_view actionId, IModelHolder& holder,
486 std::string_view payload) {
487 Key const key{std::string{modelId}, std::string{actionId}};
488 auto iter = _runners.find(key);
489 if (iter == _runners.end()) {
490 throw std::runtime_error("unknown action: " + key.first + "/" + key.second);
491 }
492 return iter->second(holder, payload);
493 }

495 /// @brief Returns whether `(modelId, actionId)` was registered with
496 /// `ActionLogPolicy<Action>::coalesce == true`.
497 ///
498 /// Used by `journal::SessionLog::checkpoint()` to decide, from the type-erased
499 /// `LogEntry` stream, which entries collapse to their latest occurrence.
500 /// Unknown pairs (never registered) default to `false` — every entry kept.
501 [[nodiscard]] bool coalesce(std::string_view modelId, std::string_view actionId) const {
502 auto iter = _coalesce.find(Key{std::string{modelId}, std::string{actionId}});
503 return iter != _coalesce.end() && iter->second;
504 }

506 /// @brief Returns the payload fingerprint `(modelId, actionId)` was
507 /// registered with, or an empty string if the pair is unregistered.
508 ///
509 /// This is the type-erased half of the journal's payload-evolution check:
510 /// `registerAction` knows the concrete `Action` and can compute
511 /// `payloadFingerprint<Action>()`; `journal::replay()` sees only strings on
512 /// a `LogEntry` and needs to ask for the current build's fingerprint by id.
513 /// The empty return is not an error -- an entry naming an unregistered
514 /// action fails at `dispatch()` with "unknown action" a moment later, which
515 /// is the better diagnostic for that case.
516 ///
517 /// @param modelId Model type-id.
518 /// @param actionId Action type-id.
519 /// @return The registered fingerprint, or `""` if the pair is unregistered.
520 [[nodiscard]] std::string schemaFor(std::string_view modelId, std::string_view actionId) const {
521 auto iter = _schema.find(Key{std::string{modelId}, std::string{actionId}});
522 return iter == _schema.end() ? std::string{} : iter->second;
523 }

525 /// @brief Returns the `{actionType: schema}` document for every action
526 /// registered under @p modelId.
527 ///
528 /// The wire's answer to `"schemas"` (`wire::makeSchemas`). Action ids are
529 /// emitted in sorted order so two calls — and two servers built from the
530 /// same sources — produce byte-identical documents; `_describe` is an
531 /// unordered map, whose iteration order is not a contract.
532 ///
533 /// A model type with no registered actions yields `{}` rather than an
534 /// error: "this type exposes no actions" and "this type does not exist"
535 /// are not distinguishable here (`ActionDispatcher` never sees a model
536 /// type that registered no action at all), and inventing a distinction
537 /// would tell an unauthenticated prober which type ids are real.
538 ///
539 /// @param modelId Model type-id to describe.
540 /// @return The `{actionType: schema}` JSON object.
541 /// @throws morph::forms::UnsatisfiableFormError if one of the model's
542 /// actions declares `formRules` no submission could satisfy —
543 /// propagated from `morph::forms::schemaJson`, which decides that
544 /// at generation time.
545 [[nodiscard]] std::string schemasJson(std::string_view modelId) const {
546 std::vector<std::string_view> actionIds;
547 for (const auto& [key, thunk] : _describe) {
548 if (key.first == modelId) {
549 actionIds.emplace_back(key.second);
550 }
551 }
552 std::ranges::sort(actionIds);
553 std::string out{"{"};
554 bool first = true;
555 for (std::string_view const actionId : actionIds) {
556 if (!first) {
557 out += ',';
558 }
559 first = false;
560 // The id is a C++ string literal from BRIDGE_REGISTER_ACTION, but
561 // it reaches a client as JSON, so it is written as JSON rather
562 // than pasted between hand-written quotes.
563 std::string quoted;
564 (void)glz::write_json(std::string{actionId}, quoted);
565 out += quoted;
566 out += ':';
567 const auto& desc = _describe.at(Key{std::string{modelId}, std::string{actionId}})();
568 out += desc.schema.empty() ? "{}" : desc.schema;
569 }

618/// @brief Registry that creates `IModelHolder` instances by string type-id.
619///
620/// Used by `RemoteServer` to instantiate models on demand from incoming
621/// `"register"` messages.
622class ModelRegistryFactory {
623public:
624 /// @brief Registers a default-construction factory for `Model` under @p modelId.
625 ///
626 /// Equivalent to `registerModel<Model>(modelId, [] { return
627 /// ModelFactory::create<Model>(); })` — the plain default-construction path
628 /// every `BRIDGE_REGISTER_MODEL` invocation uses.
629 /// @param modelId String id to register the factory under.
630 template <typename Model>
631 void registerModel(std::string_view modelId) {
632 _factories.insert_or_assign(std::string{modelId}, [] { return ModelFactory::create<Model>(); });
633 }
634
635 /// @brief Registers a custom construction hook for `Model` under @p modelId.
636 ///
637 /// The per-instance dependency-injection seam for registry-constructed
638 /// (`Socket`-mode / `RemoteServer`) models — the equivalent of
639 /// `Bridge::HandlerBinding::modelFactory` for the client-side `Local`-mode
640 /// path. @p factory runs once per incoming `"register"` message (and once
641 /// per fresh shared-instance creation) and may capture arbitrary
642 /// per-instance dependencies (an injectable clock, a secondary log handle,
643 /// a feature flag) that `Model`'s default constructor cannot reach — this
644 /// is also the seam a model whose only constructor takes arguments (i.e.
645 /// is not default-constructible at all) needs in order to be registered
646 /// for remote/socket instantiation in the first place.
647 ///
648 /// @p factory returns an owning pointer to a freshly built holder (e.g.
649 /// `std::make_unique<ModelHolder<Model>>(...)`), not a bare `Model` — the
650 /// caller controls construction end-to-end, including passing
651 /// constructor arguments `ModelHolder`'s forwarding constructor accepts.
652 /// This overload does **not** auto-attach the process-wide default action
653 /// log the way the default-construction overload does: a caller supplying
654 /// its own factory is assumed to attach whatever log/identity it needs
655 /// (via `IModelHolder::attachActionLog`) inside the closure, or to rely on
656 /// `RemoteServer`'s `LogProvider` doing so afterward, exactly as the
657 /// default path allows.
658 ///
659 /// @tparam Model Concrete model type. Used only to assert (debug builds)
660 /// that @p factory's returned holder actually reports
661 /// `typeid(Model)` — @p factory is otherwise free to build
662 /// the holder however it likes, so nothing *prevents* a
663 /// mismatched `Model`/@p factory pairing at compile time.
664 /// @tparam Factory Callable returning an owning pointer convertible to
665 /// `std::unique_ptr<IModelHolder>`; deduced from @p factory.
666 /// @param modelId String id to register the factory under.
667 /// @param factory Callable that constructs and returns a fresh holder.
668 template <typename Model, typename Factory>
669 requires std::invocable<Factory> &&
670 std::convertible_to<std::invoke_result_t<Factory>, std::unique_ptr<IModelHolder>>
671 void registerModel(std::string_view modelId, Factory factory) {
672 _factories.insert_or_assign(std::string{modelId}, [factory = std::move(factory)]() mutable {
673 std::unique_ptr<IModelHolder> holder{factory()};
674 assert(!holder ||
675 (holder->type() == std::type_index(typeid(Model)) &&
676 "registerModel<Model>(modelId, factory): factory returned a holder for a different type"));
677 return holder;
678 });
679 }
680
681 /// @brief Creates a new model holder for the type registered under @p modelId.
682 std::unique_ptr<IModelHolder> create(std::string_view modelId) {
683 auto iter = _factories.find(std::string{modelId});
684 if (iter == _factories.end()) {
685 throw std::runtime_error("unknown model type: " + std::string{modelId});
686 }
687 return iter->second();
688 }
689
690 /// @brief Returns the process-level singleton registry.
691 static ModelRegistryFactory& instance() { return defaultRegistry(); }
692
693private:
694 std::unordered_map<std::string, std::function<std::unique_ptr<IModelHolder>()>> _factories;
695};
