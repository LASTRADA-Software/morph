40/// @brief Hash functor for `std::pair<std::string, std::string>` keys used by registries.
41/// Shared between `ActionDispatcher` (server-side dispatch) and `ActionExecuteRegistry`
42/// (client-side generic-execute) since their key types are structurally identical.
43struct PairKeyHash {
44 /// @brief Combines the hashes of `key.first` and `key.second`.
45 /// @param key The pair to hash.
46 /// @return The combined hash value.
47 [[nodiscard]] std::size_t operator()(const std::pair<std::string, std::string>& key) const noexcept {
48 std::size_t seed = std::hash<std::string>{}(key.first);
49 // NOLINTNEXTLINE(readability-magic-numbers, cppcoreguidelines-avoid-magic-numbers)
50 seed ^= std::hash<std::string>{}(key.second) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
51 return seed;
52 }
53};

55/// @brief Concept satisfied by actions that expose a `bool validate() const` member.
56///
57/// Drives the default `ActionValidator::ready(...)` path: if the action supplies
58/// a `validate()` method, it is called automatically; otherwise readiness defaults
59/// to `true` (matching the existing one-shot semantics).
60template <typename A>
61concept HasValidate = requires(const A& act) {
62 { act.validate() } -> std::convertible_to<bool>;
63};

67/// @brief Per-action validator used by `BridgeHandler::set<...>` to decide
68/// whether the in-progress draft is ready to execute.
69///
70/// Resolution order (highest-priority first):
71/// 1. **Explicit specialisation** of `ActionValidator<Action>` — typically via
72/// `BRIDGE_REGISTER_VALIDATOR(Action, fn)`. Wins over everything.
73/// 2. **`bool validate() const`** member on `Action` — picked up automatically.
74/// Co-locates the predicate with the data it inspects; preferred when you
75/// own the action type.
76/// 3. **Default** — returns `true`, matching one-shot semantics (the first
77/// `set<>` lands and the action fires).
78///
79/// Validation is intentionally a property of the **action**, not the model:
80/// different actions on the same model have different readiness requirements,
81/// and keeping the predicate next to the action keeps the GUI side oblivious
82/// to model internals.
83///
84/// @code
85/// struct FormAction {
86/// double a;
87/// double b;
88/// double c;
89/// bool validate() const { return a != 0.0 && b != 0.0 && c != 0.0; }
90/// };
91/// // No macro needed; ActionValidator<FormAction>::ready calls .validate() automatically.
92/// @endcode
93///
94/// @tparam Action Concrete action type.
95template <typename Action>
96struct ActionValidator {
97 /// @brief Returns `true` if the action is in a state that should be executed.
98 ///
99 /// Auto-detects a `bool validate() const` member on @p action via the
100 /// `morph::model::detail::HasValidate` concept. Falls back to `true`.
101 ///
102 /// @param action Draft action whose readiness is being checked.
103 /// @return `true` if the action should fire, `false` to keep collecting fields.
104 static constexpr bool ready(const Action& action) {
105 if constexpr (detail::HasValidate<Action>) {
106 return action.validate();
107 } else {
108 (void)action;
109 return true;
110 }
111 }
112};

114/// @brief Thrown when a decoded action fails `ActionValidator<Action>::ready(...)`
115/// on a server-side or local execution path, before `Model::execute` runs.
116///
117/// Raised by `morph::model::detail::ActionDispatcher::registerAction`'s runner
118/// (the server dispatch path used by `RemoteServer`, every remote and Qt
119/// WebSocket topology) and by `Bridge::executeVia`'s `localOp` (the in-process
120/// path used by `LocalBackend`) — the two execution sites an in-progress action
121/// reaches without first passing through the reactive `set<>` gate
122/// (`BridgeHandler::tryFireImpl`) or the type-erased `executeJson` gate
123/// (`ActionExecuteRegistry::registerAction`), both of which already enforce
124/// `ready()` themselves. Deriving from `std::runtime_error` means existing
125/// `catch (const std::exception&)` handling — e.g. `RemoteServer::dispatchExecute`'s
126/// strand catch, which turns any thrown exception into an `err` reply — keeps
127/// working unchanged; callers that care can `catch`/`dynamic_cast` the specific
128/// type instead.
129struct ValidationError : std::runtime_error {
130 /// @brief Constructs the error with a message of the form
131 /// `"action failed validation: <modelType>/<actionType>"`.
132 /// @param modelType `ModelTraits<Model>::typeId()` of the target model.
133 /// @param actionType `ActionTraits<Action>::typeId()` of the rejected action.
134 ValidationError(std::string_view modelType, std::string_view actionType)
135 : std::runtime_error("action failed validation: " + std::string{modelType} + "/" + std::string{actionType}) {}
136};

138/// @brief Whether an action's executions are recorded to an attached action log.
139///
140/// A strong type instead of a bare `bool` so registration call sites read as
141/// intent (`Loggable::No`) rather than an unexplained `false`.
142enum class Loggable : std::uint8_t { No, Yes };
143
144/// @brief Per-action policy deciding how repeated executions are checkpointed
145/// into a durable action log.
146///
147/// Deliberately minimal for now — only `coalesce` exists, and it has no
148/// registration macro yet. Specialise directly for the rare action where only
149/// the latest occurrence should survive a checkpoint (e.g. a form-field edit
150/// fired repeatedly via `BridgeHandler::set<...>`); every other action defaults
151/// to `false`, meaning every execution is treated as a distinct, permanent fact
152/// (the right default for anything resembling a business event).
153/// @tparam Action Concrete action type.
154template <typename Action>
155struct ActionLogPolicy {
156 /// @brief If `true`, a checkpoint keeps only the most recent entry per
157 /// `(modelType, entityKey, actionType)`. If `false` (default), every
158 /// entry survives.
159 static constexpr bool coalesce = false;
160};

164/// @brief Concept satisfied by `ActionTraits<A>` specialisations that expose a
165/// `Loggable loggable` static member.
166///
167/// Lets `actionLoggable()` default to `Loggable::Yes` for hand-written
168/// `ActionTraits` specialisations (as used in tests) that predate this member,
169/// instead of requiring every existing specialisation to be updated.
170template <typename A>
171concept HasLoggableFlag = requires {
172 { ActionTraits<A>::loggable } -> std::convertible_to<Loggable>;
173};
174
175/// @brief Returns `ActionTraits<A>::loggable` if present, otherwise `Loggable::Yes`.
176///
177/// Logging every action by default (opt out via the macro's 4th argument) means
178/// new actions are captured automatically; only actions that are known to be
179/// pure queries need to opt out explicitly.
180template <typename A>
181constexpr Loggable actionLoggable() {
182 if constexpr (HasLoggableFlag<A>) {
183 return ActionTraits<A>::loggable;
184 } else {
185 return Loggable::Yes;
186 }
187}

198/// @brief Registry that maps (modelId, actionId) pairs to type-erased runner functions.
199///
200/// Used by `RemoteServer` to dispatch incoming JSON requests without knowing the
201/// concrete model or action types at the call site.
202class ActionDispatcher {
203public:
204 /// @brief Type-erased action runner: deserialises, executes, and serialises the result.
205 using Runner = std::function<std::string(IModelHolder&, std::string_view)>;
206
207 /// @brief Registers a runner for `(Model, Action)` under the given string ids.
208 ///
209 /// This is the single execution site used by `RemoteServer` (every remote and
210 /// Qt WebSocket topology) — `Model::execute()` runs here, on whichever process
211 /// actually owns @p holder. Before `Model::execute` runs, the runner reconciles
212 /// any `Quantity` fields to their declared precision
213 /// (`morph::forms::reconcileDeclaredPrecision`), overwrites any declared
214 /// computed fields from their inputs (`morph::forms::recomputeAll`,
215 /// `forms.hpp`), and enforces `ActionValidator<Action>::ready(action)`,
216 /// throwing `ValidationError` when it returns `false` — the same checks the
217 /// client bridge dispatch path (`ActionExecuteRegistry::registerAction`,
218 /// `bridge.hpp`) already performs. If a `journal::IActionLog` is attached to
219 /// @p holder (via `IModelHolder::attachActionLog`) and `Action` is loggable
220 /// (the default), the executed action is recorded automatically -- on a
221 /// successful `Model::execute` with `outcome = Outcome::Succeeded` and the
222 /// JSON result, and equally on a thrown `std::exception` (a validation
223 /// failure, a rejected write) with `outcome = Outcome::Failed` and
224 /// `error = what()`, `result` empty. Either way the exception (if any)
225 /// propagates unchanged after the entry is recorded.
226 /// @throws ValidationError if the decoded action fails `ActionValidator<Action>::ready`.
227 template <typename Model, typename Action>
228 void registerAction(std::string_view modelId, std::string_view actionId) {
229 Key const key{std::string{modelId}, std::string{actionId}};
230 _runners[key] = [](IModelHolder& holder, std::string_view payloadJson) {
231 auto action = ActionTraits<Action>::fromJson(payloadJson);
232 // Retag any Quantity fields to their declared precision so a
233 // hand-built wire payload matches the schema's advertised
234 // x-decimalPlaces, exactly as the client bridge dispatch path
235 // (ActionExecuteRegistry::registerAction, bridge.hpp) already does.
236 // No-op for actions with no Quantity members. See
237 // docs/spec/forms/forms.md.
238 ::morph::forms::reconcileDeclaredPrecision(action);
239 // Overwrite any computed fields from their declared inputs. This is
240 // the true server-side execution site for every remote and Qt
241 // WebSocket topology (RemoteServer -> ActionDispatcher::dispatch) --
242 // the one path a hand-built wire envelope reaches directly,
243 // bypassing every client-side gate. A tampered computed value on
244 // the wire is discarded here, before the validator check and
245 // Model::execute run. No-op for actions with no computedFields. See
246 // docs/spec/forms/forms.md.
247 ::morph::forms::recomputeAll(action);
248 // Enforce the action's validator on the server dispatch path — the
249 // one path an untrusted remote client can drive directly with a
250 // hand-built envelope, bypassing the client-side gates
251 // (BridgeHandler::set<>'s tryFireImpl and
252 // ActionExecuteRegistry::registerAction). ActionValidator<Action>::ready
253 // auto-detects a `bool validate() const` member and defaults to
254 // `true` for actions with no validator, so this is a no-op for
255 // unvalidated actions (zero behavior change) and a hard gate for
256 // validated ones. The exception propagates out of this lambda to
257 // ActionDispatcher::dispatch's caller (RemoteServer::dispatchExecute's
258 // strand catch turns it into an `err` reply).
259 if (!ActionValidator<Action>::ready(action)) {
260 throw ValidationError{ModelTraits<Model>::typeId(), ActionTraits<Action>::typeId()};
261 }
262 auto& model = holder.template into<Model>();
263 // Both the success and failure paths below record a journal entry
264 // (when a log is attached and Action is loggable) so a rejected or
265 // throwing execution -- a validation failure, a lost connection, a
266 // rejected write -- still leaves an audit trail, not silence. The
267 // exception is rethrown unchanged either way; only the outcome
268 // shape differs. See docs/spec/journal/journal.md, "Outcome".
269 try {
270 auto result = model.execute(action);
271 auto resultJson = ActionTraits<Action>::resultToJson(result);
272 if constexpr (detail::actionLoggable<Action>() == Loggable::Yes) {
273 if (holder.hasActionLog()) {
274 // entityKey/principal/timestampMs are filled in by recordIfAttached.
275 holder.recordIfAttached(::morph::journal::LogEntry{
276 .seq = 0,
277 .modelType = std::string{ModelTraits<Model>::typeId()},
278 .entityKey = {},
279 .actionType = std::string{ActionTraits<Action>::typeId()},
280 .payload = std::string{payloadJson},
281 .result = resultJson,
282 .outcome = ::morph::journal::Outcome::Succeeded,
283 .error = {},
284 .principal = {},
285 .timestampMs = 0,
286 });
287 }
288 }
289 return resultJson;
290 } catch (const std::exception& exc [[maybe_unused]]) {
291 if constexpr (detail::actionLoggable<Action>() == Loggable::Yes) {
292 if (holder.hasActionLog()) {
293 holder.recordIfAttached(::morph::journal::LogEntry{
294 .seq = 0,
295 .modelType = std::string{ModelTraits<Model>::typeId()},
296 .entityKey = {},
297 .actionType = std::string{ActionTraits<Action>::typeId()},
298 .payload = std::string{payloadJson},
299 .result = {},
300 .outcome = ::morph::journal::Outcome::Failed,
301 .error = exc.what(),
302 .principal = {},
303 .timestampMs = 0,
304 });
305 }
306 }
307 throw;
308 }
309 };
310 _coalesce[key] = ActionLogPolicy<Action>::coalesce;
311 }
312
313 /// @brief Dispatches an action against @p holder and returns the JSON-encoded result.
314 std::string dispatch(std::string_view modelId, std::string_view actionId, IModelHolder& holder,
315 std::string_view payload) {
316 Key const key{std::string{modelId}, std::string{actionId}};
317 auto iter = _runners.find(key);
318 if (iter == _runners.end()) {
319 throw std::runtime_error("unknown action: " + key.first + "/" + key.second);
320 }
321 return iter->second(holder, payload);
322 }
323
324 /// @brief Returns whether `(modelId, actionId)` was registered with
325 /// `ActionLogPolicy<Action>::coalesce == true`.
326 ///
327 /// Used by `journal::SessionLog::checkpoint()` to decide, from the type-erased
328 /// `LogEntry` stream, which entries collapse to their latest occurrence.
329 /// Unknown pairs (never registered) default to `false` — every entry kept.
330 [[nodiscard]] bool coalesce(std::string_view modelId, std::string_view actionId) const {
331 auto iter = _coalesce.find(Key{std::string{modelId}, std::string{actionId}});
332 return iter != _coalesce.end() && iter->second;
333 }
334
335 /// @brief Returns the process-level singleton dispatcher.
336 static ActionDispatcher& instance() { return defaultDispatcher(); }
337
338private:
339 using Key = std::pair<std::string, std::string>;
340 std::unordered_map<Key, Runner, PairKeyHash> _runners;
341 std::unordered_map<Key, bool, PairKeyHash> _coalesce;
342};

344/// @brief Registry that creates `IModelHolder` instances by string type-id.
345///
346/// Used by `RemoteServer` to instantiate models on demand from incoming
347/// `"register"` messages.
348class ModelRegistryFactory {
349public:
350 /// @brief Registers a default-construction factory for `Model` under @p modelId.
351 ///
352 /// Equivalent to `registerModel<Model>(modelId, [] { return
353 /// ModelFactory::create<Model>(); })` — the plain default-construction path
354 /// every `BRIDGE_REGISTER_MODEL` invocation uses.
355 /// @param modelId String id to register the factory under.
356 template <typename Model>
357 void registerModel(std::string_view modelId) {
358 _factories.insert_or_assign(std::string{modelId}, [] { return ModelFactory::create<Model>(); });
359 }
360
361 /// @brief Registers a custom construction hook for `Model` under @p modelId.
362 ///
363 /// The per-instance dependency-injection seam for registry-constructed
364 /// (`Socket`-mode / `RemoteServer`) models — the equivalent of
365 /// `Bridge::HandlerBinding::modelFactory` for the client-side `Local`-mode
366 /// path. @p factory runs once per incoming `"register"` message (and once
367 /// per fresh shared-instance creation) and may capture arbitrary
368 /// per-instance dependencies (an injectable clock, a secondary log handle,
369 /// a feature flag) that `Model`'s default constructor cannot reach — this
370 /// is also the seam a model whose only constructor takes arguments (i.e.
371 /// is not default-constructible at all) needs in order to be registered
372 /// for remote/socket instantiation in the first place.
373 ///
374 /// @p factory returns an owning pointer to a freshly built holder (e.g.
375 /// `std::make_unique<ModelHolder<Model>>(...)`), not a bare `Model` — the
376 /// caller controls construction end-to-end, including passing
377 /// constructor arguments `ModelHolder`'s forwarding constructor accepts.
378 /// This overload does **not** auto-attach the process-wide default action
379 /// log the way the default-construction overload does: a caller supplying
380 /// its own factory is assumed to attach whatever log/identity it needs
381 /// (via `IModelHolder::attachActionLog`) inside the closure, or to rely on
382 /// `RemoteServer`'s `LogProvider` doing so afterward, exactly as the
383 /// default path allows.
384 ///
385 /// @tparam Model Concrete model type. Used only to assert (debug builds)
386 /// that @p factory's returned holder actually reports
387 /// `typeid(Model)` — @p factory is otherwise free to build
388 /// the holder however it likes, so nothing *prevents* a
389 /// mismatched `Model`/@p factory pairing at compile time.
390 /// @tparam Factory Callable returning an owning pointer convertible to
391 /// `std::unique_ptr<IModelHolder>`; deduced from @p factory.
392 /// @param modelId String id to register the factory under.
393 /// @param factory Callable that constructs and returns a fresh holder.
394 template <typename Model, typename Factory>
395 requires std::invocable<Factory> &&
396 std::convertible_to<std::invoke_result_t<Factory>, std::unique_ptr<IModelHolder>>
397 void registerModel(std::string_view modelId, Factory factory) {
398 _factories.insert_or_assign(std::string{modelId}, [factory = std::move(factory)]() mutable {
399 std::unique_ptr<IModelHolder> holder{factory()};
400 assert(!holder
401 || (holder->type() == std::type_index(typeid(Model))
402 && "registerModel<Model>(modelId, factory): factory returned a holder for a different type"));
403 return holder;
404 });
405 }
406
407 /// @brief Creates a new model holder for the type registered under @p modelId.
408 std::unique_ptr<IModelHolder> create(std::string_view modelId) {
409 auto iter = _factories.find(std::string{modelId});
410 if (iter == _factories.end()) {
411 throw std::runtime_error("unknown model type: " + std::string{modelId});
412 }
413 return iter->second();
414 }
415
416 /// @brief Returns the process-level singleton registry.
417 static ModelRegistryFactory& instance() { return defaultRegistry(); }
418
419private:
420 std::unordered_map<std::string, std::function<std::unique_ptr<IModelHolder>()>> _factories;
421};

469// Keying the generated registrar names on `__COUNTER__` (rather than on the type spelling)
470// keeps them valid identifiers regardless of how `M`/`A` are written. A namespace-qualified
471// or template type (e.g. `app::models::Report`) would otherwise be pasted directly into the
472// identifier, producing invalid tokens like `bridge_model_reg_app::models::Report`.
473// `__LINE__` was tried first, but it is only unique *within a single physical file*: two
474// headers that each invoke this macro on the same line number (e.g. `card_model.hpp:38` and
475// `payment_model.hpp:38`) produce the same identifier once both are `#include`d into one
476// translation unit, and since C++ unnamed namespaces are per-TU (not per-file), that is a
477// hard redefinition error. `__COUNTER__` increments monotonically for the entire translation
478// unit regardless of which file expands it, so it cannot collide this way. The extra
479// indirection (`_CAT` calling `_CAT_`) is required so `__COUNTER__` is expanded to its numeric
480// value before the paste, rather than being pasted as the literal text "__COUNTER__".
481#define BRIDGE_DETAIL_CAT_(a, b) a##b
482#define BRIDGE_DETAIL_CAT(a, b) BRIDGE_DETAIL_CAT_(a, b)

484/// @brief Suppresses the two registrars whose bodies reference a model's
485/// constructor/`execute()` definitions (`registerModelOnce`,
486/// `registerActionOnce`), for a pure client that dispatches every
487/// action to a remote peer and never constructs a model locally.
488///
489/// A model's constructor and `Model::execute` are ordinary functions the
490/// compiler emits a call to inside these registrars' lambda bodies, whether
491/// or not that lambda is ever invoked at runtime — so a plain client-only
492/// build still forces the linker to resolve them, pulling in implementations
493/// (a database driver, a native UI framework, an OS-specific API) the client
494/// target may have no link path for at all, and will never call regardless.
495/// The third registrar (`registerActionExecutorOnce`, `bridge.hpp`) routes
496/// through `Bridge::executeVia` -> `IBackend::execute` (fully abstract) and
497/// needs only the action's JSON codecs plus `Model::execute`'s *declaration*
498/// (for `ActionTraits::Result`, via `decltype`) — never its definition — so
499/// it is unaffected and always emitted.
500///
501/// Defined on the `morph` target's INTERFACE via the `MORPH_CLIENT_ONLY`
502/// CMake option, never per translation unit: two TUs disagreeing on whether a
503/// model registers itself would violate ODR. See docs/spec/core/registry.md,
504/// "MORPH_CLIENT_ONLY".
505///
506/// @warning NEVER define this for a process that hosts models (a server, or
507/// any `Bridge` running `LocalBackend`) — it silently registers nothing, and
508/// the model fails at runtime with "unknown model type" rather than at
509/// compile/link time.
510#ifdef MORPH_CLIENT_ONLY
511#define MORPH_DETAIL_REGISTER_MODEL_LOCAL(M, NAME)
512#define MORPH_DETAIL_REGISTER_ACTION_LOCAL(M, A, NAME)
513#else
514#define MORPH_DETAIL_REGISTER_MODEL_LOCAL(M, NAME) \
515 namespace { \
516 [[maybe_unused]] const bool BRIDGE_DETAIL_CAT(bridge_model_reg_, __COUNTER__) = \
517 morph::model::detail::registerModelOnce<M>(NAME); \
518 }
519#define MORPH_DETAIL_REGISTER_ACTION_LOCAL(M, A, NAME) \
520 namespace { \
521 [[maybe_unused]] const bool BRIDGE_DETAIL_CAT(bridge_action_reg_, __COUNTER__) = \
522 morph::model::detail::registerActionOnce<M, A>(morph::model::ModelTraits<M>::typeId(), NAME); \
523 }
524#endif

526/// @brief Registers model type @p M with the string id @p NAME.
527///
528/// Specialises `morph::model::ModelTraits<M>` and registers a factory with the
529/// process-level `ModelRegistryFactory` at static-init time — unless
530/// `MORPH_CLIENT_ONLY` is defined, in which case that registrar is suppressed
531/// (see `MORPH_DETAIL_REGISTER_MODEL_LOCAL`'s doc comment above).
532///
533/// @param M Concrete model type.
534/// @param NAME String literal used as the type-id.
535#define BRIDGE_REGISTER_MODEL(M, NAME) \
536 template <> \
537 struct morph::model::ModelTraits<M> { \
538 static constexpr std::string_view typeId() noexcept { return NAME; } \
539 }; \
540 MORPH_DETAIL_REGISTER_MODEL_LOCAL(M, NAME)

542/// @brief Registers action type @p A (for model @p M) with the string id @p NAME.
543///
544/// Specialises `morph::model::ActionTraits<A>` with JSON codec functions and
545/// registers the action with the process-level `ActionDispatcher` at static-init time.
546/// `BRIDGE_REGISTER_MODEL(M, ...)` must be called before this macro.
547///
548/// An optional 4th argument overrides whether executions of @p A are recorded to
549/// an attached action log (see `IModelHolder::attachActionLog`); omitted, it
550/// defaults to `morph::model::Loggable::Yes`, so every action is captured unless
551/// explicitly opted out — typically for pure queries:
552/// @code
553/// BRIDGE_REGISTER_ACTION(AccountModel, Deposit, "Deposit") // logged
554/// BRIDGE_REGISTER_ACTION(AccountModel, GetAccount, "GetAccount", morph::model::Loggable::No) // opt out
