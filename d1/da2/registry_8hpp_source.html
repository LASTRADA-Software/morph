37/// @brief Hash functor for `std::pair<std::string, std::string>` keys used by registries.
38/// Shared between `ActionDispatcher` (server-side dispatch) and `ActionExecuteRegistry`
39/// (client-side generic-execute) since their key types are structurally identical.
40struct PairKeyHash {
41 /// @brief Combines the hashes of `key.first` and `key.second`.
42 /// @param key The pair to hash.
43 /// @return The combined hash value.
44 [[nodiscard]] std::size_t operator()(const std::pair<std::string, std::string>& key) const noexcept {
45 std::size_t seed = std::hash<std::string>{}(key.first);
46 // NOLINTNEXTLINE(readability-magic-numbers, cppcoreguidelines-avoid-magic-numbers)
47 seed ^= std::hash<std::string>{}(key.second) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
48 return seed;
49 }
50};

52/// @brief Concept satisfied by actions that expose a `bool validate() const` member.
53///
54/// Drives the default `ActionValidator::ready(...)` path: if the action supplies
55/// a `validate()` method, it is called automatically; otherwise readiness defaults
56/// to `true` (matching the existing one-shot semantics).
57template <typename A>
58concept HasValidate = requires(const A& act) {
59 { act.validate() } -> std::convertible_to<bool>;
60};

64/// @brief Per-action validator used by `BridgeHandler::set<...>` to decide
65/// whether the in-progress draft is ready to execute.
66///
67/// Resolution order (highest-priority first):
68/// 1. **Explicit specialisation** of `ActionValidator<Action>` — typically via
69/// `BRIDGE_REGISTER_VALIDATOR(Action, fn)`. Wins over everything.
70/// 2. **`bool validate() const`** member on `Action` — picked up automatically.
71/// Co-locates the predicate with the data it inspects; preferred when you
72/// own the action type.
73/// 3. **Default** — returns `true`, matching one-shot semantics (the first
74/// `set<>` lands and the action fires).
75///
76/// Validation is intentionally a property of the **action**, not the model:
77/// different actions on the same model have different readiness requirements,
78/// and keeping the predicate next to the action keeps the GUI side oblivious
79/// to model internals.
80///
81/// @code
82/// struct FormAction {
83/// double a;
84/// double b;
85/// double c;
86/// bool validate() const { return a != 0.0 && b != 0.0 && c != 0.0; }
87/// };
88/// // No macro needed; ActionValidator<FormAction>::ready calls .validate() automatically.
89/// @endcode
90///
91/// @tparam Action Concrete action type.
92template <typename Action>
93struct ActionValidator {
94 /// @brief Returns `true` if the action is in a state that should be executed.
95 ///
96 /// Auto-detects a `bool validate() const` member on @p action via the
97 /// `morph::model::detail::HasValidate` concept. Falls back to `true`.
98 ///
99 /// @param action Draft action whose readiness is being checked.
100 /// @return `true` if the action should fire, `false` to keep collecting fields.
101 static constexpr bool ready(const Action& action) {
102 if constexpr (detail::HasValidate<Action>) {
103 return action.validate();
104 } else {
105 (void)action;
106 return true;
107 }
108 }
109};

111/// @brief Whether an action's executions are recorded to an attached action log.
112///
113/// A strong type instead of a bare `bool` so registration call sites read as
114/// intent (`Loggable::No`) rather than an unexplained `false`.
115enum class Loggable : std::uint8_t { No, Yes };

117/// @brief Per-action policy deciding how repeated executions are checkpointed
118/// into a durable action log.
119///
120/// Deliberately minimal for now — only `coalesce` exists, and it has no
121/// registration macro yet. Specialise directly for the rare action where only
122/// the latest occurrence should survive a checkpoint (e.g. a form-field edit
123/// fired repeatedly via `BridgeHandler::set<...>`); every other action defaults
124/// to `false`, meaning every execution is treated as a distinct, permanent fact
125/// (the right default for anything resembling a business event).
126/// @tparam Action Concrete action type.
127template <typename Action>
128struct ActionLogPolicy {
129 /// @brief If `true`, a checkpoint keeps only the most recent entry per
130 /// `(modelType, entityKey, actionType)`. If `false` (default), every
131 /// entry survives.
132 static constexpr bool coalesce = false;
133};

137/// @brief Concept satisfied by `ActionTraits<A>` specialisations that expose a
138/// `Loggable loggable` static member.
139///
140/// Lets `actionLoggable()` default to `Loggable::Yes` for hand-written
141/// `ActionTraits` specialisations (as used in tests) that predate this member,
142/// instead of requiring every existing specialisation to be updated.
143template <typename A>
144concept HasLoggableFlag = requires {
145 { ActionTraits<A>::loggable } -> std::convertible_to<Loggable>;
146};
147
148/// @brief Returns `ActionTraits<A>::loggable` if present, otherwise `Loggable::Yes`.
149///
150/// Logging every action by default (opt out via the macro's 4th argument) means
151/// new actions are captured automatically; only actions that are known to be
152/// pure queries need to opt out explicitly.
153template <typename A>
154constexpr Loggable actionLoggable() {
155 if constexpr (HasLoggableFlag<A>) {
156 return ActionTraits<A>::loggable;
157 } else {
158 return Loggable::Yes;
159 }
160}

171/// @brief Registry that maps (modelId, actionId) pairs to type-erased runner functions.
172///
173/// Used by `RemoteServer` to dispatch incoming JSON requests without knowing the
174/// concrete model or action types at the call site.
175class ActionDispatcher {
176public:
177 /// @brief Type-erased action runner: deserialises, executes, and serialises the result.
178 using Runner = std::function<std::string(IModelHolder&, std::string_view)>;
179
180 /// @brief Registers a runner for `(Model, Action)` under the given string ids.
181 ///
182 /// This is the single execution site used by `RemoteServer` (every remote and
183 /// Qt WebSocket topology) — `Model::execute()` runs here, on whichever process
184 /// actually owns @p holder. If a `journal::IActionLog` is attached to @p holder
185 /// (via `IModelHolder::attachActionLog`) and `Action` is loggable (the default),
186 /// the executed action is recorded automatically after it succeeds.
187 template <typename Model, typename Action>
188 void registerAction(std::string_view modelId, std::string_view actionId) {
189 Key const key{std::string{modelId}, std::string{actionId}};
190 _runners[key] = [](IModelHolder& holder, std::string_view payloadJson) {
191 auto action = ActionTraits<Action>::fromJson(payloadJson);
192 auto& model = holder.template into<Model>();
193 auto result = model.execute(action);
194 auto resultJson = ActionTraits<Action>::resultToJson(result);
195 if constexpr (detail::actionLoggable<Action>() == Loggable::Yes) {
196 if (holder.hasActionLog()) {
197 // entityKey/principal/timestampMs are filled in by recordIfAttached.
198 holder.recordIfAttached(::morph::journal::LogEntry{
199 .seq = 0,
200 .modelType = std::string{ModelTraits<Model>::typeId()},
201 .entityKey = {},
202 .actionType = std::string{ActionTraits<Action>::typeId()},
203 .payload = std::string{payloadJson},
204 .result = resultJson,
205 .principal = {},
206 .timestampMs = 0,
207 });
208 }
209 }
210 return resultJson;
211 };
212 _coalesce[key] = ActionLogPolicy<Action>::coalesce;
213 }
214
215 /// @brief Dispatches an action against @p holder and returns the JSON-encoded result.
216 std::string dispatch(std::string_view modelId, std::string_view actionId, IModelHolder& holder,
217 std::string_view payload) {
218 Key const key{std::string{modelId}, std::string{actionId}};
219 auto iter = _runners.find(key);
220 if (iter == _runners.end()) {
221 throw std::runtime_error("unknown action: " + key.first + "/" + key.second);
222 }
223 return iter->second(holder, payload);
224 }
225
226 /// @brief Returns whether `(modelId, actionId)` was registered with
227 /// `ActionLogPolicy<Action>::coalesce == true`.
228 ///
229 /// Used by `journal::SessionLog::checkpoint()` to decide, from the type-erased
230 /// `LogEntry` stream, which entries collapse to their latest occurrence.
231 /// Unknown pairs (never registered) default to `false` — every entry kept.
232 [[nodiscard]] bool coalesce(std::string_view modelId, std::string_view actionId) const {
233 auto iter = _coalesce.find(Key{std::string{modelId}, std::string{actionId}});
234 return iter != _coalesce.end() && iter->second;
235 }
236
237 /// @brief Returns the process-level singleton dispatcher.
238 static ActionDispatcher& instance() { return defaultDispatcher(); }
239
240private:
241 using Key = std::pair<std::string, std::string>;
242 std::unordered_map<Key, Runner, PairKeyHash> _runners;
243 std::unordered_map<Key, bool, PairKeyHash> _coalesce;
244};

246/// @brief Registry that creates `IModelHolder` instances by string type-id.
247///
248/// Used by `RemoteServer` to instantiate models on demand from incoming
249/// `"register"` messages.
250class ModelRegistryFactory {
251public:
252 /// @brief Registers a default-construction factory for `Model` under @p modelId.
253 template <typename Model>
254 void registerModel(std::string_view modelId) {
255 _factories.insert_or_assign(std::string{modelId}, [] { return ModelFactory::create<Model>(); });
256 }
257
258 /// @brief Creates a new model holder for the type registered under @p modelId.
259 std::unique_ptr<IModelHolder> create(std::string_view modelId) {
260 auto iter = _factories.find(std::string{modelId});
261 if (iter == _factories.end()) {
262 throw std::runtime_error("unknown model type: " + std::string{modelId});
263 }
264 return iter->second();
265 }
266
267 /// @brief Returns the process-level singleton registry.
268 static ModelRegistryFactory& instance() { return defaultRegistry(); }
269
270private:
271 std::unordered_map<std::string, std::function<std::unique_ptr<IModelHolder>()>> _factories;
272};
