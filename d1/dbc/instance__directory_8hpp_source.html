19/// @brief How far a freshly created shared instance has got through its first action.
20///
21/// A shared instance is created *before* the action that hydrates it runs, so
22/// there is a window in which the directory holds an instance nobody yet knows
23/// to be usable. `docs/spec/core/shared_instances.md`'s Failure modes section
24/// says what must happen when that first action fails: the instance must not be
25/// handed to a *new* attacher, and is evicted from the directory the next time
26/// anyone attaches to its key.
27enum class Hydration : std::uint8_t {
28 /// @brief The first action has not settled yet.
29 pending,
30 /// @brief The first action succeeded; the instance is shareable.
31 healthy,
32 /// @brief The first action failed; the instance must not be handed out again.
33 poisoned,
34};
35
36/// @brief The hydration phase of one shared instance, as a single atomic cell.
37///
38/// One cell rather than a `firstActionPending`/`poisoned` pair, because the pair
39/// cannot be moved between its states atomically: a reader landing between
40/// "first action is no longer pending" and "…and it failed" observes a
41/// not-poisoned instance whose first action has already failed, and is handed
42/// it — the exact case the spec forbids (morph#523).
43///
44/// Owned through a `shared_ptr` and captured that way — never via a raw pointer
45/// to the owning backend — into the strand task that settles it: that task may
46/// still be running after the backend itself is destroyed. See
47/// `LocalBackend::execute`'s capture-by-`shared_ptr` rationale.
48struct HydrationState {
49 /// @brief Current phase. Only ever moves away from `pending`, exactly once.
50 std::atomic<Hydration> phase{Hydration::pending};
51
52 /// @brief Records the first action's outcome, if it has not been recorded already.
53 ///
54 /// A single compare-exchange, so the transition out of `pending` is
55 /// indivisible: a concurrent `poisoned()` sees either the whole transition
56 /// or none of it. Later actions' outcomes are ignored — only the *first*
57 /// action decides hydration.
58 /// @param succeeded `true` if the first action succeeded.
59 void settle(bool succeeded) noexcept {
60 Hydration expected = Hydration::pending;
61 phase.compare_exchange_strong(expected, succeeded ? Hydration::healthy : Hydration::poisoned);
62 }
63
64 /// @brief Whether the first action has settled as a failure.
65 /// @return `true` only in the `poisoned` phase; `false` while still `pending`.
66 [[nodiscard]] bool poisoned() const noexcept { return phase.load() == Hydration::poisoned; }
67};

69/// @brief A shared instance's directory key: `(typeId, primary)`.
70using DirectoryKey = std::pair<std::string, std::string>;
71
72/// @brief Hash functor for `DirectoryKey`.
73///
74/// Structurally identical to `registry.hpp`'s `::morph::model::detail::PairKeyHash`,
75/// but defined here rather than reused so this header depends on nothing beyond
76/// `model.hpp`: `registry.hpp` pulls in glaze and the forms/schema stack, and the
77/// instance directory is part of the async core that must stay usable without it
78/// (morph#521).
79struct DirectoryKeyHash {
80 /// @brief Combines the hashes of the type id and the primary key.
81 /// @param key The directory key to hash.
82 /// @return The combined hash value.
83 [[nodiscard]] std::size_t operator()(const DirectoryKey& key) const noexcept {
84 std::size_t seed = std::hash<std::string>{}(key.first);
85 // NOLINTNEXTLINE(readability-magic-numbers, cppcoreguidelines-avoid-magic-numbers)
86 seed ^= std::hash<std::string>{}(key.second) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
87 return seed;
88 }
89};

91/// @brief Everything a backend knows about one live model instance.
92///
93/// One record per instance, rather than one entry per instance in each of
94/// several `ModelId`-keyed maps that had to be kept in lockstep by convention
95/// (morph#523). Every field is reached through a single hash lookup, and a
96/// half-updated instance is not representable.
97struct Instance {
98 /// @brief The model itself. Never null for a record that is in the directory.
99 std::shared_ptr<::morph::model::detail::IModelHolder> holder;
100
101 /// @brief Recorded owner principal, for `IAuthorizer::authorizeInstance`.
102 ///
103 /// Empty for an ownerless instance — which is every shared instance by
104 /// design (see `RemoteServer::acquireSharedInstance`) and every instance in
105 /// a `LocalBackend`, which has no authorizer.
106 std::string owner;
107
108 /// @brief Number of outstanding attachments, or `0` for a private instance.
109 ///
110 /// A private instance is not reference-counted at all: it has no directory
111 /// key, cannot be attached to, and is destroyed by the first `release()`.
112 /// A shared instance starts at `1` and is destroyed when it returns to `0`.
113 std::size_t attachCount = 0;
114
115 /// @brief The key this instance is filed under, or empty if it is private.
116 std::optional<DirectoryKey> sharedKey;
117
118 /// @brief First-action tracking, or null if this instance has none.
119 ///
120 /// Only an instance *created* by a shared register-or-attach carries one. A
121 /// private instance has nothing to hydrate, and an instance promoted into
122 /// the directory by `promote()` has already run the action that produced its
123 /// key, so neither can be poisoned.
124 ///
125 /// It therefore doubles as the permanent mark of "this instance was created
126 /// for a directory key", which outlives `sharedKey` — `attach()` clears that
127 /// on eviction, `hydration` survives — and is what makes an evicted instance
128 /// ineligible for `promote()`. See `promote()` for why that matters.
129 std::shared_ptr<HydrationState> hydration;
130};

132/// @brief The live model instances of one backend, plus the shared-instance directory.
133///
134/// Replaces the six (`LocalBackend`) and eight (`RemoteServer`) parallel
135/// `ModelId`-keyed containers those classes used to carry, and the
136/// register-or-attach logic that was written out twice (morph#523). The
137/// invariants it maintains are stated in `docs/spec/core/shared_instances.md`;
138/// this class is where they are now enforced rather than asserted in comments.
139///
140/// @par Locking
141/// **Caller-locked, deliberately.** Every operation below assumes the owning
142/// backend already holds the mutex that guards it. That mutex covers more than
143/// this class — `RemoteServer` mutates connection scopes and checks
144/// `LimitPolicy::maxLiveModels` in the same critical section — so a lock of its
145/// own would either be redundant or, worse, let those neighbouring decisions
146/// straddle a directory change. The one piece of state that *is* internally
147/// synchronised is `HydrationState`, because it is settled from a model strand
148/// that holds no backend lock at all.
149class InstanceDirectory {
150public:
151 /// @brief Outcome of `release()`.
152 enum class Release : std::uint8_t {
153 /// @brief Another attachment remains; the instance is still live.
154 retained,
155 /// @brief The last reference went away and the instance was destroyed.
156 destroyed,
157 };
158
159 /// @brief Looks up one instance, read-only.
160 ///
161 /// Read-only because every field a backend reads — holder, owner, hydration
162 /// state — is set when the instance is filed and changed only by the
163 /// mutators below. Nothing outside this class has business writing one.
164 /// @param mid Instance id.
165 /// @return Pointer to the record, or `nullptr` if @p mid is not live.
166 [[nodiscard]] const Instance* find(::morph::exec::detail::ModelId mid) const {
167 auto iter = _instances.find(mid);
168 return iter == _instances.end() ? nullptr : &iter->second;
169 }
170
171 /// @brief Number of live instances, shared and private alike.
172 /// @return The instance count, which is what `LimitPolicy::maxLiveModels` bounds.
173 [[nodiscard]] std::size_t size() const noexcept { return _instances.size(); }
174
175 /// @brief Records a private instance: no key, no sharing, no hydration tracking.
176 /// @param mid Freshly allocated id.
177 /// @param holder The constructed model.
178 /// @param owner Owner principal for `authorizeInstance`; empty if none.
179 void insertPrivate(::morph::exec::detail::ModelId mid,
180 std::shared_ptr<::morph::model::detail::IModelHolder> holder, std::string owner = {}) {
181 _instances.insert_or_assign(mid, Instance{.holder = std::move(holder), .owner = std::move(owner)});
182 }
183
184 /// @brief Records a freshly created shared instance and files it under @p key.
185 ///
186 /// The instance starts with one attachment and with hydration `pending`, so
187 /// the outcome of its first action decides whether it may be handed to a
188 /// second attacher.
189 /// @param mid Freshly allocated id.
190 /// @param holder The constructed model.
191 /// @param key Directory key to file it under; must not already be taken.
192 /// @param owner Owner principal for `authorizeInstance`; empty if none.
193 void insertShared(::morph::exec::detail::ModelId mid, std::shared_ptr<::morph::model::detail::IModelHolder> holder,
194 DirectoryKey key, std::string owner = {}) {
195 indexKey(key, mid);
196 _instances.insert_or_assign(mid, Instance{.holder = std::move(holder),
197 .owner = std::move(owner),
198 .attachCount = 1,
199 .sharedKey = std::move(key),
200 .hydration = std::make_shared<HydrationState>()});
201 }
202
203 /// @brief Takes one more attachment to the instance filed under @p key, if any.
204 ///
205 /// A poisoned instance reports as a *miss*: it is evicted from the directory
206 /// here — lazily, at the next attach, exactly as the spec's Failure modes
207 /// section specifies — and the caller goes on to create a fresh instance.
208 /// The evicted instance itself is untouched and still live; whoever created
209 /// it releases it normally.
210 /// @param key Directory key being acquired.
211 /// @return The attached instance's id, or `std::nullopt` on a miss or an eviction.
212 [[nodiscard]] std::optional<::morph::exec::detail::ModelId> attach(const DirectoryKey& key) {
213 auto keyIter = _byKey.find(key);
214 if (keyIter == _byKey.end()) {
215 return std::nullopt;
216 }
217 auto const mid = keyIter->second;
218 Instance& inst = _instances.at(mid);
219 if (inst.hydration && inst.hydration->poisoned()) {
220 unindexKey(key);
221 inst.sharedKey.reset();
222 return std::nullopt;
223 }
224 inst.attachCount += 1;
225 return mid;
226 }
227
228 /// @brief Files an already-live, still-anonymous instance under @p key.
229 ///
230 /// The promotion half of keyed instances: an action that creates its own
231 /// entity runs on an instance that has no key yet, and the key only exists
232 /// once the result comes back. The existing holder of a key always wins, and
233 /// an instance that already holds a real key never changes it — both are
234 /// silent no-ops, because instances never mutate their own identity.
235 ///
236 /// Only an instance that has **never** held a key can be promoted, and that
237 /// is a stronger test than "has no key right now". `attach()` clears the
238 /// `sharedKey` of an instance it evicts as poisoned, so an evicted instance
239 /// looks anonymous from the directory's side — but it was created *for* its
240 /// original key and told so once, permanently: `RemoteServer` builds its
241 /// holder through `ModelRegistryFactory::create(typeId, primary)`, which
242 /// calls `IModelHolder::attachIdentity(primary)`, and attaches the action
243 /// log under that same key. Nothing updates either afterwards, so re-filing
244 /// such an instance under a second key would hand every later attacher a
245 /// model that still identifies itself as the first one — the re-keying
246 /// `docs/spec/core/shared_instances.md` ("Re-pointing, not re-keying")
247 /// forbids outright. A non-null `hydration` is exactly the mark of
248 /// "created for a directory key", so it is what disqualifies the record.
249 ///
250 /// Its own Failure modes section already says what happens instead: the
251 /// handler that hit the failed first action "does not self-heal… it keeps
252 /// its broken instance until it releases and re-attaches from scratch".
253 /// The promotion is a silent no-op like the other three, the instance stays
254 /// anonymous and unshareable, and the key the host asked for is left free
255 /// for the next attacher to create a healthy instance under.
256 ///
257 /// A promoted instance therefore never carries a `HydrationState`: the
258 /// action that produced its key has already succeeded, so there is nothing
259 /// left to poison.
260 /// @param mid Live instance to promote.
261 /// @param key Directory key to file it under.
262 /// @return `true` if the promotion happened.
263 bool promote(::morph::exec::detail::ModelId mid, DirectoryKey key) {
264 auto instIter = _instances.find(mid);
265 if (instIter == _instances.end() || _byKey.contains(key)) {
266 return false;
267 }
268 Instance& inst = instIter->second;
269 if (inst.sharedKey.has_value() || inst.hydration) {
270 return false;
271 }
272 indexKey(key, mid);
273 inst.sharedKey = std::move(key);
274 // The instance was private until now — `insertPrivate` is the only way
275 // in for a record with no `hydration`, and it files one with no
276 // attachments at all. The directory now references it, so it gets the
277 // one attachment that reference stands for.
278 inst.attachCount = 1;
279 return true;
280 }
281
282 /// @brief Releases one attachment, destroying the instance at zero.
283 ///
284 /// A private instance (`attachCount == 0`) is destroyed outright. A shared
285 /// instance is unfiled from the directory in the same step that destroys it,
286 /// so directory membership can never outlive the instance it names.
287 /// @param mid Instance to release.
288 /// @return Whether this call destroyed the instance.
289 Release release(::morph::exec::detail::ModelId mid) {
290 auto instIter = _instances.find(mid);
291 if (instIter == _instances.end()) {
292 return Release::retained;
293 }
294 Instance& inst = instIter->second;
295 if (inst.attachCount > 0) {
296 inst.attachCount -= 1;
297 if (inst.attachCount > 0) {
298 return Release::retained;
299 }
300 if (inst.sharedKey) {
301 unindexKey(*inst.sharedKey);
302 }
303 }
304 _instances.erase(instIter);
305 return Release::destroyed;
306 }
307
308 /// @brief Primary keys of the live shared instances of @p typeId.
309 ///
310 /// Served from a per-type index rather than by scanning the whole directory,
311 /// so enumerating one model type does not cost the size of every other type
312 /// put together.
313 /// @param typeId String type-id to enumerate.
314 /// @return The canonical key strings, in unspecified order.
315 [[nodiscard]] std::vector<std::string> keysOfType(const std::string& typeId) const {
316 auto typeIter = _byType.find(typeId);
317 if (typeIter == _byType.end()) {
318 return {};
319 }
320 return {typeIter->second.begin(), typeIter->second.end()};
321 }
322
323private:
324 /// @brief Adds @p key to both directory indexes. The single place either is grown.
325 /// @param key Directory key being filed.
326 /// @param mid Instance it names.
327 void indexKey(const DirectoryKey& key, ::morph::exec::detail::ModelId mid) {
328 _byKey.emplace(key, mid);
329 _byType[key.first].insert(key.second);
330 }
331
332 /// @brief Removes @p key from both directory indexes. The single place either shrinks.
333 /// @param key Directory key being unfiled.
334 void unindexKey(const DirectoryKey& key) {
335 _byKey.erase(key);
336 // A guarded `find`, not `at`: the two indexes are only ever grown and
337 // shrunk together, here and in `indexKey`, so a missing type bucket is a
338 // broken invariant -- but this runs inside `release()`, which
339 // `RemoteServer::closeConnection` calls, which transports in turn call
340 // from a (noexcept) scope-guard destructor (`net/socket_server.hpp`). A
341 // `std::out_of_range` escaping there aborts the process rather than
342 // reporting anything, so the invariant is left to the tests to police.
343 if (auto typeIter = _byType.find(key.first); typeIter != _byType.end()) {
344 typeIter->second.erase(key.second);
345 if (typeIter->second.empty()) {
346 _byType.erase(typeIter);
347 }
348 }
349 }
350
351 /// @brief Every live instance, private and shared alike. The directory's only owner of state.
352 std::unordered_map<::morph::exec::detail::ModelId, Instance, ::morph::exec::detail::ModelIdHash> _instances;
353
354 /// @brief `(typeId, primary)` -> instance. An index over `_instances`, never an owner.
355 std::unordered_map<DirectoryKey, ::morph::exec::detail::ModelId, DirectoryKeyHash> _byKey;
356
357 /// @brief `typeId` -> that type's live primaries. An index over `_byKey`, for `keysOfType`.
358 std::unordered_map<std::string, std::unordered_set<std::string>> _byType;
359};
