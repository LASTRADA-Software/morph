18/// @brief The RFC 6455 §1.3 magic GUID appended to a client's key before hashing.
19inline constexpr std::string_view kWebSocketGuid = "258EAFA5-E914-47DA-95CA-C5AB0DC85B11";
20
21/// @brief Computes the `Sec-WebSocket-Accept` value for a given client key.
22/// @param clientKey The `Sec-WebSocket-Key` header value sent by the client.
23/// @return `base64(sha1(clientKey + kWebSocketGuid))`, per RFC 6455 §1.3.
24inline std::string computeAcceptKey(std::string_view clientKey) {
25 std::string concatenated{clientKey};
26 concatenated += kWebSocketGuid;
27 auto digest = sha1Digest(concatenated);
28 return base64Encode(digest);
29}

31/// @brief Generates a random 16-byte `Sec-WebSocket-Key`, base64-encoded.
32/// @return A 24-character base64 string suitable for the `Sec-WebSocket-Key` header.
33inline std::string generateClientKey() {
34 static thread_local std::mt19937 gen{std::random_device{}()};
35 std::uniform_int_distribution<int> dist{0, 255};
36 std::array<std::uint8_t, 16> raw{};
37 for (auto& b : raw) {
38 b = static_cast<std::uint8_t>(dist(gen));
39 }
40 return base64Encode(raw);
41}

43/// @brief The pieces of a `ws://` URL relevant to opening a socket + handshake.
44struct ParsedWsUrl {
45 /// @brief Hostname or numeric address to connect to.
46 std::string host;
47 /// @brief TCP port to connect to.
48 std::uint16_t port{0};
49 /// @brief HTTP request path (defaults to `/` when the URL has none).
50 std::string path;
51};
52
53/// @brief Parses a `ws://host:port[/path]` URL.
54/// @param url The URL to parse.
55/// @return The parsed host/port/path.
56/// @throws std::runtime_error if @p url is `wss://` (unsupported — see
57/// `docs/spec/core/backend.md`'s `morph::net` section), does not start
58/// with `ws://`, has no explicit port, or has an invalid port.
59inline ParsedWsUrl parseWsUrl(std::string_view url) {
60 constexpr std::string_view kWsScheme = "ws://";
61 constexpr std::string_view kWssScheme = "wss://";
62 if (url.substr(0, kWssScheme.size()) == kWssScheme) {
63 throw std::runtime_error(
64 "parseWsUrl: wss:// is not supported by morph::net (plaintext ws:// only; TLS is future work)");
65 }
66 if (url.substr(0, kWsScheme.size()) != kWsScheme) {
67 throw std::runtime_error("parseWsUrl: URL must start with ws://");
68 }
69 std::string_view rest = url.substr(kWsScheme.size());
70 auto slashPos = rest.find('/');
71 std::string_view hostPort = (slashPos == std::string_view::npos) ? rest : rest.substr(0, slashPos);
72 std::string path = (slashPos == std::string_view::npos) ? "/" : std::string(rest.substr(slashPos));
73 auto colonPos = hostPort.rfind(':');
74 if (colonPos == std::string_view::npos || colonPos == 0) {
75 throw std::runtime_error("parseWsUrl: URL must include an explicit host and port (e.g. ws://host:1234)");
76 }
77 std::string host{hostPort.substr(0, colonPos)};
78 std::string_view portStr = hostPort.substr(colonPos + 1);
79 if (portStr.empty()) {
80 throw std::runtime_error("parseWsUrl: invalid port in URL");
81 }
82 int portVal = 0;
83 for (char ch : portStr) {
84 if (ch < '0' || ch > '9') {
85 throw std::runtime_error("parseWsUrl: invalid port in URL");
86 }
87 portVal = (portVal * 10) + (ch - '0');
88 if (portVal > 65535) {
89 throw std::runtime_error("parseWsUrl: port out of range");
90 }
91 }
92 if (portVal == 0) {
93 throw std::runtime_error("parseWsUrl: invalid port in URL");
94 }
95 return ParsedWsUrl{std::move(host), static_cast<std::uint16_t>(portVal), std::move(path)};
96}

98/// @brief Builds the client's HTTP/1.1 Upgrade request.
99/// @param url The target host/port/path.
100/// @param key The `Sec-WebSocket-Key` to send (see `generateClientKey`).
101/// @return The full request, including the trailing blank-line terminator.
102inline std::string buildClientHandshakeRequest(const ParsedWsUrl& url, std::string_view key) {
103 std::string req;
104 req += "GET " + url.path + " HTTP/1.1\r\n";
105 // static_cast<unsigned>: see TcpSocket::connect — std::to_string has no
106 // uint16_t overload, and the promotion GCC would pick trips -Wsign-promo.
107 req += "Host: " + url.host + ":" + std::to_string(static_cast<unsigned>(url.port)) + "\r\n";
108 req += "Upgrade: websocket\r\n";
109 req += "Connection: Upgrade\r\n";
110 req += "Sec-WebSocket-Key: " + std::string(key) + "\r\n";
111 req += "Sec-WebSocket-Version: 13\r\n";
112 req += "\r\n";
113 return req;
114}

116/// @brief Builds the server's HTTP/1.1 101 Switching Protocols response.
117/// @param clientKey The `Sec-WebSocket-Key` value the client sent.
118/// @return The full response, including the trailing blank-line terminator.
119inline std::string buildServerHandshakeResponse(std::string_view clientKey) {
120 std::string resp;
121 resp += "HTTP/1.1 101 Switching Protocols\r\n";
122 resp += "Upgrade: websocket\r\n";
123 resp += "Connection: Upgrade\r\n";
124 resp += "Sec-WebSocket-Accept: " + computeAcceptKey(clientKey) + "\r\n";
125 resp += "\r\n";
126 return resp;
127}

131inline std::string toLowerAscii(std::string_view s) {
132 std::string out{s};
133 for (char& c : out) {
134 if (c >= 'A' && c <= 'Z') {
135 c = static_cast<char>(c - 'A' + 'a');
136 }
137 }
138 return out;
139}

141inline std::vector<std::string_view> splitLines(std::string_view raw) {
142 std::vector<std::string_view> lines;
143 std::size_t start = 0;
144 while (start < raw.size()) {
145 auto pos = raw.find("\r\n", start);
146 if (pos == std::string_view::npos) {
147 lines.push_back(raw.substr(start));
148 break;
149 }
150 lines.push_back(raw.substr(start, pos - start));
151 start = pos + 2;
152 }
153 return lines;
154}

156inline std::string_view trimLeadingSpace(std::string_view s) {
157 std::size_t i = 0;
158 while (i < s.size() && s[i] == ' ') {
159 ++i;
160 }
161 return s.substr(i);
162}

166/// @brief The pieces of a client handshake request relevant to completing it.
167struct ClientHandshakeRequest {
168 /// @brief The `Sec-WebSocket-Key` header value.
169 std::string key;
170 /// @brief The request-line path (e.g. `/`).
171 std::string path;
172};
173
174/// @brief Parses a client's HTTP/1.1 Upgrade request (header block only, no
175/// trailing blank line).
176/// @param headerBlock The request's header lines, joined by `\r\n`, with no
177/// trailing `\r\n\r\n` (see `readHttpHeaderBlock`, Task 6).
178/// @return The extracted key and path.
179/// @throws std::runtime_error if the request line is not a `GET`, or the
180/// `Sec-WebSocket-Key`/`Upgrade` headers are missing.
181inline ClientHandshakeRequest parseClientHandshakeRequest(std::string_view headerBlock) {
182 auto lines = ws_handshake_impl::splitLines(headerBlock);
183 if (lines.empty()) {
184 throw std::runtime_error("parseClientHandshakeRequest: empty request");
185 }
186 std::string_view requestLine = lines.front();
187 if (requestLine.substr(0, 4) != "GET ") {
188 throw std::runtime_error("parseClientHandshakeRequest: expected a GET request line");
189 }
190 std::size_t const pathStart = 4;
191 std::size_t const pathEnd = requestLine.find(' ', pathStart);
192 if (pathEnd == std::string_view::npos) {
193 throw std::runtime_error("parseClientHandshakeRequest: malformed request line");
194 }
195 std::string path{requestLine.substr(pathStart, pathEnd - pathStart)};
196
197 std::string key;
198 bool hasUpgrade = false;
199 for (std::size_t i = 1; i < lines.size(); ++i) {
200 std::string_view line = lines[i];
201 auto colon = line.find(':');
202 if (colon == std::string_view::npos) {
203 continue;
204 }
205 std::string name = ws_handshake_impl::toLowerAscii(line.substr(0, colon));
206 std::string_view value = ws_handshake_impl::trimLeadingSpace(line.substr(colon + 1));
207 if (name == "sec-websocket-key") {
208 key = std::string(value);
209 } else if (name == "upgrade") {
210 hasUpgrade = true;
211 }
212 }
213 if (key.empty()) {
214 throw std::runtime_error("parseClientHandshakeRequest: missing Sec-WebSocket-Key header");
215 }
216 if (!hasUpgrade) {
217 throw std::runtime_error("parseClientHandshakeRequest: missing Upgrade header");
218 }
219 return ClientHandshakeRequest{std::move(key), std::move(path)};
220}

222/// @brief Verifies the server's HTTP/1.1 101 response against the key the
223/// client sent.
224/// @param headerBlock The response's header lines, joined by `\r\n`, with no
225/// trailing `\r\n\r\n` (see `readHttpHeaderBlock`, Task 6).
226/// @param clientKey The `Sec-WebSocket-Key` this client sent in its request.
227/// @throws std::runtime_error if the status line is not `101`, or
228/// `Sec-WebSocket-Accept` does not match `computeAcceptKey(clientKey)`.
229inline void verifyServerHandshakeResponse(std::string_view headerBlock, std::string_view clientKey) {
230 auto lines = ws_handshake_impl::splitLines(headerBlock);
231 if (lines.empty()) {
232 throw std::runtime_error("verifyServerHandshakeResponse: empty response");
233 }
234 std::string_view statusLine = lines.front();
235 if (statusLine.find(" 101 ") == std::string_view::npos) {
236 throw std::runtime_error("verifyServerHandshakeResponse: server did not return 101 Switching Protocols: " +
237 std::string(statusLine));
238 }
239 std::string accept;
240 for (std::size_t i = 1; i < lines.size(); ++i) {
241 std::string_view line = lines[i];
242 auto colon = line.find(':');
243 if (colon == std::string_view::npos) {
244 continue;
245 }
246 std::string name = ws_handshake_impl::toLowerAscii(line.substr(0, colon));
247 if (name == "sec-websocket-accept") {
248 accept = std::string(ws_handshake_impl::trimLeadingSpace(line.substr(colon + 1)));
249 }
250 }
251 std::string expected = computeAcceptKey(clientKey);
252 if (accept != expected) {
253 throw std::runtime_error("verifyServerHandshakeResponse: Sec-WebSocket-Accept mismatch");
254 }
255}

257/// @brief The result of reading an HTTP header block up through `\r\n\r\n`.
258struct HandshakeReadResult {
259 /// @brief Header lines, joined by `\r\n`, **not** including the final `\r\n\r\n`.
260 std::string header;
261 /// @brief Any bytes read past the header terminator (belong to the first
262 /// WebSocket frame(s), if the peer pipelined them with the handshake).
263 std::string leftover;
264};
265
266/// @brief Reads bytes from @p socket until the `\r\n\r\n` header terminator.
267/// @param socket Connected socket to read from.
268/// @return The header text and any leftover bytes read past the terminator.
269/// @throws std::runtime_error if the peer closes before completing the header,
270/// or if the header exceeds a 64 KiB safety cap.
271inline HandshakeReadResult readHttpHeaderBlock(TcpSocket& socket) {
272 std::string buf;
273 char chunk[4096];
274 for (;;) {
275 auto pos = buf.find("\r\n\r\n");
276 if (pos != std::string::npos) {
277 HandshakeReadResult result;
278 result.header = buf.substr(0, pos);
279 result.leftover = buf.substr(pos + 4);
280 return result;
281 }
282 if (buf.size() > 64u * 1024u) {
283 throw std::runtime_error("readHttpHeaderBlock: header exceeds maximum size (64 KiB)");
284 }
285 std::size_t got = socket.recvSome(chunk, sizeof(chunk));
286 if (got == 0) {
287 throw std::runtime_error("readHttpHeaderBlock: peer closed connection during handshake");
288 }
289 buf.append(chunk, got);
290 }
291}

293/// @brief Performs the client side of the WebSocket handshake over @p socket.
294/// @param socket Already-connected socket.
295/// @param url Parsed target URL (host/port/path) to send in the request.
296/// @return Leftover bytes read past the response header — feed these into a
297/// `WsFrameReader` before reading further from the socket.
298/// @throws std::runtime_error if the server's response is not a valid 101
299/// Switching Protocols with a matching `Sec-WebSocket-Accept`.
300inline std::string performClientHandshake(TcpSocket& socket, const ParsedWsUrl& url) {
301 std::string key = generateClientKey();
302 std::string request = buildClientHandshakeRequest(url, key);
303 socket.sendAll(request.data(), request.size());
304 HandshakeReadResult result = readHttpHeaderBlock(socket);
305 verifyServerHandshakeResponse(result.header, key);
306 return result.leftover;
307}
308
309/// @brief Performs the server side of the WebSocket handshake over @p socket.
310/// @param socket Freshly accepted socket.
311/// @return Leftover bytes read past the request header — feed these into a
312/// `WsFrameReader` before reading further from the socket.
313/// @throws std::runtime_error if the client's request is malformed or missing
314/// the required headers.
315inline std::string performServerHandshake(TcpSocket& socket) {
316 HandshakeReadResult result = readHttpHeaderBlock(socket);
317 ClientHandshakeRequest req = parseClientHandshakeRequest(result.header);
318 std::string response = buildServerHandshakeResponse(req.key);
319 socket.sendAll(response.data(), response.size());
320 return result.leftover;
321}
