293/// @brief The first letter of a word: its first code point, upper-cased where it is ASCII.
294/// @param word A non-empty word, UTF-8.
295/// @return The letter.
296[[nodiscard]] inline std::string firstLetter(std::string_view word) {
297 // The whole first code point: a lead byte says how many bytes follow it.
298 auto const lead = static_cast<unsigned char>(word.front());
299 std::size_t const length = lead >= 0xF0 ? 4 : lead >= 0xE0 ? 3 : lead >= 0xC0 ? 2 : 1;
300 std::string letter{word.substr(0, length)};
301 if (letter.size() == 1 && letter.front() >= 'a' && letter.front() <= 'z') {
302 letter.front() = static_cast<char>(letter.front() - 'a' + 'A');
303 }
304 return letter;
305}

309/// @brief The initials of a person's name: the first letter of its first and of its last word, upper-cased where the
310/// letter is ASCII. A word that ends in a full stop (a title such as `Dr.` or an abbreviation) is not a name
311/// and is skipped, unless every word is one. A one-word name has one initial.
312/// @param name The name, UTF-8.
313/// @return The initials; empty for a blank name.
314[[nodiscard]] inline std::string shellInitials(std::string_view name) {
315 std::vector<std::string_view> words;
316 for (std::size_t at = 0; at < name.size();) {
317 while (at < name.size() && (name[at] == ' ' || name[at] == '\t')) {
318 ++at;
319 }
320 std::size_t const start = at;
321 while (at < name.size() && name[at] != ' ' && name[at] != '\t') {
322 ++at;
323 }
324 if (at > start) {
325 words.push_back(name.substr(start, at - start));
326 }
327 }
328 std::vector<std::string_view> names;
329 for (std::string_view const word : words) {
330 if (!word.ends_with('.')) {
331 names.push_back(word);
332 }
333 }
334 if (names.empty()) {
335 names = words;
336 }
337 if (names.empty()) {
338 return {};
339 }
340 return names.size() == 1 ? detail::firstLetter(names.front())
341 : detail::firstLetter(names.front()) + detail::firstLetter(names.back());
342}
