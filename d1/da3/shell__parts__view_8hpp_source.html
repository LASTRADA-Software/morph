340/// @brief The first letter of a word: its first code point, upper-cased where it is ASCII.
341/// @param word A non-empty word, UTF-8.
342/// @return The letter.
343[[nodiscard]] inline std::string firstLetter(std::string_view word) {
344 // The whole first code point: a lead byte says how many bytes follow it.
345 auto const lead = static_cast<unsigned char>(word.front());
346 std::size_t const length = lead >= 0xF0 ? 4 : lead >= 0xE0 ? 3 : lead >= 0xC0 ? 2 : 1;
347 std::string letter{word.substr(0, length)};
348 if (letter.size() == 1 && letter.front() >= 'a' && letter.front() <= 'z') {
349 letter.front() = static_cast<char>(letter.front() - 'a' + 'A');
350 }
351 return letter;
352}

356/// @brief The initials of a person's name: the first letter of its first and of its last word, upper-cased where the
357/// letter is ASCII. A word that ends in a full stop (a title such as `Dr.` or an abbreviation) is not a name
358/// and is skipped, unless every word is one. A one-word name has one initial.
359/// @param name The name, UTF-8.
360/// @return The initials; empty for a blank name.
361[[nodiscard]] inline std::string shellInitials(std::string_view name) {
362 std::vector<std::string_view> words;
363 for (std::size_t at = 0; at < name.size();) {
364 while (at < name.size() && (name[at] == ' ' || name[at] == '\t')) {
365 ++at;
366 }
367 std::size_t const start = at;
368 while (at < name.size() && name[at] != ' ' && name[at] != '\t') {
369 ++at;
370 }
371 if (at > start) {
372 words.push_back(name.substr(start, at - start));
373 }
374 }
375 std::vector<std::string_view> names;
376 for (std::string_view const word : words) {
377 if (!word.ends_with('.')) {
378 names.push_back(word);
379 }
380 }
381 if (names.empty()) {
382 names = words;
383 }
384 if (names.empty()) {
385 return {};
386 }
387 return names.size() == 1 ? detail::firstLetter(names.front())
388 : detail::firstLetter(names.front()) + detail::firstLetter(names.back());
389}
