38/// @brief One locale's messages, by key, with the patterns already parsed.
39struct Catalogue {
40 /// @brief The pattern text of each key whose pattern is in the `ui/1` subset.
41 std::map<std::string, std::string, std::less<>> messages;
42 /// @brief One line per key whose pattern is outside the subset; those keys show their fallback.
43 std::vector<std::string> dropped;
44};

46/// @brief Reads the locale from a catalogue's id.
47/// @param fileId The id, such as `i18n/de-AT.json`.
48/// @return The locale (`de-AT`), or nothing when the id is not `i18n/LOCALE.json` with a locale of ASCII letters,
49/// digits and `-` that starts with a letter.
50[[nodiscard]] inline std::optional<std::string> catalogueLocale(std::string_view fileId) {
51 if (!fileId.starts_with(kCataloguePrefix) || !fileId.ends_with(kCatalogueSuffix) ||
52 fileId.size() <= kCataloguePrefix.size() + kCatalogueSuffix.size()) {
53 return std::nullopt;
54 }
55 std::string_view const locale =
56 fileId.substr(kCataloguePrefix.size(), fileId.size() - kCataloguePrefix.size() - kCatalogueSuffix.size());
57 auto const letter = [](char const symbol) {
58 return (symbol >= 'a' && symbol <= 'z') || (symbol >= 'A' && symbol <= 'Z');
59 };
60 bool const valid = letter(locale.front()) && std::ranges::all_of(locale, [&letter](char const symbol) {
61 return letter(symbol) || (symbol >= '0' && symbol <= '9') || symbol == '-';
62 });
63 return valid ? std::optional<std::string>{std::string{locale}} : std::nullopt;
64}

66/// @brief Parses a catalogue: a flat JSON object of message key to MessageFormat 2 pattern.
67///
68/// Refused: text that is not JSON or nests deeper than `kMaxJsonDepth`, anything but an object, a must-understand
69/// (`!`) key, a value that is not a string, an empty key or one longer than `kMaxNameBytes`, and more than
70/// `kMaxCatalogueKeys` keys. A pattern outside the `ui/1` subset does not refuse the catalogue: its key is left out
71/// and listed in `dropped`, so `t` shows the fallback for it.
72/// @param text The catalogue's JSON text.
73/// @return The catalogue, or the reason it is refused.
74[[nodiscard]] inline std::expected<Catalogue, std::string> parseCatalogue(std::string_view text) {
75 auto const json = parseJson(text);
76 if (!json) {
77 return std::unexpected(json.error());
78 }
79 auto const* const object = std::get_if<Json::object_t>(&json->data);
80 if (object == nullptr) {
81 return std::unexpected(std::string{"a catalogue is an object of message keys"});
82 }
83 if (auto reason = detail::mustUnderstand(*json, "the catalogue")) {
84 return std::unexpected(std::move(*reason));
85 }
86 if (object->size() > kMaxCatalogueKeys) {
87 return std::unexpected("a catalogue holds more than " + std::to_string(kMaxCatalogueKeys) + " keys");
88 }
89 Catalogue catalogue;
90 for (auto const& [key, value] : *object) {
91 if (key.empty() || key.size() > kMaxNameBytes) {
92 return std::unexpected("a message key is 1 to " + std::to_string(kMaxNameBytes) + " bytes");
93 }
94 auto const* const pattern = std::get_if<std::string>(&value.data);
95 if (pattern == nullptr) {
96 return std::unexpected("the message `" + key + "` is not a string");
97 }
98 if (auto parsed = parseMessage(*pattern); !parsed) {
99 catalogue.dropped.push_back("`" + key + "`: " + parsed.error());
100 continue;
101 }
102 catalogue.messages.emplace(key, *pattern);
103 }
104 return catalogue;
105}

107/// @brief The message catalogues of one bundle, by locale, and the provider `t` resolves through.
108class MessageCatalogues {
109public:
110 /// @brief Adds, or replaces, a locale's catalogue.
111 /// @param locale The locale.
112 /// @param catalogue Its messages.
113 void add(std::string locale, Catalogue catalogue) {
114 auto next = std::make_shared<Table>(*_table);
115 next->insert_or_assign(std::move(locale), std::move(catalogue.messages));
116 _table = std::move(next);
117 }
118
119 /// @brief The locales with a catalogue, sorted.
120 /// @return They.
121 [[nodiscard]] std::vector<std::string> locales() const {
122 std::vector<std::string> out;
123 out.reserve(_table->size());
124 for (auto const& [locale, messages] : *_table) {
125 static_cast<void>(messages);
126 out.push_back(locale);
127 }
128 return out;
129 }
130
131 /// @brief What went wrong while loading: a catalogue left out, or a pattern dropped. Each line names the file.
132 /// @return The lines.
133 [[nodiscard]] std::vector<std::string> const& problems() const noexcept { return _problems; }
134
135 /// @brief Records a loading problem.
136 /// @param line What went wrong, naming the file.
137 void note(std::string line) { _problems.push_back(std::move(line)); }
138
139 /// @brief A message's pattern in a locale: the exact locale's catalogue, then its language's (`de-AT`, then
140 /// `de`).
141 /// @param key The message key.
142 /// @param locale The BCP 47 locale.
143 /// @return The pattern, or nothing; `t` then shows the fallback.
144 [[nodiscard]] std::optional<std::string> lookup(std::string_view key, std::string_view locale) const {
145 return find(*_table, key, locale);
146 }
147
148 /// @brief A provider over the catalogues as they are now. It shares them, so it stays valid after this object is
149 /// gone, and does not see a later `add`.
150 /// @return The provider, for `FormatContext::translations` (`Session::setTranslations`).
151 [[nodiscard]] render::TranslationProvider provider() const {
152 return [table = _table](std::string_view key, std::string_view locale) { return find(*table, key, locale); };
153 }
154
155private:
156 using Table = std::map<std::string, std::map<std::string, std::string, std::less<>>, std::less<>>;
157
158 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): key, then locale, as TranslationProvider takes them.
159 [[nodiscard]] static std::optional<std::string> find(Table const& table, std::string_view key,
160 std::string_view locale) {
161 std::string_view tag = locale;
162 while (true) {
163 if (auto const messages = table.find(tag); messages != table.end()) {
164 if (auto const message = messages->second.find(key); message != messages->second.end()) {
165 return message->second;
166 }
167 }
168 auto const dash = tag.rfind('-');
169 if (dash == std::string_view::npos) {
170 return std::nullopt;
171 }
172 tag = tag.substr(0, dash);
173 }
174 }
175
176 std::shared_ptr<Table const> _table = std::make_shared<Table const>();
177 std::vector<std::string> _problems;
178};

180/// @brief Reads every message catalogue a verified bundle lists.
181///
182/// A catalogue whose id is not `i18n/LOCALE.json`, whose bytes are not among @p files, or that `parseCatalogue`
183/// refuses, is left out, and its locale shows the fallbacks; a pattern outside the subset is dropped. Each is noted
184/// in `problems()`. Nothing is refused later, at a `t`.
185/// @param manifest The bundle's manifest (`LoadedBundle::manifest`).
186/// @param files The verified files by content hash (`LoadedBundle::files`).
187/// @return The catalogues.
188[[nodiscard]] inline MessageCatalogues loadCatalogues(Manifest const& manifest,
189 std::map<std::string, std::string, std::less<>> const& files) {
190 MessageCatalogues catalogues;
191 for (ManifestFile const& file : manifest.files) {
192 if (file.kind != kMessagesFile) {
193 continue;
194 }
195 auto locale = catalogueLocale(file.id);
196 if (!locale) {
197 catalogues.note("`" + file.id + "`: a catalogue's id is i18n/<locale>.json");
198 continue;
199 }
200 auto const bytes = files.find(file.hash);
201 if (bytes == files.end()) {
202 catalogues.note("`" + file.id + "`: not among the verified files");
203 continue;
204 }
205 auto catalogue = parseCatalogue(bytes->second);
206 if (!catalogue) {
207 catalogues.note("`" + file.id + "`: " + catalogue.error());
208 continue;
209 }
210 for (std::string const& dropped : catalogue->dropped) {
211 catalogues.note("`" + file.id + "`: " + dropped);
212 }
213 catalogues.add(std::move(*locale), std::move(*catalogue));
214 }
215 return catalogues;
216}

218/// @brief A message formatted in every locale the catalogues hold: what a search over titles compares against.
219/// @param catalogues The catalogues.
220/// @param key The message key.
221/// @param fallback The source text, shown for a locale without the key.
222/// @return One text per locale, in the order of `locales()`; the fallback alone when there are no catalogues.
223// NOLINTBEGIN(bugprone-easily-swappable-parameters): key, then fallback, as `t` takes them.
224[[nodiscard]] inline std::vector<std::string> textInEveryLocale(MessageCatalogues const& catalogues,
225 std::string_view key, std::string_view fallback) {
226 // NOLINTEND(bugprone-easily-swappable-parameters)
227 std::vector<std::string> const locales = catalogues.locales();
228 std::vector<std::string> out;
229 auto const format = [](std::string_view pattern, std::string const& locale) -> std::optional<std::string> {
230 auto parsed = parseMessage(pattern);
231 if (!parsed) {
232 return std::nullopt;
233 }
234 return formatMessage(*parsed, ValueRecord{}, MessageLocale{.locale = locale, .numeric = {}}).text;
235 };
236 if (locales.empty()) {
237 out.push_back(format(fallback, "en").value_or(std::string{fallback}));
238 return out;
239 }
240 out.reserve(locales.size());
241 for (std::string const& locale : locales) {
242 std::optional<std::string> text;
243 if (auto const pattern = catalogues.lookup(key, locale)) {
244 text = format(*pattern, locale);
245 }
246 if (!text) {
247 text = format(fallback, locale);
248 }
249 out.push_back(text.value_or(std::string{fallback}));
250 }
251 return out;
252}

254/// @brief Whether a message contains @p query in any locale the catalogues hold, or in its fallback: a navigator's
255/// search, so a user finds a screen by the title they know in any language.
256///
257/// ASCII letters compare without case; every other byte compares exactly.
258/// @param catalogues The catalogues.
259/// @param key The message key.
260/// @param fallback The source text.
261/// @param query What the user typed; an empty query matches.
262/// @return Whether any locale's text contains it.
263// NOLINTBEGIN(bugprone-easily-swappable-parameters): key, fallback and query, as `t` and a search box give them.
264[[nodiscard]] inline bool matchesInAnyLocale(MessageCatalogues const& catalogues, std::string_view key,
265 std::string_view fallback, std::string_view query) {
266 // NOLINTEND(bugprone-easily-swappable-parameters)
267 auto const fold = [](std::string_view text) {
268 std::string out{text};
269 std::ranges::transform(out, out.begin(), [](char const symbol) {
270 return symbol >= 'A' && symbol <= 'Z' ? static_cast<char>(symbol - 'A' + 'a') : symbol;
271 });
272 return out;
273 };
274 std::string const needle = fold(query);
275 std::vector<std::string> texts = textInEveryLocale(catalogues, key, fallback);
276 texts.emplace_back(fallback);
277 return std::ranges::any_of(texts, [&](std::string const& text) { return fold(text).contains(needle); });
278}

282/// @brief A catalog title's parts: a plain string, or the message key of a `{"t": …}` reference and its fallback.
283struct TitleParts {
284 std::optional<std::string> text; ///< A plain string title.
285 std::optional<std::string> key; ///< A message reference's key.
286 std::optional<std::string> fallback; ///< A message reference's fallback.
287};
288
289/// @brief Splits a catalog title into its parts.
290/// @param title The title.
291/// @return Its parts; all empty for a title that is neither a string nor an object.
292[[nodiscard]] inline TitleParts titleParts(Json const& title) {
293 TitleParts parts;
294 if (std::string const* const text = std::get_if<std::string>(&title.data)) {
295 parts.text = *text;
296 return parts;
297 }
298 if (auto const* const object = std::get_if<Json::object_t>(&title.data)) {
299 for (auto const& [name, value] : *object) {
300 std::string const* const text = std::get_if<std::string>(&value.data);
301 if (text == nullptr) {
302 continue;
303 }
304 if (name == "t") {
305 parts.key = *text;
306 } else if (name == "fallback") {
307 parts.fallback = *text;
308 }
309 }
310 }
311 return parts;
312}

328/// @brief A catalog entry's title in one locale, as a shell's tab or header shows it.
329///
330/// A string title is shown as is. A message reference (`{"t": key, "fallback": …}`) is formatted in @p locale from
331/// @p catalogues — the exact locale's catalogue, then its language's — and is its fallback when no catalogue has the
332/// key. A title that is neither, or a reference with no fallback that no catalogue resolves, is the entry's id.
333/// @param entry The entry.
334/// @param catalogues The message catalogues, or null for none.
335/// @param locale The BCP 47 locale to format in.
336/// @return The title.
337[[nodiscard]] inline std::string catalogTitle(CatalogEntry const& entry, MessageCatalogues const* catalogues,
338 std::string_view locale) {
339 detail::catalogue::TitleParts const parts = detail::catalogue::titleParts(entry.title);
340 if (parts.text) {
341 return *parts.text;
342 }
343 std::string const where{locale};
344 if (parts.key && catalogues != nullptr) {
345 if (auto const pattern = catalogues->lookup(*parts.key, where)) {
346 if (auto text = detail::catalogue::formatPattern(*pattern, where)) {
347 return std::move(*text);
348 }
349 }
350 }
351 if (parts.fallback) {
352 return detail::catalogue::formatPattern(*parts.fallback, where).value_or(*parts.fallback);
353 }
354 return entry.id;
355}

357/// @brief A catalog entry's title in every locale the catalogues hold: what a search over screen titles compares
358/// against.
359///
360/// With @p catalogues, a message reference is `textInEveryLocale` of its key and fallback (the entry's id when it
361/// has none). Without, the one title is a string title as is, else a reference's fallback, else the entry's id.
362/// @param entry The entry.
363/// @param catalogues The message catalogues, or null for none.
364/// @return The titles; never empty.
365[[nodiscard]] inline std::vector<std::string> catalogTitles(CatalogEntry const& entry,
366 MessageCatalogues const* catalogues) {
367 detail::catalogue::TitleParts const parts = detail::catalogue::titleParts(entry.title);
368 if (parts.text) {
369 return {*parts.text};
370 }
371 if (parts.key && catalogues != nullptr) {
372 return textInEveryLocale(*catalogues, *parts.key, parts.fallback.value_or(entry.id));
373 }
374 return {parts.fallback.value_or(entry.id)};
375}
