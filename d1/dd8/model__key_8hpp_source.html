13/// @file
14/// Primary keys for model instances.
15///
16/// A model type declares itself *keyed* by exposing a nested `PrimaryKey` type
17/// alias; the alias is detected structurally (a `requires`-expression), never by
18/// inheritance or a marker base — the same detection style
19/// `morph::views::ViewTraits` uses for a view's `kind`/`query` members. A model
20/// without the alias is unkeyed and behaves exactly as it always has.
21///
22/// Actions say *which of their fields carries* that key via `BRIDGE_KEY_FROM`,
23/// or — for an action that creates the entity rather than naming it — which
24/// field of their *result* establishes it, via `BRIDGE_KEY_FROM_RESULT`. Actions
25/// with neither declaration are keyless and run against whichever instance the
26/// handler is already attached to; that is the common case.
27///
28/// Keys travel the wire as strings (`wire::Envelope::primary`) regardless of
29/// their C++ type, so the directory in `RemoteServer` needs exactly one map type
30/// rather than one per key type. See docs/planned/shared_model_instances.md.
31
32namespace morph::model {
33
34/// @brief Key types that are themselves a wire-encodable scalar.
35///
36/// Restricted to integral types and `std::string` because a key must round-trip
37/// losslessly through the wire's string encoding and be usable as a map key.
38/// `bool` is excluded: it carries one bit of identity, which is never a
39/// meaningful primary key and is far more likely to be a mistake.
40template <typename K>
41concept RawModelKey =
42 (std::integral<K> && !std::same_as<std::remove_cv_t<K>, bool>) || std::same_as<std::remove_cv_t<K>, std::string>;
43
44/// @brief Key types that *wrap* a `RawModelKey` — the ladder's strong ids.
45///
46/// `examples/IMPLEMENTATION.md` rule 3 requires entity identity to be a
47/// per-entity strong id type (`struct PasteId`, `struct OptionId`) exposing
48/// `hasValue()`, so that it joins the forms palette as an empty-capable field.
49/// Such a type is not integral and is not `std::string`, so before this it
50/// satisfied no arm of `ModelKey` at all — and a rung obeying rule 3 could not
51/// use `BRIDGE_MODEL_KEY`/`BRIDGE_KEY_FROM`, the very macros the framework
52/// provides for keying. Three rungs hand-wrote the traits instead, each
53/// re-stating the `*id` unwrapping convention the macro exists to hide.
54///
55/// The shape is deduced structurally rather than declared, matching how a
56/// keyed model is detected: any type with `hasValue()` and an `operator*`
57/// yielding a `RawModelKey`, and constructible back from that raw value so the
58/// key round-trips from the wire. Both ladder spellings qualify — an
59/// `std::optional`-backed id whose `operator*` returns a reference, and a
60/// sentinel-backed one (`0` means empty) whose `operator*` returns by value.
61template <typename K>
62concept WrappedModelKey = !RawModelKey<K> && requires(const K& key) {
63 { key.hasValue() } -> std::convertible_to<bool>;
64 { *key };
65 requires RawModelKey<std::remove_cvref_t<decltype(*key)>>;
66 requires requires(std::remove_cvref_t<decltype(*key)> raw) { K{std::move(raw)}; };
67};
68
69/// @brief Key types a model may declare as its `PrimaryKey`.
70///
71/// Either a wire-encodable scalar (`RawModelKey`) or a strong id wrapping one
72/// (`WrappedModelKey`).
73template <typename K>
74concept ModelKey = RawModelKey<K> || WrappedModelKey<K>;

85/// @brief The key type of a model, when one has been declared *for* it.
86///
87/// Specialised by `BRIDGE_KEY_FROM`, which deduces the type from the action
88/// member it is given — so a model does not have to say anything about keys
89/// inside its own class. The primary template is deliberately empty: a model
90/// with neither this specialisation nor a nested alias is simply unkeyed.
91/// @tparam Model Concrete model type.
92template <typename Model>
93struct ModelKeyTraits {};

97/// @brief Satisfied by a model that names its own key with a nested alias.
98template <typename M>
99concept SelfDeclaredKey = requires { typename M::PrimaryKey; } && ModelKey<typename M::PrimaryKey>;

101/// @brief Satisfied by a model whose key was deduced from a keyed action.
102template <typename M>
103concept DeducedKey =
104 requires { typename ModelKeyTraits<M>::PrimaryKey; } && ModelKey<typename ModelKeyTraits<M>::PrimaryKey>;

108/// @brief Satisfied by model types that have a primary key, however it was named.
109///
110/// Two ways in, and neither requires touching the model's own class body beyond
111/// the first: a nested `PrimaryKey` alias, or a `BRIDGE_KEY_FROM` declaration
112/// that deduces the type from the action field carrying it. Following
113/// `morph::forms`' standing rule — *infer by default, declare to override* — a
114/// nested alias wins when both are present, which is what lets a model whose
115/// key type differs from the field's type (an `int` column keyed as a
116/// `std::string`, say) state that explicitly.
117template <typename M>
118concept KeyedModel = detail::SelfDeclaredKey<M> || detail::DeducedKey<M>;

142/// @brief Encodes a primary key as its canonical wire string.
143///
144/// Integral keys are decimal; `std::string` keys pass through unchanged; a
145/// strong id encodes as whatever it wraps, so it shares a directory entry with
146/// the raw key of the same value. The encoding is total for every *engaged*
147/// key — each has exactly one representation — so two clients naming the same
148/// key always land on the same directory entry.
149/// @tparam K Key type satisfying `ModelKey`.
150/// @param key Key value to encode.
151/// @return The canonical string form of @p key.
152/// @throws std::runtime_error if @p key is a strong id with no value. Such an
153/// id names no instance, and encoding it as `""` or `"0"` would route
154/// every caller holding an unset id to a single shared instance.
155template <ModelKey K>
156[[nodiscard]] std::string keyToString(const K& key) {
157 if constexpr (WrappedModelKey<K>) {
158 // An empty strong id names no instance. Unwrapping it is undefined
159 // behaviour (`operator*` on a disengaged optional), and encoding it as
160 // "" or "0" would silently route every caller holding an unset id to
161 // one shared instance -- the worst possible failure, because it looks
162 // like it worked. Both call sites in `BridgeHandler` turn a throw here
163 // into a rejected `Completion`, so this surfaces as an error the
164 // caller can see.
165 if (!key.hasValue()) {
166 throw std::runtime_error("primary key is empty: a strong id with no value names no model instance");
167 }
168 return keyToString(*key);
169 } else if constexpr (std::same_as<std::remove_cv_t<K>, std::string>) {
170 return key;
171 } else {
172 return std::to_string(key);
173 }
174}

176/// @brief Decodes a primary key from its canonical wire string.
177///
178/// A strong id is reconstructed by decoding what it wraps and rewrapping it,
179/// so `keyFromString<PasteId>(keyToString(id)) == id` for any engaged `id`.
180/// @tparam K Key type satisfying `ModelKey`.
181/// @param text Canonical string form, as produced by `keyToString`.
182/// @return The decoded key.
183/// @throws std::runtime_error if @p text is not a valid encoding of a `K`
184/// (non-numeric text, trailing garbage, or a value out of range). A key
185/// that cannot be decoded is a protocol error, not a value to clamp:
186/// silently yielding 0 would route the caller to the wrong instance.
187template <ModelKey K>
188[[nodiscard]] K keyFromString(std::string_view text) {
189 if constexpr (WrappedModelKey<K>) {
190 return K{keyFromString<detail::UnwrappedKeyOf<K>>(text)};
191 } else if constexpr (std::same_as<std::remove_cv_t<K>, std::string>) {
192 return std::string{text};
193 } else {
194 K value{};
195 // from_chars is a [first, last) pointer API; a string_view's data()+size()
196 // is the only way to express its end, and is exactly what the standard
197 // intends here.
198 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-pointer-arithmetic)
199 auto [ptr, errc] = std::from_chars(text.data(), text.data() + text.size(), value);
200 const auto* const last = text.data() + text.size(); // NOLINT(cppcoreguidelines-pro-bounds-pointer-arithmetic)
201 if (errc != std::errc{} || ptr != last) {
202 throw std::runtime_error("invalid primary key encoding: '" + std::string{text} + "'");
203 }
204 return value;
205 }
206}

208/// @brief Declares where an action's model key comes from.
209///
210/// The primary template is the *keyless* case, which is the default and the
211/// common one: such an action says nothing about identity and runs against
212/// whichever instance its handler is already attached to. Specialise via
213/// `BRIDGE_KEY_FROM` or `BRIDGE_KEY_FROM_RESULT` rather than by hand.
214/// @tparam Action Concrete action type.
215template <typename Action>
216struct ActionKeyTraits {
217 /// @brief Whether this action carries or establishes its model's key.
218 static constexpr bool hasKey = false;
219
220 /// @brief Whether the key comes from the action's result rather than its payload.
221 static constexpr bool fromResult = false;
222};

259/// @brief Declares that action `A` is the one that defines model `M`'s primary key.
260///
261/// One line does both jobs, so the model's own class body needs to say nothing
262/// about keys: the key *type* is deduced from `MEMBER`'s type (defining
263/// `ModelKeyTraits<M>`), and `A` is recorded as an action that carries it
264/// (defining `ActionKeyTraits<A>`).
265///
266/// Executing such an action on a shareable handler attaches — or re-points —
267/// that handler to the instance holding the named key, constructing one only if
268/// no instance holds it yet. Every *keyless* action on that handler afterwards
269/// lands on the same instance, which is what makes the common case free of
270/// ceremony:
271///
272/// ```cpp
273/// BRIDGE_KEY_FROM(AccountModel, LoadAccount, &LoadAccount::id);
274///
275/// BridgeHandler<AccountModel, AllowShared> first{bridge, gui}, second{bridge, gui};
276/// first .execute(LoadAccount{.id = 32}); // constructs the instance for 32
277/// second.execute(LoadAccount{.id = 32}); // attaches to it; constructs nothing
278/// first .execute(Deposit{.amount = 100}); // keyless -> instance 32
279/// second.execute(GetBalance{}); // keyless -> instance 32, sees the 100
280/// ```
281///
282/// `MEMBER` is a pointer-to-data-member of `A` (e.g. `&LoadAccount::id`) whose
283/// type satisfies `morph::model::ModelKey`. Must appear at global scope, in
284/// exactly one translation unit, like the other `BRIDGE_REGISTER_*` macros.
285// clang-format off -- public macro surface: hand-aligned on purpose.
286// These definitions are the framework's documented API; contributors read them
287// as reference, and the continuation backslashes line up so the body is legible
288// as a block. Leaving them to the formatter means any unrelated edit nearby
289// re-wraps the whole definition, and in one case it broke a token-paste
290// invocation apart. Freeze them; realign by hand if a body changes.
291#define BRIDGE_MODEL_KEY(M, A, MEMBER) \
292 template <> \
293 struct morph::model::ActionKeyTraits<A> { \
294 static constexpr bool hasKey = true; \
295 static constexpr bool fromResult = false; \
296 static std::string key(const A& action) { return morph::model::keyToString(action.*MEMBER); } \
297 }; \
298 template <> \
299 struct morph::model::ModelKeyTraits<M> { \
300 using PrimaryKey = morph::model::detail::MemberTypeOf<decltype(MEMBER)>; \
301 }
