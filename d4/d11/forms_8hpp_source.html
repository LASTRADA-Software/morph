5/// @file forms.hpp
6/// @brief JSON-Forms-style schema generation for morph actions.
7///
8/// Given an action type `A` (a plain aggregate, as registered with
9/// `BRIDGE_REGISTER_ACTION`), this header produces a standard JSON Schema a
10/// client can render a form from at runtime — the "describe" half of an
11/// auto-generated GUI. It builds on glaze's `write_json_schema` (which
12/// already contributes types, `$defs`, per-field metadata declared via
13/// `glz::json_schema<A>`, and the `ExtUnits` stamped by
14/// `morph::units::Quantity`) and closes the two gaps glaze leaves open:
15///
16/// - **`required`** — glaze's schema writer emits no `required` array at
17/// all. `schemaJson<A>()` derives one: a member is *required* unless it is
18/// a `std::optional<...>` or its name is listed in the action's opt-out
19/// list (see below).
20/// - **`x-decimalPlaces`** — for `Quantity` members, the unit's default
21/// decimal count from `UnitTraits`, so a client knows the input step
22/// without hardcoding unit knowledge.
23/// - **`x-order`** — the member's declaration index on every property, so a
24/// renderer can lay fields out in declaration order (JSON object key order
25/// is not reliable once schemas pass through DOMs/maps).
26/// - **`x-unitAlternatives`** — for `Quantity` members whose unit system
27/// declares convertible units (`UnitTraits<E>::alternatives`): an array of
28/// `{id, display, decimals, num, den}` entries, where `num/den` is the
29/// exact alternative-to-canonical ratio. Renderers offer a unit selector
30/// and recalculate entered values exactly on switch; payloads always carry
31/// the canonical unit.
32/// - **`x-optionsAction` / `x-optionValue` / `x-optionLabel`** — for
33/// `morph::forms::Choice` members: which registered action serves the
34/// options, and which result-row fields carry the submitted value and the
35/// display label. Renderers turn these into combo boxes populated by
36/// executing the named action.
37///
38/// `morph::time::Timestamp` members need no extension keys: their schema
39/// carries the standard `"format": "date-time"` annotation.
40///
41/// @par Declaring optional fields
42/// Required is the default. An action opts individual fields out with a
43/// static member list:
44/// @code{.cpp}
45/// struct RecordMeasurement {
46/// std::int64_t sampleId = 0;
47/// Density density{};
48/// Moisture moisture{}; // may stay empty
49///
50/// static constexpr std::array optionalFields{std::string_view{"moisture"}};
51/// [[nodiscard]] bool validate() const { return morph::forms::allRequiredEngaged(*this); }
52/// };
53/// @endcode
54///
55/// @par Readiness helper
56/// `allRequiredEngaged(action)` returns `true` when every *required*
57/// empty-capable member (`Quantity`, `Choice`, `Timestamp` — anything with a
58/// `hasValue()`) is engaged. Wire it up as the action's `validate()`
59/// (the existing `ActionValidator` machinery picks that method up
60/// automatically) so the same declaration drives the schema's `required`
61/// array, the client-side submit gate, and the fielded-action readiness
62/// check. Non-quantity members are not checked — a plain `int64_t` cannot
63/// express "not filled in"; use a `Quantity` (or a custom `validate()`) when
64/// that distinction matters.

80/// @brief Concept: a field type with an internal empty state (`Quantity`,
81/// `Choice`, `Timestamp`, or any user type exposing `hasValue()`).
82template <typename T>
83concept EmptyCapableField = requires(const T& field) {
84 { field.hasValue() } -> std::convertible_to<bool>;
85};

89/// @brief Trait: is `T` a `std::optional<...>` (and therefore never required)?
90template <typename T>
91struct IsStdOptional : std::false_type {};
92
93template <typename T>
94struct IsStdOptional<std::optional<T>> : std::true_type {};
95
96template <typename T>
97inline constexpr bool isStdOptional = IsStdOptional<std::remove_cvref_t<T>>::value;

99/// @brief Concept: action declares a `static constexpr` iterable
100/// `optionalFields` list of field names.
101template <typename A>
102concept HasOptionalFields = requires {
103 std::begin(A::optionalFields);
104 std::end(A::optionalFields);
105};
106
107/// @brief Returns `true` when @p fieldName appears in `A::optionalFields`.
108template <typename A>
109[[nodiscard]] constexpr bool declaredOptional(std::string_view fieldName) noexcept {
110 if constexpr (HasOptionalFields<A>) {
111 for (std::string_view const candidate : A::optionalFields) {
112 if (candidate == fieldName) {
113 return true;
114 }
115 }
116 } else {
117 static_cast<void>(fieldName);
118 }
119 return false;
120}

122/// @brief Invokes `visitor.operator()<I>(name, member)` for every reflected
123/// member of @p action (glaze pure reflection).
124template <typename A, typename Visitor>
125// NOLINTNEXTLINE(cppcoreguidelines-missing-std-forward, cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) — member-tie iteration
126constexpr void forEachNamedMember(A&& action, Visitor&& visitor) {
127 using Plain = std::remove_cvref_t<A>;
128 constexpr auto memberCount = glz::reflect<Plain>::size;
129 auto memberTie = glz::to_tie(action);
130 [&]<std::size_t... I>(std::index_sequence<I...>) {
131 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) — index bounded by reflect::size
132 (visitor.template operator()<I>(glz::reflect<Plain>::keys[I], glz::get_member(action, get<I>(memberTie))),
133 ...);
134 }(std::make_index_sequence<memberCount>{});
135}

137/// @brief The DOM post-merge behind `schemaJson`: adds the derived `required`
138/// array, `x-order`, and `x-decimalPlaces` to a glaze-produced schema.
139///
140/// Separated from `schemaJson` so the fallback path (malformed input passes
141/// through unchanged) is directly testable.
142// NOLINTBEGIN(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) — glaze DOM requires operator[]
143template <typename A>
144[[nodiscard]] std::string mergeSchemaExtras(std::string rawSchema) {
145 // u64 number mode: the schema carries int64/uint64 bounds in $defs, which
146 // the default double-only DOM would silently round.
147 glz::generic_u64 dom{};
148 if (glz::read_json(dom, rawSchema)) {
149 return rawSchema;
150 }
151
152 glz::generic_u64::array_t requiredNames{};
153 A probe{};
154 forEachNamedMember(probe, [&]<std::size_t I>(std::string_view name, const auto& member) {
155 using Member = std::remove_cvref_t<decltype(member)>;
156 static_cast<void>(member);
157 const bool isOptional = isStdOptional<Member> || declaredOptional<A>(name);
158 if (!isOptional) {
159 requiredNames.emplace_back(std::string{name});
160 }
161 auto& property = dom["properties"][std::string{name}];
162 property["x-order"] = std::uint64_t{I};
163 if constexpr (units::isQuantity<Member>) {
164 // The field's *declared* precision: the unit default unless the
165 // field's type overrides it (Quantity<Unit::m3, 4>).
166 property["x-decimalPlaces"] = std::uint64_t{Member::declaredDecimals};
167
168 // Convertible display/entry units with their exact ratios.
169 auto const alternatives = Member::unitAlternatives();
170 if (!alternatives.empty()) {
171 glz::generic_u64::array_t list{};
172 for (auto const& alternative : alternatives) {
173 auto const meta = units::UnitTraits<std::remove_const_t<decltype(Member::unit)>>::meta(alternative.unit);
174 glz::generic_u64 entry{};
175 entry["id"] = std::string{meta.id};
176 entry["display"] = std::string{meta.display};
177 entry["decimals"] = std::uint64_t{meta.defaultDecimals};
178 entry["num"] = alternative.num;
179 entry["den"] = alternative.den;
180 list.emplace_back(std::move(entry));
181 }
182 property["x-unitAlternatives"] = list;
183 }
184 }
185 if constexpr (isChoice<Member>) {
186 // Which action serves the options, and which result-row fields
187 // carry the submitted value / display label.
188 property["x-optionsAction"] = std::string{Member::optionsAction()};
189 property["x-optionValue"] = std::string{Member::valueField()};
190 property["x-optionLabel"] = std::string{Member::labelField()};
191 }
192 });
193 // Always assign — an explicit empty array beats leaving whatever the
194 // schema writer may have emitted (or omitted) for `required`.
195 dom["required"] = requiredNames;
196
197 // value_or without a move: the copy is irrelevant (schemaJson memoises),
198 // and keeping the fallback branch inside glaze's expected avoids an
199 // untestable line here (write_json of a DOM we just built cannot fail).
200 return glz::write_json(dom).value_or(rawSchema);
201}
202// NOLINTEND(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)

206/// @brief Whether every required empty-capable member of @p action is
207/// engaged (has a value).
208///
209/// Empty-capable covers `Quantity`, `Choice`, `Timestamp`, and any user type
210/// satisfying `EmptyCapableField`. Required means: not a `std::optional<...>`
211/// member and not listed in `A::optionalFields`. Intended as the body of the
212/// action's `validate()`.
213/// @tparam A Action type (a reflectable aggregate).
214/// @param action Draft whose fields are checked.
215/// @return `true` when no required empty-capable field is empty.
216template <typename A>
217[[nodiscard]] constexpr bool allRequiredEngaged(const A& action) noexcept {
218 bool allEngaged = true;
219 detail::forEachNamedMember(action, [&]<std::size_t I>(std::string_view name, const auto& member) {
220 using Member = std::remove_cvref_t<decltype(member)>;
221 if constexpr (EmptyCapableField<Member>) {
222 if (!detail::declaredOptional<A>(name) && !member.hasValue()) {
223 allEngaged = false;
224 }
225 } else {
226 static_cast<void>(name);
227 static_cast<void>(member);
228 }
229 });
230 return allEngaged;
231}

233/// @brief Generates the JSON Schema for action type @p A, ready for a
234/// client-side form renderer.
235///
236/// glaze's `write_json_schema<A>()` output, post-processed with:
237/// - a top-level `required` array (see file docs for the rule),
238/// - `x-decimalPlaces` on every `Quantity` property (the unit's default),
239/// - `x-order` (declaration index) on every property.
240///
241/// The result is fixed per type, so it is computed once and cached. On any
242/// internal failure the unmerged glaze schema (or an empty string if even
243/// that failed) is returned rather than throwing — schema generation is a
244/// description facility, never worth crashing a server over.
245/// @tparam A Action type (a reflectable aggregate).
246/// @return The merged schema JSON.
247template <typename A>
248[[nodiscard]] std::string schemaJson() {
249 static const std::string cached =
250 detail::mergeSchemaExtras<A>(glz::write_json_schema<A>().value_or(std::string{}));
251 return cached;
252}
