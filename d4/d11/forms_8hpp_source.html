5/// @file forms/forms.hpp
6/// @brief JSON-Forms-style schema generation for morph actions.
7///
8/// Given an action type `A` (a plain aggregate, as registered with
9/// `BRIDGE_REGISTER_ACTION`), this header produces a standard JSON Schema a
10/// client can render a form from at runtime — the "describe" half of an
11/// auto-generated GUI. It builds on glaze's `write_json_schema` (which
12/// already contributes types, `$defs`, per-field metadata declared via
13/// `glz::json_schema<A>`, and the `ExtUnits` stamped by
14/// `morph::units::Quantity`) and closes the two gaps glaze leaves open:
15///
16/// - **`required`** — glaze's schema writer emits no `required` array at
17/// all. `schemaJson<A>()` derives one: a member is *required* unless it is
18/// a `std::optional<...>` or its name is listed in the action's opt-out
19/// list (see below).
20/// - **`x-decimalPlaces`** — for `Quantity` members, the unit's default
21/// decimal count from `UnitTraits`, so a client knows the input step
22/// without hardcoding unit knowledge.
23/// - **`x-order`** — the member's declaration index on every property, so a
24/// renderer can lay fields out in declaration order (JSON object key order
25/// is not reliable once schemas pass through DOMs/maps).
26/// - **`x-unitAlternatives`** — for `Quantity` members whose unit system
27/// declares convertible units (`UnitTraits<E>::relations`): an array of
28/// `{id, display, decimals, num, den}` entries, where `num/den` is the
29/// exact alternative-to-canonical ratio. Renderers offer a unit selector
30/// and recalculate entered values exactly on switch; payloads always carry
31/// the canonical unit.
32/// - **`x-optionsAction` / `x-optionValue` / `x-optionLabel`** — for
33/// `morph::forms::Choice` members: which registered action serves the
34/// options, and which result-row fields carry the submitted value and the
35/// display label. Renderers turn these into combo boxes populated by
36/// executing the named action.
37///
38/// `morph::time::Timestamp` members need no extension keys: their schema
39/// carries the standard `"format": "date-time"` annotation.
40///
41/// @par Declaring optional fields
42/// Required is the default. An action opts individual fields out with a
43/// static member list:
44/// @code{.cpp}
45/// struct RecordMeasurement {
46/// std::int64_t sampleId = 0;
47/// Density density{};
48/// Moisture moisture{}; // may stay empty
49///
50/// static constexpr std::array optionalFields{std::string_view{"moisture"}};
51/// [[nodiscard]] bool validate() const { return morph::forms::allRequiredEngaged(*this); }
52/// };
53/// @endcode
54///
55/// @par Readiness helper
56/// `allRequiredEngaged(action)` returns `true` when every *required*
57/// empty-capable member (`Quantity`, `Choice`, `Timestamp` — anything with a
58/// `hasValue()`) is engaged. Wire it up as the action's `validate()`
59/// (the existing `ActionValidator` machinery picks that method up
60/// automatically) so the same declaration drives the schema's `required`
61/// array, the client-side submit gate, and the fielded-action readiness
62/// check. Non-quantity members are not checked — a plain `int64_t` cannot
63/// express "not filled in"; use a `Quantity` (or a custom `validate()`) when
64/// that distinction matters.

80/// @brief Concept: a field type with an internal empty state (`Quantity`,
81/// `Choice`, `Timestamp`, or any user type exposing `hasValue()`).
82///
83/// `hasValue()` must be `noexcept`: `allRequiredEngaged` is `noexcept` and calls
84/// it on every member, so a throwing `hasValue()` would cross a `noexcept`
85/// boundary and call `std::terminate`. Requiring it here turns that into a
86/// compile-time rejection instead — a user field with a throwing `hasValue()`
87/// simply does not satisfy the concept.
88template <typename T>
89concept EmptyCapableField = requires(const T& field) {
90 { field.hasValue() } noexcept -> std::convertible_to<bool>;
91};

95/// @brief Trait: is `T` a `std::optional<...>` (and therefore never required)?
96template <typename T>
97struct IsStdOptional : std::false_type {};
98
99template <typename T>
100struct IsStdOptional<std::optional<T>> : std::true_type {};
101
102template <typename T>
103inline constexpr bool isStdOptional = IsStdOptional<std::remove_cvref_t<T>>::value;
104
105/// @brief Concept: action declares a `static constexpr` iterable
106/// `optionalFields` list of field names.
107template <typename A>
108concept HasOptionalFields = requires {
109 std::begin(A::optionalFields);
110 std::end(A::optionalFields);
111};
112
113/// @brief Returns `true` when @p fieldName appears in `A::optionalFields`.
114template <typename A>
115[[nodiscard]] constexpr bool declaredOptional(std::string_view fieldName) noexcept {
116 if constexpr (HasOptionalFields<A>) {
117 for (std::string_view const candidate : A::optionalFields) {
118 if (candidate == fieldName) {
119 return true;
120 }
121 }
122 } else {
123 static_cast<void>(fieldName);
124 }
125 return false;
126}

128/// @brief Invokes `visitor.operator()<I>(name, member)` for every reflected
129/// member of @p action (glaze pure reflection).
130template <typename A, typename Visitor>
131// NOLINTNEXTLINE(cppcoreguidelines-missing-std-forward, cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) — member-tie iteration
132constexpr void forEachNamedMember(A&& action, Visitor&& visitor) {
133 using Plain = std::remove_cvref_t<A>;
134 constexpr auto memberCount = glz::reflect<Plain>::size;
135 auto memberTie = glz::to_tie(action);
136 [&]<std::size_t... I>(std::index_sequence<I...>) {
137 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) — index bounded by reflect::size
138 (visitor.template operator()<I>(glz::reflect<Plain>::keys[I], glz::get_member(action, get<I>(memberTie))),
139 ...);
140 }(std::make_index_sequence<memberCount>{});
141}

143/// @brief The DOM post-merge behind `schemaJson`: adds the derived `required`
144/// array, `x-order`, and `x-decimalPlaces` to a glaze-produced schema.
145///
146/// Separated from `schemaJson` so the fallback path (malformed input passes
147/// through unchanged) is directly testable.
148// NOLINTBEGIN(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) — glaze DOM requires operator[]
149template <typename A>
150[[nodiscard]] std::string mergeSchemaExtras(std::string rawSchema) {
151 // u64 number mode: the schema carries int64/uint64 bounds in $defs, which
152 // the default double-only DOM would silently round.
153 glz::generic_u64 dom{};
154 if (glz::read_json(dom, rawSchema)) {
155 return rawSchema;
156 }
157
158 glz::generic_u64::array_t requiredNames{};
159 A probe{};
160 forEachNamedMember(probe, [&]<std::size_t I>(std::string_view name, const auto& member) {
161 using Member = std::remove_cvref_t<decltype(member)>;
162 static_cast<void>(member);
163 const bool isOptional = isStdOptional<Member> || declaredOptional<A>(name);
164 if (!isOptional) {
165 requiredNames.emplace_back(std::string{name});
166 }
167 auto& property = dom["properties"][std::string{name}];
168 property["x-order"] = std::uint64_t{I};
169 if constexpr (units::isQuantity<Member>) {
170 // The field's *declared* precision: the unit default unless the
171 // field's type overrides it (Quantity<Unit::m3, 4>).
172 property["x-decimalPlaces"] = std::uint64_t{Member::declaredDecimals};
173
174 // Convertible display/entry units with their exact ratios.
175 auto const alternatives = Member::unitAlternatives();
176 if (!alternatives.empty()) {
177 glz::generic_u64::array_t list{};
178 for (auto const& alternative : alternatives) {
179 auto const meta = units::UnitTraits<std::remove_const_t<decltype(Member::unit)>>::meta(alternative.unit);
180 glz::generic_u64 entry{};
181 entry["id"] = std::string{meta.id};
182 entry["display"] = std::string{meta.display};
183 entry["decimals"] = std::uint64_t{meta.defaultDecimals};
184 entry["num"] = alternative.num;
185 entry["den"] = alternative.den;
186 list.emplace_back(std::move(entry));
187 }
188 property["x-unitAlternatives"] = list;
189 }
190 }
191 if constexpr (isChoice<Member>) {
192 // Which action serves the options, and which result-row fields
193 // carry the submitted value / display label.
194 property["x-optionsAction"] = std::string{Member::optionsAction()};
195 property["x-optionValue"] = std::string{Member::valueField()};
196 property["x-optionLabel"] = std::string{Member::labelField()};
197 }
198 });
199 // Always assign — an explicit empty array beats leaving whatever the
200 // schema writer may have emitted (or omitted) for `required`.
201 dom["required"] = requiredNames;
202
203 // value_or without a move: the copy is irrelevant (schemaJson memoises),
204 // and keeping the fallback branch inside glaze's expected avoids an
205 // untestable line here (write_json of a DOM we just built cannot fail).
206 return glz::write_json(dom).value_or(rawSchema);
207}
208// NOLINTEND(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)

212/// @brief Retags every `Quantity` member of @p action to its **declared**
213/// precision, so a stored value matches the precision the schema
214/// advertises via `x-decimalPlaces`.
215///
216/// A wire payload carries each `Quantity` with its own runtime `dp`, which a
217/// client may set to anything. Left alone, the field is stored at the client's
218/// `dp`, silently contradicting the schema's `x-decimalPlaces` (which is the
219/// field's compile-time *declared* precision, `Quantity<U, Dec>::declaredDecimals`).
220/// Calling this on the decode path — right after `ActionTraits<A>::fromJson` and
221/// before dispatch — re-tags each `Quantity` to `declaredPrecision()` so the two
222/// agree. `atDeclaredPrecision()` only changes the value's precision tag (an
223/// exact `Rational` re-rounding to the declared decimals); an empty `Quantity`
224/// is left empty. Non-`Quantity` members are untouched.
225///
226/// This is the enforcement half of the `x-decimalPlaces` contract: the schema
227/// advertises the declared precision and the dispatch path now stores at that
228/// precision, rather than honouring whatever `dp` the client sent.
229/// @tparam A Action type (a reflectable aggregate).
230/// @param action Draft action whose `Quantity` members are retagged in place.
231template <typename A>
232constexpr void reconcileDeclaredPrecision(A& action) {
233 using Plain = std::remove_cvref_t<A>;
234 // Only actions glaze can reflect member-by-member (aggregates, or types with
235 // a `glz::meta`) can be walked here. Actions with hand-written codecs and no
236 // reflectable shape — and there is nothing to retag on them anyway — fall
237 // through as a no-op so this stays safe to call for *every* registered
238 // action from the dispatch path, not only form actions.
239 if constexpr (glz::reflectable<Plain> || glz::glaze_object_t<Plain>) {
240 constexpr auto memberCount = glz::reflect<Plain>::size;
241 auto memberTie = glz::to_tie(action);
242 [&]<std::size_t... I>(std::index_sequence<I...>) {
243 [[maybe_unused]] auto retag = [&]<std::size_t Idx>() {
244 auto& member = glz::get_member(action, get<Idx>(memberTie));
245 using Member = std::remove_cvref_t<decltype(member)>;
246 if constexpr (units::isQuantity<Member>) {
247 member = member.atDeclaredPrecision();
248 }
249 };
250 (retag.template operator()<I>(), ...);
251 }(std::make_index_sequence<memberCount>{});
252 } else {
253 static_cast<void>(action);
254 }
255}

257/// @brief Whether every required empty-capable member of @p action is
258/// engaged (has a value).
259///
260/// Empty-capable covers `Quantity`, `Choice`, `Timestamp`, and any user type
261/// satisfying `EmptyCapableField`. Required means: not a `std::optional<...>`
262/// member and not listed in `A::optionalFields`. Intended as the body of the
263/// action's `validate()`.
264/// @tparam A Action type (a reflectable aggregate).
265/// @param action Draft whose fields are checked.
266/// @return `true` when no required empty-capable field is empty.
267template <typename A>
268[[nodiscard]] constexpr bool allRequiredEngaged(const A& action) noexcept {
269 bool allEngaged = true;
270 detail::forEachNamedMember(action, [&]<std::size_t I>(std::string_view name, const auto& member) {
271 using Member = std::remove_cvref_t<decltype(member)>;
272 if constexpr (EmptyCapableField<Member>) {
273 if (!detail::declaredOptional<A>(name) && !member.hasValue()) {
274 allEngaged = false;
275 }
276 } else {
277 static_cast<void>(name);
278 static_cast<void>(member);
279 }
280 });
281 return allEngaged;
282}

284/// @brief Generates the JSON Schema for action type @p A, ready for a
285/// client-side form renderer.
286///
287/// glaze's `write_json_schema<A>()` output, post-processed with:
288/// - a top-level `required` array (see file docs for the rule),
289/// - `x-decimalPlaces` on every `Quantity` property (the unit's default),
290/// - `x-order` (declaration index) on every property.
291///
292/// The result is fixed per type, so it is computed once and cached. On any
293/// internal failure the unmerged glaze schema (or an empty string if even
294/// that failed) is returned rather than throwing — schema generation is a
295/// description facility, never worth crashing a server over.
296/// @tparam A Action type (a reflectable aggregate).
297/// @return The merged schema JSON.
298template <typename A>
299[[nodiscard]] std::string schemaJson() {
300 static const std::string cached =
301 detail::mergeSchemaExtras<A>(glz::write_json_schema<A>().value_or(std::string{}));
302 return cached;
303}
