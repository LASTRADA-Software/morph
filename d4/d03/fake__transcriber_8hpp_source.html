21/// @brief Answers each call with the next scripted outcome, and fails once the script runs out. Thread-safe.
22///
23/// Delivers before `transcribe` returns, unless `hold(true)` leaves each call outstanding until `release`. Counts the
24/// calls cancelled before they delivered.
25class FakeTranscriber final : public ITranscriber {
26public:
27 /// @brief Adds an outcome to the script.
28 /// @param outcome What the next unscripted call answers: the text, or why it failed.
29 void script(std::expected<std::string, std::string> outcome) {
30 std::scoped_lock const lock{_state->mutex};
31 _state->script.push_back(std::move(outcome));
32 }
33
34 /// @brief Sets whether calls wait for `release`.
35 /// @param held Whether they do.
36 void hold(bool held) {
37 std::scoped_lock const lock{_state->mutex};
38 _state->held = held;
39 }
40
41 /// @brief Delivers every held call that was not cancelled, and stops holding.
42 void release() {
43 std::vector<std::shared_ptr<Pending>> pending;
44 {
45 std::scoped_lock const lock{_state->mutex};
46 _state->held = false;
47 pending.swap(_state->pending);
48 }
49 for (auto const& call : pending) {
50 call->deliver();
51 }
52 }
53
54 /// @brief The calls made so far.
55 /// @return Their number.
56 [[nodiscard]] std::size_t calls() const {
57 std::scoped_lock const lock{_state->mutex};
58 return _state->inputs.size();
59 }
60
61 /// @brief The calls cancelled before they delivered.
62 /// @return Their number.
63 [[nodiscard]] std::size_t cancelled() const {
64 std::scoped_lock const lock{_state->mutex};
65 return _state->cancelled;
66 }
67
68 /// @brief What each call was given.
69 /// @return A copy of the inputs, oldest first.
70 [[nodiscard]] std::vector<TranscriptionInput> inputs() const {
71 std::scoped_lock const lock{_state->mutex};
72 return _state->inputs;
73 }
74
75 /// @brief Starts a call.
76 /// @param input The clip.
77 /// @param sink Where it delivers.
78 /// @return The call.
79 [[nodiscard]] std::shared_ptr<ITranscriberCall> transcribe(TranscriptionInput input,
80 TranscriberSink sink) override {
81 auto call = std::make_shared<Pending>(_state, std::move(sink));
82 bool held = false;
83 {
84 std::scoped_lock const lock{_state->mutex};
85 _state->inputs.push_back(std::move(input));
86 if (!_state->script.empty()) {
87 call->outcome = std::move(_state->script.front());
88 _state->script.pop_front();
89 } else {
90 call->outcome = std::unexpected(std::string{"the fake transcriber's script ran out"});
91 }
92 held = _state->held;
93 if (held) {
94 _state->pending.push_back(call);
95 }
96 }
97 if (!held) {
98 call->deliver();
99 }
100 return call;
101 }
102
103private:
104 struct State;
105
106 // Holds the state, so a call outlives the transcriber and a late cancel touches nothing freed. A finished call
107 // lets go of its sink, which usually owns whoever owns this transcriber; it is destroyed outside the lock.
108 struct Pending final : ITranscriberCall {
109 Pending(std::shared_ptr<State> shared, TranscriberSink given)
110 : state{std::move(shared)}, sink{std::move(given)} {}
111
112 void cancel() override {
113 TranscriberSink dropped;
114 std::shared_ptr<Pending> unheld;
115 {
116 std::scoped_lock const lock{state->mutex};
117 if (finished) {
118 return;
119 }
120 finished = true;
121 ++state->cancelled;
122 dropped = std::move(sink);
123 auto const position = std::ranges::find_if(
124 state->pending, [this](std::shared_ptr<Pending> const& held) { return held.get() == this; });
125 if (position != state->pending.end()) {
126 unheld = std::move(*position);
127 state->pending.erase(position);
128 }
129 }
130 }
131
132 void deliver() {
133 TranscriberSink taken;
134 {
135 std::scoped_lock const lock{state->mutex};
136 if (finished) {
137 return;
138 }
139 finished = true;
140 taken = std::move(sink);
141 }
142 if (taken.done) {
143 taken.done(std::move(outcome));
144 }
145 }
146
147 std::shared_ptr<State> state;
148 TranscriberSink sink;
149 std::expected<std::string, std::string> outcome = std::unexpected(std::string{});
150 bool finished = false;
151 };
152
153 struct State {
154 mutable std::mutex mutex;
155 std::deque<std::expected<std::string, std::string>> script;
156 std::vector<TranscriptionInput> inputs;
157 std::vector<std::shared_ptr<Pending>> pending;
158 std::size_t cancelled = 0;
159 bool held = false;
160 };
161
162 std::shared_ptr<State> _state = std::make_shared<State>();
163};
