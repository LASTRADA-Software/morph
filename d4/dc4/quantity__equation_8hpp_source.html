31/// @brief Whether a node is a plain leaf (no recorded operation).
32/// @param node The node.
33/// @return `true` for a leaf.
34[[nodiscard]] inline bool isLeafNode(const ASTNode& node) { return node.current.operation.empty(); }
35
36/// @brief Whether a node is a unit-conversion step.
37/// @param node The node.
38/// @return `true` for a conversion.
39[[nodiscard]] inline bool isConversionNode(const ASTNode& node) {
40 return node.current.operation.starts_with("convert");
41}

43/// @brief Whether a node renders as a single atom (leaf or conversion value).
44/// @param node The node.
45/// @return `true` when atomic.
46[[nodiscard]] inline bool isAtomNode(const ASTNode& node) { return isLeafNode(node) || isConversionNode(node); }

48/// @brief The value a node contributes (leaf value, else the step result).
49/// @param node The node.
50/// @return The node's optional value.
51[[nodiscard]] inline const std::optional<morph::math::Rational>& nodeValue(const ASTNode& node) {
52 return isLeafNode(node) ? node.current.lhs : node.current.result;

55/// @brief A rendered subexpression plus the precedence of its top operator.
56struct Rendered {
57 /// @brief The rendered text.
58 std::string text;
59
60 /// @brief Precedence of the outermost operator (100 for an atom).
61 int precedence{100};
62};

64/// @brief Stateful renderer for one `equation()` call.
65struct EquationRenderer {
66 /// @brief Placeholder number per reused, unnamed node (1-based).
67 std::unordered_map<const ASTNode*, std::size_t> labelIndex;
68
69 /// @brief In-edge count per node across displayed (non-opaque) paths.
70 std::unordered_map<const ASTNode*, int> refCount;
71
72 /// @brief Nodes visited during ref counting (dedup).
73 std::unordered_set<const ASTNode*> seen;
74
75 /// @brief Reused-node placeholders, in first-appearance order.
76 std::vector<const ASTNode*> placeholderOrder;
77
78 /// @brief Whether a node earns a placeholder (reused). Only ever called on
79 /// unnamed nodes recorded by `countRefs` (callers return early on
80 /// named nodes), so the lookup is total.
81 /// @param node The node (never null, never named at any call site).
82 /// @return `true` when it is a placeholder.
83 [[nodiscard]] bool isPlaceholder(const ASTNode* node) const { return refCount.at(node) >= 2; }
84
85 /// @brief Counts in-edges through displayed paths (stops at opaque nodes).
86 /// Recurses into children unconditionally; the null guard handles the
87 /// absent operands of leaf/unary/scalar steps.
88 /// @param node The current node (may be null).
89 void countRefs(const ASTNode* node) {
90 if (node == nullptr) {
91 return;
92 }
93 if (seen.contains(node)) {
94 return;
95 }
96 seen.insert(node);
97 if (node->name.has_value() || isAtomNode(*node)) {
98 return;
99 }
100 if (node->left) {
101 ++refCount[node->left.get()];
102 }
103 if (node->right) {
104 ++refCount[node->right.get()];
105 }
106 countRefs(node->left.get());
107 countRefs(node->right.get());
108 }
109
110 /// @brief Assigns placeholder labels in first-appearance order.
111 /// @param node The current node.
112 /// @param expandThis Whether to expand @p node itself (rather than label it).
113 void assignLabels(const ASTNode* node, bool expandThis) {
114 if (node == nullptr || node->name.has_value()) {
115 return;
116 }
117 if (!expandThis && isPlaceholder(node)) {
118 if (!labelIndex.contains(node)) {
119 labelIndex.emplace(node, placeholderOrder.size() + 1);
120 placeholderOrder.push_back(node);
121 }
122 if (!isAtomNode(*node)) {
123 assignLabels(node->left.get(), false);
124 assignLabels(node->right.get(), false);
125 }
126 return;
127 }
128 if (isAtomNode(*node)) {
129 return;
130 }
131 assignLabels(node->left.get(), false);
132 assignLabels(node->right.get(), false);
133 }
134
135 /// @brief Parenthesises and joins a binary subexpression.
136 /// @param op The operator token.
137 /// @param left Rendered left operand.
138 /// @param right Rendered right operand.
139 /// @return The combined rendering.
140 [[nodiscard]] static Rendered combine(const std::string& op, const Rendered& left, const Rendered& right) {
141 int const precedence = (op == "*" || op == "/") ? 2 : 1;
142 std::string const leftText = (left.precedence < precedence) ? "(" + left.text + ")" : left.text;
143 bool const rightNeedsParens =
144 (right.precedence < precedence) || (right.precedence == precedence && (op == "-" || op == "/"));
145 std::string const rightText = rightNeedsParens ? "(" + right.text + ")" : right.text;
146 return Rendered{leftText + " " + op + " " + rightText, precedence};
147 }
148
149 /// @brief Renders a unary-negation subexpression.
150 /// @param operand Rendered operand.
151 /// @return The combined rendering.
152 [[nodiscard]] static Rendered combineUnary(const Rendered& operand) {
153 std::string const text = (operand.precedence <= 1) ? "-(" + operand.text + ")" : "-" + operand.text;
154 return Rendered{text, 3};
155 }
156
157 /// @brief Renders a node symbolically (names, placeholders, inlined ops).
158 /// @param node The node.
159 /// @param expandThis Whether to expand @p node even if it is a placeholder.
160 /// @return The symbolic rendering.
161 [[nodiscard]] Rendered renderSymbolic(const ASTNode* node, bool expandThis) {
162 if (node->name.has_value()) {
163 return Rendered{"\"" + *node->name + "\"", 100};
164 }
165 if (!expandThis && isPlaceholder(node)) {
166 return Rendered{"c" + std::to_string(labelIndex.at(node)), 100};
167 }
168 if (isLeafNode(*node)) {
169 return Rendered{formatOptional(node->current.lhs), 100};
170 }
171 if (isConversionNode(*node)) {
172 return Rendered{formatOptional(node->current.result), 100};
173 }
174 Rendered const left =
175 node->left ? renderSymbolic(node->left.get(), false) : Rendered{formatOptional(node->current.lhs), 100};
176 bool const hasRight = node->right || node->current.rhs.has_value();
177 if (!hasRight) {
178 return combineUnary(left);
179 }
180 Rendered const right =
181 node->right ? renderSymbolic(node->right.get(), false) : Rendered{formatOptional(node->current.rhs), 100};
182 return combine(node->current.operation, left, right);
183 }
184
185 /// @brief Renders a node with values substituted for symbols/placeholders.
186 /// @param node The node.
187 /// @param expandThis Whether to expand @p node even if named/placeholder.
188 /// @return The substituted rendering.
189 [[nodiscard]] Rendered renderSubstituted(const ASTNode* node, bool expandThis) {
190 if (!expandThis && (node->name.has_value() || isPlaceholder(node))) {
191 return Rendered{formatOptional(nodeValue(*node)), 100};
192 }
193 if (isLeafNode(*node)) {
194 return Rendered{formatOptional(node->current.lhs), 100};
195 }
196 if (isConversionNode(*node)) {
197 return Rendered{formatOptional(node->current.result), 100};
198 }
199 Rendered const left =
200 node->left ? renderSubstituted(node->left.get(), false) : Rendered{formatOptional(node->current.lhs), 100};
201 bool const hasRight = node->right || node->current.rhs.has_value();
202 if (!hasRight) {
203 return combineUnary(left);
204 }
205 Rendered const right = node->right ? renderSubstituted(node->right.get(), false)
206 : Rendered{formatOptional(node->current.rhs), 100};
207 return combine(node->current.operation, left, right);
208 }
209
210 /// @brief Builds one `where`-legend line for a placeholder.
211 /// @param node The placeholder node.
212 /// @return The legend body (`cK = ...`).
213 [[nodiscard]] std::string legendLine(const ASTNode* node) {
214 std::string const label = "c" + std::to_string(labelIndex.at(node));
215 if (isAtomNode(*node)) {
216 return label + " = " + formatOptional(nodeValue(*node));
217 }
218 return label + " = " + renderSymbolic(node, true).text + " = " + renderSubstituted(node, true).text + " = " +
219 formatOptional(node->current.result);
220 }
221};

227template <auto U, std::uint32_t DeclaredDecimals>
228 requires UnitEnum<decltype(U)>
229std::vector<std::string> Quantity<U, DeclaredDecimals>::equation() const {
230 if (!payload) {
231 return {detail::formatOptional(payload)};
232 }
233 const detail::ASTNode* root = _ctx.node.get();
234 if (root == nullptr) {
235 return {detail::formatOptional(payload)};
236 }
237 if (root->name.has_value()) {
238 return {"\"" + *root->name + "\""};
239 }
240 if (detail::isLeafNode(*root)) {
241 return {detail::formatOptional(root->current.lhs)};
242 }
243 if (detail::isConversionNode(*root)) {
244 return {detail::formatOptional(root->current.result)};
245 }
246
247 detail::EquationRenderer renderer;
248 renderer.countRefs(root);
249 renderer.assignLabels(root, true);
250
251 std::vector<std::string> lines;
252 lines.push_back(renderer.renderSymbolic(root, true).text);
253 lines.push_back(" = " + renderer.renderSubstituted(root, true).text);
254 lines.push_back(" = " + detail::formatOptional(root->current.result));
255 for (std::size_t i = 0; i < renderer.placeholderOrder.size(); ++i) {
256 std::string const body = renderer.legendLine(renderer.placeholderOrder[i]);
257 lines.push_back((i == 0 ? "where " : " ") + body);
258 }
259 return lines;
260}
