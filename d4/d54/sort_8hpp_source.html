5/// @file table/sort.hpp
6/// @brief Typed sort keys and the chunked, stoppable merge sort over them.
7///
8/// Each column a sort or filter uses gets a `KeyColumn`: one typed array of
9/// keys plus a `CellState` per cell, built once from a snapshot. Comparing two
10/// rows is then a direct compare of two keys, with no parsing and no
11/// allocation. Exact kinds stay exact: integers, dates and booleans compare as
12/// `int64`, decimals and quantities as `Rational` (128-bit cross products);
13/// only a `Number` column holds a double.
14///
15/// Ordering rules, normative for every client and server (spec 7 §5):
16/// - an empty or invalid cell sorts after every valid one, in either direction;
17/// - each key of a chain has its own direction;
18/// - ties break by source row, so the sort is stable.
19///
20/// See `docs/spec/table/engine.md`.

110/// @brief Parses a whole decimal integer, all of @p text.
111/// @param text The text.
112/// @return The value, or nothing when @p text is not exactly an integer.
113[[nodiscard]] inline std::optional<std::int64_t> parseInteger(std::string_view text) {
114 std::int64_t value = 0;
115 auto const* const first = std::to_address(text.begin());
116 auto const* const last = std::to_address(text.end());
117 auto const [ptr, ec] = std::from_chars(first, last, value);
118 if (ec != std::errc{} || ptr != last || text.empty()) {
119 return std::nullopt;
120 }

122}

133 return std::nullopt;
134 }
135 return value;

140/// @brief One column's keys, built once and shared, immutable, between jobs.
141///
142/// Exactly one of the typed arrays is filled, chosen by the column's kind:
143/// `ints` for `Integer`, `Key`, `Date`, `DateTime` and `Bool`; `exact` for
144/// `Decimal` and `Quantity`; `reals` for `Number`; `text`, `folded` and `raw`
145/// for `Text`; `cells` for a `Custom` column with a known comparator, and
146/// `text` for one without (ordered by display text).
147struct KeyColumn {
148 /// @brief The column's kind.
149 ColumnKind kind = ColumnKind::Text;
150 /// @brief The state of each row's cell.
151 std::vector<CellState> states;
152 /// @brief Integer keys.
153 std::vector<std::int64_t> ints;
154 /// @brief Exact keys.
155 std::vector<math::Rational> exact;
156 /// @brief Floating-point keys.
157 std::vector<double> reals;
158 /// @brief Collation keys.
159 std::vector<std::string> text;
160 /// @brief Primary-strength folds, which text filters match.
161 std::vector<std::string> folded;
162 /// @brief The text as the source gave it, for case-sensitive filters.
163 std::vector<std::string> raw;
164 /// @brief Cells of a `Custom` column with a comparator.
165 std::vector<Cell> cells;
166 /// @brief The comparator of a `Custom` column, or empty.
167 CellComparator comparator;
168
169 /// @brief Whether this column orders `Custom` cells through a comparator.
170 /// @return `true` when `comparator` is set.
171 [[nodiscard]] bool usesComparator() const noexcept { return static_cast<bool>(comparator); }
172
173 /// @brief An empty key column for @p info, sized for @p rows rows.
174 /// @param info The column.
175 /// @param services Where a `Custom` column's comparator is looked up.
176 /// @param rows Row count.
177 /// @return The column, every cell `Empty` until built.
178 [[nodiscard]] static KeyColumn forColumn(ColumnInfo const& info, Services const& services, std::size_t rows) {
179 KeyColumn out;
180 out.kind = info.kind;
181 out.states.assign(rows, CellState::Empty);
182 switch (info.kind) {
183 case ColumnKind::Integer:
184 case ColumnKind::Key:
185 case ColumnKind::Date:
186 case ColumnKind::DateTime:
187 case ColumnKind::Bool:
188 out.ints.assign(rows, 0);
189 break;
190 case ColumnKind::Decimal:
191 case ColumnKind::Quantity:
192 out.exact.assign(rows, math::Rational{});
193 break;
194 case ColumnKind::Number:
195 out.reals.assign(rows, 0.0);
196 break;
197 case ColumnKind::Text:
198 out.text.resize(rows);
199 out.folded.resize(rows);
200 out.raw.resize(rows);
201 break;
202 case ColumnKind::Custom:
203 if (auto const found = services.comparators.find(info.comparator);
204 found != services.comparators.end() && found->second) {
205 out.comparator = found->second;
206 out.cells.resize(rows);
207 } else {
208 out.text.resize(rows);
209 }
210 break;
211 default:
212 break;
213 }
214 return out;
215 }
216
217 /// @brief Rebuilds the key of one row from @p cell.
218 /// @param row Row index.
219 /// @param cell The row's cell.
220 /// @param info The column.
221 /// @param services Collator and date parser; a null member means its default.
222 void set(std::size_t row, Cell const& cell, ColumnInfo const& info, Services const& services) {
223 states[row] = read(row, cell, info, services);
224 }
225
226private:
227 [[nodiscard]] CellState read(std::size_t row, Cell const& cell, ColumnInfo const& info, Services const& services) {
228 if (std::holds_alternative<std::monostate>(cell)) {
229 return CellState::Empty;
230 }
231 auto const* const textCell = std::get_if<std::string>(&cell);
232 if (textCell != nullptr && textCell->empty()) {
233 return CellState::Empty;
234 }
235 switch (kind) {
236 case ColumnKind::Integer:
237 case ColumnKind::Key:
238 return readInteger(row, cell);
239 case ColumnKind::Decimal:
240 case ColumnKind::Quantity:
241 return readExact(row, cell);
242 case ColumnKind::Number:
243 return readReal(row, cell);
244 case ColumnKind::Text: {
245 auto const shown = textCell != nullptr ? *textCell : displayText(cell);
246 auto const& collator = detail::collatorOf(services);
247 text[row] = collator.sortKey(shown);
248 folded[row] = collator.fold(shown);
249 raw[row] = shown;
250 return CellState::Valid;
251 }
252 case ColumnKind::Date:
253 case ColumnKind::DateTime:
254 return readDate(row, cell, info, services);
255 case ColumnKind::Bool:
256 return readBool(row, cell);
257 case ColumnKind::Custom:
258 if (usesComparator()) {
259 cells[row] = cell;
260 } else {
261 text[row] = detail::collatorOf(services).sortKey(displayText(cell));
262 }
263 return CellState::Valid;
264 default:
265 break;
266 }
267 return CellState::Invalid;
268 }
269
270 [[nodiscard]] CellState readInteger(std::size_t row, Cell const& cell) {
271 if (auto const* value = std::get_if<std::int64_t>(&cell)) {
272 ints[row] = *value;
273 return CellState::Valid;
274 }
275 if (auto const* value = std::get_if<math::Rational>(&cell); value != nullptr && value->isInteger()) {
276 ints[row] = value->numerator;
277 return CellState::Valid;
278 }
279 if (auto const* value = std::get_if<std::string>(&cell)) {
280 if (auto const parsed = detail::parseInteger(*value)) {
281 ints[row] = *parsed;
282 return CellState::Valid;
283 }
284 }
285 return CellState::Invalid;
286 }
287
288 [[nodiscard]] CellState readExact(std::size_t row, Cell const& cell) {
289 if (auto const* value = std::get_if<math::Rational>(&cell)) {
290 exact[row] = *value;
291 return CellState::Valid;
292 }
293 if (auto const* value = std::get_if<QuantityCell>(&cell)) {
294 // Checked: a product that does not fit int64 is an invalid cell, not
295 // a saturated value that would sort as the largest quantity.
296 auto const canonical = math::checkedMul(value->amount, value->toCanonical);
297 if (!canonical) {
298 return CellState::Invalid;
299 }
300 exact[row] = *canonical;
301 return CellState::Valid;
302 }
303 if (auto const* value = std::get_if<std::int64_t>(&cell)) {
304 exact[row] = math::Rational{*value, math::DecimalPlaces{0}};
305 return CellState::Valid;
306 }
307 if (auto const* value = std::get_if<std::string>(&cell)) {
308 if (auto const parsed = parseDecimal(*value)) {
309 exact[row] = *parsed;
310 return CellState::Valid;
311 }
312 }
313 return CellState::Invalid;
314 }
315
316 [[nodiscard]] CellState readReal(std::size_t row, Cell const& cell) {
317 std::optional<double> value;
318 if (auto const* real = std::get_if<double>(&cell)) {
319 value = *real;
320 } else if (auto const* integer = std::get_if<std::int64_t>(&cell)) {
321 value = static_cast<double>(*integer);
322 } else if (auto const* exactValue = std::get_if<math::Rational>(&cell)) {
323 value = exactValue->toDouble(math::kMaxDecimalPlaces + 1);
324 } else if (auto const* textValue = std::get_if<std::string>(&cell)) {
325 value = detail::parseReal(*textValue);
326 }
327 if (!value || std::isnan(*value)) {
328 return CellState::Invalid;
329 }
330 reals[row] = *value;
331 return CellState::Valid;
332 }
333
334 [[nodiscard]] CellState readDate(std::size_t row, Cell const& cell, ColumnInfo const& info,
335 Services const& services) {
336 if (auto const* value = std::get_if<std::int64_t>(&cell)) {
337 ints[row] = *value;
338 return CellState::Valid;
339 }
340 if (auto const* value = std::get_if<std::string>(&cell)) {
341 auto const& parser = detail::datesOf(services);
342 auto const parsed = kind == ColumnKind::Date ? parser.parseDate(*value, info.format)
343 : parser.parseDateTime(*value, info.format);
344 if (parsed) {
345 ints[row] = *parsed;
346 return CellState::Valid;
347 }
348 }
349 return CellState::Invalid;
350 }
351
352 [[nodiscard]] CellState readBool(std::size_t row, Cell const& cell) {
353 if (auto const* value = std::get_if<bool>(&cell)) {
354 ints[row] = *value ? 1 : 0;
355 return CellState::Valid;
356 }
357 if (auto const* value = std::get_if<std::int64_t>(&cell); value != nullptr && (*value == 0 || *value == 1)) {
358 ints[row] = *value;
359 return CellState::Valid;
360 }
361 if (auto const* value = std::get_if<std::string>(&cell);
362 value != nullptr && (*value == "true" || *value == "false")) {
363 ints[row] = *value == "true" ? 1 : 0;
364 return CellState::Valid;
365 }
366 return CellState::Invalid;
367 }

396/// @brief Compares two rows' keys in one column, valid before invalid in
397/// either direction.
398/// @param keys The column's keys.
399/// @param left One row.
400/// @param right Another row.
401/// @param dir The key's direction; it does not move invalid cells.
402/// @return Negative, zero or positive as row @p left orders before, with, or after @p right.
403[[nodiscard]] inline int compareKeys(KeyColumn const& keys, std::uint32_t left, std::uint32_t right,
404 SortDirection dir) {
405 bool const validA = keys.states[left] == CellState::Valid;
406 bool const validB = keys.states[right] == CellState::Valid;
407 if (validA != validB) {
408 return validA ? -1 : 1;
409 }
410 if (!validA) {
411 return 0;
412 }
413 auto const sign = [](auto ordering) -> int {
414 if (std::is_lt(ordering)) {
415 return -1;
416 }
417 return std::is_gt(ordering) ? 1 : 0;
418 };
419 int result = 0;
420 switch (keys.kind) {
421 case ColumnKind::Integer:
422 case ColumnKind::Key:
423 case ColumnKind::Date:
424 case ColumnKind::DateTime:
425 case ColumnKind::Bool:
426 result = sign(keys.ints[left] <=> keys.ints[right]);
427 break;
428 case ColumnKind::Decimal:
429 case ColumnKind::Quantity:
430 result = sign(keys.exact[left] <=> keys.exact[right]);
431 break;
432 case ColumnKind::Number:
433 result = sign(std::strong_order(keys.reals[left], keys.reals[right]));
434 break;
435 case ColumnKind::Text:
436 result = sign(keys.text[left].compare(keys.text[right]) <=> 0);
437 break;
438 case ColumnKind::Custom:
439 result = keys.usesComparator() ? sign(keys.comparator(keys.cells[left], keys.cells[right]))
440 : sign(keys.text[left].compare(keys.text[right]) <=> 0);
441 break;
442 default:
443 break;
444 }
445 return dir == SortDirection::Descending ? -result : result;
446}

448/// @brief A sort chain resolved to key columns: the comparator of a sort.
449///
450/// Holds pointers to key columns it does not own; whoever builds it keeps the
451/// columns alive for as long as it is used.
452class RowOrder {
453public:
454 /// @brief One resolved key.
455 struct Key {
456 /// @brief The column's keys.
457 KeyColumn const* keys = nullptr;
458 /// @brief The key's direction.
459 SortDirection dir = SortDirection::Ascending;
460 };
461
462 /// @brief Source order.
463 RowOrder() = default;
464
465 /// @brief An order over resolved keys, most significant first.
466 /// @param keys The keys.
467 explicit RowOrder(std::vector<Key> keys) : _keys{std::move(keys)} {}
468
469 /// @brief Whether row @p a orders before row @p b. Ties break by source
470 /// row, so this is a strict total order and every sort is stable.
471 /// @param left One row.
472 /// @param right Another row.
473 /// @return `true` when @p left comes first.
474 [[nodiscard]] bool operator()(std::uint32_t left, std::uint32_t right) const {
475 for (auto const& key : _keys) {
476 if (int const result = compareKeys(*key.keys, left, right, key.dir); result != 0) {
477 return result < 0;
478 }
479 }
480 return left < right;
481 }
482
483 /// @brief Whether this is source order.
484 /// @return `true` when no key is set.
485 [[nodiscard]] bool empty() const noexcept { return _keys.empty(); }
486
487private:
488 std::vector<Key> _keys;
489};

491/// @brief A bottom-up merge sort over row indices that can stop and resume.
492///
493/// Runs of `kRun` rows are sorted first, then merged pass by pass. `run`
494/// returns at a deadline or a stop request and continues from there on the
495/// next call, which is how the engine sorts 100,000 rows on an owner thread
496/// in frame-sized steps, or abandons a superseded sort on a worker.
497class ChunkedMergeSort {
498public:
499 /// @brief Rows per initial run.
500 static constexpr std::size_t kRun = 512;
501
502 /// @brief Prepares to sort @p rows by @p order.
503 /// @param rows Row indices to sort.
504 /// @param order The comparator; its key columns must outlive the sort.
505 ChunkedMergeSort(std::vector<std::uint32_t> rows, RowOrder order)
506 : _rows{std::move(rows)}, _order{std::move(order)} {
507 _buffer.resize(_rows.size());
508 }
509
510 /// @brief Sorts until done, the deadline passes, or a stop is requested.
511 /// @param deadline When to yield.
512 /// @param stop A stop request ends the step early.
513 /// @return `true` when the rows are sorted.
514 bool run(Deadline deadline, ::core::async::StopToken const& stop) {
515 return sortRuns(deadline, stop) && mergePasses(deadline, stop);
516 }
517
518 /// @brief Whether the sort has finished.
519 /// @return `true` once `run` returned `true`.
520 [[nodiscard]] bool done() const noexcept { return _runStart >= _rows.size() && _width >= _rows.size(); }
521
522 /// @brief The sorted rows. Valid once `done()`.
523 /// @return The rows, moved out.
524 [[nodiscard]] std::vector<std::uint32_t> take() && { return std::move(_rows); }
525
526private:
527 // Phase one: sorts each run of kRun rows in place.
528 bool sortRuns(Deadline deadline, ::core::async::StopToken const& stop) {
529 auto const count = _rows.size();
530 std::size_t work = 0;
531 while (_runStart < count) {
532 auto const end = std::min(_runStart + kRun, count);
533 std::sort(_rows.begin() + static_cast<std::ptrdiff_t>(_runStart),
534 _rows.begin() + static_cast<std::ptrdiff_t>(end), _order);
535 work += end - _runStart;
536 _runStart = end;
537 if (work >= detail::kCheckEvery && detail::shouldYield(deadline, stop)) {
538 return false;
539 }
540 }
541 return true;
542 }
543
544 // Phase two: merges runs pass by pass, doubling their width.
545 bool mergePasses(Deadline deadline, ::core::async::StopToken const& stop) {
546 auto const count = _rows.size();
547 while (_width < count) {
548 while (_mergeAt < count) {
549 if (!mergeOne(deadline, stop)) {
550 return false;
551 }
552 }
553 _rows.swap(_buffer);
554 _width *= 2;
555 _mergeAt = 0;
556 if (_width < count && detail::shouldYield(deadline, stop)) {
557 return false;
558 }
559 }
560 return true;
561 }
562
563 // Merges the two runs at `_mergeAt` into the buffer, resuming a merge a
564 // previous step left half done.
565 bool mergeOne(Deadline deadline, ::core::async::StopToken const& stop) {
566 auto const count = _rows.size();
567 if (!_merging) {
568 _left = _mergeAt;
569 _leftEnd = std::min(_mergeAt + _width, count);
570 _right = _leftEnd;
571 _rightEnd = std::min(_mergeAt + (2 * _width), count);
572 _out = _mergeAt;
573 _merging = true;
574 }
575 // The count runs across merges: a pass of narrow runs merges many
576 // short pairs, and a per-merge count would never reach a check.
577 while (_left < _leftEnd && _right < _rightEnd) {
578 _buffer[_out++] = _order(_rows[_right], _rows[_left]) ? _rows[_right++] : _rows[_left++];
579 if (++_work % detail::kCheckEvery == 0 && detail::shouldYield(deadline, stop)) {
580 return false;
581 }
582 }
583 std::copy(_rows.begin() + static_cast<std::ptrdiff_t>(_left),
584 _rows.begin() + static_cast<std::ptrdiff_t>(_leftEnd),
585 _buffer.begin() + static_cast<std::ptrdiff_t>(_out));
586 _out += _leftEnd - _left;
587 _left = _leftEnd;
588 std::copy(_rows.begin() + static_cast<std::ptrdiff_t>(_right),
589 _rows.begin() + static_cast<std::ptrdiff_t>(_rightEnd),
590 _buffer.begin() + static_cast<std::ptrdiff_t>(_out));
591 _out += _rightEnd - _right;
592 _right = _rightEnd;
593 _merging = false;
594 _mergeAt = _rightEnd;
595 return true;
596 }
597
598 std::vector<std::uint32_t> _rows;
599 std::vector<std::uint32_t> _buffer;
600 RowOrder _order;
601 std::size_t _runStart = 0;
602 std::size_t _width = kRun;
603 std::size_t _mergeAt = 0;
604 bool _merging = false;
605 std::size_t _left = 0;
606 std::size_t _leftEnd = 0;
607 std::size_t _right = 0;
608 std::size_t _rightEnd = 0;
609 std::size_t _out = 0;
610 std::size_t _work = 0;
611};

613/// @brief Sorts @p rows by @p order in one call.
614/// @param rows Row indices.
615/// @param order The comparator.
616/// @return The sorted rows.
617[[nodiscard]] inline std::vector<std::uint32_t> sortRows(std::vector<std::uint32_t> rows, RowOrder order) {
618 ChunkedMergeSort sort{std::move(rows), std::move(order)};
619 static_cast<void>(sort.run(kNoDeadline, ::core::async::StopToken{}));
620 return std::move(sort).take();
621}
