32template <class A>
33FormModel FormModel::forAction() {
34 std::string_view const typeId = model::ActionTraits<A>::typeId();
35 auto parsed = fromSchema(typeId, schemaJson<A>());
36 if (!parsed) {
37 throw std::logic_error{"FormModel::forAction: the schema of '" + std::string{typeId} +
38 "' does not read as a form (" + parsed.error().path + ": " + parsed.error().message +
39 ")"};
40 }
41 return *std::move(parsed);
42}

44/// @brief A form over action `A`: a `FormSession` built from `schemaJson<A>()`, submitting
45/// through a `BridgeHandler`, with typed `set<>`, `value()` and `lastResult()`.
46///
47/// Non-copyable and non-movable: its session and in-flight callbacks point into it.
48/// @tparam A The action type (registered with `BRIDGE_REGISTER_ACTION`).
49/// @tparam M The model type the handler talks to.
50/// @tparam S The handler's sharing policy.
51template <class A, class M, class S = bridge::NoSharing>
52class Form {
53public:
54 /// @brief The action's registered result type.
55 using Result = model::ActionTraits<A>::Result;
56
57 /// @brief The type of the member @p Member points to.
58 /// @tparam Member A pointer to a data member of `A`.
59 template <auto Member>
60 using MemberValue = bridge::detail::MemberPointerTraits<decltype(Member)>::ValueType;
61
62 /// @param runtime The runtime; its owner must be @p handler's callback executor. Borrowed.
63 /// @param handler Executes the action. Borrowed: it must outlive the form.
64 /// @param choices Fetches Choice options; may be empty when `A` has no Choice member.
65 /// @param options Locale, translations, display zone; an `accepts` gate is kept and joined
66 /// with `ActionValidator<A>`.
67 /// @throws std::logic_error from `FormModel::forAction<A>()` when `schemaJson<A>()` does not
68 /// read as a form model.
69 Form(reactive::Runtime& runtime MORPH_LIFETIMEBOUND, bridge::BridgeHandler<M, S>& handler MORPH_LIFETIMEBOUND,
70 ChoiceFetcher choices, FormSessionOptions options = {})
71 : _rt{&runtime},
72 _handler{&handler},
73 _session{runtime, FormModel::forAction<A>(),
74 [this](std::string_view, const std::string& body) { return execute(body); }, std::move(choices),
75 withValidator(std::move(options))},
76 _value{runtime,
77 [this] -> std::optional<A> {
78 std::optional<std::string> const body = _session.body();
79 return body.has_value() ? decode(*body) : std::nullopt;
80 }},
81 _lastResult{runtime, [this] -> std::optional<Result> {
82 std::optional<std::string> const& reply = _session.lastReply();
83 if (!reply.has_value()) {
84 return std::nullopt;
85 }
86 try {
87 return model::ActionTraits<A>::resultFromJson(*reply);
88 } catch (...) {
89 return std::nullopt;
90 }
91 }} {}
92
93 ~Form() = default;
94 Form(Form const&) = delete;
95 Form& operator=(Form const&) = delete;
96 Form(Form&&) = delete;
97 Form& operator=(Form&&) = delete;
98
99 /// @brief The session: fields, readiness, body, submission.
100 /// @return The session.
101 [[nodiscard]] FormSession& session() noexcept { return _session; }
102
103 /// @brief Sets one member, encoded with its field's encoder; a programmatic change, so
104 /// nothing is submitted.
105 /// @tparam Member A pointer to a data member of `A`, e.g. `&Deposit::amount`.
106 /// @param value The member's value.
107 template <auto Member>
108 void set(MemberValue<Member> value) {
109 static_assert(std::is_same_v<typename bridge::detail::MemberPointerTraits<decltype(Member)>::ClassType, A>,
110 "Form<A>::set<>: the member must belong to A");
111 _session.assign(detail::memberWireName<Member>(), glz::write_json(value).value_or(std::string{"null"}));
112 }
113
114 /// @brief The body decoded into `A`. Tracked.
115 /// @return The action while the form is ready, else `std::nullopt`.
116 [[nodiscard]] std::optional<A> value() const { return _value.get(); }
117
118 /// @brief Whether the form can submit: the session is ready, which includes `A`'s own
119 /// `validate()`. Tracked.
120 /// @return `true` when ready.
121 [[nodiscard]] bool ready() const { return _session.ready(); }
122
123 /// @brief The last successful result: the session's last reply, decoded. Tracked.
124 /// @return The typed result, or `std::nullopt` before the first success.
125 [[nodiscard]] std::optional<Result> const& lastResult() const { return _lastResult.get(); }
126
127 /// @brief Whether a submission is in flight. Tracked.
128 /// @return `true` while one is.
129 [[nodiscard]] bool pending() const { return _session.pending(); }
130
131 /// @brief The last failure. Tracked.
132 /// @return The exception, or null.
133 [[nodiscard]] std::exception_ptr error() const { return _session.lastError(); }
134
135private:
136 [[nodiscard]] static std::optional<A> decode(std::string const& body) {
137 try {
138 return model::ActionTraits<A>::fromJson(body);
139 } catch (...) {
140 return std::nullopt;
141 }
142 }
143
144 [[nodiscard]] static FormSessionOptions withValidator(FormSessionOptions options) {
145 options.accepts = [own = std::move(options.accepts)](std::string_view body) {
146 if (own && !own(body)) {
147 return false;
148 }
149 std::optional<A> const action = decode(std::string{body});
150 return action.has_value() && model::ActionValidator<A>::ready(*action);
151 };
152 return options;
153 }
154
155 [[nodiscard]] async::Completion<std::string> execute(std::string const& body) {
156 auto [completion, promise] = async::Completion<std::string>::makeSettleable(&_rt->owner());
157 auto const relay = std::make_shared<async::Completion<std::string>::Promise>(std::move(promise));
158 std::optional<A> action = decode(body);
159 if (!action) {
160 relay->reject(std::make_exception_ptr(std::invalid_argument{"Form<A>: the body does not decode"}));
161 return std::move(completion);
162 }
163 _handler->execute(*std::move(action))
164 .then(_relay,
165 [relay](Result const& result) {
166 try {
167 std::string reply = model::ActionTraits<A>::resultToJson(result);
168 relay->resolve(std::move(reply));
169 } catch (...) {
170 relay->reject(std::current_exception());
171 }
172 })
173 .onError(_relay, [relay](std::exception_ptr error) { relay->reject(std::move(error)); });
174 return std::move(completion);
175 }
176
177 reactive::Runtime* _rt;
178 bridge::BridgeHandler<M, S>* _handler;
179 // Before the session: a session ready as constructed submits from its constructor, and the
180 // handler callbacks that submission attaches are gated by this scope.
181 async::CallbackScope _relay;
182 FormSession _session;
183 reactive::Computed<std::optional<A>> _value;
184 // The session's last reply, decoded: the typed result and the JSON reply are one value, seen at once.
185 reactive::Computed<std::optional<Result>> _lastResult;
186};
