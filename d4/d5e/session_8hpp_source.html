13/// @brief Per-call session context carried through the bridge to the model.
14///
15/// `Context` is an open data bag: callers populate whichever fields make sense
16/// for the app (user identity, request tracing, locale, app-specific metadata).
17/// The framework only inspects `principal` and the action ids when consulting
18/// the configured `IAuthorizer`; everything else is passed through verbatim
19/// from caller to model.
20///
21/// On remote backends the `Context` is serialised into the wire envelope, so
22/// the server-side `RemoteServer` sees the same values the GUI sent. On the
23/// local backend it travels in-memory via the `ActionCall`.
24struct Context {
25 /// @brief Auth principal (user id).
26 ///
27 /// On the client this is whatever the caller sets and is **not** trustworthy
28 /// on its own — it is untrusted wire input. When a verifying authorizer (e.g.
29 /// `SigningAuthorizer`, `session_auth.hpp`) is installed, `RemoteServer`
30 /// **overwrites** this field with the principal extracted from a valid
31 /// `token` before dispatch, so `session::current()->principal` read inside a
32 /// model is the authenticated identity, not the client's claim.
33 std::string principal;
34
35 /// @brief Bearer credential verified server-side (empty if unauthenticated).
36 ///
37 /// Typically a signed token minted by a login action via
38 /// `session::TokenIssuer` and attached to every call (see `session_auth.hpp`
39 /// and `docs/spec/security.md`). Travels in the wire envelope's `session`.
40 std::string token;
41
42 /// @brief Stable id used for distributed tracing / log correlation. Empty if unused.
43 std::string requestId;
44
45 /// @brief BCP-47 locale tag (e.g. `en-US`, `fr-FR`) for i18n. Empty for app default.
46 std::string locale;
47
48 /// @brief Free-form bag of string→string metadata (feature flags, A/B buckets, …).
49 std::unordered_map<std::string, std::string> metadata;
50};

52/// @brief Authorizes incoming actions on a `RemoteServer`.
53///
54/// Called once per `execute` envelope, before the action is dispatched. A `false`
55/// return causes the server to reply with `err|unauthorized` (the client surfaces
56/// the error through the `.onError(...)` callback).
57///
58/// Default implementation supplied by the framework is `AllowAllAuthorizer`. Real
59/// deployments install a custom subclass that checks principal claims, action
60/// permissions, rate limits, etc.
61// NOLINTBEGIN(cppcoreguidelines-special-member-functions)
62struct IAuthorizer {
63 virtual ~IAuthorizer() = default;
64
65 /// @brief Returns `true` if @p ctx is allowed to invoke @p actionType on @p modelType.
66 ///
67 /// @param ctx Per-call session attached by the client.
68 /// @param modelType String id of the target model type.
69 /// @param actionType String id of the action being invoked.
70 /// @return `true` to allow dispatch, `false` to reject with `err|unauthorized`.
71 [[nodiscard]] virtual bool authorize(const Context& ctx,
72 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
73 std::string_view modelType,
74 std::string_view actionType) const = 0;
75
76 /// @brief Returns the authenticated principal for @p ctx, or `nullopt`.
77 ///
78 /// Called by `RemoteServer` after `authorize` succeeds. If it returns a
79 /// value, the server overwrites `Context::principal` with it before dispatch,
80 /// making the identity authoritative for model code that reads
81 /// `session::current()`. The default returns `nullopt` — for authorizers that
82 /// do not authenticate (e.g. `AllowAllAuthorizer`), `RemoteServer` **clears**
83 /// `Context::principal` to the empty string before dispatch rather than
84 /// passing the client's unverified claim through, so model code never sees an
85 /// unauthenticated principal presented as authoritative. A verifying authorizer
86 /// (`SigningAuthorizer`) overrides this to supply the token-derived identity.
87 /// @param ctx Per-call session attached by the client.
88 /// @return The verified principal to make authoritative, or `nullopt`.
89 [[nodiscard]] virtual std::optional<std::string> authenticate([[maybe_unused]] const Context& ctx) const {
90 return std::nullopt;
91 }
92
93 /// @brief Optional per-**instance** authorization hook — the multi-tenant gate.
94 ///
95 /// `authorize` sees only the model *type*, so it cannot answer "may this
96 /// caller touch *this instance* (row)?". Model instances on a `RemoteServer`
97 /// are addressable by guessable sequential ids, so without an ownership check
98 /// any authenticated caller can `execute`/`deregister` against an id it did
99 /// not create — a cross-tenant targeting gap. This hook closes it: when
100 /// installed, `RemoteServer` consults it on every `execute` **and** every
101 /// `deregister`, passing the id of the target instance and the principal
102 /// recorded as its owner at `register` time (empty if the instance was
103 /// registered by an unauthenticated caller or predates ownership tracking).
104 ///
105 /// The **default allows everything**, so an authorizer that does not override
106 /// it — including `AllowAllAuthorizer` and a plain `SigningAuthorizer` — keeps
107 /// the pre-existing behaviour exactly (no per-instance restriction). A
108 /// deployer opts into ownership enforcement by overriding this, typically to
109 /// compare @p ownerPrincipal against `ctx.principal`:
110 /// @code
111 /// bool authorizeInstance(const Context& ctx, std::string_view, std::string_view,
112 /// uint64_t, std::string_view ownerPrincipal) const override {
113 /// return ownerPrincipal.empty() || ownerPrincipal == ctx.principal;
114 /// }
115 /// @endcode
116 ///
117 /// @param ctx Per-call session (its `principal` is authoritative
118 /// only if a verifying authorizer is installed).
119 /// @param modelType Target model type id (empty for `deregister`).
120 /// @param actionType Target action type id (empty for `deregister`).
121 /// @param modelId Numeric id of the target instance.
122 /// @param ownerPrincipal Principal recorded as the instance's owner at
123 /// `register` time; empty if none was recorded.
124 /// @return `true` to allow the operation on this instance, `false` to reject.
125 [[nodiscard]] virtual bool authorizeInstance([[maybe_unused]] const Context& ctx,
126 [[maybe_unused]] std::string_view modelType,
127 [[maybe_unused]] std::string_view actionType,
128 [[maybe_unused]] std::uint64_t modelId,
129 [[maybe_unused]] std::string_view ownerPrincipal) const {
130 return true;
131 }
132};
133// NOLINTEND(cppcoreguidelines-special-member-functions)

135/// @brief Default authorizer that permits everything.
136///
137/// Wire it explicitly via `RemoteServer(pool, dispatcher, registry, allowAll)`
138/// for documentation, or rely on the server's default (which uses this type).
139struct AllowAllAuthorizer : IAuthorizer {
140 /// @brief Permits every call.
141 /// @param ctx Ignored.
142 /// @param modelType Ignored.
143 /// @param actionType Ignored.
144 /// @return Always `true`.
145 [[nodiscard]] bool authorize([[maybe_unused]] const Context& ctx,
146 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
147 [[maybe_unused]] std::string_view modelType,
148 [[maybe_unused]] std::string_view actionType) const override {
149 return true;
150 }
151};

161/// @brief Thread-local pointer to the `Context` for the currently dispatched action.
162///
163/// Installed by the backend around the model call — `LocalBackend::execute` on the
164/// local path and `RemoteServer::dispatchExecute` on the remote path — via a
165/// `ScopedContext` that clears it again on scope exit. `ActionDispatcher::dispatch`
166/// itself does not touch this pointer. Models that opt in to session-aware logic
167/// call `current()` to read it.
168inline const Context*& tlsCurrent() {
169 thread_local const Context* tls = nullptr;
170 return tls;
171}
172
173/// @brief RAII helper that sets the thread-local `Context` for its scope.
174class ScopedContext {
175public:
176 /// @brief Installs @p ctx as the thread-local context until the scope exits.
177 /// @param ctx Context whose address is stored; must outlive this object.
178 explicit ScopedContext(const Context& ctx) : _prev{tlsCurrent()} { tlsCurrent() = &ctx; }
179 /// @brief Restores the previously active thread-local context.
180 ~ScopedContext() { tlsCurrent() = _prev; }
181 ScopedContext(const ScopedContext&) = delete;
182 ScopedContext& operator=(const ScopedContext&) = delete;
183 ScopedContext(ScopedContext&&) = delete;
184 ScopedContext& operator=(ScopedContext&&) = delete;
185
186private:
187 const Context* _prev;
188};

192/// @brief Returns the active `Context` for the in-progress action, or `nullptr` if none.
193///
194/// Models that don't need session data can ignore this entirely. Models that do
195/// can read principal/locale/metadata without changing their `execute()` signature.
196inline const Context* current() noexcept {
197 return detail::tlsCurrent();
