15/// @brief Per-call session context carried through the bridge to the model.
16///
17/// `Context` is an open data bag: callers populate whichever fields make sense
18/// for the app (user identity, request tracing, locale, app-specific metadata).
19/// The framework only inspects `principal` and the action ids when consulting
20/// the configured `IAuthorizer`; everything else is passed through verbatim
21/// from caller to model.
22///
23/// On remote backends the `Context` is serialised into the wire envelope, so
24/// the server-side `RemoteServer` sees the same values the GUI sent. On the
25/// local backend it travels in-memory via the `ActionCall`.
26struct Context {
27 /// @brief Auth principal (user id).
28 ///
29 /// On the client this is whatever the caller sets and is **not** trustworthy
30 /// on its own — it is untrusted wire input. When a verifying authorizer (e.g.
31 /// `SigningAuthorizer`, `session_auth.hpp`) is installed, `RemoteServer`
32 /// **overwrites** this field with the principal extracted from a valid
33 /// `token` before dispatch, so `session::current()->principal` read inside a
34 /// model is the authenticated identity, not the client's claim.
35 std::string principal;
36
37 /// @brief Bearer credential verified server-side (empty if unauthenticated).
38 ///
39 /// Typically a signed token minted by a login action via
40 /// `session::TokenIssuer` and attached to every call (see `session_auth.hpp`
41 /// and `docs/spec/security.md`). Travels in the wire envelope's `session`.
42 std::string token;
43
44 /// @brief Stable id used for distributed tracing / log correlation. Empty if unused.
45 std::string requestId;
46
47 /// @brief BCP-47 locale tag (e.g. `en-US`, `fr-FR`) for i18n. Empty for app default.
48 std::string locale;
49
50 /// @brief Free-form bag of string→string metadata (feature flags, A/B buckets, …).
51 std::unordered_map<std::string, std::string> metadata;
52};

54/// @brief The signed-in user's verified identity, readable outside a dispatch.
55///
56/// `Context::principal` only exists *during* a dispatch — `session::current()`
57/// returns `nullptr` outside one, so UI code (a button's enabled state, a menu
58/// item's visibility) has no way to ask "who is signed in and what may they do?"
59/// without attempting the action and catching the failure. `Principal` is the
60/// longer-lived counterpart an application installs once, typically right
61/// after a successful login dispatch, and reads from UI code to shape itself
62/// instead: `bridge.currentPrincipal().hasRole("editor")`.
63///
64/// Deliberately **not** a process-wide global: the caller installs it on the
65/// specific `Bridge` (or other session-scoped object) whose backend it came
66/// from — see `Bridge::setPrincipal`/`Bridge::currentPrincipal`
67/// (`core/bridge.hpp`) — rather than one ambient value shared by every backend
68/// a process happens to hold.
69///
70/// @warning Populate this only from data the server actually returned (e.g. an
71/// action's `Result`, such as the bank example's `AuthResult`), never from a
72/// client-side guess — the server remains the sole authority for what a
73/// principal may do; this is a read-only cache of what it last told the
74/// client, not a second, independently-decided permission set. Roles can
75/// change server-side mid-session, so every dispatch is still authorized
76/// there regardless of what a stale `Principal` says client-side; this only
77/// shapes the UI, it never replaces server-side authorization.
78struct Principal {
79 /// @brief Verified identity (e.g. username), as returned by the server. Empty if signed out.
80 std::string id;
81
82 /// @brief Coarse-grained roles the app can gate UI on (e.g. `"editor"`, `"admin"`).
83 std::vector<std::string> roles;
84
85 /// @brief Free-form bag of string→string claims beyond `id`/`roles`.
86 std::unordered_map<std::string, std::string> claims;
87
88 /// @brief Returns `true` if `roles` contains @p role.
89 /// @param role Role name to look for.
90 /// @return `true` if @p role is present in `roles`.
91 [[nodiscard]] bool hasRole(std::string_view role) const { return std::ranges::find(roles, role) != roles.end(); }
92};

94/// @brief Authorizes incoming actions on a `RemoteServer`.
95///
96/// Called once per `execute` envelope, before the action is dispatched. A `false`
97/// return causes the server to reply with `err|unauthorized` (the client surfaces
98/// the error through the `.onError(...)` callback).
99///
100/// Default implementation supplied by the framework is `AllowAllAuthorizer`. Real
101/// deployments install a custom subclass that checks principal claims, action
102/// permissions, rate limits, etc.
103// NOLINTBEGIN(cppcoreguidelines-special-member-functions)
104struct IAuthorizer {
105 virtual ~IAuthorizer() = default;
106
107 /// @brief Returns `true` if @p ctx is allowed to invoke @p actionType on @p modelType.
108 ///
109 /// @param ctx Per-call session attached by the client.
110 /// @param modelType String id of the target model type.
111 /// @param actionType String id of the action being invoked.
112 /// @return `true` to allow dispatch, `false` to reject with `err|unauthorized`.
113 [[nodiscard]] virtual bool authorize(const Context& ctx,
114 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
115 std::string_view modelType, std::string_view actionType) const = 0;
116
117 /// @brief Returns the authenticated principal for @p ctx, or `nullopt`.
118 ///
119 /// Called by `RemoteServer` after `authorize` succeeds. If it returns a
120 /// value, the server overwrites `Context::principal` with it before dispatch,
121 /// making the identity authoritative for model code that reads
122 /// `session::current()`. The default returns `nullopt` — for authorizers that
123 /// do not authenticate (e.g. `AllowAllAuthorizer`), `RemoteServer` **clears**
124 /// `Context::principal` to the empty string before dispatch rather than
125 /// passing the client's unverified claim through, so model code never sees an
126 /// unauthenticated principal presented as authoritative. A verifying authorizer
127 /// (`SigningAuthorizer`) overrides this to supply the token-derived identity.
128 /// @param ctx Per-call session attached by the client.
129 /// @return The verified principal to make authoritative, or `nullopt`.
130 [[nodiscard]] virtual std::optional<std::string> authenticate([[maybe_unused]] const Context& ctx) const {
131 return std::nullopt;
132 }
133
134 /// @brief Optional per-**instance** authorization hook — the multi-tenant gate.
135 ///
136 /// `authorize` sees only the model *type*, so it cannot answer "may this
137 /// caller touch *this instance* (row)?". Model instances on a `RemoteServer`
138 /// are addressable by guessable sequential ids, so without an ownership check
139 /// any authenticated caller can `execute`/`deregister` against an id it did
140 /// not create — a cross-tenant targeting gap. This hook closes it: when
141 /// installed, `RemoteServer` consults it on every `execute` **and** every
142 /// `deregister`, passing the id of the target instance and the principal
143 /// recorded as its owner at `register` time (empty if the instance was
144 /// registered by an unauthenticated caller or predates ownership tracking).
145 ///
146 /// The **default allows everything**, so an authorizer that does not override
147 /// it — including `AllowAllAuthorizer` and a plain `SigningAuthorizer` — keeps
148 /// the pre-existing behaviour exactly (no per-instance restriction). A
149 /// deployer opts into ownership enforcement by overriding this, typically to
150 /// compare @p ownerPrincipal against `ctx.principal`:
151 /// @code
152 /// bool authorizeInstance(const Context& ctx, std::string_view, std::string_view,
153 /// uint64_t, std::string_view ownerPrincipal) const override {
154 /// return ownerPrincipal.empty() || ownerPrincipal == ctx.principal;
155 /// }
156 /// @endcode
157 ///
158 /// @param ctx Per-call session (its `principal` is authoritative
159 /// only if a verifying authorizer is installed).
160 /// @param modelType Target model type id (empty for `deregister`).
161 /// @param actionType Target action type id (empty for `deregister`).
162 /// @param modelId Numeric id of the target instance.
163 /// @param ownerPrincipal Principal recorded as the instance's owner at
164 /// `register` time; empty if none was recorded.
165 /// @return `true` to allow the operation on this instance, `false` to reject.
166 [[nodiscard]] virtual bool authorizeInstance([[maybe_unused]] const Context& ctx,
167 [[maybe_unused]] std::string_view modelType,
168 [[maybe_unused]] std::string_view actionType,
169 [[maybe_unused]] std::uint64_t modelId,
170 [[maybe_unused]] std::string_view ownerPrincipal) const {
171 return true;
172 }
173
174 /// @brief Optional gate on model **creation** — bounds *who may create* an instance.
175 ///
176 /// `authorize` and `authorizeInstance` both act on an already-existing
177 /// instance; neither can answer "may this caller create one at all?".
178 /// `RemoteServer` consults this hook on every `register` envelope, after
179 /// authenticating the caller (so @p ctx carries the *verified* principal,
180 /// not the client's raw claim) and before constructing the instance. A
181 /// `false` return replies `err "unauthorized"` and **no instance is
182 /// created** — `ModelRegistryFactory::create` never runs.
183 ///
184 /// The **default allows everything**, so an authorizer that does not
185 /// override it — including `AllowAllAuthorizer` and a plain
186 /// `SigningAuthorizer` — keeps the pre-existing behaviour exactly (any
187 /// reachable client may register any known model type). A deployer opts
188 /// into bounding registration by overriding this, typically requiring
189 /// authentication and/or restricting @p modelType:
190 /// @code
191 /// bool authorizeRegister(const Context& ctx, std::string_view modelType) const override {
192 /// return !ctx.principal.empty(); // only authenticated callers may register
193 /// }
194 /// @endcode
195 ///
196 /// @param ctx Per-call session for the register envelope; its
197 /// `principal` is already the verified identity when the
198 /// authorizer authenticates (empty otherwise).
199 /// @param modelType String id of the model type being registered.
200 /// @return `true` to allow the registration, `false` to reject with
201 /// `err "unauthorized"`.
202 [[nodiscard]] virtual bool authorizeRegister([[maybe_unused]] const Context& ctx,
203 [[maybe_unused]] std::string_view modelType) const {
204 return true;
205 }
206};
207// NOLINTEND(cppcoreguidelines-special-member-functions)

209/// @brief Default authorizer that permits everything.
210///
211/// Wire it explicitly via `RemoteServer(pool, dispatcher, registry, allowAll)`
212/// for documentation, or rely on the server's default (which uses this type).
213struct AllowAllAuthorizer : IAuthorizer {
214 /// @brief Permits every call.
215 /// @param ctx Ignored.
216 /// @param modelType Ignored.
217 /// @param actionType Ignored.
218 /// @return Always `true`.
219 [[nodiscard]] bool authorize([[maybe_unused]] const Context& ctx,
220 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
221 [[maybe_unused]] std::string_view modelType,
222 [[maybe_unused]] std::string_view actionType) const override {
223 return true;
224 }
225};

235/// @brief Thread-local pointer to the `Context` for the currently dispatched action.
236///
237/// Installed by the backend around the model call — `LocalBackend::execute` on the
238/// local path and `RemoteServer::dispatchExecute` on the remote path — via a
239/// `ScopedContext` that clears it again on scope exit. `ActionDispatcher::dispatch`
240/// itself does not touch this pointer. Models that opt in to session-aware logic
241/// call `current()` to read it.
242inline const Context*& tlsCurrent() {
243 thread_local const Context* tls = nullptr;
244 return tls;
245}
246
247/// @brief RAII helper that sets the thread-local `Context` for its scope.
248class ScopedContext {
249public:
250 /// @brief Installs @p ctx as the thread-local context until the scope exits.
251 /// @param ctx Context whose address is stored; must outlive this object.
252 explicit ScopedContext(const Context& ctx) : _prev{tlsCurrent()} { tlsCurrent() = &ctx; }
253 /// @brief Restores the previously active thread-local context.
254 ~ScopedContext() { tlsCurrent() = _prev; }
255 ScopedContext(const ScopedContext&) = delete;
256 ScopedContext& operator=(const ScopedContext&) = delete;
257 ScopedContext(ScopedContext&&) = delete;
258 ScopedContext& operator=(ScopedContext&&) = delete;
259
260private:
261 const Context* _prev;
262};

266/// @brief Returns the active `Context` for the in-progress action, or `nullptr` if none.
267///
268/// Models that don't need session data can ignore this entirely. Models that do
269/// can read principal/locale/metadata without changing their `execute()` signature.
270inline const Context* current() noexcept { return detail::tlsCurrent(); }
