40/// @brief How a collection presents its row editor.
41enum class CollectionKind : std::uint8_t {
42 Collection, ///< `v-kind: collection`: the editor is a dialog.
43 MasterDetail ///< `v-kind: master-detail`: the editor sits beside the table.
44};
45
46/// @brief Where an action's button sits.
47enum class CollectionActionScope : std::uint8_t {
48 Row, ///< On every row; the body is built from the row's `bind` entries.
49 Collection ///< Once, beside the title; the body is `{}`.
50};

52/// @brief One `v-columns` entry.
53struct CollectionColumn {
54 /// @brief The row member the column reads.
55 std::string field;
56 /// @brief The header: `label`, else `field`.
57 std::string label;
58 /// @brief The column's table kind (`kind`, by `table::ColumnKind`'s JSON names); `Text` when absent.
59 table::ColumnKind kind = table::ColumnKind::Text;
60 /// @brief `v-hidden`: carried in the rows and the table source, not shown.
61 bool hidden = false;
62 /// @brief `x-decimalPlaces`: the places an exact value without its own `dp` is shown with.
63 std::optional<std::uint32_t> decimalPlaces;
64 /// @brief The unit appended to an exact value (`ExtUnits.unitUnicode`, else `unitAscii`).
65 std::string unit;
66};

68/// @brief One `v-rowAction` or `v-actions` entry.
69struct CollectionAction {
70 /// @brief The action type executed.
71 std::string action;
72 /// @brief The button label: `label`, else the action type.
73 std::string label;

75 CollectionActionScope scope = CollectionActionScope::Row;

85/// @brief The immutable model of a `v-*` view document.
86class CollectionModel {
87public:
88 /// @brief Reads a view document.
89 /// @param viewJson The document, in the vocabulary `views::viewSchemaJson<V>()` emits.
90 /// @return The model, or the error naming the malformed key.
91 // NOLINTNEXTLINE(readability-function-cognitive-complexity): one check per `v-*` key, in document order.
92 [[nodiscard]] static std::expected<CollectionModel, SchemaError> fromSchema(std::string_view viewJson) {
93 using detail::engine::Json;
94 auto const dom = detail::engine::parseJson(viewJson);
95 if (!dom || !dom->is_object()) {
96 return std::unexpected(SchemaError{.path = "", .message = "a view document must be a JSON object"});
97 }
98 CollectionModel model{};
99 auto const* const kind = detail::engine::stringAt(*dom, "v-kind");
100 if (kind != nullptr && *kind != "collection" && *kind != "master-detail") {
101 return std::unexpected(
102 SchemaError{.path = "v-kind", .message = R"("v-kind" must be "collection" or "master-detail")"});
103 }
104 model._kind =
105 (kind != nullptr && *kind == "master-detail") ? CollectionKind::MasterDetail : CollectionKind::Collection;
106 auto const* const query = detail::engine::stringAt(*dom, "v-query");
107 if (query == nullptr) {
108 return std::unexpected(SchemaError{.path = "v-query", .message = R"(a view document needs a "v-query")"});
109 }
110 model._query = *query;
111 auto const* const title = detail::engine::stringAt(*dom, "v-title");
112 model._title = title != nullptr ? *title : model._query;
113 auto const* const rowKey = detail::engine::stringAt(*dom, "v-rowKey");
114 model._rowKey = rowKey != nullptr ? *rowKey : std::string{"id"};
115 if (Json const* const columns = detail::engine::member(*dom, "v-columns")) {
116 if (!columns->is_array()) {
117 return std::unexpected(SchemaError{.path = "v-columns", .message = R"("v-columns" must be an array)"});
118 }
119 for (Json const& entry : columns->get_array()) {
120 auto column = columnOf(entry);
121 if (!column) {
122 return std::unexpected(column.error());
123 }
124 model._columns.push_back(*std::move(column));
125 }
126 }
127 if (Json const* const rowAction = detail::engine::member(*dom, "v-rowAction")) {
128 auto action = actionOf(*rowAction, "v-rowAction");
129 if (!action) {
130 return std::unexpected(action.error());
131 }
132 model._rowAction = *std::move(action);
133 }
134 if (Json const* const actions = detail::engine::member(*dom, "v-actions")) {
135 if (!actions->is_array()) {
136 return std::unexpected(SchemaError{.path = "v-actions", .message = R"("v-actions" must be an array)"});
137 }
138 for (Json const& entry : actions->get_array()) {
139 auto action = actionOf(entry, "v-actions");
140 if (!action) {
141 return std::unexpected(action.error());
142 }
143 model._actions.push_back(*std::move(action));
144 }
145 }
146 return model;
147 }

180 [[nodiscard]] static std::expected<CollectionColumn, SchemaError> columnOf(detail::engine::Json const& entry) {
181 using detail::engine::Json;
182 auto const* const field = detail::engine::stringAt(entry, "field");
183 if (field == nullptr) {
184 return std::unexpected(SchemaError{.path = "v-columns", .message = R"(a column needs a "field")"});
185 }
186 CollectionColumn column{};
187 column.field = *field;
188 auto const* const label = detail::engine::stringAt(entry, "label");
189 column.label = label != nullptr ? *label : column.field;
190 if (auto const* const kind = detail::engine::stringAt(entry, "kind")) {
191 table::ColumnKind parsed{};
192 if (glz::read_json(parsed, detail::engine::quote(*kind))) {
193 return std::unexpected(SchemaError{.path = "v-columns." + column.field,
194 .message = "unknown column kind \"" + *kind + "\""});
195 }
196 column.kind = parsed;
197 }
198 Json const* const hidden = detail::engine::member(entry, "v-hidden");
199 column.hidden = hidden != nullptr && hidden->is_boolean() && hidden->get<bool>();
200 if (auto const places =
201 detail::engine::SchemaReader::unsignedOf(detail::engine::member(entry, "x-decimalPlaces"))) {
202 column.decimalPlaces =
203 static_cast<std::uint32_t>(std::min<std::uint64_t>(*places, math::kMaxDecimalPlaces));
204 }
205 if (Json const* const units = detail::engine::member(entry, "ExtUnits")) {
206 auto const* const unicode = detail::engine::stringAt(*units, "unitUnicode");
207 auto const* const ascii = detail::engine::stringAt(*units, "unitAscii");
208 if (unicode != nullptr && !unicode->empty()) {
209 column.unit = *unicode;
210 } else if (ascii != nullptr) {
211 column.unit = *ascii;
212 }
213 }
214 return column;
215 }
216
217 [[nodiscard]] static std::expected<CollectionAction, SchemaError> actionOf(detail::engine::Json const& node,
218 std::string const& path) {
219 auto const* const type = detail::engine::stringAt(node, "action");
220 if (type == nullptr) {
221 return std::unexpected(SchemaError{.path = path, .message = R"(an action needs an "action")"});
222 }
223 CollectionAction action{};
224 action.action = *type;
225 auto const* const label = detail::engine::stringAt(node, "label");
226 action.label = label != nullptr ? *label : action.action;
227 auto const* const scope = detail::engine::stringAt(node, "scope");
228 action.scope = (scope != nullptr && *scope == "collection") ? CollectionActionScope::Collection
229 : CollectionActionScope::Row;
230 auto const* const confirm = detail::engine::member(node, "confirm");
231 action.confirm = confirm != nullptr && confirm->is_boolean() && confirm->get<bool>();
232 if (auto const* const bind = detail::engine::member(node, "bind"); bind != nullptr && bind->is_object()) {
233 for (auto const& [field, rowField] : bind->get_object()) {
234 if (auto const* const text = rowField.get_if<std::string>()) {
235 action.bind.emplace_back(field, *text);
236 }
237 }
238 }
239 return action;
240 }

281/// @brief A cell's display text: an exact `{num, den}` value at its `dp` (else the column's places) with the
282/// column's unit, anything else as `displayText`, an absent or null member as empty.
283/// @param row The row.
284/// @param column The column.
285/// @return The text.
286[[nodiscard]] inline std::string cellText(Json const& row, CollectionColumn const& column) {
287 Json const* const value = member(row, column.field);
288 if (value == nullptr || value->is_null()) {
289 return {};
290 }
291 if (value->is_object()) {
292 auto const num = SchemaReader::signedOf(member(*value, "num"));
293 auto const den = SchemaReader::signedOf(member(*value, "den"));
294 if (num && den && *den != 0) {
295 auto const places = SchemaReader::unsignedOf(member(*value, "dp"));
296 std::uint32_t const decimals =
297 places ? static_cast<std::uint32_t>(*places) : column.decimalPlaces.value_or(0);
298 std::string const text = decimalText(
299 math::Rational{math::Numerator{*num}, math::Denominator{*den}, math::DecimalPlaces{0}}, decimals);
300 return column.unit.empty() ? text : text + " " + column.unit;
301 }
302 }
303 return displayText(*value);
304}
305
306/// @brief A row member as a table cell: an integer as `int64` (its digits as text past int64), an exact value as a
307/// `Rational`, a fraction as a `double` only in a `Number` column (its digits as text otherwise, for the
308/// engine to parse by the column's kind), text, a bool, and absent or null as empty.
309/// @param row The row.
310/// @param column The column.
311/// @return The cell.
312[[nodiscard]] inline table::Cell tableCell(Json const& row, CollectionColumn const& column) {
313 Json const* const value = member(row, column.field);
314 if (value == nullptr || value->is_null()) {
315 return std::monostate{};
316 }
317 if (auto const* const flag = value->get_if<bool>()) {
318 return *flag;
319 }
320 if (auto const* const text = value->get_if<std::string>()) {
321 return *text;
322 }
323 if (auto const* const fraction = value->get_if<double>()) {
324 if (column.kind == table::ColumnKind::Number) {
325 return *fraction;
326 }
327 return numberText(*value).value_or(std::string{});
328 }
329 if (value->get_if<std::int64_t>() != nullptr || value->get_if<std::uint64_t>() != nullptr) {
330 if (auto const whole = SchemaReader::signedOf(value)) {
331 return *whole;
332 }
333 return numberText(*value).value_or(std::string{});
334 }
335 if (value->is_object()) {
336 auto const num = SchemaReader::signedOf(member(*value, "num"));
337 auto const den = SchemaReader::signedOf(member(*value, "den"));
338 if (num && den && *den != 0) {
339 auto const places = SchemaReader::unsignedOf(member(*value, "dp")).value_or(0);
340 return math::Rational{math::Numerator{*num}, math::Denominator{*den},
341 math::DecimalPlaces{static_cast<std::uint32_t>(
342 std::min<std::uint64_t>(places, math::kMaxDecimalPlaces))}};
343 }
344 }
345 return toJson(*value);
346}

362/// @brief The reactive state of one collection: the list, the table source, the actions, the confirm step and the
363/// row editor. A controller in the sense of `docs/spec/reactive/control.md`; it draws nothing.
364///
365/// Non-copyable and non-movable: replies and the editor point into it. Destroy a view of it before the session.
366class CollectionSession {
367public:
368 /// @param runtime The runtime. Borrowed: it must outlive the session.
369 /// @param model The view model.
370 /// @param schemas Finds the row action's schema; an unknown or malformed one means the collection has no editor.
371 /// @param submit Executes the query and every action. Required.
372 /// @param choices Fetches the editor's Choice options; may be empty.
373 /// @param options The editor's session options.
374 /// @throws std::invalid_argument when @p submit is empty.
375 CollectionSession(reactive::Runtime& runtime MORPH_LIFETIMEBOUND, CollectionModel model,
376 SchemaLookup const& schemas, Submitter submit, ChoiceFetcher choices,
377 FormSessionOptions options = {})
378 : _rt{&runtime},
379 _model{std::move(model)},
380 _submit{std::move(submit)},
381 _source{std::make_shared<table::VectorSource>(detail::engine::tableColumns(_model))},
382 _openRow{runtime, std::nullopt},
383 _pending{runtime, std::nullopt},
384 _list{runtime, [this](std::string const& body) { return _submit(_model.query(), body); },
385 [] { return std::optional<std::string>{"{}"}; }},
386 _rows{runtime, [this] { return parseRows(); }},
387 _actions{
388 runtime, [this](Call call) { return _submit(call.action, std::move(call.body)); },
389 reactive::MutationOptions<Call>{
390 .concurrency = reactive::Concurrency::Exclusive, .serialKey = {}, .invalidates = {_list.link()}}} {
391 if (!_submit) {
392 throw std::invalid_argument{"CollectionSession: a Submitter is required"};
393 }
394 _syncSource = std::make_unique<reactive::Effect>(runtime, [this] { syncSource(); });
395 if (_model.rowAction().has_value() && schemas) {
396 if (auto const schema = schemas(_model.rowAction()->action)) {
397 if (auto editorModel = FormModel::fromSchema(_model.rowAction()->action, *schema)) {
398 _editor = std::make_unique<FormSession>(runtime, *std::move(editorModel), _submit,
399 std::move(choices), std::move(options));
400 _editorWatch = std::make_unique<reactive::Effect>(runtime, [this] { watchEditor(); });
401 }
402 }
403 }
404 }

409 CollectionSession(CollectionSession&&) = delete;

436 /// @brief Fires `model().actions()[action]`, on @p rowKey for a row action. A confirm-guarded action waits in
437 /// `pendingConfirm()` instead. While another action is in flight the call does nothing (the actions are
438 /// one `Exclusive` mutation, and a view disables its buttons while `actionPending()`).
439 /// @param action Index into `model().actions()`.
440 /// @param rowKey The row's exact key text, for a row action.
441 /// @throws std::out_of_range for an index past the actions.
442 void fire(std::size_t action, std::string_view rowKey = {}) {
443 CollectionAction const& entry = detail::engine::elementAt(_model.actions(), action);
444 if (entry.confirm) {
445 _pending.set(PendingConfirm{.action = action, .rowKey = std::string{rowKey}});
446 return;
447 }
448 run(entry, rowKey);
449 }

457 std::optional<PendingConfirm> const pending = _pending.peek();
458 _pending.set(std::nullopt);
459 if (pending.has_value()) {
460 run(detail::engine::elementAt(_model.actions(), pending->action), pending->rowKey);
461 }
462 }

467 /// @brief Opens the editor on a row, prefilled from the row action's `bind` entries with the row's exact values.
468 /// A prefill replaces every field, so nothing typed for another row carries over. Programmatic, so
469 /// nothing is submitted until the user edits.
470 /// @param rowKey The row's exact key text.
471 void open(std::string_view rowKey) {
472 if (!_editor) {
473 return;
474 }
475 // NOLINTNEXTLINE(bugprone-unchecked-optional-access): an editor exists only for a row action.
476 std::string const body = bindBody(*_model.rowAction(), rowKey);
477 _rt->batch([&] {
478 _openRow.set(std::string{rowKey});
479 _editor->prefill(body);
480 });
481 }

508 [[nodiscard]] std::vector<CollectionRow> parseRows() const {
509 using detail::engine::Json;
510 std::vector<CollectionRow> rows;
511 std::optional<std::string> const& reply = _list.value();
512 auto const dom = reply.has_value() ? detail::engine::parseJson(*reply) : std::nullopt;
513 if (!dom) {
514 return rows;
515 }
516 Json const* list = dom->is_array() ? &*dom : nullptr;
517 if (list == nullptr && dom->is_object()) {
518 for (auto const& [ignored, candidate] : dom->get_object()) {
519 if (candidate.is_array()) {
520 list = &candidate;
521 break;
522 }
523 }
524 }
525 if (list == nullptr) {
526 return rows;
527 }
528 for (Json const& row : list->get_array()) {
529 CollectionRow entry{};
530 Json const* const key = detail::engine::member(row, _model.rowKey());
531 entry.key = key == nullptr ? std::to_string(rows.size()) : detail::engine::displayText(*key);
532 auto const whole = key == nullptr ? std::optional<std::int64_t>{static_cast<std::int64_t>(rows.size())}
533 : detail::engine::SchemaReader::signedOf(key);
534 bool const integral = key == nullptr || key->get_if<double>() == nullptr;
535 entry.id = whole && integral ? table::RowId{*whole} : table::RowId{entry.key};
536 entry.json = detail::engine::toJson(row);
537 for (CollectionColumn const& column : _model.columns()) {
538 if (!column.hidden) {
539 entry.cells.push_back(detail::engine::cellText(row, column));
540 }
541 }
542 rows.push_back(std::move(entry));
543 }
544 return rows;
545 }

547 void syncSource() {
548 std::vector<CollectionRow> const& rows = _rows.get();
549 std::vector<table::RowId> ids;
550 std::vector<std::vector<table::Cell>> cells;
551 ids.reserve(rows.size());
552 cells.reserve(rows.size());
553 for (CollectionRow const& row : rows) {
554 ids.push_back(row.id);
555 auto const dom = detail::engine::parseJson(row.json);
556 std::vector<table::Cell> line;
557 line.reserve(_model.columns().size());
558 for (CollectionColumn const& column : _model.columns()) {
559 line.push_back(dom ? detail::engine::tableCell(*dom, column) : table::Cell{});
560 }
561 cells.push_back(std::move(line));
562 }
563 _source->setRows(std::move(ids), std::move(cells));
564 }

566 [[nodiscard]] std::string bindBody(CollectionAction const& action, std::string_view rowKey) const {
567 std::vector<CollectionRow> const& rows = _rows.peek();
568 auto const found = std::ranges::find(rows, rowKey, &CollectionRow::key);
569 auto const row = found == rows.end() ? std::nullopt : detail::engine::parseJson(found->json);
570 std::string body = "{";
571 bool first = true;
572 for (auto const& [field, rowField] : action.bind) {
573 detail::engine::Json const* const value = row ? detail::engine::member(*row, rowField) : nullptr;
574 body += (first ? "" : ",") + detail::engine::quote(field) + ":" +
575 (value == nullptr ? std::string{"null"} : detail::engine::toJson(*value));
576 first = false;
577 }
578 return body + "}";
579 }
