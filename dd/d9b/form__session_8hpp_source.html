187/// @brief The reactive state of one field: its draft and everything derived from it.
188class FieldState {
189public:
190 /// @param core The session core. Borrowed: it must outlive the field.
191 /// @param spec The field's spec. Borrowed from the core's model (or a parent's children).
192 /// @param path The instance path (`name`, `parent.member`, `rows#key.member`).
193 /// @param siblings The fields `x-optionsDependsOn` names are looked up among. Borrowed.
194 /// @param parent The containing Object or ObjectArray field, or null at the top level.
195 // NOLINTNEXTLINE(misc-no-recursion): an Object's or ObjectArray row's members are fields, to the model's depth.
196 FieldState(detail::engine::SessionCore& core MORPH_LIFETIMEBOUND, FieldSpec const& spec MORPH_LIFETIMEBOUND,
197 std::string path, std::vector<std::unique_ptr<FieldState>> const* siblings, FieldState const* parent)
198 : _core{&core},
199 _spec{&spec},
200 _path{std::move(path)},
201 _siblings{siblings},
202 _parent{parent},
203 _text{core.runtime(), std::string{}},
204 _unit{core.runtime(), spec.entryUnit},
205 _engaged{core.runtime(), false},
206 _rows{core.runtime(), std::vector<std::shared_ptr<RowState>>{}},
207 _encoded{core.runtime(), [this] { return computeEncoding(); }},
208 _error{core.runtime(),
209 [this] -> std::optional<FieldError> {
210 Encoding const& encoding = _encoded.get();
211 if (encoding.has_value() && !encoding->has_value()) {
212 return encoding->error();
213 }
214 return std::nullopt;
215 }},
216 _visible{core.runtime(),
217 [this] {
218 if (_spec->hidden) {
219 return false;
220 }
221 return _parent != nullptr ||
222 visibleVerdict(_core->model().rules(), _spec->name, _core->lookup()) != Tri::False;
223 }},
224 _readonly{core.runtime(),
225 [this] {
226 if (_spec->readOnly || (_parent != nullptr && _parent->readonly().get())) {
227 return true;
228 }
229 return _parent == nullptr &&
230 readonlyVerdict(_core->model().rules(), _spec->name, _core->lookup()) == Tri::True;
231 }},
232 _required{core.runtime(), [this] {
233 return _spec->required ||
234 (_parent == nullptr &&
235 requiredVerdict(_core->model().rules(), _spec->name, _core->lookup()) == Tri::True);
236 }} {
237 if (_spec->kind == FieldKind::Object) {
238 for (FieldSpec const& child : _spec->children) {
239 _members.push_back(
240 std::make_unique<FieldState>(core, child, _path + "." + child.name, &_members, this));
241 }
242 }
243 _label = resolveSlot(i18n::FieldSlot::Label, _spec->title);
244 _help = resolveSlot(i18n::FieldSlot::Help, _spec->help);
245 _placeholder = _spec->placeholder.empty() ? defaultPlaceholder(*_spec)
246 : resolveSlot(i18n::FieldSlot::Placeholder, _spec->placeholder);
247 }

251 FieldState& operator=(FieldState const&) = delete;
252 FieldState(FieldState&&) = delete;
253 FieldState& operator=(FieldState&&) = delete;

314 /// @brief Whether nothing is entered: a blank draft, an Object whose members are all blank, a
315 /// collection with no rows that was never engaged. Tracked.

438 /// @brief Replaces the draft with a wire value as the user's own edit, in one batch: a `load` the session does
439 /// not mark programmatic, so the field counts as changed and is validated as an edit is.
440 /// @param value The wire value; null is blank.
441 /// @return Nothing, or why the value does not fit the field: an Object takes a JSON object, a collection a JSON
442 /// array. Then nothing changes.
443 std::expected<void, std::string> edit(detail::engine::Json const& value) {
444 if (!value.is_null() && !_spec->truncated) {
445 if (_spec->kind == FieldKind::Object && !value.is_object()) {
446 return std::unexpected(std::string{"an object field takes a JSON object"});
447 }
448 if (_spec->kind == FieldKind::ObjectArray && !value.is_array()) {
449 return std::unexpected(std::string{"a list field takes a JSON array"});
450 }
451 }
452 _core->runtime().batch([&] { load(&value); });
453 return {};
454 }

456 /// @brief Replaces the draft with the decoded wire value; null or absent is blank. Does not
457 /// mark the change programmatic — the caller does, in the same batch.
458 /// @param value The wire value, or null.
459 // An Object loads its members, a collection its rows; one case per kind.
460 // NOLINTNEXTLINE(misc-no-recursion,readability-function-cognitive-complexity)
461 void load(detail::engine::Json const* value) {
462 using detail::engine::Json;
463 bool const present = value != nullptr && !value->is_null();
464 if (_spec->truncated) {
465 _text.set(present ? detail::engine::toJson(*value) : std::string{});
466 _engaged.set(present && value->is_array());
467 return;
468 }
469 if (_spec->kind == FieldKind::Object) {
470 for (auto const& entry : _members) {
471 entry->load(present ? detail::engine::member(*value, entry->spec().name) : nullptr);
472 }
473 return;
474 }
475 if (_spec->kind == FieldKind::ObjectArray) {
476 std::vector<std::shared_ptr<RowState>> loaded;
477 if (present && value->is_array()) {
478 for (Json const& element : value->get_array()) {
479 auto row = std::make_shared<RowState>(*_core, *_spec, _path, _nextRowKey++, this);
480 for (auto const& cell : row->fields()) {
481 cell->load(detail::engine::member(element, cell->spec().name));
482 }
483 row->start();
484 loaded.push_back(std::move(row));
485 }
486 }
487 _rows.set(std::move(loaded));
488 _engaged.set(present && value->is_array());
489 return;
490 }
491 std::string draft = present ? decodeScalar(*_spec, *value, _core->values()) : std::string{};
492 std::size_t const unit = _spec->entryUnit;
493 if (unit != 0 && unit < _spec->units.size() && !draft.empty()) {
494 // The wire value is in the canonical unit; the draft is shown in the entry unit.
495 draft = convertDraft(draft, _spec->units.front(), _spec->units.at(unit), _core->values().locale)
496 .value_or(std::string{});
497 }
498 _text.set(std::move(draft));
499 _unit.set(unit);
500 _engaged.set(_spec->blankAs == BlankAs::Empty && present && value->is_string());
501 }

515 /// @brief Starts what needs every sibling to exist: an `x-blankAs` field's engagement, a
516 /// Choice's options query, and every member's own. Called once, after the owner built
517 /// all its fields.
518 // NOLINTNEXTLINE(misc-no-recursion): an Object starts its members.
519 void start() {
520 for (auto const& entry : _members) {
521 entry->start();
522 }
523 if (_spec->blankAs == BlankAs::Empty) {
524 _engage = std::make_unique<reactive::Effect>(_core->runtime(), [this] {
525 if (!detail::engine::trim(_text.get()).empty() && !_engaged.peek()) {
526 _engaged.set(true);
527 }
528 });
529 }
530 if (_spec->kind == FieldKind::Choice && _spec->choice.has_value() && _core->choices()) {
531 reactive::Runtime& runtime = _core->runtime();
532 _optionsQuery = std::make_unique<reactive::Query<std::string, std::string>>(
533 runtime,
534 [this](std::string const& request) { return _core->choices()(_spec->choice->optionsAction, request); },
535 [this] { return optionsRequest(); });
536 _fetched = std::make_unique<reactive::Computed<std::vector<ChoiceOption>>>(runtime, [this] {
537 std::optional<std::string> const& reply = _optionsQuery->value();
538 return reply.has_value() ? parseOptions(*reply, *_spec->choice) : std::vector<ChoiceOption>{};
539 });
540 _clearStale = std::make_unique<reactive::Effect>(runtime, [this] { clearStaleSelection(); });
541 }

545 // Each draft is written with its length first, so no draft can be read as the end of another.
546 // NOLINTNEXTLINE(misc-no-recursion): an Object's members and a row's cells are fields.
547 void appendSnapshot(std::string& out) const {
548 std::string const& text = _text.peek();
549 out += std::to_string(text.size());
550 out += ':';
551 out += text;
552 out += std::to_string(_unit.peek());
553 out += _engaged.peek() ? '+' : '-';
554 for (auto const& entry : _members) {
555 entry->appendSnapshot(out);
556 }
557 for (auto const& row : _rows.peek()) {
558 out += '#';
559 for (auto const& cell : row->fields()) {
560 cell->appendSnapshot(out);
561 }
562 }
563 out += ';';
564 }

566 [[nodiscard]] std::string resolveSlot(i18n::FieldSlot slot, std::string const& literal) const {
567 auto const& translations = _core->options().translations;
568 std::string_view const locale = _core->options().bcp47;
569 auto const explicitKey = i18n::explicitFieldKey(_spec->i18nKey, slot);
570 if (_parent == nullptr) {
571 return render::resolveText(translations, locale, explicitKey,
572 i18n::fieldKey(_core->model().actionType(), _spec->name, slot), literal);
573 }
574 if (translations && explicitKey) {
575 if (auto hit = translations(*explicitKey, locale)) {
576 return *std::move(hit);
577 }
578 }
579 return literal;
580 }

591 /// The options request for the current parents, or nullopt while one does not encode.
592 [[nodiscard]] std::optional<std::string> optionsRequest() const {
593 std::string body = "{";
594 bool first = true;
595 // NOLINTNEXTLINE(bugprone-unchecked-optional-access): only a Choice asks, and a Choice has its spec.
596 for (std::string const& parent : _spec->choice.value().dependsOn) {
597 FieldState const* const source = sibling(parent);
598 if (source == nullptr) {

850 /// @param digits The text between the brackets.
851 /// @return The position, or `std::nullopt` unless the text is all decimal digits.
852 [[nodiscard]] static std::optional<std::size_t> rowIndexOf(std::string_view digits) {
853 std::size_t parsed = 0;
854 char const* const end = std::to_address(digits.end());
855 auto const result = std::from_chars(std::to_address(digits.begin()), end, parsed);

862 /// @brief A field by path: `name`, `parent.member`, or `rows[i].member` (a row by position).
863 /// @param path The path.
864 /// @return The field.
865 /// @throws std::out_of_range when no field has that path.
866 [[nodiscard]] FieldState& field(std::string_view path) const {
867 FieldState* current = nullptr;
868 std::span<std::unique_ptr<FieldState> const> scope = _fields;
869 std::string_view rest = path;
870 auto const missing = [&] {
871 return std::out_of_range{"FormSession::field: no field '" + std::string{path} + "'"};
872 };
873 while (!rest.empty()) {
874 auto const dot = rest.find('.');
875 std::string_view segment = rest.substr(0, dot);
876 rest = dot == std::string_view::npos ? std::string_view{} : rest.substr(dot + 1);
877 std::optional<std::size_t> index;
878 if (auto const bracket = segment.find('['); bracket != std::string_view::npos) {
879 if (!segment.ends_with(']')) {
880 throw missing();
881 }
882 index = rowIndexOf(segment.substr(bracket + 1, segment.size() - bracket - 2));
883 if (!index.has_value()) {
884 throw missing();
885 }
886 segment = segment.substr(0, bracket);
887 }
888 auto const found =
889 std::ranges::find_if(scope, [&](auto const& entry) { return entry->spec().name == segment; });
890 if (found == scope.end()) {
891 throw missing();
892 }
893 current = found->get();
894 if (index.has_value()) {
895 auto const& rows = current->rows();
896 if (*index >= rows.size() || rest.empty()) {
897 throw missing();
898 }
899 scope = rows.at(*index)->fields();
900 } else {
901 scope = current->members();
902 if (!rest.empty() && scope.empty()) {
903 throw missing();
904 }
905 }
906 }
907 if (current == nullptr) {
908 throw missing();
909 }
910 return *current;
911 }

1100 void autoSubmit() {
1101 if (_core->model().submitMode() != SubmitMode::Automatic) {
1102 return;
1103 }
1104 // While a Choice's options are in flight its selection is unjudged: a parent just changed, and the selection
1105 // may be one the new options no longer back. Waiting lets the stale selection be cleared first; the settled
1106 // query wakes this Effect again.
1107 if (std::ranges::any_of(_fields, [](auto const& field) { return field->optionsPending(); })) {
1108 return;
1109 }
1110 // The submission is Exclusive: while one is in flight a new run would be refused and its body lost. Waiting
1111 // here sends the latest body once the one in flight settles, which wakes this Effect again.
1112 if (_mutation.pending()) {
1113 return;
1114 }
1115 std::optional<std::string> const& body = _body.get();
1116 if (!body.has_value() || body == _lastSubmitted) {
1117 return;
1118 }
1119 _lastSubmitted = body;
1120 std::string const toSend = *body;
1121 runtime().untracked([&] {
1122 if (_mutation.run(toSend)) {
1123 _attempts.set(_attempts.peek() + 1);
1124 }
1125 });
1126 }
