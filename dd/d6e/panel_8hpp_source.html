71/// @brief Connects the panel document to a client, through a `HostVocabulary`: `chat.entries`, `chat.running`,
72/// `chat.onScreen` and `chat.canSend` to read, and `chat.send` and `chat.cancel` to run.
73///
74/// The commands are safe for an untrusted document: they do only what the person could do by typing, under their own
75/// session, and every action they lead to is checked on the server. A shell calls `refresh` from the client's change
76/// callback, and when the active tab changes. Neither copyable nor movable; on the runtime's owner.
77class ChatPanelBinding {
78public:
79 /// @brief Registers the values and the commands.
80 /// @param host The vocabulary. Borrowed: it outlives this binding.
81 /// @param client The client. Borrowed: it outlives this binding.
82 /// @param context Where the person is now: the open screen's context, with no text. Read at each send, so a
83 /// message
84 /// goes to the screen they are looking at when they send it. Nothing for a screen with no document
85 /// behind it, such as a host screen, which disables the input.
86 ChatPanelBinding(HostVocabulary& host, ChatClient& client, std::function<std::optional<TurnRequest>()> context)
87 : _host{&host}, _client{&client}, _context{std::move(context)} {
88 host.add(
89 HostValue{.name = "chat.entries", .version = 1, .type = "list<record>", .initial = Value{ValueList{}}});
90 host.add(HostValue{.name = "chat.running", .version = 1, .type = "bool", .initial = Value{false}});
91 host.add(HostValue{.name = "chat.onScreen", .version = 1, .type = "bool", .initial = Value{false}});
92 host.add(HostValue{.name = "chat.canSend", .version = 1, .type = "bool", .initial = Value{false}});
93 host.add(HostCommand{.name = "chat.send",
94 .version = 1,
95 .args = {HostArg{.name = "text", .type = "string"}},
96 .safeForUntrusted = true,
97 .run = [this](ValueRecord const& args) { send(args); }});
98 host.add(HostCommand{.name = "chat.cancel",
99 .version = 1,
100 .args = {},
101 .safeForUntrusted = true,
102 .run = [this](ValueRecord const& /*args*/) { _client->cancel(); }});
103 refresh();
104 }
105
106 ChatPanelBinding(ChatPanelBinding const&) = delete;
107 ChatPanelBinding& operator=(ChatPanelBinding const&) = delete;
108 ChatPanelBinding(ChatPanelBinding&&) = delete;
109 ChatPanelBinding& operator=(ChatPanelBinding&&) = delete;
110 ~ChatPanelBinding() = default;
111
112 /// @brief Publishes what the client shows now, and whether the person is on a screen.
113 void refresh() {
114 ValueList entries;
115 std::int64_t key = 0;
116 for (ChatClient::Entry const& entry : _client->entries()) {
117 entries.emplace_back(ValueRecord{{"key", Value{key++}}, {"role", roleOf(entry)}, {"text", textOf(entry)}});
118 }
119 bool const onScreen = _context && _context().has_value();
120 bool const running = _client->running();
121 _host->set("chat.entries", Value{std::move(entries)});
122 _host->set("chat.running", Value{running});
123 _host->set("chat.onScreen", Value{onScreen});
124 _host->set("chat.canSend", Value{onScreen && !running});
125 }
126
127private:
128 [[nodiscard]] static Value roleOf(ChatClient::Entry const& entry) {
129 if (entry.kind == "tool") {
130 return Value{std::string{"muted"}};
131 }
132 return Value{std::string{entry.kind == "error" ? "error" : "normal"}};
133 }
134
135 [[nodiscard]] static Value textOf(ChatClient::Entry const& entry) {
136 if (entry.kind == "user") {
137 return Value{"You: " + entry.text};
138 }
139 return Value{entry.kind == "tool" ? "Ran " + entry.text : entry.text};
140 }
141
142 void send(ValueRecord const& args) {
143 auto const given = args.find("text");
144 if (given == args.end() || _client->running() || !_context) {
145 return;
146 }
147 std::string const* const text = given->second.asString();
148 if (text == nullptr ||
149 std::ranges::all_of(*text, [](unsigned char const letter) { return std::isspace(letter); })) {
150 return;
151 }
152 std::optional<TurnRequest> request = _context();
153 if (!request) {
154 return;
155 }
156 request->text = *text;
157 _client->send(*request);
158 }
159
160 HostVocabulary* _host;
161 ChatClient* _client;
162 std::function<std::optional<TurnRequest>()> _context;
163};
