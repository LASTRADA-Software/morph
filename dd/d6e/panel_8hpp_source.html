103/// @brief Connects the panel document to a client, through a `HostVocabulary`: `chat.entries`, `chat.running`,
104/// `chat.onScreen` and `chat.canSend` to read, `chat.send` and `chat.cancel` to run, and, for the panel
105/// document alone, the dictation names.
106///
107/// The commands are safe for an untrusted document: they do only what the person could do by typing, under their own
108/// session, and every action they lead to is checked on the server. A shell calls `refresh` from the client's change
109/// callback, and when the active tab changes. Neither copyable nor movable; on the runtime's owner.
110class ChatPanelBinding {
111public:
112 /// @brief Registers the values and the commands.
113 /// @param host The vocabulary. Borrowed: it outlives this binding.
114 /// @param client The client. Borrowed: it outlives this binding.
115 /// @param context Where the person is now: the open screen's context, with no text. Read at each send, so a
116 /// message goes to the screen they are looking at when they send it. Nothing for a screen with no
117 /// document behind it, such as a host screen, which disables the input.
118 /// @param origin What the client hands back with each draft the turn proposes (`ChatClient::send`): read at
119 /// each send, with the context, so it names the tab the message goes from. May be empty, which is
120 /// 0.
121 /// @param draft The text of the draft the open screen holds, read at each send and carried as the turn's
122 /// `TurnRequest::draft`; nothing for a screen with no draft. May be empty, which is nothing.
123 /// @param recorder What records a dictated clip; null offers no dictation. Borrowed: it outlives this binding,
124 /// which sets its callbacks and clears them when it goes.
125 /// @param visible Whether the panel is in view, read at each `refresh`. A microphone never runs without its
126 /// indicator in view, so empty, like `false`, offers no dictation.
127 ChatPanelBinding(HostVocabulary& host, ChatClient& client, std::function<std::optional<TurnRequest>()> context,
128 std::function<std::uint64_t()> origin = {},
129 std::function<std::optional<std::string>()> draft = {}, IDictationRecorder* recorder = nullptr,
130 std::function<bool()> visible = {})
131 : _host{&host},
132 _client{&client},
133 _context{std::move(context)},
134 _origin{std::move(origin)},
135 _draft{std::move(draft)},
136 _recorder{recorder},
137 _visible{std::move(visible)} {
138 host.add(
139 HostValue{.name = "chat.entries", .version = 1, .type = "list<record>", .initial = Value{ValueList{}}});
140 host.add(HostValue{.name = "chat.running", .version = 1, .type = "bool", .initial = Value{false}});
141 host.add(HostValue{.name = "chat.onScreen", .version = 1, .type = "bool", .initial = Value{false}});
142 host.add(HostValue{.name = "chat.canSend", .version = 1, .type = "bool", .initial = Value{false}});
143 host.add(HostCommand{.name = "chat.send",
144 .version = 1,
145 .args = {HostArg{.name = "text", .type = "string"}},
146 .safeForUntrusted = true,
147 .run = [this](ValueRecord const& args) { send(args); }});
148 host.add(HostCommand{.name = "chat.cancel",
149 .version = 1,
150 .args = {},
151 .safeForUntrusted = true,
152 .run = [this](ValueRecord const& /*args*/) { _client->cancel(); }});
153 addDictation(host);
154 refresh();
155 }

162 /// @brief Clears the recorder's callbacks, then stops a recording, whose clip then reaches no one.
163 ~ChatPanelBinding() {
164 if (_recorder == nullptr) {
165 return;
166 }
167 _recorder->onStarted = nullptr;
168 _recorder->onProgress = nullptr;
169 _recorder->onClip = nullptr;
170 if (_state == DictationState::Recording) {
171 _recorder->stop();
172 }
173 }

175 /// @brief Publishes what the client shows now, whether the person is on a screen, and the dictation's state. Stops
176 /// a recording and drops its clip when the panel is no longer visible, the dictation offer is gone (as
177 /// after the client's `end` at sign-out) or no document is behind the panel. The microphone never runs
178 /// without its indicator; the button also stays shown while a clip is being transcribed.
179 void refresh() {
180 ValueList entries;
181 std::int64_t key = 0;
182 for (ChatClient::Entry const& entry : _client->entries()) {
183 entries.emplace_back(ValueRecord{{"key", Value{key++}}, {"role", roleOf(entry)}, {"text", textOf(entry)}});
184 }
185 bool const onScreen = _context && _context().has_value();
186 bool const running = _client->running();
187 if (_state == DictationState::Recording && !canDictate()) {
188 dropRecording();
189 }
190 _host->set("chat.entries", Value{std::move(entries)});
191 _host->set("chat.running", Value{running});
192 _host->set("chat.onScreen", Value{onScreen});
193 _host->set("chat.canSend", Value{onScreen && !running});
194 publishDictation();
195 }

197private:
198 enum class DictationState : std::uint8_t { Idle, Recording, Waiting };

200 // Registered for the panel document alone: no screen, custom screen or draft may start the microphone or read a
201 // transcript. Safe for an untrusted connection, since the panel document is the client's own.
202 void addDictation(HostVocabulary& host) {
203 std::set<std::string, std::less<>> const panel{std::string{kChatPanelId}};
204 host.add(HostValue{
205 .name = "chat.canDictate", .version = 1, .type = "bool", .initial = Value{false}, .documents = panel});
206 host.add(HostValue{.name = "chat.dictation",
207 .version = 1,
208 .type = "string",
209 .initial = Value{std::string{"idle"}},
210 .documents = panel});
211 host.add(HostValue{.name = "chat.dictationSeconds",
212 .version = 1,
213 .type = "int",
214 .initial = Value{std::int64_t{0}},
215 .documents = panel});
216 host.add(
217 HostValue{.name = "chat.dictated",
218 .version = 1,
219 .type = "record",
220 .initial = Value{ValueRecord{{"seq", Value{std::int64_t{0}}}, {"text", Value{std::string{}}}}},
221 .documents = panel});
222 host.add(HostCommand{.name = "chat.dictateStart",
223 .version = 1,
224 .args = {},
225 .safeForUntrusted = true,
226 .run = [this](ValueRecord const& /*args*/) { startDictation(); },
227 .documents = panel});
228 host.add(HostCommand{.name = "chat.dictateStop",
229 .version = 1,
230 .args = {},
231 .safeForUntrusted = true,
232 .run = [this](ValueRecord const& /*args*/) { stopDictation(); },
233 .documents = panel});
234 if (_recorder == nullptr) {
235 return;
236 }
237 _recorder->onStarted = [this](std::expected<void, std::string> started) {
238 if (!started && _state == DictationState::Recording) {
239 _state = DictationState::Idle;
240 _client->note(std::move(started.error()));
241 publishDictation();
242 }
243 };
244 _recorder->onProgress = [this](std::chrono::seconds seconds) {
245 if (_state == DictationState::Recording) {
246 _seconds = seconds.count();
247 publishDictation();
248 }
249 };
250 _recorder->onClip = [this](std::expected<std::string, std::string> clip) { clipped(std::move(clip)); };
251 }

253 [[nodiscard]] bool shown() const { return _visible && _visible(); }
254
255 [[nodiscard]] bool canDictate() const {
256 return _recorder != nullptr && _client->dictation().has_value() && !_recorder->unavailable().has_value() &&
257 _context && _context().has_value() && shown();
258 }
259
260 void publishDictation() {
261 static constexpr std::array<std::string_view, 3> kNames{"idle", "recording", "waiting"};
262 _host->set("chat.canDictate", Value{canDictate()});
263 _host->set("chat.dictation", Value{std::string{kNames.at(static_cast<std::size_t>(_state))}});
264 _host->set("chat.dictationSeconds", Value{_seconds});
265 // A transcript is not kept once the offer is gone: it is cleared, so nothing reads it after a sign-out.
266 if (_hasDictated && !_client->dictation().has_value()) {
267 _hasDictated = false;
268 _host->set("chat.dictated",
269 Value{ValueRecord{{"seq", Value{_dictatedSeq}}, {"text", Value{std::string{}}}}});
270 }
271 }
272
273 void startDictation() {
274 if (_state != DictationState::Idle || !canDictate()) {
275 return;
276 }
277 std::optional<DictationLimits> const offered = _client->dictation();
278 if (!offered.has_value()) {
279 return;
280 }
281 DictationLimits const limits = *offered;
282 _state = DictationState::Recording;
283 _seconds = 0;
284 publishDictation();
285 _recorder->start(limits);
286 }
287
288 void stopDictation() {
289 if (_state == DictationState::Recording) {
290 _recorder->stop();
291 }
292 }
293
294 // The clip a stop now delivers is dropped: `stop` is always answered by exactly one `onClip`.
295 void dropRecording() {
296 _state = DictationState::Idle;
297 _seconds = 0;
298 ++_droppedClips;
299 _recorder->stop();
300 }
301
302 void clipped(std::expected<std::string, std::string> clip) {
303 if (_droppedClips > 0) {
304 --_droppedClips;
305 return;
306 }
307 if (_state != DictationState::Recording) {
308 return;
309 }
310 if (!clip) {
311 _state = DictationState::Idle;
312 _client->note(std::move(clip.error()));
313 publishDictation();
314 return;
315 }
316 _state = DictationState::Waiting;
317 publishDictation();
318 _client->transcribe(std::move(*clip), std::nullopt,
319 [this, alive = std::weak_ptr<bool>{_alive}](std::expected<std::string, std::string> text) {
320 if (!alive.expired()) {
321 transcribed(std::move(text));
322 }
323 });
324 }
325
326 // A transcript goes into the input through `chat.dictated`, never into a turn: the panel's watch appends it.
327 void transcribed(std::expected<std::string, std::string> text) {
328 _state = DictationState::Idle;
329 if (!text) {
330 _client->note(std::move(text.error()));
331 } else if (std::ranges::all_of(*text, [](unsigned char const letter) { return std::isspace(letter); })) {
332 _client->note("Nothing was heard in the recording");
333 } else {
334 ++_dictatedSeq;
335 _hasDictated = true;
336 _host->set("chat.dictated",
337 Value{ValueRecord{{"seq", Value{_dictatedSeq}}, {"text", Value{std::move(*text)}}}});
338 }
339 publishDictation();
340 }

342 [[nodiscard]] static Value roleOf(ChatClient::Entry const& entry) {
343 if (entry.kind == "tool") {
344 return Value{std::string{"muted"}};
345 }
346 return Value{std::string{entry.kind == "error" || entry.kind == "note" ? "error" : "normal"}};
347 }
348
349 [[nodiscard]] static Value textOf(ChatClient::Entry const& entry) {
350 if (entry.kind == "user") {
351 return Value{"You: " + entry.text};
352 }
353 return Value{entry.kind == "tool" ? "Ran " + entry.text : entry.text};
354 }
355
356 void send(ValueRecord const& args) {
357 auto const given = args.find("text");
358 if (given == args.end() || _client->running() || !_context) {
359 return;
360 }
361 std::string const* const text = given->second.asString();
362 if (text == nullptr ||
363 std::ranges::all_of(*text, [](unsigned char const letter) { return std::isspace(letter); })) {
364 return;
365 }
366 std::optional<TurnRequest> request = _context();
367 if (!request) {
368 return;
369 }
370 request->text = *text;
371 if (_draft) {
372 request->draft = _draft();
373 }
374 _client->send(*request, _origin ? _origin() : std::uint64_t{0});
375 }

377 HostVocabulary* _host;
378 ChatClient* _client;
379 std::function<std::optional<TurnRequest>()> _context;
380 std::function<std::uint64_t()> _origin;
381 std::function<std::optional<std::string>()> _draft;
382 IDictationRecorder* _recorder;
383 std::function<bool()> _visible;
384 DictationState _state = DictationState::Idle;
385 std::int64_t _seconds = 0;
386 std::int64_t _dictatedSeq = 0;
387 bool _hasDictated = false;
388 std::size_t _droppedClips = 0;
389 std::shared_ptr<bool> _alive = std::make_shared<bool>(true);
