381/// @brief A view node of the document (spec 5 §9), or a structured part of one (a table column, a tab, a switch
382/// case, a menu item, an `a11y`, `drag`, `drop` or `action` record, a form field overlay).
383///
384/// Properties are kept by their document keys, so the interpreter reads a kind's properties by name; which keys a
385/// kind has, and what shape each takes, is the kind table of `docs/spec/ui/document.md`.
386struct DocNode {
387 /// @brief The node kind (`text`, `table`, …), or for a part its part kind (`column`, `tab`, `case`, …).
388 std::string kind;
389 /// @brief The element id; empty when none is given.
390 std::string id;
391 /// @brief The node's JSON Pointer.
392 std::string path;
393 /// @brief Expression properties, by key.
394 std::map<std::string, ExprPtr, std::less<>> props;
395 /// @brief Command lists, by key (`onClick`, …), including a custom node's `on` events as `on.<event>`.
396 std::map<std::string, Commands, std::less<>> events;
397 /// @brief Child node lists, by key (`children`, `header`, `loading`, `error`, `default`, `cell`, `fallback`).
398 std::map<std::string, std::vector<DocNodePtr>, std::less<>> children;
399 /// @brief Structured parts, by key (`columns`, `tabs`, `cases`, `items`, `a11y`, `action`, `fields`, …).
400 std::map<std::string, std::vector<DocNodePtr>, std::less<>> parts;
401 /// @brief Plain string properties, by key (`testId`, `surface`, `name` of a custom node, `model` and `action`
402 /// of a form, `point` of `customize`, `role` of `a11y`, a column's `id`, …).
403 std::map<std::string, std::string, std::less<>> names;
404 /// @brief Key bindings: chord to commands.
405 std::map<std::string, Commands, std::less<>> keys;
406 /// @brief Literal flags, by key (`showLabel` of an input).
407 std::map<std::string, bool, std::less<>> flags;
408 /// @brief The declarations of the scope the node creates (dialog, drawer, switch case, `forEach` or `table`
409 /// row); empty for a node that creates none.
410 std::optional<ScopeDecl> scope;
411
412 /// @brief An expression property.
413 /// @param key The key.
414 /// @return The expression, or `nullptr` when absent.
415 [[nodiscard]] ExprPtr prop(std::string_view key) const {
416 auto const found = props.find(key);
417 return found == props.end() ? nullptr : found->second;
418 }
419
420 /// @brief A command list.
421 /// @param key The key.
422 /// @return The commands, or `nullptr` when absent.
423 [[nodiscard]] Commands const* event(std::string_view key) const {

426 }

429 /// @param key The key.

433 return found == children.end() ? std::span<DocNodePtr const>{} : std::span<DocNodePtr const>{found->second};
434 }
435
436 /// @brief A list of parts.
437 /// @param key The key.
438 /// @return The parts; empty when absent.
439 [[nodiscard]] std::span<DocNodePtr const> partList(std::string_view key) const {
440 auto const found = parts.find(key);
441 return found == parts.end() ? std::span<DocNodePtr const>{} : std::span<DocNodePtr const>{found->second};

444 /// @brief A plain string property.
445 /// @param key The key.
446 /// @return The string; empty when absent.
447 [[nodiscard]] std::string_view name(std::string_view key) const {

449 return found == names.end() ? std::string_view{} : std::string_view{found->second};

451
452 /// @brief A literal flag.

454 /// @return Its value; `false` when absent.

464 [[nodiscard]] bool flag(std::string_view key, bool absent) const {
465 auto const found = flags.find(key);
466 return found == flags.end() ? absent : found->second;
467 }
