60/// @brief The facts load-time validation needs from the application's catalog. Local mode answers them from the
61/// model registry in-process, the server at registration, a remote client from the UI service's catalog.
62///
63/// A lookup left empty skips the rules that need it: `model` the unknown-model and keyed-instance rules; `action`
64/// the unknown-action, wrong-model, `refreshOn`, optimistic and offline rules.

66 /// @brief Looks up a model type by its registered type id.
67 std::function<std::optional<ModelInfo>(std::string_view typeId)> model;

84/// @brief A `Catalog::view` or `Catalog::wizard` lookup over documents by action id.
85/// @param documents The documents by action id; the lookup keeps its own copy.
86/// @return The lookup.
87[[nodiscard]] inline std::function<std::optional<std::string>(std::string_view)> documentLookup(
88 std::map<std::string, std::string, std::less<>> documents) {
89 return [known = std::make_shared<std::map<std::string, std::string, std::less<>> const>(std::move(documents))](
90 std::string_view actionId) -> std::optional<std::string> {
91 auto const found = known->find(actionId);
92 if (found == known->end()) {
93 return std::nullopt;
94 }
95 return found->second;
96 };

343/// @brief A view node of the document (spec 5 §9), or a structured part of one (a table column, a tab, a switch
344/// case, a menu item, an `a11y`, `drag`, `drop` or `action` record, a form field overlay).
345///
346/// Properties are kept by their document keys, so the interpreter reads a kind's properties by name; which keys a
347/// kind has, and what shape each takes, is the kind table of `docs/spec/ui/document.md`.
348struct DocNode {
349 /// @brief The node kind (`text`, `table`, …), or for a part its part kind (`column`, `tab`, `case`, …).
350 std::string kind;
351 /// @brief The element id; empty when none is given.
352 std::string id;
353 /// @brief The node's JSON Pointer.
354 std::string path;
355 /// @brief Expression properties, by key.
356 std::map<std::string, ExprPtr, std::less<>> props;
357 /// @brief Command lists, by key (`onClick`, …), including a custom node's `on` events as `on.<event>`.
358 std::map<std::string, Commands, std::less<>> events;
359 /// @brief Child node lists, by key (`children`, `header`, `loading`, `error`, `default`, `cell`, `fallback`).
360 std::map<std::string, std::vector<DocNodePtr>, std::less<>> children;
361 /// @brief Structured parts, by key (`columns`, `tabs`, `cases`, `items`, `a11y`, `action`, `fields`, …).
362 std::map<std::string, std::vector<DocNodePtr>, std::less<>> parts;
363 /// @brief Plain string properties, by key (`testId`, `surface`, `name` of a custom node, `model` and `action`
364 /// of a form, `point` of `customize`, `role` of `a11y`, a column's `id`, …).
365 std::map<std::string, std::string, std::less<>> names;
366 /// @brief Key bindings: chord to commands.
367 std::map<std::string, Commands, std::less<>> keys;
368 /// @brief The declarations of the scope the node creates (dialog, drawer, switch case, `forEach` or `table`
369 /// row); empty for a node that creates none.
370 std::optional<ScopeDecl> scope;
371
372 /// @brief An expression property.
373 /// @param key The key.
374 /// @return The expression, or `nullptr` when absent.

377 return found == props.end() ? nullptr : found->second;
378 }

380 /// @brief A command list.
381 /// @param key The key.

397 /// @param key The key.

402 }
403
404 /// @brief A plain string property.
405 /// @param key The key.
406 /// @return The string; empty when absent.

408 auto const found = names.find(key);
409 return found == names.end() ? std::string_view{} : std::string_view{found->second};
410 }
411};

869/// @brief Parses a key chord, and refuses one the workspace shell reserves.
870/// @param chord The chord, such as `Ctrl+Shift+K`.
871/// @param path Its JSON Pointer.
872// NOLINTNEXTLINE(readability-function-cognitive-complexity): a chord's grammar, modifiers then one key name.
873inline void checkChord(std::string const& chord, std::string const& path) {
874 static constexpr std::array<std::string_view, 4> kModifiers = {"Ctrl", "Shift", "Alt", "Meta"};
875 static constexpr std::array<std::string_view, 9> kNamedKeys = {"Enter", "Esc", "Tab", "Up", "Down",
876 "Left", "Right", "Space", "Delete"};
877 std::vector<std::string_view> parts;
878 std::size_t start = 0;
879 for (std::size_t plus = chord.find('+'); plus != std::string::npos; plus = chord.find('+', start)) {
880 parts.push_back(std::string_view{chord}.substr(start, plus - start));
881 start = plus + 1;
882 }
883 std::string_view const key = std::string_view{chord}.substr(start);
884 std::set<std::string_view> modifiers;
885 for (std::string_view const modifier : parts) {
886 if (!listed(kModifiers, modifier) || !modifiers.insert(modifier).second) {
887 refuse(path, "`" + chord + "` is not a chord: modifiers are Ctrl, Shift, Alt and Meta, once each");
888 }
889 }
890 bool const letterOrDigit =
891 key.size() == 1 && ((key.front() >= 'A' && key.front() <= 'Z') || (key.front() >= 'a' && key.front() <= 'z') ||
892 (key.front() >= '0' && key.front() <= '9'));
893 bool function = key.size() >= 2 && key.size() <= 3 && key.front() == 'F';
894 int number = 0;
895 for (char const chr : key.substr(function ? 1 : key.size())) {
896 function = function && chr >= '0' && chr <= '9';
897 number = (number * 10) + (chr - '0');
898 }
899 function = function && number >= 1 && number <= 24;
900 if (!letterOrDigit && !function && !listed(kNamedKeys, key)) {
901 refuse(path, "`" + chord + "` does not end in a key name");
902 }
903 bool const ctrlOnly = modifiers.size() == 1 && modifiers.contains("Ctrl");
904 bool const ctrlShift = modifiers.size() == 2 && modifiers.contains("Ctrl") && modifiers.contains("Shift");
905 bool const reserved = (key == "Tab" && (ctrlOnly || ctrlShift)) ||
906 (ctrlOnly && (key == "W" || key == "w" || key == "K" || key == "k"));
907 if (reserved) {
908 refuse(path, "`" + chord + "` is reserved for the workspace shell");
909 }
910}

913/// @brief Turns the JSON of a document into its model, refusing what does not parse.
914class DocumentParser {
915public:
916 /// @brief Parses the whole document.
917 /// @param json The document.
918 /// @return The model, not yet checked against its names.
919 [[nodiscard]] Document parse(Json const& json) {
920 auto const* const object = std::get_if<Json::object_t>(&json.data);
921 if (object == nullptr) {
922 refuse("", "a document is a JSON object");
923 }
924 refuseMustUnderstand(*object, "");
925 Document doc;
926 doc.vocab = text(*object, "vocab", "", true);
927 if (doc.vocab != kVocabulary) {
928 refuse("/vocab", "the vocabulary `" + doc.vocab + "` is not supported; this client reads ui/1");
929 }
930 doc.id = text(*object, "id", "", true);
931 if (doc.id.empty()) {
932 refuse("/id", "a document has a non-empty id");
933 }
934 doc.title = optionalExpr(*object, "title", "");
935 doc.identity = optionalExpr(*object, "identity", "");
936 doc.dirty = optionalExpr(*object, "dirty", "");
937 doc.showErrors = showErrors(*object, "");
938 if (Json const* const params = member(*object, "params")) {
939 for (auto const& [name, type] : objectOf(*params, "/params", "`params` maps names to types")) {
940 std::string const path = pointerAppend("/params", name);
941 checkName(name, path);
942 doc.params.emplace_back(name, typeOf(type, path));
943 }
944 }
945 doc.scope = scope(*object, "");
946 for (std::string_view const key : {"onMount", "onUnmount", "onClose", "onSave", "onBackendChange"}) {
947 if (Json const* const commands = member(*object, key)) {
948 doc.lifecycle.emplace(std::string{key}, commandList(*commands, pointerAppend("", key)));
949 }
950 }
951 if (Json const* const keys = member(*object, "keys")) {
952 doc.keys = keyBindings(*keys, "/keys");
953 }
954 if (Json const* const prefetch = member(*object, "prefetch")) {
955 doc.prefetch = nameList(*prefetch, "/prefetch");
956 }
957 if (Json const* const components = member(*object, "components")) {
958 for (auto const& [name, entry] : objectOf(*components, "/components", "`components` is an object")) {
959 doc.components.emplace(name, component(name, entry, pointerAppend("/components", name)));
960 }
961 }
962 if (Json const* const points = member(*object, "customize")) {
963 for (auto const& [name, entry] : objectOf(*points, "/customize", "`customize` is an object")) {
964 doc.customize.emplace(name, point(name, entry, pointerAppend("/customize", name), doc.id == "app"));
965 }
966 }
967 if (Json const* const view = member(*object, "view")) {
968 doc.view = node(*view, "/view");
969 } else if (doc.id != "app") {
970 refuse("", "a screen has a `view`");
971 }
972 return doc;
973 }

1094 [[nodiscard]] ScopeDecl scope(Json::object_t const& object, std::string const& path) {

1391 [[nodiscard]] Command command(Json const& json, std::string const& path) {
1392 auto const& fields = objectOf(json, path, R"(a command is an object such as {"set": ..., "value": ...})");
1393 CommandSpec const* chosen = nullptr;
1394 for (auto const& [key, value] : fields) {
1395 static_cast<void>(value);
1396 for (CommandSpec const& spec : kCommands) {
1397 if (spec.key == key) {
1398 if (chosen != nullptr) {
1399 refuse(path,
1400 "a command has one kind; found `" + std::string{chosen->key} + "` and `" + key + "`");
1401 }
1402 chosen = &spec;
1403 }
1404 }
1405 }
1406 if (chosen == nullptr) {
1407 refuse(path, "not a command of ui/1");
1408 }
1409 refuseMustUnderstand(fields, path);
1410 Command cmd;
1411 cmd.kind = chosen->kind;
1412 cmd.path = path;
1413 Json const& primary = *member(fields, chosen->key);
1414 std::string const where = pointerAppend(path, chosen->key);
1415 switch (chosen->kind) {
1416 case CommandKind::Set: {
1417 cmd.target = nameOf(primary, where, "`set` names state or an element as `name.path`", true);
1418 Json const* const value = member(fields, "value");
1419 if (value == nullptr) {
1420 refuse(path, "`set` needs a `value`");
1421 }
1422 cmd.value = expr(*value, pointerAppend(path, "value"));
1423 break;
1424 }
1425 case CommandKind::Run:
1426 cmd.target = nameOf(primary, where, "`run` names a mutation");
1427 cmd.value = optionalExpr(fields, "body", path);
1428 cmd.extra = optionalExpr(fields, "each", path);
1429 if (Json const* const binding = member(fields, "as")) {
1430 cmd.binding = nameOf(*binding, pointerAppend(path, "as"), "`as` is a name");
1431 }
1432 break;
1433 case CommandKind::Reset:
1434 case CommandKind::Refetch:
1435 case CommandKind::Open:
1436 case CommandKind::Focus:
1437 cmd.target = nameOf(primary, where, "the command names a declaration or an element");
1438 break;
1439 case CommandKind::Close:
1440 cmd.target = nameOf(primary, where, "`close` names a dialog, or `self`");
1441 break;
1442 case CommandKind::Navigate:
1443 navigate(cmd, fields, primary, path);
1444 break;
1445 case CommandKind::Return:
1446 cmd.value = expr(primary, where);
1447 break;
1448 case CommandKind::Host:
1449 cmd.target = nonEmpty(primary, where, "`host` names a registered command");
1450 cmd.value = optionalExpr(fields, "args", path);
1451 break;
1452 case CommandKind::SwitchBackend:
1453 case CommandKind::SetLocale:
1454 cmd.target = nonEmpty(primary, where, "the command takes a non-empty string");
1455 break;
1456 case CommandKind::ResetPrefs:
1457 cmd.target = nameOf(primary, where, "`resetPrefs` names a customization point, or `all`");
1458 break;
1459 case CommandKind::Notify: {
1460 cmd.value = expr(primary, where);
1461 static constexpr std::array<std::string_view, 4> kRoles = {"info", "success", "warning", "error"};
1462 cmd.option = choiceOf(fields, "role", path, kRoles);
1463 break;
1464 }
1465 case CommandKind::Seq:
1466 case CommandKind::All: {
1467 if (!std::holds_alternative<Json::array_t>(primary.data)) {
1468 refuse(where, "`" + std::string{chosen->key} + "` takes an array of commands");
1469 }
1470 cmd.commands = commandList(primary, where);
1471 break;
1472 }
1473 case CommandKind::If: {
1474 auto const* const array = std::get_if<Json::array_t>(&primary.data);
1475 if (array == nullptr || array->size() < 2 || array->size() > 3) {
1476 refuse(where, "`if` takes [condition, [then...], [else...]]");
1477 }
1478 cmd.value = expr(array->at(0), pointerAppend(where, std::size_t{0}));
1479 cmd.commands = commandList(array->at(1), pointerAppend(where, 1));
1480 if (array->size() == 3) {
1481 cmd.alternative = commandList(array->at(2), pointerAppend(where, 2));
1482 }
1483 break;
1484 }
1485 case CommandKind::CancelClose:
1486 case CommandKind::Retry:
1487 case CommandKind::SignOut:
1488 requireTrue(primary, where, chosen->key);
1489 break;
1490 case CommandKind::SignIn: {
1491 auto const* const object = std::get_if<Json::object_t>(&primary.data);
1492 Json const* const from = object == nullptr ? nullptr : member(*object, "from");
1493 if (from == nullptr) {
1494 refuse(where, R"(`signIn` is {"from": mutation})");
1495 }
1496 cmd.target = nameOf(*from, pointerAppend(where, "from"), "`signIn` names its mutation");
1497 break;
1498 }
1499 case CommandKind::Upload: {
1500 cmd.value = expr(primary, where);
1501 Json const* const destination = member(fields, "to");
1502 if (destination == nullptr) {
1503 refuse(path, "`upload` names its mutation with `to`");
1504 }
1505 cmd.target = nameOf(*destination, pointerAppend(path, "to"), "`to` names a mutation");
1506 if (Json const* const binding = member(fields, "as")) {
1507 cmd.binding = nameOf(*binding, pointerAppend(path, "as"), "`as` is a name");
1508 }
1509 if (Json const* const transfer = member(fields, "id")) {
1510 cmd.transfer = nameOf(*transfer, pointerAppend(path, "id"), "`id` is a name");
1511 }
1512 break;
1513 }
1514 case CommandKind::Download:
1515 default:
1516 download(cmd, fields, primary, path);
1517 break;
1518 }
1519 return cmd;
1520 }

1761 // NOLINTNEXTLINE(readability-function-cognitive-complexity): one case per shape a key takes.
1762 void entry(DocNode& made, KeySpec const& key, Json const& value, std::string const& path) {
1763 std::string const name{key.key};
1764 switch (key.shape) {
1765 case Shape::Expr:
1766 case Shape::Condition:
1767 made.props.emplace(name, expr(value, path));
1768 break;
1769 case Shape::Commands:
1770 made.events.emplace(name, commandList(value, path));
1771 break;
1772 case Shape::Nodes: {
1773 std::vector<DocNodePtr> nodes;
1774 if (auto const* const array = std::get_if<Json::array_t>(&value.data)) {
1775 for (std::size_t index = 0; index < array->size(); ++index) {
1776 nodes.push_back(node(array->at(index), pointerAppend(path, index)));
1777 }
1778 } else {
1779 nodes.push_back(node(value, path));
1780 }
1781 made.children.emplace(name, std::move(nodes));
1782 break;
1783 }
1784 case Shape::Text:
1785 case Shape::Name: {
1786 auto const* const string = std::get_if<std::string>(&value.data);
1787 if (string == nullptr || (key.shape == Shape::Name && !isName(*string))) {
1788 refuse(path, "`" + name + "` is " + (key.shape == Shape::Name ? "a name" : "a string"));
1789 }
1790 if (name == "id") {
1791 made.id = *string;
1792 } else {
1793 made.names.emplace(name, *string);
1794 }
1795 break;
1796 }
1797 case Shape::Part:
1798 made.parts[name].push_back(part(value, key.part, path));
1799 break;
1800 case Shape::Parts:
1801 case Shape::PartsOrExpr: {
1802 auto const* const array = std::get_if<Json::array_t>(&value.data);
1803 if (array == nullptr && key.shape == Shape::PartsOrExpr) {
1804 made.props.emplace(name, expr(value, path));
1805 break;
1806 }
1807 if (array == nullptr) {
1808 refuse(path, "`" + name + "` is an array");
1809 }
1810 auto& list = made.parts[name];
1811 for (std::size_t index = 0; index < array->size(); ++index) {
1812 list.push_back(part(array->at(index), key.part, pointerAppend(path, index)));
1813 }
1814 break;
1815 }
1816 case Shape::ExprRecord: {
1817 if (!key.part.empty()) {
1818 made.parts[name].push_back(part(value, key.part, path));
1819 break;
1820 }
1821 auto record = std::make_shared<DocNode>();
1822 record->kind = name;
1823 record->path = path;
1824 for (auto const& [field, fieldValue] : objectOf(value, path, "a record of expressions")) {
1825 record->props.emplace(field, expr(fieldValue, pointerAppend(path, field)));
1826 }
1827 made.parts[name].push_back(std::move(record));
1828 break;
1829 }
1830 case Shape::CommandRecord:
1831 for (auto const& [event, commands] : objectOf(value, path, "a record of command lists")) {
1832 made.events.emplace(std::format("{}.{}", name, event),
1833 commandList(commands, pointerAppend(path, event)));
1834 }
1835 break;
1836 case Shape::Keys:
1837 default:
1838 made.keys = keyBindings(value, path);
1839 break;
1840 }
1841 }
1842 // NOLINTEND(misc-no-recursion)

2830 std::size_t const count = _watches.size();
2831 std::vector<std::set<NameEntry const*>> observed(count);
2832 std::vector<std::set<NameEntry const*>> written(count);
2833 for (std::size_t index = 0; index < count; ++index) {

2837 reads(*watch->on, scope, observed.at(index), locals, seenLets);
2838 writes(watch->body, scope, written.at(index));
2839 }

2841 std::vector<int> state(count, 0);
2842 std::vector<std::size_t> trail;
2843 std::function<void(std::size_t)> visit = [&](std::size_t from) {
2844 state.at(from) = 1;
2845 trail.push_back(from);
2846 for (std::size_t to = 0; to < count; ++to) {
2847 bool edge = false;
2848 for (NameEntry const* entry : written.at(from)) {
2849 edge = edge || observed.at(to).contains(entry);
2850 }
2851 if (!edge) {
2852 continue;
2853 }
2854 if (state.at(to) == 1) {
2855 std::string cycle;
2856 auto const start = std::ranges::find(trail, to);
2857 for (auto step = start; step != trail.end(); ++step) {
2858 cycle += _watches.at(*step).first->path + " → ";
2859 }
2860 refuse(_watches.at(to).first->path,
2861 "the watch graph has a cycle: " + cycle + _watches.at(to).first->path);
2862 }
2863 if (state.at(to) == 0) {
2864 visit(to);
2865 }
2866 }
2867 trail.pop_back();
2868 state.at(from) = 2;
2869 };
2870 for (std::size_t index = 0; index < count; ++index) {
2871 if (state.at(index) == 0) {
2872 visit(index);
2873 }
2874 }

2890inline std::expected<Type, std::string> ScopeResolver::ref(std::string_view head,
2891 std::span<std::string const> path) const {
2892 for (auto const& [name, type] : std::views::reverse(_context.bindings)) {
2893 if (name == head) {
2894 return typeAlong(type, head, path);
2895 }
2896 }
2897 if (head == "event") {
2898 if (!_context.commands) {
2899 return std::unexpected(std::string{"`event` is read only inside a command list"});
2900 }
2901 return Type::any();
2902 }
2903 if (head == "app") {
2904 return Type::any();
2905 }
2906 if (head == "boundary") {
2907 if (!_context.boundaryError || path.size() != 1 || path.front() != "error") {
2908 return std::unexpected(std::string{"`boundary.error` is read only inside a boundary's `error`"});
2909 }
2910 return Type::of(TypeKind::String, true);
2911 }
2912 NameEntry* const entry = DocumentValidator::lookup(_context.scope, head);
2913 if (entry == nullptr) {
2914 return std::unexpected("`" + std::string{head} + "` does not resolve in this scope");
2915 }
2916 switch (entry->category) {
2917 case NameCategory::Param:
2918 return typeAlong(entry->type, head, path);
2919 case NameCategory::State:
2920 case NameCategory::Let: {
2921 auto type = _validator.derivedType(*entry, head);
2922 if (!type) {
2923 return type;
2924 }
2925 return typeAlong(*type, head, path);
2926 }
2927 case NameCategory::Query:
2928 return Type::any();
2929 case NameCategory::Mutation:
2930 return std::unexpected("`" + std::string{head} +
2931 "` is a mutation; read it with `result`, `pending` or "
2932 "`error`");
2933 case NameCategory::Model:
2934 return std::unexpected("`" + std::string{head} + "` is a model alias, not a value");
2935 case NameCategory::Transfer:
2936 if (path.empty() || !listed(kTransferValues, path.front())) {
2937 return std::unexpected("the transfer `" + std::string{head} +
2938 "` is read as `progress`, `pending` or `error`");
2939 }
2940 return Type::any();
2941 case NameCategory::Element:
2942 default: {
2943 KindSpec const* const spec = findKind(entry->element->kind, false);
2944 if (path.empty()) {
2945 return std::unexpected("the element `" + std::string{head} +
2946 "` is read through one of its values, "
2947 "such as `" +
2948 std::string{head} + ".visible`");
2949 }
2950 std::string const& value = path.front();
2951 bool const open = (spec->flags & kOpenValues) != 0;
2952 bool const known = listed(kAnyValues, value) || listed(spec->values, value) || open;
2953 if (!known) {
2954 return std::unexpected("the " + entry->element->kind + " `" + std::string{head} + "` has no value `" +
2955 value + "`");
2956 }
2957 bool const field = open && !listed(kAnyValues, value) && !listed(spec->values, value);
2958 Type const type = field ? Type::any() : elementValueType(value);
2959 return typeAlong(type, value, path.subspan(1));
2960 }
2961 }

3011/// @brief Parses a UI document and runs its load-time validation (spec 5 §3).
3012///
3013/// A document is refused when it does not parse, when a reference does not resolve, when a type does not check,
3014/// when the `let` or `watch` graph has a cycle, when an expression exceeds 256 nodes, when a node kind, command or
3015/// function is not in `ui/1`, when a must-understand key is unknown, or when a rule of spec 5 or spec 6 fails; the
3016/// list is in `docs/spec/ui/document.md`. The rules that need catalog facts run when @p options supplies them.
3017/// @param json The document's JSON.
3018/// @param options The catalog facts.
3019/// @return The document, or the first refusal, naming the document id, the JSON Pointer and the reason.
3020[[nodiscard]] inline std::expected<Document, Refusal> loadDocument(Json const& json, LoadOptions const& options = {}) {
3021 std::string documentId;
3022 if (auto const* const object = std::get_if<Json::object_t>(&json.data)) {
3023 auto const* const found = object->find("id");
3024 if (found != object->end()) {
3025 if (auto const* const text = std::get_if<std::string>(&found->second.data)) {
3026 documentId = *text;
3027 }
3028 }
3029 }
3030 try {
3031 Document doc = detail::DocumentParser{}.parse(json);
3032 detail::DocumentValidator{doc, options}.run();
3033 return doc;
3034 } catch (detail::DocumentRefused const& refused) {
3035 Refusal refusal = refused.refusal();
3036 refusal.document = documentId;
3037 return std::unexpected(std::move(refusal));
3038 }
3039}
