19/// @brief Reconstructs model state by replaying @p entries, in order, against a
20/// freshly created model instance.
21///
22/// Builds on the same `ModelRegistryFactory` / `ActionDispatcher` machinery
23/// `RemoteServer` already uses to dispatch incoming requests — no separate
24/// replay engine is needed. Because a model's entire state is exactly "initial
25/// state plus the ordered actions replayed against it", this both reconstructs
26/// state from a durable log and powers `SessionLog::undoLast()` below.
27///
28/// @param modelTypeId String type-id of the model to reconstruct (`ModelTraits<M>::typeId()`).
29/// @param entries Ordered entries to replay, typically from `IActionLog::entries()`.
30/// @param registry Model factory registry; defaults to the process-level singleton.
31/// @param dispatcher Action dispatcher; defaults to the process-level singleton.
32/// @return A freshly created holder with @p entries replayed against it.
33/// @throws std::runtime_error if @p modelTypeId or any entry's action type is unregistered.
34inline std::unique_ptr<::morph::model::detail::IModelHolder> replay(
35 std::string_view modelTypeId, const std::vector<LogEntry>& entries,
36 ::morph::model::detail::ModelRegistryFactory& registry = ::morph::model::detail::defaultRegistry(),
37 ::morph::model::detail::ActionDispatcher& dispatcher = ::morph::model::detail::defaultDispatcher()) {
38 auto holder = registry.create(modelTypeId);
39 // `registry.create` (via `ModelFactory::create`) auto-attaches the process
40 // default action log. Detach it before replaying: reconstruction re-runs the
41 // recorded actions, and without this each replayed dispatch would re-record
42 // into the live sink, corrupting the very audit trail we are reading from.
43 holder->attachActionLog(nullptr, {});
44 for (const auto& entry : entries) {
45 dispatcher.dispatch(entry.modelType, entry.actionType, *holder, entry.payload);
46 }
47 return holder;
48}

50/// @brief Full-fidelity, in-memory log of one model instance's executed actions.
51///
52/// Attach directly via `IModelHolder::attachActionLog` — every successfully
53/// executed loggable action is appended here in order, regardless of any
54/// `ActionLogPolicy::coalesce` setting. This full history is the raw material
55/// for `undoLast()`. `checkpoint()` is the one place coalescing actually
56/// happens: entries accumulated since the last checkpoint are reduced by
57/// `(modelType, entityKey, actionType)` — keeping only the latest occurrence
58/// where the action's policy says `coalesce == true`, keeping every occurrence
59/// otherwise — and only that reduced set reaches the durable sink.
60///
61/// @par Thread safety
62/// All public methods are thread-safe (guarded by an internal mutex).
63class SessionLog : public IActionLog {
64public:
65 /// @brief Appends @p entry, assigning a monotonically increasing `seq`.
66 ///
67 /// Always full fidelity: this is what `undoLast()` walks back through, so
68 /// nothing is coalesced or dropped here.
69 /// @param entry Entry to append; `seq` is overwritten regardless of the input value.
70 void append(LogEntry entry) override {
71 std::scoped_lock const lock{_mtx};
72 entry.seq = ++_nextSeq;
73 _all.push_back(std::move(entry));
74 }
75
76 /// @brief No-op — `checkpoint()` is this class's real commit point.
77 void flush() override {}
78
79 /// @brief Returns the full history (or one entity's slice of it) in append order.
80 /// @param entityKey If non-empty, restricts the result to that entity's entries.
81 /// @return Matching entries, in append order.
82 [[nodiscard]] std::vector<LogEntry> entries(std::string_view entityKey = {}) const override {
83 std::scoped_lock const lock{_mtx};
84 if (entityKey.empty()) {
85 return _all;
86 }
87 std::vector<LogEntry> out;
88 for (const auto& entry : _all) {
89 if (entry.entityKey == entityKey) {
90 out.push_back(entry);
91 }
92 }
93 return out;
94 }
95
96 /// @brief Drops the most recently appended entry and replays everything that
97 /// remains against a fresh model instance, reconstructing the state
98 /// the model was in immediately before that entry executed.
99 ///
100 /// No inverse/undo operations are needed on the action types themselves —
101 /// this reuses `replay()` over a shorter prefix of the same log. A no-op
102 /// (returns a freshly created, un-replayed holder) if the log is empty.
103 ///
104 /// @param modelTypeId String type-id of the model to reconstruct.
105 /// @param registry Model factory registry; defaults to the process-level singleton.
106 /// @param dispatcher Action dispatcher; defaults to the process-level singleton.
107 /// @return A freshly created holder with the pre-undo state replayed onto it.
108 std::unique_ptr<::morph::model::detail::IModelHolder> undoLast(
109 std::string_view modelTypeId,
110 ::morph::model::detail::ModelRegistryFactory& registry = ::morph::model::detail::defaultRegistry(),
111 ::morph::model::detail::ActionDispatcher& dispatcher = ::morph::model::detail::defaultDispatcher()) {
112 std::vector<LogEntry> remaining;
113 {
114 std::scoped_lock const lock{_mtx};
115 if (!_all.empty()) {
116 _all.pop_back();
117 _committedUpTo = std::min(_committedUpTo, _all.size());
118 }
119 remaining = _all;
120 }
121 return replay(modelTypeId, remaining, registry, dispatcher);
122 }
123
124 /// @brief Coalesces entries appended since the last checkpoint and forwards
125 /// the reduced set, in order, to @p durableSink; then flushes it.
126 ///
127 /// Advances the checkpoint regardless of whether @p durableSink->flush()
128 /// throws, matching `IOfflineQueue`-style at-least-once semantics: a
129 /// checkpoint is a forward-only commit point, not a transaction to retry.
130 /// A no-op if nothing has been appended since the last checkpoint.
131 ///
132 /// @param durableSink Receives only the coalesced entries — never the raw stream.
133 /// @param dispatcher Supplies each action's `coalesce` policy; defaults to
134 /// the process-level singleton (the same one actions were
135 /// registered against via `BRIDGE_REGISTER_ACTION`).
136 void checkpoint(IActionLog& durableSink,
137 ::morph::model::detail::ActionDispatcher& dispatcher = ::morph::model::detail::defaultDispatcher()) {
138 std::vector<LogEntry> pending;
139 {
140 std::scoped_lock const lock{_mtx};
141 if (_committedUpTo >= _all.size()) {
142 return;
143 }
144 pending.assign(_all.begin() + static_cast<std::ptrdiff_t>(_committedUpTo), _all.end());
145 _committedUpTo = _all.size();
146 }
147 for (auto& entry : coalesced(pending, dispatcher)) {
148 durableSink.append(std::move(entry));
149 }
150 durableSink.flush();
151 }
152
153private:
154 /// @brief Reduces @p pending by `(modelType, entityKey, actionType)`: the
155 /// latest occurrence overwrites earlier ones (in place, preserving
156 /// first-seen position) wherever @p dispatcher says the action
157 /// coalesces; every other entry is kept as-is.
158 static std::vector<LogEntry> coalesced(const std::vector<LogEntry>& pending,
159 ::morph::model::detail::ActionDispatcher& dispatcher) {
160 std::vector<LogEntry> out;
161 std::unordered_map<std::string, std::size_t> lastIndexForKey;
162 for (const auto& entry : pending) {
163 if (!dispatcher.coalesce(entry.modelType, entry.actionType)) {
164 out.push_back(entry);
165 continue;
166 }
167 std::string key = entry.modelType + '\0' + entry.entityKey + '\0' + entry.actionType;
168 auto iter = lastIndexForKey.find(key);
169 if (iter == lastIndexForKey.end()) {
170 lastIndexForKey.emplace(std::move(key), out.size());
171 out.push_back(entry);
172 } else {
173 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) — index from same loop
174 out[iter->second] = entry;
175 }
176 }
177 return out;
178 }
179
180 mutable std::mutex _mtx;
181 std::vector<LogEntry> _all;
182 std::size_t _committedUpTo{0};
183 uint64_t _nextSeq{0};
184};
