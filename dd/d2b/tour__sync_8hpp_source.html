22/// @brief Lists the runtime tours a server offers and keeps the list current.
23///
24/// `list` asks `morph.tours.list`. An answer makes the tours offered, loads each listed tour with `loadTourText` (one
25/// that does not load is logged at warn and left out), tells `setOnListed`'s handler, and has the session watch the
26/// record `morph.tours`. A server that never registered the model answers `unknown model type: morph.tours`: the sync
27/// then offers none, which is logged once per sync at debug, since a server without runtime tours is ordinary. Any
28/// other failure is logged at warn and keeps the last list. While offered, a `changes` event naming the record, a
29/// `resync` and a reconnect list again. Neither copyable nor movable; on the runtime's owner.
30class MORPH_API TourSync {
31public:
32 /// @brief A sync that has not listed yet.
33 /// @param dispatcher Where model calls go: the interpreter's own. Borrowed: it must outlive the sync.
34 /// @param session The session whose watch set carries `morph.tours` once offered; null for a source whose
35 /// server sends no events, such as an in-process one. Borrowed: it must outlive the sync.
36 explicit TourSync(Dispatcher& dispatcher MORPH_LIFETIMEBOUND, Session* session = nullptr);
37 TourSync(TourSync const&) = delete;
38 TourSync& operator=(TourSync const&) = delete;
39 TourSync(TourSync&&) = delete;
40 TourSync& operator=(TourSync&&) = delete;
41 /// @brief Stops the watch.
42 ~TourSync();
43
44 /// @brief Asks for the list, and learns whether the server offers runtime tours.
45 /// @param done Called once with whether they are offered; may be empty.
46 void list(std::function<void(bool offered)> done = {});
47
48 /// @brief Asks for the list again while offered; the replies of one sync arrive in call order.
49 /// @param done Called once this list's reply was handled, whatever it held; may be empty.
50 void relist(std::function<void()> done = {});
51
52 /// @brief The connection was made again: lists again while offered; probes again when the last failure was not the
53 /// absent model. The session sends the watch itself.
54 void reconnected();
55
56 /// @brief A server event: a `changes` naming `morph.tours`, or a `resync`, lists again while offered.
57 /// @param event The event.
58 void deliverEvent(wire::EventBody const& event);
59
60 /// @brief Forgets the list and the offer, stops the watch, and drops replies still in flight: a new frame's
61 /// session.
62 void reset();
63
64 /// @brief Whether the server offers runtime tours, as the last list found.
65 /// @return It.
66 [[nodiscard]] bool offered() const noexcept { return _offered; }
67
68 /// @brief The tours the last list held, loaded, in the server's order.
69 /// @return They; empty before the first list and without the model.
70 [[nodiscard]] std::vector<TourDecl> const& tours() const noexcept { return _tours; }
71
72 /// @brief Sets what is told after each list that answered.
73 /// @param onListed The handler, given the tours; empty tells nothing.
74 void setOnListed(std::function<void(std::vector<TourDecl> const&)> onListed) { _onListed = std::move(onListed); }
75
76private:
77 void call(std::function<void(std::string const&)> onReply,
78 std::function<void(std::string const&, std::string const&)> const& failed);
79 void listed(std::string const& reply);
80
81 Dispatcher* _dispatcher;
82 Session* _session;
83 std::shared_ptr<ModelChannel> _channel;
84 std::vector<TourDecl> _tours;
85 std::function<void(std::vector<TourDecl> const&)> _onListed;
86 bool _offered = false;
87 bool _absent = false;
88 bool _absentLogged = false;
89 // Last, so it stops every delivery before anything a reply would touch goes; `reset` replaces it.
90 std::unique_ptr<async::CallbackScope> _scope = std::make_unique<async::CallbackScope>();
91};
