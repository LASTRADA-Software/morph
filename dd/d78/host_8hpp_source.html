173/// @brief One mounted screen, as its host sees it.
174///
175/// The host reads `title`, `dirty` and `identity` to show the tab and the unsaved indicator, and closes the screen
176/// through `requestClose`. It never destroys a handle whose close is pending, except after `forceClose`.
177class MORPH_API ScreenHandle {
178public:
179 ScreenHandle() = default;
180 ScreenHandle(ScreenHandle const&) = delete;
181 ScreenHandle& operator=(ScreenHandle const&) = delete;
182 ScreenHandle(ScreenHandle&&) = delete;
183 ScreenHandle& operator=(ScreenHandle&&) = delete;
184 virtual ~ScreenHandle() = default;
185
186 /// @brief The screen's title: the document's `title`, as text.
187 /// @return It; it keeps its last value once the screen is closed.
188 [[nodiscard]] virtual reactive::Computed<std::string> const& title() const = 0;
189
190 /// @brief Whether the screen has unsaved changes: the document's `dirty`.
191 /// @return It; `false` once the screen is closed.
192 [[nodiscard]] virtual reactive::Computed<bool> const& dirty() const = 0;
193
194 /// @brief The screen's identity: the document's `identity`, `null` when it declares none. A host re-keys a tab
195 /// when it changes.
196 /// @return It; it keeps its last value once the screen is closed.
197 [[nodiscard]] virtual reactive::Computed<Value> const& identity() const = 0;
198
199 /// @brief The screen's head beside its title, for a shell to show above it.
200 /// @return It, valid as long as the handle: a document screen's (empty values when the document declares no
201 /// head); null for a host screen unless its handle overrides this.
202 [[nodiscard]] virtual ScreenHead* head() noexcept { return nullptr; }
203
204 /// @brief Asks the screen to close (spec 6 §3).
205 ///
206 /// @p done is called exactly once, posted to the owner, never from inside the document's own handler: with `true`
207 /// when the screen closed, `false` when it stays open. A request while one is pending joins it, and is answered
208 /// with the same outcome.
209 /// @param done Called with the outcome.
210 virtual void requestClose(std::function<void(bool closed)> done) = 0;
211
212 /// @brief Closes the screen without asking: skips `onClose` and the close guard, runs `onUnmount`, and answers
213 /// a pending `requestClose` with `true`. For application quit.
214 virtual void forceClose() = 0;
215
216 /// @brief Whether the screen can save its changes: a document screen that declares `onSave`, or a host screen
217 /// whose handle implements `save`. A close guard offers *save and close* only when this is `true`.
218 /// @return It; `false` unless a derived handle says otherwise.
219 [[nodiscard]] virtual bool canSave() const { return false; }
220
221 /// @brief Saves the screen's changes (spec 6 §15) and answers whether it is then clean.
222 ///
223 /// A document screen runs `onSave`, waits until nothing pending is left, and answers `true` when `dirty` is then
224 /// not `true`. A save while one waits joins it. A screen that cannot save answers `false` at once. @p done is not
225 /// called when the screen closes before the save has finished.
226 /// @param done Called with whether the screen is clean.
227 // NOLINTNEXTLINE(performance-unnecessary-value-param): the base only calls it; overrides keep it.
228 virtual void save(std::function<void(bool clean)> done);
229};

289/// @brief What the user chose for a dirty screen that is being closed (spec 6 §15).
290enum class CloseChoice : std::uint8_t {
291 Save, ///< Run the screen's save, and close it only if it is then clean.
292 Discard, ///< Close it without saving.
293 Cancel, ///< Keep it open.
294};

296/// @brief What decides a close the document does not decide itself: a dirty screen that declares no `onClose`.
297///
298/// It asks the user, typically with a dialog, and calls `choose` once. It offers *save and close* only when
299/// `screen.canSave()`; `Save` on a screen that cannot save keeps it open. The handle carries the choice out, so
300/// `choose` may be called after the handle is gone, and then does nothing.
301using CloseGuard = std::function<void(ScreenHandle& screen, std::function<void(CloseChoice choice)> choose)>;

398 /// @brief Starts the requests of a screen's `prefetch` queries before it mounts (spec 5 §5), so the screen mounts

474 /// @brief Answers with a cancel the call of a navigation this host handled without mounting a document screen:
475 /// the oldest unclaimed call of that screen and params, the one a mount would have adopted. Its `onCancel`
476 /// runs, and the cancel is reported to `diagnostics`.
477 ///
478 /// A host calls it once for each navigation whose `NavigateRequest::call` is set and that mounted no document
479 /// screen: one that focused an open tab, a refused mount, a host screen (which adopts no call) and a navigation

487 // How a prefetch reaches the instance the mounted screen's alias will bind.
