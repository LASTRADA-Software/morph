38/// @brief Why a screen could not be mounted.

46/// @brief One mounted screen, as its host sees it.
47///
48/// The host reads `title`, `dirty` and `identity` to show the tab and the unsaved indicator, and closes the screen
49/// through `requestClose`. It never destroys a handle whose close is pending, except after `forceClose`.
50class ScreenHandle {
51public:
52 ScreenHandle() = default;
53 ScreenHandle(ScreenHandle const&) = delete;
54 ScreenHandle& operator=(ScreenHandle const&) = delete;
55 ScreenHandle(ScreenHandle&&) = delete;
56 ScreenHandle& operator=(ScreenHandle&&) = delete;
57 virtual ~ScreenHandle() = default;
58
59 /// @brief The screen's title: the document's `title`, as text.
60 /// @return It; it keeps its last value once the screen is closed.
61 [[nodiscard]] virtual reactive::Computed<std::string> const& title() const = 0;
62
63 /// @brief Whether the screen has unsaved changes: the document's `dirty`.
64 /// @return It; `false` once the screen is closed.
65 [[nodiscard]] virtual reactive::Computed<bool> const& dirty() const = 0;
66
67 /// @brief The screen's identity: the document's `identity`, `null` when it declares none. A host re-keys a tab
68 /// when it changes.
69 /// @return It; it keeps its last value once the screen is closed.
70 [[nodiscard]] virtual reactive::Computed<Value> const& identity() const = 0;
71
72 /// @brief Asks the screen to close (spec 6 §3).
73 ///
74 /// @p done is called exactly once, posted to the owner, never from inside the document's own handler: with `true`
75 /// when the screen closed, `false` when it stays open. A request while one is pending joins it, and is answered
76 /// with the same outcome.
77 /// @param done Called with the outcome.
78 virtual void requestClose(std::function<void(bool closed)> done) = 0;
79
80 /// @brief Closes the screen without asking: skips `onClose` and the close guard, runs `onUnmount`, and answers
81 /// a pending `requestClose` with `true`. For application quit.
82 virtual void forceClose() = 0;
83};

167/// The environment `identityOf` evaluates in: the params, and nothing else. Reading anything else marks it unknown.
168class ParamsEnv final : public Environment {
169public:
170 /// @brief An environment over @p params.
171 /// @param params The params; borrowed for the environment's lifetime.
172 explicit ParamsEnv(ParamMap const& params) : _params{&params} {}
173
174 [[nodiscard]] Value ref(std::string_view head, std::span<std::string const> path) const override {
175 auto const found = _params->find(head);
176 if (found == _params->end()) {
177 _unknown = true;
178 return Value{};
179 }
180 Value value = found->second;
181 for (std::string const& step : path) {
182 ValueRecord const* const record = value.asRecord();
183 if (record == nullptr || !record->contains(step)) {
184 return Value{};
185 }
186 value = record->at(step);
187 }
188 return value;
189 }
190
191 [[nodiscard]] Value row(std::span<std::string const> path) const override { return unknown(path); }
192
193 [[nodiscard]] Value control(ExprKind form, std::string_view name) const override {
194 static_cast<void>(form);
195 static_cast<void>(name);
196 _unknown = true;
197 return Value{};
198 }
199
200 [[nodiscard]] Value pref(std::string_view point, std::span<std::string const> path) const override {
201 static_cast<void>(point);
202 return unknown(path);
203 }
204
205 [[nodiscard]] Value client(ExprKind form, std::string_view name) const override {
206 static_cast<void>(form);
207 static_cast<void>(name);
208 _unknown = true;
209 return Value{};
210 }
211
212 [[nodiscard]] FormatContext const& formatting() const override { return _formatting; }
213
214 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): Environment's signature.
215 void report(std::string_view path, std::string_view reason) const override {
216 static_cast<void>(path);
217 static_cast<void>(reason);
218 _unknown = true;
219 }
220
221 /// @brief Whether the evaluation read anything but the params, or failed.
222 /// @return It.
223 [[nodiscard]] bool unknown() const noexcept { return _unknown; }
224
225private:
226 [[nodiscard]] Value unknown(std::span<std::string const> path) const {
227 static_cast<void>(path);
228 _unknown = true;
229 return Value{};
230 }
231
232 ParamMap const* _params;
233 FormatContext _formatting;
234 mutable bool _unknown = false;
235};

249/// @brief The framework's `ScreenHost` over UI documents: it mounts them through an `Interpreter`, closes them by the
250/// protocol of spec 6 §3, and hands a document's `navigate` to `onNavigate`, which the deriving host
251/// implements.
252///
253/// It is the interpreter's `InterpreterHost` for `navigate`; every other command goes to the application's host, if
254/// one was given. Everything runs on the runtime's owner.
255class DocumentScreenHost : public ScreenHost, private InterpreterHost {
256public:
257 /// @brief A host over @p session and @p dispatcher.
258 /// @param session The session the screens share. Borrowed: it outlives the host.
259 /// @param dispatcher Where model calls go. Borrowed: it outlives the host.
260 /// @param options The documents, the backend, the close guard and the application's host.
261 /// @throws std::invalid_argument without a document lookup or a backend.
262 DocumentScreenHost(Session& session MORPH_LIFETIMEBOUND, Dispatcher& dispatcher MORPH_LIFETIMEBOUND,
263 DocumentScreenHostOptions options)
264 : _options{std::move(options)},
265 _session{&session},
266 _dispatcher{&dispatcher},
267 _interpreter{
268 session, dispatcher,
269 InterpreterOptions{.host = static_cast<InterpreterHost*>(this), .diagnostics = _options.diagnostics}} {
270 if (!_options.documents || _options.backend == nullptr) {
271 throw std::invalid_argument("DocumentScreenHost: a document lookup and a backend are required");
272 }
273 }

295 /// @brief The identity a document screen would have with @p params: its `identity` evaluated over the params.
296 /// @param screenId The screen id.
297 /// @param params The params.
298 /// @return Its canonical text, or `std::nullopt` for an unknown screen, one without `identity`, or one whose
299 /// identity reads more than its params or is `null`.
300 [[nodiscard]] std::optional<std::string> identityOf(std::string_view screenId, ParamMap const& params) override {
301 std::shared_ptr<Document const> const document = _options.documents(screenId);
302 if (!document || !document->identity) {
303 return std::nullopt;
304 }
305 detail::host::ParamsEnv const env{params};
306 Value const identity = evaluate(*document->identity, env);
307 if (env.unknown() || identity.isNull()) {
308 return std::nullopt;
309 }
310 return detail::canonicalJson(identity);
311 }

313 /// @brief Starts the requests of a screen's `prefetch` queries before it mounts (spec 5 §5), so the screen mounts
314 /// on a warm cache: what a shell does when the pointer rests on an entry for a screen that is not open.
315 ///
316 /// Each named query's body is evaluated over @p params alone and its entry warmed through the session's cache,
317 /// under the token the mounted screen's alias will have. A keyed alias's token follows from its key, and a
318 /// stateless model's is shared. A private alias gets a fresh token when the screen mounts, so nothing could read
319 /// what a prefetch warmed: its queries are skipped, as are evaluations, a key or body that reads more than the
320 /// params, and an entry that is already fresh or in flight.
321 /// @param screenId The screen id; an unknown one does nothing.
322 /// @param params The params it would mount with.
323 void prefetch(std::string_view screenId, ParamMap const& params) {
324 std::shared_ptr<Document const> const document = _options.documents(screenId);
325 if (!document) {
326 return;
327 }
328 for (std::string const& name : document->prefetch) {
329 auto const query = std::ranges::find(document->scope.queries, name, &QueryDecl::name);
330 if (query == document->scope.queries.end() || query->evaluate) {
331 continue;
332 }
333 auto const model = std::ranges::find(document->scope.models, query->model, &ModelDecl::name);
334 if (model == document->scope.models.end()) {
335 continue;
336 }
337 std::optional<Binding> const binding = bindingOf(*model, params);
338 if (!binding) {
339 continue;
340 }
341 detail::host::ParamsEnv const env{params};
342 Value const body = query->body ? evaluate(*query->body, env) : Value{ValueRecord{}};
343 if (env.unknown()) {
344 continue;
345 }
346 std::shared_ptr<ModelChannel> const channel =
347 _dispatcher->bind(model->type, binding->sharing, binding->instanceKey);
348 _session->prefetch(channel, binding->token, model->type, query->action, body);
349 }
350 }

396 // How a prefetch reaches the instance the mounted screen's alias will bind.
397 struct Binding {
398 std::string token;
399 bridge::BindSharing sharing = bridge::BindSharing::Private;
400 std::string instanceKey;
401 };
402
403 // The binding of @p model over @p params, or nothing for an instance the mounted screen will not share.
404 [[nodiscard]] std::optional<Binding> bindingOf(ModelDecl const& model, ParamMap const& params) const {
405 InstanceMode mode = model.instance;
406 if (mode == InstanceMode::Default) {
407 Catalog const& catalog = _session->options().catalog;
408 std::optional<ModelInfo> const info = catalog.model ? catalog.model(model.type) : std::nullopt;
409 mode = info && info->stateless ? InstanceMode::Shared : InstanceMode::Private;
410 }
411 if (mode == InstanceMode::Shared) {
412 return Binding{.token = "shared", .sharing = bridge::BindSharing::Private, .instanceKey = {}};
413 }
414 if (mode != InstanceMode::Keyed || !model.key) {
415 return std::nullopt;
416 }
417 detail::host::ParamsEnv const env{params};
418 std::string const key = detail::instanceKeyText(evaluate(*model.key, env));
419 if (env.unknown() || key.empty()) {
420 return std::nullopt;
421 }
422 return Binding{.token = "shared:" + key, .sharing = bridge::BindSharing::Shared, .instanceKey = key};
423 }

425 // The mount itself, under @p parent or as a backend root.
426 std::expected<std::unique_ptr<ScreenHandle>, MountError> mountAt(std::string_view screenId, ParamMap const& params,
427 ContainerWidget* parent) {
428 std::shared_ptr<Document const> document = _options.documents(screenId);
429 if (!document) {
430 return std::unexpected(MountError{.screenId = std::string{screenId}, .reason = "unknown screen"});
431 }
432 auto screen =
433 _interpreter.mount(std::move(document), params, *_options.backend, MountOptions{.parent = parent});
434 if (!screen) {
435 std::string reason = screen.error().reason;
436 if (!screen.error().path.empty()) {
437 reason = screen.error().path + ": " + reason;
438 }
439 return std::unexpected(MountError{.screenId = std::string{screenId}, .reason = std::move(reason)});
440 }
441 auto handle = std::make_unique<Handle>(std::move(*screen), _options.guard);
442 adoptCall(*handle, screenId, params);
443 return handle;
444 }

446 /// A screen opened to answer its opener (spec 6 §16): answered exactly once, by `return`, by `close: self`, or by
447 /// the screen closing any other way, which is a cancel.
448 struct Call {
449 std::string screen;
450 ParamMap params;
451 std::function<void(Value)> onResult;
452 std::function<void()> onCancel;
453 bool answered = false;
454
455 void answer(std::optional<Value> result) {
456 if (answered) {
457 return;
458 }
459 answered = true;
460 if (result) {
461 if (onResult) {
462 onResult(std::move(*result));
463 }
464 } else if (onCancel) {
465 onCancel();
466 }
467 }
468 };

470 /// The handle of a document screen. Its state is shared with the posts it makes, so a post that runs after the
471 /// handle is gone finds the screen already closed.
472 class Handle final : public ScreenHandle {
473 public:
474 Handle(std::unique_ptr<Screen> screen, CloseGuard guard)
475 : _state{std::make_shared<State>(std::move(screen), std::move(guard))} {
476 _state->self = this;
477 }
478
479 ~Handle() override {
480 _state->self = nullptr;
481 try {
482 _state->finish(true);
483 } catch (...) { // NOLINT(bugprone-empty-catch): a destructor has no one to report to; the screen is gone.
484 }
485 }
486
487 Handle(Handle const&) = delete;
488 Handle& operator=(Handle const&) = delete;
489 Handle(Handle&&) = delete;
490 Handle& operator=(Handle&&) = delete;
491
492 [[nodiscard]] reactive::Computed<std::string> const& title() const override { return _state->title; }
493 [[nodiscard]] reactive::Computed<bool> const& dirty() const override { return _state->dirty; }
494 [[nodiscard]] reactive::Computed<Value> const& identity() const override { return _state->identity; }
495
496 void requestClose(std::function<void(bool closed)> done) override { State::request(_state, std::move(done)); }
497
498 void forceClose() override { _state->finish(true); }
499
500 [[nodiscard]] Widget* root() const { return _state->screen ? &_state->screen->root() : nullptr; }
501
502 /// The call this screen answers, if it was opened as one.
503 void answers(std::shared_ptr<Call> call) { _state->call = std::move(call); }
504
505 /// Closes the screen after its own `return` or `close: self` has answered: posted, since the command that
506 /// asked for it is still running in the screen.
507 void closeItself(std::function<void(ScreenHandle&)> closed) {
508 std::weak_ptr<State> const weak = _state;
509 _state->runtime->owner().post([weak, closed = std::move(closed)] {
510 auto const held = weak.lock();
511 if (!held || !held->screen) {
512 return;
513 }
514 held->finish(true);
515 if (held->self != nullptr && closed) {
516 closed(*held->self);
517 }
518 });
519 }
520
521 /// Whether this screen answers @p call and is still open.
522 [[nodiscard]] bool answering(std::shared_ptr<Call> const& call) const {
523 return _state->screen && _state->call == call;
524 }
525
526 private:
527 struct State {
528 State(std::unique_ptr<Screen> mounted, CloseGuard closeGuard)
529 : screen{std::move(mounted)},
530 runtime{&screen->runtime()},
531 guard{std::move(closeGuard)},
532 closed{*runtime, false},
533 title{*runtime,
534 [this] {
535 if (!closed.get()) {
536 lastTitle = detail::host::titleText(screen->title());
537 }
538 return lastTitle;
539 }},
540 dirty{*runtime, [this] { return !closed.get() && screen->dirty(); }},
541 identity{*runtime, [this] {
542 if (!closed.get()) {
543 lastIdentity = screen->identity();
544 }
545 return lastIdentity;
546 }} {}
547
548 State(State const&) = delete;
549 State& operator=(State const&) = delete;
550 State(State&&) = delete;
551 State& operator=(State&&) = delete;
552 ~State() = default;
553
554 /// Starts a close, or joins the one in progress.
555 static void request(std::shared_ptr<State> const& state, std::function<void(bool)> done) {
556 if (!state->screen) {
557 state->post({std::move(done)}, true);
558 return;
559 }
560 state->waiting.push_back(std::move(done));
561 if (state->closing) {
562 return;
563 }
564 state->closing = true;
565 std::shared_ptr<bool const> const cancelled = state->screen->beginClose();
566 if (cancelled == nullptr) {
567 bool const isDirty = state->screen->dirty();
568 if (!isDirty) {
569 state->finish(true);
570 } else if (state->guard && state->self != nullptr) {
571 std::weak_ptr<State> const weak = state;
572 state->guard(*state->self, [weak](bool close) {
573 if (auto const held = weak.lock()) {
574 held->decide(close);
575 }
576 });
577 } else {
578 state->decide(false);
579 }
580 return;
581 }
582 // Closes once onClose and everything it started has ended, unless it ran cancelClose.
583 std::weak_ptr<State> const weak = state;
584 state->settled = std::make_unique<reactive::Effect>(*state->runtime, [weak, cancelled] {
585 auto const held = weak.lock();
586 if (!held || !held->screen || held->screen->pending()) {
587 return;
588 }
589 held->runtime->owner().post([weak, cancelled] {
590 if (auto const again = weak.lock()) {
591 again->settled.reset();
592 again->decide(!*cancelled);
593 }
594 });
595 });
596 }
597
598 /// Ends the close in progress: closes the screen or keeps it open, and answers every request.
599 void decide(bool close) {
600 if (!closing) {
601 return;
602 }
603 if (close) {
604 finish(true);
605 return;
606 }
607 closing = false;
608 post(std::exchange(waiting, {}), false);
609 }
610
611 /// Closes the screen now, answering every pending request with `true`.
612 void finish(bool answer) {
613 settled.reset();
614 closing = false;
615 if (call) {
616 call->answer(std::nullopt); // a no-op once `return` or `close: self` has answered
617 }
618 if (screen) {
619 // The last values a host reads once the screen is gone, whether or not it read them before.
620 lastTitle = detail::host::titleText(screen->title());
621 lastIdentity = screen->identity();
622 runtime->batch([this] { closed.set(true); });
623 screen.reset();
624 }
625 post(std::exchange(waiting, {}), answer);
626 }

664 std::function<void()> onCancel) override {
665 if (request.screen.empty()) {
666 return _options.application != nullptr &&
667 _options.application->navigate(request, std::move(onResult), std::move(onCancel));
668 }
669 if (onResult || onCancel) {
670 ValueRecord const* const params = request.params.asRecord();
671 _unclaimed.push_back(std::make_shared<Call>(Call{.screen = request.screen,
672 .params = params == nullptr ? ParamMap{} : *params,
673 .onResult = std::move(onResult),
674 .onCancel = std::move(onCancel),
675 .answered = false}));
676 }
677 onNavigate(request);

681 bool closeSelf(std::string const& screen, std::optional<Value> result) override {
682 // The innermost open call of that screen answers; a screen opened without one is the application's.
683 for (OpenCall const& open : std::views::reverse(_open)) {
684 std::shared_ptr<Call> const call = open.call.lock();
685 if (call == nullptr || call->screen != screen || call->answered || open.handle == nullptr ||
686 !open.handle->answering(call)) {
687 continue;
688 }
689 call->answer(std::move(result));
690 open.handle->closeItself([this](ScreenHandle& closed) { onScreenClosed(closed); });
691 return true;
692 }
693 return _options.application != nullptr && _options.application->closeSelf(screen, std::move(result));
694 }

696 /// Ties a mounted screen to the call that opened it: the oldest unclaimed call of that screen and params.
697 void adoptCall(Handle& handle, std::string_view screenId, ParamMap const& params) {
698 std::erase_if(_open, [](OpenCall const& open) { return open.call.expired(); });
699 auto const found = std::ranges::find_if(_unclaimed, [&](std::shared_ptr<Call> const& call) {
700 return call->screen == screenId && call->params == params;
701 });
702 if (found == _unclaimed.end()) {
703 return;
704 }
705 std::shared_ptr<Call> const call = std::move(*found);
706 _unclaimed.erase(found);
707 handle.answers(call);
708 _open.push_back(OpenCall{.call = call, .handle = &handle});
709 }
