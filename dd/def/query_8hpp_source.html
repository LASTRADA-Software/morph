49/// @brief Which rows of the view a request wants.
50struct PageRequest {

53 /// @brief Rows wanted.

57 /// @param other The request to compare with.

62/// @brief A table's sort, filters and page, as a list action's body carries them.
63///
64/// The member that holds one is marked `x-table` in the action's schema, which
65/// is how a client knows to send the table's state in it.
66struct TableQuery {
67 /// @brief The sort chain.
68 std::vector<SortKey> sort;
69 /// @brief The filter.
70 FilterSpec filters;
71 /// @brief The page.
72 PageRequest page;
73
74 /// @brief The schema keyword that marks a member of this type.
75 /// @return `"x-table"`.
76 [[nodiscard]] static constexpr std::string_view schemaMarker() noexcept { return "x-table"; }
77};

79/// @brief One page of a list action's reply.
80/// @tparam Row The row type.
81template <typename Row>
82struct Page {
83 /// @brief The rows, in view order.
84 std::vector<Row> rows;
85 /// @brief Rows in the whole filtered view; absent when the server did not
86 /// count them or the count is above its cap.
87 std::optional<std::int64_t> total;
88 /// @brief The view row of `rows[0]`.
89 std::int64_t offset = 0;
90};

92/// @brief The bounds a server applies to every `TableQuery`, an untrusted body.
93struct QueryLimits {
94 /// @brief `page.limit` above this is clamped to it.
95 std::int64_t maxLimit = 1000;
96 /// @brief `page.offset` above this is refused.
97 std::int64_t maxOffset = 10'000'000;
98 /// @brief Sort keys.
99 std::size_t maxSortKeys = 8;
100 /// @brief Filtered columns.
101 std::size_t maxFilterColumns = 32;
102 /// @brief `include` plus `exclude` entries of one column or group.
103 std::size_t maxEntries = 32;
104 /// @brief Groups.
105 std::size_t maxGroups = 8;
106 /// @brief Columns of one group.
107 std::size_t maxGroupColumns = 16;
108 /// @brief Bytes of one filter value.
109 std::size_t maxTextLength = 256;
110 /// @brief A total above this is not reported.
111 std::int64_t totalCap = 100'000;
112};

116/// @brief A `LimitExceeded` error.
117/// @param column The column concerned, or empty.
118/// @param message The description.
119/// @return The error.
120[[nodiscard]] inline TableError limitError(std::string column, std::string message) {
121 return TableError{
122 .code = TableErrorCode::LimitExceeded, .column = std::move(column), .message = std::move(message)};
123}

125/// @brief Whether any operand of an entry is longer than @p max bytes.
126/// @param entry The entry.
127/// @param max The bound.
128/// @return `true` when one is.
129[[nodiscard]] inline bool entryTooLong(FilterEntry const& entry, std::size_t max) {
130 auto const tooLong = [max](std::optional<std::string> const& value) { return value && value->size() > max; };
131 return tooLong(entry.eq) || tooLong(entry.ne) || tooLong(entry.lt) || tooLong(entry.le) || tooLong(entry.gt) ||
132 tooLong(entry.ge) || tooLong(entry.contains) || tooLong(entry.startsWith) || tooLong(entry.endsWith) ||
133 (entry.between && ((*entry.between)[0].size() > max || (*entry.between)[1].size() > max));
134}
135
136/// @brief Whether any of @p entries has an operand longer than @p max bytes.
137/// @param entries The entries.
138/// @param max The bound.
139/// @return `true` when one has.
140[[nodiscard]] inline bool anyTooLong(std::span<FilterEntry const> entries, std::size_t max) {
141 return std::ranges::any_of(entries, [max](FilterEntry const& entry) { return entryTooLong(entry, max); });
142}

144/// @brief Checks that a column exists and the principal may read it.
145/// @param columnId The column's id.
146/// @param columns The row schema's columns.
147/// @param mayRead Whether the principal may read a column; empty allows every column.
148/// @return The error, or nothing.
149[[nodiscard]] inline std::optional<TableError> checkColumn(std::string const& columnId,
150 std::span<ColumnInfo const> columns,
151 std::function<bool(std::string_view)> const& mayRead) {
152 if (!findColumn(columns, columnId)) {
153 return unknownColumn(columnId);
154 }
155 if (mayRead && !mayRead(columnId)) {
156 return TableError{.code = TableErrorCode::ColumnNotReadable,
157 .column = columnId,
158 .message = "column " + columnId + " is not readable"};
159 }
160 return std::nullopt;
161}
162
163/// @brief Checks a filter's columns, entry counts and value lengths.
164/// @param filters The filter.
165/// @param columns The row schema's columns.
166/// @param limits The bounds.
167/// @param mayRead Whether the principal may read a column.
168/// @return The first violation, or nothing.
169[[nodiscard]] inline std::optional<TableError> checkFilters(FilterSpec const& filters,
170 std::span<ColumnInfo const> columns,
171 QueryLimits const& limits,
172 std::function<bool(std::string_view)> const& mayRead) {
173 if (filters.columns.size() > limits.maxFilterColumns) {
174 return limitError({}, "too many filtered columns");
175 }
176 for (auto const& [columnId, filter] : filters.columns) {
177 if (auto error = checkColumn(columnId, columns, mayRead)) {
178 return error;
179 }
180 if (filter.include.size() + filter.exclude.size() > limits.maxEntries) {
181 return limitError(columnId, "too many filter entries");
182 }
183 if (anyTooLong(filter.include, limits.maxTextLength) || anyTooLong(filter.exclude, limits.maxTextLength)) {
184 return limitError(columnId, "filter value too long");
185 }
186 }
187 if (filters.groups.size() > limits.maxGroups) {
188 return limitError({}, "too many filter groups");
189 }
190 for (auto const& group : filters.groups) {
191 if (group.columns.size() > limits.maxGroupColumns || group.include.size() > limits.maxEntries) {
192 return limitError({}, "filter group too large");
193 }
194 for (auto const& columnId : group.columns) {
195 if (auto error = checkColumn(columnId, columns, mayRead)) {
196 return error;
197 }
198 }
199 if (anyTooLong(group.include, limits.maxTextLength)) {
200 return limitError({}, "filter value too long");
201 }
202 }
203 return std::nullopt;
204}

208/// @brief Checks @p query against the table's columns and @p limits.
209///
210/// Every column named must be one of @p columns, a closed set, and one the
211/// principal may read: a filter on a hidden column is an oracle for its value,
212/// so hiding a column is not a security control. `page.limit` is clamped;
213/// every other bound is a refusal.
214/// @param query The query, as received.
215/// @param columns The row schema's columns.
216/// @param limits The bounds.
217/// @param mayRead Whether the principal may read a column; empty allows every column.
218/// @return The query with its limit clamped, or the first violation.
219[[nodiscard]] inline std::expected<TableQuery, TableError> validate(
220 TableQuery query, std::span<ColumnInfo const> columns, QueryLimits const& limits = {},
221 std::function<bool(std::string_view)> const& mayRead = {}) {
222 if (query.page.offset < 0 || query.page.offset > limits.maxOffset) {
223 return std::unexpected(detail::limitError({}, "page.offset out of range"));
224 }
225 if (query.page.limit < 0) {
226 return std::unexpected(detail::limitError({}, "page.limit is negative"));
227 }
228 query.page.limit = std::min(query.page.limit, limits.maxLimit);
229 if (query.sort.size() > limits.maxSortKeys) {
230 return std::unexpected(detail::limitError({}, "too many sort keys"));
231 }
232 for (auto const& key : query.sort) {
233 if (auto error = detail::checkColumn(key.column, columns, mayRead)) {
234 return std::unexpected(std::move(*error));
235 }
236 }
237 if (auto error = detail::checkFilters(query.filters, columns, limits, mayRead)) {
238 return std::unexpected(std::move(*error));
239 }
240 return query;
241}

243/// @brief Escapes `%`, `_` and the escape character for a SQL `LIKE`
244/// pattern, so a database-backed server keeps them ordinary characters
245/// as the filter semantics require.
246/// @param text A filter value.
247/// @param escape The `ESCAPE` character the query declares.
248/// @return The escaped text; wrap it in `%` for `contains` and so on.
249[[nodiscard]] inline std::string escapeLikePattern(std::string_view text, char escape = '\\') {
250 std::string out;
251 out.reserve(text.size());
252 for (char const character : text) {
253 if (character == '%' || character == '_' || character == escape) {
254 out.push_back(escape);
255 }
256 out.push_back(character);
257 }
258 return out;
259}

272/// @brief The rows of one page, as source row indices.
273struct PageRows {
274 /// @brief Source rows, in view order.
275 std::vector<std::size_t> rows;
276 /// @brief Rows in the whole filtered view, unless above `QueryLimits::totalCap`.
277 std::optional<std::int64_t> total;
278 /// @brief The view row of `rows[0]`.
279 std::int64_t offset = 0;
280};

282/// @brief Runs @p query over @p source in the calling thread: the reference
283/// implementation of `server` mode.
284/// @param source The server's rows.
285/// @param query The request body's table query.
286/// @param options Limits, services and column access.
287/// @return The page, or the typed error the client shows.
288[[nodiscard]] inline std::expected<PageRows, TableError> apply(DataSource const& source, TableQuery const& query,
289 ApplyOptions const& options = {}) {
290 auto const columns = source.columns();
291 auto checked = validate(query, columns, options.limits, options.mayRead);
292 if (!checked) {
293 return std::unexpected(std::move(checked.error()));
294 }
295 // One copy of the services for this call, the filter's operands included:
296 // a collator that is not safe to share clones itself here, so concurrent
297 // calls never fold on the one the caller passed.
298 auto services = options.services.forTask();
299 auto filter = compileFilter(checked->filters, columns, services);
300 if (!filter) {
301 return std::unexpected(std::move(filter.error()));
302 }
303 detail::JobInput input;
304 input.snapshot = source.snapshot();
305 input.columns.assign(columns.begin(), columns.end());
306 input.services = std::move(services);
307 input.keys.resize(columns.size());
308 for (auto const& key : checked->sort) {
309 if (auto const column = detail::findColumn(columns, key.column)) {
310 input.needed.push_back(*column);
311 }
312 }
313 for (auto const column : filter->columnsUsed()) {
314 input.needed.push_back(column);
315 }
316 std::ranges::sort(input.needed);
317 auto const [first, last] = std::ranges::unique(input.needed);
318 input.needed.erase(first, last);
319 input.sort = checked->sort;
320 input.filter = std::make_shared<CompiledFilter const>(std::move(*filter));
321 input.previousSnapshot = input.snapshot;
322 detail::ViewJob job{std::move(input)};
323 static_cast<void>(job.step(kNoDeadline, ::core::async::StopToken{}));
324 if (job.failure()) {
325 return std::unexpected(TableError{
326 .code = TableErrorCode::ServiceFailed, .column = {}, .message = "a service failed: " + *job.failure()});
327 }
328 auto const view = job.takeResult().view;
329
330 PageRows out;
331 out.offset = checked->page.offset;
332 auto const count = static_cast<std::int64_t>(view.size());
333 if (count <= options.limits.totalCap) {
334 out.total = count;
335 }
336 auto const begin = std::min(checked->page.offset, count);
337 // Compared, not added: `maxLimit` may be as large as INT64_MAX.
338 auto const end = checked->page.limit >= count - begin ? count : begin + checked->page.limit;
339 for (auto i = begin; i < end; ++i) {
340 out.rows.push_back(view[static_cast<std::size_t>(i)]);
341 }
342 return out;
343}

358/// @brief A type glaze writes as one of its own members, as morph's `Ranged`,
359/// `Choice`, `Multiline` and `Tagged` are: its `glz::meta` value is a
360/// member pointer.
361template <typename V>
362concept MemberWrapper = requires { glz::meta<V>::value; } &&
363 std::is_member_object_pointer_v<std::remove_cvref_t<decltype(glz::meta<V>::value)>>;
364
365/// @brief The member a `MemberWrapper` is written as.
366/// @tparam V The wrapper type.
367template <typename V>
368using WrappedMember = std::remove_cvref_t<decltype(std::declval<V const&>().*(glz::meta<V>::value))>;
369
370/// @brief The column kind a member type gives.
371/// @tparam T The member type.
372/// @return The kind.
373template <typename T>
374[[nodiscard]] constexpr ColumnKind kindOf() {
375 using V = std::remove_cvref_t<T>;
376 if constexpr (IsOptional<V>::value) {
377 return kindOf<typename V::value_type>();
378 } else if constexpr (std::is_same_v<V, bool>) {
379 return ColumnKind::Bool;
380 } else if constexpr (std::integral<V>) {
381 return ColumnKind::Integer;
382 } else if constexpr (std::floating_point<V>) {
383 return ColumnKind::Number;
384 } else if constexpr (std::is_same_v<V, math::Rational>) {
385 return ColumnKind::Decimal;
386 } else if constexpr (units::isQuantity<V>) {
387 return ColumnKind::Quantity;
388 } else if constexpr (std::is_same_v<V, time::DateTime> || std::is_same_v<V, time::Timestamp>) {
389 return ColumnKind::DateTime;
390 } else if constexpr (MemberWrapper<V>) {
391 return kindOf<WrappedMember<V>>();
392 } else {
393 return ColumnKind::Text;
394 }
395}

397/// @brief An integer member as a cell.
398/// @tparam T The integer type.
399/// @param value The member.
400/// @return The cell: an `int64`, or text that an integer column reads as
401/// invalid when no `int64` holds the value (rather than a wrapped
402/// negative number).
403template <typename T>
404[[nodiscard]] Cell integerCell(T value) {
405 if constexpr (std::is_unsigned_v<T> && sizeof(T) >= sizeof(std::int64_t)) {
406 if (value > static_cast<T>(std::numeric_limits<std::int64_t>::max())) {
407 return Cell{std::to_string(value)};
408 }
409 }
410 return Cell{static_cast<std::int64_t>(value)};
411}
412
413/// @brief A member as the text glaze writes for it: an enum by its glaze name,
414/// a JSON string decoded, `null` as an empty cell.
415/// @tparam T The member type.
416/// @param value The member.
417/// @return The cell.
418template <typename T>
419[[nodiscard]] Cell writtenCell(T const& value) {
420 std::string text;
421 if (glz::write_json(value, text) || text == "null") {
422 return Cell{};
423 }
424 if (text.size() >= 2 && text.front() == '"' && text.back() == '"') {
425 std::string decoded;
426 if (!glz::read_json(decoded, text)) {
427 return Cell{std::move(decoded)};
428 }
429 }
430 return Cell{std::move(text)};
431}
432
433/// @brief A member value as a cell.
434/// @tparam T The member type.
435/// @param value The member.
436/// @return The cell.
437template <typename T>
438[[nodiscard]] Cell toCell(T const& value) {
439 using V = std::remove_cvref_t<T>;
440 if constexpr (IsOptional<V>::value) {
441 return value ? toCell(*value) : Cell{};
442 } else if constexpr (std::is_same_v<V, bool> || std::is_same_v<V, std::string> ||
443 std::is_same_v<V, math::Rational>) {
444 return Cell{value};
445 } else if constexpr (std::integral<V>) {
446 return integerCell(value);
447 } else if constexpr (std::floating_point<V>) {
448 return Cell{static_cast<double>(value)};
449 } else if constexpr (units::isQuantity<V>) {
450 return value.value() ? Cell{*value.value()} : Cell{};
451 } else if constexpr (std::is_same_v<V, time::DateTime>) {
452 return Cell{std::chrono::floor<std::chrono::seconds>(value.value).time_since_epoch().count()};
453 } else if constexpr (std::is_same_v<V, time::Timestamp>) {
454 return value.value ? toCell(*value.value) : Cell{};
455 } else if constexpr (MemberWrapper<V>) {
456 return toCell(value.*(glz::meta<V>::value));
457 } else {
458 return writtenCell(value);
459 }
460}

462/// @brief Reads one column's cell from a row.
463/// @tparam Row The row type.
464template <typename Row>
465using CellReader = Cell (*)(Row const&);
466
467/// @brief The value glaze writes under `glz::reflect<Row>::keys[I]`: the
468/// I-th declared member of a row reflected by itself, the I-th entry of
469/// a row with a `glz::meta`, which may omit, reorder or compute members.
470/// @tparam Row The row type.
471/// @tparam I The key's index.
472/// @param row The row.
473/// @return The value.
474template <typename Row, std::size_t I>
475[[nodiscard]] decltype(auto) memberAt(Row const& row) {
476 if constexpr (glz::reflectable<Row>) {
477 auto tie = glz::to_tie(row);
478 return glz::get_member(row, glz::get<I>(tie));
479 } else {
480 return glz::get_member(row, glz::get<I>(glz::reflect<Row>::values));
481 }
482}
483
484/// @brief One cell reader per key glaze writes, in key order.
485/// @tparam Row The row type.
486/// @return The readers.
487template <typename Row>
488[[nodiscard]] std::vector<CellReader<Row>> cellReaders() {
489 constexpr auto count = glz::reflect<Row>::size;
490 return [&]<std::size_t... I>(std::index_sequence<I...>) {
491 return std::vector<CellReader<Row>>{+[](Row const& row) -> Cell { return toCell(memberAt<Row, I>(row)); }...};
492 }(std::make_index_sequence<count>{});
493}
494
495/// @brief One column per key glaze writes, its kind from the value's type.
496/// @tparam Row The row type.
497/// @return The columns.
498template <typename Row>
499[[nodiscard]] std::vector<ColumnInfo> reflectedColumns() {
500 constexpr auto count = glz::reflect<Row>::size;
501 std::vector<ColumnInfo> out;
502 [&]<std::size_t... I>(std::index_sequence<I...>) {
503 (out.push_back(ColumnInfo{.id = std::string{glz::reflect<Row>::keys[I]},
504 .kind = kindOf<decltype(memberAt<Row, I>(std::declval<Row const&>()))>(),
505 .comparator = {},
506 .format = {}}),
507 ...);
508 }(std::make_index_sequence<count>{});
509 return out;
510}

514/// @brief A key projection: the `RowId` of a typed row.
515/// @tparam Row The row type.
516template <typename Row>
517using RowKey = std::function<RowId(Row const&)>;
518
519/// @brief A snapshot over a shared, immutable vector of typed rows.
520/// @tparam Row A glaze-reflectable aggregate.
521template <typename Row>
522class RowsSnapshot final : public RowSnapshot {
523public:
524 /// @brief Wraps @p rows.
525 /// @param rows The rows; shared, never modified.
526 /// @param key The key projection.
527 /// @param readers One cell reader per column.
528 RowsSnapshot(std::shared_ptr<std::vector<Row> const> rows, RowKey<Row> key,
529 std::shared_ptr<std::vector<detail::CellReader<Row>> const> readers)
530 : _rows{std::move(rows)}, _key{std::move(key)}, _readers{std::move(readers)} {}
531
532 /// @brief The number of rows.
533 /// @return Row count.
534 [[nodiscard]] std::size_t rowCount() const override { return _rows->size(); }
535
536 /// @brief A row's key.
537 /// @param row Row index.
538 /// @return The key.
539 [[nodiscard]] RowId rowId(std::size_t row) const override { return _key((*_rows)[row]); }
540
541 /// @brief One cell.
542 /// @param row Row index.
543 /// @param column Column index.
544 /// @return The cell.
545 [[nodiscard]] Cell cell(std::size_t row, std::size_t column) const override {
546 return (*_readers)[column]((*_rows)[row]);
547 }
548
549 /// @brief Hands rows `[first, last)` of @p column to @p sink in runs.
550 /// @param column Column index.
551 /// @param sink Receives the cells.
552 /// @param first First row.
553 /// @param last One past the last row.
554 void readColumn(std::size_t column, ColumnSink& sink, std::size_t first, std::size_t last) const override {
555 constexpr std::size_t kRun = 256;
556 std::vector<Cell> run;
557 run.reserve(kRun);
558 auto const reader = (*_readers)[column];
559 while (first < last) {
560 auto const end = std::min(first + kRun, last);
561 run.clear();
562 for (auto row = first; row < end; ++row) {
563 run.push_back(reader((*_rows)[row]));
564 }
565 sink.cells(first, run);
566 first = end;
567 }
568 }
569
570private:
571 std::shared_ptr<std::vector<Row> const> _rows;
572 RowKey<Row> _key;
573 std::shared_ptr<std::vector<detail::CellReader<Row>> const> _readers;
574};

576/// @brief A `DataSource` over typed rows: one column per key glaze writes,
577/// its kind inferred from the value's type.
578///
579/// `bool` is `Bool`; other integers `Integer` (an unsigned value above
580/// `INT64_MAX` is an invalid cell); floating point `Number`; `Rational`
581/// `Decimal`; a `Quantity` `Quantity`; `DateTime` and `Timestamp`
582/// `DateTime`; a type glaze writes as one of its members (`Ranged`,
583/// `Choice`, `Tagged`) that member's kind; everything else `Text` (an enum by
584/// its glaze name). A `std::optional` member's empty state, or anything glaze
585/// writes as `null`, is an empty cell. A row with a `glz::meta` has the
586/// columns its meta names. `setRows` replaces the rows and notifies `Reset`.
587/// @tparam Row A glaze-reflectable aggregate.
588template <typename Row>
589class RowsSource final : public DataSource {
590public:
591 /// @brief Builds the source.
592 /// @param rows The rows.
593 /// @param key The key projection.
594 /// @param kinds Column kinds that replace the inferred ones, by column
595 /// id (for example `Date` for an `int64` of epoch days).
596 /// @param comparators Comparator names by column id, from
597 /// `Services::comparators`; a column named here is `Custom`.
598 RowsSource(std::shared_ptr<std::vector<Row> const> rows, RowKey<Row> key,
599 std::map<std::string, ColumnKind, std::less<>> const& kinds = {},
600 std::map<std::string, std::string, std::less<>> const& comparators = {})
601 : _columns{detail::reflectedColumns<Row>()},
602 _readers{std::make_shared<std::vector<detail::CellReader<Row>> const>(detail::cellReaders<Row>())},
603 _key{std::move(key)},
604 _snapshot{std::make_shared<RowsSnapshot<Row>>(std::move(rows), _key, _readers)} {
605 for (auto& column : _columns) {

619
620 /// @brief The current rows; a pointer copy.
621 /// @return The snapshot.
622 [[nodiscard]] std::shared_ptr<RowSnapshot const> snapshot() const override { return _snapshot; }
623
624 /// @brief Registers @p listener.

627
628 /// @brief Removes @p listener; safe from inside a notification.
629 /// @param listener The listener.

632 /// @brief Replaces the rows and notifies `Reset`.
633 /// @param rows The new rows.

637 }

639private:

642 RowKey<Row> _key;
643 std::shared_ptr<RowsSnapshot<Row> const> _snapshot;

647/// @brief Runs @p query over typed rows and returns the page of rows: what a
648/// server's list action returns.
649/// @tparam Row A glaze-reflectable aggregate.
650/// @param rows The server's rows; read during the call only.
651/// @param key The key projection.
652/// @param query The request body's table query.
653/// @param options Limits, services and column access.
654/// @param kinds Column kinds that replace the inferred ones.
655/// @param comparators Comparator names by column id; a column named here is `Custom`.
656/// @return The page, or the typed error the client shows.
657template <typename Row>
658[[nodiscard]] std::expected<Page<Row>, TableError> apply(
659 std::vector<Row> const& rows, RowKey<Row> key, TableQuery const& query, ApplyOptions const& options = {},
660 std::map<std::string, ColumnKind, std::less<>> const& kinds = {},
661 std::map<std::string, std::string, std::less<>> const& comparators = {}) {
662 // Borrowed for the call: an aliasing pointer that owns nothing.
663 std::shared_ptr<std::vector<Row> const> const borrowed{std::shared_ptr<void>{}, &rows};
664 RowsSource<Row> const source{borrowed, std::move(key), kinds, comparators};
665 auto page = apply(source, query, options);
666 if (!page) {
667 return std::unexpected(std::move(page.error()));
668 }
669 Page<Row> out;
670 out.total = page->total;
671 out.offset = page->offset;
672 out.rows.reserve(page->rows.size());
673 for (auto const row : page->rows) {
674 out.rows.push_back(rows[row]);
675 }
676 return out;
677}

681/// @brief A request a `PageWindow` asks the client to send, tagged with the
682/// query it belongs to.
683struct PageFetch {
684 /// @brief The rows to request.
685 PageRequest request;
686 /// @brief The window's query when the fetch was made (`PageWindow::query`).
687 std::uint64_t query = 0;
688
689 /// @brief Member-wise equality.
690 /// @param other The fetch to compare with.
691 /// @return `true` when request and query match.
692 [[nodiscard]] bool operator==(PageFetch const& other) const = default;
693};

695/// @brief The pages a `server`-mode table holds: those around the visible
696/// rows, up to a bound, plus the request state and the last error.
697///
698/// The client calls `visible` as the view scrolls and sends the request of
699/// each fetch it returns, then hands the fetch back with its reply to
700/// `accept` (or with its error to `fail`). A change of sort or filter is a
701/// new query: call `reset`. A reply to a fetch made before the last `reset`
702/// is ignored, so a late page of the old query never shows under the new one.
703///
704/// A server may return fewer rows than a fetch asked for (`validate` clamps
705/// the limit to its `maxLimit`): the window then fetches the rest of the page
706/// from where the reply ended. A page is complete when it is full, reaches the
707/// known total, or a reply to it is empty.
708/// @tparam Row The row type.
709template <typename Row>
710class PageWindow {
711public:
712 /// @brief An empty window.
713 /// @param pageSize Rows per page (default 200).
714 /// @param maxPages Pages held at most; the farthest from the visible rows go first.
715 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters) - page size then page count, both documented above
716 explicit PageWindow(std::int64_t pageSize = 200, std::size_t maxPages = 8)
717 : _pageSize{std::max<std::int64_t>(pageSize, 1)}, _maxPages{std::max<std::size_t>(maxPages, 1)} {}
718
719 /// @brief Declares the visible rows; returns the fetches to send.
720 ///
721 /// Covers the visible rows and one page either side, skipping complete
722 /// pages, pages in flight, and pages past a known total. A page held in
723 /// part is fetched from where it ends. The returned pages are marked in
724 /// flight.
725 /// @param first First visible view row.
726 /// @param count Visible rows.
727 /// @return The fetches to send.
728 [[nodiscard]] std::vector<PageFetch> visible(std::int64_t first, std::int64_t count) {
729 _visibleFirst = std::max<std::int64_t>(first, 0);
730 _visibleCount = std::max<std::int64_t>(count, 0);
731 auto const firstPage = std::max<std::int64_t>((_visibleFirst / _pageSize) - 1, 0);
732 auto const lastPage = ((_visibleFirst + std::max<std::int64_t>(_visibleCount, 1) - 1) / _pageSize) + 1;
733 std::vector<PageFetch> out;
734 for (auto page = firstPage; page <= lastPage; ++page) {
735 if (_total && page * _pageSize >= *_total) {
736 break;
737 }
738 if (_complete.contains(page) || std::ranges::find(_inFlight, page) != _inFlight.end()) {
739 continue;
740 }
741 auto const held = _pages.find(page);
742 auto const have = held == _pages.end() ? std::int64_t{0} : static_cast<std::int64_t>(held->second.size());
743 _inFlight.push_back(page);
744 out.push_back(
745 PageFetch{.request = PageRequest{.offset = (page * _pageSize) + have, .limit = _pageSize - have},
746 .query = _query});
747 }
748 return out;
749 }
750
751 /// @brief Stores a reply.
752 /// @param fetch The fetch the reply answers, as `visible` returned it.
753 /// @param page The page the server returned.
754 /// @return `false` when the reply was ignored: its fetch belongs to an
755 /// earlier query, or is not awaited.
756 bool accept(PageFetch const& fetch, Page<Row> page) {
757 if (fetch.query != _query) {
758 return false;
759 }
760 auto const index = fetch.request.offset / _pageSize;
761 auto const flight = std::ranges::find(_inFlight, index);
762 if (flight == _inFlight.end()) {
763 return false;
764 }
765 _inFlight.erase(flight);
766 if (page.total) {
767 _total = page.total;
768 }
769 _error.reset();
770 auto& rows = _pages[index];
771 bool const replyEmpty = page.rows.empty();
772 if (fetch.request.offset == (index * _pageSize) + static_cast<std::int64_t>(rows.size())) {
773 auto const room = static_cast<std::size_t>(_pageSize) - rows.size();
774 auto const take = std::min(room, page.rows.size());
775 rows.insert(rows.end(), std::make_move_iterator(page.rows.begin()),
776 std::make_move_iterator(page.rows.begin() + static_cast<std::ptrdiff_t>(take)));
777 }
778 auto const end = (index * _pageSize) + static_cast<std::int64_t>(rows.size());
779 if (std::cmp_greater_equal(rows.size(), _pageSize) || (_total && end >= *_total) || replyEmpty) {
780 _complete.insert(index);
781 }
782 evict();
783 return true;
784 }
785
786 /// @brief Records a failed request; the error is shown until a reply arrives.
787 /// @param fetch The fetch that failed, as `visible` returned it.
788 /// @param error The server's error.
789 /// @return `false` when the failure was ignored: its fetch belongs to an earlier query.
790 bool fail(PageFetch const& fetch, TableError error) {
791 if (fetch.query != _query) {
792 return false;
793 }
794 std::erase(_inFlight, fetch.request.offset / _pageSize);
795 _error = std::move(error);
796 return true;
797 }
798
799 /// @brief Drops every page, request and error and starts a new query: the
800 /// sort or filter changed.
801 void reset() {
802 ++_query;
803 _pages.clear();
804 _complete.clear();
805 _inFlight.clear();
806 _total.reset();
807 _error.reset();
808 }
809
810 /// @brief The current query, which `reset` advances.
811 /// @return The query's number.
812 [[nodiscard]] std::uint64_t query() const noexcept { return _query; }
813
814 /// @brief The row at a view row, when it is held.
815 /// @param index View row.
816 /// @return The row, or null.
817 [[nodiscard]] Row const* row(std::int64_t index) const {
818 if (index < 0) {
819 return nullptr;
820 }
821 auto const found = _pages.find(index / _pageSize);
822 if (found == _pages.end()) {
823 return nullptr;
824 }
825 auto const offset = static_cast<std::size_t>(index % _pageSize);
826 return offset < found->second.size() ? &found->second[offset] : nullptr;
827 }
828
829 /// @brief The view's total rows, when a reply of this query reported it.
830 /// @return The total.
831 [[nodiscard]] std::optional<std::int64_t> total() const noexcept { return _total; }
832
833 /// @brief The last error, until a reply clears it.
834 /// @return The error, or nothing.
835 [[nodiscard]] std::optional<TableError> const& error() const noexcept { return _error; }
836
837 /// @brief Pages held, complete or in part.
838 /// @return Page count.
839 [[nodiscard]] std::size_t pagesHeld() const noexcept { return _pages.size(); }
840
841 /// @brief Whether any request is in flight.
842 /// @return `true` while a page is awaited.
843 [[nodiscard]] bool pending() const noexcept { return !_inFlight.empty(); }
844
845private:
846 void evict() {
847 auto const center = (_visibleFirst + (_visibleCount / 2)) / _pageSize;
848 while (_pages.size() > _maxPages) {
849 auto farthest = _pages.begin();
850 for (auto at = _pages.begin(); at != _pages.end(); ++at) {
851 if (std::abs(at->first - center) > std::abs(farthest->first - center)) {
852 farthest = at;
853 }
854 }
855 _complete.erase(farthest->first);
856 _pages.erase(farthest);
857 }
858 }
859
860 std::int64_t _pageSize;
861 std::size_t _maxPages;
862 std::uint64_t _query = 0;
863 std::map<std::int64_t, std::vector<Row>> _pages;
864 std::set<std::int64_t> _complete;
865 std::vector<std::int64_t> _inFlight;
866 std::optional<std::int64_t> _total;
867 std::optional<TableError> _error;
868 std::int64_t _visibleFirst = 0;
869 std::int64_t _visibleCount = 0;
870};
