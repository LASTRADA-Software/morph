25/// @brief Why a store refused a call.
26enum class FileError : std::uint8_t {
27 NotFound, ///< No such upload or file for this principal; a caller cannot tell the two apart.
28 TooLarge, ///< The file is larger than the store accepts.
29 OverQuota, ///< The principal's files would exceed what the store keeps for one principal.
30 BadOffset, ///< A chunk does not start where the bytes received so far end.
31 Incomplete, ///< An upload ended before every byte it announced arrived.
32};
33
34/// @brief The text of a `FileError`, as a `ui-file` refusal carries it.
35/// @param error The error.
36/// @return Its text.
37[[nodiscard]] constexpr std::string_view fileErrorText(FileError error) noexcept {
38 switch (error) {
39 case FileError::NotFound:
40 return "not found";
41 case FileError::TooLarge:
42 return "too large";
43 case FileError::OverQuota:
44 return "over quota";
45 case FileError::BadOffset:
46 return "bad offset";
47 case FileError::Incomplete:
48 default:
49 return "incomplete";
50 }
51}

72/// @brief Where uploaded and issued files live. Every call names the principal it is made for, and an upload id or a
73/// file reference works only for the principal it was made for.
74///
75/// A reference is a capability: whoever presents it, as the principal it belongs to, may read the file. It is
76/// therefore unguessable (`newFileRef`), and a store answers a reference of another principal exactly as one that

129/// @brief A new unguessable id: @p prefix, then 32 lower-case hex digits read straight from `std::random_device`.
130///
131/// `std::random_device` is read directly, never used as a seed: libstdc++, libc++ and MSVC back it with the
132/// operating system's entropy source, which is what makes the id a capability.
133/// @param prefix `file:` for a reference, `upload:` for an upload id.
134/// @return The id.
135[[nodiscard]] inline std::string newFileRef(std::string_view prefix = "file:") {
136 static constexpr std::string_view kHex = "0123456789abcdef";
137 std::random_device entropy;
138 std::string text{prefix};
139 for (int word = 0; word < 4; ++word) {
140 std::uint32_t value = entropy(); // random_device yields 32 bits of entropy per call
141 for (int nibble = 0; nibble < 8; ++nibble) {
142 text += kHex.at(value & 0xFU);
143 value >>= 4U;
144 }
145 }
146 return text;
147}

149/// @brief How much a `MemoryFileStore` keeps.
150struct MemoryFileStoreOptions {
151 /// @brief The largest file it accepts.
152 std::size_t maxFileBytes = std::size_t{64} * 1024 * 1024;
153 /// @brief The most bytes it keeps for one principal, uploads in progress included.
154 std::size_t maxBytesPerPrincipal = std::size_t{256} * 1024 * 1024;
155 /// @brief How long an upload may go without a chunk before it is dropped.
156 std::chrono::seconds uploadTtl{3600};
157 /// @brief The clock expiry reads; the steady clock when empty.
158 std::function<std::chrono::steady_clock::time_point()> clock;
159};

161/// @brief An `IFileStore` in memory: bounded per file and per principal; an upload that stalls expires.
162///
163/// Thread-safe: the UI service calls it on the server strand while an application issues references from its models'.
164class MemoryFileStore final : public IFileStore {
165public:
166 /// @brief A store with @p options.
167 /// @param options The bounds and the clock.
168 explicit MemoryFileStore(MemoryFileStoreOptions options = {}) : _options{std::move(options)} {}
169
170 /// @brief Starts an upload.
171 /// @param principal The uploader.
172 /// @param info The file's name, type and size.
173 /// @return The upload id, or `TooLarge` or `OverQuota`.
174 [[nodiscard]] std::expected<std::string, FileError> begin(std::string const& principal, FileInfo info) override {
175 std::scoped_lock const lock{_mutex};
176 expire();
177 if (info.size > _options.maxFileBytes) {
178 return std::unexpected(FileError::TooLarge);
179 }
180 if (usage(principal) + info.size > _options.maxBytesPerPrincipal) {
181 return std::unexpected(FileError::OverQuota);
182 }
183 std::string uploadId = newFileRef("upload:");
184 _uploads.emplace(uploadId,
185 Upload{.principal = principal, .info = std::move(info), .bytes = {}, .touched = now()});
186 return uploadId;
187 }
188
189 /// @brief Appends a chunk.
190 /// @param principal The uploader.
191 /// @param upload The upload id.
192 /// @param offset The bytes received so far.
193 /// @param bytes The chunk.
194 /// @return The bytes received so far, or `NotFound`, `BadOffset` or `TooLarge` (past the announced size).
195 [[nodiscard]] std::expected<std::size_t, FileError> append(std::string const& principal, std::string const& upload,
196 std::size_t offset, std::string_view bytes) override {
197 std::scoped_lock const lock{_mutex};
198 expire();
199 auto const found = _uploads.find(upload);
200 if (found == _uploads.end() || found->second.principal != principal) {
201 return std::unexpected(FileError::NotFound);
202 }
203 Upload& open = found->second;
204 if (offset != open.bytes.size()) {
205 return std::unexpected(FileError::BadOffset);
206 }
207 if (open.bytes.size() + bytes.size() > open.info.size) {
208 return std::unexpected(FileError::TooLarge);
209 }
210 open.bytes.append(bytes);
211 open.touched = now();
212 return open.bytes.size();
213 }
214
215 /// @brief Ends an upload and files it.
216 /// @param principal The uploader.
217 /// @param upload The upload id.
218 /// @return The reference, or `NotFound` or `Incomplete`.
219 [[nodiscard]] std::expected<std::string, FileError> finish(std::string const& principal,
220 std::string const& upload) override {
221 std::scoped_lock const lock{_mutex};
222 expire();
223 auto const found = _uploads.find(upload);
224 if (found == _uploads.end() || found->second.principal != principal) {
225 return std::unexpected(FileError::NotFound);
226 }
227 if (found->second.bytes.size() != found->second.info.size) {
228 return std::unexpected(FileError::Incomplete);
229 }
230 Upload done = std::move(found->second);
231 _uploads.erase(found);
232 std::string ref = newFileRef();
233 _files.emplace(ref, Stored{.principal = std::move(done.principal),
234 .info = std::move(done.info),
235 .bytes = std::move(done.bytes)});
236 return ref;
237 }
238
239 /// @brief Files bytes for a principal.
240 /// @param principal The principal.
241 /// @param info The name and type.
242 /// @param bytes The file.
243 /// @return The reference, or `TooLarge` or `OverQuota`.
244 [[nodiscard]] std::expected<std::string, FileError> issue(std::string const& principal, FileInfo info,
245 std::string bytes) override {
246 std::scoped_lock const lock{_mutex};
247 expire();
248 if (bytes.size() > _options.maxFileBytes) {
249 return std::unexpected(FileError::TooLarge);
250 }
251 if (usage(principal) + bytes.size() > _options.maxBytesPerPrincipal) {
252 return std::unexpected(FileError::OverQuota);
253 }
254 info.size = bytes.size();
255 std::string ref = newFileRef();
256 _files.emplace(ref, Stored{.principal = principal, .info = std::move(info), .bytes = std::move(bytes)});
257 return ref;
258 }
259
260 /// @brief Reads a slice.
261 /// @param principal The reader.
262 /// @param ref The reference.
263 /// @param offset The first byte.
264 /// @param length At most this many bytes.
265 /// @return The slice (empty past the end), or `NotFound`.
266 [[nodiscard]] std::expected<FileSlice, FileError> read(std::string const& principal, std::string const& ref,
267 std::size_t offset, std::size_t length) override {
268 std::scoped_lock const lock{_mutex};
269 auto const found = _files.find(ref);
270 if (found == _files.end() || found->second.principal != principal) {
271 return std::unexpected(FileError::NotFound);
272 }
273 Stored const& file = found->second;
274 std::size_t const start = std::min(offset, file.bytes.size());
275 return FileSlice{.bytes = file.bytes.substr(start, length), .info = file.info};
276 }
277
278private:
279 struct Upload {
280 std::string principal;
281 FileInfo info;
282 std::string bytes;
283 std::chrono::steady_clock::time_point touched;
284 };
285
286 struct Stored {
287 std::string principal;
288 FileInfo info;
289 std::string bytes;
290 };
291
292 [[nodiscard]] std::chrono::steady_clock::time_point now() const {
293 return _options.clock ? _options.clock() : std::chrono::steady_clock::now();
294 }
295
296 /// Drops the uploads that went without a chunk for longer than the TTL. Under the lock.
297 void expire() {
298 auto const cutoff = now() - _options.uploadTtl;
299 std::erase_if(_uploads, [cutoff](auto const& entry) { return entry.second.touched < cutoff; });
300 }
301
302 /// The bytes a principal holds: its files, and the announced sizes of its uploads. Under the lock.
303 [[nodiscard]] std::size_t usage(std::string const& principal) const {
304 std::size_t total = 0;
305 for (auto const& entry : _files) {
306 if (entry.second.principal == principal) {
307 total += entry.second.bytes.size();
308 }
309 }
310 for (auto const& entry : _uploads) {
311 if (entry.second.principal == principal) {
312 total += entry.second.info.size;
313 }
314 }
315 return total;
316 }
317
318 MemoryFileStoreOptions _options;
319 mutable std::mutex _mutex;
320 std::map<std::string, Upload, std::less<>> _uploads;
321 std::map<std::string, Stored, std::less<>> _files;
322};
