32/// @brief Where a client keeps verified files between runs, by content hash, and the digest of each application's
33/// last verified bundle.
34///
35/// A store is not trusted: `VerifiedCache` hashes what it reads back. A client calls it from whichever thread its
36/// exchange calls back on, and runs of several `open`s may overlap, so an implementation is safe to call from any
37/// thread.
38class IBundleStore {
39public:
40 IBundleStore() = default;
41 virtual ~IBundleStore() = default;
42 IBundleStore(IBundleStore const&) = delete;
43 IBundleStore& operator=(IBundleStore const&) = delete;
44 IBundleStore(IBundleStore&&) = delete;
45 IBundleStore& operator=(IBundleStore&&) = delete;
46
47 /// @brief Reads a file.
48 /// @param hash The file's content hash.
49 /// @return Its bytes, or nothing when the store has none.
50 [[nodiscard]] virtual std::optional<std::string> get(std::string_view hash) const = 0;
51
52 /// @brief Keeps a file.
53 /// @param hash The file's content hash.
54 /// @param bytes The file.
55 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the key, then the value.
56 virtual void put(std::string const& hash, std::string bytes) = 0;
57
58 /// @brief Drops a file.
59 /// @param hash The file's content hash.
60 virtual void remove(std::string_view hash) = 0;
61
62 /// @brief The digest of an application's last verified bundle.
63 /// @param application The application's id.
64 /// @return The digest, or nothing.
65 [[nodiscard]] virtual std::optional<std::string> lastDigest(std::string_view application) const = 0;
66
67 /// @brief Records an application's last verified bundle.
68 /// @param application The application's id.
69 /// @param digest The manifest's content hash.
70 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the key, then the value.
71 virtual void setLastDigest(std::string const& application, std::string digest) = 0;
72};

74/// @brief An `IBundleStore` in memory: it lasts as long as the process. Safe to call from any thread.
75class MemoryBundleStore final : public IBundleStore {
76public:
77 /// @brief Reads a file.
78 /// @param hash The file's content hash.
79 /// @return Its bytes, or nothing.
80 [[nodiscard]] std::optional<std::string> get(std::string_view hash) const override {
81 std::scoped_lock const lock{_mutex};
82 auto const found = _files.find(hash);
83 if (found == _files.end()) {
84 return std::nullopt;
85 }
86 return found->second;
87 }
88
89 /// @brief Keeps a file.
90 /// @param hash The file's content hash.
91 /// @param bytes The file.
92 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the key, then the value.
93 void put(std::string const& hash, std::string bytes) override {
94 std::scoped_lock const lock{_mutex};
95 _files.insert_or_assign(hash, std::move(bytes));
96 }
97
98 /// @brief Drops a file.
99 /// @param hash The file's content hash.
100 void remove(std::string_view hash) override {
101 std::scoped_lock const lock{_mutex};
102 if (auto const found = _files.find(hash); found != _files.end()) {
103 _files.erase(found);
104 }
105 }
106
107 /// @brief The digest of an application's last verified bundle.
108 /// @param application The application's id.
109 /// @return The digest, or nothing.
110 [[nodiscard]] std::optional<std::string> lastDigest(std::string_view application) const override {
111 std::scoped_lock const lock{_mutex};
112 auto const found = _digests.find(application);
113 if (found == _digests.end()) {
114 return std::nullopt;
115 }
116 return found->second;
117 }
118
119 /// @brief Records an application's last verified bundle.
120 /// @param application The application's id.
121 /// @param digest The manifest's content hash.
122 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the key, then the value.
123 void setLastDigest(std::string const& application, std::string digest) override {
124 std::scoped_lock const lock{_mutex};
125 _digests.insert_or_assign(application, std::move(digest));
126 }
127
128 /// @brief How many files the store holds.
129 /// @return The count.
130 [[nodiscard]] std::size_t size() const {
131 std::scoped_lock const lock{_mutex};
132 return _files.size();
133 }
134
135private:
136 mutable std::mutex _mutex;
137 std::map<std::string, std::string, std::less<>> _files;
138 std::map<std::string, std::string, std::less<>> _digests;
139};

141/// @brief A content-addressed cache that hands out only bytes that hash to the hash they are asked for.
142class VerifiedCache {
143public:
144 /// @brief A cache over @p store.
145 /// @param store The store; shared with the caller.
146 /// @throws std::invalid_argument if @p store is null.
147 explicit VerifiedCache(std::shared_ptr<IBundleStore> store) : _store{std::move(store)} {
148 if (!_store) {
149 throw std::invalid_argument("VerifiedCache: the store is null");
150 }
151 }
152
153 /// @brief Reads a file, verified. A stored file that does not hash to @p hash is dropped and reads as a miss.
154 /// @param hash The content hash.
155 /// @return The bytes, or nothing.
156 [[nodiscard]] std::optional<std::string> get(std::string const& hash) {
157 auto bytes = _store->get(hash);
158 if (bytes && contentHash(*bytes) != hash) {
159 _store->remove(hash);
160 return std::nullopt;
161 }
162 return bytes;
163 }
164
165 /// @brief Keeps a file under its content hash.
166 /// @param bytes The bytes.
167 /// @return The hash they are kept under.
168 std::string put(std::string bytes) {
169 std::string hash = contentHash(bytes);
170 _store->put(hash, std::move(bytes));
171 return hash;
172 }
173
174 /// @brief The store.
175 /// @return It.
176 [[nodiscard]] IBundleStore& store() noexcept { return *_store; }
177
178private:
179 std::shared_ptr<IBundleStore> _store;
180};

182/// @brief A bundle the client verified: what the shell mounts, and what it was verified against.
183struct LoadedBundle {
184 /// @brief The application's id, the manifest digest, and the documents (the screens it may open and the app
185 /// document) by id.
186 Bundle bundle;
187 /// @brief The `ui-hello` reply; empty when the bundle was opened offline.
188 HelloReply hello;
189 /// @brief The manifest.
190 Manifest manifest;
191 /// @brief The catalog for the principal; empty when the bundle was opened offline.
192 CatalogReply catalog;
193 /// @brief Every verified file the bundle holds, by content hash: documents, themes, message catalogues and,
194 /// for a client that loads them, components.
195 std::map<std::string, std::string, std::less<>> files;
196};

198/// @brief One reply, or why the transport could not deliver it.
199using ExchangeReply = std::expected<::morph::wire::Envelope, std::string>;

201/// @brief Sends one request and calls its callback once with the reply. The transport assigns the request's `callId`
202/// and matches the reply to it; a failure to deliver either is the error.
203using Exchange = std::function<void(::morph::wire::Envelope request, std::function<void(ExchangeReply reply)> done)>;

205/// @brief Opens an application's bundle through the UI service, verifying everything it fetches.
206///
207/// `open` runs `ui-hello`, refuses a bundle that needs a vocabulary the client does not speak, takes the manifest from
208/// the cache or fetches it by the digest and verifies it, runs `ui-catalog`, and fetches the files the principal may
209/// use that the cache does not hold, all at once, verifying each. A file that does not hash to its manifest entry,
210/// or a manifest that does not hash to the digest, refuses the whole bundle (`Unverified`). An unchanged manifest
211/// fetches nothing.
212class UiClient {
213public:
214 /// @brief A client.
215 /// @param exchange The transport.
216 /// @param client What the client speaks and carries, sent with `ui-hello`.
217 /// @param store Where verified files are kept.
218 /// @throws std::invalid_argument if @p exchange is empty or @p store is null.
219 UiClient(Exchange exchange, HelloRequest client, std::shared_ptr<IBundleStore> store)
220 : _shared{std::make_shared<Shared>(std::move(exchange), std::move(client), VerifiedCache{std::move(store)})} {
221 if (!_shared->exchange) {
222 throw std::invalid_argument("UiClient: the exchange is empty");
223 }
224 }
225
226 /// @brief What `open` delivers.
227 using Done = std::function<void(std::expected<LoadedBundle, ConnectError>)>;
228
229 /// @brief Opens the bundle. @p done is called once, from whichever thread the exchange calls back on; a
230 /// synchronous exchange calls it before `open` returns.
231 /// @param session The session every request carries: the principal's token, or none before sign-in.
232 /// @param done Called with the bundle, or why there is none.
233 void open(::morph::session::Context session, Done done) const {
234 auto const run = std::make_shared<Run>(_shared, std::move(session), std::move(done));
235 sendHello(run);
236 }
237
238 /// @brief Rebuilds the last bundle this store verified for @p application, without a connection. Files the
239 /// store no longer holds, or that no longer verify, are left out; the catalog is empty.
240 /// @param application The application's id.
241 /// @return The bundle, or `Unreachable` when no verified manifest is cached.
242 [[nodiscard]] std::expected<LoadedBundle, ConnectError> openOffline(std::string_view application) const {
243 auto const digest = _shared->cache.store().lastDigest(application);
244 if (!digest) {
245 return std::unexpected(
246 ConnectError{.kind = ConnectError::Kind::Unreachable,
247 .message = "no verified bundle of `" + std::string{application} + "` is cached"});
248 }
249 auto const text = _shared->cache.get(*digest);
250 if (!text) {
251 return std::unexpected(ConnectError{
252 .kind = ConnectError::Kind::Unreachable,
253 .message = "the cached manifest of `" + std::string{application} + "` is gone or does not verify"});
254 }
255 auto manifest = parseManifest(*text);
256 if (!manifest) {
257 return std::unexpected(ConnectError{.kind = ConnectError::Kind::Unsupported, .message = manifest.error()});
258 }
259 LoadedBundle loaded;
260 loaded.manifest = std::move(*manifest);
261 for (ManifestFile const& file : loaded.manifest.files) {
262 if (!usable(_shared->client, file.kind)) {
263 continue;
264 }
265 if (auto bytes = _shared->cache.get(file.hash)) {
266 loaded.files.emplace(file.hash, std::move(*bytes));
267 }
268 }
269 assemble(loaded, *digest);
270 return loaded;
271 }

284 /// @brief One `open`: its replies so far, and the files still in flight.
285 ///
286 /// The steps up to the catalog run one at a time. The fetches after it are in flight together, and a transport
287 /// may answer them on different threads, so what they change is guarded by `mutex`.
288 struct Run {
289 Run(std::shared_ptr<Shared> sharedState, ::morph::session::Context sessionContext, Done doneFn)
290 : shared{std::move(sharedState)}, session{std::move(sessionContext)}, done{std::move(doneFn)} {}
291
292 std::shared_ptr<Shared> shared;
293 ::morph::session::Context session;
294 Done done;
295 LoadedBundle loaded;
296 std::mutex mutex;
297 std::size_t pending = 0;
298 std::optional<ConnectError> failure;
299 };

301 /// @brief A file a run expects: its hash, and how a refusal names it.
302 struct Expected {
303 std::string hash;
304 std::string what;
305 };

307 /// @brief Whether a client fetches files of @p kind: every kind it knows, components only when it loads them.
308 [[nodiscard]] static bool usable(HelloRequest const& client, std::string_view kind) {
309 if (kind == kComponentFile) {
310 return client.components;
311 }
312 return kind == kScreenFile || kind == kAppFile || kind == kThemeFile || kind == kMessagesFile;
313 }
314
315 /// @brief Fills the bundle the shell reads from the verified files.
316 static void assemble(LoadedBundle& loaded, std::string const& digest) {
317 loaded.bundle.applicationId = loaded.manifest.application;
318 loaded.bundle.manifestDigest = digest.substr(kHashPrefix.size());
319 for (ManifestFile const& file : loaded.manifest.files) {
320 if (file.kind != kScreenFile && file.kind != kAppFile) {
321 continue;
322 }
323 if (auto const found = loaded.files.find(file.hash); found != loaded.files.end()) {
324 loaded.bundle.documents.insert_or_assign(file.id, found->second);
325 }
326 }
327 }

331 ::morph::wire::Envelope request;

337 /// @brief Sends the `ui-hello`.
338 static void sendHello(std::shared_ptr<Run> const& run) {
339 ::morph::wire::Envelope request = requestOf(run, kHelloKind);
340 request.body = detail::writeMessage(run->shared->client);

366 /// @brief The reply to a request, or the error it carries: a transport failure is `Unreachable`, an `err` is
367 /// `Refused` with the server's message.
368 [[nodiscard]] static std::expected<std::string, ConnectError> bodyOf(ExchangeReply reply) {
369 if (!reply) {
370 return std::unexpected(ConnectError{.kind = ConnectError::Kind::Unreachable, .message = reply.error()});
371 }
372 if (reply->kind != "ok") {
373 return std::unexpected(
374 ConnectError{.kind = ConnectError::Kind::Refused, .message = std::move(reply->message)});
375 }
376 return std::move(reply->body);
377 }
378
379 /// @brief A fetched file, verified against its hash, or why it is refused.
380 [[nodiscard]] static std::expected<std::string, ConnectError> verified(ExchangeReply reply,
381 Expected const& expected) {
382 auto body = bodyOf(std::move(reply));
383 if (!body) {
384 body.error().message = expected.what + ": " + body.error().message;
385 return body;
386 }
387 if (body->size() > kMaxFileBytes) {
388 return std::unexpected(ConnectError{
389 .kind = ConnectError::Kind::Unverified,
390 .message = expected.what + " is larger than " + std::to_string(kMaxFileBytes) + " bytes"});
391 }
392 if (contentHash(*body) != expected.hash) {
393 return std::unexpected(ConnectError{.kind = ConnectError::Kind::Unverified,
394 .message = expected.what + " does not match its hash"});
395 }
396 return body;
397 }

399 static void onHello(std::shared_ptr<Run> const& run, ExchangeReply reply) {
400 auto body = bodyOf(std::move(reply));
401 if (!body) {
402 ConnectError error = std::move(body.error());
403 if (error.message == "unknown envelope kind: " + std::string{kHelloKind}) {
404 error.kind = ConnectError::Kind::NoUiService;
405 error.message = "the server serves no UI";
406 } else if (error.message.starts_with("missing vocabulary: ")) {
407 error.kind = ConnectError::Kind::Unsupported;
408 }
409 fail(run, std::move(error));
410 return;
411 }
412 auto hello = parseHelloReply(*body);
413 if (!hello) {
414 fail(run, ConnectError::Kind::Unsupported, std::move(hello.error()));
415 return;
416 }
417 HelloRequest const& client = run->shared->client;
418 for (std::string const& vocabulary : hello->vocabularies) {
419 if (std::ranges::find(client.vocabularies, vocabulary) == client.vocabularies.end()) {
420 fail(run, ConnectError::Kind::Unsupported,
421 "the application needs the vocabulary " + vocabulary + ", which this client does not speak");
422 return;
423 }
424 }
425 run->loaded.hello = std::move(*hello);
426 std::string const& digest = run->loaded.hello.digest;
427 if (auto cached = run->shared->cache.get(digest)) {
428 onManifest(run, std::move(*cached));
429 return;
430 }
431 fetch(run, digest, [run](ExchangeReply fetched) {
432 auto text =
433 verified(std::move(fetched), Expected{.hash = run->loaded.hello.digest, .what = "the manifest"});
434 if (!text) {
435 fail(run, std::move(text.error()));
436 return;
437 }
438 onManifest(run, std::move(*text));
439 });
440 }
441
442 static void onManifest(std::shared_ptr<Run> const& run, std::string text) {
443 auto manifest = parseManifest(text);
444 if (!manifest) {
445 fail(run, ConnectError::Kind::Unsupported, std::move(manifest.error()));
446 return;
447 }
448 static_cast<void>(run->shared->cache.put(std::move(text)));
449 run->loaded.manifest = std::move(*manifest);
450 sendCatalog(run);
451 }
452
453 static void onCatalog(std::shared_ptr<Run> const& run, ExchangeReply reply) {
454 auto body = bodyOf(std::move(reply));
455 if (!body) {
456 fail(run, std::move(body.error()));
457 return;
458 }
459 auto catalog = parseCatalogReply(*body);
460 if (!catalog) {
461 fail(run, ConnectError::Kind::Unsupported, std::move(catalog.error()));
462 return;
463 }
464 run->loaded.catalog = std::move(*catalog);
465 std::vector<Expected> missing;
466 for (ManifestFile const& file : run->loaded.manifest.files) {
467 if (!wanted(run->shared->client, run->loaded.catalog, file)) {
468 continue;
469 }
470 if (auto cached = run->shared->cache.get(file.hash)) {
471 run->loaded.files.insert_or_assign(file.hash, std::move(*cached));
472 } else {
473 missing.push_back(Expected{.hash = file.hash, .what = "the " + file.kind + " `" + file.id + "`"});
474 }
475 }
476 // One count for each fetch and one for this loop, so a synchronous exchange that answers every fetch
477 // before the next is sent cannot settle the run early.
478 run->pending = missing.size() + 1;
479 for (Expected& expected : missing) {
480 std::string const hash = expected.hash;
481 fetch(run, hash, [run, expected = std::move(expected)](ExchangeReply fetched) {
482 auto bytes = verified(std::move(fetched), expected);
483 {
484 std::scoped_lock const lock{run->mutex};
485 if (!bytes) {
486 if (!run->failure) {
487 run->failure = std::move(bytes.error());
488 }
489 } else {
490 static_cast<void>(run->shared->cache.put(*bytes));
491 run->loaded.files.insert_or_assign(expected.hash, std::move(*bytes));
492 }
493 }
494 settle(run);
495 });
496 }
497 settle(run);
498 }
499
500 /// @brief Whether the run fetches @p file: a kind the client uses, and for a screen, one the catalog lists as
501 /// available or public, which is what the server serves this principal.
502 [[nodiscard]] static bool wanted(HelloRequest const& client, CatalogReply const& catalog,
503 ManifestFile const& file) {
504 if (!usable(client, file.kind)) {
505 return false;
506 }
507 if (file.kind != kScreenFile) {
508 return true;
509 }
510 return std::ranges::any_of(catalog.entries, [&file](CatalogEntry const& entry) {
511 return entry.id == file.id && (entry.available || entry.isPublic);
512 });
513 }
514
515 /// @brief Counts one fetch, or the loop that sent them, as settled; the last delivers the run.
516 static void settle(std::shared_ptr<Run> const& run) {
517 {
518 std::scoped_lock const lock{run->mutex};
519 if (--run->pending != 0) {
520 return;
521 }
522 }
523 // Past the last count nothing else touches the run, so it is read without the lock.
524 if (run->failure) {
525 fail(run, std::move(*run->failure));
526 return;
527 }
528 assemble(run->loaded, run->loaded.hello.digest);
529 run->shared->cache.store().setLastDigest(run->loaded.hello.application, run->loaded.hello.digest);
530 run->done(std::move(run->loaded));
531 }
