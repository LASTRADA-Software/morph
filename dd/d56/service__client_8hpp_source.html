41/// @brief Where a client keeps verified files between runs, by content hash, and the digest of each application's
42/// last verified bundle.
43///
44/// A store is not trusted: `VerifiedCache` hashes what it reads back. A client calls it from whichever thread its
45/// exchange calls back on, and runs of several `open`s may overlap, so an implementation is safe to call from any
46/// thread.
47class IBundleStore {
48public:
49 IBundleStore() = default;
50 virtual ~IBundleStore() = default;
51 IBundleStore(IBundleStore const&) = delete;
52 IBundleStore& operator=(IBundleStore const&) = delete;
53 IBundleStore(IBundleStore&&) = delete;
54 IBundleStore& operator=(IBundleStore&&) = delete;
55
56 /// @brief Reads a file.
57 /// @param hash The file's content hash.
58 /// @return Its bytes, or nothing when the store has none.
59 [[nodiscard]] virtual std::optional<std::string> get(std::string_view hash) const = 0;
60
61 /// @brief Keeps a file.
62 /// @param hash The file's content hash.
63 /// @param bytes The file.
64 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the key, then the value.
65 virtual void put(std::string const& hash, std::string bytes) = 0;
66
67 /// @brief Drops a file.
68 /// @param hash The file's content hash.
69 virtual void remove(std::string_view hash) = 0;
70
71 /// @brief The digest of an application's last verified bundle.
72 /// @param application The application's id.
73 /// @return The digest, or nothing.
74 [[nodiscard]] virtual std::optional<std::string> lastDigest(std::string_view application) const = 0;
75
76 /// @brief Records an application's last verified bundle.
77 /// @param application The application's id.
78 /// @param digest The manifest's content hash.
79 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the key, then the value.
80 virtual void setLastDigest(std::string const& application, std::string digest) = 0;
81};

183/// @brief Opens an application's bundle through the UI service, verifying everything it fetches.
184///
185/// `open` runs `ui-hello`, refuses a bundle that needs a vocabulary the client does not speak, takes the manifest from
186/// the cache or fetches it by the digest and verifies it, runs `ui-catalog`, and fetches the files the principal may
187/// use that the cache does not hold, all at once, verifying each. A file that does not hash to its manifest entry,
188/// or a manifest that does not hash to the digest, refuses the whole bundle (`Unverified`). An unchanged manifest
189/// fetches nothing.
190class MORPH_API UiClient {
191public:
192 /// @brief A client.
193 /// @param exchange The transport.
194 /// @param client What the client speaks and carries, sent with `ui-hello`.
195 /// @param store Where verified files are kept.
196 /// @throws std::invalid_argument if @p exchange is empty or @p store is null.
197 UiClient(Exchange exchange, HelloRequest client, std::shared_ptr<IBundleStore> store);
198
199 /// @brief What `open` delivers.
200 using Done = std::function<void(std::expected<LoadedBundle, ConnectError>)>;
201
202 /// @brief Opens the bundle. @p done is called once, from whichever thread the exchange calls back on; a
203 /// synchronous exchange calls it before `open` returns.
204 /// @param session The session every request carries: the principal's token, or none before sign-in.
205 /// @param done Called with the bundle, or why there is none.
206 void open(::morph::session::Context session, Done done) const;
207
208 /// @brief What `poll` delivers: the changed bundle, nothing when the digest is the one already held, or why
209 /// there is none.
210 using PollDone = std::function<void(std::expected<std::optional<LoadedBundle>, ConnectError>)>;
211
212 /// @brief One step of a development client's reload (spec 5 §12): sends `ui-hello`, and when the manifest digest
213 /// differs from @p knownDigest, opens the changed bundle as `open` does, fetching only what the cache
214 /// does not hold. @p done is called once, as `open`'s is.
215 /// @param session The session every request carries.
216 /// @param knownDigest The digest of the bundle the client has, as `HelloReply::digest` spells it.
217 /// @param done Called with the changed bundle, with nothing when the digest is unchanged, or with why the
218 /// poll failed.
219 void poll(::morph::session::Context session, std::string knownDigest, PollDone done) const;
220
221 /// @brief Rebuilds the last bundle this store verified for @p application, without a connection. Files the
222 /// store no longer holds, or that no longer verify, are left out; the catalog is empty.
223 /// @param application The application's id.
224 /// @return The bundle, or `Unreachable` when no verified manifest is cached.
225 [[nodiscard]] std::expected<LoadedBundle, ConnectError> openOffline(std::string_view application) const;
