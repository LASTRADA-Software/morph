75/// @brief 16 kHz mono 16-bit samples as a WAV file.
76/// @param samples The samples.
77/// @return The file's bytes: a 44-byte header, then the samples little-endian.
78[[nodiscard]] inline std::string wavFromPcm16(std::span<std::int16_t const> samples) {
79 std::string out;
80 out.reserve(kWavHeaderBytes + (samples.size() * 2));
81 auto const put32 = [&out](std::uint32_t const value) {
82 for (unsigned shift = 0; shift < 32U; shift += 8U) {
83 out.push_back(static_cast<char>((value >> shift) & 0xFFU));
84 }
85 };
86 auto const put16 = [&out](std::uint16_t const value) {
87 out.push_back(static_cast<char>(value & 0xFFU));
88 out.push_back(static_cast<char>((static_cast<unsigned>(value) >> 8U) & 0xFFU));
89 };
90 auto const data = static_cast<std::uint32_t>(samples.size() * 2);
91 out += "RIFF";
92 put32(36U + data);
93 out += "WAVEfmt ";
94 put32(16U);
95 put16(1U);
96 put16(1U);
97 put32(static_cast<std::uint32_t>(kSpeechSampleRate));
98 put32(static_cast<std::uint32_t>(kSpeechSampleRate) * 2U);
99 put16(2U);
100 put16(16U);
101 out += "data";
102 put32(data);
103 for (std::int16_t const sample : samples) {
104 put16(static_cast<std::uint16_t>(sample));
105 }
106 return out;
107}

109/// @brief Turns a device's samples, at any rate and channel count, into 16 kHz mono 16-bit, one chunk at a time.
110///
111/// Channels are averaged. Down-sampling averages the input samples one output sample spans, a box filter that
112/// attenuates most of what lies above the new Nyquist frequency; up-sampling interpolates between neighbours. Every
113/// position is computed in integers from the output sample's index and the samples consumed, never accumulated, so
114/// chunks give exactly what one pass would.
115class SpeechResampler {
116public:
117 /// @brief A resampler for one device format.
118 /// @param channels The device's channel count; less than 1 counts as 1.
119 /// @param sampleRate The device's sample rate; less than 1 counts as `kSpeechSampleRate`.
120 SpeechResampler(int channels, int sampleRate) // NOLINT(bugprone-easily-swappable-parameters)
121 : _channels{static_cast<std::size_t>(std::max(channels, 1))},
122 _rate{static_cast<std::uint64_t>(sampleRate > 0 ? sampleRate : kSpeechSampleRate)} {}
123
124 /// @brief Takes the next chunk.
125 /// @param interleaved Samples in [-1, 1], frame by frame; a trailing partial frame is ignored.
126 /// @return The 16 kHz mono samples the input so far completes.
127 [[nodiscard]] std::vector<std::int16_t> push(std::span<float const> interleaved) {
128 std::size_t const frames = interleaved.size() / _channels;
129 for (std::size_t frame = 0; frame < frames; ++frame) {
130 auto const samples = interleaved.subspan(frame * _channels, _channels);
131 float const sum = std::accumulate(samples.begin(), samples.end(), 0.0F);
132 _pending.push_back(sum / static_cast<float>(_channels));
133 }
134 std::vector<std::int16_t> out;
135 while (ready()) {
136 out.push_back(toPcm16(next()));
137 ++_produced;
138 }
139 // What the next output sample no longer needs.
140 auto const done = std::min(indexOf(startOf(_produced) - _consumed), _pending.size());
141 _pending.erase(_pending.begin(), _pending.begin() + static_cast<std::ptrdiff_t>(done));
142 _consumed += done;
143 return out;
144 }
145
146private:
147 static constexpr auto kSpeechRate = static_cast<std::uint64_t>(kSpeechSampleRate);
148
149 // An offset into _pending. The cast narrows on 32-bit targets and is an identity elsewhere, where GCC's
150 // -Wuseless-cast would reject it.
151 [[nodiscard]] static std::size_t indexOf(std::uint64_t const offset) {
152#if defined(__GNUC__) && !defined(__clang__)
153#pragma GCC diagnostic push
154#pragma GCC diagnostic ignored "-Wuseless-cast"
155#endif
156 return static_cast<std::size_t>(offset);
157#if defined(__GNUC__) && !defined(__clang__)
158#pragma GCC diagnostic pop
159#endif
160 }
161
162 // Where output sample @p index starts, counted in input samples from the first one ever pushed.
163 [[nodiscard]] std::uint64_t startOf(std::uint64_t const index) const { return index * _rate / kSpeechRate; }
164
165 [[nodiscard]] bool ready() const {
166 std::uint64_t const available = _consumed + _pending.size();
167 return _rate >= kSpeechRate ? startOf(_produced + 1) <= available : startOf(_produced) + 1 < available;
168 }
169
170 [[nodiscard]] float next() const {
171 auto const first = indexOf(startOf(_produced) - _consumed);
172 if (_rate >= kSpeechRate) {
173 auto const end = std::max(indexOf(startOf(_produced + 1) - _consumed), first + 1);
174 float sum = 0.0F;
175 for (std::size_t index = first; index < end; ++index) {
176 sum += _pending.at(index);
177 }
178 return sum / static_cast<float>(end - first);
179 }
180 auto const fraction = static_cast<float>((_produced * _rate) % kSpeechRate) / static_cast<float>(kSpeechRate);
181 return _pending.at(first) + ((_pending.at(first + 1) - _pending.at(first)) * fraction);
182 }
183
184 [[nodiscard]] static std::int16_t toPcm16(float const value) {
185 return static_cast<std::int16_t>(std::lround(std::clamp(value, -1.0F, 1.0F) * 32767.0F));
186 }
187
188 std::size_t _channels;
189 std::uint64_t _rate;
190 std::uint64_t _produced = 0;
191 std::uint64_t _consumed = 0;
192 std::vector<float> _pending;
193};

195/// @brief A whole buffer as 16 kHz mono 16-bit: `SpeechResampler::push` once.
196/// @param interleaved Samples in [-1, 1], frame by frame.
197/// @param channels The channel count.

199/// @return The samples.
200[[nodiscard]] inline std::vector<std::int16_t> resampleToSpeech(std::span<float const> interleaved, int channels,

212/// @brief A clip being recorded, which stops at the server's bounds: its time from the start, and the size its WAV
213/// would have. The clock is passed in, so the bounds are tested without waiting.
214class DictationClip {
215public:
216 /// @brief An empty clip.
217 /// @param limits The bounds.
218 /// @param start When recording began.
219 DictationClip(DictationLimits limits, std::chrono::steady_clock::time_point start)
220 : _limits{limits},
221 _start{start},
222 _capacity{limits.maxBytes > kWavHeaderBytes ? (limits.maxBytes - kWavHeaderBytes) / 2 : 0} {}
223
224 /// @brief Adds samples, as many as fit the byte bound.
225 /// @param samples 16 kHz mono samples; may be empty, to check the time bound alone.
226 /// @param now The time now.
227 /// @return `Full` once a bound is reached; samples past the byte bound are dropped.
228 ClipState append(std::span<std::int16_t const> samples, std::chrono::steady_clock::time_point now) {
229 if (_full) {
230 return ClipState::Full;
231 }
232 std::size_t const room = _capacity - _samples.size();
233 std::size_t const taken = std::min(room, samples.size());
234 _samples.insert(_samples.end(), samples.begin(), samples.begin() + static_cast<std::ptrdiff_t>(taken));
235 _full = _samples.size() >= _capacity || now - _start >= _limits.maxSeconds;
236 return _full ? ClipState::Full : ClipState::Recording;
237 }
238
239 /// @brief The whole seconds recorded so far.
240 /// @param now The time now.
241 /// @return They.
242 [[nodiscard]] std::chrono::seconds elapsed(std::chrono::steady_clock::time_point now) const {
243 return std::chrono::duration_cast<std::chrono::seconds>(now - _start);
244 }
245
246 /// @brief The samples held.
247 /// @return Their number.
248 [[nodiscard]] std::size_t samples() const noexcept { return _samples.size(); }
249
250 /// @brief The clip as a WAV file.
251 /// @return Its bytes: at most `DictationLimits::maxBytes`, or just the header when that bound is smaller than the
252 /// header, which holds no sample.
253 [[nodiscard]] std::string wav() const { return wavFromPcm16(_samples); }
254
255private:
256 DictationLimits _limits;
257 std::chrono::steady_clock::time_point _start;
258 std::size_t _capacity;
259 std::vector<std::int16_t> _samples;
260 bool _full = false;
261};
