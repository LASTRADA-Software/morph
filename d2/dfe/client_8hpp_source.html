30/// @brief Sends what a person says, and keeps what to show for each conversation.
31///
32/// Events are hints and the server's `TurnState` is the record: the client reads it when a turn starts, when an event
33/// says one ended, and on `resync`, and what it reads replaces what events built. What it shows lasts the session, and
34/// only for display. The server keeps the conversation itself.
35///
36/// Everything runs on the owner: every call, and the exchange's replies. A turn belongs to the conversation it started
37/// in, so switching conversations does not move it. Neither copyable nor movable.
38class ChatClient {
39public:
40 /// @brief One line to show.
41 struct Entry {
42 /// @brief `user`, `assistant`, `tool`, `error`, or `note`: a line the client adds itself, such as why a
43 /// dictation failed, which the server never sees.
44 std::string kind;
45 /// @brief Its text; a `tool` entry's is the tool's name.
46 std::string text;
47 /// @brief A `tool` entry's arguments as JSON text; empty otherwise.
48 std::string arguments;
49 /// @brief Whether a `tool` entry's action ran.
50 bool ok = false;
51 };
52
53 /// @brief A draft the server accepted, and where the turn that proposed it was started.
54 struct DraftProposal {
55 /// @brief The screen the turn was started on.
56 std::string screenId;
57 /// @brief That screen's identity, canonically encoded, as the turn's request carried it.
58 std::string identity;
59 /// @brief The `origin` the turn was started with, which a shell uses to name the tab.
60 std::uint64_t origin = 0;
61 /// @brief The accepted document as JSON text.
62 std::string document;
63 };

65 /// @brief A client over @p exchange.
66 /// @param exchange What envelopes go through; it answers on the owner.
67 /// @param session What each envelope carries, read at each send so a sign-in takes effect.
68 /// @param changed Called after anything shown has changed.
69 ChatClient(Exchange exchange, std::function<session::Context()> session, std::function<void()> changed)
70 : _exchange{std::move(exchange)}, _session{std::move(session)}, _changed{std::move(changed)} {}

78 /// @brief Asks whether this server offers this principal a chat, and dictation, which `available` and `dictation`
79 /// then answer.
80 void probe() {
81 send(std::string{kStateKind}, "{}", [this](ExchangeReply const& reply) {
82 _available = reply.has_value() && reply->kind == "ok";
83 _dictation.reset();
84 if (_available) {
85 auto const state = readTurnState(reply->body);
86 if (state && state->dictation && state->dictation->maxSeconds > 0 && state->dictation->maxBytes > 0) {
87 _dictation = DictationLimits{
88 .maxSeconds = std::chrono::seconds{static_cast<std::int64_t>(state->dictation->maxSeconds)},
89 .maxBytes = static_cast<std::size_t>(state->dictation->maxBytes)};
90 }
91 }
92 notify();
93 });
94 }

104 /// @brief Hands a dictated clip to the server, and answers with its text once the server's state says it is done.
105 ///
106 /// @p done runs exactly once: with the text when the state reads `done`; with the reason when it reads `failed`,
107 /// when the server refuses the clip, when the exchange fails, on a reconnect (the old connection's close dropped
108 /// the transcription), when `end` runs, and at once when another transcription is still waiting.
109 /// @param wav The clip, WAV bytes.
110 /// @param language The language spoken; nothing leaves it to the server's default.
111 /// @param done What gets the text or the reason.
112 void transcribe(std::string wav, std::optional<std::string> language,
113 std::function<void(std::expected<std::string, std::string>)> done) {
114 if (_transcribing) {
115 if (done) {
116 done(std::unexpected(std::string{"a dictation is already being transcribed"}));
117 }
118 return;
119 }
120 std::uint64_t const seq = ++_transcribeSeq;
121 _transcribing = Transcribing{.seq = seq, .id = 0, .done = std::move(done)};
122 TranscribeRequest const request{.audio = detail::base64Encode(wav),
123 .mediaType = std::string{kDictationMediaType},
124 .language = std::move(language)};
125 wav = std::string{};
126 send(std::string{kTranscribeKind}, writeMessage(request),
127 [this, seq](ExchangeReply const& reply) { transcribeAccepted(seq, reply); });
128 }

130 /// @brief Adds a line of the client's own to the selected conversation, such as why a dictation failed. It stays
131 /// when a running turn is redrawn, after the turn's lines, and is never sent.
132 /// @param text The line.
133 void note(std::string text) {
134 _conversations[_selected].push_back(
135 Entry{.kind = "note", .text = std::move(text), .arguments = {}, .ok = false});
136 notify();
137 }

139 /// @brief Sets what receives a draft the server accepted. Called on the owner, once per distinct document of a
140 /// turn, from an event or from the state read; whichever carries it first.
141 /// @param handler The receiver; empty drops drafts.
142 void onDraft(std::function<void(DraftProposal const&)> handler) { _onDraft = std::move(handler); }

144 /// @brief Starts a turn, and selects its conversation. Refused locally while a turn runs, as the server would.
145 /// @param request What the person said, and where they are.
146 /// @param origin An opaque number given back with each draft the turn proposes, so a shell can tell which tab
147 /// the turn belongs to. The client never reads it.
148 void send(TurnRequest const& request, std::uint64_t origin = 0) {
149 if (_active) {
150 return;
151 }
152 _selected = Key{request.screenId, request.identity};
153 Conversation& conversation = _conversations[_selected];
154 conversation.push_back(Entry{.kind = "user", .text = request.text, .arguments = {}, .ok = false});
155 _active = Active{.key = _selected, .base = conversation.size(), .origin = origin};
156 notify();
157 send(std::string{kTurnKind}, writeMessage(request), [this](ExchangeReply const& reply) { accepted(reply); });
158 }
159
160 /// @brief Asks the server to stop the running turn.
161 void cancel() {
162 if (!_active) {
163 return;
164 }
165 send(std::string{kCancelKind}, "{}", [this](ExchangeReply const& /*reply*/) { readState(); });
166 }
167
168 /// @brief Ends the chat: the server stops this connection's turn and drops the principal's conversations, and this
169 /// client forgets what it was showing. Sent ahead of a sign-out, while the session still proves who is
170 /// asking.
171 /// @param done Called once the server answered or the exchange failed, so a sign-out never waits on a dead
172 /// connection.
173 void end(std::function<void()> done) {
174 _active.reset();
175 // Answered before the conversations are cleared: what answers it may add a note, which must not outlive the
176 // sign-out.
177 if (_transcribing) {
178 transcribed(std::unexpected(std::string{"the chat ended, so the dictation stopped"}));
179 }
180 _conversations.clear();
181 _dictation.reset();
182 notify();
183 send(std::string{kEndKind}, "{}", [done = std::move(done)](ExchangeReply const& /*reply*/) {
184 if (done) {
185 done();
186 }
187 });
188 // The analyzer loses track of the temporary std::function here; `send` moves it into a callback that owns it.
189 // NOLINTNEXTLINE(clang-analyzer-cplusplus.NewDeleteLeaks)
190 }

192 /// @brief Selects which conversation `entries` shows.
193 /// @param screenId The screen.
194 /// @param identity The screen's identity's canonical text; empty for none.
195 void selectConversation(std::string screenId, std::string identity) {
196 Key key{std::move(screenId), std::move(identity)};
197 if (key != _selected) {
198 _selected = std::move(key);
199 notify();
200 }

203 /// @brief Takes an event the server sent. A topic other than the chat's and `resync` is ignored.
204 /// @param event The event.
205 void deliverEvent(wire::EventBody const& event) {
206 if (event.topic == wire::kResyncTopic) {
207 readState();
208 readTranscription();
209 return;
210 }
211 if (event.topic != kChatTopic) {
212 return;
213 }
214 if (auto const hint = readTranscriptionEvent(event.payload.str);
215 hint && hint->kind == kTranscriptionEventKind) {
216 if (_transcribing && _transcribing->id != 0 && hint->state.id == _transcribing->id) {
217 readTranscription();
218 }
219 return;
220 }
221 if (!_active) {
222 return;
223 }
224 auto const read = readTurnEvent(event.payload.str);
225 if (!read || read->turnId != _active->turnId || read->seq <= _active->seq) {
226 return;
227 }
228 _active->seq = read->seq;
229 if (read->kind == "text") {
230 // A piece joins the text before it unless a tool call came between, or the server said a round began.
231 if (read->newRound || _active->parts.empty() || _active->parts.back().kind != "text") {
232 _active->parts.push_back(TurnPart{.kind = "text", .text = read->text, .step = {}});
233 } else {
234 _active->parts.back().text += read->text;
235 }
236 } else if (read->kind == "tool") {
237 _active->parts.push_back(TurnPart{
238 .kind = "tool",
239 .text = {},
240 .step = ToolStep{.name = read->text, .ok = read->ok, .message = {}, .arguments = read->arguments}});
241 } else if (read->kind == "draft") {
242 hand(*_active, read->text);
243 return;
244 } else {
245 readState();
246 return;
247 }
248 show(*_active);
249 }
250
251 /// @brief Tells the client its connection was replaced. A running turn ended with the old connection, since a new
252 /// one has new instance ids; what was shown stays, and the conversation goes on from the next turn.
253 void reconnected() {
254 if (_active) {
255 _active->error = "the connection was lost, so this turn stopped";
256 show(*_active);
257 _active.reset();
258 }
259 if (_transcribing) {
260 transcribed(std::unexpected(std::string{"the connection was lost, so the dictation stopped"}));
261 }
262 probe();
263 }

281 // The running turn. `base` is where its entries start in its conversation, so what it shows can be redrawn.
282 struct Active {
283 Key key;
284 std::size_t base = 0;
285 std::uint64_t turnId = 0;
286 std::uint64_t seq = 0;
287 // What the turn made, in order: one part per round's text and per tool call.
288 std::vector<TurnPart> parts;
289 std::string error;
290 // What `send` was given; handed back with each draft.
291 std::uint64_t origin = 0;
292 // The last draft handed on, so the event and the state that both carry it deliver it once.
293 std::optional<std::string> handed;
294 };

296 // The transcription waiting for its text. `seq` tells this one's accept from an earlier one's; `id` is 0 until
297 // the server accepted it.
298 struct Transcribing {
299 std::uint64_t seq = 0;
300 std::uint64_t id = 0;
301 std::function<void(std::expected<std::string, std::string>)> done;
302 };
303
304 void transcribeAccepted(std::uint64_t seq, ExchangeReply const& reply) {
305 if (!_transcribing || _transcribing->seq != seq) {
306 return;
307 }
308 if (!reply || reply->kind != "ok") {
309 transcribed(std::unexpected(reply ? reply->message : reply.error()));
310 return;
311 }
312 auto const read = readTranscribeAccepted(reply->body);
313 if (!read || read->id == 0) {
314 transcribed(
315 std::unexpected(read ? std::string{"the dictation's reply names no transcription"} : read.error()));
316 return;
317 }
318 _transcribing->id = read->id;
319 readTranscription();
320 }
321
322 void readTranscription() {
323 if (!_transcribing || _transcribing->id == 0) {
324 return;
325 }
326 std::uint64_t const wanted = _transcribing->id;
327 send(std::string{kTranscriptionKind}, writeMessage(TranscriptionQuery{.id = wanted}),
328 [this, wanted](ExchangeReply const& reply) { transcriptionRead(wanted, reply); });
329 }
330
331 // A read that fails, or that answers nonsense, ends the dictation with its reason: nothing fails silently.
332 void transcriptionRead(std::uint64_t wanted, ExchangeReply const& reply) {
333 if (!_transcribing || _transcribing->id != wanted) {
334 return;
335 }
336 if (!reply) {
337 transcribed(std::unexpected(reply.error()));
338 return;
339 }
340 if (reply->kind != "ok") {
341 transcribed(std::unexpected(reply->message));
342 return;
343 }
344 auto const state = readTranscriptionState(reply->body);
345 if (!state) {
346 transcribed(std::unexpected(state.error()));
347 return;
348 }
349 if (state->status == TranscriptionStatus::Running) {
350 return;
351 }
352 if (state->status == TranscriptionStatus::Done) {
353 transcribed(state->text.value_or(std::string{}));
354 } else {
355 transcribed(std::unexpected(state->error.value_or(std::string{"the transcription failed"})));
356 }
357 }
358
359 // Clears the waiting transcription before answering, so `done` may start the next one.
360 void transcribed(std::expected<std::string, std::string> outcome) {
361 if (!_transcribing) {
362 return;
363 }
364 auto const done = std::move(_transcribing->done);
365 _transcribing.reset();
366 if (done) {
367 done(std::move(outcome));
368 }
369 }

371 // Hands a draft on unless this turn already did.
372 void hand(Active& active, std::string const& document) {
373 if (active.handed == document || !_onDraft) {
374 active.handed = document;
375 return;
376 }
377 active.handed = document;
378 _onDraft(DraftProposal{.screenId = active.key.first,
379 .identity = active.key.second,
380 .origin = active.origin,
381 .document = document});
382 }

384 void notify() const {
385 if (_changed) {
386 _changed();
387 }

390 // The callback holds a weak token, so a reply after this client is gone reaches nothing.
391 void send(std::string kind, std::string body, std::function<void(ExchangeReply const&)> then) {
392 wire::Envelope request;
393 request.kind = std::move(kind);
394 request.body = std::move(body);
395 request.session = _session ? _session() : session::Context{};
396 auto answer = [alive = std::weak_ptr<bool>{_alive}, then = std::move(then)](ExchangeReply const& reply) {
397 if (!alive.expired()) {
398 then(reply);
399 }
400 };
401 // A transport that throws rather than answers is one more failed exchange: a sign-out waits on this answer.
402 try {
403 _exchange(std::move(request), answer);
404 } catch (std::exception const& error) {
405 answer(std::unexpected(std::string{error.what()}));
406 }
407 }

409 void accepted(ExchangeReply const& reply) {
410 if (!_active) {
411 return;
412 }
413 if (!reply || reply->kind != "ok") {
414 refused(*_active, reply ? reply->message : reply.error());
415 return;
416 }
417 auto const read = readTurnAccepted(reply->body);
418 if (!read) {
419 refused(*_active, read.error());
420 return;
421 }
422 _active->turnId = read->turnId;
423 readState();
424 }
425
426 void refused(Active& active, std::string reason) {
427 active.error = std::move(reason);
428 show(active);
429 _active.reset();
430 notify();
431 }
432
433 void readState() {
434 if (!_active || _active->turnId == 0) {
435 return;
436 }
437 std::uint64_t const turnId = _active->turnId;
438 send(std::string{kStateKind}, writeMessage(StateRequest{.turnId = turnId}),
439 [this, turnId](ExchangeReply const& reply) { stateRead(turnId, reply); });
440 }
441
442 void stateRead(std::uint64_t turnId, ExchangeReply const& reply) {
443 if (!_active || _active->turnId != turnId || !reply || reply->kind != "ok") {
444 return;
445 }
446 auto const state = readTurnState(reply->body);
447 if (!state) {
448 return;
449 }
450 _active->seq = state->seq;
451 _active->error = state->error;
452 _active->parts = state->parts;
453 if (_active->parts.empty()) {
454 // A server that sends no parts is shown as it always was: the tool calls, then the text.
455 for (ToolStep const& step : state->steps) {
456 _active->parts.push_back(TurnPart{.kind = "tool", .text = {}, .step = step});
457 }
458 if (!state->text.empty()) {
459 _active->parts.push_back(TurnPart{.kind = "text", .text = state->text, .step = {}});
460 }
461 }
462 bool const running = state->status == TurnStatus::Running;
463 show(*_active);
464 if (_active && state->draft) {
465 hand(*_active, *state->draft);
466 }
467 if (!running && _active) {
468 _active.reset();
469 notify();
470 }
471 }
472
473 // Redraws the running turn's entries. Notes added since the turn began are the client's own lines: they are kept,
474 // after the turn's.
475 void show(Active const& active) {
476 Conversation& conversation = _conversations[active.key];
477 std::vector<Entry> notes;
478 for (std::size_t index = active.base; index < conversation.size(); ++index) {
479 if (conversation.at(index).kind == "note") {
480 notes.push_back(std::move(conversation.at(index)));
481 }
482 }
483 conversation.resize(active.base);
484 for (TurnPart const& part : active.parts) {
485 if (part.kind == "tool") {
486 conversation.push_back(Entry{
487 .kind = "tool", .text = part.step.name, .arguments = part.step.arguments, .ok = part.step.ok});
488 } else if (!part.text.empty()) {
489 conversation.push_back(Entry{.kind = "assistant", .text = part.text, .arguments = {}, .ok = false});
490 }
491 }
492 if (!active.error.empty()) {
493 conversation.push_back(Entry{.kind = "error", .text = active.error, .arguments = {}, .ok = false});
494 }
495 for (Entry& kept : notes) {
496 conversation.push_back(std::move(kept));
497 }
498 notify();
499 }

502 std::function<session::Context()> _session;
503 std::function<void()> _changed;
504 std::function<void(DraftProposal const&)> _onDraft;
505 std::shared_ptr<bool> _alive = std::make_shared<bool>(true);
506 std::map<Key, Conversation> _conversations;
507 Key _selected;
508 std::optional<Active> _active;
509 bool _available = false;
510 std::optional<DictationLimits> _dictation;
511 std::optional<Transcribing> _transcribing;
512 std::uint64_t _transcribeSeq = 0;
