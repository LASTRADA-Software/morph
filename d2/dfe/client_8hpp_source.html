27/// @brief Sends what a person says, and keeps what to show for each conversation.
28///
29/// Events are hints and the server's `TurnState` is the record: the client reads it when a turn starts, when an event
30/// says one ended, and on `resync`, and what it reads replaces what events built. What it shows lasts the session, and
31/// only for display. The server keeps the conversation itself.
32///
33/// Everything runs on the owner: every call, and the exchange's replies. A turn belongs to the conversation it started
34/// in, so switching conversations does not move it. Neither copyable nor movable.
35class ChatClient {
36public:
37 /// @brief One line to show.
38 struct Entry {
39 /// @brief `user`, `assistant`, `tool` or `error`.
40 std::string kind;
41 /// @brief Its text.
42 std::string text;
43 };
44
45 /// @brief A client over @p exchange.
46 /// @param exchange What envelopes go through; it answers on the owner.
47 /// @param session What each envelope carries, read at each send so a sign-in takes effect.
48 /// @param changed Called after anything shown has changed.
49 ChatClient(Exchange exchange, std::function<session::Context()> session, std::function<void()> changed)
50 : _exchange{std::move(exchange)}, _session{std::move(session)}, _changed{std::move(changed)} {}
51
52 ChatClient(ChatClient const&) = delete;
53 ChatClient& operator=(ChatClient const&) = delete;
54 ChatClient(ChatClient&&) = delete;
55 ChatClient& operator=(ChatClient&&) = delete;
56 ~ChatClient() = default;

58 /// @brief Asks whether this server offers this principal a chat, which `available` then answers.
59 void probe() {
60 send(std::string{kStateKind}, "{}", [this](ExchangeReply const& reply) {
61 _available = reply.has_value() && reply->kind == "ok";
62 notify();
63 });
64 }
65
66 /// @brief Whether the last probe found a chat.
67 /// @return Whether it did.
68 [[nodiscard]] bool available() const noexcept { return _available; }
69
70 /// @brief Starts a turn, and selects its conversation. Refused locally while a turn runs, as the server would.
71 /// @param request What the person said, and where they are.
72 void send(TurnRequest const& request) {
73 if (_active) {
74 return;
75 }
76 _selected = Key{request.screenId, request.identity};
77 Conversation& conversation = _conversations[_selected];
78 conversation.push_back(Entry{.kind = "user", .text = request.text});
79 _active = Active{.key = _selected, .base = conversation.size()};
80 notify();
81 send(std::string{kTurnKind}, writeMessage(request), [this](ExchangeReply const& reply) { accepted(reply); });
82 }
83
84 /// @brief Asks the server to stop the running turn.
85 void cancel() {
86 if (!_active) {
87 return;
88 }
89 send(std::string{kCancelKind}, "{}", [this](ExchangeReply const& /*reply*/) { readState(); });
90 }
91
92 /// @brief Ends the chat: the server stops this connection's turn and drops the principal's conversations, and this
93 /// client forgets what it was showing. Sent ahead of a sign-out, while the session still proves who is
94 /// asking.
95 /// @param done Called once the server answered or the exchange failed, so a sign-out never waits on a dead
96 /// connection.
97 void end(std::function<void()> done) {
98 _active.reset();
99 _conversations.clear();
100 notify();
101 send(std::string{kEndKind}, "{}", [done = std::move(done)](ExchangeReply const& /*reply*/) {
102 if (done) {
103 done();
104 }
105 });
106 // The analyzer loses track of the temporary std::function here; `send` moves it into a callback that owns it.
107 // NOLINTNEXTLINE(clang-analyzer-cplusplus.NewDeleteLeaks)
108 }
109
110 /// @brief Selects which conversation `entries` shows.
111 /// @param screenId The screen.
112 /// @param identity The screen's identity's canonical text; empty for none.
113 void selectConversation(std::string screenId, std::string identity) {
114 Key key{std::move(screenId), std::move(identity)};
115 if (key != _selected) {
116 _selected = std::move(key);
117 notify();
118 }
119 }
120
121 /// @brief Takes an event the server sent. A topic other than the chat's and `resync` is ignored.
122 /// @param event The event.
123 void deliverEvent(wire::EventBody const& event) {
124 if (event.topic == wire::kResyncTopic) {
125 readState();
126 return;
127 }
128 if (event.topic != kChatTopic || !_active) {
129 return;
130 }
131 auto const read = readTurnEvent(event.payload.str);
132 if (!read || read->turnId != _active->turnId || read->seq <= _active->seq) {
133 return;
134 }
135 _active->seq = read->seq;
136 if (read->kind == "text") {
137 _active->text += read->text;
138 } else if (read->kind == "tool") {
139 _active->tools.push_back(read->text);
140 } else {
141 readState();
142 return;
143 }
144 show(*_active);
145 }
146
147 /// @brief Tells the client its connection was replaced. A running turn ended with the old connection, since a new
148 /// one has new instance ids; what was shown stays, and the conversation goes on from the next turn.
149 void reconnected() {
150 if (_active) {
151 _active->error = "the connection was lost, so this turn stopped";
152 show(*_active);
153 _active.reset();
154 }
155 probe();
156 }
157
158 /// @brief What the selected conversation shows.
159 /// @return Its entries, oldest first.
160 [[nodiscard]] std::vector<Entry> const& entries() const {
161 static std::vector<Entry> const kNone;
162 auto const found = _conversations.find(_selected);
163 return found == _conversations.end() ? kNone : found->second;
164 }
165
166 /// @brief Whether a turn is running, in any conversation.
167 /// @return Whether one is.
168 [[nodiscard]] bool running() const noexcept { return _active.has_value(); }
169
170private:
171 using Key = std::pair<std::string, std::string>;
172 using Conversation = std::vector<Entry>;
173
174 // The running turn. `base` is where its entries start in its conversation, so what it shows can be redrawn.
175 struct Active {
176 Key key;
177 std::size_t base = 0;
178 std::uint64_t turnId = 0;
179 std::uint64_t seq = 0;
180 std::vector<std::string> tools;
181 std::string text;
182 std::string error;
183 };
184
185 void notify() const {
186 if (_changed) {
187 _changed();
188 }
189 }
190
191 // The callback holds a weak token, so a reply after this client is gone reaches nothing.
192 void send(std::string kind, std::string body, std::function<void(ExchangeReply const&)> then) {
193 wire::Envelope request;
194 request.kind = std::move(kind);
195 request.body = std::move(body);
196 request.session = _session ? _session() : session::Context{};
197 auto answer = [alive = std::weak_ptr<bool>{_alive}, then = std::move(then)](ExchangeReply const& reply) {
198 if (!alive.expired()) {
199 then(reply);
200 }
201 };
202 // A transport that throws rather than answers is one more failed exchange: a sign-out waits on this answer.
203 try {
204 _exchange(std::move(request), answer);
205 } catch (std::exception const& error) {
206 answer(std::unexpected(std::string{error.what()}));
207 }
208 }
209
210 void accepted(ExchangeReply const& reply) {
211 if (!_active) {
212 return;
213 }
214 if (!reply || reply->kind != "ok") {
215 refused(*_active, reply ? reply->message : reply.error());
216 return;
217 }
218 auto const read = readTurnAccepted(reply->body);
219 if (!read) {
220 refused(*_active, read.error());
221 return;
222 }
223 _active->turnId = read->turnId;
224 readState();
225 }
226
227 void refused(Active& active, std::string reason) {
228 active.error = std::move(reason);
229 show(active);
230 _active.reset();
231 notify();
232 }
233
234 void readState() {
235 if (!_active || _active->turnId == 0) {
236 return;
237 }
238 std::uint64_t const turnId = _active->turnId;
239 send(std::string{kStateKind}, writeMessage(StateRequest{.turnId = turnId}),
240 [this, turnId](ExchangeReply const& reply) { stateRead(turnId, reply); });
241 }
242
243 void stateRead(std::uint64_t turnId, ExchangeReply const& reply) {
244 if (!_active || _active->turnId != turnId || !reply || reply->kind != "ok") {
245 return;
246 }
247 auto const state = readTurnState(reply->body);
248 if (!state) {
249 return;
250 }
251 _active->seq = state->seq;
252 _active->text = state->text;
253 _active->error = state->error;
254 _active->tools.clear();
255 for (ToolStep const& step : state->steps) {
256 _active->tools.push_back(step.name);
257 }
258 bool const running = state->status == TurnStatus::Running;
259 show(*_active);
260 if (!running) {
261 _active.reset();
262 notify();
263 }
264 }
265
266 // Redraws the running turn's entries.
267 void show(Active const& active) {
268 Conversation& conversation = _conversations[active.key];
269 conversation.resize(active.base);
270 for (std::string const& tool : active.tools) {
271 conversation.push_back(Entry{.kind = "tool", .text = tool});
272 }
273 if (!active.text.empty()) {
274 conversation.push_back(Entry{.kind = "assistant", .text = active.text});
275 }
276 if (!active.error.empty()) {
277 conversation.push_back(Entry{.kind = "error", .text = active.error});
278 }
279 notify();
280 }
281
282 Exchange _exchange;
283 std::function<session::Context()> _session;
284 std::function<void()> _changed;
285 std::shared_ptr<bool> _alive = std::make_shared<bool>(true);
286 std::map<Key, Conversation> _conversations;
287 Key _selected;
288 std::optional<Active> _active;
289 bool _available = false;
290};
