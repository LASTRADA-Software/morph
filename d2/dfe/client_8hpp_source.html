27/// @brief Sends what a person says, and keeps what to show for each conversation.
28///
29/// Events are hints and the server's `TurnState` is the record: the client reads it when a turn starts, when an event
30/// says one ended, and on `resync`, and what it reads replaces what events built. What it shows lasts the session, and
31/// only for display. The server keeps the conversation itself.
32///
33/// Everything runs on the owner: every call, and the exchange's replies. A turn belongs to the conversation it started
34/// in, so switching conversations does not move it. Neither copyable nor movable.
35class ChatClient {
36public:
37 /// @brief One line to show.
38 struct Entry {
39 /// @brief `user`, `assistant`, `tool` or `error`.
40 std::string kind;
41 /// @brief Its text; a `tool` entry's is the tool's name.
42 std::string text;
43 /// @brief A `tool` entry's arguments as JSON text; empty otherwise.
44 std::string arguments;
45 /// @brief Whether a `tool` entry's action ran.
46 bool ok = false;
47 };
48
49 /// @brief A client over @p exchange.
50 /// @param exchange What envelopes go through; it answers on the owner.
51 /// @param session What each envelope carries, read at each send so a sign-in takes effect.
52 /// @param changed Called after anything shown has changed.
53 ChatClient(Exchange exchange, std::function<session::Context()> session, std::function<void()> changed)
54 : _exchange{std::move(exchange)}, _session{std::move(session)}, _changed{std::move(changed)} {}
55
56 ChatClient(ChatClient const&) = delete;
57 ChatClient& operator=(ChatClient const&) = delete;
58 ChatClient(ChatClient&&) = delete;
59 ChatClient& operator=(ChatClient&&) = delete;
60 ~ChatClient() = default;

62 /// @brief Asks whether this server offers this principal a chat, which `available` then answers.
63 void probe() {
64 send(std::string{kStateKind}, "{}", [this](ExchangeReply const& reply) {
65 _available = reply.has_value() && reply->kind == "ok";
66 notify();
67 });
68 }
69
70 /// @brief Whether the last probe found a chat.
71 /// @return Whether it did.
72 [[nodiscard]] bool available() const noexcept { return _available; }
73
74 /// @brief Starts a turn, and selects its conversation. Refused locally while a turn runs, as the server would.
75 /// @param request What the person said, and where they are.
76 void send(TurnRequest const& request) {
77 if (_active) {
78 return;
79 }
80 _selected = Key{request.screenId, request.identity};
81 Conversation& conversation = _conversations[_selected];
82 conversation.push_back(Entry{.kind = "user", .text = request.text, .arguments = {}, .ok = false});
83 _active = Active{.key = _selected, .base = conversation.size()};
84 notify();
85 send(std::string{kTurnKind}, writeMessage(request), [this](ExchangeReply const& reply) { accepted(reply); });
86 }
87
88 /// @brief Asks the server to stop the running turn.
89 void cancel() {
90 if (!_active) {
91 return;
92 }
93 send(std::string{kCancelKind}, "{}", [this](ExchangeReply const& /*reply*/) { readState(); });
94 }
95
96 /// @brief Ends the chat: the server stops this connection's turn and drops the principal's conversations, and this
97 /// client forgets what it was showing. Sent ahead of a sign-out, while the session still proves who is
98 /// asking.
99 /// @param done Called once the server answered or the exchange failed, so a sign-out never waits on a dead
100 /// connection.
101 void end(std::function<void()> done) {
102 _active.reset();
103 _conversations.clear();
104 notify();
105 send(std::string{kEndKind}, "{}", [done = std::move(done)](ExchangeReply const& /*reply*/) {
106 if (done) {
107 done();
108 }
109 });
110 // The analyzer loses track of the temporary std::function here; `send` moves it into a callback that owns it.
111 // NOLINTNEXTLINE(clang-analyzer-cplusplus.NewDeleteLeaks)
112 }
113
114 /// @brief Selects which conversation `entries` shows.
115 /// @param screenId The screen.
116 /// @param identity The screen's identity's canonical text; empty for none.
117 void selectConversation(std::string screenId, std::string identity) {
118 Key key{std::move(screenId), std::move(identity)};
119 if (key != _selected) {
120 _selected = std::move(key);
121 notify();
122 }
123 }
124
125 /// @brief Takes an event the server sent. A topic other than the chat's and `resync` is ignored.
126 /// @param event The event.
127 void deliverEvent(wire::EventBody const& event) {
128 if (event.topic == wire::kResyncTopic) {
129 readState();
130 return;
131 }
132 if (event.topic != kChatTopic || !_active) {
133 return;
134 }
135 auto const read = readTurnEvent(event.payload.str);
136 if (!read || read->turnId != _active->turnId || read->seq <= _active->seq) {
137 return;
138 }
139 _active->seq = read->seq;
140 if (read->kind == "text") {
141 // A piece joins the text before it unless a tool call came between, or the server said a round began.
142 if (read->newRound || _active->parts.empty() || _active->parts.back().kind != "text") {
143 _active->parts.push_back(TurnPart{.kind = "text", .text = read->text, .step = {}});
144 } else {
145 _active->parts.back().text += read->text;
146 }
147 } else if (read->kind == "tool") {
148 _active->parts.push_back(TurnPart{
149 .kind = "tool",
150 .text = {},
151 .step = ToolStep{.name = read->text, .ok = read->ok, .message = {}, .arguments = read->arguments}});
152 } else {
153 readState();
154 return;
155 }
156 show(*_active);
157 }
158
159 /// @brief Tells the client its connection was replaced. A running turn ended with the old connection, since a new
160 /// one has new instance ids; what was shown stays, and the conversation goes on from the next turn.
161 void reconnected() {
162 if (_active) {
163 _active->error = "the connection was lost, so this turn stopped";
164 show(*_active);
165 _active.reset();
166 }
167 probe();
168 }
169
170 /// @brief What the selected conversation shows.
171 /// @return Its entries, oldest first.
172 [[nodiscard]] std::vector<Entry> const& entries() const {
173 static std::vector<Entry> const kNone;
174 auto const found = _conversations.find(_selected);
175 return found == _conversations.end() ? kNone : found->second;
176 }
177
178 /// @brief Whether a turn is running, in any conversation.
179 /// @return Whether one is.
180 [[nodiscard]] bool running() const noexcept { return _active.has_value(); }
181
182private:
183 using Key = std::pair<std::string, std::string>;
184 using Conversation = std::vector<Entry>;
185
186 // The running turn. `base` is where its entries start in its conversation, so what it shows can be redrawn.
187 struct Active {
188 Key key;
189 std::size_t base = 0;
190 std::uint64_t turnId = 0;
191 std::uint64_t seq = 0;
192 // What the turn made, in order: one part per round's text and per tool call.
193 std::vector<TurnPart> parts;
194 std::string error;
195 };
196
197 void notify() const {
198 if (_changed) {
199 _changed();
200 }
201 }
202
203 // The callback holds a weak token, so a reply after this client is gone reaches nothing.
204 void send(std::string kind, std::string body, std::function<void(ExchangeReply const&)> then) {
205 wire::Envelope request;
206 request.kind = std::move(kind);
207 request.body = std::move(body);
208 request.session = _session ? _session() : session::Context{};
209 auto answer = [alive = std::weak_ptr<bool>{_alive}, then = std::move(then)](ExchangeReply const& reply) {
210 if (!alive.expired()) {
211 then(reply);
212 }
213 };
214 // A transport that throws rather than answers is one more failed exchange: a sign-out waits on this answer.
215 try {
216 _exchange(std::move(request), answer);
217 } catch (std::exception const& error) {
218 answer(std::unexpected(std::string{error.what()}));
219 }
220 }
221
222 void accepted(ExchangeReply const& reply) {
223 if (!_active) {
224 return;
225 }
226 if (!reply || reply->kind != "ok") {
227 refused(*_active, reply ? reply->message : reply.error());
228 return;
229 }
230 auto const read = readTurnAccepted(reply->body);
231 if (!read) {
232 refused(*_active, read.error());
233 return;
234 }
235 _active->turnId = read->turnId;
236 readState();
237 }
238
239 void refused(Active& active, std::string reason) {
240 active.error = std::move(reason);
241 show(active);
242 _active.reset();
243 notify();
244 }
245
246 void readState() {
247 if (!_active || _active->turnId == 0) {
248 return;
249 }
250 std::uint64_t const turnId = _active->turnId;
251 send(std::string{kStateKind}, writeMessage(StateRequest{.turnId = turnId}),
252 [this, turnId](ExchangeReply const& reply) { stateRead(turnId, reply); });
253 }
254
255 void stateRead(std::uint64_t turnId, ExchangeReply const& reply) {
256 if (!_active || _active->turnId != turnId || !reply || reply->kind != "ok") {
257 return;
258 }
259 auto const state = readTurnState(reply->body);
260 if (!state) {
261 return;
262 }
263 _active->seq = state->seq;
264 _active->error = state->error;
265 _active->parts = state->parts;
266 if (_active->parts.empty()) {
267 // A server that sends no parts is shown as it always was: the tool calls, then the text.
268 for (ToolStep const& step : state->steps) {
269 _active->parts.push_back(TurnPart{.kind = "tool", .text = {}, .step = step});
270 }
271 if (!state->text.empty()) {
272 _active->parts.push_back(TurnPart{.kind = "text", .text = state->text, .step = {}});
273 }
274 }
275 bool const running = state->status == TurnStatus::Running;
276 show(*_active);
277 if (!running) {
278 _active.reset();
279 notify();
280 }
281 }
282
283 // Redraws the running turn's entries.
284 void show(Active const& active) {
285 Conversation& conversation = _conversations[active.key];
286 conversation.resize(active.base);
287 for (TurnPart const& part : active.parts) {
288 if (part.kind == "tool") {
289 conversation.push_back(Entry{
290 .kind = "tool", .text = part.step.name, .arguments = part.step.arguments, .ok = part.step.ok});
291 } else if (!part.text.empty()) {
292 conversation.push_back(Entry{.kind = "assistant", .text = part.text, .arguments = {}, .ok = false});
293 }
294 }
295 if (!active.error.empty()) {
296 conversation.push_back(Entry{.kind = "error", .text = active.error, .arguments = {}, .ok = false});
297 }
298 notify();
299 }
300
301 Exchange _exchange;
302 std::function<session::Context()> _session;
303 std::function<void()> _changed;
304 std::shared_ptr<bool> _alive = std::make_shared<bool>(true);
305 std::map<Key, Conversation> _conversations;
306 Key _selected;
307 std::optional<Active> _active;
308 bool _available = false;
309};
