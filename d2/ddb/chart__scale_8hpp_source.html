24/// @brief One axis of a chart: its ends, exact.
25struct ChartRange {

32/// @brief The ticks of a value axis: their positions and the places they are shown at.
33struct ChartTicks {

40/// @brief Ten to a power, exactly.
41/// @param exponent The power, from -18 to 18.
42/// @return `10^exponent`, tagged with as many places as it has after the point.
43[[nodiscard]] inline math::Rational chartPowerOfTen(int exponent) {
44 if (exponent >= 0) {
45 return math::Rational{math::detail::powerOfTen(static_cast<std::uint32_t>(exponent)), math::DecimalPlaces{0}};
46 }
47 auto const places = static_cast<std::uint32_t>(-exponent);
48 return math::Rational{math::Numerator{1}, math::Denominator{math::detail::powerOfTen(places)},
49 math::DecimalPlaces{places}};
50}

52/// @brief The greatest integer not above a rational.
53/// @param value The rational.
54/// @return Its floor.
55[[nodiscard]] inline std::int64_t chartFloor(math::Rational const& value) {
56 std::int64_t const quotient = value.numerator / value.denominator;
57 return value.numerator % value.denominator != 0 && value.numerator < 0 ? quotient - 1 : quotient;
58}
59
60/// @brief The least integer not below a rational.
61/// @param value The rational.
62/// @return Its ceiling.
63[[nodiscard]] inline std::int64_t chartCeil(math::Rational const& value) {
64 std::int64_t const quotient = value.numerator / value.denominator;
65 return value.numerator % value.denominator != 0 && value.numerator > 0 ? quotient + 1 : quotient;
66}

68/// @brief The axis that holds every value, widened by 5 % of their span at each end so the outermost value is
69/// inside the plot rather than on its edge.
70///
71/// A span of zero is widened by one unit in the last of @p places either side; no values at all give 0 to 1.
72/// @param values The values the axis must hold.
73/// @param places The places a value is shown at, which size the widening of a zero span; at most 18.
74/// @return The range, or empty when the arithmetic overflows.
75[[nodiscard]] inline std::optional<ChartRange> chartRange(std::span<math::Rational const> values,
76 std::uint32_t places) {
77 if (values.empty()) {
78 return ChartRange{.low = math::Rational{0, math::DecimalPlaces{0}},
79 .high = math::Rational{1, math::DecimalPlaces{0}}};
80 }
81 auto const [lowest, highest] = std::ranges::minmax(values);
82 auto const span = math::checkedSub(highest, lowest);
83 if (!span) {
84 return std::nullopt;
85 }
86 static math::Rational const kTwenty{20, math::DecimalPlaces{0}};
87 auto const pad = span->numerator == 0 ? std::expected<math::Rational, math::RationalError>{chartPowerOfTen(
88 -static_cast<int>(std::min(places, math::kMaxDecimalPlaces)))}
89 : math::checkedDiv(*span, kTwenty);
90 if (!pad) {
91 return std::nullopt;
92 }
93 auto const low = math::checkedSub(lowest, *pad);
94 auto const high = math::checkedAdd(highest, *pad);
95 if (!low || !high) {
96 return std::nullopt;
97 }
98 return ChartRange{.low = *low, .high = *high};
99}

101/// @brief How many multiples of @p step lie in @p range, ends included.
102/// @param range The range.
103/// @param step The step, positive.
104/// @return The count, or empty when the arithmetic overflows.
105[[nodiscard]] inline std::optional<std::int64_t> chartTickCount(ChartRange const& range, math::Rational const& step) {
106 auto const first = math::checkedDiv(range.low, step);
107 auto const last = math::checkedDiv(range.high, step);
108 if (!first || !last) {
109 return std::nullopt;
110 }
111 return chartFloor(*last) - chartCeil(*first) + 1;
112}

114/// @brief The largest step of 1, 2 or 5 × 10ᵏ, at most @p power's power of ten, that puts three or more multiples in
115/// @p range.
116/// @param range The axis.
117/// @param power The exponent to start from: the least power of ten not below the range's span.
118/// @return The step and its exponent, or empty when none down to 10⁻¹⁸ does or the arithmetic overflows.
119[[nodiscard]] inline std::optional<std::pair<math::Rational, int>> chartTickStep(ChartRange const& range, int power) {
120 static constexpr int kLowest = -18;
121 static constexpr std::array<std::int64_t, 3> kMantissas = {5, 2, 1};
122 // Steps in descending order: 10^power, then 5, 2 and 1 times each lower power.
123 for (int exponent = power; exponent >= kLowest; --exponent) {
124 for (std::int64_t const mantissa : kMantissas) {
125 if (exponent == power && mantissa != 1) {
126 continue;
127 }
128 auto const step =
129 math::checkedMul(math::Rational{mantissa, math::DecimalPlaces{0}}, chartPowerOfTen(exponent));
130 std::optional<std::int64_t> const count =
131 step ? chartTickCount(range, *step) : std::optional<std::int64_t>{};
132 if (!count) {
133 return std::nullopt;
134 }
135 if (*count >= 3) {
136 return std::pair{*step, exponent};
137 }
138 }
139 }
140 return std::nullopt;
141}

143/// @brief The ticks of a value axis: the multiples, inside @p range, of the largest step of 1, 2 or 5 × 10ᵏ that
144/// puts three or more there.
145///
146/// Three to six ticks, except where a step of 5 × 10ᵏ would put two and the next, 2 × 10ᵏ, puts seven or eight.
147/// @param range The axis, its high end above its low end.
148/// @return The ticks, or empty when the range is too narrow for a step of 10⁻¹⁸ or the arithmetic overflows.
149[[nodiscard]] inline std::optional<ChartTicks> chartTicks(ChartRange const& range) {
150 static constexpr int kLowest = -18;
151 static constexpr int kHighest = 18;
152 auto const span = math::checkedSub(range.high, range.low);
153 if (!span || span->numerator <= 0) {
154 return std::nullopt;
155 }
156 int power = kLowest;
157 while (power < kHighest && chartPowerOfTen(power) < *span) {
158 ++power;
159 }
160 std::optional<std::pair<math::Rational, int>> const step = chartTickStep(range, power);
161 if (!step) {
162 return std::nullopt;
163 }
164 auto const first = math::checkedDiv(range.low, step->first);
165 auto const last = math::checkedDiv(range.high, step->first);
166 if (!first || !last) {
167 return std::nullopt;
168 }
169 ChartTicks ticks{.values = {}, .places = step->second < 0 ? static_cast<std::uint32_t>(-step->second) : 0U};
170 for (std::int64_t index = chartCeil(*first); index <= chartFloor(*last); ++index) {
171 auto const value = math::checkedMul(math::Rational{index, math::DecimalPlaces{0}}, step->first);
172 if (!value) {
173 return std::nullopt;
174 }
175 ticks.values.push_back(*value);
176 }
177 return ticks;
178}
