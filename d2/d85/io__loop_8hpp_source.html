22/// @file
23/// `IoLoop` — the one I/O loop an application runs, and the owner of every
24/// socket, timer and probe built on it.
25///
26/// `morph::net::SocketBackend`, `morph::net::SocketServer`,
27/// `TimeoutScheduler` and `offline::NetworkMonitor` own no thread. Each keeps
28/// its state on a `core::net::PlatformLoop` and touches it only in tasks that
29/// loop runs; its public verbs post to the loop and return. An application
30/// constructs one `IoLoop` and hands it to every one of them, so one thread
31/// carries every connection, every deadline and the connectivity probe.
32///
33/// @par Who turns the loop
34/// - **Native:** the `IoLoop` owns one thread that runs the loop until the
35/// `IoLoop` is destroyed.
36/// - **Single-threaded WebAssembly** (`__EMSCRIPTEN__` without
37/// `__EMSCRIPTEN_PTHREADS__`): there is no thread to start. The loop is
38/// host-driven — the browser's timer pumps it — and the one thread there is
39/// is always the loop's, so a verb runs its body inline.
40
41namespace morph::exec {

43/// @brief Owns a `core::net::PlatformLoop` and, natively, the one thread that
44/// runs it.
45///
46/// Cross-thread surface: `post()`, `runAndWait()`, `runningHere()` and
47/// `weak()`. Everything else about the loop — timers, sockets, flows — is
48/// touched only in tasks it runs.
49///
50/// @par Lifetime
51/// Must outlive every component built on it: a component's destructor posts
52/// its close to the loop and waits for it, which a stopped loop would never
53/// run. Destroy the components first, then the `IoLoop`.
54class IoLoop {
55 /// The loop, shared with the thread that runs it, so that an `IoLoop`
56 /// destroyed on its own thread can let that thread finish the turn it is in.
57 struct Impl {
58 ::core::net::PlatformLoop loop;
59 };
60
61public:
62 /// @brief A non-owning handle that posts to the loop only while it exists.
63 ///
64 /// For a callback that another executor runs and that may outlive the
65 /// component that armed it — a `RemoteServer` reply arriving on a pool
66 /// thread after its transport has gone.
67 class Weak {
68 public:
69 /// @brief Posts @p task if the loop still exists.
70 /// @tparam F A copyable callable taking no arguments.
71 /// @param task What to run on the loop thread.
72 /// @return `false` if the loop is gone and @p task was dropped.
73 template <class F>
74 bool post(F&& task) const {
75 std::shared_ptr<Impl> const impl = _impl.lock();
76 if (!impl) {
77 return false;
78 }
79 impl->loop.post(guarded(std::forward<F>(task)));
80 return true;
81 }
82
83 private:
84 friend class IoLoop;
85 explicit Weak(std::weak_ptr<Impl> impl) : _impl{std::move(impl)} {}
86 std::weak_ptr<Impl> _impl;
87 };
88
89#ifndef MORPH_IO_LOOP_HOST_DRIVEN
90 /// @brief Creates the loop and starts the thread that runs it.
91 IoLoop()
92 : _impl{std::make_shared<Impl>()}, _thread{[impl = _impl] {
93 // One name for every IoLoop thread: an application runs one, and
94 // its sockets, deadlines and probe all turn on it.
95 MORPH_THREAD_NAME("morph.io");
96 impl->loop.run();
97 }} {}
98
99 /// @brief Stops the loop and joins its thread.
100 ///
101 /// Work still queued is dropped by the loop's own teardown, not run. On the
102 /// loop's own thread — a task that dropped the last owner — it cannot
103 /// join: it asks the loop to stop and lets the thread end with the turn it
104 /// is in, the loop kept alive by the thread's own share of it.
105 ~IoLoop() {
106 _impl->loop.stop();
107 if (_thread.get_id() == std::this_thread::get_id()) {
108 _thread.detach();
109 } else if (_thread.joinable()) {
110 _thread.join();
111 }
112 }
113#else
114 /// @brief Creates the loop. Starts no thread: the browser's timer pumps it.
115 IoLoop() : _impl{std::make_shared<Impl>()} {}
116
117 /// @brief Destroys the loop, dropping whatever it still holds.
118 ~IoLoop() = default;
119#endif
120
121 IoLoop(const IoLoop&) = delete;
122 IoLoop& operator=(const IoLoop&) = delete;
123 IoLoop(IoLoop&&) = delete;
124 IoLoop& operator=(IoLoop&&) = delete;

126 /// @brief The loop itself, for a component arming timers or opening
127 /// sockets from inside one of its tasks.
128 /// @return The platform loop; valid for this object's lifetime.
129 [[nodiscard]] ::core::net::EventLoop& loop() noexcept { return _impl->loop; }
130
131 /// @brief Whether the calling thread is inside one of the loop's tasks.
132 ///
133 /// Natively that is the loop's own `ExecutorScope`, stated for every turn.
134 /// Under single-threaded WebAssembly there is one thread, so the answer is
135 /// always yes.
136 /// @return True where the loop's state may be touched directly.
137 [[nodiscard]] bool runningHere() const noexcept {
138#ifndef MORPH_IO_LOOP_HOST_DRIVEN
139 return ::morph::exec::runningOn(static_cast<::core::async::IExecutor const&>(_impl->loop));
140#else
141 return true;
142#endif
143 }
144
145 /// @brief Queues @p task to run on the loop thread, in a later turn.
146 ///
147 /// A throw out of @p task is logged and swallowed: one escaping a turn
148 /// would end the loop's thread.
149 /// @tparam F A copyable callable taking no arguments.
150 /// @param task What to run.
151 template <class F>
152 void post(F&& task) {
153 _impl->loop.post(guarded(std::forward<F>(task)));
154 }
155
156 /// @brief Runs @p task on the loop and returns once it has run.
157 ///
158 /// Inline when the calling thread is already on the loop, so a task of the
159 /// loop never waits on itself. For teardown and for the few verbs that must
160 /// answer with the loop's own result. A task the loop drops unrun (it was
161 /// destroyed first) ends the wait rather than hanging it.
162 /// @tparam F A copyable callable taking no arguments.
163 /// @param task What to run.
164 template <class F>
165 void runAndWait(F&& task) {
166 if (runningHere()) {
167 std::forward<F>(task)();
168 return;
169 }
170 auto done = std::make_shared<std::promise<void>>();
171 std::future<void> finished = done->get_future();
172 post([task = std::forward<F>(task), done]() mutable {
173 task();
174 done->set_value();
175 });
176 try {
177 finished.get();
178 } catch (const std::future_error&) { // NOLINT(bugprone-empty-catch)
179 // Dropped unrun, or it threw (logged by `post`): nothing is left to
180 // wait for either way.
181 }
182 }
183
184 /// @brief A handle that posts only while this loop exists.
185 /// @return The weak handle.
186 [[nodiscard]] Weak weak() const { return Weak{_impl}; }
187
188private:
189 template <class F>
190 static auto guarded(F&& task) {
191 return [task = std::forward<F>(task)]() mutable noexcept {
192 try {
193 task();
194 } catch (const std::exception& exc) {
195 ::morph::log::logError("[io-loop] task threw: {}", exc.what());
196 } catch (...) {
197 ::morph::log::logError("[io-loop] task threw an unknown exception");
198 }
199 };
200 }
201
202 std::shared_ptr<Impl> _impl;
203#ifndef MORPH_IO_LOOP_HOST_DRIVEN
204 std::thread _thread;
205#endif
206};
