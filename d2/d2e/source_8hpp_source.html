68/// @brief An `AppSource` that also hands the shell what it mounts screens with, once it delivered a bundle.
69///
70/// Opened, it holds a bridge over its backend, owned by the context's executor, and the verified bundle with its
71/// catalog. Everything it hands out is valid from a successful `ready` until `close`.
72class Source : public ui::AppSource {
73public:
74 /// @brief Where the screens' model calls go.
75 /// @return The dispatcher over the source's bridge.
76 /// @throws std::logic_error before a bundle was delivered.

79 /// @brief The catalog facts the screens load with.

83 /// @brief The screens the principal may open, in catalog order: each catalog entry that is available or public
84 /// and whose document the bundle holds.

87
88 /// @brief Whether the catalog marks @p screenId public: a screen served before sign-in, such as a login screen.
89 /// @param screenId The screen.

93 /// @brief Opens the source: the transport, the bridge and the bundle, verified.
94 /// @param context The running frontend's context.
95 /// @param ready Called once, on the owner, unless `close` comes first.

97
98 /// @brief Refused: a switch between backends is the hosting layer's (spec 6 §9).
99 /// @param backend Ignored.
100 /// @param done Called with the refusal.

102
103 /// @brief Drops a `ready` not yet delivered and releases the bridge, the transport and the bundle.
104 void close() final;
105
106 /// @brief Installs a sign-in (spec 6 §5): a session built from the current one with @p principal and @p token
107 /// replaced, carried by every later model call and UI-service request, then the catalog reloaded for it.
108 /// @param principal The principal the sign-in mutation answered.
109 /// @param token The session token it answered; empty when the source's server needs none, as in process.
110 /// @param done Called once, on the owner, once the catalog for the new session is current, with whether it
111 /// loaded; not called after `close`.
112 void signIn(std::string principal, std::string token, std::function<void(bool)> done);
113
114 /// @brief Ends the session: back to the principal before any sign-in, with no token, and the catalog reloaded
115 /// for it.
116 /// @param done As `signIn`'s.
117 void signOut(std::function<void(bool)> done);
118
119 /// @brief The lookup a `DocumentScreenHost` takes (`DocumentScreenHostOptions::documents`), over the loaded
120 /// bundle: each screen the source lists, loaded against the current catalog facts the first time it is
121 /// asked for after a load, so a shell of an application's own mounts with the same source.
122 /// @param onRefused Called with each document that does not load; may be empty.
123 /// @return The lookup, valid while the source lives; it answers null for a screen the source does not list or
124 /// whose document does not load.
125 [[nodiscard]] std::function<std::shared_ptr<ui::Document const>(std::string_view)> documents(
126 std::function<void(ui::Refusal const&)> onRefused = {});
127
128 /// @brief The bundle the catalog currently lists.
129 /// @return It; null before a bundle was delivered.
130 [[nodiscard]] ui::Bundle const* bundle() const noexcept { return _loaded ? &_loaded->bundle : nullptr; }
131
132 /// @brief The catalog for the current principal: its entries, what the principal may run, and the facts.
133 /// @return It; null before a bundle was delivered.
134 [[nodiscard]] ui::CatalogReply const* catalogReply() const noexcept {
135 return _loaded ? &_loaded->catalog : nullptr;
136 }
137
138 /// @brief The `schemas` document of every model the catalog names, by model type, or why it did not arrive: what
139 /// the explorer lists each model's actions from.
140 /// @return Them; empty before a bundle was delivered.
141 [[nodiscard]] ui::SchemasByModel const& schemasByModel() const noexcept { return _byModel; }
142
143 /// @brief The principal the source's session carries now.
144 /// @return It; empty for none.
145 [[nodiscard]] std::string const& principal() const noexcept { return _session.principal; }
146
147 /// @brief The file side channel (`files/1`) over the source's exchange, carrying its session: what a screen's
148 /// `upload` and `download` move bytes through.
149 /// @return It, or nothing before the source is open.
150 [[nodiscard]] std::optional<ui::FileTransfer> files() const;
151
152 /// @brief One step of development reload (spec 5 §12): sends `ui-hello`, and when the manifest digest changed,
153 /// fetches the changed bundle and the catalog for it. Nothing is sent while a poll is still in flight.
154 /// @param changed Called on the owner once the changed bundle is current, with the ids of the documents it
155 /// changed; not called for an unchanged digest, a failed poll, or after `close`.
156 void poll(std::function<void(std::vector<std::string> changed)> changed);
157
158 /// @brief Adds a listener called on the owner after each successful connect, the first included. On a reconnect
159 /// the source first sends `ui-hello` again, so the new connection negotiates events, then calls the
160 /// listeners.
161 /// @param listener Called on the owner; it must stay callable while the source is open.
162 void onConnected(std::function<void()> const& listener) final;
163
164 /// @brief Adds a listener for the events the server sends: what a shell hands to `Session::deliverEvent`.
165 /// @param listener Called on the owner, once per event, in the order they arrived.
166 void onEvent(std::function<void(wire::EventBody const&)> listener);
167
168 /// @brief What the session's `SessionOptions::watcher` sends the cache's watch changes through: `watcherOver` the
169 /// source's exchange, stamping the session current at each send. Empty for a source whose transport
170 /// carries no server events, such as an in-process one.
171 /// @return The watcher, or empty.
172 [[nodiscard]] ui::Watcher watcher();
173
174 /// @brief The principal before any sign-in, which `signOut` returns to.
175 /// @return It; empty for none.
176 [[nodiscard]] virtual std::string signedOutPrincipal() const { return {}; }

180 struct Transport {

190 /// @brief Makes the transport on @p context.

194
195 /// @brief How long `open` waits for the bundle before it reports the source unreachable; zero waits forever.
196 /// @return The bound.
197 [[nodiscard]] virtual std::chrono::milliseconds openTimeout() const { return std::chrono::milliseconds{0}; }
198
199 /// @brief Releases what `connect` made beside the backend; called by `close`, after the bridge is gone.
200 virtual void disconnect() {}
201
202 /// @brief Fetches the `{actionType: schema}` document of one model type, which a `form`, `collection` or
203 /// `wizard` node is built from. The default sends a `schemas` envelope over the source's exchange.
204 /// @param modelType The model type id.
205 /// @param done Called once with the document's JSON text, or why there is none.
206 virtual void fetchSchemas(std::string const& modelType,
207 std::function<void(std::expected<std::string, std::string>)> done);
208
209 /// @brief The session every request carries, for a source to fill before `open` sends anything.
210 /// @return It.
211 [[nodiscard]] session::Context& requestSession() noexcept { return _session; }

213private:
214 // The documents `documents()` loaded, by screen id, each with the text and the catalog load it was loaded for.
215 struct LoadedDocument {
216 std::string text;
217 std::uint64_t catalogLoad = 0;
218 std::shared_ptr<ui::Document const> document;
219 };
220 std::map<std::string, LoadedDocument, std::less<>> _documents;
221 // Counts the catalogs loaded, so a document loaded against an older one is loaded again.
222 std::uint64_t _catalogLoads = 0;
223 session::Context _session;
224 std::shared_ptr<bool> _alive;
225 std::unique_ptr<bridge::Bridge> _bridge;
226 std::unique_ptr<ui::BridgeDispatcher> _dispatcher;
227 // The offline queue in front of the dispatcher: what a screen's calls go through.
228 std::unique_ptr<ui::QueueingDispatcher> _queue;
229 // Retries the queue while it holds anything: the transports reconnect on their own and tell no one.
230 std::optional<reactive::TimerHandle> _replay;
231 ui::Exchange _exchange;
232 // Whether the open transport's server sends events and handles `watch`.
233 bool _serverEvents = false;
234 // Called after each connect, and with each event.
235 std::vector<std::function<void()>> _connectListeners;
236 std::vector<std::function<void(wire::EventBody const&)>> _eventListeners;
237 // A connect, on the owner: `ui-hello` again once the first bundle loaded, then the listeners.
238 void connected();
239 // Fetches the schemas of every model type the catalog names, then calls @p done; nothing after `close`.
240 void loadSchemas(std::shared_ptr<bool> const& alive, std::function<void()> done);
241 // The model types the loaded catalog names: its models, and the model of each action it has facts for.
242 [[nodiscard]] std::vector<std::string> modelTypes() const;
243 // Records each action's schema of one `{actionType: schema}` document; a document that does not parse adds none.
244 void recordSchemas(std::string const& document);
245 // The schema of each action, by action id, fetched for every model type the catalog names.
246 std::map<std::string, std::string, std::less<>> _schemas;
247 // Each model's schemas document as it arrived, or why it did not.
248 ui::SchemasByModel _byModel;
249 std::optional<ui::LoadedBundle> _loaded;
250 // The catalog the screens read through `catalog()`: replaced in place when a sign-in reloads it, so a session
251 // built over the first catalog sees the facts of the current one.
252 struct CurrentCatalog {
253 ui::Catalog facts;
254 std::map<std::string, std::string, std::less<>> schemas;
255 };
256 std::shared_ptr<CurrentCatalog> _current = std::make_shared<CurrentCatalog>();
257 // Verified files, kept across reloads: a reload fetches only what the new catalog adds.
258 std::shared_ptr<ui::IBundleStore> _store;
259 // Fetches the catalog and the bundle for the current session, then the schemas; `done` once they are current.
260 void reload(std::shared_ptr<bool> const& alive, std::function<void(std::expected<void, ui::ConnectError>)> done);
261 reactive::TimerHandle _timeout;
262 // Whether a development poll is in flight.
263 bool _polling = false;
264};
