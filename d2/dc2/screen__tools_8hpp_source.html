39/// @brief A tool name's limit, the strictest a provider's tool format is known to set.
40inline constexpr std::size_t kMaxToolName = 64;
41/// @brief The hash's length, when a name needs one.
42inline constexpr std::size_t kToolHashDigits = 8;
43
44/// @brief Whether a provider's tool format would take a name.
45/// @param name The name.
46/// @return Whether it would.
47[[nodiscard]] inline bool validToolName(std::string_view name) {
48 return !name.empty() && name.size() <= kMaxToolName && std::ranges::all_of(name, [](char const letter) {
49 return (letter >= 'a' && letter <= 'z') || (letter >= 'A' && letter <= 'Z') ||
50 (letter >= '0' && letter <= '9') || letter == '_' || letter == '-';
51 });
52}
53
54/// @brief A hash of @p text, FNV-1a: stable across runs, so a name a conversation has already used keeps its meaning.
55/// @param text The text.
56/// @return Its hash, in hexadecimal.
57[[nodiscard]] inline std::string toolHash(std::string_view text) {
58 std::uint32_t hash = 2166136261U;
59 for (char const letter : text) {
60 hash ^= static_cast<std::uint8_t>(letter);
61 hash *= 16777619U;
62 }
63 static constexpr std::array<char, 16> kDigits = {'0', '1', '2', '3', '4', '5', '6', '7',
64 '8', '9', 'a', 'b', 'c', 'd', 'e', 'f'};
65 std::string out(kToolHashDigits, '0');
66 for (std::size_t index = kToolHashDigits; index > 0; --index) {
67 out.at(index - 1) = kDigits.at(hash & 0xFU);
68 hash >>= 4U;
69 }
70 return out;
71}
72
73/// @brief A stem made valid and short enough to take a hash as its suffix.
74/// @param stem The name it starts from.
75/// @param identity What tells this tool from any other.
76/// @return The name.
77[[nodiscard]] inline std::string hashedToolName(std::string const& stem, std::string_view identity) {
78 std::string made = stem.substr(0, kMaxToolName - kToolHashDigits - 1);
79 std::ranges::replace_if(made, [](char const letter) { return !validToolName(std::string_view{&letter, 1}); }, '_');
80 return made + '_' + toolHash(identity);
81}

90// NOLINTBEGIN(misc-no-recursion): view nodes nest; parseJson bounds the depth at 64.
91// Collects what a document names, resolving each alias through the enclosing scopes as loading did.
92class ScreenWalk {
93public:
94 explicit ScreenWalk(Document const& document) {
95 enter(document.scope);
96 if (document.view) {
97 node(*document.view);
98 }
99 }
100
101 [[nodiscard]] std::vector<NamedAction> const& named() const noexcept { return _named; }
102
103private:
104 void enter(ScopeDecl const& scope) {
105 for (ModelDecl const& model : scope.models) {
106 _aliases[model.name].push_back(&model);
107 }
108 for (QueryDecl const& query : scope.queries) {
109 add(query.model, query.action, false);
110 }
111 for (MutationDecl const& mutation : scope.mutations) {
112 add(mutation.model, mutation.action, true);
113 }
114 }
115
116 void leave(ScopeDecl const& scope) {
117 for (ModelDecl const& model : scope.models) {
118 _aliases.at(model.name).pop_back();
119 }
120 }
121
122 void node(DocNode const& view) {
123 if (view.scope) {
124 enter(*view.scope);
125 }
126 // A model-bound node such as a form submits its action, so it changes data.
127 if (std::string_view const model = view.name("model"); !model.empty()) {
128 add(model, view.name("action"), true);
129 }
130 for (auto const& [key, children] : view.children) {
131 static_cast<void>(key);
132 for (DocNodePtr const& child : children) {
133 node(*child);
134 }
135 }
136 for (auto const& [key, parts] : view.parts) {
137 static_cast<void>(key);
138 for (DocNodePtr const& part : parts) {
139 node(*part);
140 }
141 }
142 if (view.scope) {
143 leave(*view.scope);
144 }
145 }
146
147 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): both come straight from the document's own fields, at the one caller.
148 void add(std::string_view alias, std::string_view action, bool changes) {
149 auto const found = _aliases.find(alias);
150 if (action.empty() || found == _aliases.end() || found->second.empty()) {
151 return;
152 }
153 ModelDecl const* const model = found->second.back();
154 auto const seen = std::ranges::find_if(
155 _named, [&](NamedAction const& named) { return named.model == model && named.action == action; });
156 if (seen != _named.end()) {
157 seen->changes = seen->changes || changes;
158 return;
159 }
160 _named.push_back(NamedAction{.model = model, .action = std::string{action}, .changes = changes});
161 }
162
163 std::map<std::string, std::vector<ModelDecl const*>, std::less<>> _aliases;
164 std::vector<NamedAction> _named;
165};
166// NOLINTEND(misc-no-recursion)

168/// @brief An action's schema as a tool's, always an object.
169/// @param schemas Each action's schema, by action id.
170/// @param action The action.
171/// @return The schema.
172[[nodiscard]] inline Json toolSchema(std::map<std::string, Json, std::less<>> const& schemas,
173 std::string const& action) {
174 auto const found = schemas.find(action);
175 Json schema = found == schemas.end() ? Json(Json::object_t{}) : found->second;
176 if (!std::holds_alternative<Json::object_t>(schema.data)) {
177 schema = Json::object_t{};
178 }
179 std::get<Json::object_t>(schema.data).insert_or_assign("type", "object");
180 return schema;
181}
182
183/// @brief What a tool says it does.
184/// @param named What the document names.
185/// @param schema The tool's schema.
186/// @return The description.
187[[nodiscard]] inline std::string toolDescription(NamedAction const& named, Json const& schema) {
188 std::string description = (named.changes ? "Changes data: runs " : "Reads data: runs ") + named.action + " on " +
189 named.model->name + ".";
190 auto const& object = std::get<Json::object_t>(schema.data);
191 if (auto const* const found = object.find("description"); found != object.end()) {
192 if (auto const* const text = std::get_if<std::string>(&found->second.data)) {
193 description += ' ' + *text;
194 }
195 }
196 return description;
197}

201/// @brief The tools of one screen: each query, mutation and model-bound node the document names, at any scope, whose
202/// alias the context bound and whose action the principal may run.
203///
204/// A tool is named `<alias>__<action>`. A name a provider's tool format would refuse, or one a tool already took, gets
205/// a shortened stem and a hash of its scope, alias and action, so it stays valid and unique. The tools are sorted by
206/// name, and an action named more than once on one alias gives one tool, offered as changing data if any naming runs
207/// it as a mutation. A credential action gets no tool: a model should not be handed a password to pass along.
208/// @param document The screen's document, loaded.
209/// @param models The aliases the client says it holds. One counts only if its scope, alias and type all match the
210/// document's declaration, so a context cannot point a tool at a model the screen does not declare.
211/// @param permitted The actions the principal may run.
212/// @param schemas Each action's schema, by action id.
213/// @param actions The catalog's action facts, when there are any.
214/// @return The tools.
215[[nodiscard]] inline std::vector<BoundTool> screenTools(
216 Document const& document, std::span<ModelRef const> models, std::set<std::string, std::less<>> const& permitted,
217 std::map<std::string, Json, std::less<>> const& schemas,
218 std::map<std::string, ActionInfo, std::less<>> const* actions = nullptr) {
219 struct Candidate {
220 std::string stem;
221 std::string identity;
222 detail::NamedAction const* named = nullptr;
223 ModelRef const* model = nullptr;
224 };
225 detail::ScreenWalk const walk{document};
226 std::vector<Candidate> candidates;
227 for (detail::NamedAction const& named : walk.named()) {
228 if (!permitted.contains(named.action)) {
229 continue;
230 }
231 if (actions != nullptr) {
232 if (auto const facts = actions->find(named.action); facts != actions->end() && facts->second.credential) {
233 continue;
234 }
235 }
236 auto const bound = std::ranges::find_if(models, [&](ModelRef const& model) {
237 return model.alias == named.model->name && model.path == named.model->path &&
238 model.modelType == named.model->type;
239 });
240 if (bound == models.end()) {
241 continue;
242 }
243 candidates.push_back(Candidate{.stem = named.model->name + "__" + named.action,
244 .identity = named.model->path + '/' + named.model->name + '/' + named.action,
245 .named = &named,
246 .model = &*bound});
247 }
248 // Sorted by a stable identity first, so which of two colliding tools keeps the plain name does not depend on
249 // order.
250 std::ranges::sort(candidates, [](Candidate const& left, Candidate const& right) {
251 return std::pair{left.stem, left.identity} < std::pair{right.stem, right.identity};
252 });
253 std::set<std::string, std::less<>> taken;
254 std::vector<BoundTool> tools;
255 tools.reserve(candidates.size());
256 for (Candidate const& candidate : candidates) {
257 std::string name = candidate.stem;
258 if (!detail::validToolName(name) || taken.contains(name)) {
259 name = detail::hashedToolName(candidate.stem, candidate.identity);
260 }
261 taken.insert(name);
262 Json schema = detail::toolSchema(schemas, candidate.named->action);
263 std::string description = detail::toolDescription(*candidate.named, schema);
264 tools.push_back(BoundTool{.spec = ToolSpec{.name = std::move(name),
265 .description = std::move(description),
266 .inputSchema = std::move(schema)},
267 .model = *candidate.model,
268 .action = candidate.named->action});
269 }
270 std::ranges::sort(tools, {}, [](BoundTool const& tool) { return tool.spec.name; });
271 return tools;
272}
