38/// @brief Why a runtime tour was refused or dropped.
39struct TourRefusal {
40 /// @brief The tour's id, as far as it could be read; empty when it could not.
41 std::string tourId;
42 /// @brief A JSON Pointer into the tour, such as `/steps/2/target`; empty for the whole tour.
43 std::string path;
44 /// @brief The reason, in one sentence, naming the step where there is one.
45 std::string reason;
46
47 /// @brief The refusal as one line.
48 /// @return The text "the tour `ID` at PATH: REASON", with the three members in place of the capitals.
49 [[nodiscard]] std::string message() const {
50 return "the tour `" + tourId + "` at " + (path.empty() ? std::string{"/"} : path) + ": " + reason;
51 }
52
53 /// @brief Memberwise equality.
54 /// @return Whether every member matches.
55 bool operator==(TourRefusal const&) const = default;
56};

78/// @brief The tours added while the server runs, beside the app document's own.
79///
80/// `add` and `replace` take one tour's JSON, the form of an entry of the app document's `tours` (`author::tourJson`
81/// writes it), and check it against the bundle served now exactly as the bundle's build checks the app document's
82/// tours, and that its `t` keys are in a catalogue of the bundle or among the server's texts. The bundle is read
83/// through the getter at every call; when its digest differs from the one last checked, every held tour is checked
84/// again first, and one that fails is dropped: logged at error and told to `setOnDropped`'s handler. Every change, a
85/// drop included, is told to `setOnChanged`'s handler after the registry's lock is released (`publishTourChanges`
86/// turns it into a change of the record `morph.tours`). The first change made while no handler is set is logged as a
87/// warning, so a server that never called `publishTourChanges` hears of it. A refusal names the runtime tour by its id
88/// and points into the submitted tour's JSON, never into the app document. The tours are kept in memory only. Safe to
89/// call from any thread.
90///
91/// The bundle getter, the server's texts (`ServedBundle::fieldTexts`) and the authoriser passed to `listFor` are
92/// called with the registry's lock held, so none of them may call back into the registry or hold a lock around a call
93/// into it. The two handlers are called after the lock is released.
94class MORPH_API TourRegistry {
95public:
96 /// @brief A registry with no tour.
97 /// @param bundle The bundle served now; called at every use, from any thread, so a development reload is seen.
98 explicit TourRegistry(std::function<std::shared_ptr<ServedBundle const>()> bundle);
99
100 /// @brief Adds a tour.
101 /// @param tour The tour's JSON text.
102 /// @return Nothing, or the refusal: a malformed tour, a name the bundle does not have, a text key nobody has,
103 /// an id the app document or another runtime tour uses, or a bundle with no app document.
104 std::expected<void, TourRefusal> add(std::string_view tour);
105
106 /// @brief Replaces a runtime tour with new content under the same id.
107 /// @param tour The tour's JSON text.
108 /// @return Nothing, or the refusal: `add`'s, or an id the registry does not hold.
109 std::expected<void, TourRefusal> replace(std::string_view tour);
110
111 /// @brief Removes a runtime tour.
112 /// @param tourId Its id.
113 /// @return Nothing, or the refusal of an id the registry does not hold.
114 std::expected<void, TourRefusal> remove(std::string_view tourId);
115
116 /// @brief The runtime tours a person may see, in registration order: those whose `screen` and every step's
117 /// `screen` the person may open (`screenAdmitted`). A tour is offered whole or not at all.
118 /// @param session The caller's session.
119 /// @param authorizer The server's authoriser.
120 /// @return The tours.
121 [[nodiscard]] ToursListed listFor(::morph::session::Context const& session,
122 ::morph::session::IAuthorizer const& authorizer);
123
124 /// @brief Checks every held tour against the bundle served now, dropping those that fail, whether or not its
125 /// digest changed: what an application calls right after it swaps the bundle.
126 /// @throws std::logic_error when the bundle getter answers no bundle.
127 void recheck();
128
129 /// @brief Sets what is told after every change.
130 /// @param onChanged The handler; empty tells nothing.
131 void setOnChanged(std::function<void()> onChanged);
132
133 /// @brief Sets what is told of each tour a bundle change dropped.
134 /// @param onDropped The handler, given the refusal; empty tells nothing.
135 void setOnDropped(std::function<void(TourRefusal const&)> onDropped);
136
137private:
138 struct Held {
139 std::string id;
140 std::uint64_t revision = 0;
141 std::string text;
142 std::vector<std::string> screens;
143 };
144
145 [[nodiscard]] static std::expected<Held, TourRefusal> checked(std::string_view tour, ServedBundle const& bundle);
146 // Checks every held tour against @p bundle and drops those that fail; answers the drops. Under the lock.
147 [[nodiscard]] std::vector<TourRefusal> recheckLocked(ServedBundle const& bundle);
148 // The bundle now, every held tour checked against it; drops into @p dropped. Under the lock.
149 [[nodiscard]] std::shared_ptr<ServedBundle const> currentLocked(std::vector<TourRefusal>& dropped);
150 void tell(bool changed, std::vector<TourRefusal> const& dropped);
151
152 std::function<std::shared_ptr<ServedBundle const>()> _bundle;
153 mutable std::mutex _mutex;
154 std::vector<Held> _tours;
155 std::string _checkedDigest;
156 std::uint64_t _revisions = 0;
157 bool _warnedUnpublished = false;
158 std::function<void()> _onChanged;
159 std::function<void(TourRefusal const&)> _onDropped;
160};
