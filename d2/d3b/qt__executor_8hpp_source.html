11/// @brief `IExecutor` implementation that posts tasks to the Qt event loop.
12///
13/// Uses `QMetaObject::invokeMethod` with `Qt::QueuedConnection` so the callable
14/// always runs on the thread that owns `QCoreApplication` (typically the GUI thread).
15///
16/// Thread-safe: `post()` may be called from any thread. No `QObject` subclass is
17/// required by the caller.
18class QtExecutor : public ::morph::exec::IExecutor {
19public:
20 /// @brief Posts @p fn to the Qt event loop for execution on the GUI thread.
21 ///
22 /// Returns immediately. @p fn is invoked asynchronously once the event loop
23 /// processes the queued event.
24 ///
25 /// @param fn Callable to execute on the GUI thread.
26 void post(std::function<void()> fn) override {
27 QMetaObject::invokeMethod(QCoreApplication::instance(), std::move(fn), Qt::QueuedConnection);
28 }
29};
