11/// @brief `IExecutor` implementation that posts tasks to a Qt event loop.
12///
13/// Uses `QMetaObject::invokeMethod` with `Qt::QueuedConnection` so the callable
14/// always runs on the thread that owns the configured context object — by
15/// default `QCoreApplication::instance()`, i.e. the GUI thread.
16///
17/// Thread-safe: `post()` may be called from any thread. No `QObject` subclass is
18/// required by the caller.
19class QtExecutor : public ::morph::exec::IExecutor {
20public:
21 /// @brief Constructs an executor that posts tasks to @p context's thread.
22 ///
23 /// @param context `QObject` whose owning thread's event loop runs posted
24 /// tasks; `QMetaObject::invokeMethod` dispatches to whichever thread
25 /// `context->thread()` reports at the time each task is posted, so tasks
26 /// posted after `context` is moved to a different thread run there.
27 /// Defaults to `QCoreApplication::instance()`, preserving the previous
28 /// GUI-thread-only behaviour. Passing `nullptr` (e.g. when constructed
29 /// before `QCoreApplication` exists) makes `post()` a no-op, matching
30 /// `QMetaObject::invokeMethod`'s own handling of a null target.
31 explicit QtExecutor(QObject* context = QCoreApplication::instance()) : _context{context} {}
32
33 /// @brief Posts @p fn to the Qt event loop for execution on the context's thread.
34 ///
35 /// Returns immediately. @p fn is invoked asynchronously once the event loop
36 /// processes the queued event.
37 ///
38 /// @param fn Callable to execute on the context object's thread.
39 void post(std::function<void()> fn) override {
40 QMetaObject::invokeMethod(_context, std::move(fn), Qt::QueuedConnection);
41 }
42
43private:
44 QObject* _context;
45};
