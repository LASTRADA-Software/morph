5/// @file forms/i18n.hpp
6/// @brief Stable, schema-derived message-key vocabulary for renderer i18n.
7///
8/// A renderer that wants to translate a piece of GUI display text (a field's
9/// label/help/placeholder, a layout group's title, a cross-field rule's
10/// violation message, a wizard's or app-shell's title) needs a **stable
11/// catalog key** to look that text up under. This header is the single
12/// source of truth for how such a key is *derived* from identifiers the
13/// schema (or the `actionType` label a renderer already has, per
14/// `docs/spec/forms/forms.md`'s renderer contract) already carries:
15/// `morph::model::ActionTraits<A>::typeId()`, a reflected wire field name, or
16/// a 0-based index into a schema array (`x-layout.groups`, `x-rules`, a
17/// wizard's steps, an app's menu).
18///
19/// No key derived here is written into the schema — a renderer computes it
20/// itself from data it already has, so the common (no-override) case adds
21/// zero bytes to any schema. See `morph::render::resolveText`
22/// (`render/i18n.hpp`) for how a derived key is looked up against a
23/// host-supplied catalog, with the schema's authored literal text as the
24/// fallback on a miss.
25

34/// @brief Which of a field's three translatable texts a message key names.
35enum class FieldSlot : std::uint8_t {
36 Label, ///< The field's display label (`<stem>.label`).
37 Help, ///< The field's help/description text (`<stem>.help`).
38 Placeholder ///< The field's empty-state placeholder text (`<stem>.placeholder`).
39};
40
41/// @brief The catalog-key suffix for @p slot.
42/// @param slot The field text slot.
43/// @return `"label"`, `"help"`, or `"placeholder"`.
44[[nodiscard]] constexpr std::string_view fieldSlotName(FieldSlot slot) noexcept {
45 switch (slot) {
46 case FieldSlot::Label:
47 return "label";
48 case FieldSlot::Help:
49 return "help";
50 case FieldSlot::Placeholder:
51 return "placeholder";
52 default:
53 return "label";
54 }
55}
56
57/// @brief Appends `.<slotName>` to a message-key stem.
58/// @param stem A message-key stem: a derived `<actionTypeId>.<wireField>` or
59/// a host-declared override (a field's `x-i18nKey`).
60/// @param slot Which field text this key names.
61/// @return `"<stem>.<slotName>"`.
62[[nodiscard]] inline std::string withSlot(std::string_view stem, FieldSlot slot) {
63 std::string key{stem};
64 key += '.';
65 key += fieldSlotName(slot);
66 return key;
67}
68
69/// @brief The derived message-key stem for a field: `<actionTypeId>.<wireField>`.
70/// @param actionTypeId `ActionTraits<A>::typeId()` of the action the field belongs to.
71/// @param wireField The field's wire key (its reflected member name).
72/// @return The stem, before the `.label` / `.help` / `.placeholder` suffix.
73[[nodiscard]] inline std::string fieldKeyStem(std::string_view actionTypeId, std::string_view wireField) {
74 std::string stem{actionTypeId};
75 stem += '.';
76 stem += wireField;
77 return stem;
78}
79
80/// @brief The fully-derived message key for one field text slot.
81/// @param actionTypeId `ActionTraits<A>::typeId()` of the action the field belongs to.
82/// @param wireField The field's wire key.
83/// @param slot Which of the field's texts this key names.
84/// @return `"<actionTypeId>.<wireField>.<slot>"`.
85[[nodiscard]] inline std::string fieldKey(std::string_view actionTypeId, std::string_view wireField, FieldSlot slot) {
86 return withSlot(fieldKeyStem(actionTypeId, wireField), slot);
87}
88
89/// @brief The explicit override key for one field text slot, when the field
90/// declares an `x-i18nKey` stem override.
91///
92/// A single `FieldMeta::i18nKey` override (see `gui_field_metadata.md`)
93/// replaces the derived `<actionTypeId>.<wireField>` stem for *all three* of
94/// a field's text slots; the `.label` / `.help` / `.placeholder` suffix is
95/// still appended per slot, exactly as it is for the derived key.
96/// @param i18nKeyOverride The field's declared `FieldMeta::i18nKey` (empty = no override).
97/// @param slot Which of the field's texts this key names.
98/// @return `std::nullopt` when @p i18nKeyOverride is empty; otherwise `"<i18nKeyOverride>.<slot>"`.
99[[nodiscard]] inline std::optional<std::string> explicitFieldKey(std::string_view i18nKeyOverride, FieldSlot slot) {
100 if (i18nKeyOverride.empty()) {
101 return std::nullopt;
102 }
103 return withSlot(i18nKeyOverride, slot);
104}

106/// @brief The derived message key for a layout group's title.
107/// @param actionTypeId `ActionTraits<A>::typeId()` of the action the group belongs to.
108/// @param groupIndex The group's 0-based index into `x-layout.groups`.
109/// @return `"<actionTypeId>.group.<groupIndex>"`.
110[[nodiscard]] inline std::string groupKey(std::string_view actionTypeId, std::size_t groupIndex) {
111 std::string key{actionTypeId};
