3/// @file
4/// @brief Structural fingerprint of an action payload's JSON shape.
5///
6/// The journal records an action's request as JSON and replays it, later,
7/// through `ActionTraits<A>::fromJson` — a *lenient* decode that ignores
8/// unknown keys and default-constructs absent ones. That leniency is deliberate
9/// (it is what makes an additive field a non-event for every already-deployed
10/// reader), but on the replay path it also means a **renamed** field silently
11/// becomes "key absent + unknown key ignored": the entry decodes, the model
12/// reconstructs, and the value that was actually recorded is gone. Nothing in
13/// the entry said which shape wrote it, so nothing could notice.
14///
15/// This header supplies the missing discriminator: a short, stable string
16/// derived from the payload struct's reflected shape, cheap enough to stamp on
17/// every recorded entry and compare on every replayed one. See
18/// `docs/spec/journal/journal.md`, "Payload schema fingerprint".
19

34/// @brief Version of the fingerprint *algorithm*, not of any one payload.
35///
36/// Stamped as the prefix of every fingerprint (`"1:…"`) so a future build that
37/// computes fingerprints differently can still recognise — and reject, or
38/// re-derive — a fingerprint written under an earlier scheme, instead of
39/// mistaking a scheme change for a payload change. Bumped only when
40/// `detail::payloadShape` changes what it emits for an unchanged struct.
41inline constexpr std::uint32_t kPayloadFingerprintScheme = 1;

45/// @brief Recursion limit for `payloadShape`.
46///
47/// A payload nested deeper than this contributes the opaque tag rather than its
48/// structure — the fingerprint stays finite for a self-referential type (a
49/// `struct Node { std::vector<Node> kids; }` is a legal Glaze payload) without
50/// needing cycle detection. Eight levels is far past any action struct in this
51/// repository or its examples.
52inline constexpr int kPayloadShapeMaxDepth = 8;
53
54/// @brief Tag emitted for a type whose JSON shape this header does not
55/// decompose — a type with a custom Glaze codec (`glz::meta` naming a
56/// value rather than an object), a function pointer, a variant.
57///
58/// Deliberately carries no type name: `glz::name_v` falls back to a
59/// `__PRETTY_FUNCTION__`/`__FUNCSIG__`-derived spelling that differs between
60/// compilers, and a fingerprint that changes when the *compiler* changes would
61/// make every journal unreadable by a peer build. See the spec's "What the
62/// fingerprint does not catch".
63inline constexpr std::string_view kOpaqueShapeTag = "x";
64
65template <typename T>
66[[nodiscard]] std::string payloadShape(int depth);

78/// @brief Renders a reflected aggregate as `(key:shape,key:shape,…)`, with the
79/// members sorted by key.
80///
81/// **Sorted, so member reordering is not a schema change.** JSON objects are
82/// unordered and the decode matches by name, so moving a field within the
83/// struct changes nothing about which bytes decode to which member; making the
84/// fingerprint order-sensitive would turn a purely cosmetic edit into a replay
85/// break for every retained journal. Renaming, adding, or removing a member
86/// does change the rendering, which is the whole point.
87///
88/// @tparam T Reflected aggregate (or `glz::meta`-declared object) type.
89/// @param depth Current recursion depth; members are rendered at `depth + 1`.
90/// @return The parenthesised, comma-joined, key-sorted member rendering.
91template <typename T>
92[[nodiscard]] std::string payloadShapeObject(int depth) {
93 using V = std::remove_cvref_t<T>;
94 constexpr std::size_t kMemberCount = glz::reflect<V>::size;
95 std::vector<std::string> members;
96 members.reserve(kMemberCount);
97 [&]<std::size_t... I>(std::index_sequence<I...>) {
98 (members.emplace_back(std::string{glz::reflect<V>::keys[I]} + ':' +
99 payloadShape<glz::refl_t<V, I>>(depth + 1)),
100 ...);
101 }(std::make_index_sequence<kMemberCount>{});
102 std::ranges::sort(members);
103 std::string out{"("};
104 bool first = true;
105 for (const auto& member : members) {
106 if (!first) {
107 out += ',';
108 }
109 first = false;
110 out += member;
111 }
112 out += ')';
113 return out;
114}

116/// @brief Renders @p T's JSON shape as a compact, compiler-independent string.
117///
118/// The grammar, in full:
119///
120/// | C++ type | Rendering |
121/// |---|---|
122/// | `bool` | `b` |
123/// | `char` | `c` (its signedness is platform-dependent; the tag is not) |
124/// | signed integral | `i` + `sizeof` (e.g. `i4`) |
125/// | unsigned integral | `u` + `sizeof` |
126/// | enumeration | `e` + `sizeof` of the underlying type |
127/// | floating point | `f` + `sizeof` |
128/// | string-like | `s` |
129/// | `std::optional<T>` | `?` + T's rendering |
130/// | map | `{` key `>` mapped `}` |
131/// | other range | `[` element `]` |
132/// | reflected object | `(` sorted `key:shape` list `)` |
133/// | anything else | `x` (see `kOpaqueShapeTag`) |
134///
135/// Every tag is derived from `std::` type traits or from Glaze's *reflected
136/// key strings*, never from a compiler-spelled type name, so two builds of the
137/// same sources on different compilers, standard libraries, or platforms
138/// render an unchanged struct identically.
139///
140/// @tparam T Type to render.
141/// @param depth Current recursion depth; at `kPayloadShapeMaxDepth` the
142/// opaque tag is emitted instead of recursing further.
143/// @return The rendering described above.
144template <typename T>
145[[nodiscard]] std::string payloadShape(int depth) {
146 using U = std::remove_cvref_t<T>;
147 if (depth > kPayloadShapeMaxDepth) {
148 return std::string{kOpaqueShapeTag};
149 }
150 if constexpr (std::is_same_v<U, bool>) {
151 return "b";
152 } else if constexpr (std::is_same_v<U, char>) {
153 return "c";
154 } else if constexpr (std::is_enum_v<U>) {
155 return "e" + std::to_string(sizeof(std::underlying_type_t<U>));
156 } else if constexpr (std::is_integral_v<U>) {
157 return (std::is_signed_v<U> ? std::string{"i"} : std::string{"u"}) + std::to_string(sizeof(U));
158 } else if constexpr (std::is_floating_point_v<U>) {
159 return "f" + std::to_string(sizeof(U));
160 } else if constexpr (glz::str_t<U> || glz::string_view_t<U>) {
161 return "s";
162 } else if constexpr (IsStdOptional<U>::value) {
163 return "?" + payloadShape<typename U::value_type>(depth + 1);
164 } else if constexpr (glz::readable_map_t<U>) {
165 return "{" + payloadShape<typename U::key_type>(depth + 1) + ">" +
166 payloadShape<typename U::mapped_type>(depth + 1) + "}";
167 } else if constexpr (glz::range<U>) {
168 return "[" + payloadShape<glz::range_value_t<U>>(depth + 1) + "]";
169 } else if constexpr (glz::reflectable<U> || glz::glaze_object_t<U>) {
170 return payloadShapeObject<U>(depth);
171 } else {
172 return std::string{kOpaqueShapeTag};
173 }
174}

176/// @brief FNV-1a over @p bytes, 64-bit.
177///
178/// Chosen for being three lines of dependency-free, byte-exact-on-every-platform
179/// arithmetic — this hash is a *discriminator*, not a security primitive, and
180/// nothing downstream must be able to trust it against an adversary who can
181/// also rewrite the journal line it is stamped on.
182/// @param bytes Input to hash.
183/// @return The 64-bit digest.
184[[nodiscard]] inline std::uint64_t fnv1a64(std::string_view bytes) noexcept {
185 constexpr std::uint64_t kOffsetBasis = 14695981039346656037ULL;
186 constexpr std::uint64_t kPrime = 1099511628211ULL;
187 std::uint64_t hash = kOffsetBasis;
188 for (const char byte : bytes) {
189 hash ^= static_cast<std::uint64_t>(static_cast<unsigned char>(byte));
190 hash *= kPrime;
191 }
192 return hash;
193}

195/// @brief Renders @p value as exactly 16 lowercase hex digits.
196/// @param value Digest to render.
197/// @return The zero-padded hex string.
198[[nodiscard]] inline std::string hex64(std::uint64_t value) {
199 constexpr std::string_view kDigits = "0123456789abcdef";
200 std::string out(16, '0');
201 for (std::size_t i = 0; i < 16; ++i) {
202 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) — fixed 16-char buffer
203 out[15 - i] = kDigits[(value >> (4 * i)) & 0xFULL];
204 }
205 return out;
206}

210/// @brief Returns the human-readable shape rendering behind
211/// `payloadFingerprint<A>()`.
212///
213/// Exposed because a fingerprint mismatch is otherwise two opaque hex strings:
214/// with this, a diagnostic (or a developer at a prompt) can print both shapes
215/// and see *which* member moved. Not stamped on entries — it is unbounded in
216/// length, whereas the fingerprint is 18 bytes.
217/// @tparam A Action payload type.
218/// @return The shape rendering; see `detail::payloadShape` for the grammar.
219template <typename A>
220[[nodiscard]] inline const std::string& payloadShapeString() {
221 static const std::string kShape = detail::payloadShape<A>(0);
222 return kShape;
223}

225/// @brief Returns `A`'s payload fingerprint: `"<scheme>:<16 hex digits>"`.
226///
227/// Two payload types fingerprint identically exactly when
228/// `payloadShapeString()` renders them identically — i.e. when they have the
229/// same member names, at the same nesting positions, with the same JSON-shape
230/// tags. Computed once per type per process (a function-local static) and
231/// returned by reference; stamping it on a journal entry is a string copy of
232/// 18 bytes.
233///
234/// @tparam A Action payload type.
235/// @return Reference to the process-lifetime fingerprint string for `A`.
236template <typename A>
237[[nodiscard]] inline const std::string& payloadFingerprint() {
238 static const std::string kFingerprint = [] {
239 return std::to_string(kPayloadFingerprintScheme) + ':' +
240 detail::hex64(detail::fnv1a64(payloadShapeString<A>()));
241 }();
242 return kFingerprint;
243}
