22/// @brief Bounds on what chats may consume.
23struct ChatLimits {
24 /// @brief Tool calls one turn may make, malformed ones included. A record's contents reach the model and could
25 /// steer it, so this is also what bounds the harm one turn can do.
26 std::size_t maxToolCallsPerTurn = 16;
27 /// @brief Turns one principal may have running.
28 std::size_t maxTurnsPerPrincipal = 2;
29 /// @brief Turns one connection may have running.
30 std::size_t maxTurnsPerConnection = 1;
31 /// @brief Conversations kept per principal; past it the least recently used is dropped.
32 std::size_t maxConversationsPerPrincipal = 32;
33 /// @brief Messages kept per conversation; past it the oldest are dropped.
34 std::size_t maxMessagesPerConversation = 200;
35 /// @brief The longest text one turn may carry.
36 std::size_t maxTextBytes = std::size_t{64} * 1024;
37 /// @brief The most of one tool's result the model is given; past it the result is cut, and the model told.
38 std::size_t maxToolResultBytes = std::size_t{64} * 1024;
39 /// @brief The most one conversation holds; past it the oldest messages are dropped. Without it, one large result
40 /// kept in a conversation would make every later request on that record too large.
41 std::size_t maxConversationBytes = std::size_t{1024} * 1024;
42 /// @brief How long an untouched conversation is kept.
43 std::chrono::minutes idleRetention{std::chrono::hours{8}};
44};

61/// @brief The conversations a server is holding, in memory, bounded by `ChatLimits`.
62///
63/// Not thread-safe: `ChatService` guards it. Nothing survives a restart, and the journal keeps what actually changed.
64class ConversationStore {
65public:
66 /// @brief An empty store.
67 /// @param limits Its bounds.
68 explicit ConversationStore(ChatLimits const& limits) : _limits{limits} {}
69
70 /// @brief A conversation's messages, oldest first. Reading it counts as using it.
71 /// @param key The conversation.
72 /// @param now The current time.
73 /// @return Its messages; none when it does not exist or has been idle too long.
74 [[nodiscard]] std::vector<Message> history(ConversationKey const& key, std::chrono::steady_clock::time_point now) {
75 auto const found = _conversations.find(key);
76 if (found == _conversations.end()) {
77 return {};
78 }
79 if (now - found->second.touched > _limits.idleRetention) {
80 _conversations.erase(found);
81 return {};
82 }
83 found->second.touched = now;
84 return found->second.messages;
85 }
86
87 /// @brief Adds a message, then applies every bound.
88 /// @param key The conversation, made if it does not exist.
89 /// @param message The message.
90 /// @param now The current time.
91 void append(ConversationKey const& key, Message message, std::chrono::steady_clock::time_point now) {
92 std::erase_if(_conversations,
93 [&](auto const& held) { return now - held.second.touched > _limits.idleRetention; });
94 Conversation& conversation = _conversations[key];
95 conversation.touched = now;
96 conversation.messages.push_back(std::move(message));
97 trim(conversation.messages);
98 evictFor(key);
99 }
100
101 /// @brief Ends every conversation of @p principal.
102 /// @param principal The principal.
103 void endAll(std::string_view principal) {
104 std::erase_if(_conversations, [&](auto const& held) { return held.first.principal == principal; });
105 }
106
107 /// @brief The conversations held.
108 /// @return Their number.
109 [[nodiscard]] std::size_t size() const noexcept { return _conversations.size(); }
110
111private:
112 struct Conversation {
113 std::vector<Message> messages;
114 std::chrono::steady_clock::time_point touched;
115 };
116
117 [[nodiscard]] static std::size_t bytesOf(Message const& message) {
118 std::size_t bytes = message.text.size() + message.opaque.size();
119 for (ToolCall const& call : message.calls) {
120 bytes += call.id.size() + call.name.size() + call.argumentsJson.size();
121 }
122 for (ToolResult const& result : message.results) {
123 bytes += result.callId.size() + result.text.size();
124 }
125 return bytes;
126 }
127
128 // Past a bound the oldest go, then whatever would leave the history starting with anything but something a person
129 // said: a provider refuses a history that opens with a reply, or with a result whose call is gone.
130 void trim(std::vector<Message>& messages) const {
131 std::size_t bytes = 0;
132 for (Message const& message : messages) {
133 bytes += bytesOf(message);
134 }
135 std::size_t drop = 0;
136 while (drop < messages.size() &&
137 (messages.size() - drop > _limits.maxMessagesPerConversation || bytes > _limits.maxConversationBytes)) {
138 bytes -= bytesOf(messages.at(drop));
139 ++drop;
140 }
141 if (drop == 0) {
142 return;
143 }
144 messages.erase(messages.begin(), messages.begin() + static_cast<std::ptrdiff_t>(drop));
145 auto const start = std::ranges::find_if(
146 messages, [](Message const& message) { return message.role == "user" && message.results.empty(); });
147 messages.erase(messages.begin(), start);
148 // A provider's opaque parts are valid only in the conversation that produced them, and this edited it.
149 for (Message& message : messages) {
150 message.opaque.clear();
151 }
152 }
153
154 // Past the bound, the principal's least recently used conversation goes, but never the one in use.
155 void evictFor(ConversationKey const& key) {
156 std::size_t held = 0;
157 auto oldest = _conversations.end();
158 for (auto current = _conversations.begin(); current != _conversations.end(); ++current) {
159 if (current->first.principal != key.principal) {
160 continue;
161 }
162 ++held;
163 if (current->first != key &&
164 (oldest == _conversations.end() || current->second.touched < oldest->second.touched)) {
165 oldest = current;
166 }
167 }
168 if (held > _limits.maxConversationsPerPrincipal && oldest != _conversations.end()) {
169 _conversations.erase(oldest);
170 }
171 }
172
173 ChatLimits _limits;
174 std::map<ConversationKey, Conversation> _conversations;
175};
