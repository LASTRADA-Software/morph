5/// @file table/edits.hpp
6/// @brief Committing edited cells: one row's edits in order, stale and error
7/// state per cell, and the row patched with the server's reply.
8///
9/// A commit runs an injected function — the table's mutation — and hands it a
10/// completion. A success carries the row as the server returned it, which
11/// replaces the row through a `RowPatcher`; a failure carries the field error,
12/// which the cell shows. Edits to one row run one at a time, in the order they
13/// were committed, so a later edit never overtakes an earlier one. A cell is
14/// stale from its commit until its edit settles.
15///
16/// With an engine attached, a row holds its place while it has an edit in
17/// flight (`Engine::beginEdit`), which a `Deferred` reorder honours.
18///
19/// See `docs/spec/table/engine.md`.
20

57/// @brief Settles one commit, on the owner thread: the row as the server
58/// returned it, or the field error.
59using EditDone = std::function<void(std::expected<std::vector<Cell>, FieldError>)>;

61/// @brief Runs a commit: the table's mutation for @p request, calling @p done
62/// once with its outcome.
63using EditCommit = std::function<void(EditRequest const& request, EditDone done)>;

65/// @brief Commits cell edits row by row, in order.
66///
67/// Owner-affine. A completion that arrives after the `CellEdits` is destroyed
68/// is ignored, and destroying it ends the engine edit of every row it still
69/// has in flight. The commit function, the patcher and the engine's handlers
70/// may commit again, or destroy this object, from inside a call it makes.
71class CellEdits {
72public:
73 /// @brief Builds the committer.
74 /// @param patcher Replaces a row with the server's reply; must outlive this object.
75 /// @param commit Runs the table's mutation.
76 /// @param engine Optional: the engine whose rows hold their place while
77 /// an edit is in flight; must outlive this object.
78 CellEdits(RowPatcher& patcher, EditCommit commit, Engine* engine = nullptr)
79 : _patcher{&patcher}, _commit{std::move(commit)}, _engine{engine} {}
80
81 CellEdits(CellEdits const&) = delete;
82 CellEdits& operator=(CellEdits const&) = delete;
83 CellEdits(CellEdits&&) = delete;
84 CellEdits& operator=(CellEdits&&) = delete;
85
86 /// @brief Ends the engine edit of every row with an edit in flight or
87 /// settling: its completion will be ignored, so nothing else would.
88 // NOLINTNEXTLINE(bugprone-exception-escape) -- a handler that throws from here terminates, as from any destructor
89 ~CellEdits() {
90 if (_engine == nullptr) {
91 return;
92 }
93 std::vector<RowId> rows{_ending.begin(), _ending.end()};
94 for (auto const& [row, queue] : _queues) {
95 rows.push_back(row);
96 }
97 _queues.clear();
98 _ending.clear();
99 for (auto const& row : rows) {
100 _engine->endEdit(row);
101 }
102 }
103
104 /// @brief Commits an edit: now, or after the row's earlier edits settle.
105 /// @param request The edit.
106 void commit(EditRequest request) {
107 auto& queue = _queues[request.row];
108 queue.push_back(std::move(request));
109 if (queue.size() == 1) {
110 start(queue.front().row);
111 }
112 }
113
114 /// @brief Whether a cell has an edit committed and not yet settled.
115 /// @param row The row's key.
116 /// @param column The column's id.
117 /// @return `true` while stale.
118 [[nodiscard]] bool stale(RowId const& row, std::string_view column) const {
119 auto const found = _queues.find(row);
120 if (found == _queues.end()) {
121 return false;
122 }
123 return std::ranges::any_of(found->second,
124 [&](EditRequest const& request) { return request.column == column; });
125 }
126
127 /// @brief The error a cell shows, until its next edit succeeds.
128 /// @param row The row's key.
129 /// @param column The column's id.
130 /// @return The message, or nothing.
131 [[nodiscard]] std::optional<std::string> error(RowId const& row, std::string_view column) const {
132 auto const found = _errors.find(row);
133 if (found == _errors.end()) {
134 return std::nullopt;
135 }
136 auto const cell = found->second.find(column);
137 return cell == found->second.end() ? std::nullopt : std::optional{cell->second};
138 }
139
140 /// @brief Edits committed and not yet settled, across rows.
141 /// @return The count.
142 [[nodiscard]] std::size_t unsettled() const {
143 std::size_t out = 0;
144 for (auto const& [row, queue] : _queues) {
145 out += queue.size();
146 }
147 return out;
148 }
149
150private:
151 void start(RowId const& row) {
152 if (_engine != nullptr) {
153 _engine->beginEdit(row);
154 }
155 // A copy: a commit that settles at once pops the queue's front, and the
156 // commit may still read its request afterwards.
157 auto const request = _queues.at(row).front();
158 // The commit runs last: it may settle, commit again or destroy this object.
159 _commit(request, _alive.guard([this, row](std::expected<std::vector<Cell>, FieldError> outcome) {
160 settle(row, std::move(outcome));
161 }));
162 }
163
164 // The queue is brought up to date before the patcher is called: patching
165 // notifies the source's listeners at once, and their handlers may commit
166 // to the same row or destroy this object.
167 void settle(RowId const& row, std::expected<std::vector<Cell>, FieldError> outcome) {
168 auto const found = _queues.find(row);
169 if (found == _queues.end() || found->second.empty()) {
170 return;
171 }
172 auto const request = std::move(found->second.front());
173 found->second.pop_front();
174 bool const more = !found->second.empty();
175 if (!more) {
176 _queues.erase(found);
177 _ending.insert(row);
178 }
179 if (outcome) {
180 if (auto const errors = _errors.find(row); errors != _errors.end()) {
181 errors->second.erase(request.column);
182 }
183 auto const alive = _alive.token();
184 _patcher->patchRow(row, std::move(*outcome));
185 if (alive.expired()) {
186 return;
187 }
188 } else {
189 auto const column = outcome.error().column.empty() ? request.column : outcome.error().column;
190 _errors[row].insert_or_assign(column, std::move(outcome.error().message));
191 }
192 if (more) {
193 start(row);
194 return;
195 }
196 _ending.erase(row);
197 if (_queues.contains(row)) {
198 // A commit made while the row was patched started its next edit.
199 return;
200 }
201 if (_engine != nullptr) {
202 _engine->endEdit(row);
203 }
204 }
205
206 RowPatcher* _patcher;
207 EditCommit _commit;
208 Engine* _engine;
209 std::unordered_map<RowId, std::deque<EditRequest>> _queues;
210 std::unordered_set<RowId> _ending; // last edit settled, row being patched
211 std::unordered_map<RowId, std::map<std::string, std::string, std::less<>>> _errors;
212 async::CallbackScope _alive;
213};
