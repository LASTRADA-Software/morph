17/// @brief Maximum accepted serialized size, in bytes, of a single wire envelope.
18///
19/// `decode` rejects any input longer than this before handing it to the JSON
20/// parser. The cap is a wire-layer denial-of-service backstop: the outer decode
21/// treats `body` as one opaque string, so a deeply-nested or enormous JSON
22/// payload smuggled *inside* `body` is invisible to any structural/depth check
23/// and would only detonate when the action codec re-parses it later (the "body
24/// double-parse"). Bounding the total message length caps that cost up
25/// front. The default of 8 MiB is generous for legitimate action payloads while
26/// keeping a single message's peak allocation bounded; transports that want a
27/// tighter bound should enforce it before calling `decode`.
28inline constexpr std::size_t kMaxEnvelopeBytes = 8u * 1024u * 1024u;
29
30/// @brief Protocol version this build of `morph` speaks.
31///
32/// Carried in `Envelope::protocolVersion` and announced by `makeHello()`.
33/// Bumped only on a **breaking** wire change (a field removal/retype, or a
34/// control-flow semantic change) — see "Action-evolution policy" in
35/// docs/spec/core/wire.md. A purely additive change (a new optional field on
36/// the envelope or on an action/result struct) does not bump this constant.
37inline constexpr std::uint32_t kProtocolVersion = 1;

39/// @brief JSON wire envelope used between any client and `RemoteServer`.
40///
41/// Supersedes the legacy pipe-delimited protocol: a single struct carries all
42/// request and reply variants, with the `kind` field acting as a discriminator.
43/// Empty/zero fields are tolerated — callers populate only what their kind needs.
44///
45/// @par Discriminator values
46/// - `"register"` — client requests model creation. Uses `typeId`, and
47/// optionally `contextKey` (the new instance's stable
48/// identity, e.g. an account id — see `RemoteServer::setLogProvider`).
49/// With `shared` set, additionally uses `primary` and becomes
50/// a register-or-attach against the shared directory.
51/// - `"attach"` — client re-points at a different `primary` of `typeId`,
52/// releasing `modelId` if non-zero. Replies `ok` with the
53/// target instance's id in `modelId`.
54/// - `"assign"` — client files live `modelId` under `primary` of `typeId`.
55/// - `"instances"` — client asks for the live shared primary keys of `typeId`.
56/// Replies `ok` with a JSON array of key strings in `body`.
57/// - `"deregister"` — client destroys an instance. Uses `modelId`.
58/// - `"execute"` — client dispatches an action. Uses `callId`, `modelId`,
59/// `modelType`, `actionType`, `body`, and optionally `session`.
60/// - `"ok"` — server success reply. Uses `callId` if present and carries
61/// the serialized result in `body`. For `register` replies,
62/// `modelId` carries the new id.
63/// - `"err"` — server failure reply. Uses `callId` if present and `message`.
64struct Envelope {
65 /// @brief Discriminator — see class docstring for valid values.
66 std::string kind;
67
68 /// @brief Correlation id for matching async replies to their requests.
69 uint64_t callId = 0;
70
71 /// @brief Model type id for `register`.
72 std::string typeId;
73
74 /// @brief Stable identity of the model instance being registered or
75 /// attached to (e.g. an account id). Empty means "no identity" —
76 /// the server-side holder gets no action log attached even if a
77 /// `LogProvider` is configured. Carried on `register` and
78 /// `attach`; ignored on every other kind.
79 std::string contextKey;
80
81 /// @brief Primary key of the instance being registered or attached to.
82 ///
83 /// Carried on `register` (when `shared` is set) and on `attach`, always as
84 /// the key's canonical string encoding (`morph::model::keyToString`)
85 /// whatever its C++ type, so the server's directory needs one map type
86 /// rather than one per key type. Empty means "no primary" — the instance is
87 /// anonymous and cannot be shared. Ignored on every other kind.
88 ///
89 /// Distinct from `contextKey`, which continues to mean only "entity key for
90 /// the action log". A keyed model will normally set both to the same value,
91 /// but conflating the fields would change behaviour for callers already
92 /// setting `contextKey` for journal purposes.
93 std::string primary;
94
95 /// @brief Whether a `register` should join the shared instance directory.
96 ///
97 /// `true` makes the request a *register-or-attach*: the server returns the
98 /// existing instance for `(typeId, primary)` if one is live, otherwise
99 /// creates it and enters it in the directory. `false` (the default, and the
100 /// value every pre-existing client sends) is today's behaviour exactly — a
101 /// private instance, invisible to the directory.
102 bool shared = false;
103
104 /// @brief Existing model instance id for `deregister`, `execute`, `ok(register)`.
105 uint64_t modelId = 0;
106
107 /// @brief Model type id for `execute` (routing key in `ActionDispatcher`).
108 std::string modelType;
109
110 /// @brief Action type id for `execute` (the second routing key).
111 std::string actionType;
112
113 /// @brief Serialized JSON of the action payload (`execute`) or of the
114 /// deserializable result (`ok`).
115 std::string body;
116
117 /// @brief Free-text error message on `err`. Empty on success.
118 std::string message;
119
120 /// @brief Session context for authorization and routing. Populated on
121 /// `execute`; ignored on every other kind.
122 ::morph::session::Context session;
123
124 /// @brief Protocol version the sender speaks.
125 ///
126 /// `0` means "unspecified / legacy peer" — the value on every envelope an
127 /// old encoder (unaware of this field) produces, and the value an old
128 /// decoder (unaware of this field) leaves untouched when it receives one
129 /// from a newer peer (ignored via `decode`'s `error_on_unknown_keys =
130 /// false`). Populated by `makeHello()` on `"hello"`; not otherwise
131 /// inspected by other `kind`s today.
132 std::uint32_t protocolVersion = 0;
133};
134

258/// @brief Replaces every ASCII control byte (0x00-0x1F, 0x7F) in @p text with a
259/// printable `\xHH` textual placeholder.
260///
261/// Applied to `makeErr`'s `message` only, and no longer for JSON validity —
262/// `detail::EscapingWriteOpts` now handles that for every field. What remains is
263/// an output-sanitization concern specific to this one field: an `err` message
264/// echoes untrusted content back (an unrecognized `Envelope::kind`, a caught
265/// exception's `what()`) and is overwhelmingly destined for a log or a console,
266/// where a raw 0x1B would carry an ANSI escape sequence into the reader's
267/// terminal. `message` is diagnostic text rather than data that must round-trip
268/// byte-for-byte, so replacing these bytes with a printable transcription costs
269/// nothing and keeps them inert.
270///
271/// Originally added for the invalid-JSON failure found by the
272/// `fuzz_dispatch_execute` harness via an `err` reply echoing an unrecognized
273/// `kind` verbatim (see docs/spec/testing_strategy.md, "Known findings"); that
274/// finding is now covered at the writer instead.
275/// @param text Untrusted text that may contain raw control bytes.
276/// @return @p text with every control byte replaced by `\xHH`.
277[[nodiscard]] inline std::string sanitizeControlChars(std::string_view text) {
278 std::string out;
279 out.reserve(text.size());
280 for (char c : text) {
281 auto byte = static_cast<unsigned char>(c);
282 if (byte < 0x20 || byte == 0x7F) {
283 static constexpr char kHex[] = "0123456789abcdef";
284 out += "\\x";
285 out += kHex[(byte >> 4) & 0xF];
286 out += kHex[byte & 0xF];
287 } else {
288 out += c;
289 }
290 }
291 return out;
292}

294/// @brief Write options that escape ASCII control bytes as `\\uXXXX` sequences.
295///
296/// glaze 7.4 leaves control bytes unescaped by default, which breaks `encode`
297/// two different ways depending on where the byte lands:
298///
299/// - **Invalid output.** RFC 8259 requires U+0000 through U+001F to be escaped,
300/// and glaze's own reader enforces that, so an envelope carrying a raw 0x0B
301/// anywhere serializes to JSON the peer's `decode` then throws on.
302/// - **Silent corruption.** Worse, the writer's chunked fast path mangles such a
303/// byte outright once the string also contains an escaped character: with a
304/// backslash or double-quote character earlier in the same string, a 0x0B at certain offsets is written
305/// as *two* 0x00 bytes. The payload is destroyed before it reaches the wire,
306/// so no amount of post-processing on the serialized form can recover it —
307/// the escaping has to happen inside the writer.
308///
309/// Enabling the option is glaze's documented remedy for exactly this (see its
310/// README, "escape control characters"); it is off by default there only because
311/// embedded nulls are hazardous for C APIs, which does not apply to a
312/// `std::string` wire field. Escaping is lossless in both directions, so any
313/// such byte round-trips unchanged rather than being replaced.
314///
315/// Applies to writing only. `decode` needs no counterpart: glaze's reader
316/// already accepts `\\uXXXX` and turns it back into the original byte.
317struct EscapingWriteOpts : glz::opts {
318 /// @brief Emit control bytes as `\\uXXXX` rather than raw.
319 // NOLINTNEXTLINE(readability-identifier-naming) — the name is glaze's, not ours; the option is matched by name.
320 bool escape_control_characters = true;
321};
322
323/// @brief Best-effort recovery of an envelope's `callId` without decoding it.
324///
325/// For the one case where a transport must answer a message it has decided
326/// *not* to parse — a frame rejected for exceeding a transport-level size cap,
327/// before `decode` is ever called. Such a reply still has to be addressed:
328/// `callId == 0` is the wire's "this is a reply to a synchronous control call"
329/// discriminator (see `QtWebSocketBackend::onTextMessage`), so an `err` sent
330/// with a zeroed id does not merely fail to resolve the execute it was meant
331/// for — it resumes whatever unrelated `register`/`deregister` happens to be
332/// parked, handing it a reply belonging to another call entirely.
333///
334/// Scans at most @p maxScanBytes, so the size cap it serves keeps its value as
335/// a cost bound; `callId` is the second field `encode` writes, so it lands well
336/// inside even a small window. A `"callId":` sequence cannot be forged from
337/// within an earlier string field, because `encode` escapes any embedded quote
338/// (yielding `\"callId\":`, which does not match).
339///
340/// @param json Raw, undecoded envelope text.
341/// @param maxScanBytes Prefix length to search. Default 1 KiB.
342/// @return The parsed `callId`, or `0` if absent, unparseable, or out of range —
343/// i.e. it degrades to exactly the behavior it replaces.
344[[nodiscard]] inline std::uint64_t peekCallId(std::string_view json, std::size_t maxScanBytes = 1024) noexcept {
345 static constexpr std::string_view kKey = "\"callId\":";
346 const std::string_view window = json.substr(0, std::min(json.size(), maxScanBytes));
347 const auto keyPos = window.find(kKey);
348 if (keyPos == std::string_view::npos) {
349 return 0;
350 }
351 std::size_t idx = keyPos + kKey.size();
352 // idx is bounded by the same condition that guards the access.
353 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
354 while (idx < window.size() && (window[idx] == ' ' || window[idx] == '\t')) {
355 ++idx;
356 }
357 std::uint64_t value = 0;
358 bool sawDigit = false;
359 for (; idx < window.size(); ++idx) {
360 // idx is bounded by the loop condition.
361 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
362 const char chr = window[idx];
363 if (chr < '0' || chr > '9') {
364 break;
365 }
366 const auto digit = static_cast<std::uint64_t>(chr - '0');
367 if (value > (std::numeric_limits<std::uint64_t>::max() - digit) / 10U) {
368 return 0; // out of range — treat as unrecoverable rather than wrap
369 }
370 value = (value * 10U) + digit;
371 sawDigit = true;
372 }
373 return sawDigit ? value : 0;
374}

404/// @brief Encodes @p env as a single JSON line.
405///
406/// Writes with `detail::EscapingWriteOpts`, so raw ASCII control bytes in any
407/// string field become `\uXXXX` escapes. Without it, a single 0x0B anywhere in
408/// `body`, `modelType`, `contextKey`, `typeId`, `actionType`, or the session's
409/// `principal`/`token` yields an envelope the peer's `decode` throws on — and,
410/// where the same string also contains a `\` or `"`, one glaze silently
411/// corrupts before it is ever sent. Both are payload-triggered transport
412/// failures, not caller errors. The escape is lossless, so such bytes still
413/// round-trip byte-for-byte.
414///
415/// @param env Envelope to serialize.
416/// @return The serialized envelope as valid, re-decodable JSON.
417/// @throws std::runtime_error on serialisation failure (should never happen for valid input).
418inline std::string encode(const Envelope& env) {
419 std::string out;
420 if (auto errCode = glz::write<detail::EscapingWriteOpts{}>(env, out)) {
421 // Not covered by this file's own test suite: every glaze write-error
422 // trigger site (`invalid_partial_key`/`unknown_key` from partial writes,
423 // `invalid_variant_object` from a `std::variant` member) is unreachable
424 // for `Envelope` -- it has no variant fields and `encode` never uses
425 // glaze's partial-write-by-key-list feature. Forcing this branch would
426 // need a fault-injection seam glaze does not expose (see
427 // LASTRADA-Software/morph#96, requesting one). Left uncovered rather
428 // than reshaping `Envelope` purely to make a test possible.
429 throw std::runtime_error("envelope encode failed: " + glz::format_error(errCode, out));
430 }
431 return out;
432}

434/// @brief Decodes @p json into an `Envelope`.
435///
436/// Enforces a wire-layer size cap: inputs longer than `kMaxEnvelopeBytes` are
437/// rejected before any parsing, bounding the cost of an oversized or
438/// deeply-nested payload smuggled inside the opaque `body` string (see
439/// `kMaxEnvelopeBytes`). This guard does **not** cap the *inner* re-parse of
440/// `body` performed later by the action codec — that codec must impose its own
441/// limits — but it does bound the total message a single `decode` will accept.
442///
443/// glaze 7.4 runs with its default options here. Note the parser does **not**
444/// reject duplicate JSON object keys: a duplicated key is accepted and the last
445/// occurrence wins. There is no glaze option to change this, so `decode` cannot
446/// enforce rejection; callers must not treat duplicate-key rejection as a
447/// security boundary (see docs/spec/wire.md).
448///
449/// @param json Serialized envelope JSON to decode.
450/// @return The decoded `Envelope`.
451/// @throws std::runtime_error if @p json exceeds `kMaxEnvelopeBytes` or is not a
452/// valid envelope.
453inline Envelope decode(std::string_view json) {
454 if (json.size() > kMaxEnvelopeBytes) {
455 throw std::runtime_error("envelope decode failed: input exceeds maximum size (" + std::to_string(json.size()) +
456 " > " + std::to_string(kMaxEnvelopeBytes) + " bytes)");
457 }
458 Envelope env{};
459 // Ignore unknown keys so the envelope is forward-compatible: a newer peer may
460 // add fields a older peer does not know, and vice versa, without a hard parse
461 // failure. Duplicate keys are still accepted last-wins (glaze offers no reject
462 // option) — see docs/spec/wire.md "Parsing guarantees and hardening".
463 // `null_terminated = false` is required, not cosmetic: `json` is a caller-supplied
464 // `string_view` with no guarantee of a trailing '\0' (e.g. a raw socket read).
465 // Left at glaze's default (true), `skip_ws` assumes it can scan past `end`
466 // looking for a terminator that may not exist — a heap-buffer-overflow on
467 // adversarial input (found by the wire_decode fuzz harness, see
468 // docs/spec/testing_strategy.md). This costs nothing: it only disables an
469 // optimization that never applied to us.
470 static constexpr glz::opts kLenient{.null_terminated = false, .error_on_unknown_keys = false};
471 if (auto errCode = glz::read<kLenient>(env, json)) {
472 throw std::runtime_error("envelope decode failed: " + glz::format_error(errCode, json));
473 }
474 return env;
475}

477/// @brief Outcome of a `"hello"` protocol-version negotiation attempt.
478enum class ProtocolNegotiationResult : std::uint8_t {
479 /// @brief The peer understood `"hello"` and accepted the announced version.
480 Negotiated,
481 /// @brief The peer does not understand `"hello"` (a pre-negotiation legacy peer).
482 LegacyPeer,
483};
484
485/// @brief Classifies a decoded reply to a `"hello"` envelope.
486///
487/// @param reply Decoded reply envelope — the result of `decode()` on the
488/// response to a `"hello"` round-trip.
489/// @return `Negotiated` if @p reply's `kind` is `"ok"`; `LegacyPeer` if it is
490/// an `"err"` whose `message` is exactly `"unknown envelope kind:
491/// hello"` — the generic unrecognised-`kind` message a pre-negotiation
492/// `RemoteServer` produces for a `kind` it does not switch on.
493/// @throws std::runtime_error if @p reply is any other `"err"` (e.g.
494/// `"protocol version unsupported"`), so the caller refuses to
495/// proceed instead of silently treating an explicit rejection as a
496/// legacy peer.
497inline ProtocolNegotiationResult interpretHelloReply(const Envelope& reply) {
498 if (reply.kind == "ok") {
499 return ProtocolNegotiationResult::Negotiated;
500 }
501 if (reply.message == "unknown envelope kind: hello") {
502 return ProtocolNegotiationResult::LegacyPeer;
503 }
504 throw std::runtime_error("protocol negotiation failed: " +
505 (reply.message.empty() ? std::string{"malformed reply"} : reply.message));
506}
