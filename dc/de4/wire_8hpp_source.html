20/// @brief Maximum accepted serialized size, in bytes, of a single wire envelope.
21///
22/// `decode` rejects any input longer than this before handing it to the JSON
23/// parser. The cap is a wire-layer denial-of-service backstop: the outer decode
24/// treats `body` as one opaque string, so a deeply-nested or enormous JSON
25/// payload smuggled *inside* `body` is invisible to any structural/depth check
26/// and would only detonate when the action codec re-parses it later (the "body
27/// double-parse"). Bounding the total message length caps that cost up
28/// front. The default of 8 MiB is generous for legitimate action payloads while
29/// keeping a single message's peak allocation bounded; transports that want a
30/// tighter bound should enforce it before calling `decode`.
31inline constexpr std::size_t kMaxEnvelopeBytes = 8u * 1024u * 1024u;
32
33/// @brief Protocol version this build of `morph` speaks.
34///
35/// Carried in `Envelope::protocolVersion` and announced by `makeHello()`.
36/// Bumped only on a **breaking** wire change (a field removal/retype, or a
37/// control-flow semantic change) — see "Action-evolution policy" in
38/// docs/spec/core/wire.md. A purely additive change (a new optional field on
39/// the envelope or on an action/result struct) does not bump this constant.
40inline constexpr std::uint32_t kProtocolVersion = 1;
41
42/// @brief JSON wire envelope used between any client and `RemoteServer`.
43///
44/// Supersedes the legacy pipe-delimited protocol: a single struct carries all
45/// request and reply variants, with the `kind` field acting as a discriminator.
46/// Empty/zero fields are tolerated — callers populate only what their kind needs.
47///
48/// @par Discriminator values
49/// - `"register"` — client requests model creation. Uses `typeId`, and
50/// optionally `contextKey` (the new instance's stable
51/// identity, e.g. an account id — see `RemoteServer::setLogProvider`).
52/// With `shared` set, additionally uses `primary` and becomes
53/// a register-or-attach against the shared directory.
54/// - `"attach"` — client re-points at a different `primary` of `typeId`,
55/// releasing `modelId` if non-zero. Replies `ok` with the
56/// target instance's id in `modelId`.
57/// - `"assign"` — client files live `modelId` under `primary` of `typeId`.
58/// - `"instances"` — client asks for the live shared primary keys of `typeId`.
59/// Replies `ok` with a JSON array of key strings in `body`.
60/// - `"schemas"` — client asks for the action descriptions of `typeId`.
61/// Replies `ok` with a `{actionType: schema}` JSON object in
62/// `body` — see "Serving action schemas" in
63/// docs/spec/core/wire.md.
64/// - `"deregister"` — client destroys an instance. Uses `modelId`.
65/// - `"execute"` — client dispatches an action. Uses `callId`, `modelId`,
66/// `modelType`, `actionType`, `body`, and optionally `session`.
67/// - `"ok"` — server success reply. Uses `callId` if present and carries
68/// the serialized result in `body`. For `register` replies,
69/// `modelId` carries the new id.
70/// - `"err"` — server failure reply. Uses `callId` if present and `message`.
71/// - `"hello"` — client opens protocol-version negotiation, once per
72/// connection before any other kind. Uses `protocolVersion`;
73/// see `makeHello()` and `interpretHelloReply()`. A peer
74/// predating negotiation answers `err "unknown envelope kind:
75/// hello"`, which is how a legacy server is detected.
76struct Envelope {
77 /// @brief Discriminator — see class docstring for valid values.
78 std::string kind;
79
80 /// @brief Correlation id for matching async replies to their requests.
81 uint64_t callId = 0;
82
83 /// @brief Model type id for `register`.
84 std::string typeId;
85
86 /// @brief Stable identity of the model instance being registered or
87 /// attached to (e.g. an account id). Empty means "no identity" —
88 /// the server-side holder gets no action log attached even if a
89 /// `LogProvider` is configured. Carried on `register` and
90 /// `attach`; ignored on every other kind.
91 std::string contextKey;
92
93 /// @brief Primary key of the instance being registered or attached to.
94 ///
95 /// Carried on `register` (when `shared` is set) and on `attach`, always as
96 /// the key's canonical string encoding (`morph::model::keyToString`)
97 /// whatever its C++ type, so the server's directory needs one map type
98 /// rather than one per key type. Empty means "no primary" — the instance is
99 /// anonymous and cannot be shared. Ignored on every other kind.
100 ///
101 /// Distinct from `contextKey`, which continues to mean only "entity key for
102 /// the action log". A keyed model will normally set both to the same value,
103 /// but conflating the fields would change behaviour for callers already
104 /// setting `contextKey` for journal purposes.
105 std::string primary;
106
107 /// @brief Whether a `register` should join the shared instance directory.
108 ///
109 /// `true` makes the request a *register-or-attach*: the server returns the
110 /// existing instance for `(typeId, primary)` if one is live, otherwise
111 /// creates it and enters it in the directory. `false` (the default, and the
112 /// value every pre-existing client sends) is today's behaviour exactly — a
113 /// private instance, invisible to the directory.
114 bool shared = false;
115
116 /// @brief Existing model instance id for `deregister`, `execute`, `ok(register)`.
117 uint64_t modelId = 0;
118
119 /// @brief Model type id for `execute` (routing key in `ActionDispatcher`).
120 std::string modelType;
121
122 /// @brief Action type id for `execute` (the second routing key).
123 std::string actionType;
124
125 /// @brief Serialized JSON of the action payload (`execute`) or of the
126 /// deserializable result (`ok`).
127 std::string body;
128
129 /// @brief Free-text error message on `err`. Empty on success.
130 std::string message;
131
132 /// @brief Session context for authorization and routing. Populated on
133 /// `execute`; ignored on every other kind.
134 ::morph::session::Context session;
135
136 /// @brief Protocol version the sender speaks.
137 ///
138 /// `0` means "unspecified / legacy peer" — the value on every envelope an
139 /// old encoder (unaware of this field) produces, and the value an old
140 /// decoder (unaware of this field) leaves untouched when it receives one
141 /// from a newer peer (ignored via `decode`'s `error_on_unknown_keys =
142 /// false`). Populated by `makeHello()` on `"hello"`; not otherwise
143 /// inspected by other `kind`s today.
144 std::uint32_t protocolVersion = 0;
145};
146
147/// @brief Inclusive protocol-version range a server advertises in reply to
148/// `"hello"`.
149///
150/// Serialized as the `"ok"` reply's `body` (JSON) when a server accepts a
151/// `"hello"` — see `RemoteServer::setSupportedVersionRange`.
152struct ProtocolRange {
153 /// @brief Oldest protocol version the server accepts.
154 std::uint32_t min = kProtocolVersion;
155
156 /// @brief Newest protocol version the server accepts.
157 std::uint32_t max = kProtocolVersion;
158};

288/// @brief Replaces every ASCII control byte (0x00-0x1F, 0x7F) in @p text with a
289/// printable `\xHH` textual placeholder.
290///
291/// Applied to `makeErr`'s `message` only, and no longer for JSON validity —
292/// `detail::EscapingWriteOpts` now handles that for every field. What remains is
293/// an output-sanitization concern specific to this one field: an `err` message
294/// echoes untrusted content back (an unrecognized `Envelope::kind`, a caught
295/// exception's `what()`) and is overwhelmingly destined for a log or a console,
296/// where a raw 0x1B would carry an ANSI escape sequence into the reader's
297/// terminal. `message` is diagnostic text rather than data that must round-trip
298/// byte-for-byte, so replacing these bytes with a printable transcription costs
299/// nothing and keeps them inert.
300///
301/// Originally added for the invalid-JSON failure found by the
302/// `fuzz_dispatch_execute` harness via an `err` reply echoing an unrecognized
303/// `kind` verbatim (see docs/spec/testing_strategy.md, "Known findings"); that
304/// finding is now covered at the writer instead.
305/// @param text Untrusted text that may contain raw control bytes.
306/// @return @p text with every control byte replaced by `\xHH`.
307[[nodiscard]] inline std::string sanitizeControlChars(std::string_view text) {
308 std::string out;
309 out.reserve(text.size());
310 for (char c : text) {
311 auto byte = static_cast<unsigned char>(c);
312 if (byte < 0x20 || byte == 0x7F) {
313 static constexpr char kHex[] = "0123456789abcdef";
314 out += "\\x";
315 out += kHex[(byte >> 4) & 0xF];
316 out += kHex[byte & 0xF];
317 } else {
318 out += c;
319 }
320 }
321 return out;
322}

324/// @brief Write options that escape ASCII control bytes as `\\uXXXX` sequences.
325///
326/// glaze 7.4 leaves control bytes unescaped by default, which breaks `encode`
327/// two different ways depending on where the byte lands:
328///
329/// - **Invalid output.** RFC 8259 requires U+0000 through U+001F to be escaped,
330/// and glaze's own reader enforces that, so an envelope carrying a raw 0x0B
331/// anywhere serializes to JSON the peer's `decode` then throws on.
332/// - **Silent corruption.** Worse, the writer's chunked fast path mangles such a
333/// byte outright once the string also contains an escaped character: with a
334/// backslash or double-quote character earlier in the same string, a 0x0B at certain offsets is written
335/// as *two* 0x00 bytes. The payload is destroyed before it reaches the wire,
336/// so no amount of post-processing on the serialized form can recover it —
337/// the escaping has to happen inside the writer.
338///
339/// Enabling the option is glaze's documented remedy for exactly this (see its
340/// README, "escape control characters"); it is off by default there only because
341/// embedded nulls are hazardous for C APIs, which does not apply to a
342/// `std::string` wire field. Escaping is lossless in both directions, so any
343/// such byte round-trips unchanged rather than being replaced.
344///
345/// Applies to writing only. `decode` needs no counterpart: glaze's reader
346/// already accepts `\\uXXXX` and turns it back into the original byte.
347struct EscapingWriteOpts : glz::opts {
348 /// @brief Emit control bytes as `\\uXXXX` rather than raw.
349 // NOLINTNEXTLINE(readability-identifier-naming) — the name is glaze's, not ours; the option is matched by name.
350 bool escape_control_characters = true;
351};

353/// @brief The `Envelope` keys `encode` omits when the member holds its default.
354///
355/// `Envelope` is a union-of-all-kinds struct: an `ok` reply uses three of its
356/// thirteen members and a `deregister` request uses two, but glaze writes every
357/// member of a struct it is handed, so a minimal reply carrying an 8-byte
358/// payload cost 255 bytes on the wire, 213 of them fields the kind does not use
359/// (morph#524).
360///
361/// Omitting a member that holds its default is **not** a protocol change. The
362/// decoder default-initialises the `Envelope` and reads with
363/// `error_on_unknown_keys = false`; a key that is absent therefore leaves the
364/// member at exactly the value the omitted key would have written. The
365/// `kProtocolVersion` bump that `docs/spec/core/wire.md` requires for a field
366/// removal or retype does not apply, because nothing is removed or retyped —
367/// see "Omitted default fields" there, and `test_wire_omitted_fields.cpp`,
368/// which decodes both forms and requires the results to be equal.
369///
370/// **Two fields are deliberately absent from this list, and for the same
371/// reason: their default is not "unset".** The rule the rest of the list obeys
372/// is that omitting a member is safe when its default means *nothing* — when
373/// "absent" and "present, holding the default" are the same statement. That
374/// holds for the payload and addressing fields: an empty `typeId` means no type
375/// id, a `modelId` of `0` means no model id (`remote.hpp` tests exactly that),
376/// and a `protocolVersion` of `0` is documented on the member as "unspecified /
377/// legacy peer" — precisely what an encoder that had never heard of the field
378/// would produce. For these two it does not hold:
379///
380/// - **`kind`** is the discriminator; an envelope without one is malformed.
381/// Keeping it unconditional also means the first key of every message is
382/// still `"kind"`.
383/// - **`callId` is a correlation field, and `0` is a live routing instruction
384/// rather than an absence.** Both transports discriminate on it: a reply with
385/// a non-zero `callId` is matched against the pending-execute map, and one
386/// with `callId == 0` is handed to whichever synchronous control call is
387/// parked (`QtWebSocketBackend::onTextMessage`,
388/// `SocketBackend::dispatchIncomingEnvelope`). A peer therefore cannot treat
389/// an absent `callId` as "no information"; it has to *reconstruct* the
390/// sentinel before it can route the frame at all. morph's own `decode` does
391/// that for free by default-initialising, which is why omitting it
392/// round-tripped cleanly in C++ and still broke the scenario driver, a second
393/// decoder that read absence as `None` (morph#524). Emitting correlation
394/// fields unconditionally costs eleven bytes on the one message shape that
395/// carries a zero id, and keeps "how do I route this frame" answerable from
396/// the frame.
397///
398/// The general form of the rule, for anyone adding a member: **elide payload,
399/// never elide identity.** If a peer has to know the field's value to decide
400/// where the message goes, it is written even at its default.
401inline constexpr std::array<std::string_view, 11> kOmittableEnvelopeKeys{
402 "typeId", "contextKey", "primary", "shared", "modelId", "modelType",
403 "actionType", "body", "message", "session", "protocolVersion"};
404
405/// @brief Whether a session context carries nothing worth sending.
406///
407/// @param ctx Session context to inspect.
408/// @return `true` when every member is empty, so the whole `session` object can
409/// be left out of the envelope.
410[[nodiscard]] inline bool isDefaultSession(const ::morph::session::Context& ctx) noexcept {
411 // A field-by-field test rather than `ctx == Context{}`, because `Context` is
412 // a plain aggregate with no `operator==` and giving it one would reach
413 // outside this header. The assertion is what stops the two drifting: a new
414 // member would otherwise be dropped from every envelope whose other session
415 // fields happen to be empty, silently and only on the wire.
416 static_assert(glz::reflect<::morph::session::Context>::size == 5,
417 "session::Context gained or lost a member: update isDefaultSession, or the new "
418 "member is silently dropped from every envelope with an otherwise-empty session.");
419 return ctx.principal.empty() && ctx.token.empty() && ctx.requestId.empty() && ctx.locale.empty() &&
420 ctx.metadata.empty();
421}

423/// @brief Collects the keys `encode` should leave out of @p env.
424///
425/// @param env Envelope about to be serialised.
426/// @param out Storage for the key list; only the returned prefix is meaningful.
427/// @return The keys to exclude, as a view into @p out.
428[[nodiscard]] inline std::span<const std::string_view> defaultValuedKeys(
429 const Envelope& env, std::array<std::string_view, kOmittableEnvelopeKeys.size()>& out) noexcept {
430 // Every name below must be a real reflected key of `Envelope`: glaze's
431 // exclude-write reports `unknown_key` for one that is not, which would turn
432 // a member rename into a throw from *every* `encode` call at runtime. This
433 // makes it a compile error instead.
434 static_assert(glz::reflect<Envelope>::size == 13,
435 "Envelope gained or lost a member: decide whether it is omittable when default "
436 "and update kOmittableEnvelopeKeys. A new member left off that list is always "
437 "written, which is merely the old behaviour; one added with a wrong spelling is "
438 "caught by the assertion below.");
439 static_assert(std::ranges::all_of(kOmittableEnvelopeKeys,
440 [](std::string_view key) {
441 return std::ranges::find(glz::reflect<Envelope>::keys, key) !=
442 std::ranges::end(glz::reflect<Envelope>::keys);
443 }),
444 "kOmittableEnvelopeKeys names a key Envelope does not have — a rename, or a typo.");
445
446 std::size_t count = 0;
447 auto omit = [&out, &count](std::string_view key) noexcept { out.at(count++) = key; };
448
449 // No `callId` arm, deliberately: see `kOmittableEnvelopeKeys`. It is a
450 // correlation field whose zero is a routing sentinel, so it is written
451 // unconditionally, exactly like `kind`.
452 if (env.typeId.empty()) {
453 omit("typeId");
454 }
455 if (env.contextKey.empty()) {
456 omit("contextKey");
457 }
458 if (env.primary.empty()) {
459 omit("primary");
460 }
461 if (!env.shared) {
462 omit("shared");
463 }
464 if (env.modelId == 0) {
465 omit("modelId");
466 }
467 if (env.modelType.empty()) {
468 omit("modelType");
469 }
470 if (env.actionType.empty()) {
471 omit("actionType");
472 }
473 if (env.body.empty()) {
474 omit("body");
475 }
476 if (env.message.empty()) {
477 omit("message");
478 }
479 if (isDefaultSession(env.session)) {
480 omit("session");
481 }
482 if (env.protocolVersion == 0) {
483 omit("protocolVersion");
484 }
485 return std::span<const std::string_view>{out.data(), count};
486}

488/// @brief Best-effort recovery of an envelope's `callId` without decoding it.
489///
490/// For the one case where a transport must answer a message it has decided
491/// *not* to parse — a frame rejected for exceeding a transport-level size cap,
492/// before `decode` is ever called. Such a reply still has to be addressed:
493/// `callId == 0` is the wire's "this is a reply to a synchronous control call"
494/// discriminator (see `QtWebSocketBackend::onTextMessage`), so an `err` sent
495/// with a zeroed id does not merely fail to resolve the execute it was meant
496/// for — it resumes whatever unrelated `register`/`deregister` happens to be
497/// parked, handing it a reply belonging to another call entirely.
498///
499/// Scans at most @p maxScanBytes, so the size cap it serves keeps its value as
500/// a cost bound; `callId` is the second field `encode` writes and is never
501/// omitted (see `kOmittableEnvelopeKeys`), so for any envelope morph encoded it
502/// lands well inside even a small window. A `"callId":` sequence cannot be
503/// forged from within an earlier string field, because `encode` escapes any
504/// embedded quote (yielding `\"callId\":`, which does not match).
505///
506/// The `0`-on-absence return is still load-bearing rather than vestigial: the
507/// input here is by definition a frame that was *not* produced under morph's
508/// own invariants — it is whatever a peer sent, at a size this side already
509/// refused to parse — so "no `callId` in the first KiB" remains a case that has
510/// to have an answer.
511///
512/// @param json Raw, undecoded envelope text.
513/// @param maxScanBytes Prefix length to search. Default 1 KiB.
514/// @return The parsed `callId`, or `0` if absent, unparseable, or out of range —
515/// i.e. it degrades to exactly the behavior it replaces.
516[[nodiscard]] inline std::uint64_t peekCallId(std::string_view json, std::size_t maxScanBytes = 1024) noexcept {
517 static constexpr std::string_view kKey = "\"callId\":";
518 const std::string_view window = json.substr(0, std::min(json.size(), maxScanBytes));
519 const auto keyPos = window.find(kKey);
520 if (keyPos == std::string_view::npos) {
521 return 0;
522 }
523 std::size_t idx = keyPos + kKey.size();
524 // idx is bounded by the same condition that guards the access.
525 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
526 while (idx < window.size() && (window[idx] == ' ' || window[idx] == '\t')) {
527 ++idx;
528 }
529 std::uint64_t value = 0;
530 bool sawDigit = false;
531 for (; idx < window.size(); ++idx) {
532 // idx is bounded by the loop condition.
533 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
534 const char chr = window[idx];
535 if (chr < '0' || chr > '9') {
536 break;
537 }
538 const auto digit = static_cast<std::uint64_t>(chr - '0');
539 if (value > (std::numeric_limits<std::uint64_t>::max() - digit) / 10U) {
540 return 0; // out of range — treat as unrecoverable rather than wrap
541 }
542 value = (value * 10U) + digit;
543 sawDigit = true;
544 }
545 return sawDigit ? value : 0;
546}

550/// @brief Builds an `err` reply envelope.
551///
552/// @p message may embed untrusted content (e.g. an unrecognized `Envelope::kind`
553/// or a caught exception's `what()`) — any raw control byte it contains is
554/// replaced with a `\xHH` placeholder. This is *output sanitization*, not JSON
555/// validity: `encode`'s `detail::EscapingWriteOpts` already guarantees validity
556/// for every field. What this stops is a raw control byte reaching a terminal or
557/// log that interprets it (see `detail::sanitizeControlChars`).
558inline Envelope makeErr(std::string message, uint64_t callId = 0) {
559 Envelope env;
560 env.kind = "err";
561 env.callId = callId;
562 env.message = detail::sanitizeControlChars(message);
563 return env;
564}

597/// @brief The serialisation primitive `encode` calls, as an injectable strategy.
598///
599/// Same shape and rationale as `morph::core::FileIoOps`: the member defaults
600/// to the real call it stands in for, so a default-constructed `WireCodecOps`
601/// is byte-for-byte what `encode` did before this seam existed.
602///
603/// @par Why this exists
604/// `encode`'s write-failure arm is unreachable through any `Envelope` value.
605/// Glaze only sets a write-time error for `invalid_partial_key`/`unknown_key`
606/// (its partial-write-by-key-list feature, which `encode` does not use) or
607/// `invalid_variant_object` (a `std::variant` member, which `Envelope` does
608/// not have). Confirmed by experiment as well as by reading glaze: five
609/// flavours of invalid UTF-8, an embedded NUL, a raw control byte and an 8 MiB
610/// payload all encode successfully.
611///
612/// That left a permanently uncovered branch guarding a real invariant. Rather
613/// than reshape `Envelope` to make a test possible, or wait on a
614/// fault-injection hook in glaze (LASTRADA-Software/morph#96 asked for one),
615/// this puts the seam on morph's side of the boundary — where the repository
616/// already puts it for the identical problem with file I/O.
617struct WireCodecOps {
618 /// @brief Serialises @p env into @p out. Mirrors the `glz::write` call
619 /// `encode` would otherwise make directly.
620 /// @return Glaze's error context; falsy on success.
621 std::function<glz::error_ctx(const Envelope& env, std::string& out)> writeEnvelope = [](const Envelope& env,
622 std::string& out) {
623 std::array<std::string_view, detail::kOmittableEnvelopeKeys.size()> omitted{};
624 return glz::write_json_exclude<detail::EscapingWriteOpts{}>(env, detail::defaultValuedKeys(env, omitted), out);
625 };
626};

628/// @brief The shared default `WireCodecOps`.
629///
630/// A function-local static, not a default-constructed temporary in `encode`'s
631/// signature: `encode` runs on every outbound message, and building a fresh
632/// `std::function` per call would put an allocation on that path purely to
633/// support a test seam.
634/// @return Reference to the process-wide default operations.
635[[nodiscard]] inline const WireCodecOps& defaultWireCodecOps() {
636 static const WireCodecOps ops;
637 return ops;
638}

640/// @brief Encodes @p env as a single JSON line.
641///
642/// Writes with `detail::EscapingWriteOpts`, so raw ASCII control bytes in any
643/// string field become `\uXXXX` escapes. Without it, a single 0x0B anywhere in
644/// `body`, `modelType`, `contextKey`, `typeId`, `actionType`, or the session's
645/// `principal`/`token` yields an envelope the peer's `decode` throws on — and,
646/// where the same string also contains a `\` or `"`, one glaze silently
647/// corrupts before it is ever sent. Both are payload-triggered transport
648/// failures, not caller errors. The escape is lossless, so such bytes still
649/// round-trip byte-for-byte.
650///
651/// @param env Envelope to serialize.
652/// @param ops Serialisation strategy; defaults to the real `glz::write` call.
653/// A test injects a failing one to exercise the throw path, which no
654/// `Envelope` value can reach — see `WireCodecOps`.
655/// @return The serialized envelope as valid, re-decodable JSON.
656/// @throws std::runtime_error on serialisation failure (should never happen for valid input).
657inline std::string encode(const Envelope& env, const WireCodecOps& ops = defaultWireCodecOps()) {
658 std::string out;
659 if (auto errCode = ops.writeEnvelope(env, out)) {
660 // Unreachable through any `Envelope` value -- see `WireCodecOps` for
661 // why, and for how a test covers this arm anyway.
662 throw std::runtime_error("envelope encode failed: " + glz::format_error(errCode, out));
663 }
664 return out;
665}

667/// @brief Decodes @p json into an `Envelope`.
668///
669/// Enforces a wire-layer size cap: inputs longer than `kMaxEnvelopeBytes` are
670/// rejected before any parsing, bounding the cost of an oversized or
671/// deeply-nested payload smuggled inside the opaque `body` string (see
672/// `kMaxEnvelopeBytes`). This guard does **not** cap the *inner* re-parse of
673/// `body` performed later by the action codec — that codec must impose its own
674/// limits — but it does bound the total message a single `decode` will accept.
675///
676/// glaze 7.4 runs with its default options here. Note the parser does **not**
677/// reject duplicate JSON object keys: a duplicated key is accepted and the last
678/// occurrence wins. There is no glaze option to change this, so `decode` cannot
679/// enforce rejection; callers must not treat duplicate-key rejection as a
680/// security boundary (see docs/spec/core/wire.md).
681///
682/// @param json Serialized envelope JSON to decode.
683/// @return The decoded `Envelope`.
684/// @throws std::runtime_error if @p json exceeds `kMaxEnvelopeBytes` or is not a
685/// valid envelope.
686inline Envelope decode(std::string_view json) {
687 if (json.size() > kMaxEnvelopeBytes) {
688 throw std::runtime_error("envelope decode failed: input exceeds maximum size (" + std::to_string(json.size()) +
689 " > " + std::to_string(kMaxEnvelopeBytes) + " bytes)");
690 }
691 Envelope env{};
692 // Ignore unknown keys so the envelope is forward-compatible: a newer peer may
693 // add fields a older peer does not know, and vice versa, without a hard parse
694 // failure. Duplicate keys are still accepted last-wins (glaze offers no reject
695 // option) — see docs/spec/core/wire.md "Parsing guarantees and hardening".
696 // `null_terminated = false` is required, not cosmetic: `json` is a caller-supplied
697 // `string_view` with no guarantee of a trailing '\0' (e.g. a raw socket read).
698 // Left at glaze's default (true), `skip_ws` assumes it can scan past `end`
699 // looking for a terminator that may not exist — a heap-buffer-overflow on
700 // adversarial input (found by the wire_decode fuzz harness, see
701 // docs/spec/testing_strategy.md). This costs nothing: it only disables an
702 // optimization that never applied to us.
703 static constexpr glz::opts kLenient{.null_terminated = false, .error_on_unknown_keys = false};
704 if (auto errCode = glz::read<kLenient>(env, json)) {
705 throw std::runtime_error("envelope decode failed: " + glz::format_error(errCode, json));
706 }
707 return env;
708}

710/// @brief Outcome of a `"hello"` protocol-version negotiation attempt.
711enum class ProtocolNegotiationResult : std::uint8_t {
712 /// @brief The peer understood `"hello"` and accepted the announced version.
713 Negotiated,
714 /// @brief The peer does not understand `"hello"` (a pre-negotiation legacy peer).
715 LegacyPeer,
716};
717
718/// @brief Classifies a decoded reply to a `"hello"` envelope.
719///
720/// @param reply Decoded reply envelope — the result of `decode()` on the
721/// response to a `"hello"` round-trip.
722/// @return `Negotiated` if @p reply's `kind` is `"ok"`; `LegacyPeer` if it is
723/// an envelope whose `message` is exactly `"unknown envelope kind:
724/// hello"` — the generic unrecognised-`kind` message a pre-negotiation
725/// `RemoteServer` produces for a `kind` it does not switch on.
726/// @throws std::runtime_error if @p reply is any other `"err"` (e.g.
727/// `"protocol version unsupported"`), so the caller refuses to
728/// proceed instead of silently treating an explicit rejection as a
729/// legacy peer.
730inline ProtocolNegotiationResult interpretHelloReply(const Envelope& reply) {
731 if (reply.kind == "ok") {
732 return ProtocolNegotiationResult::Negotiated;
733 }
734 // Matched on `message` alone: `kind` is deliberately not inspected, because a
735 // peer old enough to not know `hello` is also a peer whose error shape we do
736 // not want to depend on. Any envelope carrying exactly this message is
737 // treated as a legacy peer.
738 if (reply.message == "unknown envelope kind: hello") {
739 return ProtocolNegotiationResult::LegacyPeer;
740 }
741 throw std::runtime_error("protocol negotiation failed: " +
742 (reply.message.empty() ? std::string{"malformed reply"} : reply.message));
743}
