17/// @brief Maximum accepted serialized size, in bytes, of a single wire envelope.
18///
19/// `decode` rejects any input longer than this before handing it to the JSON
20/// parser. The cap is a wire-layer denial-of-service backstop: the outer decode
21/// treats `body` as one opaque string, so a deeply-nested or enormous JSON
22/// payload smuggled *inside* `body` is invisible to any structural/depth check
23/// and would only detonate when the action codec re-parses it later (the "body
24/// double-parse"). Bounding the total message length caps that cost up
25/// front. The default of 8 MiB is generous for legitimate action payloads while
26/// keeping a single message's peak allocation bounded; transports that want a
27/// tighter bound should enforce it before calling `decode`.
28inline constexpr std::size_t kMaxEnvelopeBytes = 8u * 1024u * 1024u;
29
30/// @brief Protocol version this build of `morph` speaks.
31///
32/// Carried in `Envelope::protocolVersion` and announced by `makeHello()`.
33/// Bumped only on a **breaking** wire change (a field removal/retype, or a
34/// control-flow semantic change) — see "Action-evolution policy" in
35/// docs/spec/core/wire.md. A purely additive change (a new optional field on
36/// the envelope or on an action/result struct) does not bump this constant.
37inline constexpr std::uint32_t kProtocolVersion = 1;

39/// @brief JSON wire envelope used between any client and `RemoteServer`.
40///
41/// Supersedes the legacy pipe-delimited protocol: a single struct carries all
42/// request and reply variants, with the `kind` field acting as a discriminator.
43/// Empty/zero fields are tolerated — callers populate only what their kind needs.
44///
45/// @par Discriminator values
46/// - `"register"` — client requests model creation. Uses `typeId`, and
47/// optionally `contextKey` (the new instance's stable
48/// identity, e.g. an account id — see `RemoteServer::setLogProvider`).
49/// - `"deregister"` — client destroys an instance. Uses `modelId`.
50/// - `"execute"` — client dispatches an action. Uses `callId`, `modelId`,
51/// `modelType`, `actionType`, `body`, and optionally `session`.
52/// - `"ok"` — server success reply. Uses `callId` if present and carries
53/// the serialized result in `body`. For `register` replies,
54/// `modelId` carries the new id.
55/// - `"err"` — server failure reply. Uses `callId` if present and `message`.
56struct Envelope {
57 /// @brief Discriminator — see class docstring for valid values.
58 std::string kind;
59
60 /// @brief Correlation id for matching async replies to their requests.
61 uint64_t callId = 0;
62
63 /// @brief Model type id for `register`.
64 std::string typeId;
65
66 /// @brief Stable identity of the model instance being registered (e.g. an
67 /// account id). Empty means "no identity" — the server-side holder
68 /// gets no action log attached even if a `LogProvider` is configured.
69 /// Ignored on every kind other than `register`.
70 std::string contextKey;
71
72 /// @brief Existing model instance id for `deregister`, `execute`, `ok(register)`.
73 uint64_t modelId = 0;
74
75 /// @brief Model type id for `execute` (routing key in `ActionDispatcher`).
76 std::string modelType;
77
78 /// @brief Action type id for `execute` (the second routing key).
79 std::string actionType;
80
81 /// @brief Serialized JSON of the action payload (`execute`) or of the
82 /// deserializable result (`ok`).
83 std::string body;
84
85 /// @brief Free-text error message on `err`. Empty on success.
86 std::string message;
87
88 /// @brief Session context for authorization and routing. Populated on
89 /// `execute`; ignored on every other kind.
90 ::morph::session::Context session;
91
92 /// @brief Protocol version the sender speaks.
93 ///
94 /// `0` means "unspecified / legacy peer" — the value on every envelope an
95 /// old encoder (unaware of this field) produces, and the value an old
96 /// decoder (unaware of this field) leaves untouched when it receives one
97 /// from a newer peer (ignored via `decode`'s `error_on_unknown_keys =
98 /// false`). Populated by `makeHello()` on `"hello"`; not otherwise
99 /// inspected by other `kind`s today.
100 std::uint32_t protocolVersion = 0;
101};
102

148/// @brief Replaces every ASCII control byte (0x00-0x1F, 0x7F) in @p text with a
149/// printable `\xHH` textual placeholder.
150///
151/// Applied to `makeErr`'s `message` only, and no longer for JSON validity —
152/// `detail::EscapingWriteOpts` now handles that for every field. What remains is
153/// an output-sanitization concern specific to this one field: an `err` message
154/// echoes untrusted content back (an unrecognized `Envelope::kind`, a caught
155/// exception's `what()`) and is overwhelmingly destined for a log or a console,
156/// where a raw 0x1B would carry an ANSI escape sequence into the reader's
157/// terminal. `message` is diagnostic text rather than data that must round-trip
158/// byte-for-byte, so replacing these bytes with a printable transcription costs
159/// nothing and keeps them inert.
160///
161/// Originally added for the invalid-JSON failure found by the
162/// `fuzz_dispatch_execute` harness via an `err` reply echoing an unrecognized
163/// `kind` verbatim (see docs/spec/testing_strategy.md, "Known findings"); that
164/// finding is now covered at the writer instead.
165/// @param text Untrusted text that may contain raw control bytes.
166/// @return @p text with every control byte replaced by `\xHH`.
167[[nodiscard]] inline std::string sanitizeControlChars(std::string_view text) {
168 std::string out;
169 out.reserve(text.size());
170 for (char c : text) {
171 auto byte = static_cast<unsigned char>(c);
172 if (byte < 0x20 || byte == 0x7F) {
173 static constexpr char kHex[] = "0123456789abcdef";
174 out += "\\x";
175 out += kHex[(byte >> 4) & 0xF];
176 out += kHex[byte & 0xF];
177 } else {
178 out += c;
179 }
180 }
181 return out;
182}

184/// @brief Write options that escape ASCII control bytes as `\\uXXXX` sequences.
185///
186/// glaze 7.4 leaves control bytes unescaped by default, which breaks `encode`
187/// two different ways depending on where the byte lands:
188///
189/// - **Invalid output.** RFC 8259 requires U+0000 through U+001F to be escaped,
190/// and glaze's own reader enforces that, so an envelope carrying a raw 0x0B
191/// anywhere serializes to JSON the peer's `decode` then throws on.
192/// - **Silent corruption.** Worse, the writer's chunked fast path mangles such a
193/// byte outright once the string also contains an escaped character: with a
194/// `\\` or `"` earlier in the same string, a 0x0B at certain offsets is written
195/// as *two* 0x00 bytes. The payload is destroyed before it reaches the wire,
196/// so no amount of post-processing on the serialized form can recover it —
197/// the escaping has to happen inside the writer.
198///
199/// Enabling the option is glaze's documented remedy for exactly this (see its
200/// README, "escape control characters"); it is off by default there only because
201/// embedded nulls are hazardous for C APIs, which does not apply to a
202/// `std::string` wire field. Escaping is lossless in both directions, so any
203/// such byte round-trips unchanged rather than being replaced.
204///
205/// Applies to writing only. `decode` needs no counterpart: glaze's reader
206/// already accepts `\\uXXXX` and turns it back into the original byte.
207struct EscapingWriteOpts : glz::opts {
208 /// @brief Emit control bytes as `\\uXXXX` rather than raw.
209 // NOLINTNEXTLINE(readability-identifier-naming) — the name is glaze's, not ours; the option is matched by name.
210 bool escape_control_characters = true;
211};

213/// @brief Best-effort recovery of an envelope's `callId` without decoding it.
214///
215/// For the one case where a transport must answer a message it has decided
216/// *not* to parse — a frame rejected for exceeding a transport-level size cap,
217/// before `decode` is ever called. Such a reply still has to be addressed:
218/// `callId == 0` is the wire's "this is a reply to a synchronous control call"
219/// discriminator (see `QtWebSocketBackend::onTextMessage`), so an `err` sent
220/// with a zeroed id does not merely fail to resolve the execute it was meant
221/// for — it resumes whatever unrelated `register`/`deregister` happens to be
222/// parked, handing it a reply belonging to another call entirely.
223///
224/// Scans at most @p maxScanBytes, so the size cap it serves keeps its value as
225/// a cost bound; `callId` is the second field `encode` writes, so it lands well
226/// inside even a small window. A `"callId":` sequence cannot be forged from
227/// within an earlier string field, because `encode` escapes any embedded quote
228/// (yielding `\"callId\":`, which does not match).
229///
230/// @param json Raw, undecoded envelope text.
231/// @param maxScanBytes Prefix length to search. Default 1 KiB.
232/// @return The parsed `callId`, or `0` if absent, unparseable, or out of range —
233/// i.e. it degrades to exactly the behavior it replaces.
234[[nodiscard]] inline std::uint64_t peekCallId(std::string_view json, std::size_t maxScanBytes = 1024) noexcept {
235 static constexpr std::string_view kKey = "\"callId\":";
236 const std::string_view window = json.substr(0, std::min(json.size(), maxScanBytes));
237 const auto keyPos = window.find(kKey);
238 if (keyPos == std::string_view::npos) {
239 return 0;
240 }
241 std::size_t idx = keyPos + kKey.size();
242 // idx is bounded by the same condition that guards the access.
243 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
244 while (idx < window.size() && (window[idx] == ' ' || window[idx] == '\t')) {
245 ++idx;
246 }
247 std::uint64_t value = 0;
248 bool sawDigit = false;
249 for (; idx < window.size(); ++idx) {
250 // idx is bounded by the loop condition.
251 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
252 const char chr = window[idx];
253 if (chr < '0' || chr > '9') {
254 break;
255 }
256 const auto digit = static_cast<std::uint64_t>(chr - '0');
257 if (value > (std::numeric_limits<std::uint64_t>::max() - digit) / 10U) {
258 return 0; // out of range — treat as unrecoverable rather than wrap
259 }
260 value = (value * 10U) + digit;
261 sawDigit = true;
262 }
263 return sawDigit ? value : 0;
264}

294/// @brief Encodes @p env as a single JSON line.
295///
296/// Writes with `detail::EscapingWriteOpts`, so raw ASCII control bytes in any
297/// string field become `\uXXXX` escapes. Without it, a single 0x0B anywhere in
298/// `body`, `modelType`, `contextKey`, `typeId`, `actionType`, or the session's
299/// `principal`/`token` yields an envelope the peer's `decode` throws on — and,
300/// where the same string also contains a `\` or `"`, one glaze silently
301/// corrupts before it is ever sent. Both are payload-triggered transport
302/// failures, not caller errors. The escape is lossless, so such bytes still
303/// round-trip byte-for-byte.
304///
305/// @param env Envelope to serialize.
306/// @return The serialized envelope as valid, re-decodable JSON.
307/// @throws std::runtime_error on serialisation failure (should never happen for valid input).
308inline std::string encode(const Envelope& env) {
309 std::string out;
310 if (auto errCode = glz::write<detail::EscapingWriteOpts{}>(env, out)) {
311 throw std::runtime_error("envelope encode failed: " + glz::format_error(errCode, out));
312 }
313 return out;
314}

316/// @brief Decodes @p json into an `Envelope`.
317///
318/// Enforces a wire-layer size cap: inputs longer than `kMaxEnvelopeBytes` are
319/// rejected before any parsing, bounding the cost of an oversized or
320/// deeply-nested payload smuggled inside the opaque `body` string (see
321/// `kMaxEnvelopeBytes`). This guard does **not** cap the *inner* re-parse of
322/// `body` performed later by the action codec — that codec must impose its own
323/// limits — but it does bound the total message a single `decode` will accept.
324///
325/// glaze 7.4 runs with its default options here. Note the parser does **not**
326/// reject duplicate JSON object keys: a duplicated key is accepted and the last
327/// occurrence wins. There is no glaze option to change this, so `decode` cannot
328/// enforce rejection; callers must not treat duplicate-key rejection as a
329/// security boundary (see docs/spec/wire.md).
330///
331/// @param json Serialized envelope JSON to decode.
332/// @return The decoded `Envelope`.
333/// @throws std::runtime_error if @p json exceeds `kMaxEnvelopeBytes` or is not a
334/// valid envelope.
335inline Envelope decode(std::string_view json) {
336 if (json.size() > kMaxEnvelopeBytes) {
337 throw std::runtime_error("envelope decode failed: input exceeds maximum size (" + std::to_string(json.size()) +
338 " > " + std::to_string(kMaxEnvelopeBytes) + " bytes)");
339 }
340 Envelope env{};
341 // Ignore unknown keys so the envelope is forward-compatible: a newer peer may
342 // add fields a older peer does not know, and vice versa, without a hard parse
343 // failure. Duplicate keys are still accepted last-wins (glaze offers no reject
344 // option) — see docs/spec/wire.md "Parsing guarantees and hardening".
345 // `null_terminated = false` is required, not cosmetic: `json` is a caller-supplied
346 // `string_view` with no guarantee of a trailing '\0' (e.g. a raw socket read).
347 // Left at glaze's default (true), `skip_ws` assumes it can scan past `end`
348 // looking for a terminator that may not exist — a heap-buffer-overflow on
349 // adversarial input (found by the wire_decode fuzz harness, see
350 // docs/spec/testing_strategy.md). This costs nothing: it only disables an
351 // optimization that never applied to us.
352 static constexpr glz::opts kLenient{.null_terminated = false, .error_on_unknown_keys = false};
353 if (auto errCode = glz::read<kLenient>(env, json)) {
354 throw std::runtime_error("envelope decode failed: " + glz::format_error(errCode, json));
355 }
356 return env;
357}

359/// @brief Outcome of a `"hello"` protocol-version negotiation attempt.
360enum class ProtocolNegotiationResult : std::uint8_t {
361 /// @brief The peer understood `"hello"` and accepted the announced version.
362 Negotiated,
363 /// @brief The peer does not understand `"hello"` (a pre-negotiation legacy peer).
364 LegacyPeer,
365};

367/// @brief Classifies a decoded reply to a `"hello"` envelope.
368///
369/// @param reply Decoded reply envelope — the result of `decode()` on the
370/// response to a `"hello"` round-trip.
371/// @return `Negotiated` if @p reply's `kind` is `"ok"`; `LegacyPeer` if it is
372/// an `"err"` whose `message` is exactly `"unknown envelope kind:
373/// hello"` — the generic unrecognised-`kind` message a pre-negotiation
374/// `RemoteServer` produces for a `kind` it does not switch on.
375/// @throws std::runtime_error if @p reply is any other `"err"` (e.g.
376/// `"protocol version unsupported"`), so the caller refuses to
377/// proceed instead of silently treating an explicit rejection as a
378/// legacy peer.
379inline ProtocolNegotiationResult interpretHelloReply(const Envelope& reply) {
380 if (reply.kind == "ok") {
381 return ProtocolNegotiationResult::Negotiated;
382 }
383 if (reply.message == "unknown envelope kind: hello") {
384 return ProtocolNegotiationResult::LegacyPeer;
385 }
386 throw std::runtime_error("protocol negotiation failed: " +
387 (reply.message.empty() ? std::string{"malformed reply"} : reply.message));
388}
