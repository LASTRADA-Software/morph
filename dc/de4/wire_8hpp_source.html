14/// @brief JSON wire envelope used between any client and `RemoteServer`.
15///
16/// Supersedes the legacy pipe-delimited protocol: a single struct carries all
17/// request and reply variants, with the `kind` field acting as a discriminator.
18/// Empty/zero fields are tolerated — callers populate only what their kind needs.
19///
20/// @par Discriminator values
21/// - `"register"` — client requests model creation. Uses `typeId`, and
22/// optionally `contextKey` (the new instance's stable
23/// identity, e.g. an account id — see `RemoteServer::setLogProvider`).
24/// - `"deregister"` — client destroys an instance. Uses `modelId`.
25/// - `"execute"` — client dispatches an action. Uses `callId`, `modelId`,
26/// `modelType`, `actionType`, `body`, and optionally `session`.
27/// - `"ok"` — server success reply. Uses `callId` if present and `result`.
28/// For `register` replies, `modelId` carries the new id.
29/// - `"err"` — server failure reply. Uses `callId` if present and `message`.
30struct Envelope {
31 /// @brief Discriminator — see class docstring for valid values.
32 std::string kind;
33
34 /// @brief Correlation id for matching async replies to their requests.
35 uint64_t callId = 0;
36
37 /// @brief Model type id for `register`.
38 std::string typeId;
39
40 /// @brief Stable identity of the model instance being registered (e.g. an
41 /// account id). Empty means "no identity" — the server-side holder
42 /// gets no action log attached even if a `LogProvider` is configured.
43 /// Ignored on every kind other than `register`.
44 std::string contextKey;
45
46 /// @brief Existing model instance id for `deregister`, `execute`, `ok(register)`.
47 uint64_t modelId = 0;
48
49 /// @brief Model type id for `execute` (routing key in `ActionDispatcher`).
50 std::string modelType;
51
52 /// @brief Action type id for `execute` (the second routing key).
53 std::string actionType;
54
55 /// @brief Serialized JSON of the action payload (`execute`) or of the
56 /// deserializable result (`ok`).
57 std::string body;
58
59 /// @brief Free-text error message on `err`. Empty on success.
60 std::string message;
61
62 /// @brief Session context for authorization and routing. Populated on
63 /// `execute`; ignored on every other kind.
64 ::morph::session::Context session;
65};
66
67/// @brief Builds a `register` envelope.
68/// @param typeId Model type id to register.
69/// @param contextKey Optional stable identity for the new instance (default: none).
70inline Envelope makeRegister(std::string typeId, std::string contextKey = {}) {
71 Envelope env;
72 env.kind = "register";
73 env.typeId = std::move(typeId);
74 env.contextKey = std::move(contextKey);
75 return env;
76}
77
78/// @brief Builds a `deregister` envelope.
79inline Envelope makeDeregister(uint64_t modelId) {
80 Envelope env;
81 env.kind = "deregister";
82 env.modelId = modelId;
83 return env;
84}
85
86/// @brief Builds an `ok` reply envelope. `body` carries the result for executes,
87/// `modelId` carries the new id for register-replies.
88inline Envelope makeOk(uint64_t callId = 0, std::string body = {}, uint64_t modelId = 0) {
89 Envelope env;
90 env.kind = "ok";
91 env.callId = callId;
92 env.body = std::move(body);
93 env.modelId = modelId;
94 return env;
95}
96
97/// @brief Builds an `err` reply envelope.
98inline Envelope makeErr(std::string message, uint64_t callId = 0) {
99 Envelope env;
100 env.kind = "err";
101 env.callId = callId;
102 env.message = std::move(message);
103 return env;
104}
105
106/// @brief Encodes @p env as a single JSON line.
107/// @throws std::runtime_error on serialisation failure (should never happen for valid input).
108inline std::string encode(const Envelope& env) {
109 std::string out;
110 if (auto errCode = glz::write_json(env, out)) {
111 throw std::runtime_error("envelope encode failed: " + glz::format_error(errCode, out));
112 }
113 return out;
114}
115
116/// @brief Decodes @p json into an `Envelope`.
117/// @throws std::runtime_error if @p json is not a valid envelope.
118inline Envelope decode(std::string_view json) {
119 Envelope env{};
120 if (auto errCode = glz::read_json(env, json)) {
121 throw std::runtime_error("envelope decode failed: " + glz::format_error(errCode, json));
122 }
123 return env;
124}
