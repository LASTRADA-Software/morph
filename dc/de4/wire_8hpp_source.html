15/// @brief Maximum accepted serialized size, in bytes, of a single wire envelope.
16///
17/// `decode` rejects any input longer than this before handing it to the JSON
18/// parser. The cap is a wire-layer denial-of-service backstop: the outer decode
19/// treats `body` as one opaque string, so a deeply-nested or enormous JSON
20/// payload smuggled *inside* `body` is invisible to any structural/depth check
21/// and would only detonate when the action codec re-parses it later (the "body
22/// double-parse"). Bounding the total message length caps that cost up
23/// front. The default of 8 MiB is generous for legitimate action payloads while
24/// keeping a single message's peak allocation bounded; transports that want a
25/// tighter bound should enforce it before calling `decode`.
26inline constexpr std::size_t kMaxEnvelopeBytes = 8u * 1024u * 1024u;
27
28/// @brief JSON wire envelope used between any client and `RemoteServer`.
29///
30/// Supersedes the legacy pipe-delimited protocol: a single struct carries all
31/// request and reply variants, with the `kind` field acting as a discriminator.
32/// Empty/zero fields are tolerated — callers populate only what their kind needs.
33///
34/// @par Discriminator values
35/// - `"register"` — client requests model creation. Uses `typeId`, and
36/// optionally `contextKey` (the new instance's stable
37/// identity, e.g. an account id — see `RemoteServer::setLogProvider`).
38/// - `"deregister"` — client destroys an instance. Uses `modelId`.
39/// - `"execute"` — client dispatches an action. Uses `callId`, `modelId`,
40/// `modelType`, `actionType`, `body`, and optionally `session`.
41/// - `"ok"` — server success reply. Uses `callId` if present and carries
42/// the serialized result in `body`. For `register` replies,
43/// `modelId` carries the new id.
44/// - `"err"` — server failure reply. Uses `callId` if present and `message`.
45struct Envelope {
46 /// @brief Discriminator — see class docstring for valid values.
47 std::string kind;
48
49 /// @brief Correlation id for matching async replies to their requests.
50 uint64_t callId = 0;
51
52 /// @brief Model type id for `register`.
53 std::string typeId;
54
55 /// @brief Stable identity of the model instance being registered (e.g. an
56 /// account id). Empty means "no identity" — the server-side holder
57 /// gets no action log attached even if a `LogProvider` is configured.
58 /// Ignored on every kind other than `register`.
59 std::string contextKey;
60
61 /// @brief Existing model instance id for `deregister`, `execute`, `ok(register)`.
62 uint64_t modelId = 0;
63
64 /// @brief Model type id for `execute` (routing key in `ActionDispatcher`).
65 std::string modelType;
66
67 /// @brief Action type id for `execute` (the second routing key).
68 std::string actionType;
69
70 /// @brief Serialized JSON of the action payload (`execute`) or of the
71 /// deserializable result (`ok`).
72 std::string body;
73
74 /// @brief Free-text error message on `err`. Empty on success.
75 std::string message;
76
77 /// @brief Session context for authorization and routing. Populated on
78 /// `execute`; ignored on every other kind.
79 ::morph::session::Context session;
80};

82/// @brief Builds a `register` envelope.
83/// @param typeId Model type id to register.
84/// @param contextKey Optional stable identity for the new instance (default: none).
85inline Envelope makeRegister(std::string typeId, std::string contextKey = {}) {
86 Envelope env;
87 env.kind = "register";
88 env.typeId = std::move(typeId);
89 env.contextKey = std::move(contextKey);
90 return env;
91}

121/// @brief Encodes @p env as a single JSON line.
122/// @throws std::runtime_error on serialisation failure (should never happen for valid input).
123inline std::string encode(const Envelope& env) {
124 std::string out;
125 if (auto errCode = glz::write_json(env, out)) {
126 throw std::runtime_error("envelope encode failed: " + glz::format_error(errCode, out));
127 }
128 return out;
129}

131/// @brief Decodes @p json into an `Envelope`.
132///
133/// Enforces a wire-layer size cap: inputs longer than `kMaxEnvelopeBytes` are
134/// rejected before any parsing, bounding the cost of an oversized or
135/// deeply-nested payload smuggled inside the opaque `body` string (see
136/// `kMaxEnvelopeBytes`). This guard does **not** cap the *inner* re-parse of
137/// `body` performed later by the action codec — that codec must impose its own
138/// limits — but it does bound the total message a single `decode` will accept.
139///
140/// glaze 7.4 runs with its default options here. Note the parser does **not**
141/// reject duplicate JSON object keys: a duplicated key is accepted and the last
142/// occurrence wins. There is no glaze option to change this, so `decode` cannot
143/// enforce rejection; callers must not treat duplicate-key rejection as a
144/// security boundary (see docs/spec/wire.md).
145///
146/// @param json Serialized envelope JSON to decode.
147/// @return The decoded `Envelope`.
148/// @throws std::runtime_error if @p json exceeds `kMaxEnvelopeBytes` or is not a
149/// valid envelope.
150inline Envelope decode(std::string_view json) {
151 if (json.size() > kMaxEnvelopeBytes) {
152 throw std::runtime_error("envelope decode failed: input exceeds maximum size (" +
153 std::to_string(json.size()) + " > " +
154 std::to_string(kMaxEnvelopeBytes) + " bytes)");
155 }
156 Envelope env{};
157 // Ignore unknown keys so the envelope is forward-compatible: a newer peer may
158 // add fields a older peer does not know, and vice versa, without a hard parse
159 // failure. Duplicate keys are still accepted last-wins (glaze offers no reject
160 // option) — see docs/spec/wire.md "Parsing guarantees and hardening".
161 static constexpr glz::opts kLenient{.error_on_unknown_keys = false};
162 if (auto errCode = glz::read<kLenient>(env, json)) {
163 throw std::runtime_error("envelope decode failed: " + glz::format_error(errCode, json));
164 }
165 return env;
166}
