36/// @brief One open tab, as an application's tab menu or a test reads it.

46/// @brief Where a workspace draws the active screen's head row.
47enum class HeadPlacement : std::uint8_t {
48 /// @brief Above each page, inside the workspace.
49 InPage,
50 /// @brief Nowhere: the host mounts `headRowForActive()` where it chooses, and a page holds only its screen.

54/// @brief Tabs over screens: a tab bar over a `WorkspaceModel`, one screen instance per tab, mounted into the tab's
55/// page through `DocumentScreenHost::mountInto`, with the screen's head row above it or, when the placement is
56/// `External`, mounted by the host instead.
57///
58/// The model holds the tab logic: tabs keep the order they were opened in, opening a screen whose identity an open
59/// tab already has focuses that tab, a singleton screen has one tab, and closing runs each screen's close protocol
60/// (spec 6 §3), one tab at a time, a walk over several tabs stopping at the first veto. The bar shows the model's tabs
61/// and takes Ctrl+Tab, Ctrl+Shift+Tab and Ctrl+W, which documents may not declare. Neither copyable nor movable;
62/// owned by the owner thread of its runtime.
63class Workspace {
64public:
65 /// @brief The shell, its tab bar appended to @p parent.
66 /// @param runtime The runtime the screens' reactive nodes belong to. Borrowed.
67 /// @param backend The backend the bar and the pages are made with. Borrowed.
68 /// @param host What mounts the screens. Borrowed; it outlives the shell.
69 /// @param options The screens' catalog facts.
70 /// @param parent The container the shell is appended to; null makes it a backend root.
71 /// @param headPlacement Where the active screen's head row is drawn. `External` leaves the host to mount
72 /// `headRowForActive()` itself, so a shell with its own top bar shows the head once.
73 Workspace(reactive::Runtime& runtime MORPH_LIFETIMEBOUND, IViewBackend& backend MORPH_LIFETIMEBOUND,
74 DocumentScreenHost& host MORPH_LIFETIMEBOUND, WorkspaceOptions options = {},
75 ContainerWidget* parent = nullptr, HeadPlacement headPlacement = HeadPlacement::InPage)
76 : _runtime{&runtime},
77 _backend{&backend},
78 _host{&host},
79 _bar{backend.createTabs(parent)},
80 _hosts{options.hosts},
81 _screens{options.screens},
82 _onChanged{std::move(options.onChanged)},
83 _headPlacement{headPlacement} {
84 // The bar and each page fill what they are given, so a screen whose root stretches fills the page.
85 _bar->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
86 _bar->setTabs({});
87 _bar->setSelected(kNone);
88 _bar->setOnSelect([this](std::size_t index) {
89 std::vector<WorkspaceTabId> const order = _model->order();
90 if (index < order.size()) {
91 _model->activate(order.at(index));
92 }
93 });
94 _bar->setKeys({"Ctrl+Tab", "Ctrl+Shift+Tab", "Ctrl+W"}, [this](std::string const& chord) {
95 if (chord == "Ctrl+Tab") {
96 next();
97 } else if (chord == "Ctrl+Shift+Tab") {
98 previous();
99 } else if (std::optional<std::size_t> const shown = active()) {
100 close(*shown, {});
101 }
102 });
103 WorkspaceModelOptions modelOptions{std::move(options), {}, {}};
104 modelOptions.onChanged = [this] { showTabs(); };
105 modelOptions.mountPage = [this](WorkspaceTabId tab, std::string_view screenId, ParamMap const& params) {
106 return mountPage(tab, screenId, params);
107 };
108 modelOptions.unmountPage = [this](WorkspaceTabId tab) {
109 std::erase(_shownOrder, tab);
110 _pages.erase(tab.value);
111 };
112 _model = std::make_unique<WorkspaceModel>(runtime, host, std::move(modelOptions));
113 }
114
115 /// @brief Unmounts every screen without asking, newest first, then the pages.
116 ~Workspace() { _model.reset(); }
117
118 Workspace(Workspace const&) = delete;
119 Workspace& operator=(Workspace const&) = delete;
120 Workspace(Workspace&&) = delete;
121 Workspace& operator=(Workspace&&) = delete;

230 /// @brief The active tab's head row, the same reactive content `headRow` builds for the body-embedded rendering,
231 /// for a host that mounts it elsewhere (a shell's own top bar, say). Never null: an invisible, empty row
232 /// stands in for "no active tab" rather than a null `Node`, since a caller mounting this may not expect to
233 /// handle one.
234 /// @return The row, reactive to the active tab changing and to its head's own content changing.
235 [[nodiscard]] Node headRowForActive() const {
236 std::optional<WorkspaceTabId> const tab = _model->active().peek();
237 if (tab) {
238 return ui::row(buildHeadRow(*tab));
239 }
240 Row empty{.children = {}, .gap = 0, .common = {}, .tone = std::nullopt};
241 empty.common.testId = "head";
242 empty.common.visible = false;
243 return ui::row(std::move(empty));

250 /// @brief The open tabs, in order.
251 /// @return Each tab's screen, identity and caption.
252 [[nodiscard]] std::vector<WorkspaceTab> tabs() const {
253 std::vector<WorkspaceTab> out;
254 for (WorkspaceTabId const tab : _model->order()) {
255 ScreenHandle const& handle = *_model->handleOf(tab);
256 out.push_back(WorkspaceTab{.screenId = std::string{screenOf(tab)},

258 .caption = captionOf(handle)});
259 }
260 return out;

291 [[nodiscard]] std::optional<WorkspaceTabId> tabAt(std::size_t index) const {
292 std::vector<WorkspaceTabId> const order = _model->order();
293 return index < order.size() ? std::optional<WorkspaceTabId>{order.at(index)} : std::nullopt;

296 [[nodiscard]] std::size_t indexOf(WorkspaceTabId tab) const {
297 std::vector<WorkspaceTabId> const order = _model->order();
298 auto const found = std::ranges::find(order, tab);
299 return found == order.end() ? kNone : static_cast<std::size_t>(found - order.begin());
300 }

302 [[nodiscard]] std::string_view screenOf(WorkspaceTabId tab) const {
303 auto const found = _pages.find(tab.value);
304 return found == _pages.end() ? std::string_view{} : std::string_view{found->second->screenId};
305 }

315 // The model's mountPage: a page slot at the end of the bar, the screen mounted into it, a host screen from the
316 // registry and a document screen from the document host, then the head row above it. The page goes with a
317 // refused mount.
318 std::expected<std::unique_ptr<ScreenHandle>, MountError> mountPage(WorkspaceTabId tab, std::string_view screenId,
319 ParamMap const& params) {
320 auto page = std::make_unique<Page>();
321 page->screenId = std::string{screenId};
322 page->slot = _backend->createSlot(_bar.get());
323 page->slot->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
324 page->slot->setVisible(false);
325 std::expected<std::unique_ptr<ScreenHandle>, MountError> handle;
326 WorkspaceScreen const facts = _screens ? _screens(screenId) : WorkspaceScreen{};
327 if (facts.host.empty()) {
328 handle = _host->mountInto(screenId, params, *page->slot);
329 } else {
330 HostScreenFactory const* const factory = _hosts == nullptr ? nullptr : _hosts->find(facts.host);
331 if (factory == nullptr) {
332 return std::unexpected(MountError{.screenId = std::string{screenId},
333 .reason = "no host screen `" + facts.host + "` is registered"});
334 }
335 handle = (*factory)(params, *page->slot);
336 }
337 if (!handle) {
338 return handle;
339 }
340 if (_headPlacement == HeadPlacement::InPage && (*handle)->head() != nullptr) {
341 // Mounted after the screen, since it reads the handle, and moved above it.
342 page->head.emplace(*_runtime, *_backend, headRow(tab), page->slot.get());
343 page->slot->moveChild(page->head->root(), 0);
344 }
345 _pages.insert_or_assign(tab.value, std::move(page));
346 _shownOrder.push_back(tab);
347 return handle;
348 }

350 // The head of a tab's screen, while the tab is open. A tracked read of the model's tabs, so a binding made while
351 // the model is still opening the tab reads the head once it holds it.
352 [[nodiscard]] ScreenHead* headOf(WorkspaceTabId tab) const {
353 static_cast<void>(_model->tabs().get());
354 ScreenHandle* const handle = _model->handleOf(tab);
355 return handle == nullptr ? nullptr : handle->head();
356 }

358 // The head row's own content, built but not yet wrapped into an immutable Node: the subtitle (muted), the
359 // badges, a spacer and the actions as buttons. Hidden while the head has none of them, so a screen without a
360 // head looks as it would without the row. Factored out of headRow() so a caller needing to set additional
361 // layout fields on the row -- which a Node, once built, can no longer be mutated to carry -- can do so on the
362 // still-mutable Row this returns, before wrapping it itself.
363 [[nodiscard]] Row buildHeadRow(WorkspaceTabId tab) const {
364 auto const subtitle = [this, tab] {
365 ScreenHead const* const head = headOf(tab);
366 return head == nullptr ? std::string{} : head->subtitle().get();
367 };
368 auto const status = [this, tab] {
369 ScreenHead const* const head = headOf(tab);
370 return head == nullptr ? std::vector<HeadBadge>{} : head->status().get();
371 };
372 auto const actions = [this, tab] {
373 ScreenHead const* const head = headOf(tab);
374 return head == nullptr ? std::vector<HeadAction>{} : head->actions().get();
375 };
376 Row row{.children = {}, .gap = 2, .common = {}, .tone = std::nullopt};
377 row.common.testId = "head";
378 // Full width, so the actions sit at the trailing edge in either placement.
379 row.common.layout.width = Sizing::stretch();
380 row.common.visible = Prop<bool>{
381 [subtitle, status, actions] { return !subtitle().empty() || !status().empty() || !actions().empty(); }};
382 Text text{.text = Prop<std::string>{subtitle}, .role = TextRole::Muted, .field = {}, .common = {}};
383 text.common.testId = "head_subtitle";
384 text.common.visible = Prop<bool>{[subtitle] { return !subtitle().empty(); }};
385 row.children.push_back(ui::text(std::move(text)));
386 row.children.push_back(ui::forEach<PlacedBadge>(
387 [status] {
388 std::vector<PlacedBadge> placed;
389 for (HeadBadge const& badge : status()) {
390 placed.push_back(PlacedBadge{.index = placed.size(), .badge = badge});
391 }
392 return placed;
393 },
394 [](PlacedBadge const& placed) { return Key{static_cast<std::int64_t>(placed.index)}; },
395 [](reactive::Signal<PlacedBadge> const& placed) {
396 Badge badge{.tone = Prop<Tone>{[&placed] { return placed.get().badge.tone; }},
397 .text = Prop<std::string>{[&placed] { return placed.get().badge.text; }},
398 .icon = placed.peek().badge.icon,
399 .common = {}};
400 badge.common.testId = "head_status_" + std::to_string(placed.peek().index);
401 return ui::badge(std::move(badge));
402 },
403 Axis::Horizontal, 1));
404 // The spacer stretches, which is what makes the row lay out at full width and pushes the actions to the end.
405 Spacer filler;
406 filler.common.testId = "head_spacer";
407 filler.common.layout.width = Sizing::stretch();
408 row.children.push_back(ui::spacer(std::move(filler)));
409 row.children.push_back(ui::forEach<HeadAction>(
410 actions, [](HeadAction const& action) { return Key{action.id}; },
411 [this, tab](reactive::Signal<HeadAction> const& action) {
412 Button button{.label = Prop<std::string>{[&action] { return action.get().label; }},
413 .emphasis = action.peek().emphasis,
414 .onClick =
415 [this, tab, actionId = action.peek().id] {
416 ScreenHandle* const handle = _model->handleOf(tab);
417 if (ScreenHead* const head = handle == nullptr ? nullptr : handle->head()) {
418 head->run(actionId);
419 }
420 },
421 .common = {}};
422 button.common.testId = "head_action_" + action.peek().id;
423 button.common.enabled = Prop<bool>{[&action] { return action.get().enabled; }};
424 return ui::button(std::move(button));
425 },
426 Axis::Horizontal, 1));
427 return row;
428 }

433 // The model changed: the bar shows its tabs in order, with their captions, and the active tab's page.
434 void showTabs() {
435 if (_model == nullptr) {
436 return;
437 }
438 std::vector<WorkspaceTabId> const order = _model->order();
439 std::optional<WorkspaceTabId> const shown = _model->active().peek();
440 std::vector<std::string> captions;
441 captions.reserve(order.size());
442 for (std::size_t index = 0; index < order.size(); ++index) {
443 auto const found = _pages.find(order.at(index).value);
444 if (found == _pages.end()) {
445 continue;
446 }
447 Page& page = *found->second;
448 // A tab the model moved (a reuseEmpty replacement) moves its page in the bar too; `_shownOrder` is the
449 // bar's own order of the pages.
450 if (index >= _shownOrder.size() || _shownOrder.at(index) != order.at(index)) {
451 std::erase(_shownOrder, order.at(index));
452 _shownOrder.insert(
453 _shownOrder.begin() + static_cast<std::ptrdiff_t>(std::min(index, _shownOrder.size())),
454 order.at(index));
455 _bar->moveChild(*page.slot, index);
456 }
457 page.slot->setVisible(shown == order.at(index));
458 captions.push_back(captionOf(*_model->handleOf(order.at(index))));
459 }
460 _bar->setTabs(captions);
461 _bar->setSelected(shown ? indexOf(*shown) : kNone);
462 // One tab needs no strip to choose it by; the chords still work with the strip hidden.
463 _bar->setBarShown(captions.size() > 1);
464 if (_onChanged) {
465 _onChanged();
466 }
467 }
468
469 reactive::Runtime* _runtime;
470 IViewBackend* _backend;
471 DocumentScreenHost* _host;
472 std::unique_ptr<TabsWidget> _bar;
473 ScreenHostRegistry const* _hosts = nullptr;
474 std::function<WorkspaceScreen(std::string_view)> _screens;
475 std::function<void()> _onChanged;
476 HeadPlacement _headPlacement = HeadPlacement::InPage;
477 std::map<std::uint64_t, std::unique_ptr<Page>> _pages;
478 // The pages in the bar's order.
479 std::vector<WorkspaceTabId> _shownOrder;
480 std::unique_ptr<WorkspaceModel> _model;
481};
