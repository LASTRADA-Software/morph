59/// @brief The parts of the active screen's head, for a host that lays them out itself.
60struct HeadParts {
61 /// @brief The eyebrow, muted, the line a host draws above the title; hidden while the head has none.

75/// @brief Tabs over screens: a tab bar over a `WorkspaceModel`, one screen instance per tab, mounted into the tab's
76/// page through `DocumentScreenHost::mountInto`, with the screen's head row above it or, when the placement is
77/// `External`, mounted by the host instead.
78///
79/// The model holds the tab logic: tabs keep the order they were opened in, opening a screen whose identity an open
80/// tab already has focuses that tab, a singleton screen has one tab, and closing runs each screen's close protocol
81/// (spec 6 §3), one tab at a time, a walk over several tabs stopping at the first veto. The bar shows the model's tabs
82/// and takes Ctrl+Tab, Ctrl+Shift+Tab and Ctrl+W, which documents may not declare. A screen a document navigates to as
83/// a dialog or a drawer is shown over the tabs (`openOverlay`). Neither copyable nor movable; owned by the owner
84/// thread of its runtime.
85class Workspace {
86public:
87 /// @brief The shell, its tab bar appended to @p parent.
88 /// @param runtime The runtime the screens' reactive nodes belong to. Borrowed.
89 /// @param backend The backend the bar and the pages are made with. Borrowed.
90 /// @param host What mounts the screens. Borrowed; it outlives the shell.
91 /// @param options The screens' catalog facts.
92 /// @param parent The container the shell is appended to; null makes it a backend root.
93 /// @param headPlacement Where the active screen's head row is drawn. `External` leaves the host to mount
94 /// `headPartsForActive()` itself, so a shell with its own top bar shows the head once.
95 Workspace(reactive::Runtime& runtime MORPH_LIFETIMEBOUND, IViewBackend& backend MORPH_LIFETIMEBOUND,
96 DocumentScreenHost& host MORPH_LIFETIMEBOUND, WorkspaceOptions options = {},
97 ContainerWidget* parent = nullptr, HeadPlacement headPlacement = HeadPlacement::InPage)
98 : _runtime{&runtime},
99 _backend{&backend},
100 _host{&host},
101 _parent{parent},
102 _bar{backend.createTabs(parent)},
103 _hosts{options.hosts},
104 _screens{options.screens},
105 _onChanged{std::move(options.onChanged)},
106 _diagnostics{options.diagnostics},
107 _headPlacement{headPlacement} {
108 // The bar and each page fill what they are given, so a screen whose root stretches fills the page.
109 _bar->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
110 _bar->setTabs({});
111 _bar->setSelected(kNone);
112 _bar->setOnSelect([this](std::size_t index) {
113 std::vector<WorkspaceTabId> const order = _model->order();
114 if (index < order.size()) {
115 _model->activate(order.at(index));
116 }
117 });
118 _bar->setKeys({"Ctrl+Tab", "Ctrl+Shift+Tab", "Ctrl+W"}, [this](std::string const& chord) {
119 if (chord == "Ctrl+Tab") {
120 next();
121 } else if (chord == "Ctrl+Shift+Tab") {
122 previous();
123 } else if (std::optional<std::size_t> const shown = active()) {
124 close(*shown, {});
125 }
126 });
127 WorkspaceModelOptions modelOptions{std::move(options), {}, {}};
128 modelOptions.onChanged = [this] { showTabs(); };
129 modelOptions.mountPage = [this](WorkspaceTabId tab, std::string_view screenId, ParamMap const& params) {
130 return mountPage(tab, screenId, params);
131 };
132 modelOptions.unmountPage = [this](WorkspaceTabId tab) {
133 std::erase(_shownOrder, tab);
134 _pages.erase(tab.value);
135 };
136 _model = std::make_unique<WorkspaceModel>(runtime, host, std::move(modelOptions));
137 }

181 /// @brief Shows a document screen over the tabs, as a dialog (`as: dialog`) or a drawer on the end side
182 /// (`as: drawer`), mounted with `DocumentScreenHost::mountInto`, so the mount adopts the navigation's call.
183 ///
184 /// It always mounts a new instance: it never focuses an open tab, so a call is never left unclaimed. The
185 /// overlay is titled by the screen's title, and the screen's head actions are its footer's buttons; its badges,
186 /// menu and tools are not shown, and a head with a menu or tools is reported once. A dialog is a sheet while the
187 /// width class is `compact`, and follows `setWidthClass` while it is open. A user's dismissal opens it again at
188 /// once and asks the screen to close (`requestClose`); it goes only when the screen closed. A screen that closes
189 /// itself goes through `screenClosed`. An overlay adds no tab, no record entry and no route.
190 /// @param request The navigation: a screen, its params, and `as` either `dialog` or `drawer`.
191 /// @return Nothing, or why nothing was shown: an `as` that is not an overlay, a host screen, or a refused mount.
192 std::expected<void, MountError> openOverlay(NavigateRequest const& request) {
193 if (request.as != "dialog" && request.as != "drawer") {
194 return std::unexpected(MountError{.screenId = request.screen,
195 .reason = "`as: " + request.as + "` is not a dialog or a drawer"});
196 }
197 if (_screens && !_screens(request.screen).host.empty()) {
198 return std::unexpected(
199 MountError{.screenId = request.screen, .reason = "a host screen opens as a tab, not an overlay"});
200 }
201 ValueRecord const* const record = request.params.asRecord();
202 ParamMap const params = record == nullptr ? ParamMap{} : *record;
203 auto overlay = std::make_unique<Overlay>();
204 overlay->id = ++_lastOverlay;
205 overlay->screenId = request.screen;
206 ContainerWidget* content = nullptr;
207 ContainerWidget* footer = nullptr;
208 if (request.as == "drawer") {
209 overlay->drawer = _backend->createDrawer(_parent, Side::End);
210 content = overlay->drawer.get();
211 footer = &overlay->drawer->footer();
212 } else {
213 overlay->dialog = _backend->createDialog(_parent);
214 content = overlay->dialog.get();
215 footer = &overlay->dialog->footer();
216 }
217 auto handle = _host->mountInto(request.screen, params, *content);
218 if (!handle) {
219 return std::unexpected(std::move(handle.error()));
220 }
221 overlay->handle = std::move(*handle);
222 showOverlay(*overlay, *footer);
223 _overlays.push_back(std::move(overlay));
224 return {};
225 }

227 /// @brief A screen the workspace shows closed itself, by `return` or `close: self`: its overlay goes, or its tab
228 /// does. What a host's `DocumentScreenHost::onScreenClosed` forwards here.
229 /// @param screen The closed screen's handle; one the workspace does not show does nothing.
230 void screenClosed(ScreenHandle const& screen) {
231 auto const overlay = std::ranges::find_if(
232 _overlays, [&screen](std::unique_ptr<Overlay> const& open) { return open->handle.get() == &screen; });
233 if (overlay != _overlays.end()) {
234 _overlays.erase(overlay);
235 return;
236 }
237 for (WorkspaceTabId const tab : _model->order()) {
238 if (_model->handleOf(tab) == &screen) {
239 // The screen is closed, so its close answers at once and the tab goes.
240 _model->close(tab, {});
241 return;
242 }
243 }
244 }

409 /// @brief What ends the active tab's head trail, so a compact top bar can stand the line's end the right distance
410 /// from the window's edge. A tracked read.

412 /// trigger is an icon button, or a last action that is icon-only or ghost is `Unframed`; any other last

594 // Titles a mounted overlay, puts its head's actions in its footer, opens it and makes a dismissal ask its screen.
595 void showOverlay(Overlay& overlay, ContainerWidget& footer) {
596 ScreenHandle const* const handle = overlay.handle.get();
597 ScreenHead* const head = overlay.handle->head();
598 overlay.scope = std::make_unique<reactive::Scope>(*_runtime);
599 overlay.scope->effect([&overlay, handle] {

601 if (overlay.dialog) {
602 overlay.dialog->setTitle(title);
603 } else {
604 overlay.drawer->setTitle(title);
605 }
606 });
607 if (head != nullptr) {
608 overlay.footer.emplace(*_runtime, *_backend, overlayActions(*head), &footer);
609 overlay.scope->effect([this, &overlay, head] {
610 bool const menu = head->menu() != nullptr && !head->menu()->get().empty();
611 bool const tools = head->tools() != nullptr && !head->tools()->get().empty();
612 if ((menu || tools) && !overlay.reported) {
613 overlay.reported = true;
614 _runtime->untracked([this, &overlay] {
615 if (_diagnostics) {
616 _diagnostics(Refusal{.document = overlay.screenId,
617 .path = {},
618 .reason = "a dialog or drawer does not show the head menu or tools "
619 "of " +
620 overlay.screenId});
621 }
622 });
623 }
624 });
625 }
626 showSheet(overlay, _backend->widthClass() == "compact");
627 std::uint64_t const overlayId = overlay.id;

630 if (shown == nullptr) {
631 return;
632 }

634 // close protocol runs, and goes only when the screen closed: a dirty screen without a guard stays.
635 shown->setOpen(true);

637 if (auto const live = alive.lock(); live != nullptr && *live && closed) {
638 removeOverlay(overlayId);
639 }
640 });
641 };
642 if (overlay.dialog) {
643 overlay.dialog->setOnDismiss(dismissed);
644 } else {
645 overlay.drawer->setOnDismiss(dismissed);
646 }
647 overlay.setOpen(true);
648 }

704 // NOLINTNEXTLINE(misc-no-recursion): a submenu is a menu one level down.
705 [[nodiscard]] static std::string shapeOf(std::vector<HeadMenuEntry> const& entries) {
706 std::string shape;
707 for (HeadMenuEntry const& entry : entries) {
708 if (entry.separator) {
709 shape += '|';
710 } else if (!entry.items.empty()) {
711 shape += '(' + shapeOf(entry.items) + ')';
712 } else {
713 shape += entry.id + ';';
714 }
715 }
716 return shape;
717 }

719 // The entry at @p path through @p entries; null past the end, which a binding reads only in the moment before
720 // the menu is drawn again for a new shape.
721 [[nodiscard]] static HeadMenuEntry const* entryAt(std::vector<HeadMenuEntry> const& entries,
722 std::vector<std::size_t> const& path) {
723 std::vector<HeadMenuEntry> const* level = &entries;
724 HeadMenuEntry const* entry = nullptr;
725 for (std::size_t const index : path) {
726 if (index >= level->size()) {
727 return nullptr;
728 }
729 entry = &level->at(index);
730 level = &entry->items;
731 }
732 return entry;
733 }

735 // The menu items of one level of the head menu: each bound to its entry by its place, which the shape keeps, and
736 // each chosen by its id, so the screen refuses an entry hidden since by name rather than running its neighbour.
737 // NOLINTNEXTLINE(misc-no-recursion): a submenu is a menu one level down.
738 [[nodiscard]] std::vector<MenuItem> headMenuItems(WorkspaceTabId tab, reactive::Signal<ShapedMenu> const& menu,
739 std::vector<HeadMenuEntry> const& level,
740 std::vector<std::size_t> const& outer) const {
741 std::vector<MenuItem> items;
742 for (std::size_t index = 0; index < level.size(); ++index) {
743 HeadMenuEntry const& entry = level.at(index);
744 MenuItem item;
745 if (entry.separator) {
746 item.separator = true;
747 items.push_back(std::move(item));
748 continue;
749 }
750 std::vector<std::size_t> path = outer;
751 path.push_back(index);
752 auto const read = [&menu, path](auto field, auto const& fallback) {
753 return [&menu, path, field, fallback] {
754 HeadMenuEntry const* const shown = entryAt(menu.get().entries, path);
755 return shown == nullptr ? fallback : std::invoke(field, *shown);
756 };
757 };
758 item.label = Prop<std::string>{read(&HeadMenuEntry::label, std::string{})};
759 item.description = Prop<std::string>{read(&HeadMenuEntry::description, std::string{})};
760 item.enabled = Prop<bool>{read(&HeadMenuEntry::enabled, false)};
761 item.checked = Prop<std::optional<bool>>{read(&HeadMenuEntry::checked, std::optional<bool>{})};
762 item.icon = entry.icon;
763 item.keys = entry.keys;
764 if (entry.items.empty()) {
765 item.onSelect = [this, tab, entryId = entry.id] {
766 ScreenHandle* const handle = _model->handleOf(tab);
767 if (ScreenHead* const head = handle == nullptr ? nullptr : handle->head()) {
768 head->choose(entryId);
769 }
770 };
771 } else {
772 item.items = headMenuItems(tab, menu, entry.items, path);
773 }
774 items.push_back(std::move(item));
775 }
776 return items;
777 }

803 // The model's mountPage: a page slot at the end of the bar, the screen mounted into it, a host screen from the
804 // registry and a document screen from the document host, then the head row above it. The page goes with a
805 // refused mount.
806 std::expected<std::unique_ptr<ScreenHandle>, MountError> mountPage(WorkspaceTabId tab, std::string_view screenId,
807 ParamMap const& params) {
808 auto page = std::make_unique<Page>();
809 page->screenId = std::string{screenId};
810 page->slot = _backend->createSlot(_bar.get());
811 page->slot->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
812 // A screen whose root keeps its content height scrolls inside the page instead of being cut at the window's
813 // edge; one whose root stretches fills the page and lays itself out.
814 page->slot->setScrollable(true);
815 page->slot->setVisible(false);
816 std::expected<std::unique_ptr<ScreenHandle>, MountError> handle;
817 WorkspaceScreen const facts = _screens ? _screens(screenId) : WorkspaceScreen{};
818 if (facts.host.empty()) {
819 handle = _host->mountInto(screenId, params, *page->slot);
820 } else {
821 HostScreenFactory const* const factory = _hosts == nullptr ? nullptr : _hosts->find(facts.host);
822 if (factory == nullptr) {
823 return std::unexpected(MountError{.screenId = std::string{screenId},
824 .reason = "no host screen `" + facts.host + "` is registered"});
825 }
826 handle = (*factory)(params, *page->slot);
827 }
828 if (!handle) {
829 return handle;
830 }
831 if (_headPlacement == HeadPlacement::InPage && (*handle)->head() != nullptr) {
832 // Mounted after the screen, since it reads the handle, and moved above it.
833 page->head.emplace(*_runtime, *_backend, headRow(tab), page->slot.get());
834 page->slot->moveChild(page->head->root(), 0);
835 }
836 _pages.insert_or_assign(tab.value, std::move(page));
837 _shownOrder.push_back(tab);
838 return handle;
839 }

998 // The head's lead, built but not yet wrapped into an immutable Node: the subtitle (muted) and the badges on one
999 // wrapping line, so they sit side by side while they fit and wrap under one another where they do not; the badges'
1000 // holder wraps too. It stretches from `kHeadLeadMinimum` units, so a trail beside it leaves it at least that much
1001 // before the trail wraps to a line of its own.
1002 //
1003 // It is built once per place and shown only while the head's placement is that place. The `Below` copy stretches
1004 // from its minimum. The `Inline` copy stretches with none, so it fills what the title's line gives it and its
1005 // subtitle wraps there, while the frame's slot, still content-sized, holds the spare width. The `End` copy is
1006 // content-sized, in the trail.

1008 std::string const suffix = idSuffix(place);
1009 auto const subtitle = [this, tab] {
1010 ScreenHead const* const head = headOf(tab);
1011 return head == nullptr ? std::string{} : head->subtitle().get();
1012 };

1014 ScreenHead const* const head = headOf(tab);
1015 return head == nullptr ? std::vector<HeadBadge>{} : head->status().get();
1016 };
1017 Row row{.children = {}, .gap = 1, .common = {}, .tone = std::nullopt};
1018 row.wrap = true;
1019 row.align = Align::Center;
1020 row.common.testId = "head_lead" + suffix;
1021 if (place == LeadPlacement::Below) {
1022 row.common.layout.width = Sizing::stretch();
1023 row.common.layout.minWidth = kHeadLeadMinimum;
1024 } else if (place == LeadPlacement::Inline) {
1025 row.common.layout.width = Sizing::stretch();
1026 }
1027 row.common.visible =
1028 Prop<bool>{[this, tab, place] { return leadHasContent(tab) && leadPlacementOf(tab) == place; }};
1029 // The runs when the screen gives them, else its plain subtitle as one run, so the text is bound to one thing.
1030 auto const spans = [this, tab, subtitle] {
1031 std::vector<TextSpan> runs = subtitleSpansOf(headOf(tab));

1033 std::string plain = subtitle();
1034 if (!plain.empty()) {
1035 runs.push_back(TextSpan{.text = std::move(plain)});
1036 }
1037 }
1038 return runs;
1039 };
1040 Text text{.text = Prop<std::string>{subtitle}, .role = TextRole::Muted, .field = {}, .common = {}};
1041 text.spans = Prop<std::vector<TextSpan>>{spans};
1042 text.common.testId = "head_subtitle" + suffix;
1043 text.common.visible = Prop<bool>{[spans] { return !spans().empty(); }};
1044 row.children.push_back(ui::text(std::move(text)));
1045 row.children.push_back(ui::forEach<PlacedBadge>(

1049 placed.push_back(PlacedBadge{.index = placed.size(), .badge = badge});

1052 },

1056 .text = Prop<std::string>{[&placed] { return placed.get().badge.text; }},
1057 .icon = placed.peek().badge.icon,
1058 .common = {}};
1059 badge.common.testId = "head_status" + suffix + "_" + std::to_string(placed.peek().index);
1060 return ui::badge(std::move(badge));
1061 },
1062 Axis::Horizontal, 1, {}, true));
1063 return row;
1064 }

1066 // The head's trail: the tools, the actions as buttons and the head's own menu behind one icon button, in one row.
1067 // It is a row of actions: its parts are the theme's action gap apart, and it wraps, so a part with nothing in it
1068 // takes none. Hidden while the head has none of them.
1069 [[nodiscard]] ActionBar buildHeadTrail(WorkspaceTabId tab) const {
1070 auto const actions = [this, tab] {
1071 ScreenHead const* const head = headOf(tab);
1072 return head == nullptr ? std::vector<HeadAction>{} : head->actions().get();
1073 };
1074 ActionBar row;
1075 // Wrapping, so a part with nothing in it (no tools, no menu) takes no gap: a non-wrapping row would leave one
1076 // after the last action, short of the edge by a step.
1077 row.wrap = true;
1078 row.common.testId = "head_trail";
1079 row.common.visible = Prop<bool>{[this, tab] { return trailHasContent(tab); }};
1080 row.align = Align::Center;
1081 row.children.push_back(ui::forEach<HeadTool>(
1082 [this, tab] { return headToolsOf(tab); }, [](HeadTool const& tool) { return Key{tool.id}; },
1083 [this, tab](reactive::Signal<HeadTool> const& tool) { return headTool(tab, tool); }, Axis::Horizontal, 1));
1084 row.children.push_back(ui::row(buildHeadLead(tab, LeadPlacement::End)));
