54/// @brief Tabs over screens: one screen instance per tab, mounted into the tab's page through
55/// `DocumentScreenHost::mountInto`.
56///
57/// Tabs keep the order they were opened in. Opening a screen whose identity an open tab already has focuses that
58/// tab; a singleton screen has one tab. Closing runs each screen's close protocol (spec 6 §3), one tab at a time,
59/// and a walk over several tabs stops at the first veto. The shell takes Ctrl+Tab, Ctrl+Shift+Tab and Ctrl+W, which
60/// documents may not declare. Neither copyable nor movable; owned by the owner thread of its runtime.
61class Workspace {
62public:
63 /// @brief The shell, its tab bar appended to @p parent.
64 /// @param runtime The runtime the screens' reactive nodes belong to. Borrowed.
65 /// @param backend The backend the bar and the pages are made with. Borrowed.
66 /// @param host What mounts the screens. Borrowed; it outlives the shell.
67 /// @param options The screens' catalog facts.
68 /// @param parent The container the shell is appended to; null makes it a backend root.
69 Workspace(reactive::Runtime& runtime MORPH_LIFETIMEBOUND, IViewBackend& backend MORPH_LIFETIMEBOUND,
70 DocumentScreenHost& host MORPH_LIFETIMEBOUND, WorkspaceOptions options = {},
71 ContainerWidget* parent = nullptr)
72 : _runtime{&runtime},
73 _backend{&backend},
74 _host{&host},
75 _options{std::move(options)},
76 _bar{backend.createTabs(parent)} {
77 _bar->setTabs({});
78 _bar->setSelected(kNone);
79 _bar->setOnSelect([this](std::size_t index) {
80 if (index < _tabs.size()) {
81 activateTab(_tabs.at(index)->id, false);
82 }
83 });
84 _bar->setKeys({"Ctrl+Tab", "Ctrl+Shift+Tab", "Ctrl+W"}, [this](std::string const& chord) {
85 if (chord == "Ctrl+Tab") {
86 next();
87 } else if (chord == "Ctrl+Shift+Tab") {
88 previous();
89 } else if (std::optional<std::size_t> const shown = active()) {
90 close(*shown, {});
91 }
92 });
93 }
94
95 /// @brief Unmounts every screen without asking, newest first.
96 ~Workspace() {
97 while (!_tabs.empty()) {
98 _tabs.pop_back();
99 }
100 }

111 /// @brief Opens a screen, or focuses the tab that already shows it.
112 ///
113 /// The tab whose screen and identity match is focused; a singleton screen's tab is focused; a `reuseEmpty`
114 /// screen opened with an identity replaces its tab that has no identity and no unsaved changes; otherwise a new
115 /// tab is opened. The tab opened or focused is the active one.
116 /// @param screenId The screen id.
117 /// @param params Its params.
118 /// @return The tab's index, or why the screen could not be mounted; a refused mount opens nothing.
119 std::expected<std::size_t, MountError> open(std::string_view screenId, ParamMap const& params) {
120 WorkspaceScreen const facts = _options.screens ? _options.screens(screenId) : WorkspaceScreen{};
121 std::optional<std::string> const identity = _host->identityOf(screenId, params);
122 for (std::size_t index = 0; index < _tabs.size(); ++index) {
123 Tab const& tab = *_tabs.at(index);
124 if (tab.screenId != screenId) {
125 continue;
126 }
127 if (facts.singleton || (identity && identityText(tab) == *identity)) {
128 activateTab(tab.id, false);
129 return index;
130 }
131 }
132 std::optional<std::uint64_t> replaced;
133 if (facts.reuseEmpty && identity) {
134 for (auto const& tab : _tabs) {
135 if (tab->screenId == screenId && tab->handle->identity().peek().isNull() &&
136 !tab->handle->dirty().peek()) {
137 replaced = tab->id;
138 break;
139 }
140 }
141 }
142 auto opened = mountTab(screenId, params);
143 if (!opened) {
144 return std::unexpected(std::move(opened.error()));
145 }
146 std::uint64_t const tabId = *opened;
147 if (replaced) {
148 std::size_t const where = indexOf(*replaced);
149 moveTab(tabId, where);
150 removeTab(*replaced, true);
151 }
152 activateTab(tabId, false);
153 return indexOf(tabId);
154 }

156 /// @brief Shows a tab and makes it the most recently used.
157 /// @param index The tab; past the last one does nothing.
158 void activate(std::size_t index) {

164 /// @brief Shows the next tab in most-recently-used order (Ctrl+Tab). Repeated calls walk the order as it was at
165 /// the first of them; any other activation starts a new walk.
166 void next() { walk(1); }

171 /// @brief Closes a tab through its screen's close protocol.
172 /// @param index The tab.
173 /// @param done Called once with whether it closed; may be empty.
174 void close(std::size_t index, std::function<void(bool)> done) {
175 if (index >= _tabs.size()) {
176 if (done) {
177 done(false);
178 }
179 return;
180 }
181 closeEach({_tabs.at(index)->id}, std::move(done));
182 }

184 /// @brief Closes every tab but one, one at a time, stopping at the first veto.
185 /// @param index The tab kept.
186 /// @param done Called once with whether every other tab closed; may be empty.
187 void closeOthers(std::size_t index, std::function<void(bool)> done) {
188 std::vector<std::uint64_t> ids;
189 ids.reserve(_tabs.size());
190 for (std::size_t where = 0; where < _tabs.size(); ++where) {
191 if (where != index) {
192 ids.push_back(_tabs.at(where)->id);
193 }
194 }
195 closeEach(std::move(ids), std::move(done));
196 }

198 /// @brief Closes every tab, one at a time, stopping at the first veto.
199 /// @param done Called once with whether every tab closed; may be empty.
200 void closeAll(std::function<void(bool)> done) {
201 std::vector<std::uint64_t> ids;
202 ids.reserve(_tabs.size());

212 std::vector<std::uint64_t> ids;

220 }

284 // Mounts a screen into a new page at the end; the page goes with a refused mount.
285 std::expected<std::uint64_t, MountError> mountTab(std::string_view screenId, ParamMap const& params) {
286 auto tab = std::make_unique<Tab>();
287 tab->id = ++_lastId;
288 tab->screenId = std::string{screenId};
289 tab->page = _backend->createSlot(_bar.get());
290 tab->page->setVisible(false);
291 auto handle = _host->mountInto(screenId, params, *tab->page);
292 if (!handle) {
293 return std::unexpected(std::move(handle.error()));
294 }
295 tab->handle = std::move(*handle);
296 tab->scope = std::make_unique<reactive::Scope>(*_runtime);
297 ScreenHandle const* const screen = tab->handle.get();
298 tab->scope->effect([this, screen] {
299 static_cast<void>(screen->title().get());
300 static_cast<void>(screen->dirty().get());
301 _runtime->untracked([this] { showCaptions(); });
302 });
303 std::uint64_t const tabId = tab->id;
304 _tabs.push_back(std::move(tab));
305 showCaptions();
306 return tabId;
307 }

309 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): a tab id, then the index it moves to.
310 void moveTab(std::uint64_t tabId, std::size_t target) {
311 std::size_t const from = indexOf(tabId);
312 if (from == kNone || from == target) {
313 return;
314 }
315 std::unique_ptr<Tab> tab = std::move(_tabs.at(from));
316 _tabs.erase(_tabs.begin() + static_cast<std::ptrdiff_t>(from));
317 _bar->moveChild(*tab->page, target);
318 _tabs.insert(_tabs.begin() + static_cast<std::ptrdiff_t>(target), std::move(tab));
319 showCaptions();
320 }
321
322 void removeTab(std::uint64_t tabId, bool force) {
323 std::size_t const where = indexOf(tabId);
324 if (where == kNone) {
325 return;
326 }
327 if (force) {
328 _tabs.at(where)->handle->forceClose();
329 }
330 _tabs.erase(_tabs.begin() + static_cast<std::ptrdiff_t>(where));
331 std::erase(_recent, tabId);
332 _walk.reset();
333 if (_active == tabId) {
334 _active.reset();
335 if (!_recent.empty()) {
336 activateTab(_recent.front(), false);
337 return;
338 }
339 }
340 showCaptions();
341 }
342
343 void activateTab(std::uint64_t tabId, bool walking) {
344 if (!walking) {
345 _walk.reset();
346 }
347 _active = tabId;
348 std::erase(_recent, tabId);
349 _recent.insert(_recent.begin(), tabId);
350 for (auto const& tab : _tabs) {
351 tab->page->setVisible(tab->id == tabId);
352 }
353 showCaptions();
354 }
355
356 void walk(int step) {
357 if (_tabs.size() < 2 || !_active) {
358 return;
359 }
360 if (!_walk) {
361 _walk = Walk{.order = _recent, .position = 0};
362 }
363 auto const count = static_cast<std::ptrdiff_t>(_walk->order.size());
364 std::ptrdiff_t const next = (((static_cast<std::ptrdiff_t>(_walk->position) + step) % count) + count) % count;
365 _walk->position = static_cast<std::size_t>(next);
366 std::uint64_t const tabId = _walk->order.at(_walk->position);
367 // A walk keeps its own order; activating moves the tab to the front of the recent list only when it ends.
368 _active = tabId;
369 for (auto const& tab : _tabs) {
370 tab->page->setVisible(tab->id == tabId);
371 }
372 std::erase(_recent, tabId);
373 _recent.insert(_recent.begin(), tabId);
374 showCaptions();
375 }
376
377 // Closes the tabs one at a time, stopping at the first that stays open.
378 void closeEach(std::vector<std::uint64_t> ids, std::function<void(bool)> done) {
379 std::erase_if(ids, [this](std::uint64_t tabId) { return indexOf(tabId) == kNone; });
380 if (ids.empty()) {
381 if (done) {
382 done(true);
383 }
384 return;
385 }
386 std::uint64_t const tabId = ids.front();
387 ids.erase(ids.begin());
388 std::weak_ptr<bool> const alive = _alive;
389 _tabs.at(indexOf(tabId))
390 ->handle->requestClose(
391 [this, alive, tabId, rest = std::move(ids), done = std::move(done)](bool closed) mutable {
392 if (alive.expired()) {
393 return;
394 }
395 if (!closed) {
396 if (done) {
397 done(false);
398 }
399 return;
400 }
401 removeTab(tabId, false);
402 closeEach(std::move(rest), std::move(done));
403 });
404 }

406 void showCaptions() {
407 std::vector<std::string> captions;
408 captions.reserve(_tabs.size());
409 for (auto const& tab : _tabs) {
410 captions.push_back(captionOf(*tab));
411 }
412 _bar->setTabs(captions);
413 std::optional<std::size_t> const shown = active();
414 _bar->setSelected(shown.value_or(kNone));
415 }
