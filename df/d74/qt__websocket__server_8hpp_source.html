20/// @brief Per-connection resource limits and bind/exposure policy enforced by
21/// `QtWebSocketServer`.
22///
23/// Declared outside `QtWebSocketServer` so its default member initialisers are
24/// fully parsed before any constructor default argument that names `Config{}`
25/// is evaluated (same rationale as `morph::qt::QtWebSocketBackendConfig` and
26/// `morph::offline::NetworkMonitorConfig`).
27struct QtWebSocketServerConfig {
28 /// @brief Max simultaneous live client connections. `0` = unbounded (today's behavior).
29 ///
30 /// A new connection beyond this count is closed immediately in `onNewConnection`,
31 /// before any message exchange and before it is tracked internally.
32 std::size_t maxConnections = 0;
33
34 /// @brief Per-frame size cap enforced before a message reaches `RemoteServer::handle()`.
35 ///
36 /// Defaults to `morph::wire::kMaxEnvelopeBytes` (the wire layer's own bound), so
37 /// an unconfigured server behaves exactly as today: the wire-layer cap is the
38 /// only one in effect. Set lower to reject oversized frames earlier, before the
39 /// cost of a pool round-trip and JSON decode.
40 std::size_t maxMessageBytes = ::morph::wire::kMaxEnvelopeBytes;
41
42 /// @brief Per-connection token-bucket rate limit, in messages per second. `0` = unbounded.
43 ///
44 /// The bucket capacity equals `messagesPerSecond` (an immediate one-second
45 /// burst is allowed right after connecting), refilling continuously at that
46 /// rate. A frame that arrives with an empty bucket is dropped — not queued,
47 /// not replied to. See `docs/spec/core/backend.md`.
48 std::size_t messagesPerSecond = 0;
49
50 /// @brief Time allowed for a newly-accepted connection to send its first text
51 /// frame before it is closed. `0` = disabled (today's behavior).
52 ///
53 /// `QWebSocketServer::newConnection()` fires only after the WebSocket (and, in
54 /// `SecureMode`, TLS) opening handshake has already completed, so in practice
55 /// this bounds time-to-first-frame after that point, not the handshake itself
56 /// — Qt exposes no earlier hook at this layer.
57 std::chrono::milliseconds handshakeTimeout{0};
58
59 /// @brief Time a connection may go without sending any frame before it is
60 /// closed. `0` = disabled (today's behavior).
61 ///
62 /// Checked by a periodic housekeeping sweep (roughly once per second), so the
63 /// actual close can lag the configured value by up to that sweep interval.
64 std::chrono::milliseconds idleTimeout{0};
65
66 /// @brief Address `listen()` binds to. Default `QHostAddress::LocalHost`
67 /// (today's behavior, unchanged).
68 QHostAddress bindAddress = QHostAddress::LocalHost;
69
70 /// @brief Deliberate opt-out of the exposure guard: set `true` only to
71 /// knowingly serve plaintext (no TLS) on a non-loopback `bindAddress`.
72 bool allowPlaintextExposure = false;
73};

75/// @brief Qt WebSocket server that bridges incoming connections to a `RemoteServer`.
76///
77/// Listens for WebSocket clients, receives their text messages, and forwards each
78/// one to `RemoteServer::handle()`. Replies are sent back to the originating client.
79///
80/// @par TLS
81/// Pass a `QSslConfiguration` to enable `wss://`. The configuration should be built
82/// from a certificate file, a Qt resource, or a byte array before being passed in.
83///
84/// @par SSL-less Qt builds (`QT_NO_SSL`, including the standard Qt-for-WebAssembly build)
85/// The constructor's `tls` parameter does not exist at all when Qt itself was
86/// configured without SSL — `QSslConfiguration` isn't a type Qt provides in that
87/// configuration. Such a server always constructs in `NonSecureMode`; see
88/// `QtWebSocketBackend`'s equivalent doc comment for the client-side rationale.
89///
90/// @par Resource limits
91/// Pass a `QtWebSocketServerConfig` to bound connection count, per-frame size,
92/// per-connection message rate, and handshake/idle time. All fields default to
93/// unbounded (except `maxMessageBytes`, which defaults to the wire-layer cap),
94/// reproducing today's behavior when omitted.
95///
96/// @par Bind address & plaintext-exposure guard
97/// `listen()` refuses — returns `false` and logs at `morph::log::LogLevel::error`
98/// — when `cfg.bindAddress` is not loopback, no TLS configuration was passed to
99/// the constructor, and `cfg.allowPlaintextExposure` is `false`. See
100/// `QtWebSocketServerConfig` and security.md's "Transport security" section.
101///
102/// @par Usage
103/// Call `listen()` to start accepting connections, and `port()` to discover the
104/// bound port (useful when @p port was 0 — let the OS assign a free port).
105/// Call `close()` or destroy the object to stop the server.
106class QtWebSocketServer : public QObject {
107 Q_OBJECT
108public:
109 /// @brief Alias for the per-connection limits configuration struct.
110 using Config = QtWebSocketServerConfig;
111
112 /// @brief Constructs the server and prepares it to listen on @p port.
113 ///
114 /// The server does not start accepting connections until `listen()` is called.
115 ///
116 /// @param server `RemoteServer` instance that processes incoming messages.
117 /// @param port TCP port to listen on. Pass 0 to let the OS pick a free port.
118 /// @param tls If non-null, enables TLS (`wss://`) with this configuration. Not
119 /// declared at all on an SSL-less Qt build (`QT_NO_SSL`) — see the
120 /// class doc comment's "SSL-less Qt builds" section.
121 /// @param cfg Per-connection resource limits. Default: everything unbounded
122 /// (today's behavior) except `maxMessageBytes`, which defaults to
123 /// the wire-layer cap.
124 /// @param parent Optional Qt parent object.
125 explicit QtWebSocketServer(::morph::backend::RemoteServer& server, quint16 port = 0,
126#ifndef QT_NO_SSL
127 std::optional<QSslConfiguration> tls = std::nullopt,
128#endif
129 QtWebSocketServerConfig cfg = QtWebSocketServerConfig{}, QObject* parent = nullptr);
130
131 /// @brief Closes the server and disconnects all clients.
132 ~QtWebSocketServer() override;
133
134 /// @brief Starts listening for incoming WebSocket connections.
135 ///
136 /// Refuses — returns `false` without binding, and logs at
137 /// `morph::log::LogLevel::error` — when the configured `bindAddress` is not
138 /// loopback, no TLS configuration was passed to the constructor, and
139 /// `allowPlaintextExposure` is `false`. See `QtWebSocketServerConfig`.
140 ///
141 /// @return `true` if the server successfully bound to the requested port.
142 bool listen();
143
144 /// @brief Returns the port the server is currently bound to.
145 ///
146 /// When constructed with port 0, this returns the OS-assigned port after
147 /// a successful `listen()` call.
148 /// @return Bound TCP port number.
149 [[nodiscard]] quint16 port() const;
150
151 /// @brief Stops accepting new connections and closes the server socket.
152 void close();
153
154 /// @brief Gracefully stops the server: refuse new work, let in-flight
155 /// replies deliver, close each client with a proper WebSocket
156 /// close frame, then hard-stop anything still connected.
157 ///
158 /// Sequence: `QWebSocketServer::pauseAccepting()` (no new connections),
159 /// `RemoteServer::beginShutdown()` (new `register`/`execute` now fail
160 /// fast with `"server shutting down"`), waiting up to @p deadline for
161 /// `RemoteServer::drainedWithin()` to report every in-flight execute has
162 /// replied, sending each still-connected client a real close frame
163 /// (`CloseCodeGoingAway`, reason `"server shutting down"`) instead of an
164 /// abort, and finally running the existing `close()` hard stop for
165 /// whatever @p deadline did not leave time to finish gracefully.
166 ///
167 /// Pumps the Qt event loop internally while it waits, so it is safe to
168 /// call from the Qt thread — which is also the thread that must run
169 /// `sendTextMessage` for replies and close frames to actually reach
170 /// clients. Purely additive and opt-in: a server that never calls this
171 /// behaves exactly as it does today, and `close()` itself is unchanged.
172 ///
173 /// @param deadline Total time budget for the whole sequence, measured
174 /// from the moment this call starts. Whatever the drain
175 /// and graceful-close steps have not finished within it
176 /// is hard-stopped exactly as `close()` does.
177 /// @return `true` if every in-flight execute drained before @p deadline
178 /// elapsed; `false` if the hard stop had to reclaim stragglers.
179 bool closeGracefully(std::chrono::milliseconds deadline);
180
181 /// @brief Qt slot called when a new client connects.
182 Q_SLOT void onNewConnection();
183
184 /// @brief Qt slot called when a connected client sends a text message.
185 /// @param message Raw text frame received from the client.
186 Q_SLOT void onTextMessage(const QString& message);
187
188 /// @brief Qt slot called when a client disconnects.
189 Q_SLOT void onDisconnected();
190
191private:
192 /// @brief Per-connection state tracked between accept and disconnect.
193 struct ClientState {
194 /// @brief The connection this state belongs to (for symmetry with the map key; unused for lookup).
195 QWebSocket* socket = nullptr;
196
197 /// @brief Current token-bucket balance for `messagesPerSecond`.
198 double tokens = 0.0;
199
200 /// @brief Last time `tokens` was refilled (used to compute elapsed time on the next frame).
201 std::chrono::steady_clock::time_point lastRefill{};
202
203 /// @brief Last time any frame was received on this connection (drives `idleTimeout`).
204 std::chrono::steady_clock::time_point lastActivity{};
205
206 /// @brief One-shot timer enforcing `handshakeTimeout`; `nullptr` once cancelled by the
207 /// first frame, or if `handshakeTimeout == 0`.
208 QTimer* handshakeTimer = nullptr;
209
210 /// @brief Connection scope `RemoteServer::openConnection()` assigned this
211 /// client at accept time, so its disconnect can reclaim every model
212 /// it registered (see `RemoteServer::closeConnection`).
213 ::morph::backend::ConnectionId cid = 0;
214 };
215
216 /// @brief Refills @p state's token bucket for elapsed time, then consumes one
217 /// token if available.
218 /// @param state Per-connection state to update.
219 /// @return `true` if a token was available (the frame is admitted), `false` if
220 /// the bucket was empty (the frame must be dropped).
221 bool consumeToken(ClientState& state);
222
223 /// @brief Qt slot: periodic sweep that closes any connection idle past `idleTimeout`.
224 Q_SLOT void onHousekeepingTick();
225
226 ::morph::backend::RemoteServer& _server;
227 quint16 _requestedPort;
228 QtWebSocketServerConfig _cfg;
229 QWebSocketServer _wsServer;
230 std::unordered_map<QWebSocket*, ClientState> _clients;
231 QTimer _housekeepingTimer;
232};
