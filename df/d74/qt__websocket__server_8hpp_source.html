14/// @brief Qt WebSocket server that bridges incoming connections to a `RemoteServer`.
15///
16/// Listens for WebSocket clients, receives their text messages, and forwards each
17/// one to `RemoteServer::handle()`. Replies are sent back to the originating client.
18///
19/// @par TLS
20/// Pass a `QSslConfiguration` to enable `wss://`. The configuration should be built
21/// from a certificate file, a Qt resource, or a byte array before being passed in.
22///
23/// @par Usage
24/// Call `listen()` to start accepting connections, and `port()` to discover the
25/// bound port (useful when @p port was 0 — let the OS assign a free port).
26/// Call `close()` or destroy the object to stop the server.
27class QtWebSocketServer : public QObject {
28 Q_OBJECT
29public:
30 /// @brief Constructs the server and prepares it to listen on @p port.
31 ///
32 /// The server does not start accepting connections until `listen()` is called.
33 ///
34 /// @param server `RemoteServer` instance that processes incoming messages.
35 /// @param port TCP port to listen on. Pass 0 to let the OS pick a free port.
36 /// @param tls If non-null, enables TLS (`wss://`) with this configuration.
37 /// @param parent Optional Qt parent object.
38 explicit QtWebSocketServer(::morph::backend::RemoteServer& server, quint16 port = 0,
39 std::optional<QSslConfiguration> tls = std::nullopt, QObject* parent = nullptr);
40
41 /// @brief Closes the server and disconnects all clients.
42 ~QtWebSocketServer() override;
43
44 /// @brief Starts listening for incoming WebSocket connections.
45 ///
46 /// @return `true` if the server successfully bound to the requested port.
47 bool listen();
48
49 /// @brief Returns the port the server is currently bound to.
50 ///
51 /// When constructed with port 0, this returns the OS-assigned port after
52 /// a successful `listen()` call.
53 /// @return Bound TCP port number.
54 [[nodiscard]] quint16 port() const;
55
56 /// @brief Stops accepting new connections and closes the server socket.
57 void close();
58
59 /// @brief Qt slot called when a new client connects.
60 Q_SLOT void onNewConnection();
61
62 /// @brief Qt slot called when a connected client sends a text message.
63 /// @param message Raw text frame received from the client.
64 Q_SLOT void onTextMessage(const QString& message);
65
66 /// @brief Qt slot called when a client disconnects.
67 Q_SLOT void onDisconnected();
68
69private:
70 ::morph::backend::RemoteServer& _server;
71 quint16 _requestedPort;
72 QWebSocketServer _wsServer;
73 std::vector<QWebSocket*> _clients;
74};
