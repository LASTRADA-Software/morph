16/// @brief How a banner draws its action.
17enum class BannerActionStyle : std::uint8_t {
18 Ghost, ///< An unframed button at the trailing edge, its label in the accent.
19 Button, ///< A framed, normal button at the trailing edge.
20 Link, ///< The action's text inside the message, underlined in the accent, wrapping with the sentence.
21};
22
23/// @brief Every style by its document name, `ghost` first.
24inline constexpr std::array<std::pair<std::string_view, BannerActionStyle>, 3> kBannerActionStyles = {{
25 {"ghost", BannerActionStyle::Ghost},
26 {"button", BannerActionStyle::Button},
27 {"link", BannerActionStyle::Link},
28}};
29
30/// @brief A style by its document name.
31/// @param name The name, such as `link`.
32/// @return The style; `nullopt` for a name that is none.
33[[nodiscard]] constexpr std::optional<BannerActionStyle> bannerActionStyleNamed(std::string_view name) noexcept {
34 for (auto const& [known, style] : kBannerActionStyles) {
35 if (known == name) {
36 return style;
37 }
38 }
39 return std::nullopt;
40}
41
42/// @brief A style's document name.
43/// @param style The style.
44/// @return Its name, such as `link`.
45[[nodiscard]] constexpr std::string_view bannerActionStyleName(BannerActionStyle style) noexcept {
46 for (auto const& [name, known] : kBannerActionStyles) {
47 if (known == style) {
48 return name;
49 }
50 }
51 return "ghost";
52}
