25/// @brief Replacements for the parts of a wizard view a host draws from its own UI kit. Each one left empty keeps
26/// the built-in part.
27struct WizardChrome {
28 /// @brief Replaces the title and the step indicator. Called once with the session, whose `title`, `current`,
29 /// `stepCount` and `stepTitle` it reads.
30 std::function<ui::Node(WizardSession&)> header;
31 /// @brief Replaces Back and Next. Called once with the session, whose `canBack`, `canNext`, `onLastStep`, `back`
32 /// and `next` are the same gates and moves the built-in buttons use.
33 std::function<ui::Node(WizardSession&)> nav;
34};
35
36/// @brief How `wizardView` draws a wizard and labels its chrome.
37struct WizardViewOptions {
38 /// @brief Every step's form view options; its overrides reach every step's fields.
39 FormViewOptions form{};
40 /// @brief The host's replacements for the view's own parts.
41 WizardChrome chrome{};
42 /// @brief The Back button's caption.
43 std::string backLabel = "Back";
44 /// @brief The Next button's caption.
45 std::string nextLabel = "Next";
46 /// @brief The note shown on the last step, which has no Next.
47 std::string lastStepText = "Last step — fill it in to finish";
48};

52/// @brief The built-in header: the title with the step's position (`Flow (1 / 2)`) and a step indicator (done,
53/// current,
54/// pending).
55/// @param session The session.
56/// @return The header.
57[[nodiscard]] inline ui::Node wizardHeader(WizardSession const& session) {
58 WizardSession const* const wizard = &session;
59 ui::Text title{.text =
60 [wizard] {
61 return wizard->title() + " (" + std::to_string(wizard->current() + 1) + " / " +
62 std::to_string(wizard->stepCount()) + ")";
63 },
64 .role = ui::TextRole::Heading};
65 title.common.testId = "wizard_title";
66 ui::Steps steps{
67 .items =
68 [wizard] {
69 std::vector<ui::Step> items;
70 std::size_t const here = wizard->current();
71 for (std::size_t i = 0; i < wizard->stepCount(); ++i) {
72 ui::StepState state = ui::StepState::Pending;
73 if (i == here) {
74 state = ui::StepState::Active;
75 } else if (wizard->stepDone(i)) {
76 state = ui::StepState::Done;
77 }
78 items.push_back(ui::Step{.label = wizard->stepTitle(i), .state = state});
79 }
80 return items;
81 },
82 .current = [wizard] { return std::optional<std::size_t>{wizard->current()}; },
83 };
84 steps.common.testId = "wizard_steps";
85 return ui::column(ui::Column{.children = {ui::text(std::move(title)), ui::steps(std::move(steps))}, .gap = 4});
86}

88/// @brief The built-in navigation: Back, Next and the last step's note.
89/// @param session The session.
90/// @param options The view options.
91/// @return The navigation row.
92[[nodiscard]] inline ui::Node wizardNav(WizardSession& session, WizardViewOptions const& options) {
93 WizardSession* const wizard = &session;
94 ui::Button back{.label = options.backLabel, .onClick = [wizard] { wizard->back(); }};
95 back.common.enabled = [wizard] { return wizard->canBack(); };
96 back.common.testId = "wizardBack";
97 ui::Button next{.label = options.nextLabel, .onClick = [wizard] { wizard->next(); }};
98 next.common.enabled = [wizard] { return wizard->canNext(); };
99 next.common.testId = "wizardNext";
100 ui::Text last{.text = options.lastStepText, .role = ui::TextRole::Muted};
101 last.common.visible = [wizard] { return wizard->onLastStep(); };
102 last.common.testId = "wizard_last";
103 return ui::row(ui::Row{
104 .children = {ui::button(std::move(back)), ui::button(std::move(next)), ui::text(std::move(last))}, .gap = 4});
105}

109/// @brief Renders @p session: the header (title with the step's position, a step indicator), the current step's
110/// `formView` (one per step, mounted while it is current; the values live in the steps' sessions, so leaving
111/// a step loses nothing), and Back and Next, enabled by `canBack` and `canNext`.
112///
113/// Test ids: `wizard_title`, `wizard_steps`, `wizard_step_<index>` (the current step's form), `wizardBack`,
114/// `wizardNext` and `wizard_last`; the step's fields have their form ids.
115///
116/// The nodes bind to the session by pointer: mount the result after the session and destroy the mount before it.
117/// @param session The session.
118/// @param options Every step's form options, the chrome and the labels.
119/// @return The view.
120[[nodiscard]] inline ui::Node wizardView(WizardSession& session, WizardViewOptions const& options = {}) {
121 WizardSession const* const wizard = &session;
122 std::vector<ui::SwitchCase> cases;
123 for (std::size_t i = 0; i < session.stepCount(); ++i) {
124 ui::Column step{.children = {formView(session.step(i), options.form)}};
125 step.common.testId = "wizard_step_" + std::to_string(i);
126 cases.push_back(
127 ui::SwitchCase{.key = ui::Key{static_cast<std::int64_t>(i)}, .node = ui::column(std::move(step))});
128 }
129 ui::Node body = ui::switchOf(ui::Switch{
130 .selector = [wizard] { return ui::Key{static_cast<std::int64_t>(wizard->current())}; },
131 .cases = std::move(cases),
132 });
133 return ui::column(ui::Column{
134 .children = {options.chrome.header ? options.chrome.header(session) : detail::engine::wizardHeader(session),
135 std::move(body),
136 options.chrome.nav ? options.chrome.nav(session) : detail::engine::wizardNav(session, options)},
137 .gap = 4});
138}
