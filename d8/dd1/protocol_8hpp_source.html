44/// @brief One thing a person said, and where they were when they said it.
45///
46/// Everything here but `text` is a hint about what the person means. It grants nothing: every action it leads to is
47/// checked under the session of the envelope that carried it.
48struct TurnRequest {
49 /// @brief The open screen's id.
50 std::string screenId;
51 /// @brief The screen's identity value, canonically encoded; empty for a screen with no record.
52 std::string identity;
53 /// @brief The screen's params as JSON text.
54 std::string paramsJson;
55 /// @brief The aliases the screen holds.
56 std::vector<ModelRef> models;
57 /// @brief What the person said.
58 std::string text;
59};

77/// @brief How far a turn has got.
78enum class TurnStatus : std::uint8_t {
79 Running, ///< Still working.
80 Done, ///< Finished.
81 Failed, ///< Stopped by a fault or a limit; `TurnState::error` says which.
82 Cancelled, ///< Stopped by the person, or by its connection closing.
83};
84
85/// @brief A turn's state as the server holds it, which outranks any event.
86struct TurnState {
87 /// @brief The turn's id; `0` when the principal has no such turn.
88 std::uint64_t turnId = 0;
89 /// @brief How far it has got.
90 TurnStatus status = TurnStatus::Running;
91 /// @brief The reply so far.
92 std::string text;
93 /// @brief Why it failed; empty otherwise.
94 std::string error;
95 /// @brief The tool calls it has made, in order.
96 std::vector<ToolStep> steps;
97 /// @brief The `TurnEvent::seq` this state covers, so a client does not apply an event it already includes.
98 std::uint64_t seq = 0;
99};

107/// @brief A hint that a turn moved. Events can be lost, so `TurnState` is the record.
108struct TurnEvent {
109 /// @brief The turn's id.
110 std::uint64_t turnId = 0;
111 /// @brief Increases by one per event of a turn.
112 std::uint64_t seq = 0;
113 /// @brief `text`, `tool`, `done` or `error`.
114 std::string kind;
115 /// @brief The text piece, the tool's name, or the reason.
116 std::string text;
117};
