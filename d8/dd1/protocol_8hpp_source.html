54/// @brief One thing a person said, and where they were when they said it.
55///
56/// Everything here but `text` is a hint about what the person means. It grants nothing: every action it leads to is
57/// checked under the session of the envelope that carried it.
58struct TurnRequest {
59 /// @brief The open screen's id.
60 std::string screenId;
61 /// @brief The screen's title as the person sees it, which is how they know the record; empty for none.
62 std::string title;
63 /// @brief The screen's identity value, canonically encoded; empty for a screen with no record.
64 std::string identity;
65 /// @brief The screen's params as JSON text.
66 std::string paramsJson;
67 /// @brief The aliases the screen holds.
68 std::vector<ModelRef> models;
69 /// @brief What the person said.
70 std::string text;
71 /// @brief The document of the draft the person has open on this screen, as JSON text; absent with none. It is the
72 /// screen's document as the assistant should read it, so a conversation of edits builds on the last
73 /// accepted draft. The server refuses a turn whose draft is over `ChatLimits::maxDraftBytes` or is not a
74 /// JSON object, and treats the text as data: it grants nothing.
75 std::optional<std::string> draft;
76};

106/// @brief How far a turn has got.
107enum class TurnStatus : std::uint8_t {
108 Running, ///< Still working.
109 Done, ///< Finished.
110 Failed, ///< Stopped by a fault or a limit; `TurnState::error` says which.
111 Cancelled, ///< Stopped by the person, or by its connection closing.

122/// @brief A turn's state as the server holds it, which outranks any event.
123struct TurnState {
124 /// @brief The turn's id; `0` when the principal has no such turn.
125 std::uint64_t turnId = 0;
126 /// @brief How far it has got.
127 TurnStatus status = TurnStatus::Running;
128 /// @brief The reply so far: the rounds' texts, a blank line between one round and the next.
129 std::string text;
130 /// @brief Why it failed; empty otherwise.
131 std::string error;
132 /// @brief The tool calls it has made, in order.
133 std::vector<ToolStep> steps;
134 /// @brief The rounds' texts and the tool calls, in the order the turn made them. Empty from a server that does not
135 /// send it, which a client shows as `steps` and then `text`.
136 std::vector<TurnPart> parts;
137 /// @brief The `TurnEvent::seq` this state covers, so a client does not apply an event it already includes.
138 std::uint64_t seq = 0;
139 /// @brief The document of the last draft the server accepted this turn, as JSON text; absent when it accepted
140 /// none.
141 /// A draft reaches a client only through here and the `draft` event, and only after the server's checks
142 /// passed it.
143 std::optional<std::string> draft;
144 /// @brief The dictation this server offers; set only in the answer to a request with no turn (the probe), and
145 /// absent when the server offers none. An older peer ignores it.
146 std::optional<DictationOffer> dictation;
147};

155/// @brief A hint that a turn moved. Events can be lost, so `TurnState` is the record.
156struct TurnEvent {
157 /// @brief The turn's id.
158 std::uint64_t turnId = 0;
159 /// @brief Increases by one per event of a turn.
160 std::uint64_t seq = 0;
161 /// @brief `text`, `tool`, `draft`, `done` or `error`.
162 std::string kind;
163 /// @brief The text piece, the tool's name, the reason, or a `draft` event's accepted document.
164 std::string text;
165 /// @brief A `tool` event's arguments as JSON text.
166 std::string arguments;
167 /// @brief A `tool` event's outcome: whether the action ran.
168 bool ok = false;
169 /// @brief Whether a `text` event is the first piece of a round that follows tool calls, so it starts a new text.
170 bool newRound = false;
171};
