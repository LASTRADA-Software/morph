21/// @brief How one sample of a device's stream is stored, in the machine's byte order.
22enum class DictationSample : std::uint8_t {
23 UInt8, ///< Unsigned, silence at 128.
24 Int16, ///< Signed 16-bit.
25 Int32, ///< Signed 32-bit.
26 Float, ///< 32-bit float in [-1, 1].

29/// @brief An open audio input: the device its bytes arrive on, and the format they are in.
30struct DictationInput {
31 /// @brief Where the samples arrive, interleaved; `readyRead` says there are more. Owned by `keepAlive`.
32 QIODevice* device = nullptr;
33 /// @brief The channel count.
34 int channels = 1;

36 int sampleRate = ui::chat::kSpeechSampleRate;

38 DictationSample sample = DictationSample::Int16;
39 /// @brief Whatever keeps the device open. Released on the recorder's thread, once control is back in its event
40 /// loop after a recording ended, never from inside the device's own `readyRead`.

42};

44/// @brief Opens an audio input, or says why it cannot.

47/// @brief Asks whether the microphone may be used, and calls `answer` with the outcome, now or later.

50/// @brief What a `QtDictationRecorder` is made from. Each empty member takes the platform's own.
