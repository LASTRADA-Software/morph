37/// @brief The stable identity of a slot: a bound property a renderer can refer to by number.
38///
39/// Whoever builds the tree assigns slot ids; the interpreter numbers every bound property of a document, and a
40/// code-generating renderer names the value it reads after it (`v.s17`). `Mounted` does not use them. A slot made

44/// @brief A slot's read together with its id, as `ui::slot` makes it; a `Prop` made from it is a slot with that id.
45/// @tparam F A callable taking no arguments.
46template <typename F>
47struct SlotBinding {
48 /// @brief The slot's id.
49 SlotId id = 0;
50 /// @brief Reads the slot's value; every signal it reads is a dependency.
51 F read;

54/// @brief A slot with an id.
55/// @tparam F A callable taking no arguments.
56/// @param slotId The slot's id.
57/// @param read Reads the value; every signal it reads is a dependency.
58/// @return What a `Prop` takes as a slot with @p slotId.
59template <typename F>
60[[nodiscard]] SlotBinding<std::decay_t<F>> slot(SlotId slotId, F&& read) {
61 return SlotBinding<std::decay_t<F>>{.id = slotId, .read = std::forward<F>(read)};

76/// @brief Whether `From` may become a constant `Prop<To>`: an implicit conversion that does not silently change
77/// the kind of value.
78///
79/// Two implicit conversions are refused because they compile and then show something else. A pointer, a string
80/// literal included, converts to `bool` as "not null", so `Prop<bool>{"false"}` would be `true`. A floating-point
81/// value converts to an integral type by truncation, so `Prop<std::int64_t>{2.9}` would be `2`. Other narrowing,
82/// such as `int` to `std::size_t`, is kept: `Prop<std::size_t> count = 0` is the ordinary way to write a count.
83///
84/// `From` is the source type as deduced for a forwarding reference; `To` is the property's value type.
85template <typename From, typename To>
86concept ConstantConvertible = std::convertible_to<From, To> &&
87 !(std::is_same_v<std::remove_cv_t<To>, bool> && std::is_pointer_v<std::decay_t<From>>) &&
88 !(std::is_integral_v<To> && std::is_floating_point_v<std::remove_cvref_t<From>>) &&
89 !IsSlotBinding<std::remove_cvref_t<From>>::value;
90

93/// @brief A node property: a constant, or a slot — a reactive read, with a stable id when it was given one.
94///
95/// A constant is set on the widget once and creates no reactive node. A slot becomes an equality-gated
96/// `reactive::Computed` plus an `reactive::Effect` calling one widget setter. An input whose value is a slot is
97/// controlled: after a user's edit, the widget shows what the slot says. A constant is not re-asserted.
98///
99/// A braced list cannot initialise a `Prop`, because the constant constructor deduces its argument's type: name
100/// the type, as in `.options = std::vector<ui::SelectOption>{{...}, {...}}`.
101/// @tparam T The property's value type.
102template <typename T>
103class Prop {
104public:
105 /// @brief The value-initialised constant.
106 Prop()
107 requires std::default_initializable<T>
108 : _value{std::in_place_index<0>} {}
109
110 /// @brief A constant.
111 ///
112 /// A callable is never a constant, even when it converts to `T`: a captureless lambda converts to a function
113 /// pointer and that to `bool`, so `Prop<bool>{[] { return false; }}` would otherwise be the constant `true`.
114 ///
115 /// A pointer does not become a `bool` constant and a floating-point value does not become an integral one; see
116 /// `detail::ConstantConvertible`.
117 /// @tparam U A non-callable type that converts to `T` without changing the kind of value.
118 /// @param value The constant.
119 template <typename U>
120 requires(!std::invocable<U&>) && detail::ConstantConvertible<U, T>
121 Prop(U&& value)
122 // A string literal converting to `T` decays here; that is the conversion.
123 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-array-to-pointer-decay)
124 : _value{std::in_place_index<0>, static_cast<T>(std::forward<U>(value))} {}
125
126 /// @brief A slot without an id.
127 ///
128 /// The read must be callable: an empty `std::function` or a null function pointer is refused here, where the
129 /// mistake is made, rather than when a mount first calls it.
130 /// @tparam F A callable taking no arguments and returning something convertible to `T`.
131 /// @param binding Evaluated by the mount inside a `Computed`; every signal it reads is a dependency.
132 /// @throws std::invalid_argument when `binding` is an empty `std::function` or a null function pointer.
133 template <typename F>
134 requires std::invocable<F&> && std::convertible_to<std::invoke_result_t<F&>, T>
135 Prop(F binding) : _value{std::in_place_index<1>, Slot{.id = std::nullopt, .read = std::move(binding)}} {
136 refuseEmpty();
137 }
138
139 /// @brief A slot with an id, as `ui::slot` makes it.
140 /// @tparam F A callable taking no arguments and returning something convertible to `T`.
141 /// @param binding The id and the read.
142 /// @throws std::invalid_argument when the read is an empty `std::function` or a null function pointer.
143 template <typename F>
144 requires std::invocable<F&> && std::convertible_to<std::invoke_result_t<F&>, T>
145 Prop(SlotBinding<F> binding)
146 : _value{std::in_place_index<1>, Slot{.id = binding.id, .read = std::move(binding.read)}} {
147 refuseEmpty();
148 }
149
150 /// @brief Whether this is a slot.
151 /// @return True for a slot, false for a constant.
152 [[nodiscard]] bool isBound() const noexcept { return _value.index() == 1; }
153
154 /// @brief The constant. Only valid when `!isBound()`.
155 /// @return The constant.
156 /// @throws std::bad_variant_access when this is a slot.
157 [[nodiscard]] T const& constant() const { return std::get<0>(_value); }
158
159 /// @brief The slot's read. Only valid when `isBound()`.
160 /// @return The read.
161 /// @throws std::bad_variant_access when this is a constant.
162 [[nodiscard]] std::function<T()> const& binding() const { return std::get<1>(_value).read; }
163
164 /// @brief The slot's id.
165 /// @return The id given through `ui::slot`; `nullopt` for a constant and for a slot made from a bare callable.
166 [[nodiscard]] std::optional<SlotId> slotId() const noexcept {
167 Slot const* const bound = std::get_if<1>(&_value);
168 return bound == nullptr ? std::nullopt : bound->id;
169 }
170
171 /// @brief The current value: the constant, or the slot read once.
172 /// @return The value.
173 [[nodiscard]] T evaluate() const { return isBound() ? std::get<1>(_value).read() : std::get<0>(_value); }
174
175private:
176 /// @brief A slot: its id, if it has one, and its read.
177 struct Slot {
178 /// @brief The id, or none.
179 std::optional<SlotId> id;
180 /// @brief The read; never empty.
181 std::function<T()> read;
182 };
183
184 /// @brief Refuses a slot whose read is empty: an empty `std::function` or a null function pointer.
185 /// @throws std::invalid_argument when the read is empty.
186 void refuseEmpty() const {
187 if (!std::get<1>(_value).read) {
188 throw std::invalid_argument{"ui::Prop: a binding must not be empty"};
189 }
190 }
191
192 std::variant<T, Slot> _value;
193};

1211/// @brief A Switch over an enumeration: each enumerator becomes the `int64` key of its case.
1212///
1213/// Enumerators are compared as `int64`: an unsigned value above `INT64_MAX` wraps to a negative key, and distinct
1214/// enumerators still give distinct keys.
1215/// @tparam E The enumeration.
1216/// @param selector Which enumerator to show; every signal it reads is a dependency.
1217/// @param cases One node per enumerator shown.
1218/// @param fallback Shown when no case matches; may be null.
1219/// @return The node.
1220/// @throws std::invalid_argument when @p selector is empty, as a `Prop` refuses an empty binding.
1221template <typename E>
1222 requires std::is_enum_v<E>
1223[[nodiscard]] Node switchOn(std::function<E()> selector, std::vector<std::pair<E, Node>> cases, Node fallback = {}) {
1224 if (!selector) {
1225 throw std::invalid_argument{"ui::switchOn: the selector must not be empty"};
1226 }
1227 std::vector<SwitchCase> keyed;
1228 keyed.reserve(cases.size());
1229 for (auto& [value, node] : cases) {
1230 keyed.push_back(SwitchCase{.key = Key{static_cast<std::int64_t>(value)}, .node = std::move(node)});
1231 }
1232 Switch spec{
1233 .selector = [current = std::move(selector)] { return Key{static_cast<std::int64_t>(current())}; },
1234 .cases = std::move(keyed),
1235 };
1236 spec.fallback = std::move(fallback);
1237 return switchOf(std::move(spec));
