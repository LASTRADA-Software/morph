5/// @file render/locale_format.hpp
6/// @brief Locale numeric-entry normalisation for the control edge.
7///
8/// A locale may render and accept e.g. `"1.050,25"`; the payload a
9/// `morph::units::Quantity` field submits is always the canonical exact
10/// `{num, den, dp}` regardless. This header is the one control-edge
11/// conversion step between the two: the exact `Rational`/`Quantity` digit
12/// routines stay entirely locale-free (they only ever see plain
13/// `.`-decimal text), and a renderer calls `normalizeLocaleNumber` once, at
14/// the point text leaves the control, before handing it to those routines.
15///
16/// @par Separators are strings, not characters
17/// Both functions take their separators as `std::string_view`, because a
18/// real locale's separator is not always one byte. fr-FR groups with U+202F
19/// (narrow no-break space) and several locales use U+00A0 — three and two
20/// UTF-8 bytes respectively. Typed as `char`, those cannot be expressed at
21/// all: the caller can only pass some single byte that never matches, so a
22/// perfectly valid `"1 050,25"` typed by a French user normalises to
23/// `std::nullopt` and the entry is reported malformed. An empty view means
24/// "this locale has no such separator" (the role `'\0'` used to play).

35/// @brief Whether every group separator in @p text sits where a group
36/// separator can legally sit.
37///
38/// The rule, in one place so it can be read on its own: a group separator is
39/// preceded by one to three digits (the first group), or by exactly three
40/// (every later one); it is followed by exactly three more digits, at each
41/// group and at the end of the integer part; and it never appears after the
42/// decimal separator. A locale with no grouping (@p groupSeparator empty) has
43/// nothing to place, so it trivially passes.
44///
45/// This is a pass of its own rather than extra state inside the normalising
46/// scan below: "the grouping is well placed" and "the digits convert" are two
47/// separate statements about the entry, and reading them as one made neither
48/// clear.
49///
50/// Characters this function does not recognise are simply not digits — the
51/// normalising scan is what rejects them, and it rejects them whatever this
52/// pass concludes.
53/// @param text The locale-formatted entry.
54/// @param decimalSeparator The locale's decimal-point string; may be empty.
55/// @param groupSeparator The locale's digit-grouping string; empty means the
56/// locale does not group.
57/// @return `true` when the grouping is well placed (or absent).
58[[nodiscard]] inline bool groupingIsWellPlaced(std::string_view text, std::string_view decimalSeparator,
59 std::string_view groupSeparator) {
60 if (groupSeparator.empty()) {
61 return true;
62 }
63 constexpr std::size_t kGroupSize = 3;
64 std::size_t digits = 0;
65 bool sawGroup = false;
66 bool sawDecimal = false;
67
68 for (std::size_t i = 0; i < text.size();) {
69 const std::string_view rest = text.substr(i);
70 if (rest.starts_with(groupSeparator)) {
71 bool const opensAGroup = sawGroup ? digits == kGroupSize : (digits >= 1 && digits <= kGroupSize);
72 if (sawDecimal || !opensAGroup) {
73 return false;
74 }
75 sawGroup = true;
76 digits = 0;
77 i += groupSeparator.size();
78 continue;
79 }
80 if (!decimalSeparator.empty() && rest.starts_with(decimalSeparator)) {
81 if (sawGroup && digits != kGroupSize) {
82 return false; // the last group of the integer part is short
83 }
84 sawDecimal = true;
85 digits = 0;
86 i += decimalSeparator.size();
87 continue;
88 }
89 // i is bounded by the loop condition.
90 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
91 char const chr = text[i];
92 digits = (chr >= '0' && chr <= '9') ? digits + 1 : 0;
93 ++i;
94 }
95
96 // An ungrouped fractional part ends the number, so the trailing check only
97 // applies when the integer part was the last thing scanned.
98 return !sawGroup || sawDecimal || digits == kGroupSize;
99}

103/// @brief Converts a locale-formatted numeric string to canonical
104/// (`-?[0-9]+(\.[0-9]+)?`) text.
105///
106/// Drops @p groupSeparator where it is correctly placed, and replaces every
107/// occurrence of @p decimalSeparator with `.`. Passing `decimalSeparator ==
108/// "."` and an empty @p groupSeparator is the identity transform (the
109/// locale-free behavior). Malformed input (a second decimal separator, a
110/// sign anywhere but the leading position of the *output*, or any character that
111/// is not a digit) yields `std::nullopt` rather than a best-effort guess. The
112/// decimal point counts as output, so a sign placed straight after the separator
113/// ("`,-5`" in a de-DE locale) is rejected -- matching the QML mirror in
114/// `src/qt/forms/qml/DynamicForm.qml`, which has always rejected it (morph#497).
115///
116/// @par Grouping is validated, not stripped (morph#574)
117/// A group separator is only dropped where a group separator can legally be:
118/// preceded by one to three digits, followed by exactly three more, and never
119/// after the decimal separator. Anything else is malformed and reported as
120/// such. Stripping unconditionally instead is a wrong *value*, not a rejected
121/// one: a de-DE user typing the US form `"1.5"` into a price field submitted
122/// `15`, and nothing downstream could tell -- the result is a perfectly valid
123/// number, ten times too large. `"1.50"` gave `150`, `"1.2.3.4"` gave `1234`,
124/// and the en-US mirror image `"1,5"` gave `15`.
125///
126/// @par The two separators must differ
127/// When @p groupSeparator is non-empty and equal to @p decimalSeparator the
128/// entry is rejected: with one string in both roles there is no reading of
129/// `"1.5"` the function could defend. This is a caller (locale-configuration)
130/// error rather than a user one, but it is reported through the return value
131/// like any other malformed entry, deliberately not through an assertion --
132/// an assertion would make the two build configurations behave differently at
133/// a control edge, and would be untestable in the one where it fires.
134///
135/// The result is `.`-decimal and digit-only, but is **not** narrowed to
136/// `-?[0-9]+(\.[0-9]+)?`: a bare "`.`", a leading "`.5`" and a trailing "`5.`"
137/// are passed through, exactly as that same QML mirror passes them. Tightening
138/// one side alone would put the two control edges back out of step, so the shape
139/// is documented here rather than changed.
140///
141/// Separators are matched as whole strings, so a multi-byte one (e.g. U+202F)
142/// works; matching them before the per-byte digit scan is what keeps their
143/// continuation bytes from being mistaken for stray non-digit characters.
144///
145/// @param text The locale-formatted entry, e.g. `"1.050,25"`.
146/// @param decimalSeparator The locale's decimal-point string, e.g. `","`.
147/// @param groupSeparator The locale's digit-grouping string, e.g. `"."`, or
148/// empty when the locale has none.
149/// @return The canonical `.`-decimal text, or `std::nullopt` when malformed.
150[[nodiscard]] inline std::optional<std::string> normalizeLocaleNumber(std::string_view text,
151 std::string_view decimalSeparator,
152 std::string_view groupSeparator) {
153 if (!groupSeparator.empty() && groupSeparator == decimalSeparator) {
154 return std::nullopt; // one string cannot play both roles: see above
155 }
156 if (!detail::groupingIsWellPlaced(text, decimalSeparator, groupSeparator)) {
157 return std::nullopt; // a separator off a group boundary: see above
158 }
159
160 std::string canonical;
161 canonical.reserve(text.size());
162 bool sawDecimal = false;
163 bool sawAnyOutput = false;
164
165 for (std::size_t i = 0; i < text.size();) {
166 const std::string_view rest = text.substr(i);
167 if (!groupSeparator.empty() && rest.starts_with(groupSeparator)) {
168 // Placement was settled above, so by here the separator is display
169 // only and is never carried into the output.
170 i += groupSeparator.size();
171 continue;
172 }
173 if (!decimalSeparator.empty() && rest.starts_with(decimalSeparator)) {
174 if (sawDecimal) {
175 return std::nullopt; // a second decimal separator: malformed
176 }
177 sawDecimal = true;
178 canonical += '.';
179 // The decimal point *is* output: without this, the `chr == '-'`
180 // guard below still believes nothing has been emitted, and a sign
181 // placed straight after the separator ("`,-5`" in a de-DE locale)
182 // is accepted as if it were leading. morph#497.
183 sawAnyOutput = true;
184 i += decimalSeparator.size();
185 continue;
186 }
187 // i is bounded by the loop condition.
188 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
189 char const chr = text[i];
190 if (chr == '-') {
191 // Leading position of the *output*: a stripped group separator
192 // before the sign would otherwise make an injected sign look
193 // leading.
194 if (sawAnyOutput) {
195 return std::nullopt; // sign injection past the leading position
196 }
197 canonical += chr;
198 } else if (chr >= '0' && chr <= '9') {
199 canonical += chr;
200 } else {
201 return std::nullopt; // any other character is malformed
202 }
203 sawAnyOutput = true;
204 ++i;
205 }
206
207 if (canonical.empty() || canonical == "-") {
208 return std::nullopt;
209 }
210 return canonical;
211}

213/// @brief Converts canonical (`.`-decimal) numeric text to locale-formatted
214/// display text, grouping the integer part in triples.
215///
216/// The display-direction inverse of `normalizeLocaleNumber`'s
217/// decimal-separator substitution, plus display-only thousands grouping
218/// (grouping is never accepted back on entry — `normalizeLocaleNumber`
219/// strips it unconditionally). Passing `decimalSeparator == "."` and an empty
220/// @p groupSeparator is the identity transform.
221/// @param canonicalText Canonical `-?[0-9]+(\.[0-9]+)?` text.
222/// @param decimalSeparator The locale's decimal-point display string.
223/// @param groupSeparator The locale's digit-grouping display string, or empty
224/// to omit grouping.
225/// @return The locale-formatted display text.
226// Mirrors normalizeLocaleNumber's parameter order; the two are inverses, so
227// diverging here would be the more confusing choice.
228// NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
229[[nodiscard]] inline std::string formatCanonicalNumber(std::string_view canonicalText,
230 std::string_view decimalSeparator,
231 std::string_view groupSeparator) {
232 bool const neg = !canonicalText.empty() && canonicalText.front() == '-';
233 std::string_view const magnitude = neg ? canonicalText.substr(1) : canonicalText;
234 auto const dot = magnitude.find('.');
235 std::string_view const wholePart = dot == std::string_view::npos ? magnitude : magnitude.substr(0, dot);
236 std::string_view const fracPart = dot == std::string_view::npos ? std::string_view{} : magnitude.substr(dot + 1);
237
238 std::string grouped;
239 grouped.reserve(wholePart.size() + ((wholePart.size() / 3) * groupSeparator.size()));
240 for (std::size_t i = 0; i < wholePart.size(); ++i) {
241 if (!groupSeparator.empty() && i != 0 && (wholePart.size() - i) % 3 == 0) {
242 grouped += groupSeparator;
243 }
244 grouped += wholePart[i];
245 }
246
247 std::string out;
248 if (neg) {
249 out += '-';
250 }
251 out += grouped;
252 if (!fracPart.empty()) {
253 out += decimalSeparator;
254 out += fracPart;
255 }
256 return out;
257}
