5/// @file render/locale_format.hpp
6/// @brief Locale numeric-entry normalisation for the control edge.
7///
8/// A locale may render and accept e.g. `"1.050,25"`; the payload a
9/// `morph::units::Quantity` field submits is always the canonical exact
10/// `{num, den, dp}` regardless. This header is the one control-edge
11/// conversion step between the two: the exact `Rational`/`Quantity` digit
12/// routines stay entirely locale-free (they only ever see plain
13/// `.`-decimal text), and a renderer calls `normalizeLocaleNumber` once, at
14/// the point text leaves the control, before handing it to those routines.
15///
16/// @par Separators are strings, not characters
17/// Both functions take their separators as `std::string_view`, because a
18/// real locale's separator is not always one byte. fr-FR groups with U+202F
19/// (narrow no-break space) and several locales use U+00A0 — three and two
20/// UTF-8 bytes respectively. Typed as `char`, those cannot be expressed at
21/// all: the caller can only pass some single byte that never matches, so a
22/// perfectly valid `"1 050,25"` typed by a French user normalises to
23/// `std::nullopt` and the entry is reported malformed. An empty view means
24/// "this locale has no such separator" (the role `'\0'` used to play).
25

33/// @brief Converts a locale-formatted numeric string to canonical
34/// (`-?[0-9]+(\.[0-9]+)?`) text.
35///
36/// Strips every occurrence of @p groupSeparator, then replaces every
37/// occurrence of @p decimalSeparator with `.`. Passing `decimalSeparator ==
38/// "."` and an empty @p groupSeparator is the identity transform (the
39/// locale-free behavior). Malformed input (a second decimal separator, a
40/// sign anywhere but the leading position, or any character that is not a
41/// digit) yields `std::nullopt` rather than a best-effort guess.
42///
43/// Separators are matched as whole strings, so a multi-byte one (e.g. U+202F)
44/// works; matching them before the per-byte digit scan is what keeps their
45/// continuation bytes from being mistaken for stray non-digit characters.
46///
47/// @param text The locale-formatted entry, e.g. `"1.050,25"`.
48/// @param decimalSeparator The locale's decimal-point string, e.g. `","`.
49/// @param groupSeparator The locale's digit-grouping string, e.g. `"."`, or
50/// empty when the locale has none.
51/// @return The canonical `.`-decimal text, or `std::nullopt` when malformed.
52[[nodiscard]] inline std::optional<std::string> normalizeLocaleNumber(std::string_view text,
53 std::string_view decimalSeparator,
54 std::string_view groupSeparator) {
55 std::string canonical;
56 canonical.reserve(text.size());
57 bool sawDecimal = false;
58 bool sawAnyOutput = false;
59
60 for (std::size_t i = 0; i < text.size();) {
61 const std::string_view rest = text.substr(i);
62 if (!groupSeparator.empty() && rest.starts_with(groupSeparator)) {
63 i += groupSeparator.size();
64 continue; // grouping is display-only; never accepted back on entry
65 }
66 if (!decimalSeparator.empty() && rest.starts_with(decimalSeparator)) {
67 if (sawDecimal) {
68 return std::nullopt; // a second decimal separator: malformed
69 }
70 sawDecimal = true;
71 canonical += '.';
72 i += decimalSeparator.size();
73 continue;
74 }
75 // i is bounded by the loop condition.
76 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
77 char const chr = text[i];
78 if (chr == '-') {
79 // Leading position of the *output*: a stripped group separator
80 // before the sign would otherwise make an injected sign look
81 // leading.
82 if (sawAnyOutput) {
83 return std::nullopt; // sign injection past the leading position
84 }
85 canonical += chr;
86 } else if (chr >= '0' && chr <= '9') {
87 canonical += chr;
88 } else {
89 return std::nullopt; // any other character is malformed
90 }
91 sawAnyOutput = true;
92 ++i;
93 }
94
95 if (canonical.empty() || canonical == "-") {
96 return std::nullopt;
97 }
98 return canonical;
99}

101/// @brief Converts canonical (`.`-decimal) numeric text to locale-formatted
102/// display text, grouping the integer part in triples.
103///
104/// The display-direction inverse of `normalizeLocaleNumber`'s
105/// decimal-separator substitution, plus display-only thousands grouping
106/// (grouping is never accepted back on entry — `normalizeLocaleNumber`
107/// strips it unconditionally). Passing `decimalSeparator == "."` and an empty
108/// @p groupSeparator is the identity transform.
109/// @param canonicalText Canonical `-?[0-9]+(\.[0-9]+)?` text.
110/// @param decimalSeparator The locale's decimal-point display string.
111/// @param groupSeparator The locale's digit-grouping display string, or empty
112/// to omit grouping.
113/// @return The locale-formatted display text.
114// Mirrors normalizeLocaleNumber's parameter order; the two are inverses, so
115// diverging here would be the more confusing choice.
116// NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
117[[nodiscard]] inline std::string formatCanonicalNumber(std::string_view canonicalText,
118 std::string_view decimalSeparator,
119 std::string_view groupSeparator) {
120 bool const neg = !canonicalText.empty() && canonicalText.front() == '-';
121 std::string_view const magnitude = neg ? canonicalText.substr(1) : canonicalText;
122 auto const dot = magnitude.find('.');
123 std::string_view const wholePart = dot == std::string_view::npos ? magnitude : magnitude.substr(0, dot);
124 std::string_view const fracPart = dot == std::string_view::npos ? std::string_view{} : magnitude.substr(dot + 1);
125
126 std::string grouped;
127 grouped.reserve(wholePart.size() + ((wholePart.size() / 3) * groupSeparator.size()));
128 for (std::size_t i = 0; i < wholePart.size(); ++i) {
129 if (!groupSeparator.empty() && i != 0 && (wholePart.size() - i) % 3 == 0) {
130 grouped += groupSeparator;
131 }
132 grouped += wholePart[i];
133 }
134
135 std::string out;
136 if (neg) {
137 out += '-';
138 }
139 out += grouped;
140 if (!fracPart.empty()) {
141 out += decimalSeparator;
142 out += fracPart;
143 }
144 return out;
145}
