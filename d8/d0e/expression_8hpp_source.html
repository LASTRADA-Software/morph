42/// @brief Whether a string is a document name: `[A-Za-z_][A-Za-z0-9_]*`.
43/// @param text The candidate.
44/// @return Whether @p text is a well-formed name.
45[[nodiscard]] constexpr bool isName(std::string_view text) noexcept {
46 if (text.empty()) {
47 return false;
48 }
49 bool first = true;
50 for (char const chr : text) {
51 bool const letter = (chr >= 'a' && chr <= 'z') || (chr >= 'A' && chr <= 'Z') || chr == '_';
52 bool const digit = chr >= '0' && chr <= '9';
53 if (!letter && (first || !digit)) {
54 return false;
55 }
56 first = false;
57 }
58 return true;
59}

61/// @brief The forms of an `expr/1` expression (spec 5 §4).
62enum class ExprKind : std::uint8_t {
63 Literal, ///< A JSON scalar, or `{"lit": v}`.
64 Ref, ///< `{"ref": "name.path"}`.
65 Row, ///< `{"row": "path"}`.
66 Query, ///< `{"query": "q"}`: a query's value.
67 Pending, ///< `{"pending": "q"}`: whether a query or mutation is in flight.
68 Error, ///< `{"error": "q"}`: a query's or mutation's error text.
69 Result, ///< `{"result": "m"}`: a mutation's last result.
70 ErrorKind, ///< `{"errorKind": "m"}`: the kind of a query's or mutation's error.
71 Can, ///< `{"can": "m"}`: whether the principal may run a mutation or query.
72 Pref, ///< `{"pref": "point.path"}`: the user's choice at a customization point.
73 Call, ///< `{"fn": "name", "args": [...]}`, and `{"t": ...}`.
74 If, ///< `{"if": [cond, then, else]}`.
75 Object, ///< `{"obj": {...}}`.
76 List, ///< `{"list": [...]}`.
77 Map, ///< `{"map": e, "as": "x", "to": e}`.
78 Filter, ///< `{"filter": e, "as": "x", "where": e}`.
79 Env, ///< `{"env": "name"}`: the client environment.
80 Session, ///< `{"session": "name"}`: a session attribute.
81 Host, ///< `{"host": "name"}`: a value the application registered with the client.
82};

134/// @brief One node of a parsed expression. Immutable once parsed; shared between the document and its readers.
135struct Expr {
136 /// @brief The form.
137 ExprKind kind = ExprKind::Literal;
138 /// @brief The JSON Pointer of the node inside its document, for refusals and diagnostics.
139 std::string path;
140 /// @brief The constant of a `Literal`.
141 Value literal;
142 /// @brief The name a form reads: a ref's head, a query or mutation, a pref's point, a comprehension's variable,
143 /// an `env`, `session` or `host` name.
144 std::string name;
145 /// @brief The path after the head of a `Ref` or `Pref`, or the path of a `Row`.
146 std::vector<std::string> fieldPath;
147 /// @brief The function of a `Call`.
148 FunctionId function = FunctionId::And;
149 /// @brief Operands: a call's arguments; `if`'s condition, then and else; a list's items; a comprehension's
150 /// source and body.
151 std::vector<ExprPtr> args;
152 /// @brief The fields of an `Object`, in the order written.
153 std::vector<std::pair<std::string, ExprPtr>> fields;
154};

266/// @brief Whether a name is one of a closed set.

268/// @param name The candidate.

275/// @brief Splits `head.a.b` into its segments.
276/// @param text The dotted path.
277/// @return The segments, or the reason the path is malformed.
278[[nodiscard]] inline std::expected<std::vector<std::string>, std::string> splitPath(std::string_view text) {
279 std::vector<std::string> segments;
280 std::size_t start = 0;
281 for (;;) {
282 std::size_t const dot = text.find('.', start);
283 std::string_view const segment =
284 text.substr(start, dot == std::string_view::npos ? std::string_view::npos : dot - start);
285 if (segment.empty()) {
286 return std::unexpected("`" + std::string{text} + "` has an empty segment");
287 }
288 segments.emplace_back(segment);
289 if (dot == std::string_view::npos) {
290 return segments;
291 }
292 start = dot + 1;
293 }
294}

296/// @brief Reads one JSON expression tree into `Expr` nodes, counting them against the size bound.
297// NOLINTBEGIN(misc-no-recursion): an expression is a tree and parsing descends it; the node bound limits the depth.
298class ExprParser {
299public:
300 /// @brief The keys a form may carry beside its own.
301 struct FormKeys {
302 /// @brief The form's own key.
303 std::string_view key;
304 /// @brief The form.
305 ExprKind kind;
306 /// @brief The keys allowed beside it.
307 std::array<std::string_view, 2> companions;
308 };

310 /// @brief Parses @p json, whose JSON Pointer is @p path.
311 /// @param json The expression.
312 /// @param path Its JSON Pointer.
313 /// @return The expression, or the refusal.
314 [[nodiscard]] std::expected<ExprPtr, Refusal> parse(Json const& json, std::string const& path) {
315 return node(json, path);
316 }
317
318private:
319 // NOLINTBEGIN(modernize-use-designated-initializers): one row per form reads as a table.
320 static constexpr std::array<FormKeys, 20> kForms = {{
321 {"lit", ExprKind::Literal, {}},
322 {"ref", ExprKind::Ref, {}},
323 {"row", ExprKind::Row, {}},
324 {"query", ExprKind::Query, {}},
325 {"pending", ExprKind::Pending, {}},
326 {"error", ExprKind::Error, {}},
327 {"result", ExprKind::Result, {}},
328 {"errorKind", ExprKind::ErrorKind, {}},
329 {"can", ExprKind::Can, {}},
330 {"pref", ExprKind::Pref, {}},
331 {"fn", ExprKind::Call, {"args", ""}},
332 {"t", ExprKind::Call, {"fallback", "args"}},
333 {"if", ExprKind::If, {}},
334 {"obj", ExprKind::Object, {}},
335 {"list", ExprKind::List, {}},
336 {"map", ExprKind::Map, {"as", "to"}},
337 {"filter", ExprKind::Filter, {"as", "where"}},
338 {"env", ExprKind::Env, {}},
339 {"session", ExprKind::Session, {}},
340 {"host", ExprKind::Host, {}},
341 }};

363 [[nodiscard]] std::expected<ExprPtr, Refusal> node(Json const& json, std::string const& path) {
364 if (++_count > kMaxExpressionNodes) {
365 return refuse(path, "the expression has more than " + std::to_string(kMaxExpressionNodes) +
366 " nodes, the bound of expr/1");
367 }
368 if (std::holds_alternative<Json::array_t>(json.data)) {
369 return refuse(path, R"(a list is written {"list": [...]}, a constant list {"lit": [...]})");
370 }
371 auto const* const object = std::get_if<Json::object_t>(&json.data);
372 if (object == nullptr) {
373 auto value = valueFromJson(json);
374 if (!value) {
375 return refuse(path, value.error());
376 }
377 return literal(std::move(*value), path);
378 }
379 auto made = std::make_shared<Expr>();
380 made->path = path;
381 if (auto refused = form(*made, *object, path)) {
382 return std::unexpected(std::move(*refused));
383 }
384 return made;
385 }

387 [[nodiscard]] std::optional<Refusal> form(Expr& made, Json::object_t const& object, std::string const& path) {
388 FormKeys const* chosen = nullptr;
389 for (auto const& [key, value] : object) {
390 static_cast<void>(value);
391 for (FormKeys const& candidate : kForms) {
392 if (candidate.key != key) {
393 continue;
394 }
395 if (chosen != nullptr) {
396 return refuse(path, "an expression has one form; found `" + std::string{chosen->key} + "` and `" +
397 key + "`")
398 .error();
399 }
400 chosen = &candidate;
401 }
402 }
403 if (chosen == nullptr) {
404 if (object.empty()) {
405 return refuse(path, "an empty object is not an expression; a constant record is {\"lit\": {}}")
406 .error();
407 }
408 return refuse(path, "`" + object.begin()->first + "` is not an expression form of expr/1").error();
409 }
410 for (auto const& [key, value] : object) {
411 static_cast<void>(value);
412 if (key != chosen->key && key != chosen->companions.at(0) && key != chosen->companions.at(1)) {
413 return refuse(pointerAppend(path, key),
414 "`" + key + "` does not belong to the `" + std::string{chosen->key} + "` form")
415 .error();
416 }
417 }
418 Json const& primary = *member(object, chosen->key);
419 std::string const primaryPath = pointerAppend(path, chosen->key);
420 made.kind = chosen->kind;
421 switch (chosen->kind) {
422 case ExprKind::Literal: {
423 auto value = valueFromJson(primary);
424 if (!value) {
425 return refuse(primaryPath, value.error()).error();
426 }
427 made.literal = std::move(*value);
428 return std::nullopt;
429 }
430 case ExprKind::Ref:
431 case ExprKind::Pref:
432 return dotted(made, primary, primaryPath);
433 case ExprKind::Row:
434 return rowPath(made, primary, primaryPath);
435 case ExprKind::Query:
436 case ExprKind::Pending:
437 case ExprKind::Error:
438 case ExprKind::Result:
439 case ExprKind::ErrorKind:
440 case ExprKind::Can:
441 return controlName(made, primary, primaryPath, chosen->key);
442 case ExprKind::Env:
443 case ExprKind::Session:
444 case ExprKind::Host:
445 return clientName(made, primary, primaryPath, chosen->key);
446 case ExprKind::Call:
447 if (chosen->key == "t") {
448 return translate(made, object, path);
449 }
450 return call(made, object, path);
451 case ExprKind::If:
452 return operands(made.args, primary, primaryPath, 3, "`if` takes [condition, then, else]");
453 case ExprKind::List:
454 return operands(made.args, primary, primaryPath, 0, "`list` takes an array of expressions");
455 case ExprKind::Object:
456 return fields(made.fields, primary, primaryPath);
457 case ExprKind::Map:
458 case ExprKind::Filter:
459 default:
460 return comprehension(made, object, path);
461 }
462 }

464 [[nodiscard]] static std::optional<Refusal> dotted(Expr& made, Json const& primary, std::string const& path) {
465 std::string const* const dottedPath = text(primary);
466 if (dottedPath == nullptr) {
467 return refuse(path, "a name and path is written as a string `name.path`").error();
468 }
469 auto segments = splitPath(*dottedPath);
470 if (!segments) {
471 return refuse(path, segments.error()).error();
472 }
473 if (!isName(segments->front())) {
474 return refuse(path, "`" + segments->front() + "` is not a name").error();
475 }
476 made.name = std::move(segments->front());
477 made.fieldPath.assign(std::make_move_iterator(segments->begin() + 1),
478 std::make_move_iterator(segments->end()));
479 return std::nullopt;
480 }

482 [[nodiscard]] static std::optional<Refusal> rowPath(Expr& made, Json const& primary, std::string const& path) {
483 std::string const* const dottedPath = text(primary);
484 if (dottedPath == nullptr) {
485 return refuse(path, "`row` names a field path as a string, \"\" for the whole row").error();
486 }
487 if (dottedPath->empty()) {
488 return std::nullopt;
489 }
490 auto segments = splitPath(*dottedPath);
491 if (!segments) {
492 return refuse(path, segments.error()).error();
493 }
494 made.fieldPath = std::move(*segments);
495 return std::nullopt;
496 }

498 [[nodiscard]] static std::optional<Refusal> controlName(Expr& made, Json const& primary, std::string const& path,
499 std::string_view key) {
500 std::string const* const name = text(primary);
501 bool const can = made.kind == ExprKind::Can;
502 bool const hostCommand = can && name != nullptr && name->starts_with("host:") && name->size() > 5;
503 if (name == nullptr || (!isName(*name) && !hostCommand)) {
504 return refuse(path, "`" + std::string{key} + "` takes the name of a " +
505 (can ? "mutation, query or `host:` command" : "query or mutation"))
506 .error();
507 }
508 made.name = *name;
509 return std::nullopt;
510 }

512 [[nodiscard]] static std::optional<Refusal> clientName(Expr& made, Json const& primary, std::string const& path,
513 std::string_view key) {
514 std::string const* const name = text(primary);
515 bool known = name != nullptr && !name->empty();
516 if (known && made.kind == ExprKind::Env) {
517 known = oneOf(kEnvNames, *name);
518 } else if (known && made.kind == ExprKind::Session) {
519 known = oneOf(kSessionNames, *name);
520 }
521 if (!known) {
522 return refuse(path, "`" + std::string{key} + "` does not read this name").error();
523 }
524 made.name = *name;
525 return std::nullopt;
526 }

528 [[nodiscard]] std::optional<Refusal> operands(std::vector<ExprPtr>& out, Json const& primary,
529 std::string const& path, std::size_t exactly, char const* shape) {
530 auto const* const array = std::get_if<Json::array_t>(&primary.data);
531 if (array == nullptr || (exactly != 0 && array->size() != exactly)) {
532 return refuse(path, shape).error();
533 }
534 for (std::size_t index = 0; index < array->size(); ++index) {
535 auto operand = node(array->at(index), pointerAppend(path, index));
536 if (!operand) {
537 return std::move(operand.error());
538 }
539 out.push_back(std::move(*operand));
540 }
541 return std::nullopt;
542 }

544 [[nodiscard]] std::optional<Refusal> fields(std::vector<std::pair<std::string, ExprPtr>>& out, Json const& primary,
545 std::string const& path) {
546 auto const* const object = std::get_if<Json::object_t>(&primary.data);
547 if (object == nullptr) {
548 return refuse(path, "`obj` takes an object of expressions").error();
549 }
550 for (auto const& [key, value] : *object) {
551 auto field = node(value, pointerAppend(path, key));
552 if (!field) {
553 return std::move(field.error());
554 }
555 out.emplace_back(key, std::move(*field));
556 }
557 return std::nullopt;
558 }

560 [[nodiscard]] std::optional<Refusal> call(Expr& made, Json::object_t const& object, std::string const& path) {
561 std::string const* const name = text(*member(object, "fn"));
562 std::string const namePath = pointerAppend(path, "fn");
563 if (name == nullptr) {
564 return refuse(namePath, "`fn` names a function as a string").error();
565 }
566 auto const index =
567 std::ranges::distance(kFunctions.begin(), std::ranges::find(kFunctions, *name, &FunctionSpec::name));
568 if (std::cmp_equal(index, kFunctions.size())) {
569 return refuse(namePath, "the function `" + *name + "` is not in expr/1").error();
570 }
571 FunctionSpec const& spec = *std::next(kFunctions.begin(), index);
572 made.function = static_cast<FunctionId>(index);
573 if (Json const* const args = member(object, "args")) {
574 if (auto refused = operands(made.args, *args, pointerAppend(path, "args"), 0,
575 "`args` takes an array of expressions")) {
576 return refused;
577 }
578 }
579 if (made.args.size() < spec.minArgs || made.args.size() > spec.maxArgs) {
580 std::string arity = std::to_string(static_cast<unsigned>(spec.minArgs));
581 if (spec.maxArgs == 255) {
582 arity += " or more";
583 } else if (spec.maxArgs != spec.minArgs) {
584 arity += " to " + std::to_string(static_cast<unsigned>(spec.maxArgs));
585 }
586 return refuse(path,
587 "`" + *name + "` takes " + arity + " arguments, not " + std::to_string(made.args.size()))
588 .error();
589 }
590 return std::nullopt;
591 }

593 [[nodiscard]] std::optional<Refusal> translate(Expr& made, Json::object_t const& object, std::string const& path) {
594 made.function = FunctionId::Translate;
595 std::string const* const key = text(*member(object, "t"));
596 Json const* const fallback = member(object, "fallback");
597 std::string const* const fallbackText = fallback == nullptr ? nullptr : text(*fallback);
598 if (key == nullptr || key->empty()) {
599 return refuse(pointerAppend(path, "t"), "`t` names a message key as a string").error();
600 }
601 if (fallbackText == nullptr) {
602 return refuse(pointerAppend(path, "fallback"), "`t` needs its `fallback` text as a string").error();
603 }
604 made.args.push_back(literal(*key, pointerAppend(path, "t")));
605 made.args.push_back(literal(*fallbackText, pointerAppend(path, "fallback")));
606 if (Json const* const args = member(object, "args")) {
607 auto record = std::make_shared<Expr>();
608 record->kind = ExprKind::Object;
609 record->path = pointerAppend(path, "args");
610 if (auto refused = fields(record->fields, *args, record->path)) {
611 return refused;
612 }
613 made.args.push_back(std::move(record));
614 }
615 return std::nullopt;
616 }
617
618 [[nodiscard]] std::optional<Refusal> comprehension(Expr& made, Json::object_t const& object,
619 std::string const& path) {
620 bool const isMap = made.kind == ExprKind::Map;
621 std::string_view const sourceKey = isMap ? "map" : "filter";
622 std::string_view const bodyKey = isMap ? "to" : "where";
623 Json const* const variable = member(object, "as");
624 Json const* const body = member(object, bodyKey);
625 std::string const* const name = variable == nullptr ? nullptr : text(*variable);
626 if (name == nullptr || !isName(*name)) {
627 return refuse(pointerAppend(path, "as"), "`" + std::string{sourceKey} + "` names its element with `as`")
628 .error();
629 }
630 if (body == nullptr) {
631 return refuse(path, "`" + std::string{sourceKey} + "` needs `" + std::string{bodyKey} + "`").error();
632 }
633 made.name = *name;
634 auto source = node(*member(object, sourceKey), pointerAppend(path, sourceKey));
635 if (!source) {
636 return std::move(source.error());
637 }
638 auto each = node(*body, pointerAppend(path, bodyKey));
639 if (!each) {
640 return std::move(each.error());
641 }
642 made.args.push_back(std::move(*source));
643 made.args.push_back(std::move(*each));
644 return std::nullopt;
645 }

647 std::size_t _count = 0;
648 // NOLINTEND(misc-no-recursion)

785/// @brief Walks an expression once, resolving names and checking the statically known types.
786// NOLINTBEGIN(misc-no-recursion): an expression is a tree and checking descends it; the node bound limits the depth.
787class ExprChecker {
788public:
789 /// @brief Checks against @p resolver.
790 /// @param resolver What the names resolve to.
791 explicit ExprChecker(NameResolver const& resolver) : _resolver{&resolver} {}
792
793 /// @brief Checks one node and its operands.
794 /// @param expr The node.
795 /// @return Its static type, or the refusal.
796 [[nodiscard]] std::expected<Type, Refusal> check(Expr const& expr) {
797 switch (expr.kind) {
798 case ExprKind::Literal:
799 return typeOfValue(expr.literal);
800 case ExprKind::Ref:
801 return reference(expr);
802 case ExprKind::Row:
803 return wrap(expr, _resolver->row(expr.fieldPath));
804 case ExprKind::Query:
805 case ExprKind::Pending:
806 case ExprKind::Error:
807 case ExprKind::Result:
808 case ExprKind::ErrorKind:
809 case ExprKind::Can:
810 return wrap(expr, _resolver->control(expr.kind, expr.name));
811 case ExprKind::Pref:
812 return wrap(expr, _resolver->pref(expr.name, expr.fieldPath));
813 case ExprKind::Env:
814 return Type::of(expr.name == "online" ? TypeKind::Bool : TypeKind::String, expr.name != "online");
815 case ExprKind::Session:
816 return Type::of(expr.name == "authenticated" ? TypeKind::Bool : TypeKind::String,
817 expr.name != "authenticated");
818 case ExprKind::Host:
819 return Type::any();
820 case ExprKind::If:
821 return choice(expr);
822 case ExprKind::Object: {
823 for (auto const& [name, field] : expr.fields) {
824 static_cast<void>(name);
825 auto checked = check(*field);
826 if (!checked) {
827 return checked;
828 }
829 }
830 return Type::of(TypeKind::Record);
831 }
832 case ExprKind::List: {
833 for (ExprPtr const& item : expr.args) {
834 auto checked = check(*item);
835 if (!checked) {
836 return checked;
837 }
838 }
839 return Type::of(TypeKind::List);
840 }
841 case ExprKind::Map:
842 case ExprKind::Filter:
843 return comprehension(expr);
844 case ExprKind::Call:
845 default:
846 return call(expr);
847 }
848 }
849
850private:
851 [[nodiscard]] static std::expected<Type, Refusal> wrap(Expr const& expr, std::expected<Type, std::string> type) {
852 if (!type) {
853 return std::unexpected(Refusal{.document = {}, .path = expr.path, .reason = std::move(type.error())});
854 }
855 return std::move(*type);
856 }
857
858 [[nodiscard]] static std::unexpected<Refusal> refuse(Expr const& expr, std::string reason) {
859 return std::unexpected(Refusal{.document = {}, .path = expr.path, .reason = std::move(reason)});
860 }
861
862 [[nodiscard]] std::expected<Type, Refusal> reference(Expr const& expr) {
863 for (auto const& [name, type] : std::views::reverse(_locals)) {
864 if (name == expr.name) {
865 return wrap(expr, typeAlong(type, expr.name, expr.fieldPath));
866 }
867 }
868 return wrap(expr, _resolver->ref(expr.name, expr.fieldPath));
869 }
870
871 [[nodiscard]] static bool admitsKind(Type const& type, KindSet accepted) {
872 return type.kind == TypeKind::Any || (accepted & kindBit(type.kind)) != 0;
873 }
874
875 [[nodiscard]] std::expected<Type, Refusal> condition(Expr const& expr, std::string_view what) {
876 auto type = check(expr);
877 if (type && !admitsKind(*type, kindBit(TypeKind::Bool))) {
878 return refuse(expr, std::string{what} + " is " + type->name() + "; expected bool");
879 }
880 return type;
881 }
882
883 [[nodiscard]] std::expected<Type, Refusal> choice(Expr const& expr) {
884 auto cond = condition(*expr.args.at(0), "the condition of `if`");
885 if (!cond) {
886 return cond;
887 }
888 auto then = check(*expr.args.at(1));
889 if (!then) {
890 return then;
891 }
892 auto otherwise = check(*expr.args.at(2));
893 if (!otherwise) {
894 return otherwise;
895 }
896 if (then->kind == otherwise->kind && then->kind != TypeKind::List) {
897 return Type::of(then->kind, then->optional || otherwise->optional);
898 }
899 return Type::any();
900 }
901
902 [[nodiscard]] std::expected<Type, Refusal> comprehension(Expr const& expr) {
903 auto source = check(*expr.args.at(0));
904 if (!source) {
905 return source;
906 }
907 if (!admitsKind(*source, kindBit(TypeKind::List))) {
908 return refuse(*expr.args.at(0), "`" + std::string{expr.kind == ExprKind::Map ? "map" : "filter"} +
909 "` runs over a list, not " + source->name());
910 }
911 Type element = source->element ? *source->element : Type::any();
912 _locals.emplace_back(expr.name, element);
913 auto body = expr.kind == ExprKind::Map ? check(*expr.args.at(1))
914 : condition(*expr.args.at(1), "the `where` of `filter`");
915 _locals.pop_back();
916 if (!body) {
917 return body;
918 }
919 if (expr.kind == ExprKind::Map) {
920 return Type::listOf(*body);
921 }
922 return Type::listOf(std::move(element));
923 }
924
925 [[nodiscard]] std::expected<Type, Refusal> call(Expr const& expr) {
926 FunctionSpec const& spec = functionSpec(expr.function);
927 std::vector<Type> types;
928 for (std::size_t index = 0; index < expr.args.size(); ++index) {
929 auto type = check(*expr.args.at(index));
930 if (!type) {
931 return type;
932 }
933 KindSet const accepted = spec.accepts.at(std::min<std::size_t>(index, 2));
934 if (!admitsKind(*type, accepted)) {
935 return refuse(*expr.args.at(index), "argument " + std::to_string(index + 1) + " of `" +
936 std::string{spec.name} + "` is " + type->name() +
937 ", which it does not take");
938 }
939 types.push_back(std::move(*type));
940 }
941 if (expr.function == FunctionId::Translate) {
942 ExprPtr const& fallback = expr.args.at(1);
943 if (std::string const* const pattern = fallback->literal.asString();
944 fallback->kind == ExprKind::Literal && pattern != nullptr) {
945 auto parsed = parseMessage(*pattern);
946 if (!parsed) {
947 return refuse(*fallback, "the fallback is outside the ui/1 message subset: " + parsed.error());
948 }
949 }
950 }
951 if (spec.result != TypeKind::Any) {
952 return Type::of(spec.result, true);
953 }
954 return numericResult(expr.function, types);
955 }
956
957 [[nodiscard]] static Type numericResult(FunctionId function, std::vector<Type> const& types) {
958 if (function == FunctionId::Coalesce || function == FunctionId::Sum || function == FunctionId::Mean ||
959 function == FunctionId::Min || function == FunctionId::Max || function == FunctionId::First) {
960 return Type::any();
961 }
962 if (function == FunctionId::Neg || function == FunctionId::Round) {
963 return Type::of(types.front().kind, true);
964 }
965 Type const& lhs = types.at(0);
966 Type const& rhs = types.at(1);
967 if (lhs.isAny() || rhs.isAny()) {
968 return Type::any();
969 }
970 bool const lhsQuantity = lhs.kind == TypeKind::Quantity;
971 bool const rhsQuantity = rhs.kind == TypeKind::Quantity;
972 if (lhsQuantity || rhsQuantity) {
973 if (function == FunctionId::Div && lhsQuantity && rhsQuantity) {
974 return Type::of(TypeKind::Decimal, true);
975 }
976 return Type::of(TypeKind::Quantity, true);
977 }
978 if (lhs.kind == TypeKind::Int && rhs.kind == TypeKind::Int && function != FunctionId::Div) {
979 return Type::of(TypeKind::Int, true);
980 }
981 return function == FunctionId::Div ? Type::any() : Type::of(TypeKind::Decimal, true);
982 }
983
984 NameResolver const* _resolver;
985 std::vector<std::pair<std::string, Type>> _locals;
986 // NOLINTEND(misc-no-recursion)
987};

1089/// @brief Reads a path out of a value: a record's field by name, a list's element by index.
1090/// @param value The value.
1091/// @param path The segments.
1092/// @return The value at the path, or `null` when a segment does not exist.
1093[[nodiscard]] inline Value walk(Value const& value, std::span<std::string const> path) {
1094 Value const* current = &value;
1095 for (std::string const& segment : path) {
1096 if (ValueList const* const items = current->asList()) {
1097 std::size_t index = 0;
1098 bool digits = !segment.empty();
1099 for (char const chr : segment) {
1100 digits = digits && chr >= '0' && chr <= '9';
1101 index = digits ? (index * 10) + static_cast<std::size_t>(chr - '0') : index;
1102 }
1103 if (!digits || index >= items->size()) {
1104 return Value{};
1105 }
1106 current = &items->at(index);
1107 continue;
1108 }
1109 current = current->field(segment);
1110 if (current == nullptr) {
1111 return Value{};
1112 }
1113 }
1114 return *current;

1160/// @brief Converts an exact value between two units of one dimension.
1161/// @param value The value in @p from.
1162/// @param from Its unit.
1163/// @param target The wanted unit.
1164/// @param units The unit table.
1165/// @return The value in @p target, its places widened by the decimal places of the conversion ratio; `std::nullopt`
1166/// when the units do not convert or the result overflows.
1167[[nodiscard]] inline std::optional<math::Rational> convertUnit(math::Rational const& value, std::string_view from,
1168 std::string_view target, UnitTable const& units) {
1169 if (from == target) {
1170 return value;
1171 }
1172 auto factor = [&units](std::string_view unit) {
1173 auto const found = units.find(unit);
1174 if (found == units.end()) {
1175 return std::pair{std::string{unit}, math::Rational{1, math::DecimalPlaces{0}}};
1176 }
1177 return std::pair{found->second.canonical, found->second.toCanonical};
1178 };
1179 auto const [fromBase, fromFactor] = factor(from);
1180 auto const [toBase, toFactor] = factor(target);
1181 if (fromBase != toBase) {
1182 return std::nullopt;
1183 }
1184 auto const ratio =
1185 math::checkedDiv(math::Rational{math::Numerator{fromFactor.numerator},
1186 math::Denominator{fromFactor.denominator}, math::DecimalPlaces{0}},
1187 math::Rational{math::Numerator{toFactor.numerator}, math::Denominator{toFactor.denominator},
1188 math::DecimalPlaces{0}});
1189 if (!ratio) {
1190 return std::nullopt;
1191 }
1192 auto converted = math::checkedMul(value, *ratio);
1193 if (!converted) {
1194 return std::nullopt;
1195 }
1196 // The places widen by the ratio's own decimal places, so 1500 mg shows as 1.500 g rather than rounding to 2 g;
1197 // a ratio that is not a terminating decimal adds none.
1198 auto denominator = static_cast<std::uint64_t>(ratio->denominator);
1199 std::uint32_t twos = 0;
1200 std::uint32_t fives = 0;
1201 for (; denominator % 2U == 0; denominator /= 2U) {
1202 ++twos;
1203 }
1204 for (; denominator % 5U == 0; denominator /= 5U) {
1205 ++fives;
1206 }
1207 std::uint32_t const extra = denominator == 1 ? std::max(twos, fives) : 0U;
1208 converted->decimalPlaces =
1209 math::DecimalPlaces{std::min(value.decimalPlaces.value + extra, math::kMaxDecimalPlaces)};
1210 return *converted;
1211}

1235/// @brief Evaluates expressions against an environment, with the comprehension variables in scope.

1243 /// @brief Evaluates one node.
1244 /// @param expr The node.
1245 /// @return Its value; `null` on a runtime error, which is reported.
1246 [[nodiscard]] Value eval(Expr const& expr) {
1247 switch (expr.kind) {
1248 case ExprKind::Literal:
1249 return expr.literal;
1250 case ExprKind::Ref:
1251 for (auto const& [name, value] : std::views::reverse(_locals)) {
1252 if (name == expr.name) {
1253 return walk(value, expr.fieldPath);
1254 }
1255 }
1256 return _env->ref(expr.name, expr.fieldPath);
1257 case ExprKind::Row:
1258 return _env->row(expr.fieldPath);
1259 case ExprKind::Query:
1260 case ExprKind::Pending:
1261 case ExprKind::Error:
1262 case ExprKind::Result:
1263 case ExprKind::ErrorKind:
1264 case ExprKind::Can:
1265 return _env->control(expr.kind, expr.name);
1266 case ExprKind::Pref:
1267 return _env->pref(expr.name, expr.fieldPath);
1268 case ExprKind::Env:
1269 case ExprKind::Session:
1270 case ExprKind::Host:
1271 return _env->client(expr.kind, expr.name);
1272 case ExprKind::If:
1273 return choice(expr);
1274 case ExprKind::Object: {
1275 ValueRecord record;
1276 for (auto const& [name, field] : expr.fields) {
1277 record.insert_or_assign(name, eval(*field));
1278 }
1279 return record;
1280 }
1281 case ExprKind::List: {
1282 ValueList items;
1283 items.reserve(expr.args.size());
1284 for (ExprPtr const& item : expr.args) {
1285 items.push_back(eval(*item));
1286 }
1287 return items;
1288 }
1289 case ExprKind::Map:
1290 case ExprKind::Filter:
1291 return comprehension(expr);
1292 case ExprKind::Call:
1293 default:
1294 return call(expr);
1295 }

1367 // NOLINTNEXTLINE(readability-function-cognitive-complexity): the four operators over plain numbers and quantities, one rule each.
1368 Value arithmetic(Expr const& expr, Value const& lhsValue, Value const& rhsValue) {
1369 auto const lhs = numeric(lhsValue);
1370 auto const rhs = numeric(rhsValue);
1371 if (!lhs || !rhs) {
1372 return mismatch(expr, "an arithmetic operand is not a number");
1373 }
1374 UnitTable const& units = _env->formatting().units;
1375 FunctionId const function = expr.function;
1376 math::Rational right = rhs->value;
1377 std::string const* unit = nullptr;
1378 if (lhs->unit != nullptr && rhs->unit != nullptr) {
1379 if (function == FunctionId::Mul) {
1380 return mismatch(expr, "a product of two quantities has no declared unit");
1381 }
1382 auto converted = convertUnit(rhs->value, *rhs->unit, *lhs->unit, units);
1383 if (!converted) {
1384 return mismatch(expr, "the units `" + *lhs->unit + "` and `" + *rhs->unit + "` do not convert");
1385 }
1386 right = *converted;
1387 unit = function == FunctionId::Div ? nullptr : lhs->unit;
1388 } else if (lhs->unit != nullptr || rhs->unit != nullptr) {
1389 bool const scales = function == FunctionId::Mul || (function == FunctionId::Div && lhs->unit != nullptr);
1390 if (!scales) {
1391 return mismatch(expr, "a quantity and a plain number do not combine this way");
1392 }
1393 unit = lhs->unit != nullptr ? lhs->unit : rhs->unit;
1394 }
1395 if (function == FunctionId::Div && right.isZero()) {
1396 return mismatch(expr, "division by zero");
1397 }
1398 std::expected<math::Rational, math::RationalError> result = math::Rational{};
1399 if (function == FunctionId::Add) {
1400 result = math::checkedAdd(lhs->value, right);
1401 } else if (function == FunctionId::Sub) {
1402 result = math::checkedSub(lhs->value, right);
1403 } else if (function == FunctionId::Mul) {
1404 result = math::checkedMul(lhs->value, right);
1405 } else {
1406 result = math::checkedDiv(lhs->value, right);
1407 }
1408 if (!result) {
1409 return mismatch(expr, "the result overflows");
1410 }
1411 // A quantity is never an `int`, so a ratio of two quantities comes back a decimal.
1412 return makeNumber(*result, lhs->integer && rhs->integer, unit);
1413 }

1415 /// Orders two values of one comparable kind: numbers exactly (quantities converted), timestamps, strings.
1416 std::optional<std::strong_ordering> order(Value const& lhs, Value const& rhs) {
1417 auto const lhsNumber = numeric(lhs);
1418 auto const rhsNumber = numeric(rhs);
1419 if (lhsNumber && rhsNumber) {
1420 if ((lhsNumber->unit == nullptr) != (rhsNumber->unit == nullptr)) {
1421 return std::nullopt;
1422 }
1423 math::Rational right = rhsNumber->value;
1424 if (lhsNumber->unit != nullptr) {
1425 auto converted = convertUnit(right, *rhsNumber->unit, *lhsNumber->unit, _env->formatting().units);
1426 if (!converted) {
1427 return std::nullopt;
1428 }
1429 right = *converted;
1430 }
1431 return lhsNumber->value <=> right;
1432 }
1433 if (time::DateTime const* const lhsTime = lhs.asTimestamp()) {
1434 if (time::DateTime const* const rhsTime = rhs.asTimestamp()) {
1435 return *lhsTime <=> *rhsTime;
1436 }
1437 return std::nullopt;
1438 }
1439 if (std::string const* const lhsText = lhs.asString()) {
1440 if (std::string const* const rhsText = rhs.asString()) {
1441 return *lhsText <=> *rhsText;
1442 }
1443 }
1444 return std::nullopt;
1445 }

1447 /// Equality over non-null values: numbers by value, lists and records element-wise, other kinds by identity.
1448 bool equal(Value const& lhs, Value const& rhs) {
1449 if (numeric(lhs) && numeric(rhs)) {
1450 auto const ordering = order(lhs, rhs);
1451 return ordering.has_value() && *ordering == std::strong_ordering::equal;
1452 }
1453 if (lhs.kind() != rhs.kind()) {
1454 return false;
1455 }
1456 if (ValueList const* const lhsItems = lhs.asList()) {
1457 return std::ranges::equal(*lhsItems, *rhs.asList(), [this](Value const& left, Value const& right) {
1458 return sameElement(left, right);
1459 });
1460 }
1461 if (ValueRecord const* const lhsFields = lhs.asRecord()) {
1462 // Records are ordered by field name, so equal records have equal sequences of fields.
1463 return std::ranges::equal(*lhsFields, *rhs.asRecord(), [this](auto const& left, auto const& right) {
1464 return left.first == right.first && sameElement(left.second, right.second);
1465 });
1466 }
1467 return lhs == rhs;
1468 }

1471 bool sameElement(Value const& left, Value const& right) {
1472 if (left.isNull() || right.isNull()) {
1473 return left.isNull() && right.isNull();

1475 return equal(left, right);

1516 Value formatTemplate(Expr const& expr, std::vector<Value> const& args) {
1517 std::string const& pattern = *args.front().asString();
1518 std::vector<std::string> shown;
1519 for (std::size_t index = 1; index < args.size(); ++index) {
1520 auto text = display(args.at(index));
1521 if (!text) {
1522 return mismatch(expr, "`format` cannot show argument " + std::to_string(index + 1));
1523 }
1524 shown.push_back(std::move(*text));
1525 }
1526 std::string out;
1527 for (std::size_t pos = 0; pos < pattern.size(); ++pos) {
1528 char const chr = pattern.at(pos);
1529 if ((chr == '{' || chr == '}') && pos + 1 < pattern.size() && pattern.at(pos + 1) == chr) {
1530 out += chr;
1531 ++pos;
1532 continue;
1533 }
1534 if (chr == '}') {
1535 return mismatch(expr, "`format`'s template has an unmatched `}`");
1536 }
1537 if (chr != '{') {
1538 out += chr;
1539 continue;
1540 }
1541 std::size_t const close = pattern.find('}', pos);
1542 std::string_view const digits = close == std::string::npos
1543 ? std::string_view{}
1544 : std::string_view{pattern}.substr(pos + 1, close - pos - 1);
1545 std::size_t index = 0;
1546 bool valid = !digits.empty() && digits.size() < 4;
1547 for (char const digit : digits) {
1548 valid = valid && digit >= '0' && digit <= '9';
1549 index = valid ? (index * 10) + static_cast<std::size_t>(digit - '0') : index;
1550 }
1551 if (!valid || index >= shown.size()) {
1552 return mismatch(expr, "`format`'s template has a placeholder with no argument");
1553 }
1554 out += shown.at(index);
1555 pos = close;
1556 }
1557 return out;
1558 }

1656 Value aggregate(Expr const& expr, ValueList const& items) {
1657 bool const mean = expr.function == FunctionId::Mean;
1658 if (items.empty()) {
1659 return mean ? Value{} : Value{std::int64_t{0}};
1660 }
1661 math::Rational total{0, math::DecimalPlaces{0}};
1662 bool integer = true;
1663 std::string const* unit = nullptr;
1664 std::uint32_t places = 0;
1665 for (std::size_t index = 0; index < items.size(); ++index) {
1666 Value const& item = items.at(index);
1667 if (item.isNull()) {
1668 return Value{};
1669 }
1670 auto const number = numeric(item);
1671 if (!number) {
1672 return mismatch(expr, "an element is not a number");
1673 }
1674 if (index == 0) {
1675 unit = number->unit;
1676 } else if ((unit == nullptr) != (number->unit == nullptr)) {
1677 return mismatch(expr, "quantities and plain numbers do not sum together");
1678 }
1679 math::Rational value = number->value;
1680 if (unit != nullptr) {
1681 auto converted = convertUnit(value, *number->unit, *unit, _env->formatting().units);
1682 if (!converted) {
1683 return mismatch(expr, "the elements' units do not convert");
1684 }
1685 value = *converted;
1686 }
1687 places = std::max(places, value.decimalPlaces.value);
1688 integer = integer && number->integer;
1689 auto next = math::checkedAdd(total, value);
1690 if (!next) {
1691 return mismatch(expr, "the sum overflows");
1692 }
1693 total = *next;
1694 }
1695 if (!mean) {
1696 return makeNumber(total, integer, unit);
1697 }
1698 auto const quotient =
1699 math::checkedDiv(total, math::Rational{static_cast<std::int64_t>(items.size()), math::DecimalPlaces{0}});
1700 // A sum of int64 values divided by their count cannot overflow, and the count is never zero here.
1701 math::Rational const rounded = math::roundToDecimalPlaces(
1702 quotient.value_or(total), math::DecimalPlaces{places}, math::RoundingMode::HalfEven);
1703 return makeNumber(rounded, integer, unit);
1704 }

1707 FunctionId const function = expr.function;
1708 if (function == FunctionId::And || function == FunctionId::Or) {
1709 return kleene(expr, function == FunctionId::Or);
1710 }
1711 if (function == FunctionId::Coalesce) {
1712 for (ExprPtr const& arg : expr.args) {
1713 Value value = eval(*arg);
1714 if (!value.isNull()) {
1715 return value;
1716 }
1717 }
1718 return Value{};
1719 }
1720 if (function == FunctionId::Engaged) {
1721 return !eval(*expr.args.front()).isNull();
1722 }

1724 args.reserve(expr.args.size());
1725 for (ExprPtr const& arg : expr.args) {
1726 args.push_back(eval(*arg));
1727 if (args.back().isNull()) {
1728 return Value{};
1729 }
1730 }
1731 return strict(expr, args);

1734 /// The functions whose every argument is required: any `null` argument has already made the result `null`.
1735 // NOLINTNEXTLINE(readability-function-cognitive-complexity): one case per function of the library.
1736 Value strict(Expr const& expr, std::vector<Value> const& args) {
1737 FormatContext const& context = _env->formatting();
1738 switch (expr.function) {
1739 case FunctionId::Not: {
1740 bool const* const flag = args.front().asBool();
1741 if (flag == nullptr) {
1742 return mismatch(expr, "`not` takes a bool");
1743 }
1744 return !*flag;
1745 }
1746 case FunctionId::Eq:
1747 return equal(args.at(0), args.at(1));
1748 case FunctionId::Ne:
1749 return !equal(args.at(0), args.at(1));
1750 case FunctionId::Lt:
1751 case FunctionId::Le:
1752 case FunctionId::Gt:
1753 case FunctionId::Ge: {
1754 auto const ordering = order(args.at(0), args.at(1));
1755 if (!ordering) {
1756 return mismatch(expr, "the operands are not of one comparable kind");
1757 }
1758 if (expr.function == FunctionId::Lt) {
1759 return std::is_lt(*ordering);
1760 }
1761 if (expr.function == FunctionId::Le) {
1762 return std::is_lteq(*ordering);
1763 }
1764 if (expr.function == FunctionId::Gt) {
1765 return std::is_gt(*ordering);
1766 }
1767 return std::is_gteq(*ordering);
1768 }
1769 case FunctionId::Add:
1770 case FunctionId::Sub:
1771 case FunctionId::Mul:
1772 case FunctionId::Div:
1773 return arithmetic(expr, args.at(0), args.at(1));
1774 case FunctionId::Neg: {
1775 auto const number = numeric(args.front());
1776 if (!number) {
1777 return mismatch(expr, "`neg` takes a number");
1778 }
1779 auto negated = math::checkedSub(math::Rational{0, number->value.decimalPlaces}, number->value);
1780 if (!negated) {
1781 return mismatch(expr, "the result overflows");
1782 }
1783 return makeNumber(*negated, number->integer, number->unit);
1784 }
1785 case FunctionId::Round: {
1786 auto const number = numeric(args.at(0));
1787 std::int64_t const* const places = args.at(1).asInt();
1788 if (!number || places == nullptr || *places < 0 ||
1789 std::cmp_greater(*places, math::kMaxDecimalPlaces)) {
1790 return mismatch(expr, "`round` takes a number and places from 0 to 18");
1791 }
1792 if (number->integer) {
1793 return args.at(0);
1794 }
1795 auto const placesValue = static_cast<std::uint32_t>(*places);
1796 std::int64_t scale = 1;
1797 for (std::uint32_t digit = 0; digit < placesValue; ++digit) {
1798 scale *= 10;
1799 }
1800 if (!math::checkedMul(number->value, math::Rational{scale, math::DecimalPlaces{0}})) {
1801 return mismatch(expr, "the result overflows");
1802 }
1803 return makeNumber(math::roundToDecimalPlaces(number->value, math::DecimalPlaces{placesValue},
1804 math::RoundingMode::HalfEven),
1805 false, number->unit);
1806 }
1807 case FunctionId::Concat: {
1808 std::string out;
1809 for (Value const& arg : args) {
1810 std::string const* const text = arg.asString();
1811 if (text == nullptr) {
1812 return mismatch(expr, "`concat` takes strings");
1813 }
1814 out += *text;
1815 }
1816 return out;
1817 }
1818 case FunctionId::Format:
1819 if (args.front().asString() == nullptr) {
1820 return mismatch(expr, "`format`'s template is a string");
1821 }
1822 return formatTemplate(expr, args);
1823 case FunctionId::Trim: {
1824 std::string const* const text = args.front().asString();
1825 if (text == nullptr) {
1826 return mismatch(expr, "`trim` takes a string");
1827 }
1828 constexpr std::string_view kSpace = " \t\n\r\f\v";
1829 std::size_t const first = text->find_first_not_of(kSpace);
1830 if (first == std::string::npos) {
1831 return std::string{};
1832 }
1833 return text->substr(first, text->find_last_not_of(kSpace) - first + 1);
1834 }
1835 case FunctionId::Length: {
1836 std::string const* const text = args.front().asString();
1837 if (text == nullptr) {
1838 return mismatch(expr, "`length` takes a string");
1839 }
1840 std::int64_t points = 0;
1841 for (char const chr : *text) {
1842 points += (static_cast<unsigned char>(chr) & 0xC0U) == 0x80U ? 0 : 1;
1843 }
1844 return points;
1845 }
1846 case FunctionId::IsEmpty: {
1847 if (std::string const* const text = args.front().asString()) {
1848 return text->empty();
1849 }
1850 if (ValueList const* const items = args.front().asList()) {
1851 return items->empty();
1852 }
1853 return mismatch(expr, "`isEmpty` takes a string or a list");
1854 }

1856 std::string const* const text = args.at(0).asString();
1857 std::string const* const part = args.at(1).asString();
1858 if (text == nullptr || part == nullptr) {
1859 return mismatch(expr, "`contains` takes two strings");
1860 }
1861 return text->contains(*part);
1862 }
1863 case FunctionId::Count:
1864 case FunctionId::Sum:
1865 case FunctionId::Mean:
1866 case FunctionId::Min:
1867 case FunctionId::Max:
1868 case FunctionId::Any:
1869 case FunctionId::All:
1870 case FunctionId::First: {
1871 ValueList const* const items = args.front().asList();
1872 if (items == nullptr) {
1873 return mismatch(expr, "the argument is not a list");
1874 }
1875 return listFunction(expr, *items);
1876 }
1877 case FunctionId::Includes:
1878 case FunctionId::IndexOf: {
1879 ValueList const* const items = args.at(0).asList();
1880 if (items == nullptr) {
1881 return mismatch(expr, "the first argument is not a list");
1882 }

1884 if (!items->at(index).isNull() && equal(items->at(index), args.at(1))) {
1885 return expr.function == FunctionId::Includes ? Value{true}
1886 : Value{static_cast<std::int64_t>(index)};
1887 }
1888 }
1889 return expr.function == FunctionId::Includes ? Value{false} : Value{};
1890 }
1891 case FunctionId::Number: {
1892 auto text = formatNumber(args.front(), context.numeric);
1893 if (!text) {
1894 return mismatch(expr, "`number` takes a number");
1895 }
1896 return std::move(*text);
1897 }
1898 case FunctionId::DecimalText: {
1899 auto const number = numeric(args.at(0));
1900 std::int64_t const* const places = args.at(1).asInt();
1901 if (!number || places == nullptr || *places < 0 ||
1902 std::cmp_greater(*places, math::kMaxDecimalPlaces)) {
1903 return mismatch(expr, "`decimal` takes a number and places from 0 to 18");
1904 }
1905 return render::formatCanonicalNumber(decimalText(number->value, static_cast<std::uint32_t>(*places)),
1906 context.numeric);
1907 }
1908 case FunctionId::QuantityText: {
1909 Quantity const* const quantity = args.at(0).asQuantity();
1910 if (quantity == nullptr) {
1911 return mismatch(expr, "`quantity` takes a quantity");
1912 }
1913 Quantity shown = *quantity;
1914 if (args.size() > 1) {
1915 std::string const* const unit = args.at(1).asString();
1916 auto converted = unit == nullptr
1917 ? std::nullopt
1918 : convertUnit(quantity->value, quantity->unit, *unit, context.units);
1919 if (!converted) {
1920 return mismatch(expr, "the quantity does not convert to that unit");
1921 }
1922 shown = Quantity{.value = *converted, .unit = *unit};
1923 }
1924 return formatNumber(shown, context.numeric).value_or(std::string{}) + " " +
1925 unitDisplay(shown.unit, context.units);
1926 }
1927 case FunctionId::Money: {
1928 auto const number = numeric(args.at(0));
1929 std::string const* const currency = args.at(1).asString();
1930 if (!number || number->unit != nullptr || currency == nullptr) {
1931 return mismatch(expr, "`money` takes an amount and a currency code");
1932 }
1933 return render::formatCanonicalNumber(decimalText(number->value, currencyDigits(*currency)),
1934 context.numeric) +
1935 " " + *currency;
1936 }
1937 case FunctionId::Date:
1938 case FunctionId::DateTime: {
1939 std::optional<time::DateTime> instant;
1940 if (time::DateTime const* const held = args.at(0).asTimestamp()) {
1941 instant = *held;
1942 } else if (std::string const* const text = args.at(0).asString()) {
1943 instant = time::DateTime::fromIso8601(*text);
1944 }
1945 std::string const* const style = args.size() > 1 ? args.at(1).asString() : nullptr;
1946 std::optional<std::string> text;
1947 if (instant && (args.size() == 1 || style != nullptr)) {
1948 text = dateText(*instant, style == nullptr ? std::string_view{"medium"} : *style,
1949 expr.function == FunctionId::DateTime);
1950 }
1951 if (!text) {
1952 return mismatch(expr, "the date functions take a timestamp and a style");
1953 }
1954 return std::move(*text);
1955 }
1956 case FunctionId::ParseDecimal: {
1957 std::string const* const text = args.at(0).asString();
1958 std::optional<std::uint32_t> declared;
1959 if (args.size() > 1) {
1960 std::int64_t const* const places = args.at(1).asInt();
1961 if (places == nullptr || !std::in_range<std::uint32_t>(*places) ||
1962 std::cmp_greater(*places, math::kMaxDecimalPlaces)) {
1963 return mismatch(expr, "`parseDecimal` takes text and places from 0 to 18");
1964 }
1965 declared = static_cast<std::uint32_t>(*places);
1966 }
1967 if (text == nullptr) {
1968 return mismatch(expr, "`parseDecimal` takes text and places from 0 to 18");
1969 }
1970 // Text that does not spell a number is what an entry holds while it is typed: `null`, not an error.
1971 if (auto parsed = render::parseDecimal(*text, context.numeric, declared)) {
1972 return Decimal{*parsed};
1973 }
1974 return Value{};
1975 }
1976 case FunctionId::Translate:
1977 if (args.at(0).asString() == nullptr || args.at(1).asString() == nullptr) {
1978 return mismatch(expr, "`t` takes a key and a fallback string");
1979 }
1980 return translate(expr, args);
1981 case FunctionId::And:
1982 case FunctionId::Or:
1983 case FunctionId::Coalesce:
1984 case FunctionId::Engaged:
1985 default:
1986 return Value{};
1987 }
1988 }

1990 Environment const* _env;
1991 std::vector<std::pair<std::string, Value>> _locals;
1992 // NOLINTEND(misc-no-recursion)
1993};
