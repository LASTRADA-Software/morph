49/// @brief Whether a string is a document name: `[A-Za-z_][A-Za-z0-9_]*`.
50/// @param text The candidate.
51/// @return Whether @p text is a well-formed name.
52[[nodiscard]] constexpr bool isName(std::string_view text) noexcept {
53 if (text.empty()) {
54 return false;
55 }
56 bool first = true;
57 for (char const chr : text) {
58 bool const letter = (chr >= 'a' && chr <= 'z') || (chr >= 'A' && chr <= 'Z') || chr == '_';
59 bool const digit = chr >= '0' && chr <= '9';
60 if (!letter && (first || !digit)) {
61 return false;
62 }
63 first = false;
64 }
65 return true;
66}

152/// @brief One node of a parsed expression. Immutable once parsed; shared between the document and its readers.
153struct Expr {
154 /// @brief The form.
155 ExprKind kind = ExprKind::Literal;
156 /// @brief The JSON Pointer of the node inside its document, for refusals and diagnostics.
157 std::string path;
158 /// @brief The constant of a `Literal`.
159 Value literal;
160 /// @brief The name a form reads: a ref's head, a query or mutation, a pref's point, a comprehension's variable,
161 /// an `env`, `session` or `host` name.
162 std::string name;
163 /// @brief The path after the head of a `Ref` or `Pref`, or the path of a `Row` or `Get`.
164 std::vector<std::string> fieldPath;
165 /// @brief The function of a `Call`.
166 FunctionId function = FunctionId::And;
167 /// @brief Operands: a call's arguments; `if`'s condition, then and else; a list's items; a comprehension's
168 /// source and body (and a `sort`'s `order`, when given); the value a `get` reads from.
169 std::vector<ExprPtr> args;
170 /// @brief The fields of an `Object`, in the order written.
171 std::vector<std::pair<std::string, ExprPtr>> fields;
172};

174/// @brief Whether a form names each element of its list with `as`, for its second operand alone.

178 return kind == ExprKind::Map || kind == ExprKind::Filter || kind == ExprKind::Sort || kind == ExprKind::Runs;

181/// @brief One run of adjacent elements of a list that share a value, as `runs` and a table's group heading read
182/// it: `{"key": <the shared value>, "count": n, "first": <element>, "last": <element>, "items": [...]}`.
183/// @param key The shared value.
184/// @param items The run's elements, in order; not empty.
185/// @return The record.
186[[nodiscard]] inline Value runRecord(Value key, ValueList items) {
187 ValueRecord record;
188 record.insert_or_assign("key", std::move(key));
189 record.insert_or_assign("count", static_cast<std::int64_t>(items.size()));
190 record.insert_or_assign("first", items.front());
191 record.insert_or_assign("last", items.back());
192 record.insert_or_assign("items", Value{std::move(items)});
193 return Value{std::move(record)};
194}

368/// @brief Splits `head.a.b` into its segments.
369/// @param text The dotted path.
370/// @return The segments, or the reason the path is malformed.
371[[nodiscard]] inline std::expected<std::vector<std::string>, std::string> splitPath(std::string_view text) {
372 std::vector<std::string> segments;
373 std::size_t start = 0;
374 for (;;) {
375 std::size_t const dot = text.find('.', start);
376 std::string_view const segment =
377 text.substr(start, dot == std::string_view::npos ? std::string_view::npos : dot - start);
378 if (segment.empty()) {
379 return std::unexpected("`" + std::string{text} + "` has an empty segment");
380 }
381 segments.emplace_back(segment);
382 if (dot == std::string_view::npos) {
383 return segments;
384 }
385 start = dot + 1;
386 }
387}

392public:
393 /// @brief The keys a form may carry beside its own.
394 struct FormKeys {
395 /// @brief The form's own key.
396 std::string_view key;
397 /// @brief The form.
398 ExprKind kind;
399 /// @brief The keys allowed beside it.
400 std::array<std::string_view, 3> companions;

403 /// @brief Parses @p json, whose JSON Pointer is @p path.
404 /// @param json The expression.
405 /// @param path Its JSON Pointer.

411private:

418 {"pending", ExprKind::Pending, {}},

459 [[nodiscard]] std::expected<ExprPtr, Refusal> node(Json const& json, std::string const& path) {
460 if (++_count > kMaxExpressionNodes) {
461 return refuse(path, "the expression has more than " + std::to_string(kMaxExpressionNodes) +
462 " nodes, the bound of expr/1");
463 }
464 if (std::holds_alternative<Json::array_t>(json.data)) {
465 return refuse(path, R"(a list is written {"list": [...]}, a constant list {"lit": [...]})");
466 }
467 auto const* const object = std::get_if<Json::object_t>(&json.data);
468 if (object == nullptr) {
469 auto value = valueFromJson(json);
470 if (!value) {
471 return refuse(path, value.error());
472 }
473 return literal(std::move(*value), path);
474 }
475 auto made = std::make_shared<Expr>();
476 made->path = path;
477 if (auto refused = form(*made, *object, path)) {
478 return std::unexpected(std::move(*refused));
479 }
480 return made;
481 }

483 [[nodiscard]] std::optional<Refusal> form(Expr& made, Json::object_t const& object, std::string const& path) {
484 FormKeys const* chosen = nullptr;
485 for (auto const& [key, value] : object) {
486 static_cast<void>(value);
487 for (FormKeys const& candidate : kForms) {
488 if (candidate.key != key) {
489 continue;
490 }
491 if (chosen != nullptr) {
492 return refuse(path, "an expression has one form; found `" + std::string{chosen->key} + "` and `" +
493 key + "`")
494 .error();
495 }
496 chosen = &candidate;
497 }
498 }
499 if (chosen == nullptr) {
500 if (object.empty()) {
501 return refuse(path, "an empty object is not an expression; a constant record is {\"lit\": {}}")
502 .error();
503 }
504 return refuse(path, "`" + object.begin()->first + "` is not an expression form of expr/1").error();
505 }
506 for (auto const& [key, value] : object) {
507 static_cast<void>(value);
508 if (key != chosen->key && !std::ranges::contains(chosen->companions, key)) {
509 return refuse(pointerAppend(path, key),
510 "`" + key + "` does not belong to the `" + std::string{chosen->key} + "` form")
511 .error();
512 }
513 }
514 Json const& primary = *member(object, chosen->key);
515 std::string const primaryPath = pointerAppend(path, chosen->key);
516 made.kind = chosen->kind;
517 switch (chosen->kind) {
518 case ExprKind::Literal: {
519 auto value = valueFromJson(primary);
520 if (!value) {
521 return refuse(primaryPath, value.error()).error();
522 }
523 made.literal = std::move(*value);
524 return std::nullopt;
525 }
526 case ExprKind::Ref:
527 case ExprKind::Pref:
528 return dotted(made, primary, primaryPath);
529 case ExprKind::Row:
530 return rowPath(made, primary, primaryPath);
531 case ExprKind::Query:
532 case ExprKind::Pending:
533 case ExprKind::Error:
534 case ExprKind::Result:
535 case ExprKind::ErrorKind:
536 case ExprKind::Can:
537 return controlName(made, primary, primaryPath, chosen->key);
538 case ExprKind::Env:
539 case ExprKind::Session:
540 case ExprKind::Host:
541 return clientName(made, primary, primaryPath, chosen->key);
542 case ExprKind::Call:
543 if (chosen->key == "t") {
544 return translate(made, object, path);
545 }
546 return call(made, object, path);
547 case ExprKind::If:
548 return operands(made.args, primary, primaryPath, 3, "`if` takes [condition, then, else]");
549 case ExprKind::List:
550 return operands(made.args, primary, primaryPath, 0, "`list` takes an array of expressions");
551 case ExprKind::Object:
552 return fields(made.fields, primary, primaryPath);
553 case ExprKind::Get:
554 return get(made, object, path);
555 case ExprKind::Map:
556 case ExprKind::Filter:
557 case ExprKind::Sort:
558 case ExprKind::Runs:
559 default:
560 return comprehension(made, object, path);
561 }
562 }

564 [[nodiscard]] static std::optional<Refusal> dotted(Expr& made, Json const& primary, std::string const& path) {
565 std::string const* const dottedPath = text(primary);
566 if (dottedPath == nullptr) {
567 return refuse(path, "a name and path is written as a string `name.path`").error();
568 }
569 auto segments = splitPath(*dottedPath);
570 if (!segments) {
571 return refuse(path, segments.error()).error();
572 }
573 if (!isName(segments->front())) {
574 return refuse(path, "`" + segments->front() + "` is not a name").error();
575 }
576 made.name = std::move(segments->front());
577 made.fieldPath.assign(std::make_move_iterator(segments->begin() + 1),
578 std::make_move_iterator(segments->end()));
579 return std::nullopt;
580 }

582 [[nodiscard]] static std::optional<Refusal> rowPath(Expr& made, Json const& primary, std::string const& path) {
583 std::string const* const dottedPath = text(primary);
584 if (dottedPath == nullptr) {

689 }

924/// @brief Walks an expression once, resolving names and checking the statically known types.
925// NOLINTBEGIN(misc-no-recursion): an expression is a tree and checking descends it; the node bound limits the depth.
926class ExprChecker {
927public:
928 /// @brief Checks against @p resolver.
929 /// @param resolver What the names resolve to.
930 explicit ExprChecker(NameResolver const& resolver) : _resolver{&resolver} {}
931
932 /// @brief Checks one node and its operands.
933 /// @param expr The node.
934 /// @return Its static type, or the refusal.
935 [[nodiscard]] std::expected<Type, Refusal> check(Expr const& expr) {
936 switch (expr.kind) {
937 case ExprKind::Literal:
938 return typeOfValue(expr.literal);
939 case ExprKind::Ref:
940 return reference(expr);
941 case ExprKind::Row:
942 return wrap(expr, _resolver->row(expr.fieldPath));
943 case ExprKind::Query:
944 case ExprKind::Pending:
945 case ExprKind::Error:
946 case ExprKind::Result:
947 case ExprKind::ErrorKind:
948 case ExprKind::Can:
949 return wrap(expr, _resolver->control(expr.kind, expr.name));
950 case ExprKind::Pref:
951 return wrap(expr, _resolver->pref(expr.name, expr.fieldPath));
952 case ExprKind::Env:
953 return Type::of(expr.name == "online" ? TypeKind::Bool : TypeKind::String, expr.name != "online");
954 case ExprKind::Session:
955 return Type::of(expr.name == "authenticated" ? TypeKind::Bool : TypeKind::String,
956 expr.name != "authenticated");
957 case ExprKind::Host:
958 return Type::any();
959 case ExprKind::If:
960 return choice(expr);
961 case ExprKind::Object: {
962 for (auto const& [name, field] : expr.fields) {
963 static_cast<void>(name);
964 auto checked = check(*field);
965 if (!checked) {
966 return checked;
967 }
968 }
969 return Type::of(TypeKind::Record);
970 }
971 case ExprKind::List: {
972 for (ExprPtr const& item : expr.args) {
973 auto checked = check(*item);
974 if (!checked) {
975 return checked;
976 }
977 }
978 return Type::of(TypeKind::List);
979 }
980 case ExprKind::Map:
981 case ExprKind::Filter:
982 case ExprKind::Sort:
983 case ExprKind::Runs:
984 return comprehension(expr);
985 case ExprKind::Get: {
986 auto source = check(*expr.args.at(0));
987 if (!source) {
988 return source;
989 }
990 if (!admitsKind(*source, kindBit(TypeKind::Record) | kindBit(TypeKind::List))) {
991 return refuse(expr, "`get` reads a path out of a record or a list, not " + source->name());
992 }
993 return Type::any();
994 }
995 case ExprKind::Call:
996 default:
997 return call(expr);
998 }
999 }

1002 [[nodiscard]] static std::expected<Type, Refusal> wrap(Expr const& expr, std::expected<Type, std::string> type) {
1003 if (!type) {
1004 return std::unexpected(Refusal{.document = {}, .path = expr.path, .reason = std::move(type.error())});
1005 }
1006 return std::move(*type);
1007 }

1010 return std::unexpected(Refusal{.document = {}, .path = expr.path, .reason = std::move(reason)});
1011 }

1023 return type.kind == TypeKind::Any || (accepted & kindBit(type.kind)) != 0;

1026 [[nodiscard]] std::expected<Type, Refusal> condition(Expr const& expr, std::string_view what) {
1027 auto type = check(expr);
1028 if (type && !admitsKind(*type, kindBit(TypeKind::Bool))) {

1034 [[nodiscard]] std::expected<Type, Refusal> choice(Expr const& expr) {
1035 auto cond = condition(*expr.args.at(0), "the condition of `if`");
1036 if (!cond) {
1037 return cond;
1038 }
1039 auto then = check(*expr.args.at(1));
1040 if (!then) {
1041 return then;
1042 }
1043 auto otherwise = check(*expr.args.at(2));
1044 if (!otherwise) {
1045 return otherwise;
1046 }
1047 if (then->kind == otherwise->kind && then->kind != TypeKind::List) {
1048 return Type::of(then->kind, then->optional || otherwise->optional);
1049 }
1050 return Type::any();

1053 [[nodiscard]] std::expected<Type, Refusal> comprehension(Expr const& expr) {
1054 auto source = check(*expr.args.at(0));
1055 if (!source) {
1056 return source;
1057 }
1058 std::string_view form = "filter";
1059 if (expr.kind == ExprKind::Map) {
1060 form = "map";
1061 } else if (expr.kind == ExprKind::Sort) {
1062 form = "sort";
1063 } else if (expr.kind == ExprKind::Runs) {
1064 form = "runs";
1065 }
1066 if (!admitsKind(*source, kindBit(TypeKind::List))) {
1067 return refuse(*expr.args.at(0), "`" + std::string{form} + "` runs over a list, not " + source->name());
1068 }
1069 Type element = source->element ? *source->element : Type::any();
1070 _locals.emplace_back(expr.name, element);
1071 auto body = expr.kind == ExprKind::Filter ? condition(*expr.args.at(1), "the `where` of `filter`")
1072 : check(*expr.args.at(1));
1073 _locals.pop_back();
1074 if (!body) {
1075 return body;
1076 }
1077 if (expr.kind == ExprKind::Map) {
1078 return Type::listOf(*body);
1079 }
1080 if (expr.kind == ExprKind::Runs) {
1081 return Type::listOf(Type::of(TypeKind::Record));
1082 }
1083 if (expr.kind == ExprKind::Sort && expr.args.size() > 2) {
1084 auto order = check(*expr.args.at(2));
1085 if (!order) {
1086 return order;
1087 }
1088 if (!admitsKind(*order, kindBit(TypeKind::String))) {
1089 return refuse(*expr.args.at(2), "the `order` of `sort` is " + order->name() + "; expected string");
1090 }
1091 }
1092 return Type::listOf(std::move(element));
1093 }

1095 [[nodiscard]] std::expected<Type, Refusal> call(Expr const& expr) {
1096 FunctionSpec const& spec = functionSpec(expr.function);
1097 std::vector<Type> types;
1098 for (std::size_t index = 0; index < expr.args.size(); ++index) {
1099 auto type = check(*expr.args.at(index));
1100 if (!type) {
1101 return type;
1102 }
1103 KindSet const accepted = spec.accepts.at(std::min<std::size_t>(index, 2));
1104 if (!admitsKind(*type, accepted)) {
1105 return refuse(*expr.args.at(index), "argument " + std::to_string(index + 1) + " of `" +
1106 std::string{spec.name} + "` is " + type->name() +
1107 ", which it does not take");
1108 }
1109 types.push_back(std::move(*type));
1110 }
1111 if (auto refused = literalRefusal(expr)) {
1112 return std::unexpected(std::move(*refused));
1113 }
1114 if (spec.result != TypeKind::Any) {
1115 return Type::of(spec.result, true);
1116 }
1117 if (expr.function == FunctionId::ConcatLists) {
1118 return concatenated(types);
1119 }
1120 if (expr.function == FunctionId::Slice) {
1121 Type const& source = types.front();
1122 return Type::listOf(source.element ? *source.element : Type::any(), true);
1123 }
1124 return numericResult(expr.function, types);
1125 }

1127 /// The checks of an argument written as a literal: a rounding mode and a `dateAdd` unit are checked at load,
1128 /// because a wrong one is a wrong value, not a wrong look; a `t` fallback must be in the message subset.
1129 [[nodiscard]] static std::optional<Refusal> literalRefusal(Expr const& expr) {
1130 auto const literalText = [&expr](std::size_t index) -> std::string const* {
1131 if (index >= expr.args.size() || expr.args.at(index)->kind != ExprKind::Literal) {
1132 return nullptr;
1133 }
1134 return expr.args.at(index)->literal.asString();
1135 };
1136 if (auto const position = modeArgument(expr.function)) {
1137 std::string const* const mode = literalText(*position);
1138 if (mode != nullptr && !roundingMode(Value{*mode})) {
1139 return refuse(*expr.args.at(*position), "the rounding mode is `halfEven` or `halfUp`").error();
1140 }
1141 }
1142 if (std::string const* const unit = literalText(2);
1143 expr.function == FunctionId::DateAdd && unit != nullptr && !oneOf(kDateUnits, *unit)) {
1144 return refuse(*expr.args.at(2), "the unit is `days`, `weeks`, `months`, `years`, `hours` or `minutes`")
1145 .error();
1146 }
1147 if (std::string const* const pattern = literalText(1);
1148 expr.function == FunctionId::Translate && pattern != nullptr) {
1149 auto parsed = parseMessage(*pattern);
1150 if (!parsed) {
1151 return refuse(*expr.args.at(1), "the fallback is outside the ui/1 message subset: " + parsed.error())
1152 .error();
1153 }
1154 }
1155 return std::nullopt;
1156 }

1158 /// `concatLists`'s type: `list<T>` when every element type that is known is `T`, else `list<any>`.
1159 [[nodiscard]] static Type concatenated(std::vector<Type> const& types) {
1160 std::optional<Type> common;
1161 for (Type const& type : types) {
1162 if (type.isAny() || !type.element || type.element->isAny()) {
1163 continue;
1164 }
1165 if (!common) {
1166 common = *type.element;
1167 } else if (!(*common == *type.element)) {
1168 return Type::listOf(Type::any(), true);
1169 }
1170 }
1171 return Type::listOf(common.value_or(Type::any()), true);
1172 }

1379 return {.form = DateForm::Unreadable, .date = {}, .instant = {}};
1380 }

1382 return {};

1386 unsigned parsed = 0;

1390 if (error != std::errc{} || stop != end) {
1391 return std::nullopt;
1392 }
1393 return parsed;
1394 };

1396 auto const month = field(5, 2);
1397 auto const day = field(8, 2);
1398 if (year && month && day) {
1399 std::chrono::year_month_day const date{std::chrono::year{static_cast<int>(*year)},
1400 std::chrono::month{*month}, std::chrono::day{*day}};

1403 }
1404 return {.form = DateForm::Unreadable, .date = {}, .instant = {}};
1405 }
1406 }
1407 if (auto const parsed = time::DateTime::fromIso8601(*text)) {
1408 return {.form = DateForm::InstantText, .date = {}, .instant = *parsed};
1409 }
1410 return {.form = DateForm::Unreadable, .date = {}, .instant = {}};

1456/// @brief Converts an exact value between two units of one dimension.
1457/// @param value The value in @p from.
1458/// @param from Its unit.
1459/// @param target The wanted unit.
1460/// @param units The unit table.
1461/// @return The value in @p target, its places widened by the decimal places of the conversion ratio; `std::nullopt`
1462/// when the units do not convert or the result overflows.
1463[[nodiscard]] inline std::optional<math::Rational> convertUnit(math::Rational const& value, std::string_view from,
1464 std::string_view target, UnitTable const& units) {
1465 if (from == target) {
1466 return value;
1467 }
1468 auto factor = [&units](std::string_view unit) {
1469 auto const found = units.find(unit);
1470 if (found == units.end()) {
1471 return std::pair{std::string{unit}, math::Rational{1, math::DecimalPlaces{0}}};
1472 }
1473 return std::pair{found->second.canonical, found->second.toCanonical};
1474 };
1475 auto const [fromBase, fromFactor] = factor(from);
1476 auto const [toBase, toFactor] = factor(target);
1477 if (fromBase != toBase) {
1478 return std::nullopt;
1479 }
1480 auto const ratio =
1481 math::checkedDiv(math::Rational{math::Numerator{fromFactor.numerator},
1482 math::Denominator{fromFactor.denominator}, math::DecimalPlaces{0}},
1483 math::Rational{math::Numerator{toFactor.numerator}, math::Denominator{toFactor.denominator},
1484 math::DecimalPlaces{0}});
1485 if (!ratio) {
1486 return std::nullopt;
1487 }
1488 auto converted = math::checkedMul(value, *ratio);
1489 if (!converted) {
1490 return std::nullopt;
1491 }
1492 // The places widen by the ratio's own decimal places, so 1500 mg shows as 1.500 g rather than rounding to 2 g;
1493 // a ratio that is not a terminating decimal adds none.
1494 auto denominator = static_cast<std::uint64_t>(ratio->denominator);
1495 std::uint32_t twos = 0;
1496 std::uint32_t fives = 0;
1497 for (; denominator % 2U == 0; denominator /= 2U) {
1498 ++twos;
1499 }
1500 for (; denominator % 5U == 0; denominator /= 5U) {
1501 ++fives;
1502 }
1503 std::uint32_t const extra = denominator == 1 ? std::max(twos, fives) : 0U;
1504 converted->decimalPlaces =
1505 math::DecimalPlaces{std::min(value.decimalPlaces.value + extra, math::kMaxDecimalPlaces)};
1506 return *converted;
1507}

1582/// @brief The least common multiple of exact values' denominators, over which each value is an integer.
1583/// @param values The values, canonical.

1588 auto const denominator = static_cast<std::uint64_t>(value.denominator);
1589 math::detail::U128 const product = math::detail::mulU64(common, denominator / std::gcd(common, denominator));
1590 if (product.hi != 0 || product.lo > static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max())) {
1591 return std::nullopt;

1598/// @brief A sample read for `stdev` and `rsd`: its moments, `null` without a report, or the reason to report.
1599using SampleReading = std::expected<std::optional<Moments>, std::string>;

1601/// @brief The exact moments of a sample of numbers, or of quantities converted to the first element's unit.
1602///
1603/// A `null` element makes the sample `null`; an element that is not a number, a quantity among plain numbers or units
1604/// that do not convert is a mismatch; a common denominator, an element or a sum that leaves its width overflows.
1605/// @param items The sample.
1606/// @param units The unit table.
1607/// @return The moments; empty for a `null` element; the reason for a mismatch or an overflow.
1608[[nodiscard]] inline SampleReading momentsOf(ValueList const& items, UnitTable const& units) {
1609 using math::detail::U128;
1610 std::vector<math::Rational> values;
1611 values.reserve(items.size());
1612 Moments moments;
1613 for (std::size_t index = 0; index < items.size(); ++index) {
1614 auto const number = numeric(items.at(index));
1615 if (items.at(index).isNull()) {
1616 return std::optional<Moments>{};
1617 }
1618 if (!number) {
1619 return std::unexpected(std::string{"an element is not a number"});
1620 }
1621 if (index == 0) {
1622 moments.unit = number->unit;
1623 } else if ((moments.unit == nullptr) != (number->unit == nullptr)) {
1624 return std::unexpected(std::string{"quantities and plain numbers do not combine"});
1625 }
1626 auto converted = moments.unit == nullptr ? std::optional{number->value}
1627 : convertUnit(number->value, *number->unit, *moments.unit, units);
1628 if (!converted) {
1629 return std::unexpected(std::string{"the elements' units do not convert"});
1630 }
1631 values.push_back(*converted);
1632 }
1633 auto const scale = commonDenominator(values);
1634 if (!scale) {
1635 return std::unexpected(std::string{"the statistics overflow"});
1636 }
1637 moments.scale = *scale;
1638 U128 positive{};
1639 U128 negative{};
1640 for (math::Rational const& value : values) {
1641 auto const denominator = static_cast<std::uint64_t>(value.denominator);
1642 U128 const scaled = math::detail::mulU64(math::detail::absU64(value.numerator), moments.scale / denominator);
1643 auto squares = scaled.hi == 0
1644 ? math::detail::checkedAddU128(moments.squares, math::detail::mulU64(scaled.lo, scaled.lo))
1645 : std::nullopt;
1646 auto sum = math::detail::checkedAddU128(value.numerator < 0 ? negative : positive, scaled);
1647 if (!squares || !sum) {
1648 return std::unexpected(std::string{"the statistics overflow"});
1649 }
1650 moments.squares = *squares;
1651 (value.numerator < 0 ? negative : positive) = *sum;
1652 }
1653 moments.count = values.size();
1654 moments.sumMagnitude = positive >= negative ? math::detail::wrappingSubU128(positive, negative)
1655 : math::detail::wrappingSubU128(negative, positive);
1656 return std::optional{moments};
1657}

1659/// @brief The standard deviation or the RSD of a sample, correctly rounded at @p places.
1660///
1661/// With `n` elements `X = x * D`, `V = n * sum X^2 - (sum X)^2`: the sample standard deviation is the root of
1662/// `V * 10^(2 * places) / (n * (n - 1) * D^2)`, and the RSD in percent the root of
1663/// `10^(4 + 2 * places) * n * V / ((n - 1) * (sum X)^2)`, where `D` cancels. Nothing is rounded before the root.
1664/// @param moments The sample's moments, two or more elements.
1665/// @param relative Whether to compute the RSD.
1666/// @param places The places of the result.
1667/// @param mode The tie-breaking rule.
1668/// @return The result, or the reason it cannot be computed.
1669[[nodiscard]] inline std::expected<math::Rational, std::string> spreadOf(Moments const& moments, bool relative,
1670 std::uint32_t places,
1671 math::RoundingMode mode) {
1672 using math::detail::checkedMulU128;
1673 using math::detail::U128;
1674 std::string const overflow = "the statistics overflow";
1675 U128 const count{.hi = 0, .lo = moments.count};
1676 U128 const degrees{.hi = 0, .lo = moments.count - 1};
1677 auto const scaledSquares = checkedMulU128(count, moments.squares);
1678 auto const squaredSum = checkedMulU128(moments.sumMagnitude, moments.sumMagnitude);
1679 // n * sum X^2 >= (sum X)^2 by Cauchy-Schwarz, so the difference is never negative.
1680 auto const spread =
1681 scaledSquares && squaredSum ? math::detail::checkedSubU128(*scaledSquares, *squaredSum) : std::nullopt;
1682 if (!spread) {
1683 return std::unexpected(overflow);
1684 }
1685 std::optional<U128> numerator;
1686 std::optional<U128> denominator;
1687 if (relative) {
1688 if (moments.sumMagnitude == U128{}) {
1689 return std::unexpected(std::string{"the RSD of values whose mean is zero"});
1690 }
1691 auto const power = math::detail::powerOfTenU128(4 + (2 * places));
1692 auto const scaled = power ? checkedMulU128(*power, count) : std::nullopt;
1693 numerator = scaled ? checkedMulU128(*scaled, *spread) : std::nullopt;
1694 denominator = checkedMulU128(degrees, *squaredSum);
1695 } else {
1696 auto const power = math::detail::powerOfTenU128(2 * places);
1697 numerator = power ? checkedMulU128(*spread, *power) : std::nullopt;
1698 auto const pairs = checkedMulU128(count, degrees);
1699 denominator =
1700 pairs ? checkedMulU128(*pairs, math::detail::mulU64(moments.scale, moments.scale)) : std::nullopt;
1701 }
1702 auto const root = numerator && denominator
1703 ? math::detail::roundSqrtOfFraction(*numerator, *denominator, places, mode)
1704 : std::nullopt;
1705 if (!root) {
1706 return std::unexpected(overflow);
1707 }
1708 return *root;
1709}

1780 /// @brief Orders two values as sorting does: as `lt` orders them, except that strings compare case-folded as
1781 /// the table engine's default collator folds them, and by their own bytes when they fold the same, so
1782 /// the order never depends on the source order.
1783 /// @param lhs A value, not `null`.
1784 /// @param rhs A value, not `null`.
1785 /// @return The ordering; `nullopt` when the two are not of one comparable kind.
1786 [[nodiscard]] std::optional<std::strong_ordering> sortOrder(Value const& lhs, Value const& rhs) {
1787 std::string const* const lhsText = lhs.asString();

1790 table::CodePointCollator const collator;
1791 std::strong_ordering const folded = collator.fold(*lhsText) <=> collator.fold(*rhsText);
1792 return folded != std::strong_ordering::equal ? folded : *lhsText <=> *rhsText;
1793 }

1797 /// @brief A stable order of rows by their sort keys: by each key in turn, `null` last in either direction, and
1798 /// a descending key reversing the comparison, not the rows, so rows with equal keys keep their order.
1799 /// @param keys Each row's keys, most significant first; every row has one per entry of @p descending.
1800 /// @param descending Whether each key sorts descending.
1801 /// @return The rows' indices in sorted order; `nullopt` when a key's values are not of one comparable kind.
1802 [[nodiscard]] std::optional<std::vector<std::size_t>> sortRows(std::vector<std::vector<Value>> const& keys,
1803 std::vector<bool> const& descending) {
1804 // Every value of a key is checked against the key's first value, so the comparison below never meets a
1805 // pair it cannot order and the sort's ordering is strict and weak.
1806 for (std::size_t level = 0; level < descending.size(); ++level) {
1807 if (!orderable(keys, level)) {
1808 return std::nullopt;
1809 }
1810 }
1811 std::vector<std::size_t> indices(keys.size());
1812 // NOLINTNEXTLINE(modernize-use-ranges): std::ranges::iota is missing from Emscripten's libc++.
1813 std::iota(indices.begin(), indices.end(), std::size_t{0});
1814 std::ranges::stable_sort(indices, [&](std::size_t left, std::size_t right) {
1815 for (std::size_t level = 0; level < descending.size(); ++level) {
1816 std::strong_ordering const ordering =
1817 keyOrder(keys.at(left).at(level), keys.at(right).at(level), descending.at(level));
1818 if (ordering != std::strong_ordering::equal) {
1819 return ordering == std::strong_ordering::less;
1820 }
1821 }
1822 return false;
1823 });
1824 return indices;
1825 }

2201 Value arithmetic(Expr const& expr, Value const& lhsValue, Value const& rhsValue) {
2202 auto const lhs = numeric(lhsValue);
2203 auto const rhs = numeric(rhsValue);
2204 if (!lhs || !rhs) {
2205 return mismatch(expr, "an arithmetic operand is not a number");
2206 }
2207 UnitTable const& units = _env->formatting().units;
2208 FunctionId const function = expr.function;
2209 math::Rational right = rhs->value;
2210 std::string const* unit = nullptr;
2211 if (lhs->unit != nullptr && rhs->unit != nullptr) {
2212 if (function == FunctionId::Mul) {
2213 return mismatch(expr, "a product of two quantities has no declared unit");
2214 }

2216 if (!converted) {
2217 return mismatch(expr, "the units `" + *lhs->unit + "` and `" + *rhs->unit + "` do not convert");
2218 }
2219 right = *converted;
2220 unit = function == FunctionId::Div ? nullptr : lhs->unit;
2221 } else if (lhs->unit != nullptr || rhs->unit != nullptr) {
2222 bool const scales = function == FunctionId::Mul || (function == FunctionId::Div && lhs->unit != nullptr);
2223 if (!scales) {
2224 return mismatch(expr, "a quantity and a plain number do not combine this way");
2225 }
2226 unit = lhs->unit != nullptr ? lhs->unit : rhs->unit;
2227 }
2228 if (function == FunctionId::Div && right.isZero()) {
2229 return mismatch(expr, "division by zero");
2230 }
2231 std::expected<math::Rational, math::RationalError> result = math::Rational{};
2232 if (function == FunctionId::Add) {
2233 result = math::checkedAdd(lhs->value, right);
