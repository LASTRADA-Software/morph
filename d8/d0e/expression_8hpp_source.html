40/// @brief Whether a string is a document name: `[A-Za-z_][A-Za-z0-9_]*`.
41/// @param text The candidate.
42/// @return Whether @p text is a well-formed name.
43[[nodiscard]] constexpr bool isName(std::string_view text) noexcept {
44 if (text.empty()) {
45 return false;
46 }
47 bool first = true;
48 for (char const chr : text) {
49 bool const letter = (chr >= 'a' && chr <= 'z') || (chr >= 'A' && chr <= 'Z') || chr == '_';
50 bool const digit = chr >= '0' && chr <= '9';
51 if (!letter && (first || !digit)) {
52 return false;
53 }
54 first = false;
55 }
56 return true;
57}

59/// @brief The forms of an `expr/1` expression (spec 5 §4).
60enum class ExprKind : std::uint8_t {
61 Literal, ///< A JSON scalar, or `{"lit": v}`.
62 Ref, ///< `{"ref": "name.path"}`.
63 Row, ///< `{"row": "path"}`.
64 Query, ///< `{"query": "q"}`: a query's value.
65 Pending, ///< `{"pending": "q"}`: whether a query or mutation is in flight.
66 Error, ///< `{"error": "q"}`: a query's or mutation's error text.
67 Result, ///< `{"result": "m"}`: a mutation's last result.
68 ErrorKind, ///< `{"errorKind": "m"}`: the kind of a query's or mutation's error.
69 Refusal, ///< `{"refusal": "m"}`: a query's or mutation's action refusal, `null` or `{code, args}`.
70 Can, ///< `{"can": "m"}`: whether the principal may run a mutation or query.
71 Pref, ///< `{"pref": "point.path"}`: the user's choice at a customization point.
72 Call, ///< `{"fn": "name", "args": [...]}`, and `{"t": ...}`.
73 If, ///< `{"if": [cond, then, else]}`.
74 Object, ///< `{"obj": {...}}`.
75 List, ///< `{"list": [...]}`.
76 Map, ///< `{"map": e, "as": "x", "to": e}`.
77 Filter, ///< `{"filter": e, "as": "x", "where": e}`.
78 Sort, ///< `{"sort": e, "as": "x", "by": e, "order": e}`.
79 Runs, ///< `{"runs": e, "as": "x", "by": e}`.
80 Env, ///< `{"env": "name"}`: the client environment.
81 Session, ///< `{"session": "name"}`: a session attribute.
82 Host, ///< `{"host": "name"}`: a value the application registered with the client.
83 Get, ///< `{"get": e, "path": "a.b"}`: a path read out of a computed value.
84};

145/// @brief One node of a parsed expression. Immutable once parsed; shared between the document and its readers.
146struct Expr {
147 /// @brief The form.
148 ExprKind kind = ExprKind::Literal;
149 /// @brief The JSON Pointer of the node inside its document, for refusals and diagnostics.
150 std::string path;
151 /// @brief The constant of a `Literal`.
152 Value literal;
153 /// @brief The name a form reads: a ref's head, a query or mutation, a pref's point, a comprehension's variable,
154 /// an `env`, `session` or `host` name.
155 std::string name;
156 /// @brief The path after the head of a `Ref` or `Pref`, or the path of a `Row` or `Get`.
157 std::vector<std::string> fieldPath;
158 /// @brief The function of a `Call`.
159 FunctionId function = FunctionId::And;
160 /// @brief Operands: a call's arguments; `if`'s condition, then and else; a list's items; a comprehension's
161 /// source and body (and a `sort`'s `order`, when given); the value a `get` reads from.
162 std::vector<ExprPtr> args;
163 /// @brief The fields of an `Object`, in the order written.
164 std::vector<std::pair<std::string, ExprPtr>> fields;
165};
