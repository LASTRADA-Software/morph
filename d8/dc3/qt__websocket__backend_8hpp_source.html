21/// @brief Reconnect tuning for `QtWebSocketBackend`.
22///
23/// Declared outside `QtWebSocketBackend` so its default initialisers are fully
24/// parsed before any constructor default argument that names `Config{}` is
25/// evaluated (same rationale as `morph::offline::NetworkMonitorConfig`).
26struct QtWebSocketBackendConfig {
27 /// @brief Whether to attempt automatic reconnect after an unsolicited disconnect.
28 bool reconnectEnabled = true;
29
30 /// @brief Delay before the first reconnect attempt after a disconnect.
31 std::chrono::milliseconds initialReconnectDelay = std::chrono::milliseconds{500};
32
33 /// @brief Upper bound on the exponential backoff between reconnect attempts.
34 std::chrono::milliseconds maxReconnectDelay = std::chrono::seconds{30};
35
36 /// @brief Multiplier applied to the delay after each failed attempt.
37 double backoffMultiplier = 2.0;
38};

40/// @brief `IBackend` implementation that communicates with a `RemoteServer` over WebSocket.
41///
42/// `registerModel()` is synchronous (blocks the calling thread via a nested
43/// `QEventLoop` until the server replies). `deregisterModel()` is fire-and-forget
44/// (it sends the message without waiting, avoiding a nested event loop during
45/// destruction). `execute()` is asynchronous: it assigns a call-id, sends the
46/// message, and resolves the returned `Completion` when the matching reply arrives.
47///
48/// @par TLS
49/// Pass a `QSslConfiguration` to enable `wss://`. Build it with `tlsVerifyingConfig()`
50/// (CA-verified — the recommended production default) or `tlsPinnedConfig()`
51/// (pinned-certificate — the correct choice for a self-signed deployment), both in
52/// `qt_tls.hpp`. `tlsInsecureNoVerify()` disables peer verification entirely and is
53/// for local development and tests only — see security.md's "Transport security" section.
54///
55/// @par Threading
56/// Must be used from the Qt event loop thread. `execute()` and the internal
57/// message handler are both called on that thread.
58class QtWebSocketBackend : public ::morph::backend::detail::IBackend {
59public:
60 /// @brief Alias for the reconnect configuration struct.
61 using Config = QtWebSocketBackendConfig;
62
63 /// @brief Constructs the backend and opens a WebSocket connection to @p serverUrl.
64 ///
65 /// @param serverUrl `ws://` or `wss://` URL of the remote `RemoteServer`.
66 /// @param dispatcher Action dispatcher (defaults to the process-level singleton).
67 /// @param registry Model registry (defaults to the process-level singleton).
68 /// @param tls If non-null, enables TLS and applies this configuration.
69 /// @param cfg Reconnect tuning. Default: enabled, 500ms initial / 30s cap, 2x backoff.
70 explicit QtWebSocketBackend(
71 QUrl serverUrl,
72 ::morph::model::detail::ActionDispatcher& dispatcher = ::morph::model::detail::defaultDispatcher(),
73 ::morph::model::detail::ModelRegistryFactory& registry = ::morph::model::detail::defaultRegistry(),
74 std::optional<QSslConfiguration> tls = std::nullopt, Config cfg = Config{});
75
76 /// @brief Closes the socket and cleans up pending operations.
77 ~QtWebSocketBackend() override;

79 /// @brief Pumps the Qt event loop until the socket is connected or @p timeoutMs elapses.
80 ///
81 /// Must be called on the Qt event loop thread after construction.
82 ///
83 /// @param timeoutMs Maximum time to wait in milliseconds.
84 /// @return `true` if connected before the timeout, `false` otherwise.
85 bool waitForConnected(int timeoutMs = 5000);
86
87 /// @brief Sends a `"hello"` envelope to the server and classifies its reply.
88 ///
89 /// Synchronous, like `registerModel` — blocks the calling (Qt event loop)
90 /// thread via a nested `QEventLoop` until the reply arrives. Intended to be
91 /// called once, after `waitForConnected()` returns `true` and before any
92 /// `registerModel`/`execute` call; nothing enforces that ordering.
93 ///
94 /// @return `Negotiated` if the server accepted `kProtocolVersion`;
95 /// `LegacyPeer` if the server does not understand `"hello"`.
96 /// @throws std::runtime_error if the server explicitly rejects the version,
97 /// or if the socket is not connected (or disconnects mid-call).
98 ::morph::wire::ProtocolNegotiationResult negotiateProtocolVersion();
99
100 /// @brief Sends a `register` message to the server and blocks until the reply arrives.
101 ///
102 /// @param typeId String type-id of the model to register.
103 /// @param factory Ignored — model construction is delegated to the server.
104 /// @return `ModelId` assigned by the server.
105 /// @throws std::runtime_error if the server replies with an error or the socket is not connected.
106 ::morph::exec::detail::ModelId registerModel(
107 const std::string& typeId,
108 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory) override;
109
110 /// @brief Sends a shared (register-or-attach) `register` and blocks for the reply.
111 ///
112 /// An empty primary degrades to the private path.
113 /// @param typeId String type-id of the model.
114 /// @param factory Ignored — model construction is delegated to the server.
115 /// @param identity Entity key for the action log plus the directory primary key.
116 /// @return `ModelId` of the shared (or newly created) instance.
117 /// @throws std::runtime_error if the server errors or the socket is not connected.
118 ::morph::exec::detail::ModelId registerModelShared(
119 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
120 ::morph::backend::detail::InstanceIdentity identity) override;
121
122 /// @brief Sends an `attach` and blocks for the reply, re-pointing from @p current.
123 /// @param typeId String type-id of the model.
124 /// @param factory Ignored — model construction is delegated to the server.
125 /// @param identity Entity key for the action log plus the directory primary key.
126 /// @param current Instance currently held, or `ModelId{0}` if none.
127 /// @return `ModelId` of the instance now attached to.
128 /// @throws std::runtime_error if the server errors or the socket is not connected.
129 ::morph::exec::detail::ModelId attachModel(
130 const std::string& typeId, std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
131 ::morph::backend::detail::InstanceIdentity identity, ::morph::exec::detail::ModelId current) override;
132
133 /// @brief Files a live server-side instance under @p primary.
134 /// @param mid Live instance to promote.
135 /// @param typeId Model type id.
136 /// @param primary Canonical string encoding of the key to file it under.
137 void assignPrimary(::morph::exec::detail::ModelId mid, const std::string& typeId,
138 std::string_view primary) override;

140 /// @brief Asks the server for the live shared primary keys of @p typeId.
141 /// @param typeId String type-id to enumerate.
142 /// @return Canonical key strings of the live shared instances.
143 /// @throws std::runtime_error if the server errors or the socket is not connected.
144 std::vector<std::string> listInstances(const std::string& typeId) override;

146 /// @brief Sends a `deregister` message fire-and-forget (does not wait for a reply).
147 ///
148 /// No acknowledgement is awaited, which avoids a nested `QEventLoop` during
149 /// destruction (that can trip Qt asserts). Note the server performs no
150 /// connection-scoped cleanup: an undelivered or lost `deregister` leaves the
151 /// model registered on the server indefinitely.
152 ///
153 /// @param mid Id of the model to remove on the server.
154 void deregisterModel(::morph::exec::detail::ModelId mid) override;
155
156 /// @brief Sends an `execute` message and returns a `Completion` that resolves on reply.
157 ///
158 /// Assigns a monotonically increasing call-id so that concurrent calls can be
159 /// matched to their replies. The `Completion` callbacks are posted via @p cbExec.
160 ///
161 /// @param mid Target model id on the server.
162 /// @param call Bundled action; `serializeAction` and `deserializeResult` are used.
163 /// @param cbExec Executor for delivering the completion callbacks.
164 /// @return Completion resolved asynchronously when the server reply arrives.
165 ::morph::async::Completion<std::shared_ptr<void>> execute(::morph::exec::detail::ModelId mid,
166 ::morph::backend::detail::ActionCall call,
167 ::morph::exec::IExecutor* cbExec) override;
168
169 /// @brief No-op — this backend holds no local model objects.
170 void notifyBackendChanged() override {}
171
172 /// @brief Resolves every pending execute call's `Completion` with @p exc.
173 ///
174 /// Called by `Bridge::switchBackend()` on the outgoing backend, by `~Bridge`,
175 /// and internally when the socket disconnects. Late replies arriving for
176 /// already-cancelled call ids are dropped silently.
177 ///
178 /// @param exc Exception delivered to every pending completion's error sink.
179 void cancelPending(const std::exception_ptr& exc) override;
180
181 /// @brief Installs the handler `Bridge` uses to re-register handlers after a reconnect.
182 /// @param handler Callable invoked on the Qt thread after every successful reconnect.
183 /// Pass `nullptr` to clear.
184 void setReconnectHandler(const std::function<void()>& handler) override;

186private:
