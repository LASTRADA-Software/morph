20/// @brief Reconnect tuning for `QtWebSocketBackend`.
21///
22/// Declared outside `QtWebSocketBackend` so its default initialisers are fully
23/// parsed before any constructor default argument that names `Config{}` is
24/// evaluated (same rationale as `morph::offline::NetworkMonitorConfig`).
25struct QtWebSocketBackendConfig {
26 /// @brief Whether to attempt automatic reconnect after an unsolicited disconnect.
27 bool reconnectEnabled = true;
28
29 /// @brief Delay before the first reconnect attempt after a disconnect.
30 std::chrono::milliseconds initialReconnectDelay = std::chrono::milliseconds{500};
31
32 /// @brief Upper bound on the exponential backoff between reconnect attempts.
33 std::chrono::milliseconds maxReconnectDelay = std::chrono::seconds{30};
34
35 /// @brief Multiplier applied to the delay after each failed attempt.
36 double backoffMultiplier = 2.0;
37};

39/// @brief `IBackend` implementation that communicates with a `RemoteServer` over WebSocket.
40///
41/// All `registerModel()` and `deregisterModel()` calls are synchronous (block the
42/// calling thread via a nested `QEventLoop` until the server replies). `execute()`
43/// is asynchronous: it assigns a call-id, sends the message, and resolves the
44/// returned `Completion` when the matching reply arrives.
45///
46/// @par TLS
47/// Pass a `QSslConfiguration` to enable `wss://`. For self-signed certificates
48/// set `QSslSocket::VerifyNone` on the configuration before passing it in.
49///
50/// @par Threading
51/// Must be used from the Qt event loop thread. `execute()` and the internal
52/// message handler are both called on that thread.
53class QtWebSocketBackend : public ::morph::backend::detail::IBackend {
54public:
55 /// @brief Alias for the reconnect configuration struct.
56 using Config = QtWebSocketBackendConfig;
57
58 /// @brief Constructs the backend and opens a WebSocket connection to @p serverUrl.
59 ///
60 /// @param serverUrl `ws://` or `wss://` URL of the remote `RemoteServer`.
61 /// @param dispatcher Action dispatcher (defaults to the process-level singleton).
62 /// @param registry Model registry (defaults to the process-level singleton).
63 /// @param tls If non-null, enables TLS and applies this configuration.
64 /// @param cfg Reconnect tuning. Default: enabled, 500ms initial / 30s cap, 2x backoff.
65 explicit QtWebSocketBackend(
66 QUrl serverUrl,
67 ::morph::model::detail::ActionDispatcher& dispatcher = ::morph::model::detail::defaultDispatcher(),
68 ::morph::model::detail::ModelRegistryFactory& registry = ::morph::model::detail::defaultRegistry(),
69 std::optional<QSslConfiguration> tls = std::nullopt,
70 Config cfg = Config{});
71
72 /// @brief Closes the socket and cleans up pending operations.
73 ~QtWebSocketBackend() override;

75 /// @brief Pumps the Qt event loop until the socket is connected or @p timeoutMs elapses.
76 ///
77 /// Must be called on the Qt event loop thread after construction.
78 ///
79 /// @param timeoutMs Maximum time to wait in milliseconds.
80 /// @return `true` if connected before the timeout, `false` otherwise.
81 bool waitForConnected(int timeoutMs = 5000);

83 /// @brief Sends a `register` message to the server and blocks until the reply arrives.
84 ///
85 /// @param typeId String type-id of the model to register.
86 /// @param factory Ignored — model construction is delegated to the server.
87 /// @return `ModelId` assigned by the server.
88 /// @throws std::runtime_error if the server replies with an error or the socket is not connected.
89 ::morph::exec::detail::ModelId registerModel(
90 const std::string& typeId,

93 /// @brief Sends a `deregister` message and blocks until the server acknowledges.
94 ///
95 /// @param mid Id of the model to remove on the server.
96 void deregisterModel(::morph::exec::detail::ModelId mid) override;
97
98 /// @brief Sends an `execute` message and returns a `Completion` that resolves on reply.
99 ///
100 /// Assigns a monotonically increasing call-id so that concurrent calls can be
101 /// matched to their replies. The `Completion` callbacks are posted via @p cbExec.
102 ///
103 /// @param mid Target model id on the server.
104 /// @param call Bundled action; `serializeAction` and `deserializeResult` are used.
105 /// @param cbExec Executor for delivering the completion callbacks.
106 /// @return Completion resolved asynchronously when the server reply arrives.
107 ::morph::async::Completion<std::shared_ptr<void>> execute(::morph::exec::detail::ModelId mid,
108 ::morph::backend::detail::ActionCall call,
109 ::morph::exec::IExecutor* cbExec) override;
110
111 /// @brief No-op — this backend holds no local model objects.
112 void notifyBackendChanged() override {}
113
114 /// @brief Resolves every pending execute call's `Completion` with @p exc.
115 ///
116 /// Called by `Bridge::switchBackend()` on the outgoing backend, by `~Bridge`,
117 /// and internally when the socket disconnects. Late replies arriving for
118 /// already-cancelled call ids are dropped silently.
119 ///
120 /// @param exc Exception delivered to every pending completion's error sink.
121 void cancelPending(const std::exception_ptr& exc) override;
122
123 /// @brief Installs the handler `Bridge` uses to re-register handlers after a reconnect.
124 /// @param handler Callable invoked on the Qt thread after every successful reconnect.
125 /// Pass `nullptr` to clear.
126 void setReconnectHandler(const std::function<void()>& handler) override;
