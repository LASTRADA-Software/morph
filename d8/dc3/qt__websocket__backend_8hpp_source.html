27/// parsed before any constructor default argument that names `Config{}` is
28/// evaluated (same rationale as `morph::offline::NetworkMonitorConfig`).

30 /// @brief Whether to attempt automatic reconnect after an unsolicited disconnect.
31 bool reconnectEnabled = true;
32
33 /// @brief Delay before the first reconnect attempt after a disconnect.
34 std::chrono::milliseconds initialReconnectDelay = std::chrono::milliseconds{500};
35
36 /// @brief Upper bound on the exponential backoff between reconnect attempts.
37 std::chrono::milliseconds maxReconnectDelay = std::chrono::seconds{30};
38
39 /// @brief Multiplier applied to the delay after each failed attempt.
40 double backoffMultiplier = 2.0;

42 /// @brief Opt in to `registerModelAsync` (see its doc comment on `IBackend`).
43 ///
44 /// Defaults to `false`: `Bridge::registerHandler()` then falls back to the
45 /// synchronous `registerModel`, exactly as before this feature existed —
46 /// every existing embedder (a desktop Qt client, this backend's own test
47 /// suite) keeps registering synchronously, immediately usable the line
48 /// after `BridgeHandler`'s constructor returns.
49 ///
50 /// Set `true` only for a build where that synchronous guarantee cannot
51 /// hold at all — a WASM main thread, where the nested `QEventLoop`
52 /// `registerModel` relies on aborts the page outright. Doing so is a
53 /// deliberate trade: the caller must then wait for registration to
54 /// complete (e.g. gate the UI on it) before firing an action through that
55 /// handler, since `executeVia` fails fast with "handler not bound" for an
56 /// unbound binding rather than queuing or blocking.
57 bool asyncRegistrationEnabled = false;
58};

60/// @brief `IBackend` implementation that communicates with a `RemoteServer` over WebSocket.
61///
62/// `registerModel()` is synchronous (blocks the calling thread via a nested
63/// `QEventLoop` until the server replies) -- unusable on a WASM main thread,
64/// which Qt refuses to spin a nested loop on at all. `registerModelAsync()`
65/// is the non-blocking alternative `Bridge::registerHandler()` prefers when
66/// available (see `IBackend::registerModelAsync`'s doc comment): it assigns a
67/// call-id, sends the message, returns immediately, and invokes exactly one
68/// of its `onRegistered`/`onError` callbacks once the matching reply arrives
69/// -- the same call-id-matching mechanism `execute()` already uses.
70/// `deregisterModel()` is fire-and-forget (it sends the message without
71/// waiting, avoiding a nested event loop during destruction). `execute()` is
72/// asynchronous: it assigns a call-id, sends the message, and resolves the
73/// returned `Completion` when the matching reply arrives.
74///
75/// @par TLS
76/// Pass a `QSslConfiguration` to enable `wss://`. Build it with `tlsVerifyingConfig()`
77/// (CA-verified — the recommended production default) or `tlsPinnedConfig()`
78/// (pinned-certificate — the correct choice for a self-signed deployment), both in
79/// `qt_tls.hpp`. `tlsInsecureNoVerify()` disables peer verification entirely and is
80/// for local development and tests only — see security.md's "Transport security" section.
81///
82/// @par SSL-less Qt builds (`QT_NO_SSL`, including the standard Qt-for-WebAssembly build)
83/// The constructor's `tls` parameter (and the `_tls` member) does not exist at all
84/// when Qt itself was configured without SSL — `QSslConfiguration` isn't a type
85/// Qt provides in that configuration, so there is no value to accept or ignore.
86/// `wss://` still works on such a build regardless: in a WASM/browser
87/// deployment the browser terminates TLS before Qt's `QWebSocket` ever sees
88/// the connection, so the only thing genuinely unavailable is the ability to
89/// *configure* TLS from C++ (client certificates, pinning, etc.) — plain
90/// `wss://` and `ws://` both still connect normally.
91///
92/// @par Threading
93/// Must be used from the Qt event loop thread. `execute()` and the internal
94/// message handler are both called on that thread.
95class QtWebSocketBackend : public ::morph::backend::detail::IBackend {

152 /// @brief Sends a `register` message and returns without blocking; the
153 /// reply is matched later, asynchronously, by `callId`.
154 ///
155 /// The non-blocking counterpart to `registerModel`/`registerModelWithContext`
156 /// (see `IBackend::registerModelAsync`'s doc comment for why this exists):
157 /// `registerModel` blocks the calling thread in a nested `QEventLoop` via
158 /// `sendSync`, which a WASM main thread cannot do at all. This instead
159 /// assigns a fresh `callId` (the same counter `execute()` uses), sends the
160 /// `register` envelope, and returns `true` immediately; the reply is
161 /// matched via `_pendingRegistrations` when `onTextMessage` sees it (no
162 /// protocol change needed — the server already echoes `callId` on every
163 /// reply, `register` included). Exactly one of @p onRegistered / @p onError
164 /// fires, on the Qt event loop thread, once the reply arrives — or never,
165 /// if the socket disconnects first without ever reconnecting and
166 /// `cancelPending` is never called again for this id (a disconnect
167 /// *before* a reconnect calls `cancelPending`, which does invoke @p onError
168 /// — see `cancelPending`'s doc comment).
169 ///
170 /// If the socket has not finished connecting yet, the request is **queued**
171 /// rather than failed: this is exactly the ordering a single-threaded WASM
172 /// client must use, since it can never block waiting for the connection to
173 /// settle (a `BridgeHandler` constructed the moment the backend is wired
174 /// up, before the first `connected` signal). The queued request is sent —
175 /// with a call-id assigned at that point, not now — the moment `connected`
176 /// fires next (first connect included), in FIFO order. If the socket never
177 /// connects at all and the backend is torn down first, the queued entry is
178 /// still resolved: `~QtWebSocketBackend` calls `cancelPending`, which drains
179 /// `_queuedRegistrations` too and invokes @p onError exactly once for each —
180 /// no call-id was ever assigned, but the callback still fires, the same
181 /// guarantee an already-sent (call-id-bearing) registration gets.
182 ///
183 /// @param typeId String type-id of the model to instantiate.
184 /// @param factory Unused — this backend holds no local model to construct;
185 /// the server instantiates the model from `typeId`.
186 /// @param contextKey Stable identity of the new instance; travels in the wire envelope.
187 /// @param onRegistered Invoked with the server-assigned `ModelId` on success.
188 /// @param onError Invoked with a diagnostic message on failure or disconnect.
189 /// @return `true` always — this backend has an async path (`false` is never returned).
190 bool registerModelAsync(const std::string& typeId,
191 std::function<std::unique_ptr<::morph::model::detail::IModelHolder>()> factory,
192 std::string_view contextKey, std::function<void(::morph::exec::detail::ModelId)> onRegistered,
193 std::function<void(const std::string&)> onError) override;

256 /// @brief Sends a `deregister` message fire-and-forget (does not wait for a reply).
257 ///
258 /// No acknowledgement is awaited, which avoids a nested `QEventLoop` during
259 /// destruction (that can trip Qt asserts). Note the server performs no
260 /// connection-scoped cleanup: an undelivered or lost `deregister` leaves the
261 /// model registered on the server indefinitely.
262 ///
263 /// Assigned a real, non-zero `callId` from the same counter/namespace
264 /// `execute()`/`registerModelAsync()` use (see issue #65): `callId == 0`
265 /// is reserved for a parked synchronous control call's reply, and a
266 /// fire-and-forget `deregister` sharing that sentinel could otherwise have
267 /// its own stray "ok" reply handed to an unrelated `registerModel`'s
268 /// parked `sendSync` loop if the two land back to back on the same
269 /// connection. The reply is tracked in `_pendingDeregisters` purely so it
270 /// can be recognised and dropped in `onTextMessage`; nothing observes it.
271 ///
272 /// @param mid Id of the model to remove on the server.
273 void deregisterModel(::morph::exec::detail::ModelId mid) override;

380 /// @brief One in-flight `registerModelAsync` call, keyed by `callId`.
381 ///
382 /// Kept separate from `PendingExecute`/`_pending` (a different `callId`
383 /// namespace would be a protocol change; this shares the same namespace
384 /// and counter, just a different local map) because a register reply's
385 /// shape (`modelId`, no `deserialize` step) differs from an execute
386 /// reply's.
387 struct PendingRegistration {
388 std::function<void(::morph::exec::detail::ModelId)> onRegistered;
389 std::function<void(const std::string&)> onError;
390 };
391 std::unordered_map<uint64_t, PendingRegistration> _pendingRegistrations;

393 /// @brief One `registerModelAsync` call made before the socket had
394 /// finished connecting. No call-id is assigned until the request
395 /// is actually sent (from `flushQueuedRegistrations`), so a queued
396 /// entry that never gets to fire (backend destroyed first) needs no
397 /// cancellation bookkeeping.
398 struct QueuedRegistration {
399 std::string typeId;
400 std::string contextKey;
401 std::function<void(::morph::exec::detail::ModelId)> onRegistered;
402 std::function<void(const std::string&)> onError;
403 };
404 std::vector<QueuedRegistration> _queuedRegistrations;
405
406 /// @brief Call-ids of `deregister` envelopes still awaiting their (unused)
407 /// reply (see issue #65).
408 ///
409 /// `deregisterModel` is fire-and-forget: nobody observes the reply, but it
410 /// still needs a real, non-zero `callId` so `onTextMessage` can recognise
411 /// and drop it explicitly, rather than letting it fall through to the
412 /// `callId == 0` branch and collide with a parked `sendSync` waiter. A
413 /// late reply for an id no longer in this set (already dropped, or the
414 /// backend was cancelled/destroyed) is simply ignored — nothing to clean
415 /// up either way.
416 std::unordered_set<uint64_t> _pendingDeregisters;
417
418 /// @brief One in-flight `assignPrimaryAsync` call, keyed by `callId`.
419 ///
420 /// Mirrors `PendingRegistration`/`_pendingRegistrations`: same `callId`
421 /// namespace and counter, separate map because an `assign` reply is
422 /// matched the same way a `register`/`registerShared` reply is (a bare
423 /// `modelId`, echoing back the instance that was promoted).
424 struct PendingAssign {
425 std::function<void(::morph::exec::detail::ModelId)> onRegistered;
426 std::function<void(const std::string&)> onError;
427 };
428 std::unordered_map<uint64_t, PendingAssign> _pendingAssigns;
