120/// @brief Looks a variable up in the process's environment.
121/// @param name The variable.
122/// @return Its value; nothing when it is not set.
123[[nodiscard]] inline std::optional<std::string> processEnvironment(std::string_view name) {
124 // NOLINTNEXTLINE(concurrency-mt-unsafe): read at startup, before a server's threads run.
125 char const* const value = std::getenv(std::string{name}.c_str());
126 return value == nullptr ? std::nullopt : std::optional<std::string>{std::string{value}};
127}

129/// @brief The key a provider block names: inline, from the environment, or none.
130/// @param given The inline `apiKey`.
131/// @param variable The `apiKeyEnv`.
132/// @param environment How the variable is looked up.
133/// @param where What the block is, for the refusal.
134/// @return The key, empty for none, or why not: both named, or a variable that is not set.
135[[nodiscard]] inline std::expected<std::string, std::string> keyOf(std::optional<std::string> given,
136 std::optional<std::string> const& variable,
137 EnvironmentLookup const& environment,
138 std::string_view where) {
139 if (given && variable) {
140 return std::unexpected(std::string{where} + " names both `apiKey` and `apiKeyEnv`");
141 }
142 if (given) {
143 return std::move(*given);
144 }
145 if (!variable) {
146 return std::string{};
147 }
148 auto key = environment ? environment(*variable) : std::nullopt;
149 if (!key) {
150 return std::unexpected("the environment variable `" + *variable + "` named by `apiKeyEnv` is not set");
151 }
152 return std::move(*key);
153}

157/// @brief Reads a chat's configuration from JSON text.
158///
159/// A key the format does not have is refused, so a misspelled limit cannot silently leave the default in force. What
160/// is left out keeps its default. The key comes from `apiKeyEnv`, the name of an environment variable, or from an
161/// inline `apiKey`; naming both is refused. A refusal says what is wrong without quoting the text, which may hold the
162/// key.
163/// @param text The JSON text.
164/// @param environment How `apiKeyEnv` is looked up; the process's environment by default.
165/// @return The configuration, or why not.
166[[nodiscard]] inline std::expected<ChatConfig, std::string> parseChatConfig(
167 std::string_view text, EnvironmentLookup const& environment = detail::processEnvironment) {
168 detail::ConfigFile file;
169 if (auto const error = glz::read<glz::opts{.null_terminated = false, .error_on_unknown_keys = true}>(file, text)) {
170 // Without the text, so the reason cannot quote a key held in it.
171 return std::unexpected("the chat configuration cannot be read: " + std::string{glz::format_error(error)});
172 }
173 ChatConfig config;
174 config.provider.name = std::move(file.provider.name);
175 config.provider.baseUrl = std::move(file.provider.baseUrl);
176 config.provider.model = std::move(file.provider.model);
177 config.systemPrompt = std::move(file.systemPrompt);
178 auto providerKey =
179 detail::keyOf(std::move(file.provider.apiKey), file.provider.apiKeyEnv, environment, "the chat configuration");
180 if (!providerKey) {
181 return std::unexpected(std::move(providerKey.error()));
182 }
183 config.provider.apiKey = std::move(*providerKey);
184 if (file.transcription) {
185 detail::TranscriptionFile& block = *file.transcription;
186 auto key = detail::keyOf(std::move(block.apiKey), block.apiKeyEnv, environment,
187 "the chat configuration's `transcription`");
188 if (!key) {
189 return std::unexpected(std::move(key.error()));
190 }
191 config.transcription = TranscriptionConfig{.name = std::move(block.name),
192 .baseUrl = std::move(block.baseUrl),
193 .model = std::move(block.model),
194 .apiKey = std::move(*key),
195 .language = std::move(block.language)};
196 }
197 if (file.provider.maxTokens) {
198 if (*file.provider.maxTokens <= 0) {
199 return std::unexpected(std::string{"the chat configuration's `maxTokens` must be above 0"});
200 }
201 config.provider.maxTokens = file.provider.maxTokens;
202 }
203 // A limit of 0 would refuse every turn, so it can only be a mistake.
204 auto const apply = [](std::optional<std::size_t> const& given, std::size_t& into, std::string_view name,
205 std::string& problem) {
206 if (given) {
207 if (*given == 0 && problem.empty()) {
208 problem = "the chat configuration's `" + std::string{name} + "` must be above 0";
209 }
210 into = *given;
211 }
212 };
213 std::string problem;
214 detail::LimitsFile const& limits = file.limits;
215 apply(limits.maxToolCallsPerTurn, config.limits.maxToolCallsPerTurn, "maxToolCallsPerTurn", problem);
216 apply(limits.maxTurnsPerPrincipal, config.limits.maxTurnsPerPrincipal, "maxTurnsPerPrincipal", problem);
217 apply(limits.maxTurnsPerConnection, config.limits.maxTurnsPerConnection, "maxTurnsPerConnection", problem);
218 apply(limits.maxConversationsPerPrincipal, config.limits.maxConversationsPerPrincipal,
219 "maxConversationsPerPrincipal", problem);
220 apply(limits.maxMessagesPerConversation, config.limits.maxMessagesPerConversation, "maxMessagesPerConversation",
221 problem);
222 apply(limits.maxTextBytes, config.limits.maxTextBytes, "maxTextBytes", problem);
223 apply(limits.maxToolResultBytes, config.limits.maxToolResultBytes, "maxToolResultBytes", problem);
224 apply(limits.maxConversationBytes, config.limits.maxConversationBytes, "maxConversationBytes", problem);
225 apply(limits.maxProposalBytes, config.limits.maxProposalBytes, "maxProposalBytes", problem);
226 apply(limits.maxDraftBytes, config.limits.maxDraftBytes, "maxDraftBytes", problem);
227 apply(limits.maxInlineDocumentBytes, config.limits.maxInlineDocumentBytes, "maxInlineDocumentBytes", problem);
228 apply(limits.maxProposalsPerTurn, config.limits.maxProposalsPerTurn, "maxProposalsPerTurn", problem);
229 apply(limits.maxDictationSeconds, config.limits.maxDictationSeconds, "maxDictationSeconds", problem);
230 apply(limits.maxDictationBytes, config.limits.maxDictationBytes, "maxDictationBytes", problem);
231 apply(limits.maxTranscriptionsPerPrincipal, config.limits.maxTranscriptionsPerPrincipal,
232 "maxTranscriptionsPerPrincipal", problem);
233 if (!problem.empty()) {
234 return std::unexpected(std::move(problem));
235 }
236 if (config.limits.maxDictationBytes > kMaxDictationBytes) {
237 return std::unexpected("the chat configuration's `maxDictationBytes` must be at most " +
238 std::to_string(kMaxDictationBytes));
239 }
240 if (limits.transcriptionRetentionSeconds) {
241 if (*limits.transcriptionRetentionSeconds <= 0) {
242 return std::unexpected(
243 std::string{"the chat configuration's `transcriptionRetentionSeconds` must be above 0"});
244 }
245 config.limits.transcriptionRetention = std::chrono::seconds{*limits.transcriptionRetentionSeconds};
246 }
247 if (limits.toolCallTimeoutSeconds) {
248 if (*limits.toolCallTimeoutSeconds <= 0) {
249 return std::unexpected(std::string{"the chat configuration's `toolCallTimeoutSeconds` must be above 0"});
250 }
251 config.limits.toolCallTimeout = std::chrono::seconds{*limits.toolCallTimeoutSeconds};
252 }
253 if (limits.idleRetentionMinutes) {
254 if (*limits.idleRetentionMinutes <= 0) {
255 return std::unexpected(std::string{"the chat configuration's `idleRetentionMinutes` must be above 0"});
256 }
257 config.limits.idleRetention = std::chrono::minutes{*limits.idleRetentionMinutes};
258 }
259 return config;
260}

262/// @brief Reads a chat's configuration from a JSON file. See `parseChatConfig`.
263/// @param path The file.
264/// @param environment How `apiKeyEnv` is looked up; the process's environment by default.
265/// @return The configuration, or why not.
266[[nodiscard]] inline std::expected<ChatConfig, std::string> loadChatConfig(
267 std::filesystem::path const& path, EnvironmentLookup const& environment = detail::processEnvironment) {
268 std::ifstream const file{path};
269 if (!file) {
270 return std::unexpected("the chat configuration file `" + path.string() + "` cannot be opened");
271 }
272 std::ostringstream text;
273 text << file.rdbuf();
274 return parseChatConfig(text.str(), environment);
275}
