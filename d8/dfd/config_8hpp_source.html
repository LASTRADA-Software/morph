85/// @brief Looks a variable up in the process's environment.

87/// @return Its value; nothing when it is not set.
88[[nodiscard]] inline std::optional<std::string> processEnvironment(std::string_view name) {
89 // NOLINTNEXTLINE(concurrency-mt-unsafe): read at startup, before a server's threads run.

96/// @brief Reads a chat's configuration from JSON text.
97///
98/// A key the format does not have is refused, so a misspelled limit cannot silently leave the default in force. What
99/// is left out keeps its default. The key comes from `apiKeyEnv`, the name of an environment variable, or from an
100/// inline `apiKey`; naming both is refused. A refusal says what is wrong without quoting the text, which may hold the
101/// key.
102/// @param text The JSON text.
103/// @param environment How `apiKeyEnv` is looked up; the process's environment by default.
104/// @return The configuration, or why not.
105[[nodiscard]] inline std::expected<ChatConfig, std::string> parseChatConfig(
106 std::string_view text, EnvironmentLookup const& environment = detail::processEnvironment) {
107 detail::ConfigFile file;
108 if (auto const error = glz::read<glz::opts{.null_terminated = false, .error_on_unknown_keys = true}>(file, text)) {
109 // Without the text, so the reason cannot quote a key held in it.
110 return std::unexpected("the chat configuration cannot be read: " + std::string{glz::format_error(error)});
111 }
112 ChatConfig config;
113 config.provider.name = std::move(file.provider.name);
114 config.provider.baseUrl = std::move(file.provider.baseUrl);
115 config.provider.model = std::move(file.provider.model);
116 config.systemPrompt = std::move(file.systemPrompt);
117 if (file.provider.apiKey && file.provider.apiKeyEnv) {
118 return std::unexpected(std::string{"the chat configuration names both `apiKey` and `apiKeyEnv`"});
119 }
120 if (file.provider.apiKey) {
121 config.provider.apiKey = std::move(*file.provider.apiKey);
122 } else if (file.provider.apiKeyEnv) {
123 auto key = environment ? environment(*file.provider.apiKeyEnv) : std::nullopt;
124 if (!key) {
125 return std::unexpected("the environment variable `" + *file.provider.apiKeyEnv +
126 "` named by `apiKeyEnv` is not set");
127 }
128 config.provider.apiKey = std::move(*key);
129 }
130 if (file.provider.maxTokens) {
131 if (*file.provider.maxTokens <= 0) {
132 return std::unexpected(std::string{"the chat configuration's `maxTokens` must be above 0"});
133 }
134 config.provider.maxTokens = file.provider.maxTokens;
135 }
136 // A limit of 0 would refuse every turn, so it can only be a mistake.
137 auto const apply = [](std::optional<std::size_t> const& given, std::size_t& into, std::string_view name,
138 std::string& problem) {
139 if (given) {
140 if (*given == 0 && problem.empty()) {
141 problem = "the chat configuration's `" + std::string{name} + "` must be above 0";
142 }
143 into = *given;
144 }
145 };
146 std::string problem;
147 detail::LimitsFile const& limits = file.limits;
148 apply(limits.maxToolCallsPerTurn, config.limits.maxToolCallsPerTurn, "maxToolCallsPerTurn", problem);
149 apply(limits.maxTurnsPerPrincipal, config.limits.maxTurnsPerPrincipal, "maxTurnsPerPrincipal", problem);
150 apply(limits.maxTurnsPerConnection, config.limits.maxTurnsPerConnection, "maxTurnsPerConnection", problem);
151 apply(limits.maxConversationsPerPrincipal, config.limits.maxConversationsPerPrincipal,
152 "maxConversationsPerPrincipal", problem);
153 apply(limits.maxMessagesPerConversation, config.limits.maxMessagesPerConversation, "maxMessagesPerConversation",
154 problem);
155 apply(limits.maxTextBytes, config.limits.maxTextBytes, "maxTextBytes", problem);
156 apply(limits.maxToolResultBytes, config.limits.maxToolResultBytes, "maxToolResultBytes", problem);
157 apply(limits.maxConversationBytes, config.limits.maxConversationBytes, "maxConversationBytes", problem);
158 if (!problem.empty()) {
159 return std::unexpected(std::move(problem));
160 }
161 if (limits.idleRetentionMinutes) {
162 if (*limits.idleRetentionMinutes <= 0) {
163 return std::unexpected(std::string{"the chat configuration's `idleRetentionMinutes` must be above 0"});
164 }
165 config.limits.idleRetention = std::chrono::minutes{*limits.idleRetentionMinutes};
166 }
167 return config;
168}

170/// @brief Reads a chat's configuration from a JSON file. See `parseChatConfig`.
171/// @param path The file.
172/// @param environment How `apiKeyEnv` is looked up; the process's environment by default.
173/// @return The configuration, or why not.
174[[nodiscard]] inline std::expected<ChatConfig, std::string> loadChatConfig(
175 std::filesystem::path const& path, EnvironmentLookup const& environment = detail::processEnvironment) {
176 std::ifstream const file{path};
177 if (!file) {
178 return std::unexpected("the chat configuration file `" + path.string() + "` cannot be opened");
179 }
180 std::ostringstream text;
181 text << file.rdbuf();
182 return parseChatConfig(text.str(), environment);
183}
