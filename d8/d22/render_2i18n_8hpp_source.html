5/// @file render/i18n.hpp
6/// @brief The renderer-side translation-catalog seam over morph's derived
7/// message keys (`morph::forms::i18n`, `forms/i18n.hpp`).
8///
9/// `morph::render` is client-side only and never appears on the wire — it is
10/// the namespace the planned per-field widget-override registry
11/// (`gui_renderer_toolkit.md`'s `SlotRegistry`) will eventually share. morph
12/// ships this seam and the resolution algorithm below; it defines **no**
13/// translation storage format. A host adapts whatever catalog it already
14/// owns (Qt `QTranslator`/`.qm`, a JSON bundle, a database) into the one
15/// `TranslationProvider` signature.

24/// @brief A host-supplied catalog lookup: `key`/`bcp47Locale` -> translated
25/// text, or `std::nullopt` on a miss.
26///
27/// A default-constructed (empty) `TranslationProvider` means "no catalog
28/// installed" — `resolveText` treats it the same as a provider that misses
29/// on every key.
30using TranslationProvider =
31 std::function<std::optional<std::string>(std::string_view key, std::string_view bcp47Locale)>;
32
33/// @brief Resolves one display slot's text: explicit key, then derived key,
34/// then the schema literal.
35///
36/// Tried in order, most specific first:
37/// 1. @p explicitKey (a host-declared `x-i18nKey`-derived key), if present;
38/// 2. @p derivedKey (`morph::forms::i18n::fieldKey` or one of its siblings);
39/// 3. a miss at both falls back to @p schemaLiteral — the schema's
40/// authored `title` / `description` / `x-placeholder` / group or step
41/// title, unchanged.
42/// An empty/unset @p provider ("no catalog installed") skips straight to
43/// @p schemaLiteral, matching an unconfigured renderer's behavior exactly.
44/// @param provider The host's catalog lookup.
45/// @param bcp47Locale The locale to resolve against (e.g. `"fr-FR"`).
46/// @param explicitKey The field/group/rule/step's declared `x-i18nKey`-based
47/// key, or `std::nullopt` when none is declared.
48/// @param derivedKey The mechanically-derived key for this slot.
49/// @param schemaLiteral The schema's authored fallback text for this slot.
50/// @return The resolved display text.
51[[nodiscard]] inline std::string resolveText(const TranslationProvider& provider, std::string_view bcp47Locale,
52 const std::optional<std::string>& explicitKey,
53 std::string_view derivedKey, std::string_view schemaLiteral) {
54 if (provider) {
55 if (explicitKey.has_value()) {
56 if (auto hit = provider(*explicitKey, bcp47Locale)) {
57 return *hit;
58 }
59 }
60 if (auto hit = provider(derivedKey, bcp47Locale)) {
61 return *hit;
62 }
63 }
64 return std::string{schemaLiteral};
65}
