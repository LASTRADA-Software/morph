46/// @brief How the conformance cases read and drive one backend under test.
47///
48/// The driving operations act as a user would, through the backend's own input path where it has one; each may run
49/// pending work before and after it. Reading operations report what the widget shows now.
50///
51/// Whether a handler can run after its widget was destroyed is observable only for a backend whose input can be
52/// delivered later than the driving call, such as a queued connection; a probe whose `click` delivers before it
53/// returns passes that case without exercising it.
54class ConformanceProbe {
55public:
56 ConformanceProbe() = default;
57 virtual ~ConformanceProbe() = default;
58 ConformanceProbe(ConformanceProbe const&) = delete;
59 ConformanceProbe& operator=(ConformanceProbe const&) = delete;
60 ConformanceProbe(ConformanceProbe&&) = delete;
61 ConformanceProbe& operator=(ConformanceProbe&&) = delete;
62
63 /// @brief The backend under test.
64 /// @return The backend every case mounts on.
65 virtual IViewBackend& backend() = 0;
66
67 /// @brief The runtime the cases' signals and mounts belong to.
68 /// @return The runtime, whose owner `settle` drives.
69 virtual reactive::Runtime& runtime() = 0;
70
71 /// @brief Runs posted flushes and pending native events until nothing is left.
72 virtual void settle() = 0;
73
74 /// @brief The text a widget shows: a Text's or TextInput's text, a Button's or Checkbox's label, and the label of
75 /// the option a Select marks.
76 /// @param widget The widget.
77 /// @return The text, UTF-8; empty for a Select that marks no option.
78 [[nodiscard]] virtual std::string textOf(Widget const& widget) = 0;
79
80 /// @brief The widget's own visibility flag, as its last `setVisible` left it.
81 ///
82 /// Only the widget's own flag counts: a hidden ancestor, a collapsed panel or a closed dialog around it does
83 /// not make this false. That is what `RecordingBackend` records, and it lets a case tell which Tabs page slot
84 /// the mount hid.
85 /// @param widget The widget.
86 /// @return True for a new widget; false once `setVisible(false)` reached it and until `setVisible(true)` does.
87 [[nodiscard]] virtual bool visibleOf(Widget const& widget) = 0;
88
89 /// @brief Whether a widget accepts input, by its own flag: a disabled ancestor does not make this false, which
90 /// is what `RecordingBackend` records.
91 /// @param widget The widget.
92 /// @return True for a new widget; false once `setEnabled(false)` reached it and until `setEnabled(true)` does.
93 [[nodiscard]] virtual bool enabledOf(Widget const& widget) = 0;
94
95 /// @brief How many children a container has.
96 /// @param container The container.
97 /// @return The count.
98 [[nodiscard]] virtual std::size_t childCount(ContainerWidget const& container) = 0;

100 /// @brief A container's child.
101 /// @param container The container.
102 /// @param index The position.
103 /// @return The child, or null past the end. A case drives it as it drives a root.

105
106 /// @brief Activates a button as a user would.
107 /// @param widget The button.
108 virtual void click(Widget& widget) = 0;
109

113 /// @param text The text to end up with.

115
116 /// @brief Drags one widget onto another as a user would.
117 /// @param source The dragged widget.
118 /// @param target The widget it is dropped on.
119 virtual void drag(Widget& source, Widget& target) = 0;
120
121 /// @brief Dismisses a dialog or drawer as a user would (Esc on the TUI), or a dismissible banner.
122 /// @param dialog The dialog, drawer or banner.

192/// @brief Collects a case's checks and keeps the first failure, so a case reads top to bottom.
193class Checks {
194public:
195 /// @brief Checks that a text is what it should be.
196 /// @param what What is checked, for the message.
197 /// @param expected The text it should be.
198 /// @param actual The text it is.
199 void text(std::string_view what, std::string_view expected, std::string_view actual) {
200 if (expected != actual) {
201 fail(std::string{what} + ": expected '" + std::string{expected} + "', got '" + std::string{actual} + "'");
202 }
203 }
204
205 /// @brief Checks a condition.
206 /// @param holds The condition.
207 /// @param what The failure message when it does not hold.
208 void that(bool holds, std::string_view what) {
209 if (!holds) {
210 fail(std::string{what});
211 }
212 }
213
214 /// @brief The case's verdict.
215 /// @return `nullopt` when every check held, else the first failure.
216 [[nodiscard]] std::optional<std::string> result() const { return _failure; }
217
218private:
219 void fail(std::string message) {
220 if (!_failure.has_value()) {
221 _failure = std::move(message);
222 }
223 }
224
225 std::optional<std::string> _failure;
226};

240/// @brief A ForEach of Texts, one per entry, keyed by id.
241/// @param rows The entries; must outlive the node's mounts.
242/// @param builds Counts the row views built, when not null; must outlive the node's mounts.
243/// @return The node.
244[[nodiscard]] inline Node entryList(reactive::Signal<std::vector<Entry>> const& rows, int* builds = nullptr) {
245 return ui::forEach<Entry>(
246 rows, [](Entry const& entry) { return Key{entry.id}; },
247 [builds](reactive::Signal<Entry> const& entry) {
248 if (builds != nullptr) {
249 ++*builds;
250 }
251 return ui::text({.text = [&entry] { return entry.get().label; }});
252 });
253}

288/// @param probe The backend under test.

297/// @brief The texts of a container's children, joined by commas.
298/// @param probe The backend under test.
299/// @param container The container.
300/// @return For example `a,b,c`; a missing child reads `<null>`.
301[[nodiscard]] inline std::string childTexts(ConformanceProbe& probe, ContainerWidget const& container) {
302 std::string texts;
303 for (std::size_t i = 0; i < probe.childCount(container); ++i) {
304 if (i != 0) {
305 texts += ',';
306 }
307 Widget const* const child = probe.childAt(container, i);
308 texts += child == nullptr ? std::string{"<null>"} : probe.textOf(*child);
309 }
310 return texts;
311}

313/// @brief The texts of the rows an `entryTable` shows as selected, joined by commas.
314/// @param probe The backend under test.
315/// @param table The table.
316/// @return For example `a,c`; a row without a first cell reads `<null>`.
317[[nodiscard]] inline std::string selectedTexts(ConformanceProbe& probe, Widget const& table) {
318 std::string texts;
319 for (std::size_t const row : probe.selectedRows(table)) {
320 if (!texts.empty()) {
321 texts += ',';
322 }
323 Widget const* const cell = childOf(probe, childOf(probe, &table, row), 0);
324 texts += cell == nullptr ? std::string{"<null>"} : probe.textOf(*cell);
325 }
326 return texts;
327}

329/// @brief Keys as text, for a failure message.
330/// @param keys The keys.
331/// @return For example `1,"b"`.
332[[nodiscard]] inline std::string keyTexts(std::vector<Key> const& keys) {
333 std::string texts;

336 texts += ',';
