33/// @brief How the conformance cases read and drive one backend under test.
34///
35/// The driving operations act as a user would, through the backend's own input path where it has one; each may run
36/// pending work before and after it. Reading operations report what the widget shows now.
37///
38/// Whether a handler can run after its widget was destroyed is observable only for a backend whose input can be
39/// delivered later than the driving call, such as a queued connection; a probe whose `click` delivers before it
40/// returns passes that case without exercising it.
41class ConformanceProbe {
42public:
43 ConformanceProbe() = default;
44 virtual ~ConformanceProbe() = default;
45 ConformanceProbe(ConformanceProbe const&) = delete;
46 ConformanceProbe& operator=(ConformanceProbe const&) = delete;
47 ConformanceProbe(ConformanceProbe&&) = delete;
48 ConformanceProbe& operator=(ConformanceProbe&&) = delete;
49
50 /// @brief The backend under test.
51 /// @return The backend every case mounts on.
52 virtual IViewBackend& backend() = 0;
53
54 /// @brief The runtime the cases' signals and mounts belong to.
55 /// @return The runtime, whose owner `settle` drives.
56 virtual reactive::Runtime& runtime() = 0;
57
58 /// @brief Runs posted flushes and pending native events until nothing is left.
59 virtual void settle() = 0;
60
61 /// @brief The text a widget shows: a Text's or TextInput's text, a Button's or Checkbox's label, and the label of
62 /// the option a Select marks.
63 /// @param widget The widget.
64 /// @return The text, UTF-8; empty for a Select that marks no option.
65 [[nodiscard]] virtual std::string textOf(Widget const& widget) = 0;
66
67 /// @brief The widget's own visibility flag, as its last `setVisible` left it.
68 ///
69 /// Only the widget's own flag counts: a hidden ancestor, a collapsed panel or a closed dialog around it does
70 /// not make this false. That is what `RecordingBackend` records, and it lets a case tell which Tabs page slot
71 /// the mount hid.
72 /// @param widget The widget.
73 /// @return True for a new widget; false once `setVisible(false)` reached it and until `setVisible(true)` does.
74 [[nodiscard]] virtual bool visibleOf(Widget const& widget) = 0;
75
76 /// @brief Whether a widget accepts input, by its own flag: a disabled ancestor does not make this false, which
77 /// is what `RecordingBackend` records.
78 /// @param widget The widget.
79 /// @return True for a new widget; false once `setEnabled(false)` reached it and until `setEnabled(true)` does.
80 [[nodiscard]] virtual bool enabledOf(Widget const& widget) = 0;
81
82 /// @brief How many children a container has.
83 /// @param container The container.
84 /// @return The count.
85 [[nodiscard]] virtual std::size_t childCount(ContainerWidget const& container) = 0;
86
87 /// @brief A container's child.
88 /// @param container The container.
89 /// @param index The position.
90 /// @return The child, or null past the end. A case drives it as it drives a root.
91 [[nodiscard]] virtual Widget* childAt(ContainerWidget const& container, std::size_t index) = 0;
92
93 /// @brief Activates a button as a user would.
94 /// @param widget The button.
95 virtual void click(Widget& widget) = 0;
96
97 /// @brief Replaces a text field's text as a user typing would: whatever the field held is cleared first (select
98 /// all, or delete), so it ends up holding exactly @p text. `onChange` may see intermediate texts.
99 /// @param widget The field.
100 /// @param text The text to end up with.
101 virtual void type(Widget& widget, std::string_view text) = 0;
102
103 /// @brief Drags one widget onto another as a user would.
104 /// @param source The dragged widget.
105 /// @param target The widget it is dropped on.
106 virtual void drag(Widget& source, Widget& target) = 0;
107
108 /// @brief Dismisses a dialog as a user would (Esc on the TUI).
109 /// @param dialog The dialog.
110 virtual void dismiss(Widget& dialog) = 0;
111
112 /// @brief Selects rows of a table as a user would, so that the user's selection is exactly these rows.
113 /// @param table The table.
114 /// @param rows Positions among the table's current rows, in the order the user picks them: at least one, and
115 /// one only for a `Single` table.
116 virtual void selectRows(Widget& table, std::vector<std::size_t> const& rows) = 0;
117
118 /// @brief The rows a table shows as selected.
119 /// @param table The table.
120 /// @return Their positions among the table's current rows, ascending.
121 [[nodiscard]] virtual std::vector<std::size_t> selectedRows(Widget const& table) = 0;
122
123 /// @brief Moves a text field's cursor as a user would (the arrow keys, a click), without changing the text.
124 /// @param field The text field.
125 /// @param position A byte offset into the text it shows.
126 virtual void moveCursor(Widget& field, std::size_t position) = 0;
127
128 /// @brief Where a text field's cursor is.
129 /// @param field The text field.
130 /// @return A byte offset into the text it shows.
131 [[nodiscard]] virtual std::size_t cursorOf(Widget const& field) = 0;
132};

144/// @brief Collects a case's checks and keeps the first failure, so a case reads top to bottom.
145class Checks {
146public:
147 /// @brief Checks that a text is what it should be.
148 /// @param what What is checked, for the message.
149 /// @param expected The text it should be.
150 /// @param actual The text it is.
151 void text(std::string_view what, std::string_view expected, std::string_view actual) {
152 if (expected != actual) {
153 fail(std::string{what} + ": expected '" + std::string{expected} + "', got '" + std::string{actual} + "'");
154 }
155 }
156
157 /// @brief Checks a condition.
158 /// @param holds The condition.
159 /// @param what The failure message when it does not hold.
160 void that(bool holds, std::string_view what) {
161 if (!holds) {
162 fail(std::string{what});
163 }
164 }
165
166 /// @brief The case's verdict.
167 /// @return `nullopt` when every check held, else the first failure.
168 [[nodiscard]] std::optional<std::string> result() const { return _failure; }
169
170private:
171 void fail(std::string message) {
172 if (!_failure.has_value()) {
173 _failure = std::move(message);
174 }
175 }
176
177 std::optional<std::string> _failure;
178};

192/// @brief A ForEach of Texts, one per entry, keyed by id.
193/// @param rows The entries; must outlive the node's mounts.
194/// @param builds Counts the row views built, when not null; must outlive the node's mounts.
195/// @return The node.
196[[nodiscard]] inline Node entryList(reactive::Signal<std::vector<Entry>> const& rows, int* builds = nullptr) {
197 return ui::forEach<Entry>(
198 rows, [](Entry const& entry) { return Key{entry.id}; },
199 [builds](reactive::Signal<Entry> const& entry) {
200 if (builds != nullptr) {
201 ++*builds;
202 }
203 return ui::text({.text = [&entry] { return entry.get().label; }});
204 });
205}

227/// @brief The root widget of a mounted container node, as the container it is.
228///
229/// The root of a mounted Switch, Tabs, Column, ForEach or Dialog is the widget the backend's `createSlot`,
230/// `createTabs`, `createStack` or `createDialog` returned, and each of those is a `ContainerWidget` by its declared
231/// type, so the downcast cannot fail for any backend that compiles.
232/// @param view The mount, of a container node.
233/// @return The root widget as a container.
234[[nodiscard]] inline ContainerWidget const& rootContainer(Mounted const& view) {
235 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-static-cast-downcast): a container by the factory's type, see above
236 return static_cast<ContainerWidget const&>(view.root());
237}

239/// @brief A child of a widget that should be a container.
240/// @param probe The backend under test.
241/// @param parent The widget; null, or a widget that is no container, has no children.
242/// @param index The position.
243/// @return The child, or null.
244[[nodiscard]] inline Widget* childOf(ConformanceProbe& probe, Widget const* parent, std::size_t index) {
245 ContainerWidget const* const container = parent == nullptr ? nullptr : asContainer(*parent);
246 return container == nullptr ? nullptr : probe.childAt(*container, index);
247}

249/// @brief The texts of a container's children, joined by commas.
250/// @param probe The backend under test.
251/// @param container The container.
252/// @return For example `a,b,c`; a missing child reads `<null>`.
253[[nodiscard]] inline std::string childTexts(ConformanceProbe& probe, ContainerWidget const& container) {
254 std::string texts;
255 for (std::size_t i = 0; i < probe.childCount(container); ++i) {
256 if (i != 0) {
257 texts += ',';
258 }
259 Widget const* const child = probe.childAt(container, i);
260 texts += child == nullptr ? std::string{"<null>"} : probe.textOf(*child);
261 }
262 return texts;
263}

265/// @brief The texts of the rows an `entryTable` shows as selected, joined by commas.
266/// @param probe The backend under test.
267/// @param table The table.
268/// @return For example `a,c`; a row without a first cell reads `<null>`.
269[[nodiscard]] inline std::string selectedTexts(ConformanceProbe& probe, Widget const& table) {
270 std::string texts;
271 for (std::size_t const row : probe.selectedRows(table)) {
272 if (!texts.empty()) {
273 texts += ',';
274 }
275 Widget const* const cell = childOf(probe, childOf(probe, &table, row), 0);
276 texts += cell == nullptr ? std::string{"<null>"} : probe.textOf(*cell);
277 }
278 return texts;
279}

281/// @brief Keys as text, for a failure message.
282/// @param keys The keys.
283/// @return For example `1,"b"`.
284[[nodiscard]] inline std::string keyTexts(std::vector<Key> const& keys) {
285 std::string texts;
286 for (Key const& key : keys) {
287 if (!texts.empty()) {
288 texts += ',';
289 }
290 if (auto const* const number = std::get_if<std::int64_t>(&key); number != nullptr) {

293 texts += '"' + std::get<std::string>(key) + '"';
294 }
295 }
296 return texts;

311/// @param probe The backend under test.

1018/// @brief What `LifeSentinel` reports about the handler that holds it.
1019struct HandlerLife {
1020 /// @brief The sentinel inside the handler while that handler runs; null otherwise.
1021 void const* running = nullptr;

1023 bool freedWhileRunning = false;

1026/// @brief Captured by value in a handler, tells whether the backend destroyed the handler while it ran.
1027///
1028/// Every copy of the handler holds its own sentinel; only the destruction of the copy that is running counts.
1029class LifeSentinel {
1030public:
1031 /// @param life Where to report; must outlive every copy.
1032 explicit LifeSentinel(HandlerLife& life) noexcept : _life{&life} {}
1033
1034 /// @brief Reports its own destruction when it belongs to the running handler.
1035 ~LifeSentinel() {
1036 if (_life->running == this) {
1037 _life->freedWhileRunning = true;
1038 }
1039 }
1040
1041 /// @brief A copy, for another copy of the handler.
1042 LifeSentinel(LifeSentinel const&) = default;

1060/// @brief Case: a handler may destroy its own widget; the backend keeps the running handler alive until it returns,
1061/// and stays usable after.

1065 HandlerLife life;

1072 std::optional<Mounted>* const own = &view;
1073 ++clicks;
1074 report->running = &sentinel;
1075 own->reset();

1077 }}));
1078 probe.settle();
1079 probe.click(view->root());
1080 probe.settle();
1081 Checks checks;
1082 checks.that(clicks == 1,
1083 "the click that destroys its own button ran the action " + std::to_string(clicks) + " times");
