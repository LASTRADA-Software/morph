27/// @brief `SQLITE_TRANSIENT` (`((sqlite3_destructor_type)-1)` in `sqlite3.h`)

55/// @brief Reference SQLite-backed `IOfflineQueue`: persists `payload`,
56/// `idempotencyKey`, and `attempts` across process restarts.
57///
58/// Schema (one table, `morph_offline_queue`):
59///
60/// ```sql
61/// CREATE TABLE IF NOT EXISTS morph_offline_queue (
62/// id INTEGER PRIMARY KEY AUTOINCREMENT,
63/// payload TEXT NOT NULL,
64/// idempotency_key TEXT NOT NULL DEFAULT '',
65/// attempts INTEGER NOT NULL DEFAULT 0,
66/// enqueued_at INTEGER NOT NULL
67/// );
68/// CREATE UNIQUE INDEX IF NOT EXISTS ix_queue_idem
69/// ON morph_offline_queue(idempotency_key) WHERE idempotency_key <> '';
70/// ```
71///
72/// `id` is `AUTOINCREMENT`, so ids are never reused and a re-opened queue
73/// re-presents each row under its **stored** id — stable across restarts.
74/// `QueueItem::id` remains queue-local (per `docs/spec/offline/offline.md`);
75/// cross-restart identity is carried by `idempotencyKey`, not `id`.
76///
77/// The partial unique index gives insert-time dedup for a non-empty
78/// `idempotencyKey`: a re-enqueue of the same key is a no-op that returns the
79/// existing row's id (`INSERT ... ON CONFLICT ... DO NOTHING`, then a lookup
80/// on a no-op conflict). Empty keys (the default) are exempt, so keyless
81/// items behave exactly as `InMemoryOfflineQueue` does — never deduplicated.
82///
83/// @par Crash safety
84/// `drain()` never deletes, so a crash between `drain()` and `markDone()`
85/// loses nothing. Each write (`enqueue`, `markDone`, `setAttempts`,
86/// `setIdempotencyKey`) is its own committed statement; `PRAGMA
87/// journal_mode=WAL` (set once, at construction) gives the durability.
88///
89/// @par Thread safety
90/// All operations serialise on an internal mutex around the connection, so
91/// this is safe to share between the application's enqueue-on-failure write
92/// path and `SyncWorker`'s drain/replay read path.
93class SqliteOfflineQueue : public IOfflineQueue {
94public:
95 using IOfflineQueue::enqueue; // keep the two-arg overload visible
96
97 /// @brief Opens (or creates) the queue database at @p path, creating the
98 /// schema if it does not already exist.
99 /// @param path SQLite database file.
100 /// @param maxDepth Maximum number of pending rows `enqueue()` will admit
101 /// before throwing `OfflineQueueFullError`; `std::nullopt` (the
102 /// default) means unbounded. Not persisted in the database itself
103 /// — a per-construction parameter, so a reopen must pass it again
104 /// to keep the same cap enforced.
105 /// @throws SqliteOfflineQueueError if the database cannot be opened or
106 /// the schema cannot be created.
107 explicit SqliteOfflineQueue(std::filesystem::path path, std::optional<std::size_t> maxDepth = std::nullopt)
108 : _path{std::move(path)}, _maxDepth{maxDepth} {
109 if (sqlite3_open(_path.string().c_str(), &_db) != SQLITE_OK) {
110 std::string msg = "SqliteOfflineQueue: failed to open " + _path.string() + ": " +
111 (_db != nullptr ? sqlite3_errmsg(_db) : "unknown error");
112 if (_db != nullptr) {
113 sqlite3_close(_db);
114 _db = nullptr;
115 }
116 throw SqliteOfflineQueueError{msg};
117 }
118 execOrThrow("PRAGMA journal_mode=WAL;");
119 execOrThrow(
120 "CREATE TABLE IF NOT EXISTS morph_offline_queue ("
121 " id INTEGER PRIMARY KEY AUTOINCREMENT,"
122 " payload TEXT NOT NULL,"
123 " idempotency_key TEXT NOT NULL DEFAULT '',"
124 " attempts INTEGER NOT NULL DEFAULT 0,"
125 " enqueued_at INTEGER NOT NULL"
126 ");");
127 execOrThrow(
128 "CREATE UNIQUE INDEX IF NOT EXISTS ix_queue_idem "
129 "ON morph_offline_queue(idempotency_key) WHERE idempotency_key <> '';");
130 }
131
132 /// @brief Closes the underlying SQLite connection.
133 ~SqliteOfflineQueue() override {
134 if (_db != nullptr) {
135 sqlite3_close(_db);
136 }
137 }

144 /// @brief Inserts @p payload with an empty idempotency key.
145 /// @param payload Serialised action to persist.
146 /// @return The new row's id (`SELECT last_insert_rowid()`).
147 /// @throws OfflineQueueFullError if the queue is already at `maxDepth()`.
148 uint64_t enqueue(std::string payload) override {
149 std::scoped_lock const lock{_mtx};
150 checkCapacityLocked();
151 detail::StatementGuard guard{
152 prepare("INSERT INTO morph_offline_queue (payload, idempotency_key, attempts, enqueued_at) "
153 "VALUES (?, '', 0, ?);")};
154 bindText(guard.get(), 1, payload);
155 bindInt64(guard.get(), 2, nowMillis());
156 stepOrThrow(guard.get(), "enqueue");
157 return static_cast<uint64_t>(sqlite3_last_insert_rowid(_db));
158 }
159
160 /// @brief Inserts @p payload carrying @p idempotencyKey in one write. A
161 /// non-empty key already present on a row is deduplicated: the
162 /// existing row's id is returned and nothing new is inserted.
163 /// @param payload Serialised action to persist.
164 /// @param idempotencyKey Stable dedup token; empty means "no dedup".
165 /// @return The new row's id, or the existing row's id on a dedup hit.
166 /// @throws OfflineQueueFullError if the queue is already at `maxDepth()`.
167 /// Checked before the insert is attempted, so a call that would
168 /// have resolved to a dedup hit (inserting nothing) can also be
169 /// rejected when the queue happens to be full at the same time —
170 /// a documented, accepted conservatism rather than an extra
171 /// round trip to special-case it.
172 uint64_t enqueue(std::string payload, std::string idempotencyKey) override {
173 std::scoped_lock const lock{_mtx};
174 if (idempotencyKey.empty()) {
175 checkCapacityLocked();
176 detail::StatementGuard guard{
177 prepare("INSERT INTO morph_offline_queue (payload, idempotency_key, attempts, enqueued_at) "
178 "VALUES (?, '', 0, ?);")};
179 bindText(guard.get(), 1, payload);
180 bindInt64(guard.get(), 2, nowMillis());
181 stepOrThrow(guard.get(), "enqueue");
182 return static_cast<uint64_t>(sqlite3_last_insert_rowid(_db));
183 }
184
185 checkCapacityLocked();
186 detail::StatementGuard insertGuard{
187 prepare("INSERT INTO morph_offline_queue (payload, idempotency_key, attempts, enqueued_at) "
188 "VALUES (?, ?, 0, ?) "
189 "ON CONFLICT(idempotency_key) WHERE idempotency_key <> '' DO NOTHING;")};
190 bindText(insertGuard.get(), 1, payload);
191 bindText(insertGuard.get(), 2, idempotencyKey);
192 bindInt64(insertGuard.get(), 3, nowMillis());
193 stepOrThrow(insertGuard.get(), "enqueue");
194 if (sqlite3_changes(_db) > 0) {
195 return static_cast<uint64_t>(sqlite3_last_insert_rowid(_db));
196 }
197
198 // Conflict fired (DO NOTHING) -- a row for this key already exists.
199 detail::StatementGuard lookupGuard{prepare("SELECT id FROM morph_offline_queue WHERE idempotency_key = ?;")};
200 bindText(lookupGuard.get(), 1, idempotencyKey);
201 int const lookupResult = sqlite3_step(lookupGuard.get());
202 if (lookupResult != SQLITE_ROW) {
203 // The conflict proved a row exists, so anything but a row here is a
204 // real failure. Returning the `0` this used to fall through with
205 // would hand the caller an id that matches nothing: `markDone(0)`
206 // silently deletes no row, and the item is stranded in the queue
207 // forever with no error ever reported.
208 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: enqueue could not resolve the existing id "
209 "for a deduplicated idempotency key: "} +
210 sqlite3_errmsg(_db)};
211 }
212 return static_cast<uint64_t>(sqlite3_column_int64(lookupGuard.get(), 0));
213 }
214
215 /// @brief Returns all pending rows in ascending-id (enqueue) order.
216 /// @return Snapshot of all pending items; the table is unchanged.
217 std::vector<QueueItem> drain() const override {
218 std::scoped_lock const lock{_mtx};
219 detail::StatementGuard guard{
220 prepare("SELECT id, payload, idempotency_key, attempts FROM morph_offline_queue ORDER BY id;")};
221 std::vector<QueueItem> out;
222 for (;;) {
223 int const stepResult = sqlite3_step(guard.get());
224 if (stepResult == SQLITE_DONE) {
225 break;
226 }
227 if (stepResult != SQLITE_ROW) {
228 // `while (step() == SQLITE_ROW)` treated an I/O error, a corrupt
229 // page, or SQLITE_BUSY as "no more rows", so drain() returned a
230 // silently truncated set that the caller takes for the complete
231 // list of pending work -- and SyncWorker then reports a clean
232 // pass over a queue it never fully read.
233 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: drain failed part-way through after "} +
234 std::to_string(out.size()) + " row(s): " + sqlite3_errmsg(_db)};
235 }
236 QueueItem item;
237 item.id = static_cast<uint64_t>(sqlite3_column_int64(guard.get(), 0));
238 item.payload = textColumn(guard.get(), 1);
239 item.idempotencyKey = textColumn(guard.get(), 2);
240 item.attempts = static_cast<uint32_t>(sqlite3_column_int64(guard.get(), 3));
241 out.push_back(std::move(item));
242 }
243 return out;
244 }

246 /// @brief Deletes the row identified by @p itemId. No-op if absent.
247 /// @param itemId Id returned by the corresponding `enqueue()` call.
248 void markDone(uint64_t itemId) override {
249 std::scoped_lock const lock{_mtx};
250 detail::StatementGuard guard{prepare("DELETE FROM morph_offline_queue WHERE id = ?;")};
251 bindInt64(guard.get(), 1, static_cast<std::int64_t>(itemId));
252 stepOrThrow(guard.get(), "markDone");
253 }

255 /// @brief Persists an updated attempt count for @p itemId. No-op if absent.
256 /// @param itemId Id of the item whose count changed.
257 /// @param attempts New cumulative attempt count to store.
258 void setAttempts(uint64_t itemId, uint32_t attempts) override {
259 std::scoped_lock const lock{_mtx};
260 detail::StatementGuard guard{prepare("UPDATE morph_offline_queue SET attempts = ? WHERE id = ?;")};
261 bindInt64(guard.get(), 1, static_cast<std::int64_t>(attempts));
262 bindInt64(guard.get(), 2, static_cast<std::int64_t>(itemId));
263 stepOrThrow(guard.get(), "setAttempts");
264 }

274 /// @return The capacity `enqueue()` enforces, or `std::nullopt` if none.
275 std::optional<std::size_t> maxDepth() const override { return _maxDepth; }

277protected:
278 /// @brief Stamps an idempotency key onto an already-inserted row. No-op if
279 /// @p itemId is absent. Reachable only if a caller invokes the base
280 /// `IOfflineQueue::enqueue(payload, key)` default through an
281 /// `IOfflineQueue&` -- this class's own `enqueue(payload, key)`
282 /// override above stamps the key inline in the same INSERT instead.
283 /// @param itemId Id of the row to stamp.
284 /// @param idempotencyKey Key to store.
285 void setIdempotencyKey(uint64_t itemId, std::string idempotencyKey) override {
286 std::scoped_lock const lock{_mtx};
287 detail::StatementGuard guard{prepare("UPDATE morph_offline_queue SET idempotency_key = ? WHERE id = ?;")};
288 bindText(guard.get(), 1, idempotencyKey);
289 bindInt64(guard.get(), 2, static_cast<std::int64_t>(itemId));
290 stepOrThrow(guard.get(), "setIdempotencyKey");
291 }

293private:
294 void execOrThrow(const char* sql) {
295 char* errMsg = nullptr;
296 if (sqlite3_exec(_db, sql, nullptr, nullptr, &errMsg) != SQLITE_OK) {
297 std::string msg = errMsg != nullptr ? errMsg : "unknown sqlite error";
298 sqlite3_free(errMsg);
299 throw SqliteOfflineQueueError{"SqliteOfflineQueue: " + msg};
300 }
301 }
302
303 sqlite3_stmt* prepare(const char* sql) const {
304 sqlite3_stmt* stmt = nullptr;
305 if (sqlite3_prepare_v2(_db, sql, -1, &stmt, nullptr) != SQLITE_OK) {
306 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: prepare failed: "} + sqlite3_errmsg(_db)};
307 }
308 return stmt;
309 }
310
311 void bindText(sqlite3_stmt* stmt, int index, const std::string& value) const {
312 if (sqlite3_bind_text(stmt, index, value.c_str(), -1, detail::kSqliteTransient) != SQLITE_OK) {
313 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: bind failed: "} + sqlite3_errmsg(_db)};
314 }
315 }
316
317 void bindInt64(sqlite3_stmt* stmt, int index, std::int64_t value) const {
318 if (sqlite3_bind_int64(stmt, index, value) != SQLITE_OK) {
319 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: bind failed: "} + sqlite3_errmsg(_db)};
320 }
321 }
322
323 void stepOrThrow(sqlite3_stmt* stmt, const char* what) const {
324 // A busy/error code is treated the same as reaching the end -- a
325 // production consumer wanting to distinguish SQLITE_BUSY should retry
326 // instead, but a single in-process mutex around the whole connection
327 // makes SQLITE_BUSY practically unreachable for this reference queue.
328 if (sqlite3_step(stmt) != SQLITE_DONE) {
329 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: "} + what +
330 " failed: " + sqlite3_errmsg(_db)};
331 }
332 }
333
334 static std::string textColumn(sqlite3_stmt* stmt, int index) {
335 const auto* text = reinterpret_cast<const char*>(sqlite3_column_text(stmt, index));
336 return text != nullptr ? std::string{text} : std::string{};
337 }
338
339 static std::int64_t nowMillis() {
340 return std::chrono::duration_cast<std::chrono::milliseconds>(
341 std::chrono::system_clock::now().time_since_epoch())
342 .count();
343 }

345 /// @brief Returns the current row count. Caller must hold `_mtx`.
346 /// @return `COUNT(*)` against `morph_offline_queue`.
347 std::size_t countLocked() const {
348 detail::StatementGuard guard{prepare("SELECT COUNT(*) FROM morph_offline_queue;")};
349 sqlite3_step(guard.get());
350 return static_cast<std::size_t>(sqlite3_column_int64(guard.get(), 0));
351 }
352
353 /// @brief Throws `OfflineQueueFullError` if the queue is already at
354 /// `maxDepth()`. Caller must hold `_mtx`. No-op if unbounded.
355 void checkCapacityLocked() const {
356 if (!_maxDepth) {
357 return;
358 }
359 std::size_t const current = countLocked();
360 if (current >= *_maxDepth) {
361 ::morph::observe::detail::emitMetric(::morph::observe::Metric::queueOverflow,
362 static_cast<double>(current));
363 throw OfflineQueueFullError{*_maxDepth, current};
364 }
365 }
366
367 std::filesystem::path _path;
368 mutable sqlite3* _db = nullptr;
369 mutable std::mutex _mtx;
370 std::optional<std::size_t> _maxDepth;
371};
