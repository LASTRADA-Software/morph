39/// @brief RAII wrapper that finalizes a `sqlite3_stmt*` on scope exit,
40/// including when an exception unwinds past it.

42public:
43 /// @brief Takes over finalization of @p stmt.
44 /// @param stmt Statement to finalize on destruction. Borrowed for the
45 /// guard's whole lifetime — it is `sqlite3_finalize`d, not
46 /// copied, so it must stay valid until the guard is destroyed.

48 ~StatementGuard() { sqlite3_finalize(_stmt); }

51 StatementGuard& operator=(const StatementGuard&) = delete;
52 StatementGuard(StatementGuard&&) = delete;
53 StatementGuard& operator=(StatementGuard&&) = delete;

63/// @brief Reference SQLite-backed `IOfflineQueue`: persists `payload`,
64/// `idempotencyKey`, and `attempts` across process restarts.
65///
66/// Schema (one table, `morph_offline_queue`):
67///
68/// ```sql
69/// CREATE TABLE IF NOT EXISTS morph_offline_queue (
70/// id INTEGER PRIMARY KEY AUTOINCREMENT,
71/// payload TEXT NOT NULL,
72/// idempotency_key TEXT NOT NULL DEFAULT '',
73/// attempts INTEGER NOT NULL DEFAULT 0,
74/// enqueued_at INTEGER NOT NULL
75/// );
76/// CREATE UNIQUE INDEX IF NOT EXISTS ix_queue_idem
77/// ON morph_offline_queue(idempotency_key) WHERE idempotency_key <> '';
78/// ```
79///
80/// `id` is `AUTOINCREMENT`, so ids are never reused and a re-opened queue
81/// re-presents each row under its **stored** id — stable across restarts.
82/// `QueueItem::id` remains queue-local (per `docs/spec/offline/offline.md`);
83/// cross-restart identity is carried by `idempotencyKey`, not `id`.
84///
85/// The partial unique index gives insert-time dedup for a non-empty
86/// `idempotencyKey`: a re-enqueue of the same key is a no-op that returns the
87/// existing row's id (`INSERT ... ON CONFLICT ... DO NOTHING`, then a lookup
88/// on a no-op conflict). Empty keys (the default) are exempt, so keyless
89/// items behave exactly as `InMemoryOfflineQueue` does — never deduplicated.
90///
91/// @par Crash safety
92/// `drain()` never deletes, so a crash between `drain()` and `markDone()`
93/// loses nothing. Each write (`enqueue`, `markDone`, `setAttempts`,
94/// `setIdempotencyKey`) is its own committed statement; `PRAGMA
95/// journal_mode=WAL` and `PRAGMA synchronous=FULL` (both set once, at
96/// construction, and `journal_mode` read back and verified rather than
97/// trusted -- some filesystems, e.g. NFS, silently fall back to `delete`
98/// mode) give the durability. `PRAGMA busy_timeout` is also set at
99/// construction: this class's own mutex makes `SQLITE_BUSY` unreachable for
100/// a single instance, but the class documents no single-opener restriction.
101/// Construction also fsyncs the containing directory once, after
102/// `sqlite3_open()` and every schema-setup `PRAGMA`/`CREATE` succeed, closing
103/// the same fresh-file-creation directory-durability gap `FileIoOps::syncPath`
104/// closes for `FileActionLog`/`FileOfflineQueue`.
105///
106/// @par Thread safety
107/// All operations serialise on an internal mutex around the connection, so
108/// this is safe to share between the application's enqueue-on-failure write
109/// path and `SyncWorker`'s drain/replay read path.
110class SqliteOfflineQueue : public IOfflineQueue {
111public:
112 using IOfflineQueue::enqueue; // keep the two-arg overload visible
113
114 /// @brief `PRAGMA busy_timeout` set at construction (morph#532): how long
115 /// a statement blocks on `SQLITE_BUSY` before giving up, in
116 /// milliseconds.
117 static constexpr int kBusyTimeoutMillis = 5000;
118
119 /// @brief How hard each committed statement is pushed onto the platter.
120 enum class Synchronous : std::uint8_t {
121 /// @brief `PRAGMA synchronous=NORMAL` — the default, and SQLite's own
122 /// recommendation under WAL. A crash or power loss can lose the
123 /// most recent commits, which this queue's at-least-once
124 /// delivery plus `idempotencyKey` dedup already absorb.
125 normal,
126 /// @brief `PRAGMA synchronous=FULL` — fsync on every committed
127 /// statement. Measured ~18x slower per mutation (~0.08ms to
128 /// ~1.44ms on NVMe/btrfs, sqlite 3.53.4), and every mutation here
129 /// is its own commit, so a 200-item `SyncWorker` drain goes from
130 /// ~16ms to ~290ms — all of it under this class's mutex, where it
131 /// blocks the producer's `enqueue()` too. Opt in when losing the
132 /// last few commits is genuinely unacceptable.
133 full,
134 };
135
136 /// @brief SQLite's numeric `PRAGMA synchronous` level for this connection,
137 /// as read back at construction.
138 ///
139 /// `0` OFF, `1` NORMAL, `2` FULL, `3` EXTRA. Reported rather than inferred
140 /// from the constructor argument: the argument is what was asked for, this
141 /// is what SQLite confirmed.
142 /// @return The level in force on this queue's own connection.
143 [[nodiscard]] int synchronousLevel() const noexcept { return _synchronousLevel; }
144
145 /// @brief The `journal_mode` this database actually ended up in.
146 ///
147 /// Normally `"wal"`. A filesystem without the shared-memory support WAL
148 /// needs reports `"delete"`; `:memory:` and the temp spellings report
149 /// `"memory"`. Construction warns rather than throws on anything but
150 /// `"wal"` (durability is carried by `PRAGMA synchronous`), so this is how
151 /// a caller that cares can tell which mode it got.
152 /// @return The mode string as SQLite reported it at construction.
153 [[nodiscard]] const std::string& journalMode() const noexcept { return _journalMode; }
154
155 /// @brief Opens (or creates) the queue database at @p path, creating the
156 /// schema if it does not already exist.
157 /// @param path SQLite database file.
158 /// @param maxDepth Maximum number of pending rows `enqueue()` will admit
159 /// before throwing `OfflineQueueFullError`; `std::nullopt` (the
160 /// default) means unbounded. Not persisted in the database itself
161 /// — a per-construction parameter, so a reopen must pass it again
162 /// to keep the same cap enforced.
163 /// @param ioOps Injectable directory-fsync primitive (`FileIoOps::syncPath`
164 /// only -- the rest of this class talks to SQLite directly, never
165 /// through `FileIoOps`); defaults to the real syscall. Test-only
166 /// seam, mirroring `FileActionLog`/`FileOfflineQueue`'s own
167 /// `FileIoOps` parameter, for forcing the directory-fsync failure
168 /// branch below without needing a real OS-level failure.
169 /// @param synchronous `PRAGMA synchronous` level; see `Synchronous`.
170 /// Defaults to `normal`.
171 /// @param busyTimeout How long a statement waits out a lock another
172 /// connection holds before returning `SQLITE_BUSY`. Defaults to
173 /// `kBusyTimeoutMillis`. This class's own mutex makes `SQLITE_BUSY`
174 /// unreachable for a single instance, so the timeout only matters
175 /// when something else has the database open — but when it does, the
176 /// wait happens under that mutex and blocks every other caller of
177 /// this instance, a Qt GUI thread included. Pass `0` to restore
178 /// fail-fast.
179 /// @throws SqliteOfflineQueueError if the database cannot be opened, the
180 /// schema cannot be created, or the containing directory cannot
181 /// be fsynced after `sqlite3_open()` creates a brand-new file. A
182 /// `journal_mode` that does not end up as `wal` is **warned about,
183 /// not thrown** — see the constructor body.
184 explicit SqliteOfflineQueue(std::filesystem::path path, std::optional<std::size_t> maxDepth = std::nullopt,
185 ::morph::core::FileIoOps ioOps = {}, Synchronous synchronous = Synchronous::normal,
186 std::chrono::milliseconds busyTimeout = std::chrono::milliseconds{kBusyTimeoutMillis})
187 : _path{std::move(path)},
188 _maxDepth{maxDepth},
189 _io{std::move(ioOps)},
190 _synchronous{synchronous},
191 _busyTimeout{busyTimeout} {
192 if (sqlite3_open(_path.string().c_str(), &_db) != SQLITE_OK) {
193 std::string msg = "SqliteOfflineQueue: failed to open " + _path.string() + ": " +
194 (_db != nullptr ? sqlite3_errmsg(_db) : "unknown error");
195 if (_db != nullptr) {
196 sqlite3_close(_db);
197 _db = nullptr;
198 }
199 throw SqliteOfflineQueueError{msg};
200 }
201 // _db is a live, open connection from here on -- if any schema-setup
202 // statement below throws, this constructor never completes, so
203 // ~SqliteOfflineQueue() never runs to close it. Close it here before
204 // rethrowing, the same discipline the open-failure branch above
205 // already applies to its own failure path.
206 try {
207 // FIRST, before anything that can return SQLITE_BUSY. Converting a
208 // database to WAL needs an exclusive lock, so with a second
209 // connection open `PRAGMA journal_mode=WAL` is itself a BUSY
210 // candidate -- issued after the timeout it waits, issued before it
211 // fails instantly with "database is locked" (measured: 12ms to
212 // throw without, a full 1001ms wait with a 1000ms timeout set
213 // first). Setting it last, as an earlier revision did, left the
214 // multi-opener case it was added for failing exactly as before.
215 execOrThrow(("PRAGMA busy_timeout=" + std::to_string(_busyTimeout.count()) + ";").c_str());
216
217 // Before journal_mode, and unconditional. SQLite's default for the
218 // rollback journal is already FULL; it is WAL that lowers it to
219 // NORMAL, so raising it here first means the setting holds whether
220 // or not WAL takes. Ordered the other way round, a database that
221 // fell back to the rollback journal kept whatever synchronous it
222 // happened to have.
223 execOrThrow(
224 ("PRAGMA synchronous=" + std::string{_synchronous == Synchronous::full ? "FULL" : "NORMAL"} + ";")
225 .c_str());
226 // Read back for the same reason journal_mode is: sqlite3_exec
227 // discards the row a PRAGMA returns, so a setting that did not take
228 // would otherwise be invisible. `synchronous` is a *connection*
229 // property, not a database one, so this is also the only way a
230 // caller can confirm the level it asked for is the level in force.
231 {
232 detail::StatementGuard const guard{prepare("PRAGMA synchronous;")};
233 if (sqlite3_step(guard.get()) != SQLITE_ROW) {
234 throw SqliteOfflineQueueError{"SqliteOfflineQueue: failed to read back synchronous"};
235 }
236 _synchronousLevel = sqlite3_column_int(guard.get(), 0);
237 }
238
239 execOrThrow("PRAGMA journal_mode=WAL;");
240 // execOrThrow() discards sqlite3_exec's row callback, so a silent
241 // fallback would otherwise go unnoticed (morph#532). Read the
242 // pragma back through a real prepared statement rather than
243 // trusting the set.
244 //
245 // Warn, do not throw. An earlier revision made a non-`wal` result a
246 // hard construction failure, which refused configurations that were
247 // never actually less durable: `journal_mode` reports `memory` for
248 // `:memory:` and the temp/"" spellings, and `delete` on filesystems
249 // without the shared-memory WAL needs (NFS, CIFS/SMB, some overlay,
250 // 9p and Docker mounts). All of those persist correctly through the
251 // rollback journal -- whose durability is exactly what the
252 // `synchronous` pragma above now guarantees regardless of mode. The
253 // real call sites are not hypothetical either: kanban's
254 // `enableOfflineQueue()` builds one of these from a user-supplied
255 // path, so a throw here means an NFS home directory cannot open the
256 // app at all.
257 {
258 detail::StatementGuard const guard{prepare("PRAGMA journal_mode;")};
259 if (sqlite3_step(guard.get()) != SQLITE_ROW) {
260 throw SqliteOfflineQueueError{"SqliteOfflineQueue: failed to read back journal_mode"};
261 }
262 _journalMode = textColumn(guard.get(), 0);
263 if (_journalMode != "wal") {
264 ::morph::log::logWarn(
265 "SqliteOfflineQueue: PRAGMA journal_mode=WAL did not take for {} (got '{}'); continuing on "
266 "that mode -- durability is carried by PRAGMA synchronous, but concurrent readers and the "
267 "writer will not overlap as they do under WAL",
268 _path.string(), _journalMode);
269 }
270 }
271 execOrThrow(
272 "CREATE TABLE IF NOT EXISTS morph_offline_queue ("
273 " id INTEGER PRIMARY KEY AUTOINCREMENT,"
274 " payload TEXT NOT NULL,"
275 " idempotency_key TEXT NOT NULL DEFAULT '',"
276 " attempts INTEGER NOT NULL DEFAULT 0,"
277 " enqueued_at INTEGER NOT NULL"
278 ");");
279 execOrThrow(
280 "CREATE UNIQUE INDEX IF NOT EXISTS ix_queue_idem "
281 "ON morph_offline_queue(idempotency_key) WHERE idempotency_key <> '';");
282 // sqlite3_open() above creates `_path` (and, once journal_mode=WAL
283 // took, its "-wal"/"-shm" siblings) if it did not already exist --
284 // a fresh directory entry that SQLite's own internal fsyncs of the
285 // *file's* contents never make durable, the identical
286 // directory-vs-file-fsync gap morph#532 closed for
287 // `FileActionLog`/`FileOfflineQueue` (see `FileIoOps::syncPath`'s
288 // own docs). Unconditional: harmless when the file already
289 // existed, since syncing an unchanged directory is a cheap no-op.
290 //
291 // Which directory, though, is SQLite's to answer rather than
292 // `_path`'s. `:memory:`, `""` and the `file::memory:` URI spellings
293 // open no file at all, and `parent_path()` is empty for them --
294 // which `FileIoOps::syncPath` resolves to `"."`, so deriving the
295 // directory from `_path` meant fsyncing the *process's current
296 // working directory* and reporting the result as this queue's: a
297 // warning naming a database that is not on disk where a CWD cannot
298 // be fsynced, and a refusal to construct an in-memory queue at all
299 // where that fsync genuinely fails. `sqlite3_db_filename` reports
300 // an empty name for exactly those spellings, so it distinguishes
301 // "no backing file, nothing to sync" from a real path without this
302 // class having to re-parse SQLite's own filename grammar.
303 const char* const backingFile = sqlite3_db_filename(_db, "main");
304 if (backingFile != nullptr && *backingFile != '\0') {
305 std::filesystem::path const backingPath{backingFile};
306 auto const dirSync = ::morph::core::classifyDirectorySync(_io.syncPath(backingPath.parent_path()));
307 if (dirSync == ::morph::core::DirectorySync::failed) {
308 throw SqliteOfflineQueueError{"SqliteOfflineQueue: failed to fsync directory after creating " +
309 backingPath.string()};
310 }
311 if (dirSync == ::morph::core::DirectorySync::unsupported) {
312 ::morph::log::logWarn(
313 "SqliteOfflineQueue: cannot fsync the directory containing {}; SQLite's own fsyncs still "
314 "cover the database contents, but its directory entry is only as durable as this filesystem "
315 "makes it",
316 backingPath.string());
317 }
318 }
319 } catch (...) {
320 sqlite3_close(_db);
321 _db = nullptr;
322 throw;
323 }
324 }

338 /// @brief Inserts @p payload with an empty idempotency key.
339 /// @param payload Serialised action to persist.
340 /// @return The new row's id (`SELECT last_insert_rowid()`).
341 /// @throws OfflineQueueFullError if the queue is already at `maxDepth()`.
342 [[nodiscard]] uint64_t enqueue(std::string payload) override {
343 std::scoped_lock const lock{_mtx};
344 checkCapacityLocked();
345 detail::StatementGuard guard{
346 prepare("INSERT INTO morph_offline_queue (payload, idempotency_key, attempts, enqueued_at) "
347 "VALUES (?, '', 0, ?);")};
348 bindText(guard.get(), 1, payload);
349 bindInt64(guard.get(), 2, nowMillis());
350 stepOrThrow(guard.get(), "enqueue");
351 return static_cast<uint64_t>(sqlite3_last_insert_rowid(_db));
352 }
353
354 /// @brief Inserts @p payload carrying @p idempotencyKey in one write. A
355 /// non-empty key already present on a row is deduplicated: the
356 /// existing row's id is returned and nothing new is inserted.
357 /// @param payload Serialised action to persist.
358 /// @param idempotencyKey Stable dedup token; empty means "no dedup".
359 /// @return The new row's id, or the existing row's id on a dedup hit.
360 /// @throws OfflineQueueFullError if the queue is already at `maxDepth()`.
361 /// Checked before the insert is attempted, so a call that would
362 /// have resolved to a dedup hit (inserting nothing) can also be
363 /// rejected when the queue happens to be full at the same time —
364 /// a documented, accepted conservatism rather than an extra
365 /// round trip to special-case it.
366 [[nodiscard]] uint64_t enqueue(std::string payload, std::string idempotencyKey) override {
367 std::scoped_lock const lock{_mtx};
368 if (idempotencyKey.empty()) {
369 checkCapacityLocked();
370 detail::StatementGuard guard{
371 prepare("INSERT INTO morph_offline_queue (payload, idempotency_key, attempts, enqueued_at) "
372 "VALUES (?, '', 0, ?);")};
373 bindText(guard.get(), 1, payload);
374 bindInt64(guard.get(), 2, nowMillis());
375 stepOrThrow(guard.get(), "enqueue");
376 return static_cast<uint64_t>(sqlite3_last_insert_rowid(_db));
377 }
378
379 checkCapacityLocked();
380 detail::StatementGuard insertGuard{
381 prepare("INSERT INTO morph_offline_queue (payload, idempotency_key, attempts, enqueued_at) "
382 "VALUES (?, ?, 0, ?) "
383 "ON CONFLICT(idempotency_key) WHERE idempotency_key <> '' DO NOTHING;")};
384 bindText(insertGuard.get(), 1, payload);
385 bindText(insertGuard.get(), 2, idempotencyKey);
386 bindInt64(insertGuard.get(), 3, nowMillis());
387 stepOrThrow(insertGuard.get(), "enqueue");
388 if (sqlite3_changes(_db) > 0) {
389 return static_cast<uint64_t>(sqlite3_last_insert_rowid(_db));
390 }
391
392 // Conflict fired (DO NOTHING) -- a row for this key already exists.
393 detail::StatementGuard lookupGuard{prepare("SELECT id FROM morph_offline_queue WHERE idempotency_key = ?;")};
394 bindText(lookupGuard.get(), 1, idempotencyKey);
395 int const lookupResult = sqlite3_step(lookupGuard.get());
396 if (lookupResult != SQLITE_ROW) {
397 // The conflict proved a row exists, so anything but a row here is a
398 // real failure. Returning the `0` this used to fall through with
399 // would hand the caller an id that matches nothing: `markDone(0)`
400 // silently deletes no row, and the item is stranded in the queue
401 // forever with no error ever reported.
402 //
403 // COVERAGE GAP, documented not closed: reaching this branch needs
404 // a second writer on the same underlying file to delete this
405 // exact row in the narrow window between the INSERT above and
406 // this SELECT -- a genuine cross-connection race, not reachable
407 // through this class's own single-instance mutex or its own
408 // single-connection API at all. Unlike the drain()/prepare()/
409 // stepOrThrow() gaps closed elsewhere in this file via a
410 // second-connection helper, this window opens and closes within
411 // one function call on one thread, with no existing seam to pause
412 // mid-statement and no SQLite-level hook (commit hook / custom
413 // VFS) in this codebase to force it deterministically. A tight
414 // busy-loop second thread could occasionally win the race, but
415 // that is a fragile, non-deterministic test for one low-value
416 // branch arm -- left open and documented rather than forced.
417 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: enqueue could not resolve the existing id "
418 "for a deduplicated idempotency key: "} +
419 sqlite3_errmsg(_db)};
420 }
421 return static_cast<uint64_t>(sqlite3_column_int64(lookupGuard.get(), 0));
422 }

424 /// @brief Returns all pending rows in ascending-id (enqueue) order.
425 /// @return Snapshot of all pending items; the table is unchanged.
426 [[nodiscard]] std::vector<QueueItem> drain() const override {
427 std::scoped_lock const lock{_mtx};
428 detail::StatementGuard guard{
429 prepare("SELECT id, payload, idempotency_key, attempts FROM morph_offline_queue ORDER BY id;")};
430 std::vector<QueueItem> out;
431 for (;;) {
432 int const stepResult = sqlite3_step(guard.get());
433 if (stepResult == SQLITE_DONE) {
434 break;
435 }
436 if (stepResult != SQLITE_ROW) {
437 // `while (step() == SQLITE_ROW)` treated an I/O error, a corrupt
438 // page, or SQLITE_BUSY as "no more rows", so drain() returned a
439 // silently truncated set that the caller takes for the complete
440 // list of pending work -- and SyncWorker then reports a clean
441 // pass over a queue it never fully read.
442 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: drain failed part-way through after "} +
443 std::to_string(out.size()) + " row(s): " + sqlite3_errmsg(_db)};
444 }
445 QueueItem item;
446 item.id = static_cast<uint64_t>(sqlite3_column_int64(guard.get(), 0));
447 item.payload = textColumn(guard.get(), 1);
448 item.idempotencyKey = textColumn(guard.get(), 2);
449 item.attempts = static_cast<uint32_t>(sqlite3_column_int64(guard.get(), 3));
450 out.push_back(std::move(item));
451 }
452 return out;
453 }

482 /// @brief Returns the configured maximum depth, or `std::nullopt` if unbounded.
483 /// @return The capacity `enqueue()` enforces, or `std::nullopt` if none.
484 [[nodiscard]] std::optional<std::size_t> maxDepth() const override { return _maxDepth; }
485
486protected:
487 /// @brief Stamps an idempotency key onto an already-inserted row. No-op if
488 /// @p itemId is absent. Reachable only if a caller invokes the base
489 /// `IOfflineQueue::enqueue(payload, key)` default through an
490 /// `IOfflineQueue&` -- this class's own `enqueue(payload, key)`
491 /// override above stamps the key inline in the same INSERT instead.
492 /// @param itemId Id of the row to stamp.
493 /// @param idempotencyKey Key to store.
494 void setIdempotencyKey(uint64_t itemId, std::string idempotencyKey) override {
495 std::scoped_lock const lock{_mtx};
496 // `WHERE NOT EXISTS (...)` rather than a bare UPDATE: the partial
497 // unique index `ix_queue_idem` rejects stamping a non-empty key a
498 // pending row already holds, and a bare UPDATE turned that into a
499 // thrown SqliteOfflineQueueError -- while this class's own
500 // `enqueue(payload, key)` resolves the identical conflict silently, by
501 // keeping the existing row (morph#249). Same conflict, two answers,
502 // and only one of them matched the documented dedup contract.
503 //
504 // Throwing also bought nothing. This hook is called by the *base*
505 // `IOfflineQueue::enqueue(payload, key)` default, which has already
506 // inserted the row by the time it stamps; on a conflict the row exists
507 // either way, so the throw added an exception to an outcome it could
508 // not undo. Leaving the key unset is the same end state without it.
509 detail::StatementGuard const guard{
510 prepare("UPDATE morph_offline_queue SET idempotency_key = ?1 WHERE id = ?2 "
511 "AND (?1 = '' OR NOT EXISTS (SELECT 1 FROM morph_offline_queue "
512 "WHERE idempotency_key = ?1 AND id <> ?2));")};
513 bindText(guard.get(), 1, idempotencyKey);
514 bindInt64(guard.get(), 2, static_cast<std::int64_t>(itemId));
515 stepOrThrow(guard.get(), "setIdempotencyKey");
516 }

519 void execOrThrow(const char* sql) {
520 char* errMsg = nullptr;
521 if (sqlite3_exec(_db, sql, nullptr, nullptr, &errMsg) != SQLITE_OK) {
522 std::string msg = errMsg != nullptr ? errMsg : "unknown sqlite error";
523 sqlite3_free(errMsg);
524 throw SqliteOfflineQueueError{"SqliteOfflineQueue: " + msg};
525 }
526 }

528 sqlite3_stmt* prepare(const char* sql) const {
529 sqlite3_stmt* stmt = nullptr;
530 if (sqlite3_prepare_v2(_db, sql, -1, &stmt, nullptr) != SQLITE_OK) {
531 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: prepare failed: "} + sqlite3_errmsg(_db)};
532 }
533 return stmt;
534 }
535
536 // COVERAGE GAP, documented not closed: this throw's only realistic
537 // non-misuse trigger is SQLITE_TOOBIG, raised when `value` exceeds
538 // SQLite's SQLITE_LIMIT_LENGTH -- confirmed ~2 GB on this SDK's SQLite
539 // 3.51 via sqlite3_limit() (a 1,000,000,001-byte string was confirmed to
540 // still bind SQLITE_OK). This class does not expose the raw sqlite3*
541 // handle for a test to lower that limit via sqlite3_limit(), so the only
542 // way to reach this branch is allocating and binding an actual >2 GB
543 // std::string -- kSqliteTransient below means SQLite additionally copies
544 // it, so the peak footprint is >4 GB for one assertion. Evaluated and
545 // declined as not worth the memory/CI risk on an 8 GB development
546 // machine for a single low-value branch that a legitimate offline-queue
547 // payload will never approach in practice.
548 void bindText(sqlite3_stmt* stmt, int index, const std::string& value) const {
549 // An explicit length (not -1) is required so a NUL inside `value` --
550 // legitimate, since payload/idempotencyKey are opaque strings the
551 // caller controls the serialisation of (morph#531) -- doesn't tell
552 // SQLite to measure only up to that byte and silently truncate.
553 if (value.size() > static_cast<std::size_t>(INT_MAX)) {
554 throw SqliteOfflineQueueError{"SqliteOfflineQueue: value exceeds INT_MAX bytes"};
555 }
556 if (sqlite3_bind_text(stmt, index, value.c_str(), static_cast<int>(value.size()), detail::kSqliteTransient) !=
557 SQLITE_OK) {
558 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: bind failed: "} + sqlite3_errmsg(_db)};
559 }
560 }
561
562 void bindInt64(sqlite3_stmt* stmt, int index, std::int64_t value) const {
563 if (sqlite3_bind_int64(stmt, index, value) != SQLITE_OK) {
564 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: bind failed: "} + sqlite3_errmsg(_db)};
565 }
566 }
567
568 void stepOrThrow(sqlite3_stmt* stmt, const char* what) const {
569 // Anything but SQLITE_DONE throws, so SQLITE_BUSY is not distinguished
570 // from a genuine error -- a production consumer wanting to retry on busy
571 // would need to split them. A single in-process mutex around the whole
572 // connection makes SQLITE_BUSY practically unreachable for this
573 // reference queue, which is why the distinction is not made here.
574 if (sqlite3_step(stmt) != SQLITE_DONE) {
575 throw SqliteOfflineQueueError{std::string{"SqliteOfflineQueue: "} + what +
576 " failed: " + sqlite3_errmsg(_db)};
577 }
578 }
579
580 static std::string textColumn(sqlite3_stmt* stmt, int index) {
581 // sqlite3_column_bytes() gives the real stored length; constructing a
582 // std::string from the raw `const char*` alone would stop at the
583 // first NUL and silently truncate a NUL-bearing payload or
584 // idempotency key on the way back out (morph#531) -- the read-side
585 // half of the same truncation bindText() above fixes on the write
586 // side.
587 const auto* text = reinterpret_cast<const char*>(sqlite3_column_text(stmt, index));
588 return text != nullptr ? std::string{text, static_cast<std::size_t>(sqlite3_column_bytes(stmt, index))}
589 : std::string{};
590 }

593 return std::chrono::duration_cast<std::chrono::milliseconds>(
594 std::chrono::system_clock::now().time_since_epoch())
595 .count();

598 /// @brief Returns the current row count. Caller must hold `_mtx`.
599 /// @return `COUNT(*)` against `morph_offline_queue`.
600 std::size_t countLocked() const {
601 detail::StatementGuard guard{prepare("SELECT COUNT(*) FROM morph_offline_queue;")};
602 sqlite3_step(guard.get());
603 return static_cast<std::size_t>(sqlite3_column_int64(guard.get(), 0));
604 }

606 /// @brief Throws `OfflineQueueFullError` if the queue is already at
607 /// `maxDepth()`. Caller must hold `_mtx`. No-op if unbounded.
608 void checkCapacityLocked() const {
609 if (!_maxDepth) {
610 return;
611 }
612 std::size_t const current = countLocked();
613 if (current >= *_maxDepth) {
614 ::morph::observe::detail::emitMetric(::morph::observe::Metric::queueOverflow,
615 static_cast<double>(current));
616 throw OfflineQueueFullError{*_maxDepth, current};
617 }
618 }
