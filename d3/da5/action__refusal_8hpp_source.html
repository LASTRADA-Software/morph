15/// @brief One named argument of an `ActionRefusal`: a fact the refusal's sentence needs.

24/// @brief The exception an action throws to refuse itself with a code the client can word.
25///
26/// The message is for logs and tests and is never shown to a person; the code and the arguments are what a screen
27/// words from its message catalogue.
28class ActionRefusal : public std::runtime_error {
29public:
30 /// @brief A refusal.
31 /// @param code The application's dotted code, such as `lims.billing.invoiceLeased`.
32 /// @param args The named arguments of the sentence.
33 /// @param message The text for logs, returned by `what()`.
34 ActionRefusal(std::string code, std::vector<RefusalArg> args, std::string message)
35 : std::runtime_error{std::move(message)}, _code{std::move(code)}, _args{std::move(args)} {}
36
37 /// @brief The refusal's code.
38 /// @return The code.
39 [[nodiscard]] std::string const& code() const noexcept { return _code; }
