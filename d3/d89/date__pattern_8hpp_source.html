27/// @brief A token of a date pattern.
28enum class Token : std::uint8_t {
29 Year, ///< `yyyy`: four digits.
30 Month, ///< `MM`: two digits.
31 MonthName, ///< `MMM`: `Jan` to `Dec`, in English, read without regard to case.
32 Day, ///< `dd`: two digits.
33 Hour, ///< `HH`: two digits, 00 to 23.
34 Minute, ///< `mm`: two digits.
35 Second, ///< `ss`: two digits.
36 Offset, ///< `xxx`: the zone's offset from UTC, `+02:00`.
37 Literal, ///< One of `-`, `/`, `.`, `:`, `,` and a space.
38};
39
40/// @brief One element of a parsed pattern.
41struct Element {
42 /// @brief What it is.
43 Token token = Token::Literal;
44 /// @brief The character of a literal.
45 char literal = ' ';
46};

48/// @brief The English month abbreviations `MMM` writes and reads.
49inline constexpr std::array<std::string_view, 12> kMonths = {"Jan", "Feb", "Mar", "Apr", "May", "Jun",
50 "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"};

52/// @brief A run of one letter as a token: its letter and its length.
53/// @param letter The letter.
54/// @param count How many times it repeats.
55/// @return The token; `nullopt` for a run that is none.
56[[nodiscard]] inline std::optional<Token> tokenOf(char letter, std::size_t count) noexcept {
57 if (letter == 'y' && count == 4) {
58 return Token::Year;
59 }
60 if (letter == 'M' && count == 2) {
61 return Token::Month;
62 }
63 if (letter == 'M' && count == 3) {
64 return Token::MonthName;
65 }
66 if (letter == 'x' && count == 3) {
67 return Token::Offset;
68 }
69 if (count != 2) {
70 return std::nullopt;
71 }
72 switch (letter) {
73 case 'd':
74 return Token::Day;
75 case 'H':
76 return Token::Hour;
77 case 'm':
78 return Token::Minute;
79 case 's':
80 return Token::Second;
81 default:
82 return std::nullopt;
83 }
84}

86/// @brief A pattern split into its tokens and literals.
87/// @param pattern The pattern.
88/// @return The elements; `nullopt` when it holds anything but the tokens and the literals.
89[[nodiscard]] inline std::optional<std::vector<Element>> elementsOf(std::string_view pattern) {
90 std::vector<Element> elements;
91 for (std::size_t cursor = 0; cursor < pattern.size();) {
92 char const character = pattern.at(cursor);
93 if (std::string_view{"-/.:, "}.contains(character)) {
94 elements.push_back(Element{.token = Token::Literal, .literal = character});
95 ++cursor;
96 continue;
97 }
98 std::size_t run = cursor;
99 while (run < pattern.size() && pattern.at(run) == character) {
100 ++run;
101 }
102 std::optional<Token> const token = tokenOf(character, run - cursor);
103 if (!token) {
104 return std::nullopt;
105 }
106 elements.push_back(Element{.token = *token});
107 cursor = run;
108 }
109 return elements;
110}

112/// @brief Whether a token is written as digits.
113/// @param token The token.
114/// @return True for every token but `MMM`, `xxx` and a literal.
115[[nodiscard]] constexpr bool numeric(Token token) noexcept {
116 return token != Token::MonthName && token != Token::Offset && token != Token::Literal;
117}

119/// @brief Reads @p count digits of @p text cursor @p cursor.
120/// @param text The text.
121/// @param cursor Where to read; moved past the digits.
122/// @param count How many digits.
123/// @return The number; `nullopt` when there are not that many digits.
124[[nodiscard]] inline std::optional<int> digits(std::string_view text, std::size_t& cursor, std::size_t count) {
125 if (cursor + count > text.size()) {
126 return std::nullopt;
127 }
128 int value = 0;
129 for (std::size_t index = 0; index < count; ++index) {
130 char const digit = text.at(cursor + index);
131 if (digit < '0' || digit > '9') {
132 return std::nullopt;
133 }
134 value = (value * 10) + (digit - '0');
135 }
136 cursor += count;
137 return value;
138}

140/// @brief The fields a pattern reads out of a text.
141struct Fields {
142 int year = 1970; ///< `yyyy`.
143 unsigned month = 1; ///< `MM` or `MMM`.
144 unsigned day = 1; ///< `dd`.
145 int hour = 0; ///< `HH`.
146 int minute = 0; ///< `mm`.
147 int second = 0; ///< `ss`.
148 std::optional<int> offset; ///< `xxx`, in minutes.
149};

151/// @brief Reads `MMM` out of @p text cursor @p cursor, without regard to case.
152/// @param text The text.
153/// @param cursor Where to read; moved past it.
154/// @return The month, 1 to 12; `nullopt` when no month's name is there.
155[[nodiscard]] inline std::optional<unsigned> monthName(std::string_view text, std::size_t& cursor) {
156 if (cursor + 3 > text.size()) {
157 return std::nullopt;
158 }
159 auto const lower = [](char letter) { return std::tolower(static_cast<unsigned char>(letter)); };
160 for (std::size_t month = 0; month < kMonths.size(); ++month) {
161 std::string_view const name = kMonths.at(month);
162 if (lower(text.at(cursor)) == lower(name.at(0)) && lower(text.at(cursor + 1)) == lower(name.at(1)) &&
163 lower(text.at(cursor + 2)) == lower(name.at(2))) {
164 cursor += 3;
165 return static_cast<unsigned>(month + 1);
166 }
167 }
168 return std::nullopt;
169}

171/// @brief Reads `xxx` out of @p text cursor @p cursor: a sign, two hour digits, `:` and two minute digits.
172/// @param text The text.
173/// @param cursor Where to read; moved past it.
174/// @return The offset from UTC in minutes; `nullopt` when none is there.
175[[nodiscard]] inline std::optional<int> offset(std::string_view text, std::size_t& cursor) {
176 if (cursor >= text.size() || (text.at(cursor) != '+' && text.at(cursor) != '-')) {
177 return std::nullopt;
178 }
179 int const sign = text.at(cursor) == '-' ? -1 : 1;
180 ++cursor;
181 std::optional<int> const hours = digits(text, cursor, 2);
182 if (!hours || cursor >= text.size() || text.at(cursor) != ':') {
183 return std::nullopt;
184 }
185 ++cursor;
186 std::optional<int> const minutes = digits(text, cursor, 2);
187 if (!minutes || *hours > 23 || *minutes > 59) {
188 return std::nullopt;
189 }
190 return sign * ((*hours * 60) + *minutes);
191}

193/// @brief Reads one element of a pattern out of @p text cursor @p cursor into @p fields.
194/// @param element The element.
195/// @param text The text.
196/// @param cursor Where to read; moved past it.
197/// @param fields Where the value goes.
198/// @return Whether the text holds the element there.
199[[nodiscard]] inline bool readElement(Element const& element, std::string_view text, std::size_t& cursor,
200 Fields& fields) {
201 if (element.token == Token::Literal) {
202 if (cursor >= text.size() || text.at(cursor) != element.literal) {
203 return false;
204 }
205 ++cursor;
206 return true;
207 }
208 if (element.token == Token::MonthName) {
209 std::optional<unsigned> const month = monthName(text, cursor);
210 fields.month = month.value_or(0);
211 return month.has_value();
212 }
213 if (element.token == Token::Offset) {
214 fields.offset = offset(text, cursor);
215 return fields.offset.has_value();
216 }
217 std::optional<int> const value = digits(text, cursor, element.token == Token::Year ? 4 : 2);
218 if (!value) {
219 return false;
220 }
221 if (element.token == Token::Year) {
222 fields.year = *value;
223 } else if (element.token == Token::Month) {
224 fields.month = static_cast<unsigned>(*value);
225 } else if (element.token == Token::Day) {
226 fields.day = static_cast<unsigned>(*value);
227 } else if (element.token == Token::Hour) {
228 fields.hour = *value;
229 } else if (element.token == Token::Minute) {
230 fields.minute = *value;
231 } else {
232 fields.second = *value;
233 }
234 return true;
235}

237/// @brief The instant a text names in a pattern's elements.
238/// @param elements The pattern.
239/// @param text The text.
240/// @param offsetMinutes The display zone's offset, used when the pattern holds no `xxx`.
241/// @return The UTC instant; `nullopt` for text that names none, a day that does not exist among them.
242[[nodiscard]] inline std::optional<time::DateTime> read(std::vector<Element> const& elements, std::string_view text,
243 std::int32_t offsetMinutes) {
244 Fields fields;
245 std::size_t cursor = 0;
246 for (Element const& element : elements) {
247 if (!readElement(element, text, cursor, fields)) {
248 return std::nullopt;
249 }
250 }
251 std::chrono::year_month_day const date{std::chrono::year{fields.year}, std::chrono::month{fields.month},
252 std::chrono::day{fields.day}};
253 if (cursor != text.size() || !date.ok() || fields.hour > 23 || fields.minute > 59 || fields.second > 59) {
254 return std::nullopt;
255 }
256 auto const local = std::chrono::sys_days{date} + std::chrono::hours{fields.hour} +
257 std::chrono::minutes{fields.minute} + std::chrono::seconds{fields.second};
258 auto const utc = local - std::chrono::minutes{fields.offset.value_or(offsetMinutes)};
259 return time::DateTime{std::chrono::time_point_cast<std::chrono::milliseconds>(utc)};
260}

262/// @brief @p value as two digits, or four for a year.
263/// @param value The number.
264/// @param width How many digits.
265/// @return The digits, zero-padded.
266// NOLINTNEXTLINE(bugprone-easily-swappable-parameters): a number, then how wide to write it.
267[[nodiscard]] inline std::string padded(long long value, std::size_t width) {
268 std::string text = std::to_string(value < 0 ? -value : value);
269 if (text.size() < width) {
270 text.insert(0, width - text.size(), '0');
271 }
272 return value < 0 ? "-" + text : text;
273}

277/// @brief Checks a date pattern: only the tokens `yyyy`, `MM`, `MMM`, `dd`, `HH`, `mm`, `ss` and `xxx` and the
278/// literals `-`, `/`, `.`, `:`, `,` and a space; `yyyy`, `dd` and one of `MM` or `MMM` once each; for a date
279/// and time also `HH` and `mm` once each and `ss` and `xxx` cursor most once, and for a date no time token; and
280/// no two tokens touching without a literal between them, except `MMM` beside a number. So every text it writes
281/// reads back to the same instant.
282/// @param pattern The pattern, such as `dd-MMM-yyyy HH:mm`.
283/// @param withTime Whether it is a date-and-time pattern rather than a date pattern.
284/// @return Nothing, or why the pattern is refused.
285[[nodiscard]] inline std::expected<void, std::string> checkDatePattern(std::string_view pattern, bool withTime) {
286 using detail::dates::Token;
287 auto const elements = detail::dates::elementsOf(pattern);
288 if (!elements) {
289 return std::unexpected{
290 "a date pattern holds only yyyy, MM, MMM, dd, HH, mm, ss, xxx and the literals - / . : , "
291 "and a space"};
292 }
293 std::array<int, 9> counts{};
294 for (std::size_t index = 0; index < elements->size(); ++index) {
295 Token const token = elements->at(index).token;
296 ++counts.at(static_cast<std::size_t>(token));
297 if (index > 0 && token != Token::Literal) {
298 Token const before = elements->at(index - 1).token;
299 bool const touching = before != Token::Literal;
300 bool const nameBesideNumber = (token == Token::MonthName && detail::dates::numeric(before)) ||
301 (before == Token::MonthName && detail::dates::numeric(token));
302 if (touching && !nameBesideNumber) {
303 return std::unexpected{"two tokens of a date pattern touch without a literal between them"};
304 }
305 }
306 }
307 auto const count = [&counts](Token token) { return counts.at(static_cast<std::size_t>(token)); };
308 if (count(Token::Year) != 1 || count(Token::Day) != 1 || count(Token::Month) + count(Token::MonthName) != 1) {
309 return std::unexpected{"a date pattern holds yyyy, dd, and MM or MMM, each once"};
310 }
311 bool const anyTime = count(Token::Hour) + count(Token::Minute) + count(Token::Second) + count(Token::Offset) > 0;
312 if (!withTime && anyTime) {
313 return std::unexpected{"a date pattern holds no time token"};
314 }
315 if (withTime && (count(Token::Hour) != 1 || count(Token::Minute) != 1 || count(Token::Second) > 1 ||
316 count(Token::Offset) > 1)) {
317 return std::unexpected{"a date-and-time pattern holds HH and mm once each, and ss and xxx cursor most once"};
318 }
319 return {};
320}

322/// @brief Writes an instant in a pattern.
323/// @param pattern A pattern `checkDatePattern` takes.
324/// @param local The instant, already moved into the display zone.
325/// @param offsetMinutes The display zone's offset from UTC, which `xxx` writes as `+02:00` (`+00:00` for UTC).
326/// @return The text, such as `24-Oct-2026 09:41`; empty for a pattern that is not one.
327[[nodiscard]] inline std::string formatDate(std::string_view pattern, time::DateTime local,
328 std::int32_t offsetMinutes) {
329 using detail::dates::Token;
330 auto const elements = detail::dates::elementsOf(pattern);
331 if (!elements) {
332 return {};
333 }
334 auto const day = std::chrono::floor<std::chrono::days>(local.value);
335 std::chrono::year_month_day const date{day};
336 std::chrono::hh_mm_ss const clock{std::chrono::floor<std::chrono::seconds>(local.value - day)};
337 std::string text;
338 for (detail::dates::Element const& element : *elements) {
339 switch (element.token) {
340 case Token::Literal:
341 text += element.literal;
342 break;
343 case Token::Year:
344 text += detail::dates::padded(static_cast<int>(date.year()), 4);
345 break;
346 case Token::Month:
347 text += detail::dates::padded(static_cast<unsigned>(date.month()), 2);
348 break;
349 case Token::MonthName:
350 text += detail::dates::kMonths.at(static_cast<unsigned>(date.month()) - 1);
351 break;
352 case Token::Day:
353 text += detail::dates::padded(static_cast<unsigned>(date.day()), 2);
354 break;
355 case Token::Hour:
356 text += detail::dates::padded(clock.hours().count(), 2);
357 break;
358 case Token::Minute:
359 text += detail::dates::padded(clock.minutes().count(), 2);
360 break;
361 case Token::Second:
362 text += detail::dates::padded(clock.seconds().count(), 2);
363 break;
364 case Token::Offset:
365 default: {
366 std::int32_t const minutes = offsetMinutes < 0 ? -offsetMinutes : offsetMinutes;
367 text += (offsetMinutes < 0 ? "-" : "+") + detail::dates::padded(minutes / 60, 2) + ":" +
368 detail::dates::padded(minutes % 60, 2);
369 break;
370 }
371 }
372 }
373 return text;
374}

376/// @brief Reads the instant a text names: in the pattern, then as ISO 8601 (`yyyy-MM-dd`, `yyyy-MM-dd HH:mm` or
377/// `yyyy-MM-ddTHH:mm` in the display zone, or a full instant such as `2026-10-24T09:41:00Z`), so a pasted ISO
378/// date still works. A pattern with `xxx` takes the instant from the offset the text names; any other reads it
379/// in the display zone.
380/// @param pattern The pattern; empty reads ISO only.
381/// @param text The text.
382/// @param offsetMinutes The display zone's offset from UTC.
383/// @return The UTC instant; `nullopt` for text that names none, such as `31-Feb-2026`.
384// NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the pattern, then the text, as `formatDate` takes them.
385[[nodiscard]] inline std::optional<time::DateTime> parseDate(std::string_view pattern, std::string_view text,
386 std::int32_t offsetMinutes) {
387 for (std::string_view const form : {pattern, std::string_view{"yyyy-MM-dd HH:mm"},
388 std::string_view{"yyyy-MM-ddTHH:mm"}, std::string_view{"yyyy-MM-dd"}}) {
389 if (form.empty()) {
390 continue;
391 }
392 std::optional<std::vector<detail::dates::Element>> elements;
393 if (form == "yyyy-MM-ddTHH:mm") {
394 elements = detail::dates::elementsOf("yyyy-MM-dd HH:mm");
395 if (elements) {
396 elements->at(5).literal = 'T'; // the space between the date and the clock
397 }
398 } else {
399 elements = detail::dates::elementsOf(form);
400 }
401 if (elements) {
402 if (auto instant = detail::dates::read(*elements, text, offsetMinutes)) {
403 return instant;
404 }
405 }
406 }
407 return time::DateTime::fromIso8601(text);
408}
