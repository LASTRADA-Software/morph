22/// @brief Answers each call with the next scripted response, and fails once the script runs out. Thread-safe.
23///
24/// Delivers before `stream` returns, unless `hold(true)` leaves each call outstanding until `release`.
25class FakeChatProvider final : public IProvider {
26public:
27 /// @brief Adds a response to the script.
28 /// @param response What the next unscripted call answers.
29 void script(Response response) {
30 std::scoped_lock const lock{_state->mutex};
31 _state->script.push_back(std::move(response));
32 }
33
34 /// @brief Answers every call past the script with @p response, so a turn can go on calling tools indefinitely.
35 /// @param response The response.
36 void repeat(Response response) {
37 std::scoped_lock const lock{_state->mutex};
38 _state->forever = std::move(response);
39 }
40
41 /// @brief Sets whether calls wait for `release`.
42 /// @param held Whether they do.
43 void hold(bool held) {
44 std::scoped_lock const lock{_state->mutex};
45 _state->held = held;
46 }
47
48 /// @brief Delivers every held call that was not cancelled, and stops holding.
49 void release() {
50 std::vector<std::shared_ptr<Pending>> pending;
51 {
52 std::scoped_lock const lock{_state->mutex};
53 _state->held = false;
54 pending.swap(_state->pending);
55 }
56 for (auto const& call : pending) {
57 call->deliver();
58 }
59 }
60
61 /// @brief The calls made so far.
62 /// @return Their number.
63 [[nodiscard]] std::size_t calls() const {
64 std::scoped_lock const lock{_state->mutex};

66 }
67
68 /// @brief The calls cancelled so far.
69 /// @return Their number.

74
75 /// @brief What each call was asked.

82 /// @brief Starts a call.
83 /// @param request What to ask.
84 /// @param sink Where it delivers.
85 /// @return The call.
86 [[nodiscard]] std::shared_ptr<IProviderCall> stream(Request request, ProviderSink sink) override {
87 auto call = std::make_shared<Pending>(_state, std::move(sink));
88 bool held = false;
89 {
90 std::scoped_lock const lock{_state->mutex};
91 _state->requests.push_back(std::move(request));
92 if (!_state->script.empty()) {
93 call->outcome = std::move(_state->script.front());
94 _state->script.pop_front();
95 } else if (_state->forever) {
96 call->outcome = *_state->forever;
97 } else {
98 call->outcome = std::unexpected(std::string{"the fake provider's script ran out"});
99 }
100 held = _state->held;
101 if (held) {
102 _state->pending.push_back(call);
103 }
104 }
105 if (!held) {
106 call->deliver();
107 }
108 return call;
109 }

111private:
112 struct State;

114 // Holds the state, so a call outlives the provider and a late cancel touches nothing freed. A finished call lets
115 // go of its sink and leaves the held list: the sink usually owns whoever owns this provider, so keeping either
116 // would hold the two alive in a cycle. Both are destroyed outside the lock, since freeing the sink can free the
117 // provider.
118 struct Pending final : IProviderCall {
119 Pending(std::shared_ptr<State> shared, ProviderSink given)
120 : state{std::move(shared)}, sink{std::move(given)} {}
121
122 void cancel() override {
123 ProviderSink dropped;
124 std::shared_ptr<Pending> unheld;
125 {
126 std::scoped_lock const lock{state->mutex};
127 if (finished) {
128 return;
129 }
130 finished = true;
131 ++state->cancelled;
132 dropped = std::move(sink);
133 auto const position = std::ranges::find_if(
134 state->pending, [this](std::shared_ptr<Pending> const& held) { return held.get() == this; });
135 if (position != state->pending.end()) {
136 unheld = std::move(*position);
137 state->pending.erase(position);
138 }
139 }
140 }
141
142 void deliver() {
143 ProviderSink taken;
144 {
145 std::scoped_lock const lock{state->mutex};
146 if (finished) {
147 return;
148 }
149 finished = true;
150 taken = std::move(sink);
151 }
152 if (outcome && !outcome->text.empty() && taken.text) {
153 taken.text(outcome->text);
154 }
155 if (taken.done) {
156 taken.done(std::move(outcome));
157 }
158 }
159
160 std::shared_ptr<State> state;
161 ProviderSink sink;
162 std::expected<Response, std::string> outcome = std::unexpected(std::string{});
163 bool finished = false;
164 };
165
166 struct State {
167 mutable std::mutex mutex;
168 std::deque<Response> script;
169 std::optional<Response> forever;
170 std::vector<Request> requests;
171 std::vector<std::shared_ptr<Pending>> pending;
172 std::size_t cancelled = 0;
173 bool held = false;
174 };
175
176 std::shared_ptr<State> _state = std::make_shared<State>();
177};
