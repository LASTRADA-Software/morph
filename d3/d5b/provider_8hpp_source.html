18/// @brief A tool offered to the model.

28/// @brief A call the model made.
29struct ToolCall {
30 /// @brief The provider's id for it, which its result names.
31 std::string id;
32 /// @brief The tool's name, which may name no tool: a model's output is untrusted.

38/// @brief What a tool call came to.
39struct ToolResult {
40 /// @brief The call's `ToolCall::id`.
41 std::string callId;
42 /// @brief The result's text, or why it did not run.
43 std::string text;
44 /// @brief Whether it failed.
45 bool isError = false;
46};

48/// @brief One message of a conversation.
49struct Message {
50 /// @brief `user` or `assistant`.
51 std::string role;
52 /// @brief Its text.
53 std::string text;
54 /// @brief The calls an assistant message made.
55 std::vector<ToolCall> calls;
56 /// @brief The results a user message reports.
57 std::vector<ToolResult> results;
58 /// @brief What a provider said it must get back unchanged for an assistant message, from `Response::opaque`; empty
59 /// once history around it has been edited, since it is valid only in the conversation that produced it.
60 std::string opaque;
61};

63/// @brief What one provider call is asked.
64struct Request {
65 /// @brief The system text.
66 std::string system;
67 /// @brief The conversation so far, oldest first.
68 std::vector<Message> messages;
69 /// @brief The tools it may call.
70 std::vector<ToolSpec> tools;
71};
72
73/// @brief What one provider call answered.
74struct Response {
75 /// @brief The whole text.
76 std::string text;
77 /// @brief The tools it wants called; none ends the turn.
78 std::vector<ToolCall> calls;
79 /// @brief Whatever this provider needs replayed unchanged for this message, such as a model's reasoning
80 /// signatures.
81 /// Only the provider that wrote it reads it.
82 std::string opaque;
83};

85/// @brief An outstanding provider call.
86class IProviderCall {

95 /// @brief Stops the call. Its sink hears nothing after this returns, unless a delivery had already begun. Callable
96 /// from any thread, more than once.

100/// @brief Where a provider call delivers.
101struct ProviderSink {
102 /// @brief A piece of the text, from any thread, in order.
103 std::function<void(std::string const&)> text;
104 /// @brief The outcome, or why there is none, from any thread, exactly once unless cancelled. It may run before
105 /// `IProvider::stream` returns.
106 std::function<void(std::expected<Response, std::string>)> done;
107};

109/// @brief An AI model behind a chat.
110class IProvider {
111public:

115 IProvider(IProvider&&) = delete;
116 IProvider& operator=(IProvider&&) = delete;
117 virtual ~IProvider() = default;

119 /// @brief Starts a call. Callable from any thread.
120 /// @param request What to ask.
121 /// @param sink Where it delivers.
