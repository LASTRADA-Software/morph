17/// @file session/session_auth.hpp
18/// @brief Opt-in authenticated sessions: signed bearer tokens and a verifying
19/// `IAuthorizer` so a `RemoteServer` can trust the caller's identity.
20///
21/// Include this only if you want authentication. The token is a signed,
22/// stateless bearer credential — `base64url(claimsJson) "." base64url(mac)` —
23/// minted app-side by a login action via `TokenIssuer` and verified per request
24/// by `SigningAuthorizer`, which also feeds the verified principal back to
25/// `RemoteServer` so `session::current()->principal` is authoritative inside a
26/// model. The MAC primitive is pluggable (`MacFunction`); a self-contained,
27/// test-vector-verified HMAC-SHA256 (`hmacSha256`) is the default. See
28/// `docs/spec/security.md` for the full trust model and its limits.

42/// @brief SHA-256 (FIPS 180-4). Returns the raw 32-byte digest.
43/// @param data Message bytes to hash.
44/// @return 32-byte digest as a byte string.
45inline std::string sha256(std::string_view data) {
46 static constexpr std::array<uint32_t, 64> kRound = {
47 0x428a2f98, 0x71374491, 0xb5c0fbcf, 0xe9b5dba5, 0x3956c25b, 0x59f111f1, 0x923f82a4, 0xab1c5ed5,
48 0xd807aa98, 0x12835b01, 0x243185be, 0x550c7dc3, 0x72be5d74, 0x80deb1fe, 0x9bdc06a7, 0xc19bf174,
49 0xe49b69c1, 0xefbe4786, 0x0fc19dc6, 0x240ca1cc, 0x2de92c6f, 0x4a7484aa, 0x5cb0a9dc, 0x76f988da,
50 0x983e5152, 0xa831c66d, 0xb00327c8, 0xbf597fc7, 0xc6e00bf3, 0xd5a79147, 0x06ca6351, 0x14292967,
51 0x27b70a85, 0x2e1b2138, 0x4d2c6dfc, 0x53380d13, 0x650a7354, 0x766a0abb, 0x81c2c92e, 0x92722c85,
52 0xa2bfe8a1, 0xa81a664b, 0xc24b8b70, 0xc76c51a3, 0xd192e819, 0xd6990624, 0xf40e3585, 0x106aa070,
53 0x19a4c116, 0x1e376c08, 0x2748774c, 0x34b0bcb5, 0x391c0cb3, 0x4ed8aa4a, 0x5b9cca4f, 0x682e6ff3,
54 0x748f82ee, 0x78a5636f, 0x84c87814, 0x8cc70208, 0x90befffa, 0xa4506ceb, 0xbef9a3f7, 0xc67178f2};
55
56 std::array<uint32_t, 8> hash = {0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a,
57 0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19};
58
59 std::vector<uint8_t> msg(data.begin(), data.end());
60 const uint64_t bitLen = msg.size() * 8;
61 msg.push_back(0x80);
62 while (msg.size() % 64 != 56) {
63 msg.push_back(0x00);
64 }
65 for (int shift = 56; shift >= 0; shift -= 8) {
66 msg.push_back(static_cast<uint8_t>((bitLen >> shift) & 0xff));
67 }
68
69 auto rotr = [](uint32_t val, uint32_t bits) { return (val >> bits) | (val << (32 - bits)); };
70
71 for (std::size_t chunk = 0; chunk < msg.size(); chunk += 64) {
72 std::array<uint32_t, 64> w{};
73 for (std::size_t i = 0; i < 16; ++i) {
74 w[i] = (static_cast<uint32_t>(msg[chunk + (i * 4)]) << 24) |
75 (static_cast<uint32_t>(msg[chunk + (i * 4) + 1]) << 16) |
76 (static_cast<uint32_t>(msg[chunk + (i * 4) + 2]) << 8) |
77 static_cast<uint32_t>(msg[chunk + (i * 4) + 3]);
78 }
79 for (std::size_t i = 16; i < 64; ++i) {
80 const uint32_t s0 = rotr(w[i - 15], 7) ^ rotr(w[i - 15], 18) ^ (w[i - 15] >> 3);
81 const uint32_t s1 = rotr(w[i - 2], 17) ^ rotr(w[i - 2], 19) ^ (w[i - 2] >> 10);
82 w[i] = w[i - 16] + s0 + w[i - 7] + s1;
83 }
84
85 uint32_t a = hash[0];
86 uint32_t b = hash[1];
87 uint32_t c = hash[2];
88 uint32_t d = hash[3];
89 uint32_t e = hash[4];
90 uint32_t f = hash[5];
91 uint32_t g = hash[6];
92 uint32_t h = hash[7];
93 for (std::size_t i = 0; i < 64; ++i) {
94 const uint32_t bigS1 = rotr(e, 6) ^ rotr(e, 11) ^ rotr(e, 25);
95 const uint32_t choose = (e & f) ^ (~e & g);
96 const uint32_t t1 = h + bigS1 + choose + kRound[i] + w[i];
97 const uint32_t bigS0 = rotr(a, 2) ^ rotr(a, 13) ^ rotr(a, 22);
98 const uint32_t maj = (a & b) ^ (a & c) ^ (b & c);
99 const uint32_t t2 = bigS0 + maj;
100 h = g;
101 g = f;
102 f = e;
103 e = d + t1;
104 d = c;
105 c = b;
106 b = a;
107 a = t1 + t2;
108 }
109 hash[0] += a;
110 hash[1] += b;
111 hash[2] += c;
112 hash[3] += d;
113 hash[4] += e;
114 hash[5] += f;
115 hash[6] += g;
116 hash[7] += h;
117 }
118
119 std::string out(32, '\0');
120 for (std::size_t i = 0; i < 8; ++i) {
121 out[(i * 4)] = static_cast<char>((hash[i] >> 24) & 0xff);
122 out[(i * 4) + 1] = static_cast<char>((hash[i] >> 16) & 0xff);
123 out[(i * 4) + 2] = static_cast<char>((hash[i] >> 8) & 0xff);
124 out[(i * 4) + 3] = static_cast<char>(hash[i] & 0xff);
125 }
126 return out;
127}

129/// @brief HMAC-SHA256 (RFC 2104). Returns the raw 32-byte MAC.
130/// @param key Secret key bytes.
131/// @param msg Message bytes to authenticate.
132/// @return 32-byte MAC as a byte string.
133inline std::string hmacSha256Raw(std::string_view key, std::string_view msg) {
134 constexpr std::size_t blockSize = 64;
135 std::string block{key};
136 if (block.size() > blockSize) {
137 block = sha256(block);
138 }
139 block.resize(blockSize, '\0');
140
141 std::string inner(blockSize, '\0');
142 std::string outer(blockSize, '\0');
143 for (std::size_t i = 0; i < blockSize; ++i) {
144 inner[i] = static_cast<char>(static_cast<uint8_t>(block[i]) ^ 0x36);
145 outer[i] = static_cast<char>(static_cast<uint8_t>(block[i]) ^ 0x5c);
146 }
147 const std::string innerHash = sha256(inner + std::string{msg});
148 return sha256(outer + innerHash);
149}

151/// @brief Constant-time byte-string equality (guards against MAC timing leaks).
152/// @param lhs First string.
153/// @param rhs Second string.
154/// @return `true` iff the strings are byte-for-byte equal.
155inline bool constantTimeEquals(std::string_view lhs, std::string_view rhs) {
156 if (lhs.size() != rhs.size()) {
157 return false; // length is not secret
158 }
159 uint8_t diff = 0;
160 for (std::size_t i = 0; i < lhs.size(); ++i) {
161 diff = uint8_t(diff | (uint8_t(lhs[i]) ^ uint8_t(rhs[i])));
162 }
163 return diff == 0;
164}

166/// @brief The url-safe base64 alphabet (RFC 4648 §5): `-` and `_` replace `+`/`/`.
167inline constexpr std::string_view kB64Alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
168
169/// @brief base64url-encodes @p in without padding (RFC 4648 §5).
170/// @param in Bytes to encode.
171/// @return The base64url text.
172inline std::string base64UrlEncode(std::string_view in) {
173 std::string out;
174 std::size_t i = 0;
175 for (; i + 3 <= in.size(); i += 3) {
176 const uint32_t triple = (static_cast<uint32_t>(static_cast<uint8_t>(in[i])) << 16) |
177 (static_cast<uint32_t>(static_cast<uint8_t>(in[i + 1])) << 8) |
178 static_cast<uint32_t>(static_cast<uint8_t>(in[i + 2]));
179 out.push_back(kB64Alphabet[(triple >> 18) & 0x3f]);
180 out.push_back(kB64Alphabet[(triple >> 12) & 0x3f]);
181 out.push_back(kB64Alphabet[(triple >> 6) & 0x3f]);
182 out.push_back(kB64Alphabet[triple & 0x3f]);
183 }
184 const std::size_t rem = in.size() - i;
185 if (rem == 1) {
186 const uint32_t triple = static_cast<uint32_t>(static_cast<uint8_t>(in[i])) << 16;
187 out.push_back(kB64Alphabet[(triple >> 18) & 0x3f]);
188 out.push_back(kB64Alphabet[(triple >> 12) & 0x3f]);
189 } else if (rem == 2) {
190 const uint32_t triple = (static_cast<uint32_t>(static_cast<uint8_t>(in[i])) << 16) |
191 (static_cast<uint32_t>(static_cast<uint8_t>(in[i + 1])) << 8);
192 out.push_back(kB64Alphabet[(triple >> 18) & 0x3f]);
193 out.push_back(kB64Alphabet[(triple >> 12) & 0x3f]);
194 out.push_back(kB64Alphabet[(triple >> 6) & 0x3f]);
195 }
196 return out;
197}

199/// @brief Decodes canonical base64url text (no padding). Returns nullopt on invalid input.
200///
201/// Canonical decoding is enforced to remove token-string malleability: a token
202/// is a signed byte string, but base64url is a *bit*-oriented encoding, so a
203/// naive decoder that silently discards the leftover bits of the final symbol
204/// would map several distinct encodings onto the same bytes (e.g. the trailing
205/// character could be perturbed within the discarded low bits without changing
206/// the decoded MAC). This function rejects any such non-canonical input:
207///
208/// - A length `% 4 == 1` is impossible for real base64url output and is rejected.
209/// - The leftover bits that do not form a whole output byte (2 bits for a
210/// 1-remainder group, 4 bits for a 2-remainder group) **must be zero**; a
211/// nonzero remainder means the encoding is not the canonical one this decoder
212/// would itself produce, so it is rejected rather than truncated.
213///
214/// The result is thus a bijection over valid tokens: exactly one string decodes
215/// to any given byte sequence.
216/// @param in base64url text.
217/// @return Decoded bytes, or nullopt if @p in has an invalid character, an
218/// impossible length, or nonzero leftover (non-canonical) bits.
219inline std::optional<std::string> base64UrlDecode(std::string_view in) {
220 auto valueOf = [](char chr) -> int {
221 const auto pos = kB64Alphabet.find(chr);
222 return pos == std::string_view::npos ? -1 : static_cast<int>(pos);
223 };
224 // A base64url group of 4 symbols encodes 3 bytes; the only valid tail
225 // remainders are 0 (whole groups), 2 (→1 byte), or 3 (→2 bytes). A
226 // remainder of 1 symbol cannot be produced by the encoder and is rejected.
227 if (in.size() % 4 == 1) {
228 return std::nullopt;
229 }
230 std::string out;
231 uint32_t buffer = 0;
232 int bits = 0;
233 for (const char chr : in) {
234 const int val = valueOf(chr);
235 if (val < 0) {
236 return std::nullopt;
237 }
238 buffer = (buffer << 6) | static_cast<uint32_t>(val);
239 bits += 6;
240 if (bits >= 8) {
241 bits -= 8;
242 out.push_back(static_cast<char>((buffer >> bits) & 0xff));
243 }
244 }
245 // Reject non-canonical encodings: any bit left in `buffer` below the byte
246 // boundary must be zero, otherwise this input is a distinct encoding of the
247 // same bytes (malleability) rather than the canonical one.
248 if (bits > 0 && (buffer & ((1U << bits) - 1)) != 0) {
249 return std::nullopt;
250 }
251 return out;
252}

282/// @brief Verified claims carried by a session token.
283///
284/// Serialised to JSON (Glaze) as the signed payload. Extend with app claims as
285/// needed — unknown fields are ignored on read, so adding claims is compatible.
286struct SessionToken {
287 /// @brief Authenticated user/principal id.
288 std::string principal;
289 /// @brief Issue time, ms since epoch. If positive, `verify` rejects a token
290 /// whose `issuedAtMs` is more than `kClockSkewMs` in the future
291 /// (not-yet-valid). `0` (unset) disables the not-before check.
292 int64_t issuedAtMs = 0;
293 /// @brief Expiry, ms since epoch. **Must be strictly positive** — a token
294 /// must carry a real expiry. `verify` treats `expiresAtMs <= 0`
295 /// (including the default `0`) as invalid/expired, so a zero token is
296 /// never an eternal credential.
297 int64_t expiresAtMs = 0;
298 /// @brief Coarse-grained roles the app can key authorization policy on.
299 std::vector<std::string> roles;
300};
301
302/// @brief Allowed clock skew (ms) between issuer and verifier for the
303/// not-before (`issuedAtMs`) check. Keeps a token minted a moment ahead
304/// of the verifier's clock from being spuriously rejected.
305inline constexpr int64_t kClockSkewMs = 60'000; // 60s
306
307/// @brief Why token verification failed.
308enum class AuthError : std::uint8_t {
309 Malformed, ///< Not `payload.sig`, bad base64url, or unparseable claims.
310 BadSignature, ///< MAC did not match — forged or tampered.
311 Expired, ///< Missing/non-positive `expiresAtMs`, or it is in the past.
312 NotYetValid, ///< `issuedAtMs` is set and more than `kClockSkewMs` in the future.
313};

340/// @brief Write options that escape ASCII control bytes as `\\uXXXX` sequences.

353/// round-trips through `TokenVerifier::verify` unchanged.

355 /// @brief Emit control bytes as `\\uXXXX` rather than raw.
356 // NOLINTNEXTLINE(readability-identifier-naming) — glaze's option name, matched by name.
357 bool escape_control_characters = true;
358};

362/// @brief Mints signed bearer tokens from claims using a shared secret.
363///
364/// Wire format: `base64url(claimsJson) "." base64url(mac(secret, payload))`.
365class TokenIssuer {
366public:
367#ifdef MORPH_REQUIRE_VETTED_HMAC
368 /// @brief Constructs an issuer over @p secret using @p mac.
369 ///
370 /// Built with `MORPH_REQUIRE_VETTED_HMAC` defined: @p mac has **no
371 /// default** here, so a call site that relied on the reference
372 /// `hmacSha256` default fails to compile, forcing an explicit vetted
373 /// `MacFunction` injection. See `docs/spec/security.md` ("MAC-primitive
374 /// recommended wiring").
375 /// @param secret Shared secret (same value the verifier uses).
376 /// @param mac MAC primitive (required — no default under this build option).
377 explicit TokenIssuer(std::string secret, MacFunction mac) : _secret{std::move(secret)}, _mac{std::move(mac)} {}
378#else
379 /// @brief Constructs an issuer over @p secret using @p mac.
380 /// @param secret Shared secret (same value the verifier uses).
381 /// @param mac MAC primitive; defaults to `hmacSha256`.
382 explicit TokenIssuer(std::string secret, MacFunction mac = hmacSha256)
383 : _secret{std::move(secret)}, _mac{std::move(mac)} {}
384#endif
385
386 /// @brief Serialises @p claims and returns a signed token string.
387 ///
388 /// Writes with `detail::EscapingWriteOpts` so a raw ASCII control byte in
389 /// `principal`/`roles` round-trips through `TokenVerifier::verify` instead
390 /// of producing invalid JSON (or, alongside an escaped `\`/`"`, silently
391 /// corrupted JSON) — see that struct's doc comment.
392 /// @param claims Claims to embed and sign.
393 /// @return The signed `payload.sig` token.
394 /// @throws TokenIssuanceError on encode failure (see `detail::throwOnGlazeError`
395 /// for why this is not realistically reachable for `SessionToken`).
396 [[nodiscard]] std::string issue(const SessionToken& claims) const {
397 std::string json;
398 detail::throwOnGlazeError(glz::write<detail::EscapingWriteOpts{}>(claims, json), json);
399 const std::string payload = detail::base64UrlEncode(json);
400 const std::string sig = detail::base64UrlEncode(_mac(_secret, payload));
401 return payload + "." + sig;
402 }
403
404private:
405 std::string _secret;
406 MacFunction _mac;
407};

409/// @brief Verifies signed bearer tokens against a shared secret.
410class TokenVerifier {
411public:
412#ifdef MORPH_REQUIRE_VETTED_HMAC
413 /// @brief Constructs a verifier over @p secret using @p mac.
414 ///
415 /// Built with `MORPH_REQUIRE_VETTED_HMAC` defined: @p mac has **no
416 /// default** here, so a call site that relied on the reference
417 /// `hmacSha256` default fails to compile, forcing an explicit vetted
418 /// `MacFunction` injection. See `docs/spec/security.md` ("MAC-primitive
419 /// recommended wiring").
420 /// @param secret Shared secret (same value the issuer used).
421 /// @param mac MAC primitive (required — no default under this build option).
422 explicit TokenVerifier(std::string secret, MacFunction mac) : _secret{std::move(secret)}, _mac{std::move(mac)} {}
423#else
424 /// @brief Constructs a verifier over @p secret using @p mac.
425 /// @param secret Shared secret (same value the issuer used).
426 /// @param mac MAC primitive; defaults to `hmacSha256`.
427 explicit TokenVerifier(std::string secret, MacFunction mac = hmacSha256)
428 : _secret{std::move(secret)}, _mac{std::move(mac)} {}
429#endif
430
431 /// @brief Verifies @p token's signature and expiry and returns its claims.
432 ///
433 /// The MAC is checked *before* the payload is parsed, so untrusted JSON is
434 /// never handed to the parser until authenticity is established.
435 /// @param token Token string (`payload.sig`).
436 /// @param nowMs Current time (ms since epoch) to check expiry against.
437 /// @return The verified claims, or an `AuthError`.
438 [[nodiscard]] std::expected<SessionToken, AuthError> verify(std::string_view token, int64_t nowMs) const {
439 const auto dot = token.find('.');
440 if (dot == std::string_view::npos || token.find('.', dot + 1) != std::string_view::npos) {
441 return std::unexpected(AuthError::Malformed);
442 }
443 const std::string_view payload = token.substr(0, dot);
444 const auto sig = detail::base64UrlDecode(token.substr(dot + 1));
445 if (!sig) {
446 return std::unexpected(AuthError::Malformed);
447 }
448 const std::string expectedMac = _mac(_secret, std::string{payload});
449 if (!detail::constantTimeEquals(*sig, expectedMac)) {
450 return std::unexpected(AuthError::BadSignature);
451 }
452 const auto json = detail::base64UrlDecode(payload);
453 if (!json) {
454 return std::unexpected(AuthError::Malformed);
455 }
456 SessionToken claims;
457 // Ignore unknown claims so a token minted by a newer issuer (with extra
458 // application claims) still verifies against an older verifier — the
459 // forward-compatibility the SessionToken doc promises. See security.md.
460 static constexpr glz::opts kLenient{.error_on_unknown_keys = false};
461 if (glz::read<kLenient>(claims, *json)) {
462 return std::unexpected(AuthError::Malformed);
463 }
464 // A token MUST carry a strictly-positive expiry. A missing/zero (or
465 // negative) `expiresAtMs` is treated as already-expired rather than
466 // "eternal", so a default-constructed or zeroed token is never an
467 // unbounded bearer credential. See docs/spec/security.md.
468 if (claims.expiresAtMs <= 0 || nowMs > claims.expiresAtMs) {
469 return std::unexpected(AuthError::Expired);
470 }
471 // Not-before / issued-at check. When `issuedAtMs` is set (positive), a
472 // token whose issue time is more than `kClockSkewMs` in the future is
473 // not yet valid — it was minted against a clock ahead of ours beyond the
474 // tolerated skew. An unset (`0`) or non-positive `issuedAtMs` skips this
475 // check (issue time is optional/informational).
476 if (claims.issuedAtMs > 0 && claims.issuedAtMs - nowMs > kClockSkewMs) {
477 return std::unexpected(AuthError::NotYetValid);
478 }
479 return claims;
480 }
481
482private:
483 std::string _secret;
484 MacFunction _mac;
485};

487/// @brief `IAuthorizer` that authenticates via a signed `Context::token`.
488///
489/// `authorize` returns `true` only for a token with a valid signature and
490/// unexpired claims (and, if a policy is supplied, one the policy admits).
491/// `authenticate` returns the verified principal so `RemoteServer` makes it
492/// authoritative. Install it on the server:
493/// @code
494/// auto authz = std::make_shared<morph::session::SigningAuthorizer>(sharedSecret);
495/// auto server = std::make_shared<morph::backend::RemoteServer>(pool, authz);
496/// @endcode

500 ///
501 /// Receives the verified token plus the target ids; return `false` to deny.
502 /// The default (empty) admits any validly-signed, unexpired token.
503 using Policy =
504 std::function<bool(const SessionToken& claims, std::string_view modelType, std::string_view actionType)>;

506#ifdef MORPH_REQUIRE_VETTED_HMAC
507 /// @brief Constructs the authorizer.
508 ///
509 /// Built with `MORPH_REQUIRE_VETTED_HMAC` defined: @p mac has **no
510 /// default**, so a call site that relied on the reference `hmacSha256`
511 /// default fails to compile. See `docs/spec/security.md` ("MAC-primitive
512 /// recommended wiring").
513 /// @param secret Shared secret used to verify tokens.
514 /// @param mac MAC primitive (required — no default under this build option).
515 /// @param clock Time source (ms since epoch) for expiry; defaults to system time.
516 /// @param policy Optional per-request policy over verified claims.
517 explicit SigningAuthorizer(std::string secret, MacFunction mac, Clock clock = systemClockMs, Policy policy = {})

519#else
520 /// @brief Constructs the authorizer.
521 /// @param secret Shared secret used to verify tokens.
522 /// @param mac MAC primitive; defaults to `hmacSha256`.
523 /// @param clock Time source (ms since epoch) for expiry; defaults to system time.
524 /// @param policy Optional per-request policy over verified claims.
525 explicit SigningAuthorizer(std::string secret, MacFunction mac = hmacSha256, Clock clock = systemClockMs,
526 Policy policy = {})
527 : _verifier{std::move(secret), std::move(mac)}, _clock{std::move(clock)}, _policy{std::move(policy)} {}
528#endif
529
530 /// @brief Allows the call iff @p ctx carries a valid token the policy admits.
531 /// @param ctx Per-call session (its `token` is verified).
532 /// @param modelType Target model type id (passed to the policy).
533 /// @param actionType Target action type id (passed to the policy).
534 /// @return `true` to allow dispatch, `false` to reject.
535 [[nodiscard]] bool authorize(const Context& ctx,
536 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters)
537 std::string_view modelType, std::string_view actionType) const override {
538 const auto claims = _verifier.verify(ctx.token, _clock());
539 if (!claims) {
540 return false;
541 }
542 return !_policy || _policy(*claims, modelType, actionType);
543 }
544
545 /// @brief Returns the verified principal so the server can make it authoritative.
546 /// @param ctx Per-call session (its `token` is verified).
547 /// @return The token's principal if valid, else `nullopt`.
548 [[nodiscard]] std::optional<std::string> authenticate(const Context& ctx) const override {
549 const auto claims = _verifier.verify(ctx.token, _clock());
550 if (!claims) {
551 return std::nullopt;
552 }
553 return claims->principal;
554 }
555
556private:
557 TokenVerifier _verifier;
558 Clock _clock;
559 Policy _policy;
560};
