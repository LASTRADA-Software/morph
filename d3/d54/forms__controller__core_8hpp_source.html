5/// @file
6/// Forms-specific facade over `morph::qt::bridge::GenericModelBridgeCore`:
7/// gives the shipped Qt/QML forms renderer's controller the two operation
8/// names `DynamicForm.qml` actually calls -- `submitIfValid` and
9/// `fetchOptions` -- over the same generic dispatch the composed core
10/// already provides. A concrete `QObject`/`QML_ELEMENT` wrapper per app (Qt
11/// cannot register a class *template* for QML) forwards to this facade and
12/// turns its callbacks into signals -- see
13/// `examples/forms/gui_qml/FormsController.hpp` for the reference wrapper.
14///
15/// Composes rather than inherits `GenericModelBridgeCore`: neither type
16/// declares a virtual destructor (there is no polymorphic dispatch anywhere
17/// in this seam -- every consumer names its concrete `Model` type), and
18/// public inheritance from a class not designed for it invites deleting a
19/// derived object through a base pointer, which is undefined behaviour
20/// without one. Composition sidesteps the question entirely.
21///
22/// See `morph::qt::bridge::GenericModelBridgeCore` for the constructor
23/// overloads (owning vs. composing a `Bridge`) and the underlying dispatch
24/// mechanism.
25
26#include <morph/qt/bridge/generic_model_bridge_core.hpp>
27#include <string>
28#include <utility>
29
30namespace morph::qt::forms {
31
32/// @brief `DynamicForm.qml`-facing facade over
33/// `morph::qt::bridge::GenericModelBridgeCore`, naming its one generic
34/// dispatch operation `submitIfValid`/`fetchOptions` instead of
35/// `execute`, and carrying the `{actionType: schema}` document
36/// `DynamicForm.qml` renders from -- a forms-specific concept the
37/// composed core deliberately does not know about.
38///
39/// @tparam Model The registered model type (`BRIDGE_REGISTER_MODEL`) whose
40/// actions the shipped `DynamicForm.qml` renders.
41/// @tparam Sharing Forwarded to `GenericModelBridgeCore` unchanged --
42/// `morph::bridge::NoSharing` (the default) or
43/// `morph::bridge::AllowShared`. Every shipped forms
44/// controller uses the default; `AllowShared` compiles and
45/// type-checks but has no exercising caller in this repo yet.
46template <typename Model, typename Sharing = ::morph::bridge::NoSharing>
47class FormsControllerCore {
48public:
49 /// @brief Constructs the core with its own private, always-local `Bridge`.
50 /// See `GenericModelBridgeCore`'s own constructor for the full
51 /// contract.
52 /// @param schemasJson The full schema set the QML renderer will parse.
53 explicit FormsControllerCore(std::string schemasJson) : _schemasJson{std::move(schemasJson)} {}
54
55 /// @brief Constructs the core over a caller-supplied `Bridge`/executor.
56 /// See `GenericModelBridgeCore`'s own constructor for the full
57 /// contract.
58 /// @param bridge The bridge to register this core's handler on. Must
59 /// outlive this core.
60 /// @param guiExec Executor used to deliver `Completion` callbacks.
61 /// Must outlive this core.
62 /// @param schemasJson The full schema set the QML renderer will parse.
63 FormsControllerCore(::morph::bridge::Bridge& bridge, ::morph::exec::IExecutor* guiExec, std::string schemasJson)
64 : _core{bridge, guiExec}, _schemasJson{std::move(schemasJson)} {}
65
66 /// @brief The `{actionType: schema}` JSON supplied at construction.
67 /// @return A reference to the cached schema-set JSON.
68 [[nodiscard]] const std::string& schemasJson() const noexcept { return _schemasJson; }
69
70 /// @brief Dispatches @p bodyJson as @p actionType's body via the generic
71 /// `executeJson` path, invoking @p onReply / @p onError on the GUI
72 /// thread once the reply arrives.
73 /// @tparam OnReply Callable invoked with the result JSON (`std::string`) on success.
74 /// @tparam OnError Callable invoked with the `std::exception_ptr` on failure.
75 /// @param actionType Registered action type id.
76 /// @param bodyJson Fully-assembled JSON body for the action.
77 /// @param onReply Success callback.
78 /// @param onError Failure callback.
79 template <typename OnReply, typename OnError>
80 void submitIfValid(const std::string& actionType, const std::string& bodyJson, OnReply onReply, OnError onError) {
81 _core.execute(actionType, bodyJson, std::move(onReply), std::move(onError));
82 }
83
84 /// @brief Executes @p optionsAction with @p bodyJson to fetch a `Choice`
85 /// field's combo-box options, via the same generic `executeJson`
86 /// path `submitIfValid` uses. @p optionsAction is a parameter rather
87 /// than a hardcoded id, and @p bodyJson is a true pass-through (not
88 /// always `"{}"`), so a dependent `Choice` (`x-optionsDependsOn`)
89 /// can send `{parentField: value, ...}` instead of an empty body.
90 /// @tparam OnReply Callable invoked with the options-action result JSON on success.
91 /// @tparam OnError Callable invoked with the `std::exception_ptr` on failure.
92 /// @param optionsAction Registered action type id that serves the options.
93 /// @param bodyJson Fully-assembled JSON body for the options action
94 /// (`"{}"` for an independent `Choice`).
95 /// @param onReply Success callback.
96 /// @param onError Failure callback.
97 template <typename OnReply, typename OnError>
98 void fetchOptions(const std::string& optionsAction, const std::string& bodyJson, OnReply onReply,
99 OnError onError) {
100 _core.execute(optionsAction, bodyJson, std::move(onReply), std::move(onError));
101 }
102
103private:
104 ::morph::qt::bridge::GenericModelBridgeCore<Model, Sharing> _core;
105 std::string _schemasJson;
106};
