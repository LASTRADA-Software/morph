5/// @file
6/// Model-agnostic core of the shipped Qt/QML forms renderer's controller:
7/// owns the Bridge/BridgeHandler/executor wiring `examples/forms/gui_qml`'s
8/// `FormsController` used to hardcode per-app, and exposes the two
9/// operations `DynamicForm.qml` needs -- submit and options-fetch --
10/// generically over `BridgeHandler<Model>::executeJson`, so an app depends on
11/// this directly instead of re-deriving the wiring. A concrete
12/// `QObject`/`QML_ELEMENT` wrapper per app (Qt cannot register a class
13/// *template* for QML) forwards to this core and turns its callbacks into
14/// signals -- see `examples/forms/gui_qml/FormsController.hpp` for the
15/// reference wrapper.
16

27/// @brief Owns the Bridge/BridgeHandler/executor plumbing behind a
28/// schema-driven QML forms controller, generic over the model type.
29///
30/// @tparam Model The registered model type (`BRIDGE_REGISTER_MODEL`) whose
31/// actions the shipped `DynamicForm.qml` renders.
32template <typename Model>
33class FormsControllerCore {
34public:
35 /// @brief Constructs the core with the app's pre-assembled `{actionType:
36 /// schema}` JSON (e.g. hand-written like `lab::schemasJson()`).
37 /// @param schemasJson The full schema set the QML renderer will parse.
38 explicit FormsControllerCore(std::string schemasJson) : _schemasJson{std::move(schemasJson)} {}
39
40 /// @brief The `{actionType: schema}` JSON supplied at construction.
41 /// @return A reference to the cached schema-set JSON.
42 [[nodiscard]] const std::string& schemasJson() const noexcept { return _schemasJson; }
43
44 /// @brief Dispatches @p bodyJson as @p actionType's body via the generic
45 /// `executeJson` path, invoking @p onReply / @p onError on the GUI
46 /// thread once the reply arrives.
47 /// @tparam OnReply Callable invoked with the result JSON (`std::string`) on success.
48 /// @tparam OnError Callable invoked with the `std::exception_ptr` on failure.
49 /// @param actionType Registered action type id.
50 /// @param bodyJson Fully-assembled JSON body for the action.
51 /// @param onReply Success callback.
52 /// @param onError Failure callback.
53 template <typename OnReply, typename OnError>
54 void submitIfValid(std::string actionType, std::string bodyJson, OnReply onReply, OnError onError) {
55 _handler.executeJson(actionType, bodyJson)
56 .then([onReply = std::move(onReply)](std::string resultJson) mutable { onReply(std::move(resultJson)); })
57 .onError([onError = std::move(onError)](const std::exception_ptr& err) mutable { onError(err); });
58 }
59
60 /// @brief Executes @p optionsAction with @p bodyJson to fetch a `Choice`
61 /// field's combo-box options, via the same generic `executeJson`
62 /// path `submitIfValid` uses -- @p optionsAction is never
63 /// hardcoded, unlike the pre-factoring example controller, and
64 /// @p bodyJson is a true pass-through (not always `"{}"`), so a
65 /// dependent `Choice` (`x-optionsDependsOn`) can send
66 /// `{parentField: value, ...}` instead of an empty body.
67 /// @tparam OnReply Callable invoked with the options-action result JSON on success.
68 /// @tparam OnError Callable invoked with the `std::exception_ptr` on failure.
69 /// @param optionsAction Registered action type id that serves the options.
70 /// @param bodyJson Fully-assembled JSON body for the options action
71 /// (`"{}"` for an independent `Choice`).
72 /// @param onReply Success callback.
73 /// @param onError Failure callback.
74 template <typename OnReply, typename OnError>
75 void fetchOptions(std::string optionsAction, std::string bodyJson, OnReply onReply, OnError onError) {
76 _handler.executeJson(optionsAction, bodyJson)
77 .then([onReply = std::move(onReply)](std::string resultJson) mutable { onReply(std::move(resultJson)); })
78 .onError([onError = std::move(onError)](const std::exception_ptr& err) mutable { onError(err); });
79 }
80
81private:
82 // Declaration order matters for destruction, exactly as in the
83 // pre-factoring FormsController: _handler/_bridge must tear down before
84 // _pool/_gui.
85 morph::exec::ThreadPoolExecutor _pool{2};
86 ::morph::qt::QtExecutor _gui;
87 morph::bridge::Bridge _bridge{std::make_unique<morph::backend::LocalBackend>(_pool)};
88 morph::bridge::BridgeHandler<Model> _handler{_bridge, &_gui};
89 std::string _schemasJson;
90};
