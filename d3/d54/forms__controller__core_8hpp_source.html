5/// @file
6/// Model-agnostic core of the shipped Qt/QML forms renderer's controller:
7/// owns (or composes over) the Bridge/BridgeHandler/executor wiring
8/// `examples/forms/gui_qml`'s `FormsController` used to hardcode per-app, and
9/// exposes the two operations `DynamicForm.qml` needs -- submit and
10/// options-fetch -- generically over `BridgeHandler<Model>::executeJson`, so
11/// an app depends on this directly instead of re-deriving the wiring. A
12/// concrete `QObject`/`QML_ELEMENT` wrapper per app (Qt cannot register a
13/// class *template* for QML) forwards to this core and turns its callbacks
14/// into signals -- see `examples/forms/gui_qml/FormsController.hpp` for the
15/// reference wrapper.
16///
17/// Two constructor overloads decide who owns the `Bridge`:
18/// - The single-argument (schema-only) constructor builds and owns a private
19/// `ThreadPoolExecutor` + `QtExecutor` + `Bridge` over a `LocalBackend`,
20/// exactly as before -- the convenient default for a demo or an app that
21/// has no `Bridge` of its own.
22/// - The `(Bridge&, IExecutor*, schemasJson)` constructor composes over a
23/// caller-supplied `Bridge`/executor instead -- the caller decides the
24/// deployment mode (`LocalBackend`, `SimulatedRemoteBackend`,
25/// `QtWebSocketBackend`, ...) and this core never builds a second, always-
26/// local `Bridge` of its own. The caller's `Bridge`/executor must outlive
27/// this core.

40/// @brief Owns, or composes over, the Bridge/BridgeHandler/executor plumbing
41/// behind a schema-driven QML forms controller, generic over the
42/// model type.
43///
44/// @tparam Model The registered model type (`BRIDGE_REGISTER_MODEL`) whose
45/// actions the shipped `DynamicForm.qml` renders.
46template <typename Model>
47class FormsControllerCore {
48public:
49 /// @brief Constructs the core with its own private, always-local `Bridge`
50 /// (`ThreadPoolExecutor` + `QtExecutor` + `LocalBackend`), and the
51 /// app's pre-assembled `{actionType: schema}` JSON (e.g.
52 /// hand-written like `lab::schemasJson()`).
53 ///
54 /// Use the `(Bridge&, IExecutor*, schemasJson)` overload instead when the
55 /// app already has a `Bridge` (remote/socket mode, or one shared across
56 /// multiple presenters) that this core should compose over rather than
57 /// duplicate.
58 /// @param schemasJson The full schema set the QML renderer will parse.
59 explicit FormsControllerCore(std::string schemasJson)
60 : _owned{std::in_place}, _handler{_owned->bridge, &_owned->gui}, _schemasJson{std::move(schemasJson)} {}
61
62 /// @brief Constructs the core over a caller-supplied `Bridge`/executor,
63 /// instead of building a private, always-local one.
64 ///
65 /// The core registers a `BridgeHandler<Model>` on @p bridge exactly as
66 /// the owning constructor's internal one does, so `submitIfValid`/
67 /// `fetchOptions` dispatch through whatever backend @p bridge currently
68 /// has installed (`LocalBackend`, `SimulatedRemoteBackend`,
69 /// `QtWebSocketBackend`, ...) -- including a backend @p bridge switches
70 /// to later via `Bridge::switchBackend`, since the registered handler
71 /// re-registers itself automatically.
72 ///
73 /// @param bridge The bridge to register this core's handler on. Must
74 /// outlive this core.
75 /// @param guiExec Executor used to deliver `Completion` callbacks
76 /// (e.g. a `QtExecutor` for the GUI thread). Must
77 /// outlive this core.
78 /// @param schemasJson The full schema set the QML renderer will parse.
79 FormsControllerCore(morph::bridge::Bridge& bridge, ::morph::exec::IExecutor* guiExec, std::string schemasJson)
80 : _handler{bridge, guiExec}, _schemasJson{std::move(schemasJson)} {}
81
82 /// @brief The `{actionType: schema}` JSON supplied at construction.
83 /// @return A reference to the cached schema-set JSON.
84 [[nodiscard]] const std::string& schemasJson() const noexcept { return _schemasJson; }
85
86 /// @brief Dispatches @p bodyJson as @p actionType's body via the generic
87 /// `executeJson` path, invoking @p onReply / @p onError on the GUI
88 /// thread once the reply arrives.
89 /// @tparam OnReply Callable invoked with the result JSON (`std::string`) on success.
90 /// @tparam OnError Callable invoked with the `std::exception_ptr` on failure.
91 /// @param actionType Registered action type id.
92 /// @param bodyJson Fully-assembled JSON body for the action.
93 /// @param onReply Success callback.
94 /// @param onError Failure callback.
95 template <typename OnReply, typename OnError>
96 void submitIfValid(std::string actionType, std::string bodyJson, OnReply onReply, OnError onError) {
97 _handler.executeJson(actionType, bodyJson)
98 .then([onReply = std::move(onReply)](std::string resultJson) mutable { onReply(std::move(resultJson)); })
99 .onError([onError = std::move(onError)](const std::exception_ptr& err) mutable { onError(err); });
100 }
101
102 /// @brief Executes @p optionsAction with @p bodyJson to fetch a `Choice`
103 /// field's combo-box options, via the same generic `executeJson`
104 /// path `submitIfValid` uses -- @p optionsAction is never
105 /// hardcoded, unlike the pre-factoring example controller, and
106 /// @p bodyJson is a true pass-through (not always `"{}"`), so a
107 /// dependent `Choice` (`x-optionsDependsOn`) can send
108 /// `{parentField: value, ...}` instead of an empty body.
109 /// @tparam OnReply Callable invoked with the options-action result JSON on success.
110 /// @tparam OnError Callable invoked with the `std::exception_ptr` on failure.
111 /// @param optionsAction Registered action type id that serves the options.
112 /// @param bodyJson Fully-assembled JSON body for the options action
113 /// (`"{}"` for an independent `Choice`).
114 /// @param onReply Success callback.
115 /// @param onError Failure callback.
116 template <typename OnReply, typename OnError>
117 void fetchOptions(std::string optionsAction, std::string bodyJson, OnReply onReply, OnError onError) {
118 _handler.executeJson(optionsAction, bodyJson)
119 .then([onReply = std::move(onReply)](std::string resultJson) mutable { onReply(std::move(resultJson)); })
120 .onError([onError = std::move(onError)](const std::exception_ptr& err) mutable { onError(err); });
121 }
122
123private:
124 /// @brief The private pool/executor/backend bundle the schema-only
125 /// constructor builds and owns, exactly as `FormsControllerCore`
126 /// always did before the `(Bridge&, IExecutor*, ...)` overload
127 /// existed. Absent (`_owned` unengaged) when the core instead
128 /// composes over a caller-supplied `Bridge`/executor.
129 ///
130 /// Declaration order within the struct matters for destruction: `bridge`
131 /// must tear down before `pool`/`gui`, exactly as the pre-factoring
132 /// `FormsController` required.
133 struct OwnedBridge {
134 morph::exec::ThreadPoolExecutor pool{2};
135 ::morph::qt::QtExecutor gui;
136 morph::bridge::Bridge bridge{std::make_unique<morph::backend::LocalBackend>(pool)};
137 };
138
139 // Declaration order matters for destruction: _handler must tear down
140 // before _owned (its bridge and executor, when this core owns them), so
141 // _owned is declared first and _handler after it. When the caller-
142 // supplied-Bridge constructor is used, _owned stays unengaged and
143 // _handler instead references the caller's Bridge/executor directly --
144 // the caller is responsible for outliving _handler in that case.
145 std::optional<OwnedBridge> _owned;
146 morph::bridge::BridgeHandler<Model> _handler;
147 std::string _schemasJson;
148};
