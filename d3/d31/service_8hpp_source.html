410/// @brief Whether an authoriser is asked about one of the UI service's reads: `ui`'s `hello`, `catalog`, `fetch` or
411/// `texts`.
412///
413/// An application that serves a login screen admits these for a session with no principal yet, so a client can greet
414/// the service, read the catalog, fetch the login screen and show its fields' texts in the person's language.
415/// Admitting them discloses no other screen: `ui-catalog` lists, and `ui-fetch` serves, a non-public screen only to a
416/// principal who may run its queries, which the authoriser still decides. `ui-texts` answers the application's
417/// words, as a bundled message catalogue holds them, and no data.
418/// @param modelType The model type asked about.
419/// @param actionType The action asked about.
420/// @return Whether it is one of those reads.
421[[nodiscard]] constexpr bool isUiServiceRead(std::string_view modelType, std::string_view actionType) noexcept {
422 return modelType == "ui" &&
423 (actionType == "hello" || actionType == "catalog" || actionType == "fetch" || actionType == "texts");
424}

426/// @brief The UI service: answers `ui-hello`, `ui-catalog` and `ui-fetch` from one `ServedBundle`, and `ui-evaluate`
427/// through the server's `ActionDispatcher`, as an envelope service of a `RemoteServer`
428/// (`ServerConfig::services`).
429///
430/// Every request is authorised (`docs/spec/ui/service.md`, "Authorisation"):
431///
432/// | Kind | Gate |
433/// |---|---|
434/// | `ui-hello` | `authorize(session, "ui", "hello")` |
435/// | `ui-catalog` | `authorize(session, "ui", "catalog")` |
436/// | `ui-fetch` of the manifest or a public file | the `ui-hello` gate |
437/// | `ui-fetch` of any other file | `authorize(session, "ui", "fetch")`; for a screen, also the catalog's rule |
438/// | `ui-evaluate` | `authorize(session, modelType, actionType)`: as the action it evaluates |
439/// | `ui-texts` | `authorize(session, "ui", "texts")` |
440///
441/// A refusal is `err "unauthorized"`. A request body that is too large, malformed or carries a must-understand key
442/// is refused with the parser's reason.
443class MORPH_API UiService final : public ::morph::backend::IEnvelopeService {
444public:
445 /// @brief Serves @p bundle, and evaluates through @p dispatcher.
446 /// @param bundle The bundle; shared, never changed.
447 /// @param dispatcher The dispatcher `ui-evaluate` reads the actions from: the server's own. Borrowed: it must
448 /// outlive the service.
449 /// @throws std::invalid_argument if @p bundle is null.
450 /// @param files The file side channel's store; `nullptr` serves no `ui-file`.
451 explicit UiService(std::shared_ptr<ServedBundle const> bundle,
452 ::morph::model::detail::ActionDispatcher& dispatcher MORPH_LIFETIMEBOUND =
453 ::morph::model::detail::defaultDispatcher(),
454 std::shared_ptr<IFileStore> files = nullptr);
455
456 /// @brief Whether @p kind is one of the service's kinds.
457 /// @param kind The envelope kind.
458 /// @return `true` for `ui-hello`, `ui-catalog`, `ui-fetch`, `ui-evaluate` and `ui-texts`, and for `ui-file` when
459 /// the service has a file store.
460 [[nodiscard]] bool serves(std::string_view kind) const override;
461
462 /// @brief Answers one request with no connection behind it: a `ui-hello` offer of events is accepted in the
463 /// reply, and recorded nowhere.
464 /// @param request The envelope, its principal verified.
465 /// @param authorizer The server's authoriser.
466 /// @return The encoded reply.
467 [[nodiscard]] std::string handle(::morph::wire::Envelope const& request,
468 ::morph::session::IAuthorizer const& authorizer) override;
469
470 /// @brief Answers one request; a `ui-hello` that offers events and is answered `ok` sets `facts.events`.
471 ///
472 /// A hello that does not offer them leaves the facts as they are, so a later hello on the same connection, such
473 /// as a development reload's poll, does not withdraw what the first negotiated.
474 /// @param request The envelope, its principal verified.
475 /// @param authorizer The server's authoriser.
476 /// @param facts The facts of the connection the request arrived on.
477 /// @return The encoded reply.
478 [[nodiscard]] std::string handle(::morph::wire::Envelope const& request,
479 ::morph::session::IAuthorizer const& authorizer,
480 ::morph::backend::ConnectionFacts& facts) override;

507 [[nodiscard]] static bool gate(::morph::session::IAuthorizer const& authorizer,
