95/// @brief An application's bundle, built and immutable: what `UiService` serves.
96struct ServedBundle {
97 /// @brief The application.
98 ApplicationInfo application;
99 /// @brief The vocabularies the bundle needs, sorted; always `expr/1` and `ui/1`.
100 std::vector<std::string> vocabularies;
101 /// @brief The manifest's text.
102 std::string manifest;
103 /// @brief The manifest's content hash, which `ui-hello` replies with.
104 std::string digest;
105 /// @brief Every file by its content hash, the manifest included.
106 std::map<std::string, ServedFile, std::less<>> files;
107 /// @brief The catalog's screens, in registration order.
108 std::vector<ServedScreen> screens;
109 /// @brief The actions the app document names.
110 std::vector<ActionRef> appActions;
111 /// @brief The model types the app document names.
112 std::vector<std::string> appModelTypes;
113 /// @brief Every action any served document names, sorted and unique: the candidates of `permitted`.
114 std::vector<ActionRef> allActions;
115 /// @brief The model facts the registry's catalog gave, for every model type a document names; absent when it
116 /// has no model lookup.
117 std::optional<std::map<std::string, ModelInfo, std::less<>>> models;
118 /// @brief The action facts the registry's catalog gave, for every action a document names; absent when it has
119 /// no action lookup.
120 std::optional<std::map<std::string, ActionInfo, std::less<>>> actions;
121 /// @brief The view documents the registry's catalog gave, by the action of their `v-query`, for every action a
122 /// document names; absent when it has no view lookup.
123 std::optional<std::map<std::string, std::string, std::less<>>> views;
124 /// @brief The wizard documents the registry's catalog gave, by the action of their first step, for every action
125 /// a document names; absent when it has no wizard lookup.
126 std::optional<std::map<std::string, std::string, std::less<>>> wizards;
127};

131/// @brief Whether @p bytes can be served: within `kMaxFileBytes`, and within the envelope bound once escaped into
132/// an `ok` reply.
133/// @param bytes The file.
134/// @return Why it cannot, or nothing.
135[[nodiscard]] inline std::optional<std::string> checkServable(std::string const& bytes) {
136 if (bytes.size() > kMaxFileBytes) {
137 return "the file is larger than " + std::to_string(kMaxFileBytes) + " bytes";
138 }
139 if (::morph::wire::encode(::morph::wire::makeOk(0, bytes)).size() > ::morph::wire::kMaxEnvelopeBytes) {
140 return std::string{"the file does not fit one envelope once escaped"};
141 }
142 return std::nullopt;
143}

145/// @brief Checks a catalog entry's own fields.
146/// @param entry The entry.
147/// @return Why it is refused, or nothing.
148[[nodiscard]] inline std::optional<std::string> checkEntry(CatalogEntry const& entry) {
149 if (entry.id.empty()) {
150 return std::string{"a catalog entry has an id"};
151 }
152 if (auto reason = checkName(entry.id, "a catalog entry id")) {
153 return reason;
154 }
155 if (entry.whenDenied != "hide" && entry.whenDenied != "disable") {
156 return "`whenDenied` is `hide` or `disable`, not `" + entry.whenDenied + "`";
157 }
158 return std::nullopt;
159}

163/// @brief Collects the actions and model types a loaded document names, resolving each query's, mutation's and
164/// model-bound node's alias through the enclosing scopes.
165///
166/// Load-time validation has already resolved every alias and requires every alias to name a type, so each lookup
167/// here finds one.
168class DocumentActions {
169public:
170 /// @brief Walks @p document.
171 /// @param document The loaded document.
172 explicit DocumentActions(Document const& document) {
173 enter(document.scope, true);
174 if (document.view) {
175 node(*document.view);
176 }
177 }
178
179 /// @brief The top-level queries.
180 /// @return Them, in declaration order.
181 [[nodiscard]] std::vector<ActionRef> const& topQueries() const noexcept { return _top; }
182
183 /// @brief Every action named.
184 /// @return Them, sorted and unique.
185 [[nodiscard]] std::vector<ActionRef> actions() const { return {_all.begin(), _all.end()}; }
186
187 /// @brief Every model type named.
188 /// @return Them, sorted and unique.
189 [[nodiscard]] std::vector<std::string> modelTypes() const { return {_models.begin(), _models.end()}; }
190
191private:
192 void enter(ScopeDecl const& scope, bool top) {
193 for (ModelDecl const& model : scope.models) {
194 _types.try_emplace(model.name).first->second.push_back(model.type);
195 _models.insert(model.type);
196 }
197 for (QueryDecl const& query : scope.queries) {
198 ActionRef ref{.modelType = resolve(query.model), .action = query.action};
199 if (top) {
200 _top.push_back(ref);
201 }
202 _all.insert(std::move(ref));
203 }
204 for (MutationDecl const& mutation : scope.mutations) {
205 _all.insert(ActionRef{.modelType = resolve(mutation.model), .action = mutation.action});
206 }
207 }
208
209 void leave(ScopeDecl const& scope) {
210 for (ModelDecl const& model : scope.models) {
211 _types.at(model.name).pop_back();
212 }
213 }
214
215 void node(DocNode const& view) {
216 if (view.scope) {
217 enter(*view.scope, false);
218 }
219 if (std::string_view const model = view.name("model"); !model.empty()) {
220 _all.insert(ActionRef{.modelType = resolve(model), .action = std::string{view.name("action")}});
221 }
222 for (auto const& [key, children] : view.children) {
223 static_cast<void>(key);
224 for (DocNodePtr const& child : children) {
225 node(*child);
226 }
227 }
228 for (auto const& [key, parts] : view.parts) {
229 static_cast<void>(key);
230 for (DocNodePtr const& part : parts) {
231 node(*part);
232 }
233 }
234 if (view.scope) {
235 leave(*view.scope);
236 }
237 }
238
239 /// @brief The type of the innermost alias named @p alias. Load-time validation has resolved it already, so the
240 /// lookup cannot miss; `at` throws rather than read past the end if that ever changes.
241 [[nodiscard]] std::string const& resolve(std::string_view alias) const {
242 std::vector<std::string> const& types = _types.at(std::string{alias});
243 return types.at(types.size() - 1);
244 }
245
246 /// @brief The types an alias names in the open scopes, innermost last.
247 std::map<std::string, std::vector<std::string>, std::less<>> _types;
248 std::vector<ActionRef> _top;
249 std::set<ActionRef> _all;
250 std::set<std::string> _models;
251};

255/// @brief Whether the principal of @p session may run every action of @p refs.
256/// @param authorizer The authoriser.
257/// @param session The session, its principal verified.
258/// @param refs The actions.
259/// @return Whether it may run all of them; `true` for none.
260[[nodiscard]] inline bool permitsAll(::morph::session::IAuthorizer const& authorizer,
261 ::morph::session::Context const& session, std::vector<ActionRef> const& refs) {
262 return std::ranges::all_of(
263 refs, [&](ActionRef const& ref) { return authorizer.authorize(session, ref.modelType, ref.action); });

280/// @brief Collects an application's screens and files and builds the bundle the UI service serves.
281///
282/// Built once, before the service answers anything; `build` copies everything into an immutable `ServedBundle`.
283/// A screen authored with the C++ builders is added by its emitted text, so an authoring registry feeds this one
284/// through `addScreen` without either depending on the other.
285class UiRegistry {
286public:
287 /// @brief An empty registry.
288 /// @param application The application `ui-hello` names.
289 /// @param catalog The catalog facts: load-time validation reads them at registration, and the
290 /// `ui-catalog` reply carries them to the client. A lookup left empty is neither checked nor
291 /// served.
292 explicit UiRegistry(ApplicationInfo application, Catalog catalog = {})
293 : _application{std::move(application)}, _catalog{std::move(catalog)} {}
294
295 /// @brief Registers a screen: its catalog entry and its document.
296 ///
297 /// Refused when the entry's id is empty, longer than `kMaxNameBytes`, already registered, or not the document's
298 /// id; when `whenDenied` is neither `hide` nor `disable`; when the entry names a `host` (a host screen has no
299 /// document: `addEntry`); when the document is `app` (`addFile`); when the text is too large to serve; and when
300 /// load-time validation refuses the document.
301 /// @param entry The catalog entry.
302 /// @param text The document's JSON text, served byte for byte.
303 /// @return Nothing, or the refusal.
304 std::expected<void, Refusal> addScreen(CatalogEntry entry, std::string text) {
305 if (auto refusal = checkNewEntry(entry)) {
306 return std::unexpected(std::move(*refusal));
307 }
308 if (entry.host) {
309 return refuse(entry.id, "a host screen has no document; it is added with addEntry");
310 }
311 if (auto reason = detail::checkServable(text)) {
312 return refuse(entry.id, std::move(*reason));
313 }
314 auto loaded = loadDocumentText(text, LoadOptions{.catalog = _catalog});
315 if (!loaded) {
316 return std::unexpected(std::move(loaded.error()));
317 }
318 if (loaded->id != entry.id) {
319 return refuse(entry.id, "the entry's id is not the document's id `" + loaded->id + "`");
320 }
321 if (!loaded->view) {
322 return refuse(entry.id, "the app document is added with addFile");
323 }
324 logWarnings(*loaded);
325 detail::DocumentActions const named{*loaded};
326 Screen screen{.served = ServedScreen{.entry = std::move(entry),
327 .hash = contentHash(text),
328 .topQueries = named.topQueries(),
329 .actions = named.actions(),
330 .modelTypes = named.modelTypes()},
331 .text = std::move(text),
332 .components = {}};
333 for (auto const& [name, component] : loaded->components) {
334 static_cast<void>(name);
335 if (!component.qml.empty()) {
336 screen.components.emplace_back(pointerAppend(component.path, "qml"), component.qml);
337 }
338 }
339 _screens.push_back(std::move(screen));
340 return {};
341 }
342
343 /// @brief Lists an entry that has no document: a host screen (`host`), or one the application has not built
344 /// (`available: false`).
345 ///
346 /// Refused on the same entry rules as `addScreen`, and when the entry is neither a host screen nor unavailable.
347 /// @param entry The catalog entry.
348 /// @return Nothing, or the refusal.
349 std::expected<void, Refusal> addEntry(CatalogEntry entry) {
350 if (auto refusal = checkNewEntry(entry)) {
351 return std::unexpected(std::move(*refusal));
352 }
353 if (!entry.host && entry.available) {
354 return refuse(entry.id, "an entry without a document is a host screen or unavailable");
355 }
356 _screens.push_back(Screen{
357 .served =
358 ServedScreen{.entry = std::move(entry), .hash = {}, .topQueries = {}, .actions = {}, .modelTypes = {}},
359 .text = {},
360 .components = {}});
361 return {};
362 }

364 /// @brief Adds a file that is not a screen: the app document (`app`, whose id is `app`), a theme (`theme`), a
365 /// message catalogue (`messages`), a delivered component (`component`) or an icon (`icon`).
366 ///
367 /// Refused for another kind, an empty or over-long id, a kind and id already added, a file too large to serve,
368 /// an app document that load-time validation refuses or whose id is not `app`, a message catalogue whose id is
369 /// not `i18n/LOCALE.json`, that `parseCatalogue` refuses, or that has a pattern outside the `ui/1` subset, and a
370 /// theme file that is neither a theme index `parseThemeIndex` takes nor a token file `parseTokens` takes, and an
371 /// icon that is not SVG text or is larger than `kMaxIconBytes`.
372 /// @param file The file.
373 /// @return Nothing, or the refusal.
374 std::expected<void, Refusal> addFile(BundleFile file) {
375 if (file.kind != kAppFile && file.kind != kThemeFile && file.kind != kMessagesFile &&
376 file.kind != kComponentFile && file.kind != kIconFile) {
377 return refuse(file.id,
378 "a file is an app document, a theme, a message catalogue, a component or an icon, "
379 "not `" +
380 file.kind + "`");
381 }
382 if (file.id.empty()) {
383 return refuse(file.id, "a file has an id");
384 }
385 if (auto reason = detail::checkName(file.id, "a file id")) {
386 return refuse(file.id, std::move(*reason));
387 }
388 if (std::ranges::any_of(
389 _files, [&file](BundleFile const& added) { return added.kind == file.kind && added.id == file.id; })) {
390 return refuse(file.id, "the " + file.kind + " `" + file.id + "` is already added");
391 }
392 if (auto reason = detail::checkServable(file.bytes)) {
393 return refuse(file.id, std::move(*reason));
394 }
395 if (auto added = addKind(file); !added) {
396 return added;
397 }
398 _files.push_back(std::move(file));
399 return {};
400 }

402 /// @brief Adds a vocabulary the bundle needs beyond `ui/1` and `expr/1`, such as `forms/1` or `files/1`. A
403 /// client that does not speak it is refused at `ui-hello`.
404 /// @param vocabulary The vocabulary.
405 void requireVocabulary(std::string vocabulary) { _vocabularies.insert(std::move(vocabulary)); }
406
407 /// @brief Builds the bundle.
408 ///
409 /// Refused when a screen names a delivered component (`qml`) whose hash no `component` file has, when the
410 /// bundle holds more than `kMaxBundleFiles` files or entries, and when it needs more than `kMaxListItems`
411 /// vocabularies or a vocabulary name longer than `kMaxNameBytes`.
412 /// @return The bundle, or the refusal.
413 [[nodiscard]] std::expected<std::shared_ptr<ServedBundle const>, Refusal> build() const {
414 if (auto refusal = checkBuildable()) {
415 return std::unexpected(std::move(*refusal));
416 }
417 auto bundle = std::make_shared<ServedBundle>();
418 bundle->application = _application;
419 bundle->vocabularies.assign(_vocabularies.begin(), _vocabularies.end());
420 bundle->appActions = _appActions;
421 bundle->appModelTypes = _appModelTypes;
422 serveFiles(*bundle);
423 addFacts(*bundle);
424 return bundle;
425 }

427private:
428 /// @brief Checks every theme index: each combination of its variants, densities and surfaces over `base`
429 /// resolves from the bundle's token files (`checkTheme`).
430 /// @return The refusal, naming the index and the combination, or nothing.
431 [[nodiscard]] std::optional<Refusal> checkThemes() const {
432 TokenFileLookup const lookup = [this](std::string_view fileId) -> TokenSet const* {
433 auto const found = _tokenFiles.find(fileId);
434 return found == _tokenFiles.end() ? nullptr : &found->second;
435 };
436 for (auto const& [indexId, index] : _themeIndexes) {
437 if (auto reason = checkTheme(index, lookup)) {
438 return Refusal{.document = indexId, .path = {}, .reason = "the theme: " + std::move(*reason)};
439 }
440 for (auto const& [name, fileId] : index.icons) {
441 bool const held = std::ranges::any_of(
442 _files, [&fileId](BundleFile const& file) { return file.kind == kIconFile && file.id == fileId; });
443 if (!held) {
444 return Refusal{.document = indexId,
445 .path = {},
446 .reason = std::format("the theme index names the icon `{}` as `{}`, which is not "
447 "an icon file of the bundle",
448 name, fileId)};
449 }
450 }
451 }
452 return std::nullopt;
453 }

455 /// @brief A registered screen and what `build` needs of its document.
456 struct Screen {
457 /// @brief What the bundle serves.
458 ServedScreen served;
459 /// @brief The document's text; empty for an entry without one.
460 std::string text;
461 /// @brief The delivered components the document names: their `qml` pointer and hash.
462 std::vector<std::pair<std::string, std::string>> components;

465 // The checks and the indexing one kind of file adds to addFile's common ones.
466 std::expected<void, Refusal> addKind(BundleFile const& file) {
467 if (file.kind == kMessagesFile) {
468 return checkCatalogue(file);
469 }
470 if (file.kind == kThemeFile) {
471 return addTheme(file);
472 }
473 if (file.kind == kAppFile) {
474 return addApp(file);
475 }
476 if (file.kind == kIconFile) {
477 return checkIcon(file);
478 }
479 return {};
480 }

482 // An icon is SVG text, small enough for a theme to carry a set of them.
483 [[nodiscard]] static std::expected<void, Refusal> checkIcon(BundleFile const& file) {
484 if (file.bytes.size() > kMaxIconBytes) {
485 return refuse(file.id, std::format("an icon is at most {} bytes of SVG; this one is {}", kMaxIconBytes,
486 file.bytes.size()));
487 }
488 if (!file.bytes.contains("<svg")) {
489 return refuse(file.id, "an icon is SVG text, and this one has no <svg> element");
490 }
491 return {};
492 }
493
494 [[nodiscard]] static std::expected<void, Refusal> checkCatalogue(BundleFile const& file) {
495 if (!catalogueLocale(file.id)) {
496 return refuse(file.id, "a message catalogue's id is i18n/<locale>.json");
497 }
498 auto catalogue = parseCatalogue(file.bytes);
499 if (!catalogue) {
500 return refuse(file.id, std::move(catalogue.error()));
501 }
502 if (!catalogue->dropped.empty()) {
503 return refuse(file.id, "a message is outside the ui/1 subset: " + catalogue->dropped.front());
504 }
505 return {};
506 }
507
508 std::expected<void, Refusal> addTheme(BundleFile const& file) {
509 if (isThemeIndex(file.bytes)) {
510 auto index = parseThemeIndex(file.bytes);
511 if (!index) {
512 return refuse(file.id, "the theme index: " + index.error());
513 }
514 _themeIndexes.emplace_back(file.id, std::move(*index));
515 return {};
516 }
517 auto tokens = parseTokens(file.bytes);
518 if (!tokens) {
519 return refuse(file.id, "the token file: " + tokens.error());
520 }
521 _tokenFiles.emplace(file.id, std::move(*tokens));
522 return {};
523 }

525 std::expected<void, Refusal> addApp(BundleFile const& file) {
526 auto loaded = loadDocumentText(file.bytes, LoadOptions{.catalog = _catalog});
527 if (!loaded) {
528 return std::unexpected(std::move(loaded.error()));
529 }
530 if (loaded->id != kAppFile) {
531 return refuse(file.id, "the app document's id is `app`");
532 }
533 logWarnings(*loaded);
534 detail::DocumentActions const named{*loaded};
535 _appActions = named.actions();
536 _appModelTypes = named.modelTypes();
537 return {};
538 }

548 /// @brief A refusal of the registration of @p document.
549 /// @param document The screen or file id.
550 /// @param reason Why.
551 /// @return The refusal, as an `std::unexpected`.
552 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): the id first, then the reason, at every call.
553 [[nodiscard]] static std::unexpected<Refusal> refuse(std::string document, std::string reason) {
554 return std::unexpected(Refusal{.document = std::move(document), .path = {}, .reason = std::move(reason)});

557 /// @brief Checks an entry about to be registered: its own fields, and that its id is new.
558 /// @param entry The entry.
559 /// @return The refusal, or nothing.
560 [[nodiscard]] std::optional<Refusal> checkNewEntry(CatalogEntry const& entry) const {
561 if (auto reason = detail::checkEntry(entry)) {
562 return Refusal{.document = entry.id, .path = {}, .reason = std::move(*reason)};
563 }
564 if (std::ranges::any_of(_screens, [&](Screen const& screen) { return screen.served.entry.id == entry.id; })) {
565 return Refusal{.document = entry.id, .path = {}, .reason = "a catalog entry with this id is registered"};
566 }
567 return std::nullopt;
568 }

570 /// @brief What `build` refuses: a delivered component missing from the bundle, a theme some combination of which
571 /// does not resolve, and the bounds.
572 /// @return The refusal, or nothing.
573 [[nodiscard]] std::optional<Refusal> checkBuildable() const {
574 if (auto refusal = checkThemes()) {
575 return refusal;
576 }
577 std::set<std::string, std::less<>> componentHashes;
578 for (BundleFile const& file : _files) {
579 if (file.kind == kComponentFile) {
580 componentHashes.insert(contentHash(file.bytes));
581 }
582 }
583 for (Screen const& screen : _screens) {
584 for (auto const& [path, hash] : screen.components) {
585 if (!componentHashes.contains(hash)) {
586 return Refusal{.document = screen.served.entry.id,
587 .path = path,
588 .reason = "the delivered component is not in the bundle"};
589 }
590 }
591 }
592 if (_screens.size() > kMaxBundleFiles || _screens.size() + _files.size() > kMaxBundleFiles) {
593 return Refusal{
594 .document = {},
595 .path = {},
596 .reason = "the bundle holds more than " + std::to_string(kMaxBundleFiles) + " files or entries"};
597 }
598 std::vector<std::string> const vocabularies{_vocabularies.begin(), _vocabularies.end()};
599 if (auto reason = detail::checkNames(vocabularies, kMaxListItems, "the vocabularies", "a vocabulary")) {
600 return Refusal{.document = {}, .path = {}, .reason = std::move(*reason)};
601 }
602 return std::nullopt;
603 }

605 /// @brief Fills @p bundle's screens, files, manifest and digest.
606 /// @param bundle The bundle being built.
607 void serveFiles(ServedBundle& bundle) const {
608 Manifest manifest{.application = _application.id, .version = _application.version, .files = {}};
609 auto serve = [&bundle, &manifest](BundleFile const& file, std::optional<std::size_t> screen) {
610 std::string const hash = contentHash(file.bytes);
611 manifest.files.push_back(ManifestFile{
612 .id = file.id, .kind = file.kind, .hash = hash, .size = file.bytes.size(), .isPublic = file.isPublic});
613 ServedFile& served = bundle.files.try_emplace(hash).first->second;

617 };

643 /// @brief Fills @p bundle's action candidates and, from the registry's catalog, its facts.
644 /// @param bundle The bundle being built, its screens already served.
645 void addFacts(ServedBundle& bundle) const {
646 std::set<ActionRef> all{_appActions.begin(), _appActions.end()};
647 std::set<std::string, std::less<>> modelTypes{_appModelTypes.begin(), _appModelTypes.end()};
648 for (ServedScreen const& screen : bundle.screens) {
649 all.insert(screen.actions.begin(), screen.actions.end());
650 modelTypes.insert(screen.modelTypes.begin(), screen.modelTypes.end());
651 }
652 bundle.allActions.assign(all.begin(), all.end());
653 if (_catalog.model) {
654 auto& known = bundle.models.emplace();
655 for (std::string const& type : modelTypes) {
656 // Load-time validation refused every document naming a type the lookup does not know.
657 known.emplace(type, _catalog.model(type).value_or(ModelInfo{}));
658 }
659 }
660 if (_catalog.action) {
661 auto& known = bundle.actions.emplace();
662 for (ActionRef const& ref : all) {
663 // Load-time validation refused every document naming an action the lookup does not know.
664 known.emplace(ref.action, _catalog.action(ref.action).value_or(ActionInfo{}));
665 }
666 }
667 auto const documents = [&all](auto const& lookup, auto& out) {
668 if (!lookup) {
669 return;
670 }
671 auto& known = out.emplace();
672 for (ActionRef const& ref : all) {
673 if (std::optional<std::string> document = lookup(ref.action)) {
674 known.emplace(ref.action, std::move(*document));
675 }
676 }
677 };
678 documents(_catalog.view, bundle.views);
679 documents(_catalog.wizard, bundle.wizards);
680 }

695/// @brief Whether an authoriser is asked about one of the UI service's reads: `ui`'s `hello`, `catalog` or `fetch`.
696///
697/// An application that serves a login screen admits these for a session with no principal yet, so a client can greet
698/// the service, read the catalog and fetch the login screen. Admitting them discloses no other screen: `ui-catalog`
699/// lists, and `ui-fetch` serves, a non-public screen only to a principal who may run its queries, which the
700/// authoriser still decides.
701/// @param modelType The model type asked about.
702/// @param actionType The action asked about.
703/// @return Whether it is one of those reads.
704[[nodiscard]] constexpr bool isUiServiceRead(std::string_view modelType, std::string_view actionType) noexcept {
705 return modelType == "ui" && (actionType == "hello" || actionType == "catalog" || actionType == "fetch");
706}

708/// @brief The UI service: answers `ui-hello`, `ui-catalog` and `ui-fetch` from one `ServedBundle`, and `ui-evaluate`
709/// through the server's `ActionDispatcher`, as an envelope service of a `RemoteServer`
710/// (`ServerConfig::services`).
711///
712/// Every request is authorised (`docs/spec/ui/service.md`, "Authorisation"):
713///
714/// | Kind | Gate |
715/// |---|---|
716/// | `ui-hello` | `authorize(session, "ui", "hello")` |
717/// | `ui-catalog` | `authorize(session, "ui", "catalog")` |
718/// | `ui-fetch` of the manifest or a public file | the `ui-hello` gate |
719/// | `ui-fetch` of any other file | `authorize(session, "ui", "fetch")`; for a screen, also the catalog's rule |
720/// | `ui-evaluate` | `authorize(session, modelType, actionType)`: as the action it evaluates |
721///
722/// A refusal is `err "unauthorized"`. A request body that is too large, malformed or carries a must-understand key
723/// is refused with the parser's reason.
724class UiService final : public ::morph::backend::IEnvelopeService {
725public:
726 /// @brief Serves @p bundle, and evaluates through @p dispatcher.
727 /// @param bundle The bundle; shared, never changed.
728 /// @param dispatcher The dispatcher `ui-evaluate` reads the actions from: the server's own. Borrowed: it must
729 /// outlive the service.
730 /// @throws std::invalid_argument if @p bundle is null.
731 /// @param files The file side channel's store; `nullptr` serves no `ui-file`.
732 explicit UiService(std::shared_ptr<ServedBundle const> bundle,
733 ::morph::model::detail::ActionDispatcher& dispatcher MORPH_LIFETIMEBOUND =
734 ::morph::model::detail::defaultDispatcher(),
735 std::shared_ptr<IFileStore> files = nullptr)
736 : _bundle{std::move(bundle)}, _dispatcher{&dispatcher}, _files{std::move(files)} {
737 if (!_bundle) {
738 throw std::invalid_argument("UiService: the bundle is null");
739 }
740 }
741
742 /// @brief Whether @p kind is one of the service's kinds.
743 /// @param kind The envelope kind.
744 /// @return `true` for `ui-hello`, `ui-catalog`, `ui-fetch` and `ui-evaluate`.
745 [[nodiscard]] bool serves(std::string_view kind) const override {
746 return kind == kHelloKind || kind == kCatalogKind || kind == kFetchKind || kind == kEvaluateKind ||
747 (kind == kFileKind && _files != nullptr);
748 }

750 /// @brief Answers one request with no connection behind it: a `ui-hello` offer of events is accepted in the
751 /// reply, and recorded nowhere.
752 /// @param request The envelope, its principal verified.
753 /// @param authorizer The server's authoriser.
754 /// @return The encoded reply.
755 [[nodiscard]] std::string handle(::morph::wire::Envelope const& request,
756 ::morph::session::IAuthorizer const& authorizer) override {
757 ::morph::backend::ConnectionFacts facts;
758 return handle(request, authorizer, facts);
759 }
760
761 /// @brief Answers one request; a `ui-hello` that offers events and is answered `ok` sets `facts.events`.
762 ///
763 /// A hello that does not offer them leaves the facts as they are, so a later hello on the same connection, such
764 /// as a development reload's poll, does not withdraw what the first negotiated.
765 /// @param request The envelope, its principal verified.
766 /// @param authorizer The server's authoriser.
767 /// @param facts The facts of the connection the request arrived on.
768 /// @return The encoded reply.
769 [[nodiscard]] std::string handle(::morph::wire::Envelope const& request,
770 ::morph::session::IAuthorizer const& authorizer,
771 ::morph::backend::ConnectionFacts& facts) override {
772 if (request.kind == kHelloKind) {
773 return hello(request, authorizer, facts);
774 }
775 if (request.kind == kCatalogKind) {
776 return catalog(request, authorizer);
777 }
778 if (request.kind == kEvaluateKind) {
779 return evaluate(request, authorizer);
780 }
781 if (request.kind == kFileKind) {
782 return file(request, authorizer);
783 }
784 return fetch(request, authorizer);
785 }

807 [[nodiscard]] std::string hello(::morph::wire::Envelope const& request,
808 ::morph::session::IAuthorizer const& authorizer,
809 ::morph::backend::ConnectionFacts& facts) const {
810 if (!gate(authorizer, request, kHelloAction)) {
811 return err(request, "unauthorized");
812 }
813 auto client = parseHelloRequest(request.body);
814 if (!client) {
815 return err(request, std::move(client.error()));
816 }
817 std::string missing;
818 for (std::string const& vocabulary : _bundle->vocabularies) {
819 if (std::ranges::find(client->vocabularies, vocabulary) == client->vocabularies.end()) {
820 missing += (missing.empty() ? "" : ", ") + vocabulary;
821 }
822 }
823 if (!missing.empty()) {
824 return err(request, "missing vocabulary: " + missing);
825 }
826 if (client->events) {
827 facts.events = true;
828 }
829 return ok(request, detail::writeMessage(HelloReply{.application = _bundle->application.id,
830 .version = _bundle->application.version,
831 .vocabularies = _bundle->vocabularies,
832 .digest = _bundle->digest,
833 .events = client->events}));
834 }

836 /// @brief Evaluates the action body @p request carries (spec 5 §7): decoded, reconciled and recomputed, never
837 /// executed, and answered with its field errors. Synchronous on the server strand, like the other kinds.
838 /// @param request The envelope: `modelType`, `actionType` and the action's JSON `body`.
839 /// @param authorizer The server's authoriser.
840 /// @return The encoded reply: an `EvaluateReply`, or `err` for a refusal or a body that does not decode.
841 [[nodiscard]] std::string evaluate(::morph::wire::Envelope const& request,
842 ::morph::session::IAuthorizer const& authorizer) const {
843 if (request.modelType.empty() || request.actionType.empty()) {
844 return err(request, "ui-evaluate names no modelType and actionType");
845 }
846 if (!authorizer.authorize(request.session, request.modelType, request.actionType)) {
847 return err(request, "unauthorized");
848 }
849 auto evaluation = _dispatcher->evaluate(request.modelType, request.actionType, request.body);
850 EvaluateReply reply{.body = {}, .errors = std::move(evaluation.errors), .ready = evaluation.ready};
851 if (auto const error = glz::read_json(reply.body, evaluation.body)) {
852 return err(request, "ui-evaluate: " + glz::format_error(error, evaluation.body));
853 }
854 return ok(request, detail::writeMessage(reply));
855 }

857 /// @brief One step of an upload or a download through the file side channel (spec 6 §18).
858 ///
859 /// `begin` is authorised as the mutation that will reference the file; every later step is authorised by the
860 /// upload id or reference, which belong to the principal that created them. A refusal of the store is `err` with
861 /// its text; an id or reference of another principal is `not found`, like one that does not exist.
862 /// @param request The envelope; its body is a `FileRequest`.
863 /// @param authorizer The server's authoriser.
864 /// @return The encoded reply: a `FileReply`, or `err`.
865 [[nodiscard]] std::string file(::morph::wire::Envelope const& request,
866 ::morph::session::IAuthorizer const& authorizer) const {
867 auto parsed = parseFileRequest(request.body);
868 if (!parsed) {
869 return err(request, std::move(parsed.error()));
870 }
871 FileRequest& step = *parsed;
872 std::string const& principal = request.session.principal;
873 auto const refused = [&request](FileError error) { return err(request, std::string{fileErrorText(error)}); };
874 FileReply reply;
875 if (step.op == "begin") {
876 if (step.model.empty() || step.action.empty()) {
877 return err(request, "ui-file begin names no model and action");
878 }
879 if (!authorizer.authorize(request.session, step.model, step.action)) {
880 return err(request, "unauthorized");
881 }
882 auto upload = _files->begin(
883 principal, FileInfo{.name = std::move(step.name), .type = std::move(step.type), .size = step.size});
884 if (!upload) {
885 return refused(upload.error());
886 }
887 reply.upload = std::move(*upload);
888 } else if (step.op == "chunk") {
889 auto bytes = ::morph::session::detail::base64UrlDecode(step.data);
890 if (!bytes) {
891 return err(request, "ui-file chunk data is not base64url");
892 }
893 if (bytes->size() > kMaxFileChunkBytes) {
894 return err(request, "ui-file chunk is longer than " + std::to_string(kMaxFileChunkBytes) + " bytes");
895 }
896 auto received = _files->append(principal, step.upload, step.offset, *bytes);
897 if (!received) {
898 return refused(received.error());
899 }
900 reply.received = *received;
901 } else if (step.op == "end") {
902 auto ref = _files->finish(principal, step.upload);
903 if (!ref) {
904 return refused(ref.error());
905 }
906 reply.ref = std::move(*ref);
907 } else {
908 auto slice = _files->read(principal, step.ref, step.offset, std::min(step.length, kMaxFileChunkBytes));
909 if (!slice) {
910 return refused(slice.error());
911 }
912 reply.data = ::morph::session::detail::base64UrlEncode(slice->bytes);
913 reply.total = slice->info.size;
914 reply.name = std::move(slice->info.name);
915 reply.type = std::move(slice->info.type);
916 }
917 return ok(request, detail::writeMessage(reply));
918 }

920 [[nodiscard]] std::string catalog(::morph::wire::Envelope const& request,
921 ::morph::session::IAuthorizer const& authorizer) const {
922 if (!gate(authorizer, request, kCatalogAction)) {
923 return err(request, "unauthorized");
924 }
925 CatalogReply reply;
926 Fetchable const fetchable = listEntries(reply, request.session, authorizer);
927 for (ActionRef const& ref : _bundle->allActions) {
928 if (authorizer.authorize(request.session, ref.modelType, ref.action)) {
929 reply.permitted.push_back(ref.action);
930 }
931 }
932 if (_bundle->models) {
933 auto& known = reply.models.emplace();
934 for (auto const& [type, info] : *_bundle->models) {
935 if (fetchable.modelTypes.contains(type)) {
936 known.emplace(type, info);
937 }
938 }
939 }
940 if (_bundle->actions) {
941 auto& known = reply.actions.emplace();
942 for (auto const& [action, info] : *_bundle->actions) {
943 if (fetchable.actions.contains(action)) {
944 known.emplace(action, info);
945 }
946 }
947 }
948 // A collection's or wizard's document goes with the action that names it, to whoever may fetch that action.
949 auto const documents = [&fetchable](auto const& served, auto& out) {
950 if (!served) {
951 return;
952 }
953 auto& known = out.emplace();
954 for (auto const& [action, document] : *served) {
955 if (fetchable.actions.contains(action)) {
956 known.emplace(action, document);
957 }
958 }
959 };
960 documents(_bundle->views, reply.views);
961 documents(_bundle->wizards, reply.wizards);
962 return ok(request, detail::writeMessage(reply));
963 }

965 /// @brief Lists the entries the principal sees into @p reply.
966 /// @param reply The reply being built.
967 /// @param session The session, its principal verified.
968 /// @param authorizer The server's authoriser.
969 /// @return What the documents the principal may fetch name.
970 [[nodiscard]] Fetchable listEntries(CatalogReply& reply, ::morph::session::Context const& session,
971 ::morph::session::IAuthorizer const& authorizer) const {
972 Fetchable fetchable{.modelTypes = {_bundle->appModelTypes.begin(), _bundle->appModelTypes.end()},
973 .actions = {}};
974 for (ActionRef const& ref : _bundle->appActions) {
975 fetchable.actions.insert(ref.action);
976 }
977 for (ServedScreen const& screen : _bundle->screens) {
978 CatalogEntry entry = screen.entry;
979 if (entry.available && !detail::permitsAll(authorizer, session, screen.topQueries)) {
980 if (entry.whenDenied == "hide") {
981 continue;
982 }
983 entry.available = false;
984 entry.reason = "denied";
985 }
986 if (entry.available || entry.isPublic) {
987 fetchable.modelTypes.insert(screen.modelTypes.begin(), screen.modelTypes.end());
988 for (ActionRef const& ref : screen.actions) {
989 fetchable.actions.insert(ref.action);
990 }
991 }
992 reply.entries.push_back(std::move(entry));
993 }
994 return fetchable;
995 }

997 [[nodiscard]] std::string fetch(::morph::wire::Envelope const& request,
998 ::morph::session::IAuthorizer const& authorizer) const {
999 auto wanted = parseFetchRequest(request.body);
1000 if (!wanted) {
1001 return err(request, std::move(wanted.error()));
1002 }
1003 auto const found = _bundle->files.find(wanted->hash);
1004 bool const isPublic = found != _bundle->files.end() &&
1005 std::ranges::any_of(found->second.holders,
1006 [](ServedFile::Holder const& holder) { return holder.isPublic; });
1007 if (isPublic) {
1008 return gate(authorizer, request, kHelloAction) ? ok(request, found->second.bytes)
1009 : err(request, "unauthorized");
1010 }
1011 // The fetch gate comes before the lookup's answer, so a caller it refuses cannot learn which hashes exist.
1012 if (!gate(authorizer, request, kFetchAction)) {
1013 return err(request, "unauthorized");
1014 }
1015 if (found == _bundle->files.end()) {
1016 return err(request, "not found");
1017 }
1018 bool const admitted = std::ranges::any_of(found->second.holders, [&](ServedFile::Holder const& holder) {
1019 return !holder.screen || screenAdmits(_bundle->screens.at(*holder.screen), request.session, authorizer);
1020 });
1021 return admitted ? ok(request, found->second.bytes) : err(request, "unauthorized");
1022 }

1024 /// @brief Whether the principal may fetch @p screen's document: it is available, and the catalog lists it as

1027 ::morph::session::IAuthorizer const& authorizer) {
1028 return screen.entry.available && detail::permitsAll(authorizer, session, screen.topQueries);
1029 }
