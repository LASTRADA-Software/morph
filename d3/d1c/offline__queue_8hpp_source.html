17/// @brief An item stored in the offline queue.
18///
19/// The payload is an opaque string — the caller controls the serialisation
20/// format (JSON, binary-hex, plain text, etc.).
21struct QueueItem {
22 /// @brief Stable identifier assigned at enqueue time.
23 ///
24 /// Local to *this* queue instance; it is **not** a cross-subsystem
25 /// idempotency key (a durable queue re-presents the same logical op with a
26 /// fresh `id` after a restart, and the journal never sees it). Use
27 /// `idempotencyKey` to dedup a replay against already-applied ops.
28 uint64_t id{};
29
30 /// @brief Opaque serialised representation of the queued action.
31 std::string payload;
32
33 /// @brief Optional caller-supplied idempotency key that is **stable across
34 /// subsystems and process restarts** for one logical operation.
35 ///
36 /// Empty by default. When set, it is the shared dedup token that lets a
37 /// replay consumer recognise an op the journal (or a previous replay) has
38 /// already applied and skip it, instead of double-applying. The offline
39 /// queue and the journal replay have no shared identity otherwise — the
40 /// queue's `id` is queue-local and the journal's `seq` is journal-local —
41 /// so a host that replays through both paths must wire this key (see
42 /// `docs/spec/offline.md`, "Idempotency key: deduping against the journal").
43 /// The value is opaque to the queue; a good choice is a stable content hash
44 /// or a client-minted operation id (e.g. a UUID) reused if the same op is
45 /// re-enqueued. The queue does not interpret, require, or enforce
46 /// uniqueness on it — enforcement is the replay consumer's job.
47 std::string idempotencyKey;
48
49 /// @brief Durable retry count for this item, authoritative when the queue
50 /// persists it.
51 ///
52 /// Defaults to `0`. A queue that overrides `IOfflineQueue::setAttempts()`
53 /// to store this value makes the retry budget survive a process restart:
54 /// `SyncWorker` seeds its own attempt counter from the larger of this
55 /// field and its in-memory count, and writes the updated count back via
56 /// `setAttempts()` after every failed replay. A queue that leaves
57 /// `setAttempts()` as the default no-op never advances this field, so
58 /// `SyncWorker`'s in-memory counter stays authoritative — today's
59 /// process-local retry behavior, unchanged.
60 uint32_t attempts{0};
61};
62
63/// @brief Thrown by `enqueue()` when the queue is at its configured `maxDepth()`.
64///
65/// `IOfflineQueue` enforces a reject-newest overflow policy: once the queue
66/// holds `maxDepth()` items, a further `enqueue()` throws instead of silently
67/// evicting an older item or invoking an app-defined callback. Evicting the
68/// oldest item would destroy data the caller believes durable and break
69/// replay ordering with no error raised anywhere; a host that genuinely wants
70/// different eviction semantics already has the seam for it — subclass
71/// `IOfflineQueue` directly rather than layering a second policy mechanism on
72/// top of this one.
73struct OfflineQueueFullError : std::runtime_error {
74 /// @param maxDepthValue The configured capacity that was reached.
75 /// @param currentSizeValue Number of pending items at the time of rejection
76 /// (equal to maxDepth for a well-behaved implementation).
77 OfflineQueueFullError(std::size_t maxDepthValue, std::size_t currentSizeValue)
78 : std::runtime_error("IOfflineQueue: enqueue rejected, queue is at capacity (" +
79 std::to_string(currentSizeValue) + "/" + std::to_string(maxDepthValue) + ")"),
80 maxDepth{maxDepthValue}, currentSize{currentSizeValue} {}
81
82 /// @brief The configured capacity that was reached.
83 std::size_t maxDepth;
84 /// @brief Number of pending items at the time of rejection.
85 std::size_t currentSize;
86};

90/// @brief Interface for durable storage of actions that could not be delivered.
91///
92/// Items accumulate while the system is offline and are replayed by `SyncWorker`
93/// on reconnect. The interface is intentionally minimal so that implementations
94/// can range from in-memory (`InMemoryOfflineQueue`) to SQLite or file-backed stores.
95// NOLINTBEGIN(cppcoreguidelines-special-member-functions)
96struct IOfflineQueue {
97 virtual ~IOfflineQueue() = default;
98
99 /// @brief Appends @p payload to the queue with no idempotency key.
100 ///
101 /// @param payload Serialised action to persist.
102 /// @return A stable id that can be passed to `markDone()`.
103 virtual uint64_t enqueue(std::string payload) = 0;
104
105 /// @brief Appends @p payload carrying the cross-subsystem @p idempotencyKey.
106 ///
107 /// The key is stored on the resulting `QueueItem::idempotencyKey` so a
108 /// replay consumer can dedup this op against ones the journal (or a prior
109 /// replay) already applied (see `QueueItem::idempotencyKey` and
110 /// `docs/spec/offline.md`). The queue neither interprets nor enforces
111 /// uniqueness on the key.
112 ///
113 /// The default implementation delegates to `enqueue(std::move(payload))`
114 /// and then stamps the key via `setIdempotencyKey`, so existing
115 /// `IOfflineQueue` implementations keep working without overriding it.
116 /// @param payload Serialised action to persist.
117 /// @param idempotencyKey Stable dedup token for the logical op; may be empty.
118 /// @return A stable id that can be passed to `markDone()`.
119#pragma GCC diagnostic push
120#pragma GCC diagnostic ignored "-Woverloaded-virtual"
121 virtual uint64_t enqueue(std::string payload, std::string idempotencyKey) {
122 const uint64_t itemId = enqueue(std::move(payload));
123 setIdempotencyKey(itemId, std::move(idempotencyKey));
124 return itemId;
125 }
126#pragma GCC diagnostic pop
127
128 /// @brief Returns all pending items in enqueue order without removing them.
129 ///
130 /// Items remain in the queue until `markDone()` is called. It is safe to
131 /// call `drain()` multiple times — items survive a crash between `drain()`
132 /// and the corresponding `markDone()` call.
133 /// @return Snapshot of all pending items.
134 virtual std::vector<QueueItem> drain() const = 0;
135
136 /// @brief Removes the item identified by @p itemId.
137 ///
138 /// No-op if @p itemId is not found.
139 /// @param itemId Id returned by the corresponding `enqueue()` call.
140 virtual void markDone(uint64_t itemId) = 0;
141
142 /// @brief Returns the number of pending items without removing them.
143 ///
144 /// Default implementation calls `drain().size()` — correct but O(n) and
145 /// allocates a full snapshot vector to answer a size query. Override for
146 /// an O(1) or index-backed answer.
147 /// @return Current pending item count.
148 virtual std::size_t size() const { return drain().size(); }
149
150 /// @brief Returns the configured maximum depth, or `std::nullopt` if unbounded.
151 ///
152 /// Default: `std::nullopt` (unbounded) — preserves current behavior for any
153 /// `IOfflineQueue` subclass that predates this method.
154 /// @return The capacity `enqueue()` enforces, or `std::nullopt` if none.
155 virtual std::optional<std::size_t> maxDepth() const { return std::nullopt; }
156
157 /// @brief Persists an updated attempt count for an item. Default: no-op.
158 ///
159 /// A durable queue overrides this to store the count so the retry budget
160 /// survives a restart — `SyncWorker` calls it after every failed replay,
161 /// and reads the persisted value back through `QueueItem::attempts` on
162 /// the next `drain()` (this run, or after a restart). `InMemoryOfflineQueue`
163 /// overrides it to update the in-deque item; a queue that leaves it as
164 /// the default no-op keeps the pre-existing process-local retry behavior
165 /// (`SyncWorker`'s own in-memory counter is then always authoritative,
166 /// since `QueueItem::attempts` never advances).
167 /// @param itemId Id of the item whose attempt count changed.
168 /// @param attempts New cumulative attempt count to persist.
169 virtual void setAttempts([[maybe_unused]] uint64_t itemId, [[maybe_unused]] uint32_t attempts) {}
170
171protected:
172 /// @brief Stamps an idempotency key onto an already-enqueued item.
173 ///
174 /// Called by the default `enqueue(payload, key)` overload so implementations
175 /// that only override the single-argument `enqueue` still support keys. The
176 /// default is a no-op (an implementation with no per-item key storage simply
177 /// drops it); `InMemoryOfflineQueue` overrides it to record the key.
178 /// @param itemId Id of the item to stamp.
179 /// @param idempotencyKey Key to store; ignored by the default no-op.
180 virtual void setIdempotencyKey([[maybe_unused]] uint64_t itemId, [[maybe_unused]] std::string idempotencyKey) {}
181};
182// NOLINTEND(cppcoreguidelines-special-member-functions)

184// ── In-memory implementation ──────────────────────────────────────────────────

186/// @brief Thread-safe in-memory implementation of `IOfflineQueue`.
187///
188/// Suitable for testing and for applications that do not require persistence
189/// across process restarts. Items are stored in a `std::deque` protected by a mutex.
190class InMemoryOfflineQueue : public IOfflineQueue {
191public:
192 using IOfflineQueue::enqueue; // keep the two-arg overload visible
193
194 /// @brief Constructs an in-memory queue, optionally bounded.
195 /// @param maxDepth Maximum number of pending items `enqueue()` will admit
196 /// before throwing `OfflineQueueFullError`; `std::nullopt` (the
197 /// default) means unbounded.
198 explicit InMemoryOfflineQueue(std::optional<std::size_t> maxDepth = std::nullopt) : _maxDepth{maxDepth} {}
199
200 /// @brief Appends @p payload and returns a monotonically increasing id.
201 /// @param payload Serialised action to store.
202 /// @return Unique id for this item.
203 uint64_t enqueue(std::string payload) override { return enqueue(std::move(payload), {}); }
204
205 /// @brief Appends @p payload carrying @p idempotencyKey and returns a
206 /// monotonically increasing id.
207 /// @param payload Serialised action to store.
208 /// @param idempotencyKey Stable dedup token; stored verbatim on the item.
209 /// @return Unique id for this item.
210 /// @throws OfflineQueueFullError if the queue is already at `maxDepth()`.
211 uint64_t enqueue(std::string payload, std::string idempotencyKey) override {
212 std::scoped_lock const lock{_mtx};
213 if (_maxDepth && _items.size() >= *_maxDepth) {
214 ::morph::observe::detail::emitMetric(::morph::observe::Metric::queueOverflow,
215 static_cast<double>(_items.size()));
216 throw OfflineQueueFullError{*_maxDepth, _items.size()};
217 }
218 uint64_t const itemId = ++_nextId;
219 _items.push_back(
220 QueueItem{.id = itemId, .payload = std::move(payload), .idempotencyKey = std::move(idempotencyKey)});
221 return itemId;
222 }
223
224 /// @brief Returns a snapshot of all pending items. Thread-safe.
225 /// @return Copy of all items in insertion order.
226 std::vector<QueueItem> drain() const override {
227 std::scoped_lock const lock{_mtx};
228 return std::vector<QueueItem>{_items.begin(), _items.end()};
229 }
230
231 /// @brief Returns the number of pending items. Thread-safe.
232 /// @return Current pending item count.
233 std::size_t size() const override {
234 std::scoped_lock const lock{_mtx};
235 return _items.size();
236 }
237
238 /// @brief Returns the configured maximum depth, or `std::nullopt` if unbounded.
239 /// @return The capacity `enqueue()` enforces, or `std::nullopt` if none.
240 std::optional<std::size_t> maxDepth() const override { return _maxDepth; }
241
242 /// @brief Removes the item with @p itemId from the queue. Thread-safe.
243 ///
244 /// No-op if @p itemId is not found.
245 /// @param itemId Id of the item to remove.
246 void markDone(uint64_t itemId) override {
247 std::scoped_lock const lock{_mtx};
248 auto iter = std::ranges::find_if(_items, [itemId](const QueueItem& item) { return item.id == itemId; });
249 if (iter != _items.end()) {
250 _items.erase(iter);
251 }
252 }
253
254 /// @brief Updates the persisted attempt count on the in-deque item with
255 /// @p itemId. No-op if @p itemId is not found. Thread-safe.
256 ///
257 /// `InMemoryOfflineQueue` does not itself survive a process restart, but
258 /// overriding this hook lets a *fresh* `SyncWorker` constructed over the
259 /// same instance observe a durable-style cumulative attempt count — used
260 /// to simulate cross-restart dead-lettering in tests.
261 /// @param itemId Id of the item to update.
262 /// @param attempts New attempt count to store.
263 void setAttempts(uint64_t itemId, uint32_t attempts) override {
264 std::scoped_lock const lock{_mtx};
265 auto iter = std::ranges::find_if(_items, [itemId](const QueueItem& item) { return item.id == itemId; });
266 if (iter != _items.end()) {
267 iter->attempts = attempts;
268 }
269 }
270
271private:
272 mutable std::mutex _mtx;
273 std::deque<QueueItem> _items;
274 uint64_t _nextId{0};
275 std::optional<std::size_t> _maxDepth;
276};
