19/// @brief A replay attempt count, as a type distinct from a queue item id.
20///
21/// `setAttempts(itemId, attempts)` takes two integers that mean entirely
22/// different things, and before this type existed both were plain unsigned
23/// integers, mutually convertible in either direction. Transposing them
24/// compiled, and — because `setAttempts()` on an unknown id is a documented
25/// no-op — it also *ran*: the real item's count never advanced, nothing threw,
26/// and the defect surfaced much later as a retry budget that never exhausts.
27///
28/// `Attempts` removes the hazard rather than suppressing the warning about it.
29/// It is implicitly constructible from a narrow integer, so `setAttempts(id, 3)`
30/// reads exactly as it did before, and **deliberately not constructible from a
31/// 64-bit integer**, which is what a `QueueItem::id` is — so the transposed
32/// call does not compile. A genuinely 64-bit count is still expressible, with
33/// the narrowing spelled out at the call site:
34/// `setAttempts(id, static_cast<std::uint32_t>(count))`.
35class Attempts {
36public:
37 /// @brief Constructs a zero attempt count.
38 constexpr Attempts() noexcept = default;
39
40 /// @brief Implicitly wraps a narrow integral attempt count.
41 ///
42 /// Constrained to integral types **strictly narrower than a queue item
43 /// id**, which is the whole point of the class: `uint64_t` (and any other
44 /// 64-bit integer) is rejected, so passing an item id where an attempt
45 /// count belongs is a compile error rather than a silent no-op. `bool` is
46 /// excluded because a boolean is never an attempt count.
47 ///
48 /// @tparam Count Integral type of the incoming count.
49 /// @param count The attempt count to wrap.
50 template <typename Count>
51 requires(std::integral<Count> && !std::same_as<std::remove_cv_t<Count>, bool> &&
52 sizeof(Count) < sizeof(std::uint64_t))
53 constexpr Attempts(Count count) noexcept : _value{static_cast<std::uint32_t>(count)} {}
54
55 /// @brief Returns the wrapped count.
56 /// @return The attempt count as a plain `uint32_t`.
57 [[nodiscard]] constexpr std::uint32_t value() const noexcept { return _value; }
58
59 /// @brief Compares two attempt counts.
60 /// @param other The count to compare against.
61 /// @return `true` when both wrap the same value.
62 [[nodiscard]] constexpr bool operator==(const Attempts& other) const noexcept = default;
63
64private:
65 std::uint32_t _value{0};
66};

68static_assert(!std::is_constructible_v<Attempts, std::uint64_t>,
69 "Attempts must not be constructible from a queue item id, or setAttempts()'s two "
70 "parameters become mutually convertible again and a transposed call compiles.");
71static_assert(std::is_convertible_v<std::uint32_t, Attempts>,
72 "Attempts must stay implicitly constructible from a narrow count, so existing "
73 "setAttempts(id, n) call sites keep reading the way they did.");

75/// @brief An item stored in the offline queue.
76///
77/// The payload is an opaque string — the caller controls the serialisation
78/// format (JSON, binary-hex, plain text, etc.).
79struct QueueItem {
80 /// @brief Stable identifier assigned at enqueue time.
81 ///
82 /// Local to *this* queue instance; it is **not** a cross-subsystem
83 /// idempotency key (it is queue-local and the journal never sees it; both
84 /// shipped durable queues re-present the *stored* id after a restart). Use
85 /// `idempotencyKey` to dedup a replay against already-applied ops.
86 uint64_t id{};
87
88 /// @brief Opaque serialised representation of the queued action.
89 std::string payload;
90
91 /// @brief Optional caller-supplied idempotency key that is **stable across
92 /// subsystems and process restarts** for one logical operation.
93 ///
94 /// Empty by default. When set, it is the shared dedup token that lets a
95 /// replay consumer recognise an op the journal (or a previous replay) has
96 /// already applied and skip it, instead of double-applying. The offline
97 /// queue and the journal replay have no shared identity otherwise — the
98 /// queue's `id` is queue-local and the journal's `seq` is journal-local —
99 /// so a host that replays through both paths must wire this key (see
100 /// `docs/spec/offline/offline.md`, "`idempotencyKey`: deduping against the journal").
101 /// The value is opaque to the queue; a good choice is a stable content hash
102 /// or a client-minted operation id (e.g. a UUID) reused if the same op is
103 /// re-enqueued.
104 ///
105 /// The queue never *interprets* the key, and the interface does not
106 /// **require** an implementation to enforce uniqueness on it — a replay
107 /// consumer must therefore always dedup on the key itself, because a
108 /// conforming queue may hand it the same key twice. An implementation is
109 /// nonetheless **permitted** to dedup a non-empty key at enqueue time as a
110 /// strengthening of that floor; see `IOfflineQueue::enqueue(std::string,
111 /// std::string)` for exactly what a dedup hit does.
112 std::string idempotencyKey;
113
114 /// @brief Durable retry count for this item, authoritative when the queue
115 /// persists it.
116 ///
117 /// Defaults to `0`. A queue that overrides `IOfflineQueue::setAttempts()`
118 /// to store this value makes the retry budget survive a process restart:
119 /// `SyncWorker` seeds its own attempt counter from the larger of this
120 /// field and its in-memory count, and writes the updated count back via
121 /// `setAttempts()` after every failed replay. A queue that leaves
122 /// `setAttempts()` as the default no-op never advances this field, so
123 /// `SyncWorker`'s in-memory counter stays authoritative — today's
124 /// process-local retry behavior, unchanged.
125 uint32_t attempts{0};
126};
127
128/// @brief Thrown by `enqueue()` when the queue is at its configured `maxDepth()`.
129///
130/// `IOfflineQueue` enforces a reject-newest overflow policy: once the queue
131/// holds `maxDepth()` items, a further `enqueue()` throws instead of silently
132/// evicting an older item or invoking an app-defined callback. Evicting the
133/// oldest item would destroy data the caller believes durable and break
134/// replay ordering with no error raised anywhere; a host that genuinely wants
135/// different eviction semantics already has the seam for it — subclass
136/// `IOfflineQueue` directly rather than layering a second policy mechanism on
137/// top of this one.
138struct OfflineQueueFullError : std::runtime_error {
139 /// @param maxDepthValue The configured capacity that was reached.
140 /// @param currentSizeValue Number of pending items at the time of rejection
141 /// (equal to maxDepth for a well-behaved implementation).
142 OfflineQueueFullError(std::size_t maxDepthValue, std::size_t currentSizeValue)
143 : std::runtime_error("IOfflineQueue: enqueue rejected, queue is at capacity (" +
144 std::to_string(currentSizeValue) + "/" + std::to_string(maxDepthValue) + ")"),
145 maxDepth{maxDepthValue},
146 currentSize{currentSizeValue} {}
147
148 /// @brief The configured capacity that was reached.
149 std::size_t maxDepth;
150 /// @brief Number of pending items at the time of rejection.
151 std::size_t currentSize;
152};

156/// @brief Interface for durable storage of actions that could not be delivered.
157///
158/// Items accumulate while the system is offline and are replayed by `SyncWorker`
159/// on reconnect. The interface is intentionally minimal so that implementations
160/// can range from in-memory (`InMemoryOfflineQueue`) to SQLite or file-backed stores.
161// NOLINTBEGIN(cppcoreguidelines-special-member-functions)
162struct IOfflineQueue {
163 virtual ~IOfflineQueue() = default;
164
165 /// @brief Appends @p payload to the queue with no idempotency key.
166 ///
167 /// @param payload Serialised action to persist.
168 /// @return A stable id that can be passed to `markDone()`.
169 [[nodiscard]] virtual uint64_t enqueue(std::string payload) = 0;
170
171 /// @brief Appends @p payload carrying the cross-subsystem @p idempotencyKey.
172 ///
173 /// The key is stored on the resulting `QueueItem::idempotencyKey` so a
174 /// replay consumer can dedup this op against ones the journal (or a prior
175 /// replay) already applied (see `QueueItem::idempotencyKey` and
176 /// `docs/spec/offline/offline.md`).
177 ///
178 /// @par Uniqueness is a floor, not a prohibition
179 /// The queue never interprets the key, and this interface does not
180 /// *require* uniqueness enforcement: a conforming implementation may store
181 /// the same non-empty key twice, so a replay consumer must dedup on the key
182 /// regardless. Enqueue-time dedup is an allowed **strengthening**, not a
183 /// contract violation. Of the implementations morph ships,
184 /// `InMemoryOfflineQueue` never dedups, while `FileOfflineQueue` (linear
185 /// scan) and `SqliteOfflineQueue` (partial unique index) both do.
186 ///
187 /// @par What a dedup hit does, in every implementation that dedups
188 /// - It applies only to a **non-empty** key already carried by a *pending*
189 /// item. An empty key is never a dedup token — two empty-key enqueues
190 /// always produce two distinct items, in every implementation.
191 /// - The call **succeeds** and returns the **existing** item's id rather
192 /// than a fresh one, so the return value is not a reliable signal that
193 /// anything was stored.
194 /// - It is **first-write-wins with silent payload loss**: @p payload is
195 /// discarded, the pending item keeps the payload it already had, and no
196 /// error is raised and nothing is reported to the caller. A caller that
197 /// re-enqueues a *corrected* payload under an unchanged key therefore
198 /// loses the correction — mint a new key when the payload changes.
199 /// - `markDone()` releases the key: once the pending item is gone, the same
200 /// key enqueues normally again.
201 /// - It survives a restart in a durable queue — re-enqueuing a key that a
202 /// pending persisted item still carries is a hit after a reopen.
203 ///
204 /// These guarantees are pinned for every shipped implementation by
205 /// `tests/offline_queue_conformance.hpp`, which is told which policy each
206 /// implementation has and asserts it.
207 ///
208 /// The default implementation delegates to `enqueue(std::move(payload))`
209 /// and then stamps the key via `setIdempotencyKey`, so existing
210 /// `IOfflineQueue` implementations keep working without overriding it.
211 ///
212 /// @par Non-atomicity of this default (implementors, read this)
213 /// This default makes **two separate virtual calls**
214 /// (`enqueue(payload)`, then `setIdempotencyKey(itemId, key)`), each
215 /// independently locking and unlocking whatever internal mutex the
216 /// concrete class uses — nothing is held across both calls. A concurrent
217 /// `markDone(itemId)` racing in the window between them could erase the
218 /// item before the key is stamped (silently dropping the key rather than
219 /// throwing), and a concurrent `drain()` in that same window can observe
220 /// the item with an *empty* idempotency key. A subclass that stores keys
221 /// and cares about atomicity **must override the two-argument `enqueue`**
222 /// and stamp the key under the same lock acquisition that inserts the
223 /// item, exactly as every implementation morph ships already does
224 /// (`FileOfflineQueue`, `InMemoryOfflineQueue`, `SqliteOfflineQueue` all
225 /// override it inline) — this base default exists only so a subclass
226 /// that has not been updated yet keeps compiling and working for the
227 /// non-concurrent case, not as an atomicity guarantee.
228 /// @param payload Serialised action to persist.
229 /// @param idempotencyKey Stable dedup token for the logical op; may be empty.
230 /// @return A stable id that can be passed to `markDone()`.
231#pragma GCC diagnostic push
232#pragma GCC diagnostic ignored "-Woverloaded-virtual"
233 [[nodiscard]] virtual uint64_t enqueue(std::string payload, std::string idempotencyKey) {
234 const uint64_t itemId = enqueue(std::move(payload));
235 setIdempotencyKey(itemId, std::move(idempotencyKey));
236 return itemId;
237 }
238#pragma GCC diagnostic pop
239
240 /// @brief Returns all pending items in enqueue order without removing them.
241 ///
242 /// Items remain in the queue until `markDone()` is called. It is safe to
243 /// call `drain()` multiple times — items survive a crash between `drain()`
244 /// and the corresponding `markDone()` call.
245 ///
246 /// @par Enqueue order is the implementation's to keep; the id does not imply it
247 /// "Enqueue order" is a property this method **requires** and that
248 /// `QueueItem::id` does not supply. Every implementation morph ships mints
249 /// ids that happen to increase with insertion — an in-memory counter, a
250 /// file offset, SQLite's `INTEGER PRIMARY KEY AUTOINCREMENT` — so each of
251 /// them satisfies this method by presenting rows ordered by id. That is a
252 /// property of those three stores, not of the type: a `uint64_t` neither
253 /// promises monotonicity nor rules out an id minted from a GUID, a content
254 /// hash, or a sharded sequence, and a store that reuses the id of a removed
255 /// row does not order correctly either. An implementation over such a store
256 /// must carry its own insertion sequence and order on *that*; ordering by
257 /// the id would compile, survive a casual smoke test, and replay a user's
258 /// actions out of order. `tests/offline_queue_conformance.hpp` asserts the
259 /// ordering, so an implementation that gets this wrong fails there.
260 /// @return Snapshot of all pending items.
261 [[nodiscard]] virtual std::vector<QueueItem> drain() const = 0;
262
263 /// @brief Removes the item identified by @p itemId.
264 ///
265 /// No-op if @p itemId is not found.
266 /// @param itemId Id returned by the corresponding `enqueue()` call.
267 virtual void markDone(uint64_t itemId) = 0;
268
269 /// @brief Returns the number of pending items without removing them.
270 ///
271 /// Default implementation calls `drain().size()` — correct but O(n) and
272 /// allocates a full snapshot vector to answer a size query. Override for
273 /// an O(1) or index-backed answer.
274 /// @return Current pending item count.
275 [[nodiscard]] virtual std::size_t size() const { return drain().size(); }
276
277 /// @brief Returns the configured maximum depth, or `std::nullopt` if unbounded.
278 ///
279 /// Default: `std::nullopt` (unbounded) — preserves current behavior for any
280 /// `IOfflineQueue` subclass that predates this method.
281 /// @return The capacity `enqueue()` enforces, or `std::nullopt` if none.
282 [[nodiscard]] virtual std::optional<std::size_t> maxDepth() const { return std::nullopt; }
283
284 /// @brief Persists an updated attempt count for an item. Default: no-op.
285 ///
286 /// A durable queue overrides this to store the count so the retry budget
287 /// survives a restart — `SyncWorker` calls it after every failed replay,
288 /// and reads the persisted value back through `QueueItem::attempts` on
289 /// the next `drain()` (this run, or after a restart). `InMemoryOfflineQueue`
290 /// overrides it to update the in-deque item; a queue that leaves it as
291 /// the default no-op keeps the pre-existing process-local retry behavior
292 /// (`SyncWorker`'s own in-memory counter is then always authoritative,
293 /// since `QueueItem::attempts` never advances).
294 /// @param itemId Id of the item whose attempt count changed.
295 /// @param attempts New cumulative attempt count to persist. Typed, not a
296 /// bare integer, so a transposed call does not compile — see
297 /// `Attempts` for what that silently did before.
298 virtual void setAttempts([[maybe_unused]] uint64_t itemId, [[maybe_unused]] Attempts attempts) {}
299
300protected:
301 /// @brief Stamps an idempotency key onto an already-enqueued item.
302 ///
303 /// Called by the default `enqueue(payload, key)` overload so implementations
304 /// that only override the single-argument `enqueue` still support keys. The
305 /// default is a no-op (an implementation with no per-item key storage simply
306 /// drops it); `InMemoryOfflineQueue` overrides it to record the key.
307 /// @param itemId Id of the item to stamp.
308 /// @param idempotencyKey Key to store; ignored by the default no-op.
309 virtual void setIdempotencyKey([[maybe_unused]] uint64_t itemId, [[maybe_unused]] std::string idempotencyKey) {}
310};
311// NOLINTEND(cppcoreguidelines-special-member-functions)

313// The point of `Attempts`, stated as something the compiler checks rather than
314// as a comment. The second assertion is what keeps the first from being
315// vacuous: if `Attempts` were an alias for `uint32_t` the transposed call would
316// compile and the first assertion would fail, and if it were made explicit the
317// ordinary call would stop compiling and the second would fail. Both directions
318// have to hold.
319static_assert(!std::is_invocable_v<decltype(&IOfflineQueue::setAttempts), IOfflineQueue&, uint32_t, uint64_t>,
320 "setAttempts's parameters must not be transposable: writing an item id into an attempt count is "
321 "silent, because setAttempts on an unknown id is a documented no-op.");
322static_assert(std::is_invocable_v<decltype(&IOfflineQueue::setAttempts), IOfflineQueue&, uint64_t, uint32_t>,
323 "setAttempts must still accept a plain count in the right order.");

327/// @brief Thread-safe in-memory implementation of `IOfflineQueue`.
328///
329/// Suitable for testing and for applications that do not require persistence
330/// across process restarts. Items are stored in a `std::deque` protected by a mutex.
331class InMemoryOfflineQueue : public IOfflineQueue {
332public:
333 using IOfflineQueue::enqueue; // keep the two-arg overload visible
334
335 /// @brief Constructs an in-memory queue, optionally bounded.
336 /// @param maxDepth Maximum number of pending items `enqueue()` will admit
337 /// before throwing `OfflineQueueFullError`; `std::nullopt` (the
338 /// default) means unbounded.
339 explicit InMemoryOfflineQueue(std::optional<std::size_t> maxDepth = std::nullopt) : _maxDepth{maxDepth} {}
340
341 /// @brief Appends @p payload and returns a monotonically increasing id.
342 /// @param payload Serialised action to store.
343 /// @return Unique id for this item.
344 [[nodiscard]] uint64_t enqueue(std::string payload) override { return enqueue(std::move(payload), {}); }
345
346 /// @brief Appends @p payload carrying @p idempotencyKey and returns a
347 /// monotonically increasing id.
348 /// @param payload Serialised action to store.
349 /// @param idempotencyKey Stable dedup token; stored verbatim on the item.
350 /// @return Unique id for this item.
351 /// @throws OfflineQueueFullError if the queue is already at `maxDepth()`.
352 [[nodiscard]] uint64_t enqueue(std::string payload, std::string idempotencyKey) override {
353 std::scoped_lock const lock{_mtx};
354 if (_maxDepth && _items.size() >= *_maxDepth) {
355 ::morph::observe::detail::emitMetric(::morph::observe::Metric::queueOverflow,
356 static_cast<double>(_items.size()));
357 throw OfflineQueueFullError{*_maxDepth, _items.size()};
358 }
359 uint64_t const itemId = ++_nextId;
360 _items.push_back(
361 QueueItem{.id = itemId, .payload = std::move(payload), .idempotencyKey = std::move(idempotencyKey)});
362 return itemId;
363 }
364
365 /// @brief Returns a snapshot of all pending items. Thread-safe.
366 /// @return Copy of all items in insertion order.
367 [[nodiscard]] std::vector<QueueItem> drain() const override {
368 std::scoped_lock const lock{_mtx};
369 return std::vector<QueueItem>{_items.begin(), _items.end()};
370 }
371
372 /// @brief Returns the number of pending items. Thread-safe.
373 /// @return Current pending item count.
374 [[nodiscard]] std::size_t size() const override {
375 std::scoped_lock const lock{_mtx};
376 return _items.size();
377 }
378
379 /// @brief Returns the configured maximum depth, or `std::nullopt` if unbounded.
380 /// @return The capacity `enqueue()` enforces, or `std::nullopt` if none.
381 [[nodiscard]] std::optional<std::size_t> maxDepth() const override { return _maxDepth; }
382
383 /// @brief Removes the item with @p itemId from the queue. Thread-safe.
384 ///
385 /// No-op if @p itemId is not found.
386 /// @param itemId Id of the item to remove.
387 void markDone(uint64_t itemId) override {
388 std::scoped_lock const lock{_mtx};
389 auto iter = std::ranges::find_if(_items, [itemId](const QueueItem& item) { return item.id == itemId; });
390 if (iter != _items.end()) {
391 _items.erase(iter);
392 }
393 }
394
395 /// @brief Updates the persisted attempt count on the in-deque item with
396 /// @p itemId. No-op if @p itemId is not found. Thread-safe.
397 ///
398 /// `InMemoryOfflineQueue` does not itself survive a process restart, but
399 /// overriding this hook lets a *fresh* `SyncWorker` constructed over the
400 /// same instance observe a durable-style cumulative attempt count — used
401 /// to simulate cross-restart dead-lettering in tests.
402 /// @param itemId Id of the item to update.
403 /// @param attempts New attempt count to store.
404 void setAttempts(uint64_t itemId, Attempts attempts) override {
405 std::scoped_lock const lock{_mtx};
406 auto iter = std::ranges::find_if(_items, [itemId](const QueueItem& item) { return item.id == itemId; });
407 if (iter != _items.end()) {
408 iter->attempts = attempts.value();
409 }
410 }
411
412private:
413 mutable std::mutex _mtx;
414 std::deque<QueueItem> _items;
415 uint64_t _nextId{0};
416 std::optional<std::size_t> _maxDepth;
417};
