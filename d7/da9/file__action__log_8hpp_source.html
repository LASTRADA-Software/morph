24/// @brief Append-only, newline-delimited-JSON `IActionLog` backed by a local file.
25///
26/// Each entry is written as one `journal::toJson`-encoded line. `flush()` flushes
27/// the C stdio buffer and then issues a real fsync (POSIX `fsync` / Windows
28/// `_commit`), so a crash immediately after `flush()` returns cannot lose data —
29/// this is the "local file" sink from the original request, and the natural
30/// target for `SessionLog::checkpoint()` at a "Save" action.
31///
32/// @par Process-local `seq`
33/// Like `InMemoryActionLog`, `seq` is assigned fresh per process instance — it
34/// does not resume from the highest `seq` already on disk if the file is
35/// reopened. Entries remain correctly ordered on disk regardless (append-only),
36/// but `seq` alone is not a cross-restart unique key; use `entries()`' natural
37/// file order for that.
38///
39/// @par Thread safety
40/// All public methods are thread-safe (guarded by an internal mutex). Safe to
41/// use from multiple threads within one process; not safe for multiple
42/// processes to append to the same path concurrently.
43class FileActionLog : public IActionLog {
44public:
45 /// @brief Opens (creating if necessary) @p path for appending.
46 /// @param path File to append entries to.
47 /// @throws std::runtime_error if the file cannot be opened.
48 explicit FileActionLog(std::filesystem::path path) : _path{std::move(path)} {
49 _file = std::fopen(_path.string().c_str(), "a");
50 if (_file == nullptr) {
51 throw std::runtime_error("FileActionLog: failed to open " + _path.string());
52 }
53 }
54
55 /// @brief Closes the underlying file.
56 ///
57 /// `_file` is always non-null here: the constructor either finishes with a
58 /// valid handle or throws before completing (in which case this destructor
59 /// never runs), and copy/move are deleted, so there is no path that could
60 /// null it out afterwards.
61 // NOLINTNEXTLINE(cert-err33-c) — destructor context, can't propagate errors
62 ~FileActionLog() override { std::fclose(_file); }
63
64 FileActionLog(const FileActionLog&) = delete;
65 FileActionLog& operator=(const FileActionLog&) = delete;
66 FileActionLog(FileActionLog&&) = delete;
67 FileActionLog& operator=(FileActionLog&&) = delete;
68
69 /// @brief Appends @p entry as one JSON line. Buffered until `flush()`. Thread-safe.
70 /// @param entry Entry to append; `seq` is overwritten regardless of the input value.
71 void append(LogEntry entry) override {
72 std::scoped_lock const lock{_mtx};
73 entry.seq = ++_nextSeq;
74 auto line = toJson(entry);
75 line.push_back('\n');
76 // NOLINTNEXTLINE(cert-err33-c) — append-only; fwrite errors checked via subsequent fsync
77 std::fwrite(line.data(), 1, line.size(), _file);
78 }
79
80 /// @brief Flushes stdio's buffer, then fsyncs the file descriptor. Thread-safe.
81 void flush() override {
82 std::scoped_lock const lock{_mtx};
83 // NOLINTNEXTLINE(cert-err33-c) — errors logged by caller after flush
84 std::fflush(_file);
85#ifdef _WIN32
86 _commit(_fileno(_file));
87#else
88 ::fsync(fileno(_file));
89#endif
90 }
91
92 /// @brief Re-reads the file from disk and decodes every line. Thread-safe.
93 ///
94 /// Reads whatever is currently on disk, including anything written but not
95 /// yet `flush()`ed if the platform's stdio buffering has already handed it
96 /// to the OS — callers that need a guaranteed-durable view should `flush()`
97 /// first.
98 /// @param entityKey If non-empty, restricts the result to that entity's entries.
99 /// @return Matching entries, in on-disk (append) order.
100 [[nodiscard]] std::vector<LogEntry> entries(std::string_view entityKey = {}) const override {
101 std::scoped_lock const lock{_mtx};
102 std::ifstream in{_path};
103 std::vector<std::string> lines;
104 std::string line;
105 while (std::getline(in, line)) {
106 if (!line.empty()) {
107 lines.push_back(line);
108 }
109 }
110 std::vector<LogEntry> out;
111 for (std::size_t i = 0; i < lines.size(); ++i) {
112 LogEntry entry;
113 try {
114 entry = fromJson(lines[i]);
115 } catch (const std::exception& exc) {
116 // A crash between `append`'s `fwrite` and the next flush can leave
117 // a truncated final line. Tolerate exactly that — skip a malformed
118 // *trailing* line so the rest of the log stays readable — but a
119 // malformed line mid-file is genuine corruption and is re-thrown.
120 if (i + 1 == lines.size()) {
121 ::morph::log::logWarn("FileActionLog: skipping malformed trailing line in " +
122 _path.string() + ": " + std::string{exc.what()});
123 break;
124 }
125 throw;
126 }
127 if (entityKey.empty() || entry.entityKey == entityKey) {
128 out.push_back(std::move(entry));
129 }
130 }
131 return out;
132 }
133
134private:
135 std::filesystem::path _path;
136 std::FILE* _file = nullptr;
137 mutable std::mutex _mtx;
138 uint64_t _nextSeq{0};
139};
