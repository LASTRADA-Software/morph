23/// @brief Append-only, newline-delimited-JSON `IActionLog` backed by a local file.
24///
25/// Each entry is written as one `journal::toJson`-encoded line. `flush()` flushes
26/// the C stdio buffer and then issues a real fsync (POSIX `fsync` / Windows
27/// `_commit`), so a crash immediately after `flush()` returns cannot lose data —
28/// this is the "local file" sink from the original request, and the natural
29/// target for `SessionLog::checkpoint()` at a "Save" action.
30///
31/// @par Process-local `seq`
32/// Like `InMemoryActionLog`, `seq` is assigned fresh per process instance — it
33/// does not resume from the highest `seq` already on disk if the file is
34/// reopened. Entries remain correctly ordered on disk regardless (append-only),
35/// but `seq` alone is not a cross-restart unique key; use `entries()`' natural
36/// file order for that.
37///
38/// @par Thread safety
39/// All public methods are thread-safe (guarded by an internal mutex). Safe to
40/// use from multiple threads within one process; not safe for multiple
41/// processes to append to the same path concurrently.
42class FileActionLog : public IActionLog {
43public:
44 /// @brief Opens (creating if necessary) @p path for appending.
45 /// @param path File to append entries to.
46 /// @throws std::runtime_error if the file cannot be opened.
47 explicit FileActionLog(std::filesystem::path path) : _path{std::move(path)} {
48 _file = std::fopen(_path.string().c_str(), "a");
49 if (_file == nullptr) {
50 throw std::runtime_error("FileActionLog: failed to open " + _path.string());
51 }
52 }
53
54 /// @brief Closes the underlying file.
55 ///
56 /// `_file` is always non-null here: the constructor either finishes with a
57 /// valid handle or throws before completing (in which case this destructor
58 /// never runs), and copy/move are deleted, so there is no path that could
59 /// null it out afterwards.
60 ~FileActionLog() override { std::fclose(_file); }
61
62 FileActionLog(const FileActionLog&) = delete;
63 FileActionLog& operator=(const FileActionLog&) = delete;
64 FileActionLog(FileActionLog&&) = delete;
65 FileActionLog& operator=(FileActionLog&&) = delete;
66
67 /// @brief Appends @p entry as one JSON line. Buffered until `flush()`. Thread-safe.
68 /// @param entry Entry to append; `seq` is overwritten regardless of the input value.
69 void append(LogEntry entry) override {
70 std::scoped_lock lock{_mtx};
71 entry.seq = ++_nextSeq;
72 auto line = toJson(entry);
73 line.push_back('\n');
74 std::fwrite(line.data(), 1, line.size(), _file);
75 }
76
77 /// @brief Flushes stdio's buffer, then fsyncs the file descriptor. Thread-safe.
78 void flush() override {
79 std::scoped_lock lock{_mtx};
80 std::fflush(_file);
81#if defined(_WIN32)
82 _commit(_fileno(_file));
83#else
84 ::fsync(fileno(_file));
85#endif
86 }
87
88 /// @brief Re-reads the file from disk and decodes every line. Thread-safe.
89 ///
90 /// Reads whatever is currently on disk, including anything written but not
91 /// yet `flush()`ed if the platform's stdio buffering has already handed it
92 /// to the OS — callers that need a guaranteed-durable view should `flush()`
93 /// first.
94 /// @param entityKey If non-empty, restricts the result to that entity's entries.
95 /// @return Matching entries, in on-disk (append) order.
96 [[nodiscard]] std::vector<LogEntry> entries(std::string_view entityKey = {}) const override {
97 std::scoped_lock lock{_mtx};
98 std::ifstream in{_path};
99 std::vector<LogEntry> out;
100 std::string line;
101 while (std::getline(in, line)) {
102 if (line.empty()) {
103 continue;
104 }
105 auto entry = fromJson(line);
106 if (entityKey.empty() || entry.entityKey == entityKey) {
107 out.push_back(std::move(entry));
108 }
109 }
110 return out;
111 }
112
113private:
114 std::filesystem::path _path;
115 std::FILE* _file = nullptr;
116 mutable std::mutex _mtx;
117 uint64_t _nextSeq{0};
118};
