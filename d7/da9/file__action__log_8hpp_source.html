26/// @brief Append-only, newline-delimited-JSON `IActionLog` backed by a local file.
27///
28/// Each entry is written as one `journal::toJson`-encoded line. `flush()` flushes
29/// the C stdio buffer and then issues a real fsync (POSIX `fsync` / Windows
30/// `_commit`), so a crash immediately after `flush()` returns cannot lose data —
31/// this is the "local file" sink from the original request, and the natural
32/// target for `SessionLog::checkpoint()` at a "Save" action. Dedups `append()` on
33/// a non-empty `LogEntry::idempotencyKey`, rebuilding the seen-key set from disk
34/// at open time — see `IActionLog`'s class docs and the constructor's docs.
35///
36/// @par Process-local `seq`
37/// Like `InMemoryActionLog`, `seq` is assigned fresh per process instance — it
38/// does not resume from the highest `seq` already on disk if the file is
39/// reopened. Entries remain correctly ordered on disk regardless (append-only),
40/// but `seq` alone is not a cross-restart unique key; use `entries()`' natural
41/// file order for that.
42///
43/// @par Thread safety
44/// All public methods are thread-safe (guarded by an internal mutex). Safe to
45/// use from multiple threads within one process; not safe for multiple
46/// processes to append to the same path concurrently.
47///
48/// @par Rotation
49/// `rotate()` seals the active file (rename to a host-chosen path) and
50/// reopens a fresh, empty one at the original path — the seam a host uses to
51/// implement its own retention policy. See `rotate()`'s own docs.
52class FileActionLog : public IActionLog {
53public:
54 /// @brief Opens (creating if necessary) @p path for appending.
55 ///
56 /// Also rebuilds the `idempotencyKey` dedup set (see `append()`) from
57 /// whatever is already on disk at @p path — an O(n) scan of the existing
58 /// file's contents, paid once here, not on every `append()`.
59 /// @param path File to append entries to.
60 /// @throws std::runtime_error if the file cannot be opened.
61 /// @throws SerializationError if an existing file at @p path has a malformed
62 /// *interior* line (a malformed trailing line is tolerated — see
63 /// `entries()`).
64 explicit FileActionLog(std::filesystem::path path) : _path{std::move(path)} {
65 // Discard a torn trailing record before anything else touches the file.
66 // The file is opened "a", so the next append() would otherwise start
67 // writing at the exact byte the truncated JSON stopped at, with no
68 // separating newline: the two would merge into one line that swallows
69 // the new entry, and once a *further* append pushes that merged line out
70 // of trailing position, entries()' tolerance no longer applies and it
71 // throws -- from this very constructor, leaving the journal permanently
72 // unopenable. FileOfflineQueue heals the same damage in compact(); this
73 // is FileActionLog's equivalent.
74 repairTornTail();
75
76 // Rebuild the idempotencyKey dedup set from whatever is already durably on
77 // disk, so a re-relayed outbox row is recognised even after this process
78 // restarts (not just within one FileActionLog instance's lifetime). Reuses
79 // entries()'s existing torn-trailing-line tolerance; a malformed *interior*
80 // line still throws SerializationError here, same as calling entries()
81 // directly would (see entries()'s docs).
82 //
83 // Done before _file is opened: entries() reads through its own
84 // std::ifstream, independent of _file, so a throw here leaves no file
85 // handle open. Opening _file first and rebuilding the dedup set after
86 // would leak it on this path -- the constructor never completes, so
87 // the destructor never runs to close what fopen() already opened.
88 // entries() is deliberately the final overrider here: copy/move are
89 // deleted and nothing derives from this class, so there is no more-
90 // derived override for the call to bypass.
91 // NOLINTNEXTLINE(clang-analyzer-optin.cplusplus.VirtualCall)
92 for (const auto& existing : entries()) {
93 if (!existing.idempotencyKey.empty()) {
94 _seenIdempotencyKeys.insert(existing.idempotencyKey);
95 }
96 }
97 _file = std::fopen(_path.string().c_str(), "a");
98 if (_file == nullptr) {
99 throw std::runtime_error("FileActionLog: failed to open " + _path.string());
100 }
101 }
102
103 /// @brief Closes the underlying file.
104 ///
105 /// `_file` is normally non-null: the constructor either finishes with a
106 /// valid handle or throws before completing (in which case this destructor
107 /// never runs), and copy/move are deleted. The one exception is a `rotate()`
108 /// whose reopen failed after the rename succeeded — it throws with the
109 /// handle left null rather than dangling, so the null check here is
110 /// reachable and load-bearing, not defensive noise.
111 // NOLINTNEXTLINE(cert-err33-c) — destructor context, can't propagate errors
112 ~FileActionLog() override {
113 if (_file != nullptr) {
114 // Destructor context: there is nothing to propagate a failure to.
115 // NOLINTNEXTLINE(cert-err33-c, cppcoreguidelines-owning-memory)
116 std::fclose(_file);
117 }
118 }

125 /// @brief Appends @p entry as one JSON line. Buffered until `flush()`. Thread-safe.
126 ///
127 /// The entry's `idempotencyKey` is only remembered as *seen* once `flush()`
128 /// confirms it reached the disk (see `flush()`), so a write that fails is
129 /// still retryable. Recording the key up front — before the bytes were
130 /// durable — silently deduplicated the retry of a row that was never
131 /// written, turning a transient I/O error into permanent data loss.
132 ///
133 /// @param entry Entry to append; `seq` is overwritten regardless of the input value.
134 /// @throws std::runtime_error if the write fails, or if the log has no open
135 /// file (only reachable after a `rotate()` that failed to reopen).
136 void append(LogEntry entry) override {
137 std::scoped_lock const lock{_mtx};
138 requireOpen("append");
139 if (!entry.idempotencyKey.empty() && (_seenIdempotencyKeys.contains(entry.idempotencyKey) ||
140 _unflushedIdempotencyKeys.contains(entry.idempotencyKey))) {
141 return; // already recorded; a re-relayed duplicate is a safe no-op
142 }
143 entry.seq = ++_nextSeq;
144 auto line = toJson(entry);
145 line.push_back('\n');
146 if (std::fwrite(line.data(), 1, line.size(), _file) != line.size()) {
147 throw std::runtime_error("FileActionLog::append: short write to " + _path.string());
148 }
149 if (!entry.idempotencyKey.empty()) {
150 _unflushedIdempotencyKeys.insert(std::move(entry.idempotencyKey));
151 }
152 }
153
154 /// @brief Flushes stdio's buffer, then fsyncs the file descriptor. Thread-safe.
155 ///
156 /// Throws rather than returning a status because the interface is
157 /// `void`-returning and the failure is not one a caller may ignore:
158 /// `OutboxRelay::relay()` calls `markRelayed()` immediately after this, and
159 /// on a silent failure would record rows as relayed in the model's own store
160 /// while nothing reached the durable sink — the rows are then gone from the
161 /// outbox and absent from the log. Throwing aborts `relay()` before
162 /// `markRelayed`, leaving the batch to be retried.
163 ///
164 /// On failure the keys buffered since the last successful flush are
165 /// forgotten, so a retry writes them again instead of being deduplicated
166 /// away. A duplicated audit row is recoverable; a dropped one is not.
167 ///
168 /// @throws std::runtime_error if the buffer flush or the fsync fails, or if
169 /// the log has no open file.
170 void flush() override {
171 std::scoped_lock const lock{_mtx};
172 requireOpen("flush");
173 if (std::fflush(_file) != 0) {
174 _unflushedIdempotencyKeys.clear();
175 throw std::runtime_error("FileActionLog::flush: failed to flush " + _path.string());
176 }
177#ifdef _WIN32
178 int const syncResult = _commit(_fileno(_file));
179#else
180 int const syncResult = ::fsync(fileno(_file));
181#endif
182 if (syncResult != 0) {
183 _unflushedIdempotencyKeys.clear();
184 throw std::runtime_error("FileActionLog::flush: failed to fsync " + _path.string());
185 }
186 // Durable now: promote this window's keys so they survive as dedup state.
187 for (const auto& key : _unflushedIdempotencyKeys) {
188 _seenIdempotencyKeys.insert(key);
189 }
190 _unflushedIdempotencyKeys.clear();
191 }

193 /// @brief Re-reads the file from disk and decodes every line. Thread-safe.
194 ///
195 /// Reads whatever is currently on disk, including anything written but not
196 /// yet `flush()`ed if the platform's stdio buffering has already handed it
197 /// to the OS — callers that need a guaranteed-durable view should `flush()`
198 /// first.
199 /// @param entityKey If non-empty, restricts the result to that entity's entries.
200 /// @return Matching entries, in on-disk (append) order.
201 [[nodiscard]] std::vector<LogEntry> entries(std::string_view entityKey = {}) const override {
202 std::scoped_lock const lock{_mtx};
203 std::ifstream in{_path};
204 std::vector<std::string> lines;
205 std::string line;
206 while (std::getline(in, line)) {
207 if (!line.empty()) {
208 lines.push_back(line);
209 }
210 }
211 std::vector<LogEntry> out;
212 for (std::size_t i = 0; i < lines.size(); ++i) {
213 LogEntry entry;
214 try {
215 entry = fromJson(lines[i]);
216 } catch (const std::exception& exc) {
217 // A crash between `append`'s `fwrite` and the next flush can leave
218 // a truncated final line. Tolerate exactly that — skip a malformed
219 // *trailing* line so the rest of the log stays readable — but a
220 // malformed line mid-file is genuine corruption and is re-thrown.
221 if (i + 1 == lines.size()) {
222 ::morph::log::logWarn("FileActionLog: skipping malformed trailing line in " + _path.string() +
223 ": " + std::string{exc.what()});
224 break;
225 }
226 throw;
227 }
228 if (entityKey.empty() || entry.entityKey == entityKey) {
229 out.push_back(std::move(entry));
230 }
231 }
232 return out;
233 }

235 /// @brief Seals the active file and reopens a fresh, empty one at the same path.
236 ///
237 /// Flushes (`fflush` + `fsync`/`_commit`) and closes the current active
238 /// file, renames it to @p sealedPath, then reopens a fresh, empty active
239 /// file at the original path. Thread-safe — guarded by the same mutex as
240 /// `append()`/`flush()`/`entries()`, so no in-flight `append()` call is
241 /// ever split across the sealed and the new active file. `entries()`
242 /// keeps reading only the (post-rotation, empty) active file; composing
243 /// full history across a rotation is a host-side recipe (concatenate each
244 /// sealed segment's `entries()` oldest-to-newest, then the active file's),
245 /// not a new API.
246 ///
247 /// @par Crash safety
248 /// The rename is a single atomic filesystem operation. A crash before it
249 /// completes leaves the pre-rotation active file untouched, as if
250 /// `rotate()` was never called; a crash after leaves the sealed file plus
251 /// a freshly recreated, empty active file. Either way no line is ever torn
252 /// across the two files, so the existing torn-line rule keeps applying
253 /// independently per file.
254 ///
255 /// @param sealedPath Destination path for the sealed segment. Platform
256 /// `rename` semantics decide what happens if a filesystem entry
257 /// already exists there (POSIX silently replaces it) — pass a path
258 /// that does not collide with an existing segment.
259 /// @throws std::runtime_error if the pre-rotation flush/fsync fails (raised
260 /// before anything is closed or renamed, so no segment is ever
261 /// sealed around entries that never reached the disk); if renaming
262 /// to @p sealedPath fails — in that case the *original* active file
263 /// is reopened in place (still holding every entry recorded before
264 /// the call, so no data is lost; the rotation simply did not
265 /// happen) — or if reopening the active path afterward fails
266 /// outright (only possible if the rename itself succeeded and the
267 /// original path's directory then became unwritable). After that
268 /// last case the log has no open file: `append()`, `flush()`, and a
269 /// further `rotate()` all throw with that diagnosis rather than
270 /// dereferencing a null handle, and destruction is still safe.
271 void rotate(const std::filesystem::path& sealedPath) {
272 std::scoped_lock const lock{_mtx};
273 requireOpen("rotate");
274 // Same durability steps as flush()'s body, inlined here because
275 // flush() itself takes _mtx and this is already under it. A failure is
276 // raised before the file is closed and renamed, so a segment is never
277 // sealed around entries that never reached the disk.
278 if (std::fflush(_file) != 0) {
279 throw std::runtime_error("FileActionLog::rotate: failed to flush " + _path.string());
280 }
281#ifdef _WIN32
282 int const syncResult = _commit(_fileno(_file));
283#else
284 int const syncResult = ::fsync(fileno(_file));
285#endif
286 if (syncResult != 0) {
287 throw std::runtime_error("FileActionLog::rotate: failed to fsync " + _path.string());
288 }
289 // Everything buffered is now durable in the segment about to be sealed.
290 for (const auto& key : _unflushedIdempotencyKeys) {
291 _seenIdempotencyKeys.insert(key);
292 }
293 _unflushedIdempotencyKeys.clear();
294 // NOLINTNEXTLINE(cert-err33-c) — the data is already fsynced above
295 std::fclose(_file);
296 _file = nullptr;
297
298 std::error_code renameError;
299 std::filesystem::rename(_path, sealedPath, renameError);
300
301 // Reopen the active path regardless of the rename's outcome: on
302 // success this creates a fresh empty file; on failure it reopens the
303 // same pre-rotation file (still holding every prior entry), so a
304 // failed rotation never leaves the log unusable.
305 _file = std::fopen(_path.string().c_str(), "a");
306 if (_file == nullptr) {
307 throw std::runtime_error("FileActionLog::rotate: failed to reopen " + _path.string() + " after " +
308 (renameError ? "a failed" : "a successful") + " rename to " +
309 sealedPath.string());
310 }
311 if (renameError) {
312 throw std::runtime_error("FileActionLog::rotate: failed to rename " + _path.string() + " to " +
313 sealedPath.string() + ": " + renameError.message());
314 }
315 }

317private:
318 /// Throws if no file handle is open. Reachable only after a `rotate()` whose
319 /// reopen failed: that path deliberately leaves `_file` null rather than
320 /// dangling, so every entry point has to say so instead of dereferencing it.
321 void requireOpen(std::string_view what) const {
322 if (_file == nullptr) {
323 throw std::runtime_error("FileActionLog::" + std::string{what} + ": " + _path.string() +
324 " is not open (a previous rotate() failed to reopen it)");
325 }
326 }

328 /// Truncates any bytes following the last newline in the file.
329 ///
330 /// A crash between `append()`'s `fwrite` and the next `flush()` can leave a
331 /// partial record at the end. Because every complete record is written
332 /// newline-terminated in a single `fwrite`, whatever follows the final
333 /// newline is by construction an incomplete record and never a whole one —
334 /// which makes discarding it safe: it can only remove bytes that no reader
335 /// could ever have decoded. Complete records, including a malformed
336 /// *interior* line, are left exactly as they are; diagnosing those stays
337 /// `entries()`' job.
338 void repairTornTail() const {
339 std::error_code errorCode;
340 auto const size = std::filesystem::file_size(_path, errorCode);
341 if (errorCode || size == 0) {
342 return; // absent or empty: nothing to repair
343 }
344 std::ifstream input{_path, std::ios::binary};
345 if (!input) {
346 return;
347 }
348 std::uintmax_t intactEnd = 0;
349 std::uintmax_t offset = 0;
350 std::string line;
351 while (std::getline(input, line)) {
352 offset += line.size();
353 if (input.eof()) {
354 break; // no trailing newline: this line is the torn remainder
355 }
356 ++offset; // the '\n' getline consumed
357 intactEnd = offset;
358 }
359 if (intactEnd == size) {
360 return;
361 }
362 std::filesystem::resize_file(_path, intactEnd, errorCode);
363 if (errorCode) {
364 ::morph::log::logWarn("FileActionLog: could not truncate torn trailing record in " + _path.string() +
365 ": " + errorCode.message());
366 return;
367 }
368 ::morph::log::logWarn("FileActionLog: discarded " + std::to_string(size - intactEnd) +
369 " byte(s) of a torn trailing record in " + _path.string());
370 }
371
372 std::filesystem::path _path;
373 std::FILE* _file = nullptr;
374 mutable std::mutex _mtx;
375 uint64_t _nextSeq{0};
376 /// Keys confirmed durable by a successful `flush()`.
377 std::unordered_set<std::string> _seenIdempotencyKeys;
378 /// Keys written since the last successful `flush()`. Promoted into
379 /// `_seenIdempotencyKeys` on success, discarded on failure so the
380 /// corresponding rows stay retryable.
381 std::unordered_set<std::string> _unflushedIdempotencyKeys;
382};
