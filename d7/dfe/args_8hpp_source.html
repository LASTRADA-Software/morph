16/// @brief What the generic client reads from its command line.
17struct Args {
18 /// @brief `--server URL`: the server to fetch the screens from.

30 /// @brief `--dev`, or `MORPH_UI_DEV=1`: development mode, which reloads the screens when the server's documents
31 /// change (spec 5 §12). A client built without the development tools ignores it.
32 bool dev = false;
33};

35/// @brief Reads the client's options from @p args and removes them, leaving the rest for the frontend.
36///
37/// Each value option is `--name value` or `--name=value`, and the last one given wins. `args[0]` is the program
38/// name, and nothing after a `--` is read.
39/// @param args The command line; what was read is removed.
40/// @return The options, or why they are refused: a value option without its value, or not exactly one of
41/// `--server` and `--app`. @p args is unchanged on a refusal.
42[[nodiscard]] std::expected<Args, std::string> readArgs(std::vector<std::string>& args);
