36/// @brief Reads the document of a screen the bundle does not hold, such as a person's own custom screen.
37///
38/// Called on the turn's worker with the turn's session and the screen's id, and answers through the callback, from any
39/// thread, exactly once: the document and its base, or why there is none, which fails the turn. It is asked only for
40/// an id the bundle has no screen of, once per turn, and it must answer for the session's person alone. One that has
41/// not answered within `ChatLimits::toolCallTimeout` fails the turn, and a later answer is ignored.

43 std::function<void(std::expected<ChatScreenSource, std::string>)>)>;

53/// @brief What a `ChatService` is built with.
54struct MORPH_API ChatOptions {

58 /// offers no dictation. Needs a provider, since the text goes into the chat's input.
59 std::shared_ptr<ITranscriber> transcriber;
60 /// @brief The language a clip is transcribed in when its request names none; nothing leaves it to the transcriber.
61 std::optional<std::string> transcriptionLanguage;
62 /// @brief Where turns run. Borrowed: it outlives the service and every turn. Required with a provider.
63 exec::IExecutor* worker = nullptr;
64 /// @brief The bundle screens are looked up in. Required with a provider.
65 std::shared_ptr<ServedBundle const> bundle;
66 /// @brief The bounds.
67 ChatLimits limits;
68 /// @brief What the assistant is told about how to behave.
69 std::string systemPrompt = std::string{kDefaultSystemPrompt};
70 /// @brief The clock retention reads; empty for `steady_clock`.
71 std::function<std::chrono::steady_clock::time_point()> now;
72 /// @brief Tools offered beside the ones the open screen names. A turn offers each one whose `offer` accepts it,
73 /// and a call of one runs its handler in place of an action. Their names must be valid, and distinct from
74 /// each other; see `ChatTool`.
75 std::vector<ChatTool> tools;
76 /// @brief Where the document of a screen outside the bundle is read, so the chat can offer such a screen's
77 /// actions. Empty: a screen the bundle does not hold is not available to the chat.
78 ChatDocumentSource documentSource;
79 /// @brief The dispatcher the server's actions are registered on. Borrowed.
80 ::morph::model::detail::ActionDispatcher* dispatcher = &::morph::model::detail::defaultDispatcher();
81
82 /// @brief Takes the limits and the system prompt from a configuration. An empty prompt keeps the default.
83 /// @param config The configuration, however the application filled it.
84 void configure(ChatConfig const& config);
85};

87/// @brief Answers the chat kinds, and runs each turn's loop on the worker: read what the screen offers, ask the
88/// provider, run the tools it calls, and repeat until it has no more.
89///
90/// Each tool call is an execute carrying the client's session, so it passes every check the GUI's own calls pass. It
91/// runs unscoped and names the instance the client's context gave, so it reaches the screen's own instance and leaves
92/// the connection's principal baseline alone. A running action therefore cannot be interrupted: cancelling a turn
93/// stops it between actions.
94class MORPH_API ChatService final : public backend::IEnvelopeService {
95public:
96 /// @brief A service over @p options.
97 /// @param options What it is built with.
98 /// @throws std::invalid_argument with a provider but no worker or bundle, with a transcriber but no provider, a
99 /// dictation limit of 0 or a `maxDictationBytes` above `kMaxDictationBytes`, or with an extra tool whose
100 /// name is invalid or taken by another, or which has no handler.
101 explicit ChatService(ChatOptions options);
102
103 /// @brief Stops every running turn and cancels every running transcription, announcing none. A provider or
104 /// transcriber answering later reaches nothing.
105 ~ChatService() override;
106
107 ChatService(ChatService const&) = delete;
108 ChatService& operator=(ChatService const&) = delete;
109 ChatService(ChatService&&) = delete;
110 ChatService& operator=(ChatService&&) = delete;
111
112 /// @brief Whether this service answers @p kind.
113 /// @param kind The envelope's kind.
114 /// @return `true` for the chat kinds, when there is a provider.
115 [[nodiscard]] bool serves(std::string_view kind) const override;
116
117 /// @brief Answers one envelope with no connection behind it.
118 /// @param request The envelope, its principal verified.
119 /// @param authorizer The server's authoriser.
120 /// @return The encoded reply.
121 [[nodiscard]] std::string handle(wire::Envelope const& request, session::IAuthorizer const& authorizer) override;
122
123 /// @brief Answers one envelope.
124 /// @param request The envelope, its principal verified.
125 /// @param authorizer The server's authoriser.
126 /// @param facts The facts of the connection it arrived on.
127 /// @return The encoded reply.
128 [[nodiscard]] std::string handle(wire::Envelope const& request, session::IAuthorizer const& authorizer,
129 backend::ConnectionFacts& facts) override;
130
131 /// @brief Keeps a handle to the server, and warns about actions that would go unrecorded.
132 /// @param server The server.
133 void bound(std::weak_ptr<backend::RemoteServer> const& server) override;
134
135 /// @brief Stops the connection's running turns, and cancels its running transcriptions and forgets them, since no
136 /// one can read them.
137 /// @param cid The connection.
138 void connectionClosed(backend::ConnectionId cid) override;
139
140private:
141 struct Turn;
142 struct Shared;
143 struct Transcription;
144 using TurnPtr = std::shared_ptr<Turn>;
145 using EnvelopeReply = std::function<void(wire::Envelope const&)>;
146
147 [[nodiscard]] static std::string err(wire::Envelope const& request, std::string message);
148
149 std::shared_ptr<Shared> _shared;
150};
