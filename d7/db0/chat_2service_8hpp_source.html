61/// @brief Answers the chat kinds, and runs each turn's loop on the worker: read what the screen offers, ask the
62/// provider, run the tools it calls, and repeat until it has no more.
63///
64/// Each tool call is an execute carrying the client's session, so it passes every check the GUI's own calls pass. It
65/// runs unscoped and names the instance the client's context gave, so it reaches the screen's own instance and leaves
66/// the connection's principal baseline alone. A running action therefore cannot be interrupted: cancelling a turn
67/// stops it between actions.
68class MORPH_API ChatService final : public backend::IEnvelopeService {
69public:
70 /// @brief A service over @p options.
71 /// @param options What it is built with.
72 /// @throws std::invalid_argument with a provider but no worker or bundle.
73 explicit ChatService(ChatOptions options);
74
75 /// @brief Stops every running turn. A provider call answering later reaches nothing.
76 ~ChatService() override;
77
78 ChatService(ChatService const&) = delete;
79 ChatService& operator=(ChatService const&) = delete;
80 ChatService(ChatService&&) = delete;
81 ChatService& operator=(ChatService&&) = delete;
82
83 /// @brief Whether this service answers @p kind.
84 /// @param kind The envelope's kind.
85 /// @return `true` for the chat kinds, when there is a provider.
86 [[nodiscard]] bool serves(std::string_view kind) const override;
87
88 /// @brief Answers one envelope with no connection behind it.
89 /// @param request The envelope, its principal verified.
90 /// @param authorizer The server's authoriser.
91 /// @return The encoded reply.
92 [[nodiscard]] std::string handle(wire::Envelope const& request, session::IAuthorizer const& authorizer) override;
93
94 /// @brief Answers one envelope.
95 /// @param request The envelope, its principal verified.
96 /// @param authorizer The server's authoriser.

98 /// @return The encoded reply.
99 [[nodiscard]] std::string handle(wire::Envelope const& request, session::IAuthorizer const& authorizer,
100 backend::ConnectionFacts& facts) override;
101
102 /// @brief Keeps a handle to the server, and warns about actions that would go unrecorded.
103 /// @param server The server.
104 void bound(std::weak_ptr<backend::RemoteServer> const& server) override;
105
106 /// @brief Stops the connection's running turns.
107 /// @param cid The connection.
108 void connectionClosed(backend::ConnectionId cid) override;
109
110private:

117
118 std::shared_ptr<Shared> _shared;
119};
