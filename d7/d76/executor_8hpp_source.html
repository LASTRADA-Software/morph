18/// @brief Abstract executor interface.
19///
20/// Concrete implementations decide *where* and *when* posted tasks run
21/// (thread pool, main thread, Qt event loop, …).
22// NOLINTBEGIN(cppcoreguidelines-special-member-functions)
23struct IExecutor {
24 virtual ~IExecutor() = default;
25
26 /// @brief Schedules @p task for asynchronous execution.
27 ///
28 /// Thread-safe. The task is invoked at some point after this call returns.
29 /// Any exception thrown by the task is silently swallowed unless the
30 /// implementation documents otherwise.
31 /// @param task Callable to execute.
32 virtual void post(std::function<void()> task) = 0;
33};
34// NOLINTEND(cppcoreguidelines-special-member-functions)
35
36/// @brief Multi-threaded executor backed by a fixed-size thread pool.
37///
38/// Tasks are placed in a FIFO queue and consumed by worker threads.
39/// Exceptions thrown by tasks are caught and logged via `morph::log` (they do
40/// not propagate out of the worker or abort sibling tasks).
41/// The destructor blocks until all worker threads have exited.
42// NOLINTNEXTLINE(cppcoreguidelines-special-member-functions)
43class ThreadPoolExecutor : public IExecutor {
44public:
45 /// @brief Constructs the pool with @p n worker threads.
46 ///
47 /// @p n is clamped to a minimum of 1: a pool with zero workers would accept
48 /// posted tasks that could never run (every `post()` would hang forever), so
49 /// `ThreadPoolExecutor(0)` yields a usable single-worker pool rather than a
50 /// silently dead one.
51 /// @param n Number of threads to spawn. Values below 1 are clamped to 1.
52 explicit ThreadPoolExecutor(std::size_t n) {
53 std::size_t const workers = n == 0 ? 1 : n;
54 for (std::size_t i = 0; i < workers; ++i) {
55 _workers.emplace_back([this] { loop(); });
56 }
57 }
58
59 /// @brief Signals all workers to stop and joins them.
60 ///
61 /// Workers drain the queue before exiting: after `_stop` is set they keep
62 /// popping and running already-queued tasks and only return once the queue is
63 /// empty, so the join blocks until every task queued before destruction has
64 /// run. Tasks `post()`ed concurrently with or after destruction race this and
65 /// may be silently lost.
66 ~ThreadPoolExecutor() override {
67 {
68 std::scoped_lock const lock{_m};
69 _stop = true;
70 }
71 _cv.notify_all();
72 for (auto& worker : _workers) {
73 worker.join();
74 }
75 }
76
77 /// @brief Enqueues @p task for execution on one of the pool threads.
78 /// @param task Callable to execute. Thread-safe.
79 void post(std::function<void()> task) override {
80 {
81 std::scoped_lock const lock{_m};
82 _q.push(std::move(task));
83 }
84 _cv.notify_one();
85 }
86
87private:
88 void loop() {
89 for (;;) {
90 std::function<void()> task;
91 {
92 std::unique_lock lock{_m};
93 _cv.wait(lock, [this] { return _stop || !_q.empty(); });
94 if (_stop && _q.empty()) {
95 return;
96 }
97 task = std::move(_q.front());
98 _q.pop();
99 }
100 try {
101 task();
102 } catch (const std::exception& exc) {
103 // A task failure must not kill the worker or unrelated tasks, but
104 // it must not vanish either — log and carry on.
105 ::morph::log::logError("[thread-pool] task threw: " + std::string{exc.what()});
106 } catch (...) {
107 ::morph::log::logError("[thread-pool] task threw unknown exception");
108 }
109 }
110 }
111 std::mutex _m;
112 std::condition_variable _cv;
113 std::queue<std::function<void()>> _q;
114 std::vector<std::thread> _workers;
115 bool _stop = false;
116};
117
118/// @brief Single-thread executor intended for use on the calling (main) thread.
119///
120/// Tasks posted from other threads are collected and dispatched only when
121/// the caller invokes `runFor()`. Useful in tests or event loops that have no
122/// native dispatcher.
123class MainThreadExecutor : public IExecutor {
124public:
125 /// @brief Enqueues @p task to be run on the next `runFor()` call.
126 ///
127 /// Thread-safe. The task is *not* executed immediately.
128 /// @param task Callable to execute.
129 void post(std::function<void()> task) override {
130 {
131 std::scoped_lock const lock{_m};
132 _q.push(std::move(task));
133 }
134 _cv.notify_all();
135 }
136
137 /// @brief Pumps the task queue for up to @p timeout.
138 ///
139 /// Runs queued tasks one by one. It does **not** return early when the queue
140 /// drains: while time remains before the deadline it blocks waiting for
141 /// newly posted tasks, and only returns once the deadline is reached (with
142 /// any still-queued tasks left for a subsequent call — this is a pump, not a
143 /// flush). Exceptions thrown by tasks are logged and execution continues with
144 /// the next task.
145 ///
146 /// Must be called from the thread that "owns" this executor.
147 /// @param timeout Maximum wall-clock time to spend draining.
148 void runFor(std::chrono::milliseconds timeout) {
149 auto deadline = std::chrono::steady_clock::now() + timeout;
150 while (std::chrono::steady_clock::now() < deadline) {
151 std::function<void()> task;
152 {
153 std::unique_lock lock{_m};
154 if (!_cv.wait_until(lock, deadline, [this] { return !_q.empty(); })) {
155 return;
156 }
157 task = std::move(_q.front());
158 _q.pop();
159 }
160 try {
161 task();
162 } catch (const std::exception& exc) {
163 ::morph::log::logError("[main-thread] callback threw: " + std::string{exc.what()});
164 }
165 }
166 }
167
168private:
169 std::mutex _m;
170 std::condition_variable _cv;
171 std::queue<std::function<void()>> _q;
172};
