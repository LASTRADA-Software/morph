18/// @brief Abstract executor interface.
19///
20/// Concrete implementations decide *where* and *when* posted tasks run
21/// (thread pool, main thread, Qt event loop, …).
22struct IExecutor {
23 virtual ~IExecutor() = default;
24
25 /// @brief Schedules @p task for asynchronous execution.
26 ///
27 /// Thread-safe. The task is invoked at some point after this call returns.
28 /// Any exception thrown by the task is silently swallowed unless the
29 /// implementation documents otherwise.
30 /// @param task Callable to execute.
31 virtual void post(std::function<void()> task) = 0;
32};
33
34/// @brief Multi-threaded executor backed by a fixed-size thread pool.
35///
36/// Tasks are placed in a FIFO queue and consumed by worker threads.
37/// Exceptions thrown by tasks are silently caught and discarded.
38/// The destructor blocks until all worker threads have exited.
39class ThreadPoolExecutor : public IExecutor {
40public:
41 /// @brief Constructs the pool with @p n worker threads.
42 /// @param n Number of threads to spawn. Must be > 0.
43 explicit ThreadPoolExecutor(std::size_t n) {
44 for (std::size_t i = 0; i < n; ++i) {
45 _workers.emplace_back([this] { loop(); });
46 }
47 }
48
49 /// @brief Signals all workers to stop and joins them.
50 ///
51 /// Remaining queued tasks that have not started are dropped.
52 ~ThreadPoolExecutor() override {
53 {
54 std::scoped_lock lock{_m};
55 _stop = true;
56 }
57 _cv.notify_all();
58 for (auto& worker : _workers) {
59 worker.join();
60 }
61 }
62
63 /// @brief Enqueues @p task for execution on one of the pool threads.
64 /// @param task Callable to execute. Thread-safe.
65 void post(std::function<void()> task) override {
66 {
67 std::scoped_lock lock{_m};
68 _q.push(std::move(task));
69 }
70 _cv.notify_one();
71 }
72
73private:
74 void loop() {
75 for (;;) {
76 std::function<void()> task;
77 {
78 std::unique_lock lock{_m};
79 _cv.wait(lock, [this] { return _stop || !_q.empty(); });
80 if (_stop && _q.empty()) {
81 return;
82 }
83 task = std::move(_q.front());
84 _q.pop();
85 }
86 try {
87 task();
88 } catch (...) {
89 }
90 }
91 }
92 std::mutex _m;
93 std::condition_variable _cv;
94 std::queue<std::function<void()>> _q;
95 std::vector<std::thread> _workers;
96 bool _stop = false;
97};
98
99/// @brief Single-thread executor intended for use on the calling (main) thread.
100///
101/// Tasks posted from other threads are collected and dispatched only when
102/// the caller invokes `runFor()`. Useful in tests or event loops that have no
103/// native dispatcher.
104class MainThreadExecutor : public IExecutor {
105public:
106 /// @brief Enqueues @p task to be run on the next `runFor()` call.
107 ///
108 /// Thread-safe. The task is *not* executed immediately.
109 /// @param task Callable to execute.
110 void post(std::function<void()> task) override {
111 {
112 std::scoped_lock lock{_m};
113 _q.push(std::move(task));
114 }
115 _cv.notify_all();
116 }
117
118 /// @brief Drains the task queue for up to @p timeout.
119 ///
120 /// Runs queued tasks one by one until the deadline is reached or the queue
121 /// is empty. Exceptions thrown by tasks are logged and execution continues
122 /// with the next task.
123 ///
124 /// Must be called from the thread that "owns" this executor.
125 /// @param timeout Maximum wall-clock time to spend draining.
126 void runFor(std::chrono::milliseconds timeout) {
127 auto deadline = std::chrono::steady_clock::now() + timeout;
128 while (std::chrono::steady_clock::now() < deadline) {
129 std::function<void()> task;
130 {
131 std::unique_lock lock{_m};
132 if (!_cv.wait_until(lock, deadline, [this] { return !_q.empty(); })) {
133 return;
134 }
135 task = std::move(_q.front());
136 _q.pop();
137 }
138 try {
139 task();
140 } catch (const std::exception& exc) {
141 ::morph::log::logError("[main-thread] callback threw: " + std::string{exc.what()});
142 }
143 }
144 }
145
146private:
147 std::mutex _m;
148 std::condition_variable _cv;
149 std::queue<std::function<void()>> _q;
150};
