27/// @brief A `core::async::IExecutor` over a morph executor: how a core-cpp
28/// strand's pump, and an `ExecutorScope`, reach a thread pool, a
29/// main-thread pump or Qt.
30///
31/// Every `morph::exec::IExecutor` owns exactly one (`IExecutor::coreExecutor`),
32/// so an executor has one core-cpp identity for its whole life.
33///
34/// A strand hands its base one bare coroutine handle per turn, which this
35/// posts as a lambda holding nothing else: trivially copyable, so it fits a
36/// `std::function`'s small buffer and a turn costs no allocation. The lambda
37/// does not refer to this object, so it may be destroyed while a pump it
38/// posted is still queued: the pump finds its strand closed and ends.
39class CoreExecutorOver final : public ::core::async::IExecutor {
40public:
41 /// @param executor Where every resumption is posted. Borrowed: it must
42 /// outlive every strand over this object and run what it queued.
43 explicit CoreExecutorOver(::morph::exec::IExecutor& executor MORPH_LIFETIMEBOUND) : _executor{&executor} {}
44
45 using ::core::async::IExecutor::submit;
46
47 /// @brief Posts @p handle's resumption.
48 /// @param handle The coroutine to resume; borrowed.
49 void submit(std::coroutine_handle<> handle) override;
50
51 /// @brief Posts @p work's resumption. An executor that drops the task
52 /// unrun releases its claim, which frees a chain nobody owns.
53 /// @param work The coroutine to resume, and its claim.
54 void submit(::core::async::ParkedWork work) override;
55
56private:
57 ::morph::exec::IExecutor* _executor;
58};

62/// @brief Abstract executor interface.
63///
64/// Concrete implementations decide *where* and *when* posted tasks run
65/// (thread pool, main thread, Qt event loop, …).
66///
67/// An executor is an identity, so it is neither copyable nor movable: it owns
68/// the core-cpp adapter (`coreExecutor()`) that holds a pointer back to it.
69struct IExecutor {
70 IExecutor() = default;
71 IExecutor(const IExecutor&) = delete;
72 IExecutor& operator=(const IExecutor&) = delete;
73 IExecutor(IExecutor&&) = delete;
74 IExecutor& operator=(IExecutor&&) = delete;
75 virtual ~IExecutor() = default;
76
77 /// @brief Schedules @p task for asynchronous execution.
78 ///
79 /// Thread-safe. The task is invoked at some point after this call returns.
80 /// Any exception thrown by the task is silently swallowed unless the
81 /// implementation documents otherwise.
82 /// @param task Callable to execute.
83 virtual void post(std::function<void()> task) = 0;
84
85 /// @brief Whether this executor runs one task at a time, each after the
86 /// tasks posted before it: a strand, a GUI loop, a pumped
87 /// `MainThreadExecutor`.
88 ///
89 /// What an owner must be: a component whose state is touched only in its
90 /// owner's tasks needs those tasks never to overlap. A pool runs several
91 /// at once, and the inline executor runs each wherever it is posted from.
92 /// @return `true` unless an override says otherwise.
93 [[nodiscard]] virtual bool isSerial() const noexcept { return true; }
94
95 /// @brief This executor as a core-cpp executor: the identity an
96 /// `core::async::ExecutorScope` names.
97 /// @return The adapter that posts every resumption through `post()`. The
98 /// same object for the executor's whole life.
99 [[nodiscard]] ::core::async::IExecutor& coreExecutor() noexcept { return _core; }
100
101private:
102 // Stores only a pointer to this object, so constructing it from `*this`
103 // before this object is complete is sound.
104 detail::CoreExecutorOver _core{*this};
105};

118/// @brief Whether the calling thread is inside a task of @p executor.
119///
120/// True inside a task that @p executor's own worker is running, and inside
121/// anything that task calls synchronously, including a task posted to
122/// `detail::inlineExecutor()`, which states no scope of its own. Every scope in
123/// force is consulted, not only the innermost: a strand's batch runs inside the
124/// scope of the pool worker that drives it, and a task on that strand is still
125/// a task of the pool.
126///
127/// Cheap enough for a debug-build assertion: a walk over the calling thread's
128/// scope chain, which is a few entries deep.
129/// @param executor The executor asked about.
130/// @return True while a task of @p executor is running on the calling thread.
131[[nodiscard]] inline bool runningOn(IExecutor& executor) noexcept {
132 auto const* const identity = &executor.coreExecutor();
133 return ::core::async::ExecutorScope::anyInForce(
134 [identity](::core::async::ExecutorScope const& scope) { return &scope.executor() == identity; });
135}

137/// @brief Whether the calling thread is inside a task of the core-cpp
138/// executor @p executor: a `core::net::EventLoop`'s turn, a core-cpp
139/// strand's batch, or anything else that states an `ExecutorScope`.
140///
141/// The same walk as the overload above, for an owner that is a core-cpp
142/// executor rather than a morph one — `IoLoop::loop()` is the one morph hands
143/// out.
144/// @param executor The executor asked about.
145/// @return True while a task of @p executor is running on the calling thread.
146[[nodiscard]] inline bool runningOn(::core::async::IExecutor const& executor) noexcept {
147 return ::core::async::ExecutorScope::anyInForce(
148 [&executor](::core::async::ExecutorScope const& scope) { return &scope.executor() == &executor; });
149}

151/// @brief Multi-threaded executor backed by a fixed-size thread pool.
152///
153/// Tasks are placed in a FIFO queue and consumed by worker threads.
154/// Exceptions thrown by tasks are caught and logged via `morph::log` (they do
155/// not propagate out of the worker or abort sibling tasks).
156/// The destructor blocks until all worker threads have exited.
157// NOLINTNEXTLINE(cppcoreguidelines-special-member-functions)
158class ThreadPoolExecutor : public IExecutor {
159public:
160 /// @brief Constructs the pool with @p n worker threads.
161 ///
162 /// @p n is clamped to a minimum of 1: a pool with zero workers would accept
163 /// posted tasks that could never run (every `post()` would hang forever), so
164 /// `ThreadPoolExecutor(0)` yields a usable single-worker pool rather than a
165 /// silently dead one.
166 /// @param n Number of threads to spawn. Values below 1 are clamped to 1.
167 explicit ThreadPoolExecutor(std::size_t n) {
168 std::size_t const workers = n == 0 ? 1 : n;
169 for (std::size_t i = 0; i < workers; ++i) {
170 _workers.emplace_back([this] { loop(); });
171 }
172 }
173
174 /// @brief Signals all workers to stop and joins them.
175 ///
176 /// Workers drain the queue before exiting: after `_stop` is set they keep
177 /// popping and running already-queued tasks and only return once the queue is
178 /// empty, so the join blocks until every task queued before destruction has
179 /// run. Tasks `post()`ed concurrently with or after destruction race this and
180 /// may be silently lost.
181 ~ThreadPoolExecutor() override {
182 {
183 // notify_all() while still holding _m, not after releasing it: a
184 // waiter woken by a notify issued after unlocking could reacquire
185 // the lock, see _stop, return from wait(), and let this object be
186 // destroyed while this thread is still physically inside the
187 // notify call -- a real data race on the condition variable's
188 // internal state (this pattern is what post()'s own notify_one(),
189 // below, must avoid too, for the same reason).
190 std::scoped_lock const lock{_m};
191 _stop = true;
192 _cv.notify_all();
193 }
194 for (auto& worker : _workers) {
195 worker.join();
196 }
197 }
198
199 /// @brief Enqueues @p task for execution on one of the pool threads.
200 /// @param task Callable to execute. Thread-safe.
201 void post(std::function<void()> task) override {
202 std::scoped_lock const lock{_m};
203 _q.push(std::move(task));
204 _cv.notify_one();
205 }
206
207 /// @brief Not serial: the workers run tasks at once.
208 /// @return `false`.
209 [[nodiscard]] bool isSerial() const noexcept override { return false; }
210
211private:
212 void loop() {
213 for (;;) {
214 std::function<void()> task;
215 {
216 std::unique_lock lock{_m};
217 _cv.wait(lock, [this] { return _stop || !_q.empty(); });
218 if (_stop && _q.empty()) {
219 return;
220 }
221 task = std::move(_q.front());
222 _q.pop();
223 }
224 try {
225 // Stated per task, not once per worker: the scope ends with the
226 // task, on a throw too, so a thread never carries this pool's
227 // identity into anything it runs afterwards.
228 ::core::async::ExecutorScope const scope{coreExecutor()};
229 task();
230 } catch (const std::exception& exc) {
231 // A task failure must not kill the worker or unrelated tasks, but
232 // it must not vanish either — log and carry on.
233 ::morph::log::logError("[thread-pool] task threw: " + std::string{exc.what()});
234 } catch (...) {
235 ::morph::log::logError("[thread-pool] task threw unknown exception");
236 }
237 }
238 }
239 std::mutex _m;
240 std::condition_variable _cv;
241 std::queue<std::function<void()>> _q;
242 std::vector<std::thread> _workers;
243 bool _stop = false;
244};

246/// @brief Single-thread executor intended for use on the calling (main) thread.
247///
248/// Tasks posted from other threads are collected and dispatched only when
249/// the caller invokes `runFor()`. Useful in tests or event loops that have no
250/// native dispatcher.
251class MainThreadExecutor : public IExecutor {
252public:
253 /// @brief Enqueues @p task to be run on the next `runFor()` call.
254 ///
255 /// Thread-safe. The task is *not* executed immediately.
256 /// @param task Callable to execute.
257 void post(std::function<void()> task) override {
258 std::scoped_lock const lock{_m};
259 _q.push(std::move(task));
260 // Notified with the lock held: a consumer that polls (`runOnce()`) can
261 // take the task the moment the lock is released and destroy this
262 // executor, and a notify after that would touch a destroyed condition
263 // variable.
264 _cv.notify_all();
265 }
266
267 /// @brief Pumps the task queue for up to @p timeout.
268 ///
269 /// Runs queued tasks one by one. It does **not** return early when the queue
270 /// drains: while time remains before the deadline it blocks waiting for
271 /// newly posted tasks, and only returns once the deadline is reached (with
272 /// any still-queued tasks left for a subsequent call — this is a pump, not a
273 /// flush). Exceptions thrown by tasks are logged and execution continues with
274 /// the next task.
275 ///
276 /// Must be called from the thread that "owns" this executor.
277 /// @param timeout Maximum wall-clock time to spend draining.
278 void runFor(std::chrono::milliseconds timeout) {
279 auto deadline = std::chrono::steady_clock::now() + timeout;
280 while (std::chrono::steady_clock::now() < deadline) {
281 std::function<void()> task;
282 {
283 std::unique_lock lock{_m};
284 if (!_cv.wait_until(lock, deadline, [this] { return !_q.empty(); })) {
285 return;
286 }
287 task = std::move(_q.front());
288 _q.pop();
289 }
290 runTask(std::move(task));
291 }
292 }
293
294 /// @brief Dequeues and runs at most one pending task, without blocking.
295 ///
296 /// Never waits for a task to appear: if the queue is empty, returns `false`
297 /// immediately. Otherwise pops exactly one task, runs it, and returns `true`
298 /// whether or not that task threw. Unlike `runFor()`, there is no wall-clock
299 /// wait — this is the single-step primitive for callers (e.g. tests or
300 /// external event loops) that want deterministic, one-task-at-a-time control
301 /// instead of a timed pump.
302 ///
303 /// Must be called from the thread that "owns" this executor.
304 /// @return `true` if a task was dequeued and run, `false` if the queue was empty.
305 bool runOnce() {
306 std::function<void()> task;
307 {
308 std::scoped_lock const lock{_m};
309 if (_q.empty()) {
310 return false;
311 }
312 task = std::move(_q.front());
313 _q.pop();
314 }
315 runTask(std::move(task));
316 return true;
317 }
318
319 /// @brief Runs tasks until the queue is empty, with no wall-clock timeout.
320 ///
321 /// Repeatedly calls the same dequeue-and-invoke step as `runOnce()` until the
322 /// queue reports empty, then returns. It does **not** wait for tasks posted
323 /// after draining begins: a task that posts new work while running extends
324 /// the drain (the new task is still queued and gets consumed), but `drain()`
325 /// never blocks waiting for an *externally* posted task the way `runFor()`
326 /// does — it simply stops once the queue is observed empty.
327 ///
328 /// Must be called from the thread that "owns" this executor.
329 void drain() {
330 while (runOnce()) {
331 // Keep consuming until the queue reports empty.
332 }
333 }
334
335private:
336 void runTask(std::function<void()> task) {
337 try {
338 // The pumping thread is the caller's own, so the scope must end
339 // with the task: the caller runs other code on this thread between
340 // pumps.
341 ::core::async::ExecutorScope const scope{coreExecutor()};
342 task();
343 } catch (const std::exception& exc) {
344 ::morph::log::logError("[main-thread] callback threw: " + std::string{exc.what()});
345 } catch (...) {
346 // Mirrors ThreadPoolExecutor::loop's own catch-all. Without it a
347 // non-`std::exception` throw unwinds out of runFor()/runOnce()/
348 // drain(), each of which documents that a throwing task is logged
349 // and the pump continues -- runOnce() promises to return `true`
350 // "whether or not that task threw", and would not return at all.
351 ::morph::log::logError("[main-thread] callback threw unknown exception");
352 }
353 }
354
355 std::mutex _m;
356 std::condition_variable _cv;
357 std::queue<std::function<void()>> _q;
358};

362/// @brief Executor that runs each posted task on the posting thread, at once.
363///
364/// The base of an `OwnerStrand` that should run each task on the thread that
365/// posts it (a test's owner with no pool). It is not serial
366/// (`isSerial()` is `false`), so it cannot be a `Completion`'s owner: a
367/// completion delivered "wherever the producer settled" has no one place its
368/// callbacks are attached and run.
369///
370/// @warning Not a general-purpose executor. Posting to it re-enters the caller,
371/// so a handler that takes a lock the posting frame already holds
372/// self-deadlocks. Application code that wants "run it now" should
373/// call the function instead of posting it.
374///
375/// States no `ExecutorScope`: the task runs on the caller's thread, so
376/// whatever scope is already in force is the right answer to "where am I
377/// running". A task posted here from inside a pool task is still on the pool
378/// (`runningOn(pool)` stays true); one posted from a thread inside no
379/// executor's task is on none.
380// NOLINTNEXTLINE(cppcoreguidelines-special-member-functions)
381class InlineExecutor : public IExecutor {
382public:
383 /// @brief Runs @p task immediately, on the calling thread.
384 ///
385 /// An empty @p task is ignored rather than invoked, since `std::function`'s
386 /// empty state is a legal value to move around and calling it is undefined.
387 /// @param task Callable to execute.
388 void post(std::function<void()> task) override {
389 if (task) {
390 task();
391 }
392 }
393
394 /// @brief Not serial: a task runs on whichever thread posts it, beside
395 /// any other running elsewhere.
396 /// @return `false`.
397 [[nodiscard]] bool isSerial() const noexcept override { return false; }
398};
399
400/// @brief The process-wide `InlineExecutor`.
401///
402/// A function-local static, so it is constructed on first use and outlives
403/// everything built over it.
404/// @return Reference to the shared inline executor.
405inline IExecutor& inlineExecutor() {
406 static InlineExecutor executor;
407 return executor;
408}
