18/// @brief Abstract executor interface.
19///
20/// Concrete implementations decide *where* and *when* posted tasks run
21/// (thread pool, main thread, Qt event loop, …).
22// NOLINTBEGIN(cppcoreguidelines-special-member-functions)
23struct IExecutor {
24 virtual ~IExecutor() = default;
25
26 /// @brief Schedules @p task for asynchronous execution.
27 ///
28 /// Thread-safe. The task is invoked at some point after this call returns.
29 /// Any exception thrown by the task is silently swallowed unless the
30 /// implementation documents otherwise.
31 /// @param task Callable to execute.
32 virtual void post(std::function<void()> task) = 0;
33};
34// NOLINTEND(cppcoreguidelines-special-member-functions)
35
36/// @brief Multi-threaded executor backed by a fixed-size thread pool.
37///
38/// Tasks are placed in a FIFO queue and consumed by worker threads.
39/// Exceptions thrown by tasks are caught and logged via `morph::log` (they do
40/// not propagate out of the worker or abort sibling tasks).
41/// The destructor blocks until all worker threads have exited.
42// NOLINTNEXTLINE(cppcoreguidelines-special-member-functions)
43class ThreadPoolExecutor : public IExecutor {
44public:
45 /// @brief Constructs the pool with @p n worker threads.
46 ///
47 /// @p n is clamped to a minimum of 1: a pool with zero workers would accept
48 /// posted tasks that could never run (every `post()` would hang forever), so
49 /// `ThreadPoolExecutor(0)` yields a usable single-worker pool rather than a
50 /// silently dead one.
51 /// @param n Number of threads to spawn. Values below 1 are clamped to 1.
52 explicit ThreadPoolExecutor(std::size_t n) {
53 std::size_t const workers = n == 0 ? 1 : n;
54 for (std::size_t i = 0; i < workers; ++i) {
55 _workers.emplace_back([this] { loop(); });
56 }
57 }
58
59 /// @brief Signals all workers to stop and joins them.
60 ///
61 /// Workers drain the queue before exiting: after `_stop` is set they keep
62 /// popping and running already-queued tasks and only return once the queue is
63 /// empty, so the join blocks until every task queued before destruction has
64 /// run. Tasks `post()`ed concurrently with or after destruction race this and
65 /// may be silently lost.
66 ~ThreadPoolExecutor() override {
67 {
68 // notify_all() while still holding _m, not after releasing it: a
69 // waiter woken by a notify issued after unlocking could reacquire
70 // the lock, see _stop, return from wait(), and let this object be
71 // destroyed while this thread is still physically inside the
72 // notify call -- a real data race on the condition variable's
73 // internal state (this pattern is what post()'s own notify_one(),
74 // below, must avoid too, for the same reason).
75 std::scoped_lock const lock{_m};
76 _stop = true;
77 _cv.notify_all();
78 }
79 for (auto& worker : _workers) {
80 worker.join();
81 }
82 }
83
84 /// @brief Enqueues @p task for execution on one of the pool threads.
85 /// @param task Callable to execute. Thread-safe.
86 void post(std::function<void()> task) override {
87 std::scoped_lock const lock{_m};
88 _q.push(std::move(task));
89 _cv.notify_one();
90 }
91
92private:
93 void loop() {
94 for (;;) {
95 std::function<void()> task;
96 {
97 std::unique_lock lock{_m};
98 _cv.wait(lock, [this] { return _stop || !_q.empty(); });
99 if (_stop && _q.empty()) {
100 return;
101 }
102 task = std::move(_q.front());
103 _q.pop();
104 }
105 try {
106 task();
107 } catch (const std::exception& exc) {
108 // A task failure must not kill the worker or unrelated tasks, but
109 // it must not vanish either — log and carry on.
110 ::morph::log::logError("[thread-pool] task threw: " + std::string{exc.what()});
111 } catch (...) {
112 ::morph::log::logError("[thread-pool] task threw unknown exception");
113 }
114 }
115 }
116 std::mutex _m;
117 std::condition_variable _cv;
118 std::queue<std::function<void()>> _q;
119 std::vector<std::thread> _workers;
120 bool _stop = false;
121};

123/// @brief Single-thread executor intended for use on the calling (main) thread.
124///
125/// Tasks posted from other threads are collected and dispatched only when
126/// the caller invokes `runFor()`. Useful in tests or event loops that have no
127/// native dispatcher.
128class MainThreadExecutor : public IExecutor {
129public:
130 /// @brief Enqueues @p task to be run on the next `runFor()` call.
131 ///
132 /// Thread-safe. The task is *not* executed immediately.
133 /// @param task Callable to execute.
134 void post(std::function<void()> task) override {
135 {
136 std::scoped_lock const lock{_m};
137 _q.push(std::move(task));
138 }
139 _cv.notify_all();
140 }
141
142 /// @brief Pumps the task queue for up to @p timeout.
143 ///
144 /// Runs queued tasks one by one. It does **not** return early when the queue
145 /// drains: while time remains before the deadline it blocks waiting for
146 /// newly posted tasks, and only returns once the deadline is reached (with
147 /// any still-queued tasks left for a subsequent call — this is a pump, not a
148 /// flush). Exceptions thrown by tasks are logged and execution continues with
149 /// the next task.
150 ///
151 /// Must be called from the thread that "owns" this executor.
152 /// @param timeout Maximum wall-clock time to spend draining.
153 void runFor(std::chrono::milliseconds timeout) {
154 auto deadline = std::chrono::steady_clock::now() + timeout;
155 while (std::chrono::steady_clock::now() < deadline) {
156 std::function<void()> task;
157 {
158 std::unique_lock lock{_m};
159 if (!_cv.wait_until(lock, deadline, [this] { return !_q.empty(); })) {
160 return;
161 }
162 task = std::move(_q.front());
163 _q.pop();
164 }
165 runTask(std::move(task));
166 }
167 }
168
169 /// @brief Dequeues and runs at most one pending task, without blocking.
170 ///
171 /// Never waits for a task to appear: if the queue is empty, returns `false`
172 /// immediately. Otherwise pops exactly one task, runs it, and returns `true`
173 /// whether or not that task threw. Unlike `runFor()`, there is no wall-clock
174 /// wait — this is the single-step primitive for callers (e.g. tests or
175 /// external event loops) that want deterministic, one-task-at-a-time control
176 /// instead of a timed pump.
177 ///
178 /// Must be called from the thread that "owns" this executor.
179 /// @return `true` if a task was dequeued and run, `false` if the queue was empty.
180 bool runOnce() {
181 std::function<void()> task;
182 {
183 std::scoped_lock const lock{_m};
184 if (_q.empty()) {
185 return false;
186 }
187 task = std::move(_q.front());
188 _q.pop();
189 }
190 runTask(std::move(task));
191 return true;
192 }
193
194 /// @brief Runs tasks until the queue is empty, with no wall-clock timeout.
195 ///
196 /// Repeatedly calls the same dequeue-and-invoke step as `runOnce()` until the
197 /// queue reports empty, then returns. It does **not** wait for tasks posted
198 /// after draining begins: a task that posts new work while running extends
199 /// the drain (the new task is still queued and gets consumed), but `drain()`
200 /// never blocks waiting for an *externally* posted task the way `runFor()`
201 /// does — it simply stops once the queue is observed empty.
202 ///
203 /// Must be called from the thread that "owns" this executor.
204 void drain() {
205 while (runOnce()) {
206 // Keep consuming until the queue reports empty.
207 }
208 }
209
210private:
211 void runTask(std::function<void()> task) {
212 try {
213 task();
214 } catch (const std::exception& exc) {
215 ::morph::log::logError("[main-thread] callback threw: " + std::string{exc.what()});
216 }
217 }
218
219 std::mutex _m;
220 std::condition_variable _cv;
221 std::queue<std::function<void()>> _q;
222};
