44/// @brief What a frontend lends the source it opens and the screens it mounts.

58 /// @return The runtime.

126/// @brief Where the screens and the models are: a server (remote) or the application linked in-process (local).
127///
128/// `main` builds one from its arguments and hands it to `Frontend::run`. Every call is made on the context's owner
129/// executor, and every callback it is given must be called there too.

131public:

133 virtual ~AppSource() = default;
134 AppSource(AppSource const&) = delete;
135 AppSource& operator=(AppSource const&) = delete;
136 AppSource(AppSource&&) = delete;
137 AppSource& operator=(AppSource&&) = delete;
138
139 /// @brief Connects (or, local, sets up the models), signs the user in natively if the application needs it, and
140 /// delivers the bundle.
141 /// @param context The running frontend's context; its executor is every bridge's callback executor.
142 /// @param ready Called once, on the owner, now or later, with the bundle or why there is none. A source that is
143 /// closed first never calls it.
144 virtual void open(AppContext& context, std::function<void(std::expected<Bundle, ConnectError>)> ready) = 0;
145
146 /// @brief Replaces the dispatch target under the mounted screens: `Bridge::switchBackend`, posted to the owner.

152 /// @brief Releases everything `open` made (connections, bridges, handlers) and drops a `ready` not yet
153 /// delivered. Called by `runApp` once the screens are gone and before the frontend's runtime is; the

158/// @brief A frontend: owns an event loop and a renderer, and runs one source's screens.
159class Frontend {
160public:
161 Frontend() = default;
162 virtual ~Frontend() = default;
163 Frontend(Frontend const&) = delete;
164 Frontend& operator=(Frontend const&) = delete;
165 Frontend(Frontend&&) = delete;
166 Frontend& operator=(Frontend&&) = delete;
167
168 /// @brief The name `selectFrontend` matches.
169 /// @return The name.
170 [[nodiscard]] virtual std::string_view name() const = 0;
171
172 /// @brief Builds the runtime and its executor, opens @p source, mounts the app shell and drives the loop until
173 /// `AppContext::quit`.
174 ///
175 /// Tears down in reverse: the screens, then the source's connections, then the runtime, then the loop and the
176 /// toolkit. A frontend leaves the first two to `runApp`, which fixes that order, and keeps the rest by
177 /// declaring its members or locals in construction order.
178 /// @param source Where the screens and the models are; it outlives the call.
179 /// @return The exit code passed to `quit`; a frontend may also end the run on its own (end of input, an
180 /// interrupt) with a code it documents.
181 virtual int run(AppSource& source) = 0;
182};

194/// @brief Why no frontend was selected.
195struct FrontendError {
196 /// @brief What went wrong.
197 enum class Kind : std::uint8_t {
198 MissingName, ///< A `--ui` without a name.
199 UnknownName, ///< `--ui` or `MORPH_UI` names a frontend this binary was not built with.
200 NoneUsable, ///< Nothing was named, and no built frontend is usable here.
201 NotMade, ///< The chosen option has no `make`, or its `make` returned null.
202 };
203 /// @brief What went wrong.
204 Kind kind = Kind::NoneUsable;
205 /// @brief The name asked for, or the option's name for `NotMade`; empty otherwise.
206 std::string requested;
207 /// @brief The names of the frontends this binary was built with, in order.
208 std::vector<std::string> built;
209
210 /// @brief A message for the user, naming the built frontends.
211 /// @return The message.
212 [[nodiscard]] std::string message() const {
213 std::string names;
214 for (std::string const& name : built) {
215 names += names.empty() ? name : ", " + name;
216 }
217 if (names.empty()) {
218 names = "none";
219 }
220 switch (kind) {
221 case Kind::MissingName:
222 return "morph::ui: --ui needs a frontend name (built: " + names + ")";
223 case Kind::UnknownName:
224 return "morph::ui: no frontend named '" + requested + "' is built (built: " + names + ")";
225 case Kind::NotMade:
226 return "morph::ui: frontend '" + requested + "' could not be made";
227 case Kind::NoneUsable:
228 default:
229 return "morph::ui: no built frontend is usable here (built: " + names + ")";
230 }
231 }
232};

234/// @brief Reads an environment variable: its value, or `nullopt` when it is unset.
235using EnvironmentReader = std::function<std::optional<std::string>(std::string_view name)>;
236
237/// @brief The process environment, read through `std::getenv`.
238/// @return A reader over the process environment.
239[[nodiscard]] inline EnvironmentReader processEnvironment() {
240 return [](std::string_view name) -> std::optional<std::string> {
241 std::string const key{name};
242 // getenv races only with a concurrent setenv; frontend selection runs in main before any thread exists.
243 // MSVC deprecates getenv (C4996) in favour of _dupenv_s; a consumer built with /WX must not fail on it.
244 // clang-cl reports the same deprecation as -Wdeprecated-declarations, which the MSVC pragma does not reach.
245 // The MSVC pragma stays on the line just before the call, the only line it covers.
246#if defined(__clang__) && defined(_WIN32)
247#pragma clang diagnostic push
248#pragma clang diagnostic ignored "-Wdeprecated-declarations"
249#endif
250#ifdef _MSC_VER
251#pragma warning(suppress : 4996)
252#endif
253 char const* const value = std::getenv(key.c_str()); // NOLINT(concurrency-mt-unsafe)
254#if defined(__clang__) && defined(_WIN32)
255#pragma clang diagnostic pop
256#endif
257 if (value == nullptr) {
258 return std::nullopt;
259 }
260 return std::string{value};
261 };
262}

266/// @brief The command line's choice of frontend, and the arguments that made it.
267struct CommandLineChoice {
268 /// @brief The last name given, or none.
269 std::optional<std::string> name;
270 /// @brief Whether a `--ui` was given without a name after it, or the last name given is empty.
271 bool missingName = false;
272 /// @brief The positions in the arguments of every `--ui=<name>`, `--ui` and the name after it, ascending.
273 std::vector<std::size_t> used;
274};
275
276/// @brief Reads `--ui=<name>` and `--ui <name>` from the arguments, up to a `--`.
277///
278/// The last one wins. An argument starting with `--` is never taken as the name after `--ui`: it is the next option,
279/// and the `--ui` before it has no name.
280/// @param args The arguments; a program name among them is never a `--ui` flag.
281/// @return The choice.
282[[nodiscard]] inline CommandLineChoice readCommandLine(std::span<std::string const> args) {
283 CommandLineChoice choice;
284 bool nameFollows = false;
285 for (std::size_t index = 0; std::string const& raw : args) {
286 std::string_view const arg = raw;
287 if (nameFollows) {
288 nameFollows = false;
289 if (arg.starts_with("--")) {
290 choice.missingName = true;
291 break;
292 }
293 choice.name = std::string{arg};
294 choice.used.push_back(index);
295 } else if (arg == "--") {
296 break;
297 } else if (arg == "--ui") {
298 nameFollows = true;
299 choice.used.push_back(index);
300 } else if (arg.starts_with("--ui=")) {
301 choice.name = std::string{arg.substr(std::string_view{"--ui="}.size())};
302 choice.used.push_back(index);
303 }
304 ++index;
305 }
306 choice.missingName = choice.missingName || nameFollows || (choice.name.has_value() && choice.name->empty());
307 return choice;
308}

312/// @brief Picks the frontend to run: the last `--ui=<name>` (or `--ui <name>`) before a `--`, else a non-empty
313/// `MORPH_UI`, else the first option in the given order whose `usable()` holds.
314///
315/// A frontend named on the command line or in `MORPH_UI` is used even when its `usable()` is false: the user asked
316/// for it. On success every `--ui` argument (with the name after a bare `--ui`) is removed from @p args, and the rest
317/// keep their order, for the caller to pass on; on an error @p args is left as it was.
318/// @param built The frontends this binary was built with, in order of preference.
319/// @param args The command-line arguments, with or without the program name first.
320/// @param env Reads `MORPH_UI`; the process environment by default. An empty reader reads nothing.
321/// @return The frontend, made; or why there is none.
322[[nodiscard]] inline std::expected<std::unique_ptr<Frontend>, FrontendError> selectFrontend(
323 std::span<FrontendOption const> built, std::vector<std::string>& args,
324 EnvironmentReader const& env = processEnvironment()) {
325 std::vector<std::string> names;
326 names.reserve(built.size());
327 for (FrontendOption const& option : built) {
328 names.push_back(option.name);
329 }
330 auto const failure = [&names](FrontendError::Kind kind, std::string requested) {
331 return std::unexpected{FrontendError{.kind = kind, .requested = std::move(requested), .built = names}};
332 };
333 detail::CommandLineChoice const choice = detail::readCommandLine(args);
334 if (choice.missingName) {
335 return failure(FrontendError::Kind::MissingName, {});
336 }
337 std::optional<std::string> requested = choice.name;
338 if (!requested.has_value() && env) {
339 if (std::optional<std::string> fromEnvironment = env("MORPH_UI");
340 fromEnvironment.has_value() && !fromEnvironment->empty()) {
341 requested = std::move(fromEnvironment);
342 }
343 }
344 FrontendOption const* chosen = nullptr;
345 if (requested.has_value()) {
346 auto const found = std::ranges::find(built, *requested, &FrontendOption::name);
347 if (found == built.end()) {
348 return failure(FrontendError::Kind::UnknownName, *requested);
349 }
350 chosen = &*found;
351 } else {
352 auto const found = std::ranges::find_if(
353 built, [](FrontendOption const& candidate) { return !candidate.usable || candidate.usable(); });
354 if (found == built.end()) {
355 return failure(FrontendError::Kind::NoneUsable, {});
356 }
357 chosen = &*found;
358 }
359 std::unique_ptr<Frontend> frontend = chosen->make ? chosen->make() : nullptr;
360 if (frontend == nullptr) {
361 return failure(FrontendError::Kind::NotMade, chosen->name);
362 }
363 for (std::size_t const index : choice.used | std::views::reverse) {
364 args.erase(args.begin() + static_cast<std::ptrdiff_t>(index));
365 }
366 return frontend;
367}

370/// did not open. Destroying it unmounts all of it.
371class MountedShell {
372public:
373 MountedShell() = default;
374 virtual ~MountedShell() = default;
375 MountedShell(MountedShell const&) = delete;

377 MountedShell(MountedShell&&) = delete;
378 MountedShell& operator=(MountedShell&&) = delete;

381/// @brief Mounts the shell for what the source delivered. Called once, on the owner; it may return null to mount
382/// nothing (after calling `AppContext::quit`, for example).
383using ShellFactory =
384 std::function<std::unique_ptr<MountedShell>(AppContext& context, std::expected<Bundle, ConnectError> opened)>;
385
386/// @brief Runs one source inside a frontend: opens it, mounts the shell when it is ready, runs the loop, and tears
387/// down in order.
388///
389/// The order is fixed here so that every frontend keeps it. When the loop returns, or throws, the shell is destroyed
390/// first, while the source's connections that its screens use are still open, and the source is closed next, while
391/// the runtime and the executor its handlers were made with are still alive; the frontend destroys those after this
392/// returns.
393///
394/// The source's `ready` may come during `open` (a local source) or later from the loop (a remote one). A second
395/// `ready` is ignored, and one that comes after the run has ended does nothing. A @p mountShell that throws quits the
396/// context with exit code 1, and its exception leaves this call once everything is torn down.
397/// @param context The frontend's context; its runtime must outlive this call.
398/// @param source The source to open; closed before this returns.
399/// @param mountShell Mounts the shell for what the source delivered.
400/// @param loop Runs the frontend's event loop and returns the exit code once `AppContext::quit` ends it.
401/// @return What @p loop returns.
402/// @throws std::invalid_argument when @p mountShell or @p loop is empty; whatever `open`, @p mountShell, @p loop or
403/// `close` throws, the first of them.
404[[nodiscard]] inline int runApp(AppContext& context, AppSource& source, ShellFactory const& mountShell,
405 std::function<int()> const& loop) {
406 if (!mountShell || !loop) {
407 throw std::invalid_argument{"morph::ui::runApp: the shell factory and the loop must not be empty"};
408 }
409 struct Run {
410 std::unique_ptr<MountedShell> shell;
411 bool opened = false;
412 std::exception_ptr mountFailure;
413 };
414 auto const run = std::make_shared<Run>();
415 std::exception_ptr failure;
416 int exitCode = 0;
417 try {
418 source.open(context,
419 [weak = std::weak_ptr{run}, &context, &mountShell](std::expected<Bundle, ConnectError> opened) {
420 std::shared_ptr<Run> const live = weak.lock();
421 if (live == nullptr || live->opened) {
422 return;
423 }
424 live->opened = true;
425 try {
426 live->shell = mountShell(context, std::move(opened));
427 } catch (...) {
428 live->mountFailure = std::current_exception();
429 context.quit(1);
430 }
431 });
432 exitCode = loop();
433 } catch (...) {
434 failure = std::current_exception();
435 }
436 // The shell goes first, and the run counts as opened from here on, so a `ready` that arrives while the source
437 // closes mounts nothing; one that arrives after this returns finds the run gone.
438 std::exception_ptr const mountFailure = run->mountFailure;
439 run->shell.reset();
440 run->opened = true;
441 try {
442 source.close();
443 } catch (...) {
444 if (!failure) {
445 failure = std::current_exception();
446 }
447 }
448 if (!failure) {
449 failure = mountFailure;
450 }
451 if (failure) {
452 std::rethrow_exception(failure);
453 }
454 return exitCode;
455}
