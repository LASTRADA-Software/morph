5/// @file util/datetime.hpp
6/// @brief UTC timestamps for morph actions.
7///
8/// Two types, mirroring the `Rational`/`Quantity` split:
9///
10/// - `morph::time::DateTime` — the *value*: a UTC instant with millisecond
11/// precision over `std::chrono::sys_time` (adapted from LASTRADA
12/// `Toolbox/Chrono.hpp`). Cheap to copy, ordered, duration arithmetic.
13/// - `morph::time::Timestamp` — the *field*: an optionally-empty `DateTime`
14/// with the same one-kind-of-empty semantics as `Quantity` (a form draft
15/// starts blank; required-ness is derived by `morph::forms`).
16///
17/// @par Wire format
18/// A `DateTime` travels as the ISO-8601 string
19/// `"YYYY-MM-DDTHH:MM:SS[.mmm]Z"` (UTC only; the fractional part and the
20/// trailing `Z` are optional on input, always written on output). Parsing is
21/// a strict, hand-rolled routine — no locale, no `std::chrono::parse`
22/// dependency — so behaviour is identical across libstdc++, libc++ (WASM),
23/// and MSVC. Unlike the `Rational` codec, which clamps hostile input into a
24/// valid value, a malformed timestamp has no meaningful clamp: the codec
25/// rejects it as a JSON **read error**.
26///
27/// @par Schema
28/// A `Timestamp` field renders as `{"type": ["string","null"],
29/// "format": "date-time"}` — the standard JSON-Schema vocabulary form
30/// renderers key on (the demo clients render a date-time input for it).
31

45/// @brief A UTC instant with millisecond precision.
46///
47/// The default state is the Unix epoch — `DateTime` itself has no empty
48/// state; use `Timestamp` for fields that may be blank.
49struct DateTime {
50 /// @brief The underlying UTC instant (Unix time, leap seconds ignored).
51 std::chrono::sys_time<std::chrono::milliseconds> value;
52
53 /// @brief Constructs the Unix epoch (1970-01-01T00:00:00.000Z).
54 constexpr DateTime() noexcept = default;
55
56 /// @brief Wraps an existing UTC instant.
57 /// @param instant The UTC instant to wrap.
58 constexpr DateTime(std::chrono::sys_time<std::chrono::milliseconds> instant) noexcept : value{instant} {}
59
60 /// @brief Composes a UTC instant from calendar/clock components.
61 /// @param year Calendar year.
62 /// @param month Month-of-year in [1, 12].
63 /// @param day Day-of-month in [1, 31].
64 /// @param hour Hours in [0, 23].
65 /// @param minute Minutes in [0, 59].
66 /// @param second Seconds in [0, 59].
67 /// @param millisecond Milliseconds in [0, 999].
68 constexpr DateTime(std::chrono::year year, std::chrono::month month, std::chrono::day day,
69 std::chrono::hours hour, std::chrono::minutes minute, std::chrono::seconds second,
70 std::chrono::milliseconds millisecond = std::chrono::milliseconds{0}) noexcept
71 : value{static_cast<std::chrono::sys_days>(std::chrono::year_month_day{year, month, day}) + hour + minute +
72 second + millisecond} {}
73
74 /// @brief The current UTC instant, truncated to milliseconds.
75 /// @return The current UTC date-time.
76 [[nodiscard]] static DateTime now() noexcept {
77 return DateTime{std::chrono::time_point_cast<std::chrono::milliseconds>(std::chrono::system_clock::now())};
78 }

80 /// @brief Renders as ISO-8601 UTC: `YYYY-MM-DDTHH:MM:SS.mmmZ`.
81 /// @return The formatted instant.
82 [[nodiscard]] std::string toIso8601() const {
83 auto const wholeDays = std::chrono::floor<std::chrono::days>(value);
84 auto const calendar = std::chrono::year_month_day{wholeDays};
85 auto const clock = std::chrono::hh_mm_ss{value - wholeDays};
86 return std::format("{:04}-{:02}-{:02}T{:02}:{:02}:{:02}.{:03}Z", static_cast<int>(calendar.year()),
87 static_cast<unsigned>(calendar.month()), static_cast<unsigned>(calendar.day()),
88 clock.hours().count(), clock.minutes().count(), clock.seconds().count(),
89 clock.subseconds().count());
90 }
91
92 /// @brief Strictly parses ISO-8601 UTC: `YYYY-MM-DDTHH:MM:SS`, optionally
93 /// followed by `.m`/`.mm`/`.mmm` and an optional trailing `Z`.
94 ///
95 /// The calendar date must exist (2026-02-30 is rejected) and clock
96 /// components must be in range; no time zones other than `Z`, no
97 /// lowercase separators, no trailing input. The fixed-width date and clock
98 /// fields (month, day, hour, minute, second, fraction) reject a leading
99 /// sign; only the year field may carry a `-` (negative years).
100 /// @param text Candidate ISO-8601 string.
101 /// @return The parsed instant, or `std::nullopt` when @p text is malformed.
102 // NOLINTBEGIN(cppcoreguidelines-pro-bounds-pointer-arithmetic, cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
103 [[nodiscard]] static std::optional<DateTime> fromIso8601(std::string_view text) noexcept {
104 // Callers only pass offsets within the length-checked prefix below.
105 // `allowSign` gates the leading '-'/'+' that `std::from_chars` would
106 // otherwise accept: the whole-string year is the only field where a
107 // sign is legitimate (negative years). For the fixed-width date/clock
108 // fields a leading sign is a sign-injection attack — "T-5:30:15" would
109 // read hour = -5 and silently shift to a different valid instant — so
110 // those callers reject it and the value is forced non-negative.
111 auto const number = [&text](std::size_t offset, std::size_t count,
112 bool allowSign) noexcept -> std::optional<int> {
113 if (!allowSign && count > 0 && offset < text.size() &&
114 (text[offset] == '-' || text[offset] == '+')) {
115 return std::nullopt;
116 }
117 int parsed = 0;
118 auto const* first = text.data() + offset;
119 auto const [end, errc] = std::from_chars(first, first + count, parsed);
120 if (errc != std::errc{} || end != first + count) {
121 return std::nullopt;
122 }
123 return parsed;
124 };
125 auto const separator = [&text](std::size_t offset, char expected) noexcept {
126 return offset < text.size() && text[offset] == expected;
127 };
128
129 if (text.size() < 19 || !separator(4, '-') || !separator(7, '-') || !separator(10, 'T') ||
130 !separator(13, ':') || !separator(16, ':')) {
131 return std::nullopt;
132 }
133 auto const year = number(0, 4, /*allowSign=*/true);
134 auto const month = number(5, 2, /*allowSign=*/false);
135 auto const day = number(8, 2, /*allowSign=*/false);
136 auto const hour = number(11, 2, /*allowSign=*/false);
137 auto const minute = number(14, 2, /*allowSign=*/false);
138 auto const second = number(17, 2, /*allowSign=*/false);
139 if (!year || !month || !day || !hour || !minute || !second) {
140 return std::nullopt;
141 }
142
143 std::size_t cursor = 19;
144 int milliseconds = 0;
145 if (separator(cursor, '.')) {
146 ++cursor;
147 std::size_t digits = 0;
148 int fraction = 0;
149 while (digits < 3 && cursor < text.size() && text[cursor] >= '0' && text[cursor] <= '9') {
150 fraction = (fraction * 10) + (text[cursor] - '0');
151 ++cursor;
152 ++digits;
153 }
154 if (digits == 0) {
155 return std::nullopt;
156 }
157 for (std::size_t pad = digits; pad < 3; ++pad) {
158 fraction *= 10;
159 }
160 milliseconds = fraction;
161 }
162 if (separator(cursor, 'Z')) {
163 ++cursor;
164 }
165 if (cursor != text.size()) {
166 return std::nullopt;
167 }
168
169 auto const calendar = std::chrono::year_month_day{std::chrono::year{*year},
170 std::chrono::month{static_cast<unsigned>(*month)},
171 std::chrono::day{static_cast<unsigned>(*day)}};
172 if (!calendar.ok() || *hour > 23 || *minute > 59 || *second > 59) {
173 return std::nullopt;
174 }
175 return DateTime{std::chrono::year{*year},
176 std::chrono::month{static_cast<unsigned>(*month)},
177 std::chrono::day{static_cast<unsigned>(*day)},
178 std::chrono::hours{*hour},
179 std::chrono::minutes{*minute},
180 std::chrono::seconds{*second},
181 std::chrono::milliseconds{milliseconds}};
182 }
183 // NOLINTEND(cppcoreguidelines-pro-bounds-pointer-arithmetic, cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
184
185 /// @brief Ordering on the underlying instant.
186 /// @param other Instant to compare against.
187 /// @return The relative ordering of the two instants.
188 [[nodiscard]] constexpr std::strong_ordering operator<=>(const DateTime& other) const noexcept = default;

190 /// @brief Shifts this instant by an arbitrary chrono duration.
191 /// @param delta Duration to add (may be negative).
192 /// @return `*this`.
193 template <typename Rep, typename Period>
194 constexpr DateTime& operator+=(std::chrono::duration<Rep, Period> delta) noexcept {
195 value += std::chrono::duration_cast<std::chrono::milliseconds>(delta);
196 return *this;
197 }
198
199 /// @brief Shifts this instant backwards by an arbitrary chrono duration.
200 /// @param delta Duration to subtract (may be negative).
201 /// @return `*this`.
202 template <typename Rep, typename Period>
203 constexpr DateTime& operator-=(std::chrono::duration<Rep, Period> delta) noexcept {
204 value -= std::chrono::duration_cast<std::chrono::milliseconds>(delta);
205 return *this;
206 }

208 /// @brief An instant @p delta after @p lhs.
209 /// @param lhs Starting instant.
210 /// @param delta Duration offset.
211 /// @return The shifted instant.
212 template <typename Rep, typename Period>
213 [[nodiscard]] friend constexpr DateTime operator+(DateTime lhs, std::chrono::duration<Rep, Period> delta) noexcept {
214 lhs += delta;
215 return lhs;
216 }
217
218 /// @brief An instant @p delta before @p lhs.
219 /// @param lhs Starting instant.
220 /// @param delta Duration offset.
221 /// @return The shifted instant.
222 template <typename Rep, typename Period>
223 [[nodiscard]] friend constexpr DateTime operator-(DateTime lhs, std::chrono::duration<Rep, Period> delta) noexcept {
224 lhs -= delta;
225 return lhs;
226 }
227
228 /// @brief The signed difference between two instants.
229 /// @param lhs Minuend.
230 /// @param rhs Subtrahend.
231 /// @return `lhs - rhs` as a chrono duration.
232 [[nodiscard]] friend constexpr auto operator-(const DateTime& lhs, const DateTime& rhs) noexcept {
233 return lhs.value - rhs.value;

237/// @brief An optionally-empty timestamp field.
238///
239/// The blank state ("not entered / not recorded") lives inside, exactly like
240/// `morph::units::Quantity`: action structs never wrap a `Timestamp` in
241/// `std::optional`, and a non-optional `Timestamp` member is *required* by
242/// the `morph::forms` rules.
243struct Timestamp {
244 /// @brief The payload; `std::nullopt` means "not entered".
245 std::optional<DateTime> value;
246
247 /// @brief Constructs the empty state.
248 constexpr Timestamp() noexcept = default;
249
250 /// @brief Engages with @p instant.
251 /// @param instant The instant to hold.
252 constexpr Timestamp(DateTime instant) noexcept : value{instant} {}
253
254 /// @brief Adopts an optional payload as-is.
255 /// @param payload Engaged or empty payload.
256 constexpr Timestamp(std::optional<DateTime> payload) noexcept : value{payload} {}
257
258 /// @brief An engaged timestamp holding the current UTC instant.
259 /// @return `Timestamp{DateTime::now()}`.
260 [[nodiscard]] static Timestamp now() noexcept { return Timestamp{DateTime::now()}; }
261
262 /// @brief Whether an instant has been entered.
263 /// @return `true` if the payload is engaged.
264 [[nodiscard]] constexpr bool hasValue() const noexcept { return value.has_value(); }
265
266 /// @brief Unchecked access to the engaged instant (UB when empty, exactly
267 /// like `std::optional` — the unchecked contract is the point).
268 /// @return The engaged instant.
269 // NOLINTNEXTLINE(bugprone-unchecked-optional-access)
270 [[nodiscard]] constexpr const DateTime& operator*() const noexcept { return *value; }
271
272 /// @brief Ordering/equality on the payload; empty sorts before engaged.
273 /// @param other Timestamp to compare against.
274 /// @return The ordering of the two payloads.
275 [[nodiscard]] constexpr auto operator<=>(const Timestamp& other) const noexcept = default;
276};
277
278/// @brief The signed duration between two engaged timestamps.
279/// @param lhs Minuend.
280/// @param rhs Subtrahend.
281/// @return `lhs - rhs` as a chrono duration, or `std::nullopt` if either is empty.
282[[nodiscard]] constexpr std::optional<std::chrono::milliseconds> operator-(const Timestamp& lhs,
283 const Timestamp& rhs) noexcept {
284 if (!lhs.value || !rhs.value) {
285 return std::nullopt;
286 }
287 return *lhs.value - *rhs.value;
288}

298/// @brief Reads a `DateTime` from a JSON ISO-8601 string; malformed content
299/// sets a syntax error on the context (it is not clampable input).
300template <>
301struct from<JSON, morph::time::DateTime> {
302 template <auto Opts>
303 static void op(auto&& value, is_context auto&& ctx, auto&& iter, auto end) {
304 std::string str{};
305 parse<JSON>::op<Opts>(str, ctx, iter, end);
306 if (bool(ctx.error)) [[unlikely]] {
307 return;
308 }
309 if (auto parsed = morph::time::DateTime::fromIso8601(str); parsed.has_value()) {
310 value = *parsed;
311 return;
312 }
313 ctx.error = error_code::syntax_error;
314 }
315};
316
317/// @brief Writes a `DateTime` as its ISO-8601 JSON string.
318template <>
319struct to<JSON, morph::time::DateTime> {
320 template <auto Opts>
321 static void op(auto&& value, is_context auto&& ctx, auto&&... args) {
322 serialize<JSON>::op<Opts>(value.toIso8601(), ctx, args...);
323 }
324};

354/// @brief std::format support for `morph::time::DateTime` (empty spec only):
355/// renders the ISO-8601 UTC form.
356template <>
357struct std::formatter<morph::time::DateTime> {
358 // NOLINTBEGIN(readability-convert-member-functions-to-static) — std::formatter requires non-static methods
359 constexpr auto parse(std::format_parse_context& ctx) {
360 // The parse context always contains the terminating '}' (the format
361 // machinery's contract), so the range is never empty here.
362 if (*ctx.begin() != '}') {
363 throw std::format_error{"morph::time::DateTime accepts only the empty format spec '{}'"};
364 }
365 return ctx.begin();
366 }
367
368 auto format(const morph::time::DateTime& value, std::format_context& ctx) const {
369 return std::format_to(ctx.out(), "{}", value.toIso8601());
370 }
371};
