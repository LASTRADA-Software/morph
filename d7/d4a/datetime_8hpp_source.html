5/// @file datetime.hpp
6/// @brief UTC timestamps for morph actions.
7///
8/// Two types, mirroring the `Rational`/`Quantity` split:
9///
10/// - `morph::time::DateTime` — the *value*: a UTC instant with millisecond
11/// precision over `std::chrono::sys_time` (adapted from LASTRADA
12/// `Toolbox/Chrono.hpp`). Cheap to copy, ordered, duration arithmetic.
13/// - `morph::time::Timestamp` — the *field*: an optionally-empty `DateTime`
14/// with the same one-kind-of-empty semantics as `Quantity` (a form draft
15/// starts blank; required-ness is derived by `morph::forms`).
16///
17/// @par Wire format
18/// A `DateTime` travels as the ISO-8601 string
19/// `"YYYY-MM-DDTHH:MM:SS[.mmm]Z"` (UTC only; the fractional part and the
20/// trailing `Z` are optional on input, always written on output). Parsing is
21/// a strict, hand-rolled routine — no locale, no `std::chrono::parse`
22/// dependency — so behaviour is identical across libstdc++, libc++ (WASM),
23/// and MSVC. Unlike the `Rational` codec, which clamps hostile input into a
24/// valid value, a malformed timestamp has no meaningful clamp: the codec
25/// rejects it as a JSON **read error**.
26///
27/// @par Schema
28/// A `Timestamp` field renders as `{"type": ["string","null"],
29/// "format": "date-time"}` — the standard JSON-Schema vocabulary form
30/// renderers key on (the demo clients render a date-time input for it).

45/// @brief A UTC instant with millisecond precision.
46///
47/// The default state is the Unix epoch — `DateTime` itself has no empty
48/// state; use `Timestamp` for fields that may be blank.
49struct DateTime {
50 /// @brief The underlying UTC instant (Unix time, leap seconds ignored).
51 std::chrono::sys_time<std::chrono::milliseconds> value;
52
53 /// @brief Constructs the Unix epoch (1970-01-01T00:00:00.000Z).
54 constexpr DateTime() noexcept = default;
55
56 /// @brief Wraps an existing UTC instant.
57 /// @param instant The UTC instant to wrap.
58 constexpr DateTime(std::chrono::sys_time<std::chrono::milliseconds> instant) noexcept : value{instant} {}
59
60 /// @brief Composes a UTC instant from calendar/clock components.
61 /// @param year Calendar year.
62 /// @param month Month-of-year in [1, 12].
63 /// @param day Day-of-month in [1, 31].
64 /// @param hour Hours in [0, 23].
65 /// @param minute Minutes in [0, 59].
66 /// @param second Seconds in [0, 59].
67 /// @param millisecond Milliseconds in [0, 999].
68 constexpr DateTime(std::chrono::year year, std::chrono::month month, std::chrono::day day,
69 std::chrono::hours hour, std::chrono::minutes minute, std::chrono::seconds second,
70 std::chrono::milliseconds millisecond = std::chrono::milliseconds{0}) noexcept
71 : value{static_cast<std::chrono::sys_days>(std::chrono::year_month_day{year, month, day}) + hour + minute +
72 second + millisecond} {}
73
74 /// @brief The current UTC instant, truncated to milliseconds.
75 /// @return The current UTC date-time.
76 [[nodiscard]] static DateTime now() noexcept {
77 return DateTime{std::chrono::time_point_cast<std::chrono::milliseconds>(std::chrono::system_clock::now())};
78 }
79
80 /// @brief Renders as ISO-8601 UTC: `YYYY-MM-DDTHH:MM:SS.mmmZ`.
81 /// @return The formatted instant.
82 [[nodiscard]] std::string toIso8601() const {
83 auto const wholeDays = std::chrono::floor<std::chrono::days>(value);
84 auto const calendar = std::chrono::year_month_day{wholeDays};
85 auto const clock = std::chrono::hh_mm_ss{value - wholeDays};
86 return std::format("{:04}-{:02}-{:02}T{:02}:{:02}:{:02}.{:03}Z", static_cast<int>(calendar.year()),
87 static_cast<unsigned>(calendar.month()), static_cast<unsigned>(calendar.day()),
88 clock.hours().count(), clock.minutes().count(), clock.seconds().count(),
89 clock.subseconds().count());
90 }
91
92 /// @brief Strictly parses ISO-8601 UTC: `YYYY-MM-DDTHH:MM:SS`, optionally
93 /// followed by `.m`/`.mm`/`.mmm` and an optional trailing `Z`.
94 ///
95 /// The calendar date must exist (2026-02-30 is rejected) and clock
96 /// components must be in range; no time zones other than `Z`, no
97 /// lowercase separators, no trailing input.
98 /// @param text Candidate ISO-8601 string.
99 /// @return The parsed instant, or `std::nullopt` when @p text is malformed.
100 [[nodiscard]] static std::optional<DateTime> fromIso8601(std::string_view text) noexcept {
101 // Callers only pass offsets within the length-checked prefix below.
102 auto const number = [&text](std::size_t offset, std::size_t count) noexcept -> std::optional<int> {
103 int parsed = 0;
104 auto const* first = text.data() + offset;
105 auto const [end, errc] = std::from_chars(first, first + count, parsed);
106 if (errc != std::errc{} || end != first + count) {
107 return std::nullopt;
108 }
109 return parsed;
110 };
111 auto const separator = [&text](std::size_t offset, char expected) noexcept {
112 return offset < text.size() && text[offset] == expected;
113 };
114
115 if (text.size() < 19 || !separator(4, '-') || !separator(7, '-') || !separator(10, 'T') ||
116 !separator(13, ':') || !separator(16, ':')) {
117 return std::nullopt;
118 }
119 auto const year = number(0, 4);
120 auto const month = number(5, 2);
121 auto const day = number(8, 2);
122 auto const hour = number(11, 2);
123 auto const minute = number(14, 2);
124 auto const second = number(17, 2);
125 if (!year || !month || !day || !hour || !minute || !second) {
126 return std::nullopt;
127 }
128
129 std::size_t cursor = 19;
130 int milliseconds = 0;
131 if (separator(cursor, '.')) {
132 ++cursor;
133 std::size_t digits = 0;
134 int fraction = 0;
135 while (digits < 3 && cursor < text.size() && text[cursor] >= '0' && text[cursor] <= '9') {
136 fraction = (fraction * 10) + (text[cursor] - '0');
137 ++cursor;
138 ++digits;
139 }
140 if (digits == 0) {
141 return std::nullopt;
142 }
143 for (std::size_t pad = digits; pad < 3; ++pad) {
144 fraction *= 10;
145 }
146 milliseconds = fraction;
147 }
148 if (separator(cursor, 'Z')) {
149 ++cursor;
150 }
151 if (cursor != text.size()) {
152 return std::nullopt;
153 }
154
155 auto const calendar = std::chrono::year_month_day{std::chrono::year{*year},
156 std::chrono::month{static_cast<unsigned>(*month)},
157 std::chrono::day{static_cast<unsigned>(*day)}};
158 if (!calendar.ok() || *hour > 23 || *minute > 59 || *second > 59) {
159 return std::nullopt;
160 }
161 return DateTime{std::chrono::year{*year},
162 std::chrono::month{static_cast<unsigned>(*month)},
163 std::chrono::day{static_cast<unsigned>(*day)},
164 std::chrono::hours{*hour},
165 std::chrono::minutes{*minute},
166 std::chrono::seconds{*second},
167 std::chrono::milliseconds{milliseconds}};
168 }
169
170 /// @brief Ordering on the underlying instant.
171 /// @param other Instant to compare against.
172 /// @return The relative ordering of the two instants.
173 [[nodiscard]] constexpr std::strong_ordering operator<=>(const DateTime& other) const noexcept = default;

175 /// @brief Shifts this instant by an arbitrary chrono duration.
176 /// @param delta Duration to add (may be negative).
177 /// @return `*this`.
178 template <typename Rep, typename Period>
179 constexpr DateTime& operator+=(std::chrono::duration<Rep, Period> delta) noexcept {
180 value += std::chrono::duration_cast<std::chrono::milliseconds>(delta);
181 return *this;
182 }
183
184 /// @brief Shifts this instant backwards by an arbitrary chrono duration.
185 /// @param delta Duration to subtract (may be negative).
186 /// @return `*this`.
187 template <typename Rep, typename Period>
188 constexpr DateTime& operator-=(std::chrono::duration<Rep, Period> delta) noexcept {
189 value -= std::chrono::duration_cast<std::chrono::milliseconds>(delta);
190 return *this;
191 }
192
193 /// @brief An instant @p delta after @p lhs.
194 /// @param lhs Starting instant.
195 /// @param delta Duration offset.
196 /// @return The shifted instant.
197 template <typename Rep, typename Period>
198 [[nodiscard]] friend constexpr DateTime operator+(DateTime lhs, std::chrono::duration<Rep, Period> delta) noexcept {
199 lhs += delta;
200 return lhs;
201 }
202
203 /// @brief An instant @p delta before @p lhs.
204 /// @param lhs Starting instant.
205 /// @param delta Duration offset.
206 /// @return The shifted instant.
207 template <typename Rep, typename Period>
208 [[nodiscard]] friend constexpr DateTime operator-(DateTime lhs, std::chrono::duration<Rep, Period> delta) noexcept {
209 lhs -= delta;
210 return lhs;
211 }
212
213 /// @brief The signed difference between two instants.
214 /// @param lhs Minuend.
215 /// @param rhs Subtrahend.
216 /// @return `lhs - rhs` as a chrono duration.
217 [[nodiscard]] friend constexpr auto operator-(const DateTime& lhs, const DateTime& rhs) noexcept {
218 return lhs.value - rhs.value;
219 }

222/// @brief An optionally-empty timestamp field.
223///
224/// The blank state ("not entered / not recorded") lives inside, exactly like
225/// `morph::units::Quantity`: action structs never wrap a `Timestamp` in
226/// `std::optional`, and a non-optional `Timestamp` member is *required* by
227/// the `morph::forms` rules.
228struct Timestamp {
229 /// @brief The payload; `std::nullopt` means "not entered".
230 std::optional<DateTime> value;
231
232 /// @brief Constructs the empty state.
233 constexpr Timestamp() noexcept = default;
234
235 /// @brief Engages with @p instant.
236 /// @param instant The instant to hold.
237 constexpr Timestamp(DateTime instant) noexcept : value{instant} {}
238
239 /// @brief Adopts an optional payload as-is.
240 /// @param payload Engaged or empty payload.
241 constexpr Timestamp(std::optional<DateTime> payload) noexcept : value{payload} {}
242
243 /// @brief An engaged timestamp holding the current UTC instant.
244 /// @return `Timestamp{DateTime::now()}`.
245 [[nodiscard]] static Timestamp now() noexcept { return Timestamp{DateTime::now()}; }
246
247 /// @brief Whether an instant has been entered.
248 /// @return `true` if the payload is engaged.
249 [[nodiscard]] constexpr bool hasValue() const noexcept { return value.has_value(); }
250
251 /// @brief Unchecked access to the engaged instant (UB when empty, exactly
252 /// like `std::optional` — the unchecked contract is the point).
253 /// @return The engaged instant.
254 // NOLINTNEXTLINE(bugprone-unchecked-optional-access)
255 [[nodiscard]] constexpr const DateTime& operator*() const noexcept { return *value; }
256
257 /// @brief Ordering/equality on the payload; empty sorts before engaged.
258 /// @param other Timestamp to compare against.
259 /// @return The ordering of the two payloads.
260 [[nodiscard]] constexpr auto operator<=>(const Timestamp& other) const noexcept = default;
261};

271/// @brief Reads a `DateTime` from a JSON ISO-8601 string; malformed content
272/// sets a syntax error on the context (it is not clampable input).
273template <>
274struct from<JSON, morph::time::DateTime> {
275 template <auto Opts>
276 static void op(auto&& value, is_context auto&& ctx, auto&& iter, auto end) {
277 std::string text{};
278 parse<JSON>::op<Opts>(text, ctx, iter, end);
279 if (bool(ctx.error)) [[unlikely]] {
280 return;
281 }
282 if (auto parsed = morph::time::DateTime::fromIso8601(text); parsed.has_value()) {
283 value = *parsed;
284 return;
285 }
286 ctx.error = error_code::syntax_error;
287 }
288};

327/// @brief std::format support for `morph::time::DateTime` (empty spec only):
328/// renders the ISO-8601 UTC form.
329template <>
330struct std::formatter<morph::time::DateTime> {
331 constexpr auto parse(std::format_parse_context& ctx) {
332 // The parse context always contains the terminating '}' (the format
333 // machinery's contract), so the range is never empty here.
334 if (*ctx.begin() != '}') {
335 throw std::format_error{"morph::time::DateTime accepts only the empty format spec '{}'"};
336 }
337 return ctx.begin();
338 }
339

341 return std::format_to(ctx.out(), "{}", value.toIso8601());
342 }
