5/// @file util/datetime.hpp
6/// @brief UTC timestamps for morph actions.
7///
8/// Two types, mirroring the `Rational`/`Quantity` split:
9///
10/// - `morph::time::DateTime` — the *value*: a UTC instant with millisecond
11/// precision over `std::chrono::sys_time` (adapted from LASTRADA
12/// `Toolbox/Chrono.hpp`). Cheap to copy, ordered, duration arithmetic.
13/// - `morph::time::Timestamp` — the *field*: an optionally-empty `DateTime`
14/// with the same one-kind-of-empty semantics as `Quantity` (a form draft
15/// starts blank; required-ness is derived by `morph::forms`).
16///
17/// @par Wire format
18/// A `DateTime` travels as the ISO-8601 string
19/// `"YYYY-MM-DDTHH:MM:SS[.mmm]Z"` (UTC only; the fractional part and the
20/// trailing `Z` are optional on input, always written on output). Parsing is
21/// a strict, hand-rolled routine — no locale, no `std::chrono::parse`
22/// dependency — so behaviour is identical across libstdc++, libc++ (WASM),
23/// and MSVC. Unlike the `Rational` codec, which clamps hostile input into a
24/// valid value, a malformed timestamp has no meaningful clamp: the codec
25/// rejects it as a JSON **read error**.
26///
27/// @par Schema
28/// A `Timestamp` field renders as `{"type": ["string","null"],
29/// "format": "date-time"}` — the standard JSON-Schema vocabulary form
30/// renderers key on (the demo clients render a date-time input for it).

51/// @brief Process-wide override for `DateTime::now()`, guarded by a mutex.
52///
53/// Mirrors `morph::journal::detail::defaultActionLogState()` — the same
54/// "mutex-guarded function-local static" shape used for the action-log and
55/// logger override seams elsewhere in the codebase. An empty `std::function`
56/// (the default) means "no override installed"; `DateTime::now()` falls back
57/// to `std::chrono::system_clock::now()` in that case. Forward-declared here
58/// (before `DateTime` is a complete type -- a function signature only needs
59/// `DateTime` to be declared, not defined) and defined out-of-line just after
60/// the `DateTime` struct, before `DateTime::now()`'s first caller needs it.
61[[nodiscard]] std::pair<std::mutex, std::function<DateTime()>>& nowOverrideState();

65/// @brief A UTC instant with millisecond precision.
66///
67/// The default state is the Unix epoch — `DateTime` itself has no empty
68/// state; use `Timestamp` for fields that may be blank.
69struct DateTime {
70 /// @brief The underlying UTC instant (Unix time, leap seconds ignored).
71 std::chrono::sys_time<std::chrono::milliseconds> value;
72
73 /// @brief Constructs the Unix epoch (1970-01-01T00:00:00.000Z).
74 constexpr DateTime() noexcept = default;
75
76 /// @brief Wraps an existing UTC instant.
77 /// @param instant The UTC instant to wrap.
78 constexpr DateTime(std::chrono::sys_time<std::chrono::milliseconds> instant) noexcept : value{instant} {}
79
80 /// @brief Composes a UTC instant from calendar/clock components.
81 /// @param year Calendar year.
82 /// @param month Month-of-year in [1, 12].
83 /// @param day Day-of-month in [1, 31].
84 /// @param hour Hours in [0, 23].
85 /// @param minute Minutes in [0, 59].
86 /// @param second Seconds in [0, 59].
87 /// @param millisecond Milliseconds in [0, 999].
88 constexpr DateTime(std::chrono::year year, std::chrono::month month, std::chrono::day day, std::chrono::hours hour,
89 std::chrono::minutes minute, std::chrono::seconds second,
90 std::chrono::milliseconds millisecond = std::chrono::milliseconds{0}) noexcept
91 : value{static_cast<std::chrono::sys_days>(std::chrono::year_month_day{year, month, day}) + hour + minute +
92 second + millisecond} {}
93
94 /// @brief The current UTC instant, truncated to milliseconds.
95 ///
96 /// Consults the process-wide clock override installed via `setNowOverride`/
97 /// `ScopedNowOverride`, if any; otherwise calls
98 /// `std::chrono::system_clock::now()` directly. This is the seam that lets
99 /// registry-constructed (remotely instantiated) models get a deterministic
100 /// "now" in tests without any constructor parameter: production code calls
101 /// `DateTime::now()` exactly as before, and a test installs an override for
102 /// its scope. See `ScopedNowOverride`.
103 ///
104 /// @warning This function is `noexcept`. If an installed override callable
105 /// throws, that exception escapes a `noexcept` function and `std::terminate()`
106 /// is called — an override must not throw. The override also runs while the
107 /// internal mutex is held, so it must not itself call `DateTime::now()` /
108 /// `Timestamp::now()`, `setNowOverride`, or construct a `ScopedNowOverride`:
109 /// the mutex is non-recursive and any of those would self-deadlock (the same
110 /// constraint `morph::log`'s sink callback documents for the same reason).
111 /// @return The current UTC date-time, or the overridden instant if one is installed.
112 [[nodiscard]] static DateTime now() noexcept {
113 auto& [mtx, overrideFn] = detail::nowOverrideState();
114 {
115 std::scoped_lock const lock{mtx};
116 if (overrideFn) {
117 return overrideFn();
118 }
119 }
120 return DateTime{std::chrono::time_point_cast<std::chrono::milliseconds>(std::chrono::system_clock::now())};
121 }
122
123 /// @brief Renders as ISO-8601 UTC: `YYYY-MM-DDTHH:MM:SS.mmmZ`.
124 /// @return The formatted instant.
125 [[nodiscard]] std::string toIso8601() const {
126 auto const wholeDays = std::chrono::floor<std::chrono::days>(value);
127 auto const calendar = std::chrono::year_month_day{wholeDays};
128 auto const clock = std::chrono::hh_mm_ss{value - wholeDays};
129 return std::format("{:04}-{:02}-{:02}T{:02}:{:02}:{:02}.{:03}Z", static_cast<int>(calendar.year()),
130 static_cast<unsigned>(calendar.month()), static_cast<unsigned>(calendar.day()),
131 clock.hours().count(), clock.minutes().count(), clock.seconds().count(),
132 clock.subseconds().count());
133 }
134
135 /// @brief Strictly parses ISO-8601 UTC: `YYYY-MM-DDTHH:MM:SS`, optionally
136 /// followed by `.m`/`.mm`/`.mmm` and an optional trailing `Z`.
137 ///
138 /// The calendar date must exist (2026-02-30 is rejected) and clock
139 /// components must be in range; no time zones other than `Z`, no
140 /// lowercase separators, no trailing input. The fixed-width date and clock
141 /// fields (month, day, hour, minute, second, fraction) reject a leading
142 /// sign; only the year field may carry a `-` (negative years).
143 /// @param text Candidate ISO-8601 string.
144 /// @return The parsed instant, or `std::nullopt` when @p text is malformed.
145 // NOLINTBEGIN(cppcoreguidelines-pro-bounds-pointer-arithmetic, cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
146 [[nodiscard]] static std::optional<DateTime> fromIso8601(std::string_view text) noexcept {
147 // Callers only pass offsets within the length-checked prefix below.
148 // `allowSign` gates the leading '-'/'+' that `std::from_chars` would
149 // otherwise accept: the whole-string year is the only field where a
150 // sign is legitimate (negative years). For the fixed-width date/clock
151 // fields a leading sign is a sign-injection attack — "T-5:30:15" would
152 // read hour = -5 and silently shift to a different valid instant — so
153 // those callers reject it and the value is forced non-negative.
154 auto const number = [&text](std::size_t offset, std::size_t count,
155 bool allowSign) noexcept -> std::optional<int> {
156 if (!allowSign && count > 0 && offset < text.size() && (text[offset] == '-' || text[offset] == '+')) {
157 return std::nullopt;
158 }
159 int parsed = 0;
160 auto const* first = text.data() + offset;
161 auto const [end, errc] = std::from_chars(first, first + count, parsed);
162 if (errc != std::errc{} || end != first + count) {
163 return std::nullopt;
164 }
165 return parsed;
166 };
167 auto const separator = [&text](std::size_t offset, char expected) noexcept {
168 return offset < text.size() && text[offset] == expected;
169 };
170
171 if (text.size() < 19 || !separator(4, '-') || !separator(7, '-') || !separator(10, 'T') ||
172 !separator(13, ':') || !separator(16, ':')) {
173 return std::nullopt;
174 }
175 auto const year = number(0, 4, /*allowSign=*/true);
176 auto const month = number(5, 2, /*allowSign=*/false);
177 auto const day = number(8, 2, /*allowSign=*/false);
178 auto const hour = number(11, 2, /*allowSign=*/false);
179 auto const minute = number(14, 2, /*allowSign=*/false);
180 auto const second = number(17, 2, /*allowSign=*/false);
181 if (!year || !month || !day || !hour || !minute || !second) {
182 return std::nullopt;
183 }
184
185 std::size_t cursor = 19;
186 int milliseconds = 0;
187 if (separator(cursor, '.')) {
188 ++cursor;
189 std::size_t digits = 0;
190 int fraction = 0;
191 while (digits < 3 && cursor < text.size() && text[cursor] >= '0' && text[cursor] <= '9') {
192 fraction = (fraction * 10) + (text[cursor] - '0');
193 ++cursor;
194 ++digits;
195 }
196 if (digits == 0) {
197 return std::nullopt;
198 }
199 for (std::size_t pad = digits; pad < 3; ++pad) {
200 fraction *= 10;
201 }
202 milliseconds = fraction;
203 }
204 if (separator(cursor, 'Z')) {
205 ++cursor;
206 }
207 if (cursor != text.size()) {
208 return std::nullopt;
209 }
210
211 auto const calendar =
212 std::chrono::year_month_day{std::chrono::year{*year}, std::chrono::month{static_cast<unsigned>(*month)},
213 std::chrono::day{static_cast<unsigned>(*day)}};
214 if (!calendar.ok() || *hour > 23 || *minute > 59 || *second > 59) {
215 return std::nullopt;
216 }
217 return DateTime{std::chrono::year{*year},
218 std::chrono::month{static_cast<unsigned>(*month)},
219 std::chrono::day{static_cast<unsigned>(*day)},
220 std::chrono::hours{*hour},
221 std::chrono::minutes{*minute},
222 std::chrono::seconds{*second},
223 std::chrono::milliseconds{milliseconds}};
224 }
225 // NOLINTEND(cppcoreguidelines-pro-bounds-pointer-arithmetic, cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
226
227 /// @brief Ordering on the underlying instant.
228 /// @param other Instant to compare against.
229 /// @return The relative ordering of the two instants.
230 [[nodiscard]] constexpr std::strong_ordering operator<=>(const DateTime& other) const noexcept = default;
231
232 /// @brief Shifts this instant by an arbitrary chrono duration.
233 /// @param delta Duration to add (may be negative).
234 /// @return `*this`.
235 template <typename Rep, typename Period>
236 constexpr DateTime& operator+=(std::chrono::duration<Rep, Period> delta) noexcept {
237 value += std::chrono::duration_cast<std::chrono::milliseconds>(delta);
238 return *this;
239 }
240
241 /// @brief Shifts this instant backwards by an arbitrary chrono duration.
242 /// @param delta Duration to subtract (may be negative).
243 /// @return `*this`.
244 template <typename Rep, typename Period>
245 constexpr DateTime& operator-=(std::chrono::duration<Rep, Period> delta) noexcept {
246 value -= std::chrono::duration_cast<std::chrono::milliseconds>(delta);
247 return *this;
248 }
249
250 /// @brief An instant @p delta after @p lhs.
251 /// @param lhs Starting instant.
252 /// @param delta Duration offset.
253 /// @return The shifted instant.
254 template <typename Rep, typename Period>
255 [[nodiscard]] friend constexpr DateTime operator+(DateTime lhs,
256 std::chrono::duration<Rep, Period> delta) noexcept {
257 lhs += delta;
258 return lhs;
259 }
260
261 /// @brief An instant @p delta before @p lhs.
262 /// @param lhs Starting instant.
263 /// @param delta Duration offset.
264 /// @return The shifted instant.
265 template <typename Rep, typename Period>
266 [[nodiscard]] friend constexpr DateTime operator-(DateTime lhs,
267 std::chrono::duration<Rep, Period> delta) noexcept {
268 lhs -= delta;
269 return lhs;
270 }
271
272 /// @brief The signed difference between two instants.
273 /// @param lhs Minuend.
274 /// @param rhs Subtrahend.
275 /// @return `lhs - rhs` as a chrono duration.
276 [[nodiscard]] friend constexpr auto operator-(const DateTime& lhs, const DateTime& rhs) noexcept {
277 return lhs.value - rhs.value;
278 }
279};

296/// @brief Installs @p clock as the process-wide override for `DateTime::now()`
297/// (and therefore `Timestamp::now()`, which delegates to it).
298///
299/// Every call to `DateTime::now()` — including inside a registry-constructed
300/// (remotely instantiated) model that has no constructor parameter to receive
301/// an injected clock — consults this override before falling back to
302/// `std::chrono::system_clock::now()`. This is the seam
303/// `docs/spec/util/datetime.md` describes for deterministic
304/// testing of time-dependent model behavior: a test fixes "now" once via
305/// `ScopedNowOverride` and every `DateTime::now()`/`Timestamp::now()` call for
306/// its lifetime — direct or from inside a model under test — observes the
307/// fixed instant. Thread-safe.
308///
309/// @warning @p clock must not throw (`DateTime::now()` is `noexcept` and calls
310/// it directly — an exception escaping @p clock terminates the process) and
311/// must not call `DateTime::now()`/`Timestamp::now()`, `setNowOverride`, or
312/// construct a `ScopedNowOverride` itself: `now()` holds a non-recursive mutex
313/// while invoking @p clock, so any of those self-deadlocks.
314/// @param clock Callable returning the instant `DateTime::now()` should report,
315/// or an empty `std::function` to clear the override (restore
316/// real wall-clock time).
317inline void setNowOverride(std::function<DateTime()> clock) {
318 auto& [mtx, slot] = detail::nowOverrideState();
319 std::scoped_lock const lock{mtx};
320 slot = std::move(clock);
321}
322
323/// @brief RAII helper that fixes `DateTime::now()` to a constant instant (or a
324/// custom callable) for its lifetime and restores the previous
325/// override on destruction.
326///
327/// Mirrors `morph::journal::ScopedActionLog` and `morph::log::ScopedLoggerOverride`
328/// — the same scoped-install-then-restore pattern used for the codebase's other
329/// process-wide override seams, so one test's fixed "now" never leaks into the
330/// next.
331///
332/// @code
333/// {
334/// morph::time::ScopedNowOverride guard{someFixedInstant};
335/// // ... DateTime::now()/Timestamp::now(), anywhere, return someFixedInstant ...
336/// } // previous override (or real wall-clock time) restored here
337/// @endcode
338class ScopedNowOverride {
339public:
340 /// @brief Installs a clock that always returns @p fixedInstant.
341 /// @param fixedInstant The constant instant `DateTime::now()` should report for this scope.
342 explicit ScopedNowOverride(DateTime fixedInstant) : ScopedNowOverride{[fixedInstant] { return fixedInstant; }} {}
343
344 /// @brief Installs an arbitrary clock callable, saving whatever override was there before.
345 ///
346 /// @warning See `setNowOverride`'s warning: @p clock must not throw and must
347 /// not call `DateTime::now()`/`Timestamp::now()` or install another override.
348 /// @param clock Callable returning the instant `DateTime::now()` should report for this scope.
349 explicit ScopedNowOverride(std::function<DateTime()> clock) {
350 auto& [mtx, slot] = detail::nowOverrideState();
351 std::scoped_lock const lock{mtx};
352 _previous = std::move(slot);
353 slot = std::move(clock);
354 }
355
356 /// @brief Restores the saved override (or "no override", if none was installed before).
357 ~ScopedNowOverride() { setNowOverride(std::move(_previous)); }
358
359 ScopedNowOverride(const ScopedNowOverride&) = delete;
360 ScopedNowOverride& operator=(const ScopedNowOverride&) = delete;
361 ScopedNowOverride(ScopedNowOverride&&) = delete;
362 ScopedNowOverride& operator=(ScopedNowOverride&&) = delete;
363
364private:
365 std::function<DateTime()> _previous;
366};

368/// @brief An optionally-empty timestamp field.
369///
370/// The blank state ("not entered / not recorded") lives inside, exactly like
371/// `morph::units::Quantity`: action structs never wrap a `Timestamp` in
372/// `std::optional`, and a non-optional `Timestamp` member is *required* by
373/// the `morph::forms` rules.
374struct Timestamp {
375 /// @brief The payload; `std::nullopt` means "not entered".
376 std::optional<DateTime> value;
377
378 /// @brief Constructs the empty state.
379 constexpr Timestamp() noexcept = default;
380
381 /// @brief Engages with @p instant.
382 /// @param instant The instant to hold.
383 constexpr Timestamp(DateTime instant) noexcept : value{instant} {}
384
385 /// @brief Adopts an optional payload as-is.
386 /// @param payload Engaged or empty payload.
387 constexpr Timestamp(std::optional<DateTime> payload) noexcept : value{payload} {}
388
389 /// @brief An engaged timestamp holding the current UTC instant.
390 /// @return `Timestamp{DateTime::now()}`.
391 [[nodiscard]] static Timestamp now() noexcept { return Timestamp{DateTime::now()}; }
392
393 /// @brief Whether an instant has been entered.
394 /// @return `true` if the payload is engaged.
395 [[nodiscard]] constexpr bool hasValue() const noexcept { return value.has_value(); }
396
397 /// @brief Unchecked access to the engaged instant (UB when empty, exactly
398 /// like `std::optional` — the unchecked contract is the point).
399 /// @return The engaged instant.
400 // NOLINTNEXTLINE(bugprone-unchecked-optional-access)
401 [[nodiscard]] constexpr const DateTime& operator*() const noexcept { return *value; }
402
403 /// @brief Ordering/equality on the payload; empty sorts before engaged.
404 /// @param other Timestamp to compare against.
405 /// @return The ordering of the two payloads.
406 [[nodiscard]] constexpr auto operator<=>(const Timestamp& other) const noexcept = default;
407};
408
409/// @brief The signed duration between two engaged timestamps.
410/// @param lhs Minuend.
411/// @param rhs Subtrahend.
412/// @return `lhs - rhs` as a chrono duration, or `std::nullopt` if either is empty.
413[[nodiscard]] constexpr std::optional<std::chrono::milliseconds> operator-(const Timestamp& lhs,
414 const Timestamp& rhs) noexcept {
415 if (!lhs.value || !rhs.value) {
416 return std::nullopt;
417 }
418 return *lhs.value - *rhs.value;
419}

429/// @brief Reads a `DateTime` from a JSON ISO-8601 string; malformed content
430/// sets a syntax error on the context (it is not clampable input).
431template <>
432struct from<JSON, morph::time::DateTime> {
433 template <auto Opts>
434 static void op(auto&& value, is_context auto&& ctx, auto&& iter, auto end) {
435 std::string str{};
436 parse<JSON>::op<Opts>(str, ctx, iter, end);
437 if (bool(ctx.error)) [[unlikely]] {
438 return;
439 }
440 if (auto parsed = morph::time::DateTime::fromIso8601(str); parsed.has_value()) {
441 value = *parsed;
442 return;
443 }
444 ctx.error = error_code::syntax_error;
445 }
446};
