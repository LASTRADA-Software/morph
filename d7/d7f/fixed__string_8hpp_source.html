14/// @brief A structural, NTTP-capable compile-time string.
15///
16/// A literal type with public array storage, so it satisfies the rules for a
17/// non-type template parameter. Two `FixedString` values of the same length
18/// with the same characters compare equal as template arguments, so a type
19/// parameterised on a `FixedString` written identically in two translation
20/// units is one and the same type.
21///
22/// This is the single canonical definition shared by the forms layer
23/// (`morph::forms::FixedString`, used by `Choice`) and the units layer
24/// (`morph::units::detail::FixedString`, used by `NamedQuantity`), which alias
25/// it rather than redefining their own copies.
26///
27/// @tparam N Storage size including the terminating null.
28template <std::size_t N>
29struct FixedString {
30 /// @brief Character storage (null-terminated).
31 std::array<char, N> data{};
32
33 /// @brief Captures a string literal.
34 /// @param literal The literal to copy, e.g. `"ListSamples"`.
35 // NOLINTNEXTLINE(modernize-avoid-c-arrays, cppcoreguidelines-avoid-c-arrays, cppcoreguidelines-pro-bounds-constant-array-index, cppcoreguidelines-pro-bounds-avoid-unchecked-container-access) — string literals ARE C arrays.
36 consteval FixedString(const char (&literal)[N]) noexcept {
37 for (std::size_t index = 0; index < N; ++index) {
38 // NOLINTNEXTLINE(cppcoreguidelines-pro-bounds-constant-array-index, cppcoreguidelines-pro-bounds-avoid-unchecked-container-access)
39 data[index] = literal[index];
40 }
41 }
42
43 /// @brief A view of the string (without the terminating null).
44 /// @return The string contents.
45 [[nodiscard]] constexpr std::string_view view() const noexcept { return {data.data(), N - 1}; }
46};
47
