29/// @brief The window width class `{"env": "widthClass"}` reads (spec 3 §8).
30/// @param logicalWidth A width in logical pixels.
31/// @return `"compact"` below 600, `"medium"` below 1200, else `"expanded"`.
32[[nodiscard]] std::string widthClassOf(double logicalWidth);

34/// @brief What a `Backend` is made with beside its engine and host item.
35struct BackendOptions {
36 /// @brief The restyle map and the custom components; null renders every kind with its `MorphUi` component and
37 /// loads no custom component. Borrowed: it outlives the backend.
38 Components* components = nullptr;
39 /// @brief Called with the host item's width class (`widthClassOf`) once at construction and again each time
40 /// the class changes, not on every resize; empty calls nothing.
41 std::function<void(std::string const& widthClass)> onWidthClass;
42};

44/// @brief A widget the application's bundle or the client provides by name (spec 5 §10), in place of the built-in
45/// fallback a renderer without it shows.
46class CustomWidget : public ui::Widget {
47public:
48 /// @brief Sets a text prop: `string`, and `int`, `exact` and `json`, which cross as text so that no value loses
49 /// precision.
50 /// @param name The prop.
51 /// @param value Its text.
52 // NOLINTNEXTLINE(bugprone-easily-swappable-parameters): a prop's name, then its value, as every setter takes them.
53 virtual void setTextProp(std::string_view name, std::string_view value) = 0;
54
55 /// @brief Sets a `bool` prop.
56 /// @param name The prop.
57 /// @param value Its value.
58 virtual void setBoolProp(std::string_view name, bool value) = 0;
59
60 /// @brief Sets a `number` prop: the one kind of value that crosses as a double.
61 /// @param name The prop.
62 /// @param value Its value.
63 virtual void setNumberProp(std::string_view name, double value) = 0;
64
65 /// @brief Hands over the handler of the component's events.
66 /// @param onEvent Called with the event's name and its payload as text (a key as `i:<int64>` or `s:<text>`,
67 /// anything else as the component wrote it); empty does nothing.
68 virtual void setOnEvent(std::function<void(std::string event, std::string payload)> onEvent) = 0;
69};

71/// @brief What a `Backend::createCustom` call asks for.
72struct CustomRequest {
73 /// @brief The component's name, as the document's `components` declares it.
74 std::string name;
75 /// @brief The props the document declares for it; a component that does not declare each one is refused.
76 std::vector<std::string> props;
77 /// @brief The lowest version of a host component that will do; a bundle component ignores it.
78 int minVersion = 0;
79};

81/// @brief `ui::IViewBackend` over Qt Quick: each widget is a Qt Quick item made from its kind's `MorphUi`
82/// component, or from the component the restyle map names for that kind.
83///
84/// Every root a mount makes is appended to a column the backend adds to the host item, so several roots share it
85/// top to bottom. Neither copyable nor movable. It belongs to the thread of its engine, and every widget it makes
86/// is destroyed before it.
87class Backend final : public ui::IViewBackend {
88public:
89 /// @brief A backend whose roots fill @p host.
90 /// @param engine The engine its components are compiled in. Borrowed.
91 /// @param host The item its roots are shown in. Borrowed.
92 /// @param options The components and the width class callback.
93 Backend(QQmlEngine& engine MORPH_LIFETIMEBOUND, QQuickItem& host MORPH_LIFETIMEBOUND, BackendOptions options = {});
94 /// @brief Removes the roots' column from the host.
95 ~Backend() override;
96 Backend(Backend const&) = delete;
97 Backend& operator=(Backend const&) = delete;
98 Backend(Backend&&) = delete;
99 Backend& operator=(Backend&&) = delete;
100
101 /// @brief The Qt Quick item a widget of this backend shows.
102 /// @param widget The widget.
103 /// @return Its item; null for a widget this backend did not make.
104 [[nodiscard]] static QQuickItem* itemOf(ui::Widget const& widget) noexcept;
105
106 /// @brief Loads a custom component by name.
107 ///
108 /// A host component loads into this backend's engine; a bundle component must match its digest and compiles in
109 /// the components' contained engine. A component that cannot be loaded — not registered, a host component below
110 /// `minVersion`, a digest that does not match, a compile error, a declared prop it lacks — is reported once
111 /// through the components' failure callback, and the caller shows the fallback.
112 /// @param parent The container to append it to; null makes a root.
113 /// @param request The component and the props the document declares.
114 /// @return The widget, or null when the fallback is shown instead.
115 [[nodiscard]] std::unique_ptr<CustomWidget> createCustom(ui::ContainerWidget* parent,
116 CustomRequest const& request);
