29/// @brief The window width class `{"env": "widthClass"}` reads (spec 3 §8).
30/// @param logicalWidth A width in logical pixels.
31/// @return `"compact"` below 600, `"medium"` below 1200, else `"expanded"`.
32[[nodiscard]] std::string widthClassOf(double logicalWidth);

34/// @brief What a `Backend` is made with beside its engine and host item.
35struct BackendOptions {
36 /// @brief The restyle map and the custom components; null renders every kind with its `MorphUi` component and
37 /// loads no custom component. Borrowed: it outlives the backend.
38 Components* components = nullptr;
39 /// @brief Called with the host item's width class (`widthClassOf`) once at construction and again each time
40 /// the class changes, not on every resize; empty calls nothing.
41 std::function<void(std::string const& widthClass)> onWidthClass;
42};

48/// @brief `ui::IViewBackend` over Qt Quick: each widget is a Qt Quick item made from its kind's `MorphUi`
49/// component, or from the component the restyle map names for that kind.
50///
51/// Every root a mount makes is appended to a column the backend adds to the host item, so several roots share it
52/// top to bottom. Neither copyable nor movable. It belongs to the thread of its engine, and every widget it makes
53/// is destroyed before it.
54class Backend final : public ui::IViewBackend {
55public:
56 /// @brief A backend whose roots fill @p host.
57 /// @param engine The engine its components are compiled in. Borrowed.
58 /// @param host The item its roots are shown in. Borrowed.
59 /// @param options The components and the width class callback.
60 Backend(QQmlEngine& engine MORPH_LIFETIMEBOUND, QQuickItem& host MORPH_LIFETIMEBOUND, BackendOptions options = {});
61 /// @brief Removes the roots' column from the host.
62 ~Backend() override;
63 Backend(Backend const&) = delete;
64 Backend& operator=(Backend const&) = delete;
65 Backend(Backend&&) = delete;
66 Backend& operator=(Backend&&) = delete;

73 /// @brief Loads a custom component by name.
74 ///
75 /// A host component loads into this backend's engine; a bundle component must match its digest and compiles in
76 /// the components' contained engine. A component that cannot be loaded — not registered, a host component below
77 /// `minVersion`, a digest that does not match, a compile error, a declared prop it lacks — is reported once
78 /// through the components' failure callback, and the caller shows the fallback.
79 /// @param parent The container to append it to; null makes a root.
80 /// @param request The component and the props the document declares.
81 /// @return The widget, or null when the fallback is shown instead.
82 [[nodiscard]] std::unique_ptr<CustomWidget> createCustom(ui::ContainerWidget* parent,
