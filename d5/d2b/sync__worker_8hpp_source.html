18/// @brief Aggregated result returned by `SyncWorker::run()`.
19struct SyncResult {
20 /// @brief Number of items successfully replayed and removed from the queue.
21 int successful = 0;
22
23 /// @brief Number of items that failed and were left in the queue for retry.
24 int failed = 0;
25
26 /// @brief Number of items that exhausted their retry budget and were dropped
27 /// from the queue (handed to the `DeadLetterSink` if one is set,
28 /// otherwise logged via `morph::log::logError`).
29 int deadLettered = 0;
30};

32/// @brief Replays queued actions from an `IOfflineQueue` on reconnect.
33///
34/// `SyncWorker` drains the queue and calls a caller-supplied replay function
35/// for each item. The framework has no knowledge of what "replay" means —
36/// that is entirely the caller's domain. Retry semantics are baked in with
37/// hard-coded sensible defaults; there are intentionally no setters.
38///
39/// @par Retry & dead-letter (built-in defaults)
40/// - Each item is retried up to **5 cumulative attempts**. The count is seeded
41/// from the larger of the drained `QueueItem::attempts` (durable, if the
42/// queue persists it) and this worker's own in-memory count, so a queue that
43/// overrides `IOfflineQueue::setAttempts()` makes the budget survive a
44/// process restart; a queue that does not (the default no-op) keeps the
45/// count purely in-memory, exactly as before.
46/// - After every failed attempt, the new count is written back through
47/// `IOfflineQueue::setAttempts()` (a no-op unless the queue overrides it) so
48/// a persisting queue's next `drain()` — this run or after a restart — sees
49/// the updated value.
50/// - Items that fail their 5th cumulative attempt are dropped from the queue.
51/// If a `DeadLetterSink` is set, it is invoked with the exhausted item
52/// instead of the default log line; if unset, the item is logged at
53/// `morph::log::LogLevel::error` (the payload appears in the log line) —
54/// today's behavior, unchanged.
55/// - Items that succeed reset their attempt counter implicitly (they're removed).
56/// - There is no public knob on the retry cap — if your application needs
57/// different retry math, wrap or replace `SyncWorker`. The framework's
58/// promise is "obvious, safe defaults that the GUI never has to think about."
59///
60/// @par ReplayFunction contract
61/// - Return `true` → item successfully processed; it is removed from the queue.
62/// - Return `false` → item failed; attempt counter incremented; left in queue if
63/// under the cap, otherwise dropped (dead-lettered).
64/// - Throw → treated as failure (same path as returning `false`).
65///
66/// @par Thread safety
67/// `run()` is safe to call from any thread. Concurrent calls are serialised
68/// by an internal mutex — the second caller blocks until the first `run()` completes.
69class SyncWorker {
70public:
71 /// @brief Callable that attempts to replay a single queued item.
72 using ReplayFunction = std::function<bool(const std::string& payload)>;
73
74 /// @brief Callable invoked when an item exhausts its retry budget, just
75 /// before it is removed from the queue.
76 ///
77 /// Receives the exhausted item (payload, idempotencyKey, and its final
78 /// cumulative attempt count). If unset, `SyncWorker` keeps the default
79 /// behavior: log at `morph::log::LogLevel::error` and drop. A throwing
80 /// sink is caught and logged — the item is still removed.
81 using DeadLetterSink = std::function<void(const QueueItem& poisoned)>;
82
83 /// @brief Constructs a worker that drains @p queue using @p replay.
84 /// @param queue Queue to drain on each `run()` call.
85 /// @param replay Function called for each pending item.
86 /// @param deadLetterSink Optional hook invoked with the exhausted item
87 /// instead of the default log-and-drop path when an
88 /// item exhausts its retry budget. Default: unset.
89 SyncWorker(IOfflineQueue& queue, ReplayFunction replay, DeadLetterSink deadLetterSink = nullptr)
90 : _queue{queue}, _replay{std::move(replay)}, _deadLetterSink{std::move(deadLetterSink)} {}
91
92 /// @brief Drains the queue, replaying each item via the replay function.
93 ///
94 /// Concurrent calls are serialised. The call blocks until all pending items
95 /// have been processed or `stop()` is signalled.
96 ///
97 /// If `stop()` was called before `run()` acquired the lock, `run()` returns
98 /// immediately with an empty result and resets the stop flag.
99 ///
100 /// @return Counts of successful / failed / dead-lettered replays.
101 SyncResult run() {
102 std::scoped_lock const runLock{_runMtx};
103 bool const wasStoppedBeforeRun = _stopped.exchange(false);
104 SyncResult result;
105 if (wasStoppedBeforeRun) {
106 return result;
107 }
108
109 auto items = _queue.drain();
110 ::morph::observe::detail::emitMetric(::morph::observe::Metric::queueDepth, static_cast<double>(items.size()));
111 for (auto& item : items) {
112 if (_stopped.load()) {
113 break;
114 }
115 bool succeeded = false;
116 try {
117 succeeded = _replay(item.payload);
118 } catch (...) {
119 succeeded = false;
120 }
121 if (succeeded) {
122 _queue.markDone(item.id);
123 _attempts.erase(item.id);
124 ++result.successful;
125 continue;
126 }
127 // The effective count is the larger of the persisted item.attempts
128 // (0 unless the queue overrides setAttempts) and this worker's own
129 // in-memory count, so a durable queue's cross-restart count is
130 // never regressed by a fresh (empty) in-memory map.
131 auto& counter = _attempts[item.id];
132 counter = std::max(counter, item.attempts);
133 ++counter;
134 _queue.setAttempts(item.id, counter);
135 if (counter >= kMaxAttempts) {
136 if (_deadLetterSink) {
137 QueueItem poisoned = item;
138 poisoned.attempts = counter;
139 try {
140 _deadLetterSink(poisoned);
141 } catch (...) {
142 ::morph::log::logError("[sync_worker] dead-letter sink threw while handling payload: " +
143 item.payload);
144 }
145 } else {
146 ::morph::log::logError("[sync_worker] dropping payload after " + std::to_string(kMaxAttempts) +
147 " failed attempts: " + item.payload);
148 }
149 _queue.markDone(item.id);
150 _attempts.erase(item.id);
151 ++result.deadLettered;
152 } else {
153 ++result.failed;
154 }
155 }
156 return result;
157 }
158
159 /// @brief Signals an in-progress `run()` to stop after the current item.
160 ///
161 /// Thread-safe. The flag is automatically reset at the start of the next
162 /// `run()` call, so stopping is one-shot.
163 void stop() { _stopped.store(true); }
164
165private:
166 /// @brief Cap on per-item cumulative retry attempts. Intentionally
167 /// hard-coded — see class docs.
168 static constexpr uint32_t kMaxAttempts = 5;
169
170 IOfflineQueue& _queue;
171 ReplayFunction _replay;
172 DeadLetterSink _deadLetterSink;
173 std::mutex _runMtx;
174 std::atomic<bool> _stopped{false};
175 std::unordered_map<uint64_t, uint32_t> _attempts;
176};
