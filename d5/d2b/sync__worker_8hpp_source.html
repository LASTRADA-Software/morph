22/// @brief Replays queued actions from an `IOfflineQueue` on reconnect.
23///
24/// `SyncWorker` drains the queue and calls a caller-supplied replay function
25/// for each item. The framework has no knowledge of what "replay" means —
26/// that is entirely the caller's domain.
27///
28/// @par ReplayFunction contract
29/// - Return `true` → item successfully processed; it is removed from the queue.
30/// - Return `false` → item failed; it is left in the queue for the next `run()`.
31/// - Throw → treated as failure; exception is swallowed, item stays.
32///
33/// @par Thread safety
34/// `run()` is safe to call from any thread. Concurrent calls are serialised
35/// by an internal mutex — the second caller blocks until the first `run()` completes.
36class SyncWorker {
37public:
38 /// @brief Callable that attempts to replay a single queued item.
39 using ReplayFunction = std::function<bool(const std::string& payload)>;
40
41 /// @brief Constructs a worker that drains @p queue using @p replay.
42 /// @param queue Queue to drain on each `run()` call.
43 /// @param replay Function called for each pending item.
44 SyncWorker(IOfflineQueue& queue, ReplayFunction replay) : _queue{queue}, _replay{std::move(replay)} {}
45
46 /// @brief Drains the queue, replaying each item via the replay function.
47 ///
48 /// Concurrent calls are serialised. The call blocks until all pending items
49 /// have been processed or `stop()` is signalled.
50 ///
51 /// If `stop()` was called before `run()` acquired the lock, `run()` returns
52 /// immediately with an empty result and resets the stop flag.
53 ///
54 /// @return Counts of successful and failed replays.
55 SyncResult run() {
56 std::scoped_lock runLock{_runMtx};
57 bool wasStoppedBeforeRun = _stopped.exchange(false);
58 SyncResult result;
59 if (wasStoppedBeforeRun) {
60 return result;
61 }
62
63 for (auto& item : _queue.drain()) {
64 if (_stopped.load()) {
65 break;
66 }
67 bool succeeded = false;
68 try {
69 succeeded = _replay(item.payload);
70 } catch (...) {
71 succeeded = false;
72 }
73 if (succeeded) {
74 _queue.markDone(item.id);
75 ++result.successful;
76 } else {
77 ++result.failed;
78 }
79 }
80 return result;
81 }
82
83 /// @brief Signals an in-progress `run()` to stop after the current item.
84 ///
85 /// Thread-safe. The flag is automatically reset at the start of the next
86 /// `run()` call, so stopping is one-shot.
87 void stop() { _stopped.store(true); }
88
89private:
90 IOfflineQueue& _queue;
91 ReplayFunction _replay;
92 std::mutex _runMtx;
93 std::atomic<bool> _stopped{false};
94};
