16/// @brief How much a button stands out; each backend maps it to its theme.
17enum class ButtonEmphasis : std::uint8_t {
18 Normal, ///< An ordinary action.
19 Primary, ///< The screen's main action.
20 Danger, ///< An action that destroys or cannot be undone.
21 Ghost, ///< An action embedded in other content, with no chrome of its own.
22 /// @brief A destructive action that is not the main one: outlined, its label in the error colour, so the
23 /// screen's filled action stays the one that stands out.
24 DangerOutline,
25};
26
27/// @brief Every emphasis by its document name, `normal` first.
28inline constexpr std::array<std::pair<std::string_view, ButtonEmphasis>, 5> kButtonEmphases = {{
29 {"normal", ButtonEmphasis::Normal},
30 {"primary", ButtonEmphasis::Primary},
31 {"danger", ButtonEmphasis::Danger},
32 {"ghost", ButtonEmphasis::Ghost},
33 {"dangerOutline", ButtonEmphasis::DangerOutline},
34}};
35
36/// @brief An emphasis by its document name.
37/// @param name The name, such as `dangerOutline`.
38/// @return The emphasis; `nullopt` for a name that is none.
39[[nodiscard]] constexpr std::optional<ButtonEmphasis> emphasisNamed(std::string_view name) noexcept {
40 for (auto const& [known, emphasis] : kButtonEmphases) {
41 if (known == name) {
42 return emphasis;
43 }
44 }
45 return std::nullopt;
46}
47
48/// @brief An emphasis's document name.
49/// @param emphasis The emphasis.
50/// @return Its name, such as `dangerOutline`.
51[[nodiscard]] constexpr std::string_view emphasisName(ButtonEmphasis emphasis) noexcept {
52 for (auto const& [name, known] : kButtonEmphases) {
53 if (known == emphasis) {
54 return name;
55 }
56 }
57 return "normal";
58}
