91/// @brief Stores custom screens. The application's implementation is where every policy over them lives.
92///
93/// `list` answers the screens a person may open, which may include screens others saved. `put` and `remove` refuse
94/// what the person may not do, with a reason for the person. An id longer than `kMaxNameBytes` is shown to the person
95/// cut, so `get` and `remove` could not be given it back: a store keeps its ids within `kMaxNameBytes` (the model's
96/// own are 39 bytes). The model calls a store only with a person the server
97/// derived and with a document that passed every check. Every call carries the verified session too
98/// (`CustomScreenKey::session`, `CustomScreenScope::session`), so a policy may depend on more than the person, such as
99/// a role the session's token grants. A store owns the executor its completions are delivered on.
100///
101/// An entry `list` answers whose id lacks `kCustomScreenIdPrefix` is listed broken, and `get` and `remove` refuse its
102/// id before the store is called: the model cannot reach it, so cleaning it up is the store's.
103///
104/// The model's checks keep a person from saving a document that exceeds their permissions; they do not decide who
105/// may overwrite whose screen. That is the store's: see `put`.
106class ICustomScreenStore {
107public:
108 ICustomScreenStore() = default;
109 ICustomScreenStore(ICustomScreenStore const&) = delete;
110 ICustomScreenStore& operator=(ICustomScreenStore const&) = delete;
111 ICustomScreenStore(ICustomScreenStore&&) = delete;
112 ICustomScreenStore& operator=(ICustomScreenStore&&) = delete;
113 virtual ~ICustomScreenStore() = default;
114
115 /// @brief The custom screens a person may open in an application.
116 /// @param scope The person, the application and the verified session.
117 /// @return The screens, in the store's order.
118 [[nodiscard]] virtual async::Completion<std::vector<CustomScreen>> list(CustomScreenScope scope) = 0;
119
120 /// @brief One custom screen the person may open.
121 /// @param key The entry.
122 /// @return The screen, or the refusal.
123 [[nodiscard]] virtual async::Completion<std::expected<CustomScreen, CustomScreenRefusal>> get(
124 CustomScreenKey key) = 0;
125
126 /// @brief Stores a custom screen, replacing one of the same key.
127 ///
128 /// The key's id may name a screen this person can open but not change (a screen another person shared with them):
129 /// the store must refuse that write, since the model checks the document and not the right to replace what is
130 /// stored. `screen.savedBy` is the person the model derived, the caller, not the screen's first author.
131 /// @param key The entry.
132 /// @param screen The screen.
133 /// @return Nothing, or the refusal.
134 [[nodiscard]] virtual async::Completion<std::expected<void, CustomScreenRefusal>> put(CustomScreenKey key,
135 CustomScreen screen) = 0;
136
137 /// @brief Forgets a custom screen.
138 /// @param key The entry.
139 /// @return Nothing, or the refusal.
140 [[nodiscard]] virtual async::Completion<std::expected<void, CustomScreenRefusal>> remove(CustomScreenKey key) = 0;
141};

143/// @brief The framework's store: a map held in memory, on its own strand, each person seeing and changing only
144/// their own screens. It keys every entry by the person and reads nothing else of the session.
145///
146/// Its entries last as long as it does. Each call is a task on the strand, and its completion is delivered there. It
147/// bounds neither the number of entries nor their total size, so it serves tests and examples, not a deployment.
148class MemoryCustomScreenStore final : public ICustomScreenStore {
149public:
150 /// @brief A store whose strand runs on @p base.
151 /// @param base Where the strand's tasks run. Borrowed: it must outlive the store.
152 explicit MemoryCustomScreenStore(exec::IExecutor& base MORPH_LIFETIMEBOUND) : _strand{base} {}
153
154 /// @brief The custom screens the person saved in the application; the session is not read.
155 /// @param scope The person, the application and the verified session.
156 /// @return The screens, ordered by id.
157 [[nodiscard]] async::Completion<std::vector<CustomScreen>> list(CustomScreenScope scope) override {
158 auto [completion, promise] = async::Completion<std::vector<CustomScreen>>::makeSettleable(&_strand);
159 _strand.postTask(
160 [this, who = std::move(scope.person), where = std::move(scope.app), settle = std::move(promise)] mutable {
161 std::vector<CustomScreen> out;
162 for (auto const& [key, screen] : _entries) {
163 if (std::get<0>(key) == who && std::get<1>(key) == where) {
164 out.push_back(screen);
165 }
166 }
167 settle.resolve(std::move(out));
168 });
169 return std::move(completion);
170 }
171
172 /// @brief One of the person's own screens.
173 /// @param key The entry.
174 /// @return The screen, or `no such custom screen`.
175 [[nodiscard]] async::Completion<std::expected<CustomScreen, CustomScreenRefusal>> get(
176 CustomScreenKey key) override {
177 auto [completion, promise] =
178 async::Completion<std::expected<CustomScreen, CustomScreenRefusal>>::makeSettleable(&_strand);
179 _strand.postTask([this, entry = tie(key), settle = std::move(promise)] mutable {
180 auto const found = _entries.find(entry);
181 if (found == _entries.end()) {
182 settle.resolve(std::unexpected(CustomScreenRefusal{.reason = "no such custom screen"}));
183 return;
184 }
185 settle.resolve(found->second);
186 });
187 return std::move(completion);
188 }
189
190 /// @brief Stores a screen under the person's key.
191 /// @param key The entry.
192 /// @param screen The screen.
193 /// @return Nothing: the memory store refuses no write.
194 [[nodiscard]] async::Completion<std::expected<void, CustomScreenRefusal>> put(CustomScreenKey key,
195 CustomScreen screen) override {
196 auto [completion, promise] =
197 async::Completion<std::expected<void, CustomScreenRefusal>>::makeSettleable(&_strand);
198 _strand.postTask([this, entry = tie(key), kept = std::move(screen), settle = std::move(promise)] mutable {
199 _entries.insert_or_assign(std::move(entry), std::move(kept));
200 settle.resolve(std::expected<void, CustomScreenRefusal>{});
201 });
202 return std::move(completion);
203 }
204
205 /// @brief Forgets one of the person's own screens.
206 /// @param key The entry.
207 /// @return Nothing, or `no such custom screen`.
208 [[nodiscard]] async::Completion<std::expected<void, CustomScreenRefusal>> remove(CustomScreenKey key) override {
209 auto [completion, promise] =
210 async::Completion<std::expected<void, CustomScreenRefusal>>::makeSettleable(&_strand);
211 _strand.postTask([this, entry = tie(key), settle = std::move(promise)] mutable {
212 if (_entries.erase(entry) == 0) {
213 settle.resolve(std::unexpected(CustomScreenRefusal{.reason = "no such custom screen"}));
214 return;
215 }
216 settle.resolve(std::expected<void, CustomScreenRefusal>{});
217 });
218 return std::move(completion);
219 }
220
221private:
222 using Key = std::tuple<std::string, std::string, std::string>;
223
224 [[nodiscard]] static Key tie(CustomScreenKey const& key) { return Key{key.person, key.app, key.id}; }
225
226 std::map<Key, CustomScreen> _entries;
227 /// After the map, so it is destroyed first: a task running on it reads the map.
228 exec::OwnerStrand _strand;
229};

231/// @brief Checks that @p custom uses nothing its base screen does not: the base rule.
232///
233/// Compared: the actions it names, wherever declared and component fallbacks included, each on its model type and in
234/// a position of the same kind (a query, a mutation, a model-backed table, a collection, a wizard, or a form by its
235/// literal `submit`: none, `explicit` or `auto`); the commands run without a press (from a lifecycle list, a watch,
236/// a component's `on` event, a form whose `submit` is not `explicit`, or the `onSuccess` and `onError` of a mutation
237/// that a `run`, an `upload` or a `signIn` among them reaches, each name resolved in the scope that declares the
238/// mutation), by trigger. A form with no `submit` keeps its action schema's mode, which this check cannot see and
239/// which is automatic unless the schema says otherwise, so it counts as submitting itself.
240///
241/// Also compared: the client-side commands `upload`, `download`, `switchBackend`, `signIn`, `signOut` and
242/// `resetPrefs`; every `host` name, whether a host command, a host value or a host component; each `host` command
243/// as a whole, name and arguments, which must be identical to one of the base's; the delivered components, by name
244/// and content hash; and the params, which must be the base's exactly. Every `upload` of the custom document must
245/// send exactly `{"ref": "event"}` and sit directly in the `onPick` of a `filePicker` that opens files (no `mode`, or
246/// `open`) or in a `dropZone`'s `onDrop`, so it is always a file the person just chose. A host command's arguments
247/// must be literals. Text, layout, styles, states and expressions are not compared.
248/// @param base The base screen, loaded.
249/// @param baseText The base screen's JSON text.
250/// @param custom The custom document, loaded.
251/// @param customText The text @p custom was loaded from, which the `host` names are read from.
252/// @return The first use the base does not have, with its JSON Pointer, or nothing.
253/// @throws std::logic_error When a document names a model alias it does not declare, which only a document that
254/// skipped load-time validation can.
255[[nodiscard]] MORPH_API std::optional<Refusal> checkBaseRule(Document const& base, std::string_view baseText,
256 Document const& custom, std::string_view customText);

300/// @brief The longest `broken.reason` of a summary, in bytes; a longer one is cut and ends with an ellipsis.
301inline constexpr std::size_t kMaxBrokenReasonBytes = 1024;
302
303/// @brief The longest `attributes` record, in bytes, as JSON. A `save` with a longer one is refused, and a stored
304/// screen whose record is longer is listed broken without it.
305inline constexpr std::size_t kMaxAttributesBytes = 1024;
306
307/// @brief The most bytes one summary takes before escaping. Every field is capped when the summary is built: `id`,
308/// `title`, `basedOn`, `savedBy` and `broken.pointer` at `kMaxNameBytes`, `broken.reason` at
309/// `kMaxBrokenReasonBytes`, a cut one ending with an ellipsis, and `attributes` at `kMaxAttributesBytes`;
310/// `hash` is 71 bytes (`sha256:` and 64 digits). That is 5 * 256 + 1024 + 1024 + 71 = 3399 bytes of text;
311/// the keys, the punctuation and `savedAt` take under 300 more, and 4 KiB leaves a margin.
312inline constexpr std::size_t kMaxSummaryBytes = std::size_t{4} * 1024;
313
314/// @brief The most a byte of a summary grows to in an envelope: the reply is written with control bytes escaped, so
315/// one becomes a six-byte escape sequence, and the envelope then doubles each backslash of the reply, making
316/// seven.
317inline constexpr std::size_t kMaxEscapedGrowth = 7;
318
319/// @brief The bytes of a `get` reply that are not its document: one capped summary, at most
320/// `kMaxEscapedGrowth * kMaxSummaryBytes` = 28 KiB once escaped, with a margin to 64 KiB.
321inline constexpr std::size_t kCustomScreenReplyOverhead = std::size_t{64} * 1024;
322
323/// @brief The custom screens' bounds.
324struct CustomScreensOptions {
325 /// @brief The longest custom document, in bytes. At most `kMaxFileBytes`, and small enough that a `get` reply
326 /// fits `wire::kMaxEnvelopeBytes`: the document is escaped once into the reply's JSON and the reply is
327 /// escaped again as the envelope's body, so a document of quotes and backslashes grows four-fold, and
328 /// `4 * maxDocumentBytes + kCustomScreenReplyOverhead` must not exceed the envelope bound.
329 std::size_t maxDocumentBytes = std::size_t{256} * 1024;
330 /// @brief The most verdicts `VerdictCache` keeps.
331 std::size_t verdictCacheSize = 1024;
332 /// @brief The most summaries `list` answers; it says `truncated` when the store held more. A capped summary is at
333 /// most `kMaxSummaryBytes` and grows at most `kMaxEscapedGrowth`-fold in an envelope, so
334 /// `maxListed * kMaxEscapedGrowth * kMaxSummaryBytes + kCustomScreenReplyOverhead` must not exceed
335 /// `wire::kMaxEnvelopeBytes`: at most 290. The default of 200 takes at most 5.8 MB.
336 std::size_t maxListed = 200;
337};

339/// @brief The checks' verdicts by `verdictKey`, bounded, dropping the least recently used.
340///
341/// A list of custom screens runs every stored document through the checks, and every save refetches every list, so

350 VerdictCache& operator=(VerdictCache const&) = delete;
351 VerdictCache(VerdictCache&&) = delete;
352 VerdictCache& operator=(VerdictCache&&) = delete;
353 ~VerdictCache() = default;
354
355 /// @brief The verdict kept under @p key, which becomes the most recently used.
356 /// @param key The key.

359

364
365 /// @brief How many verdicts are kept.
366 /// @return The count.
