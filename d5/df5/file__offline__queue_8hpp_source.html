40/// @brief One NDJSON line of `FileOfflineQueue`'s on-disk log.
41///
42/// `op == "put"` upserts `id`'s current `(payload, idempotencyKey, attempts)`;
43/// `op == "done"` tombstones `id`. Replaying every line in file order and
44/// applying each in turn (a later "put" overwrites an earlier one for the same
45/// `id`; a "done" removes it) reconstructs the live item set — the
46/// last-write-wins-per-id compaction `docs/spec/offline/offline.md` describes
47/// for this variant.
48struct FileQueueRecord {
49 std::string op;
50 uint64_t id{};
51 std::string payload;
52 std::string idempotencyKey;
53 uint32_t attempts{0};
54};

56inline void throwOnGlazeError(const glz::error_ctx& errCode, std::string_view context) {
57 if (errCode) {
58 throw FileOfflineQueueError{glz::format_error(errCode, context)};
59 }
60}

62/// @brief Write options that escape ASCII control bytes as `\\uXXXX` sequences.
63///
64/// glaze 7.4 leaves control bytes (0x00-0x1F) unescaped by default, which
65/// breaks a `FileQueueRecord` carrying one in `payload`/`idempotencyKey` two
66/// ways: RFC 8259 requires those bytes escaped, so the raw byte alone yields
67/// JSON `fromJson`'s `glz::read` throws on for any non-trailing line (a
68/// permanent `FileOfflineQueueError` on every later `open()`, per this file's
69/// torn-trailing-line tolerance); worse, once the same string also contains an
70/// escaped `\` or `"`, glaze's chunked writer path silently rewrites the
71/// control byte as two 0x00 bytes, corrupting the payload before it ever
72/// reaches disk. Mirrors `morph::wire::detail::EscapingWriteOpts` (`core/wire.hpp`)
73/// exactly; duplicated here (rather than shared) so this header stays free of
74/// a `core/` dependency. Escaping is lossless, so any such byte still
75/// round-trips through `fromJson` unchanged.
76struct EscapingWriteOpts : glz::opts {
77 /// @brief Emit control bytes as `\\uXXXX` rather than raw.
78 // NOLINTNEXTLINE(readability-identifier-naming) — glaze's option name, matched by name.
79 bool escape_control_characters = true;
80};
81
82/// @brief Encodes @p record as JSON, escaping control bytes in `payload`/`idempotencyKey`.
83inline std::string toJson(const FileQueueRecord& record) {
84 std::string out;
85 throwOnGlazeError(glz::write<EscapingWriteOpts{}>(record, out), out);
86 return out;
87}
88
89inline FileQueueRecord fromJson(std::string_view json) {
90 FileQueueRecord record{};
91 // null_terminated=false: json is a caller-supplied view with no guaranteed
92 // trailing '\0' — see the identical fix + rationale on morph::wire::decode
93 // (wire.hpp), whose fuzz harness found the resulting heap-buffer-overflow in
94 // glaze's skip_ws. glz::read_json (used elsewhere in this file) hardcodes
95 // glz::opts{} and offers no way to override this, hence the explicit glz::read<>.
96 static constexpr glz::opts kUnpadded{.null_terminated = false};
97 throwOnGlazeError(glz::read<kUnpadded>(record, json), json);
98 return record;
99}

103/// @brief Reference append-only, NDJSON-backed `IOfflineQueue` that persists
104/// `payload`, `idempotencyKey`, and `attempts` across process restarts
105/// with no third-party dependency.
106///
107/// Each mutation (`enqueue`, `markDone`, `setAttempts`, `setIdempotencyKey`)
108/// appends one JSON line and immediately `fflush`+`fsync`s it, so it is a
109/// committed transaction before the call returns. On construction, the file
110/// is replayed line by line (last-write-wins per id — a later "put" overwrites
111/// an earlier one, a "done" tombstones it) and then rewritten in compacted
112/// form (one "put" line per surviving item), which both bounds file growth and
113/// heals a torn trailing line left by a crash mid-write — the same tolerance
114/// `FileActionLog::entries()` (`include/morph/journal/file_action_log.hpp`)
115/// gives a malformed *trailing* line: it is logged and skipped, not thrown: a
116/// malformed line anywhere else is genuine corruption and is rethrown.
117///
118/// New ids resume from the highest id ever seen in the file (including
119/// tombstoned ones), so a fresh item can never collide with an old tombstone
120/// — unlike `FileActionLog`'s process-local `seq`, which does not need this
121/// property because it never reuses/removes entries.
122///
123/// @par Thread safety
124/// All public methods are thread-safe (guarded by an internal mutex). Not
125/// safe for multiple processes to open the same path concurrently — same
126/// restriction as `FileActionLog`.
127///
128/// @par Idempotency-key dedup
129/// A keyed `enqueue` does a linear scan over the currently-pending items to
130/// look for a matching `idempotencyKey` — O(pending items) per call. That is
131/// fine at the queue depths this reference implementation targets; a host
132/// with high-volume keyed enqueues should prefer `SqliteOfflineQueue`, whose
133/// dedup is index-backed.
134class FileOfflineQueue : public IOfflineQueue {
135public:
136 using IOfflineQueue::enqueue; // keep the two-arg overload visible
137
138 /// @brief Opens (or creates) the queue log at @p path, replaying and
139 /// compacting whatever is already there.
140 /// @param path NDJSON file to store queue state in.
141 /// @param ioOps Injectable file-I/O primitives; defaults to the real
142 /// syscalls. Test-only seam — see `morph::core::FileIoOps`'s own
143 /// docs — for forcing the failure branches that need a real
144 /// OS-level I/O error to reach.
145 /// @param maxDepth Maximum number of pending items `enqueue()` will admit
146 /// before throwing `OfflineQueueFullError`; `std::nullopt` (the
147 /// default) means unbounded. Not persisted in the file itself — a
148 /// per-construction parameter, so a reopen must pass it again to
149 /// keep the same cap enforced.
150 /// @throws FileOfflineQueueError if @p path exists but contains a
151 /// malformed non-trailing line.
152 /// @throws std::runtime_error if @p path cannot be opened/rewritten.
153 explicit FileOfflineQueue(std::filesystem::path path, ::morph::core::FileIoOps ioOps = {},
154 std::optional<std::size_t> maxDepth = std::nullopt)
155 : _path{std::move(path)}, _io{std::move(ioOps)}, _maxDepth{maxDepth} {
156 load();
157 compact();
158 _file = _io.fopen(_path.string(), "a");
159 if (_file == nullptr) {
160 throw std::runtime_error("FileOfflineQueue: failed to open " + _path.string());
161 }
162 }
163
164 /// @brief Closes the underlying file.
165 // NOLINTNEXTLINE(cert-err33-c) — destructor context, can't propagate errors
166 ~FileOfflineQueue() override {
167 if (_file != nullptr) {
168 std::fclose(_file);
169 }
170 }

177 /// @brief Appends @p payload with no idempotency key.
178 /// @param payload Serialised action to persist.
179 /// @return A stable id that can be passed to `markDone()`.
180 uint64_t enqueue(std::string payload) override { return enqueue(std::move(payload), {}); }
181
182 /// @brief Appends @p payload carrying @p idempotencyKey. A non-empty key
183 /// already present on a pending item is deduplicated: the existing
184 /// item's id is returned and nothing new is written.
185 /// @param payload Serialised action to persist.
186 /// @param idempotencyKey Stable dedup token; empty means "no dedup".
187 /// @return The new item's id, or the existing item's id on a dedup hit.
188 /// @throws OfflineQueueFullError if the queue is already at `maxDepth()`
189 /// (a dedup hit above bypasses this check and always succeeds).
190 uint64_t enqueue(std::string payload, std::string idempotencyKey) override {
191 std::scoped_lock const lock{_mtx};
192 if (!idempotencyKey.empty()) {
193 for (const auto& [existingId, item] : _items) {
194 if (item.idempotencyKey == idempotencyKey) {
195 return existingId;
196 }
197 }
198 }
199 if (_maxDepth && _items.size() >= *_maxDepth) {
200 ::morph::observe::detail::emitMetric(::morph::observe::Metric::queueOverflow,
201 static_cast<double>(_items.size()));
202 throw OfflineQueueFullError{*_maxDepth, _items.size()};
203 }
204 uint64_t const itemId = ++_nextId;
205 QueueItem item{.id = itemId, .payload = std::move(payload), .idempotencyKey = std::move(idempotencyKey)};
206 appendPut(item);
207 _items.emplace(itemId, std::move(item));
208 return itemId;
209 }
210
211 /// @brief Returns all pending items in ascending-id (enqueue) order.
212 /// @return Snapshot of all pending items; the file itself is unchanged.
213 std::vector<QueueItem> drain() const override {
214 std::scoped_lock const lock{_mtx};
215 std::vector<QueueItem> out;
216 out.reserve(_items.size());
217 for (const auto& [id, item] : _items) {
218 out.push_back(item);
219 }
220 return out;
221 }
222
223 /// @brief Returns the number of pending items. Thread-safe.
224 /// @return Current pending item count.
225 std::size_t size() const override {
226 std::scoped_lock const lock{_mtx};
227 return _items.size();
228 }
229
230 /// @brief Returns the configured maximum depth, or `std::nullopt` if unbounded.
231 /// @return The capacity `enqueue()` enforces, or `std::nullopt` if none.
232 std::optional<std::size_t> maxDepth() const override { return _maxDepth; }
233
234 /// @brief Tombstones @p itemId. No-op if not found.
235 /// @param itemId Id returned by the corresponding `enqueue()` call.
236 void markDone(uint64_t itemId) override {
237 std::scoped_lock const lock{_mtx};
238 if (_items.erase(itemId) == 0) {
239 return;
240 }
241 appendDone(itemId);
242 }
243
244 /// @brief Persists an updated attempt count for @p itemId. No-op if not found.
245 /// @param itemId Id of the item whose count changed.
246 /// @param attempts New cumulative attempt count to store.
247 void setAttempts(uint64_t itemId, uint32_t attempts) override {
248 std::scoped_lock const lock{_mtx};
249 auto iter = _items.find(itemId);
250 if (iter == _items.end()) {
251 return;
252 }
253 iter->second.attempts = attempts;
254 appendPut(iter->second);
255 }
256
257protected:
258 /// @brief Stamps an idempotency key onto an already-enqueued item. No-op
259 /// if @p itemId is not found. Reachable only if a caller invokes
260 /// the base `IOfflineQueue::enqueue(payload, key)` default through
261 /// an `IOfflineQueue&` — this class's own `enqueue(payload, key)`
262 /// override above stamps the key inline instead.
263 /// @param itemId Id of the item to stamp.
264 /// @param idempotencyKey Key to store.
265 void setIdempotencyKey(uint64_t itemId, std::string idempotencyKey) override {
266 std::scoped_lock const lock{_mtx};
267 auto iter = _items.find(itemId);
268 if (iter == _items.end()) {
269 return;
270 }
271 iter->second.idempotencyKey = std::move(idempotencyKey);
272 appendPut(iter->second);
273 }

275private:
276 void appendPut(const QueueItem& item) {
277 detail::FileQueueRecord const record{.op = "put",
278 .id = item.id,
279 .payload = item.payload,
280 .idempotencyKey = item.idempotencyKey,
281 .attempts = item.attempts};
282 writeLine(detail::toJson(record));
283 }
284
285 void appendDone(uint64_t itemId) {
286 detail::FileQueueRecord const record{
287 .op = "done", .id = itemId, .payload = {}, .idempotencyKey = {}, .attempts = 0};
288 writeLine(detail::toJson(record));
289 }

291 /// Every mutation is documented as a committed transaction by the time the
292 /// call returns, so a failure to get the bytes down has to be raised rather
293 /// than swallowed: a caller told an item was enqueued, or marked done, must
294 /// not have that silently be untrue after a restart.
295 void writeLine(const std::string& json) {
296 std::string line = json;
297 line.push_back('\n');
298 if (_io.fwrite(line.data(), line.size(), _file) != line.size()) {
299 throw std::runtime_error("FileOfflineQueue: short write to " + _path.string());
300 }
301 syncFile(_file, _path.string());
302 }

304 void syncFile(std::FILE* file, const std::string& what) const {
305 if (_io.fflush(file) != 0) {
306 throw std::runtime_error("FileOfflineQueue: failed to flush " + what);
307 }
308 if (_io.fsync(file) != 0) {
309 throw std::runtime_error("FileOfflineQueue: failed to fsync " + what);
310 }
311 }

313 /// @brief Reads whatever is on disk and replays it into `_items`/`_nextId`.
314 void load() {
315 if (!std::filesystem::exists(_path)) {
316 return;
317 }
318 std::ifstream in{_path};
319 std::vector<std::string> lines;
320 std::string line;
321 while (std::getline(in, line)) {
322 if (!line.empty()) {
323 lines.push_back(line);
324 }
325 }
326 uint64_t highestId = 0;
327 for (std::size_t i = 0; i < lines.size(); ++i) {
328 detail::FileQueueRecord record;
329 try {
330 record = detail::fromJson(lines[i]);
331 } catch (const std::exception& exc) {
332 if (i + 1 == lines.size()) {
333 ::morph::log::logWarn("FileOfflineQueue: skipping malformed trailing line in " + _path.string() +
334 ": " + std::string{exc.what()});
335 break;
336 }
337 throw;
338 }
339 highestId = std::max(highestId, record.id);
340 if (record.op == "done") {
341 _items.erase(record.id);
342 } else {
343 _items[record.id] = QueueItem{.id = record.id,
344 .payload = record.payload,
345 .idempotencyKey = record.idempotencyKey,
346 .attempts = record.attempts};
347 }
348 }
349 _nextId = highestId;
350 }
351
352 /// @brief Rewrites the file with exactly one "put" line per surviving
353 /// item, collapsing whatever history `load()` just replayed.
354 /// Called once from the constructor, after `load()` and before the
355 /// append-mode `_file` handle is opened for new writes.
356 void compact() {
357 std::string const tmp = _path.string() + ".compact-tmp";
358 std::FILE* out = _io.fopen(tmp, "w");
359 if (out == nullptr) {
360 throw std::runtime_error("FileOfflineQueue: failed to open " + tmp + " for compaction");
361 }
362 auto writeRecord = [&](const detail::FileQueueRecord& record) {
363 std::string outLine = detail::toJson(record);
364 outLine.push_back('\n');
365 if (_io.fwrite(outLine.data(), outLine.size(), out) != outLine.size()) {
366 // Closing on the way out of a throw; nothing to report a
367 // close failure to.
368 // NOLINTNEXTLINE(cert-err33-c, cppcoreguidelines-owning-memory)
369 std::fclose(out);
370 throw std::runtime_error("FileOfflineQueue: short write during compaction of " + _path.string());
371 }
372 };
373
374 for (const auto& [entryId, item] : _items) {
375 writeRecord(detail::FileQueueRecord{.op = "put",
376 .id = item.id,
377 .payload = item.payload,
378 .idempotencyKey = item.idempotencyKey,
379 .attempts = item.attempts});
380 }
381
382 // Carry the id high-water mark across the rewrite. `load()` derives
383 // _nextId from the ids it sees, and compaction drops every tombstone, so
384 // without this the mark silently regresses to the highest *surviving*
385 // id: enqueue 1 and 2, markDone(2), restart (compacts to just id 1),
386 // restart again -> _nextId == 1 and the next enqueue reissues id 2, the
387 // id of an item that was completed and acknowledged. That breaks the
388 // "new ids never collide with an old tombstone" invariant this class
389 // documents, and a stale in-flight reference to the old id 2 would then
390 // silently address a different item.
391 //
392 // Recorded as a "done" for the mark itself rather than a new record
393 // type: `load()` already raises highestId for every id it reads and
394 // erasing an id that is not present is a no-op, so this needs no reader
395 // change and stays readable by an older build. Emitted only when the
396 // mark exceeds every surviving id -- writing "done" for an id that a
397 // "put" line above just restored would delete it on the next load.
398 uint64_t const maxSurviving = _items.empty() ? 0 : _items.rbegin()->first;
399 if (_nextId > maxSurviving) {
400 writeRecord(detail::FileQueueRecord{
401 .op = "done", .id = _nextId, .payload = {}, .idempotencyKey = {}, .attempts = 0});
402 }
403
404 syncFile(out, tmp);
405 // NOLINTNEXTLINE(cert-err33-c, cppcoreguidelines-owning-memory) — the data is already fsynced above
406 std::fclose(out);
407 std::filesystem::rename(tmp, _path);
408 }

415 uint64_t _nextId{0};
416 std::optional<std::size_t> _maxDepth;
417};
