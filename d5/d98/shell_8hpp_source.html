30/// @brief What a sign-in installs: the principal, and the session token its server needs (empty for none).
31struct SignedIn {
32 /// @brief The principal.
33 std::string principal;
34 /// @brief The session token; empty when the server needs none, as in process.
35 std::string token;
36};

38/// @brief Maps a sign-in mutation's reply to the principal and token to install; `std::nullopt` refuses the sign-in.
39using SignInMapper = std::function<std::optional<SignedIn>(ui::Value const& reply)>;

41/// @brief The default sign-in mapping: a reply `{ok, principal, token}`, refused when `ok` is `false` or there is no
42/// principal; `token` may be absent.
43/// @param reply The sign-in mutation's reply.
44/// @return The principal and token, or `std::nullopt`.
45[[nodiscard]] std::optional<SignedIn> defaultSignIn(ui::Value const& reply);

47/// @brief The application's side of the commands a document host does not answer itself (spec 6): sign-in and
48/// sign-out, which install a principal into the source, and the file commands, which read a picked file from
49/// disk and save a downloaded one, moving the bytes over the source's file side channel or inline.
50///
51/// A shell of an application's own builds one and passes it as `DocumentScreenHostOptions::application`;
52/// `mountShell` builds its own.
53class ApplicationHost final : public ui::InterpreterHost {
54public:
55 /// @brief A host over @p source.
56 /// @param source The source sign-ins are installed into and files move through. Borrowed: it outlives the
57 /// host.
58 /// @param signIn Maps a sign-in reply to what is installed. Required.
59 /// @param downloads Where a `download` saves a file; the working directory when empty.
60 /// @param signedIn Called with what a sign-in installs, for the shell to install it and reload.
61 /// @param signedOut Called on a `signOut`.
62 /// @param onProblem Called with a refused sign-in's reason; may be empty.
63 /// @throws std::invalid_argument when @p signIn is empty.
64 ApplicationHost(Source& source, SignInMapper signIn, std::filesystem::path downloads,
65 std::function<void(SignedIn)> signedIn, std::function<void()> signedOut,
66 std::function<void(std::string const&)> onProblem = {});
67
68 /// @brief Maps the reply and hands a principal to `signedIn`.
69 /// @param reply The sign-in mutation's reply.
70 /// @return Whether it signed in: the mapper answered a principal.
71 bool signIn(ui::Value const& reply) override;
72
73 /// @brief Calls `signedOut`.
74 void signOut() override;
75
76 /// @brief Reads the picked file from disk and sends it through the source's file store.
77 /// @param request The file and the mutation it is for.
78 /// @param progress Called with the progress, as an exact fraction.
79 /// @param done Called with the file's reference, or the failure.
80 /// @return Whether it can: the source has a file side channel and the file is a path.
81 bool uploadFor(ui::UploadRequest const& request, std::function<void(ui::Value)> progress,

84 /// @brief Fetches a file by reference and saves it in the downloads directory. A preview is not supported.

113
114 /// @brief Calls the notify sink if one is set.
115 /// @param text The message.

117 /// @return Whether the sink showed it; `false` when none is set.
118 bool notify(std::string const& text, std::string const& role) override;

126 std::function<void(std::string const&)> _onProblem;

130/// @brief What a client shell reports, and how it starts.
131struct ShellOptions {
132 /// @brief The screen mounted first; empty for the first the source lists.
133 std::string startScreen;
134 /// @brief The start screen's params.
135 ui::ParamMap startParams;
136 /// @brief Called with each problem: why the source did not open, a document that did not load, a refused mount,
137 /// an interpreter diagnostic. Empty ignores them; the shell shows the first three either way.
138 std::function<void(std::string const& problem)> onProblem;
139 /// @brief Called with the id of each screen once it is mounted.
140 std::function<void(std::string const& screenId)> onMounted;
141 /// @brief Where a screen's `download` saves a file, under the name the screen suggests; the working directory
142 /// when empty.
143 std::filesystem::path downloads;
144 /// @brief The history the shown screen's route is kept in (spec 3 §10), such as the browser's: back and forward
145 /// open the route of the entry moved to, and the route it holds at start is opened once the session has
146 /// the screen. Borrowed: it outlives the shell; null keeps no history.
147 ui::IHistory* history = nullptr;
148 /// @brief Development mode (spec 5 §12): every second the shell polls the source's `ui-hello` digest, and when it
149 /// changed, remounts every open tab whose document changed, keeping its state. A shell built without
150 /// the development tools never polls. A production client leaves it off: the digest is then read once.
151 bool dev = false;
152 /// @brief Called after each development reload with the ids of the screens it remounted; empty ignores it.
153 std::function<void(std::vector<std::string> const& remounted)> onReloaded;
154 /// @brief Maps a sign-in mutation's reply to the principal and token the shell installs. Required.
155 SignInMapper signIn = defaultSignIn;
156};

158/// @brief Mounts the client's shell for what @p source delivered.
159///
160/// On a bundle: loads each screen of `source.screens()` with the source's catalog facts, opens the start screen as a
161/// tab, and on a document's `navigate` or a navigator choice opens the screen asked for as a tab (or focuses the tab
162/// that shows it), from a task posted to the owner, never from inside the handler that asked. A sign-in or sign-out
163/// closes every tab and rebuilds the frame for the new catalog. On a `ConnectError`, or when no screen can be mounted,
164/// shows why as text.
165/// @param context The frontend's context.
166/// @param backend The view backend the screens mount on. Borrowed: it outlives the shell.
167/// @param opened What the source delivered.
168/// @param source The source; it outlives the shell.
169/// @param options The start screen and the observers.
170/// @return The shell; destroying it unmounts everything.
171/// @throws std::invalid_argument when `options.signIn` is empty.
172[[nodiscard]] std::unique_ptr<ui::MountedShell> mountShell(ui::AppContext& context, ui::IViewBackend& backend,
173 std::expected<ui::Bundle, ui::ConnectError> opened,
174 Source& source, ShellOptions options);
