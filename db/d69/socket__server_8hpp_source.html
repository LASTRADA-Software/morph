33/// @brief Raw-socket WebSocket server front for `morph::backend::RemoteServer`.
34///
35/// Mirrors `morph::qt::QtWebSocketServer`'s behavior without Qt: for each
36/// accepted connection it performs the RFC 6455 handshake, then reads framed
37/// text messages and forwards each to `RemoteServer::handle()`; the reply is
38/// written back to the originating socket, or dropped if the connection is
39/// gone by the time the reply is ready.
40///
41/// @par Threading
42/// Owns one accept thread plus one thread per accepted connection — there is
43/// no shared event loop. `RemoteServer::handle()` replies arrive on the
44/// server's worker-pool thread and are marshalled back onto the owning
45/// connection's own write path, serialized by a per-connection mutex.
46///
47/// @par Lifetime
48/// Holds `RemoteServer& _server` by reference, exactly like
49/// `QtWebSocketServer` — the server's owning `shared_ptr` must outlive this
50/// object (see `docs/spec/core/backend.md`'s Lifetime & ownership section).
51class SocketServer {
52public:
53 /// @brief Alias for the config struct.
54 using Config = SocketServerConfig;
55
56 /// @brief Constructs the server; does not start listening.
57 /// @param server `RemoteServer` instance that processes incoming messages.
58 /// Borrowed, not owned: it must outlive this server (see
59 /// `docs/spec/concurrency_and_lifetimes.md`, "Destruction
60 /// ordering").
61 /// @param port TCP port to listen on. Pass 0 to let the OS pick a free port.
62 /// @param cfg Backlog tuning. Default: 64-connection backlog.
63 // Copy/move are implicitly deleted by the non-copyable/non-movable
64 // std::mutex/std::thread members below — no explicit `= delete` needed
65 // (matches the rest of the codebase's convention, e.g. `LocalBackend`).
66 SocketServer(::morph::backend::RemoteServer& server MORPH_LIFETIMEBOUND, std::uint16_t port, Config cfg = {})
67 : _server{server}, _requestedPort{port}, _cfg{cfg} {}

69 /// @brief Stops accepting and closes every client connection.
70 ~SocketServer() { close(); }

72 /// @brief Starts listening for incoming WebSocket connections.
73 ///
74 /// Fails closed: the accept loop's only way to stop waiting is the wakeup
75 /// pipe (see `acceptLoop()`), so if the pipe cannot be created — `pipe(2)`
76 /// answering `EMFILE`/`ENFILE` — this reports failure and spawns no thread
77 /// rather than starting a loop nothing could ever interrupt.
78 /// @return `true` if the server bound the requested port *and* armed its
79 /// wakeup pipe.
80 bool listen() {
81 try {
82 _listenSocket = ::morph::net::detail::TcpSocket::listen(_requestedPort, _cfg.backlog);
83 } catch (const std::exception&) {
84 return false;
85 }
86 // Non-blocking before the loop can ever reach ::accept -- see
87 // TcpSocket::tryAccept()'s doc comment for why a poll() readiness
88 // report is not a promise that accept() will not park.
89 if (!_listenSocket.setNonBlocking() || !_wakeup.open()) {
90 _listenSocket = ::morph::net::detail::TcpSocket{};
91 return false;
92 }
93 _closing.store(false);
94 _acceptThread = std::thread{[this] { acceptLoop(); }};
95 return true;
96 }

98 /// @brief Returns the port the server is currently bound to.
99 /// @return Bound TCP port (OS-assigned when constructed with port 0), or `0` before `listen()` succeeds.
100 [[nodiscard]] std::uint16_t port() const { return _listenSocket.boundPort(); }

102 /// @brief Stops accepting new connections and closes every client connection.
103 ///
104 /// Releases the listening socket once the accept thread has joined, so
105 /// `port()` reads `0` afterwards and the bound port is free to rebind —
106 /// the same post-close observation `QtWebSocketServer` makes.
107 ///
108 /// Idempotent: safe to call more than once (including implicitly, via the
109 /// destructor, after an explicit call), and safe to call from several
110 /// threads at once **on a live object** — concurrent callers are
111 /// serialized, and every one of them returns only once the teardown is
112 /// complete. Racing a call against the *destructor* is still the caller's
113 /// problem (see `docs/spec/concurrency_and_lifetimes.md`, "Destruction
114 /// ordering"): no lock inside this object can outlive the object holding
115 /// it.
116 void close() {
117 // Serialize the whole body, not just the guard below (morph#451).
118 // `_closing.exchange` alone cannot exclude a second caller: the loser
119 // still sees a *joinable* accept thread — the winner has not joined it
120 // yet and cannot have, since that thread is parked in poll() until the
121 // winner's wakeup byte releases it — so it falls through and calls
122 // join() on the same std::thread the winner is joining. That is
123 // UB, and it is not a theoretical one: on Linux/glibc the loser parks
124 // forever in pthread_join's futex on a descriptor the winner already
125 // reaped; on macOS/libc++ it throws std::system_error. Holding the
126 // mutex across the join makes the second caller wait for the first and
127 // then observe !joinable(), which is exactly the outcome the guard was
128 // written to produce.
129 //
130 // Deadlock-free because nothing inside this class calls close():
131 // neither acceptLoop() nor a per-client thread does, so no thread this
132 // function joins can be waiting on this mutex.
133 std::scoped_lock const teardownLock{_closeMtx};
134 // Kept as-is despite the mutex: acceptLoop() reads `_closing` from the
135 // accept thread, which never takes `_closeMtx`.
136 bool const wasAlreadyClosing = _closing.exchange(true);
137 if (wasAlreadyClosing && !_acceptThread.joinable()) {
138 return;
139 }
140 // The accept loop's wakeup, and the reason this teardown terminates at
141 // all (morph#437). It used to be `_listenSocket.shutdownBoth()`, which
142 // works only because Linux happens to kick a parked accept(2) when the
143 // listening socket is shut down -- macOS/BSD do not, so the join below
144 // never returned there. Nothing about that was arranged by this code.
145 // The loop now parks in poll() on this pipe as well as on the listener,
146 // so one byte here ends it on every platform.
147 //
148 // Runs at most once per listen()/close() cycle and never blocks: the
149 // guard above returns before reaching it on a second call, and the
150 // write end is non-blocking regardless.
151 _wakeup.signal();
152 if (_acceptThread.joinable()) {
153 _acceptThread.join();
154 }
155 // Only now that the accept thread is gone, so no fd number released
156 // here can be reused under a poll()/accept() still holding it.
157 // Dropping the listener is what keeps close()'s documented "stops
158 // accepting new connections" true without shutdownBoth(): left open,
159 // the kernel would keep completing handshakes into a backlog nobody
160 // drains, and a client would hang in the WebSocket Upgrade read
161 // instead of failing fast. `port()` therefore reads 0 after close(),
162 // which is what QtWebSocketServer already does.
163 _listenSocket = ::morph::net::detail::TcpSocket{};
164 _wakeup.reset();
165 std::vector<std::shared_ptr<ClientConnection>> clients;
166 std::vector<std::thread> threads;
167 {
168 std::scoped_lock lock{_clientsMtx};
169 clients = _clients;
170 threads.swap(_clientThreads);
171 _clients.clear();
172 }
173 for (auto& client : clients) {
174 // `closed` first, so no client thread starts a *new* send; then
175 // shutdown without taking writeMtx. Taking it here would deadlock
176 // exactly when the shutdown is most needed: a client thread blocked
177 // in sendAll against a stalled peer's full socket buffer holds
178 // writeMtx for as long as that send is stuck, and close() has no
179 // timeout -- it is reached from the destructor. shutdownBoth() is
180 // documented safe from any thread and is itself the mechanism that
181 // unblocks that send (it fails, sendText catches, the thread exits
182 // and releases the lock). Locking to "protect" the socket would
183 // therefore wait on the very thing it is trying to interrupt.
184 client->closed.store(true);
185 client->socket.shutdownBoth();
186 }
187 for (auto& t : threads) {
188 if (t.joinable()) {
189 t.join();
190 }
191 }
192 }

195 struct ClientConnection {
196 ClientConnection(::morph::net::detail::TcpSocket sock, ::morph::backend::ConnectionId connectionId)
197 : socket{std::move(sock)}, cid{connectionId} {}
198 ::morph::net::detail::TcpSocket socket;
199 /// Scope every `register` on this connection belongs to, so dropping

205 void sendText(const std::string& payload) {
206 std::scoped_lock lock{writeMtx};
207 if (closed.load() || !socket.valid()) {
208 return;
209 }
210 try {
211 std::string frame = ::morph::net::detail::encodeWsFrame(::morph::net::detail::WsOpcode::kText, payload,
212 /*mask=*/false);
213 socket.sendAll(frame.data(), frame.size());
214 } catch (const std::exception&) {
215 closed.store(true);
216 }
217 }

220 void acceptLoop() {
221 for (;;) {
222 std::array<pollfd, kPollFdCount> fds{};
223 pollfd& listenPfd = fds.front();
224 pollfd& wakeupPfd = fds.back();
225 listenPfd.fd = _listenSocket.nativeHandle();
226 listenPfd.events = POLLIN;
227 wakeupPfd.fd = _wakeup.readFd();
228 wakeupPfd.events = POLLIN;
229 // No timeout: the loop has an explicit wakeup now, so it has no
230 // reason to surface periodically and re-check anything.
231 if (::poll(fds.data(), kPollFdCount, -1) < 0) {
232 if (errno == EINTR) {
233 continue; // a delivered signal, not a failure -- as in TcpSocket::accept()
234 }
235 return; // poll() itself is broken; there is nothing left to wait on
236 }
237 if (wakeupPfd.revents != 0) {
238 return; // close() rang the wakeup pipe
239 }
240 if (listenPfd.revents == 0) {
241 continue;
242 }
243 std::optional<::morph::net::detail::TcpSocket> clientSocket;
244 try {
245 clientSocket = _listenSocket.tryAccept();
246 } catch (const std::exception&) {
247 return; // the listener is unusable (server closing)
248 }
249 if (!clientSocket) {
250 continue; // readiness went stale before the accept; wait again
251 }
252 if (_closing.load()) {
253 return;
254 }
255 auto conn = std::make_shared<ClientConnection>(std::move(*clientSocket), _server.openConnection());
256 std::thread clientThread{[this, conn] { clientLoop(conn); }};
257 {
258 std::scoped_lock lock{_clientsMtx};
259 _clients.push_back(conn);
260 _clientThreads.push_back(std::move(clientThread));
261 }
262 }
263 }

265 void clientLoop(const std::shared_ptr<ClientConnection>& conn) {
266 // Reclaim this connection's models however the loop exits — failed
267 // handshake, peer close, read error, or shutdown via close(). Without
268 // it every model registered over this transport outlived its connection
269 // forever: the scope machinery was wired into QtWebSocketServer only,
270 // and this server dispatched through the unscoped two-argument
271 // handle(). closeConnection() is idempotent, so the redundant call
272 // during close() (which joins these threads) is harmless.
273 struct ScopeGuard {
274 ScopeGuard(::morph::backend::RemoteServer& srv MORPH_LIFETIMEBOUND,
275 ::morph::backend::ConnectionId connectionId)
276 : server{srv}, cid{connectionId} {}
277 ~ScopeGuard() { server.closeConnection(cid); }
278 ScopeGuard(const ScopeGuard&) = delete;
279 ScopeGuard& operator=(const ScopeGuard&) = delete;
280 ScopeGuard(ScopeGuard&&) = delete;
281 ScopeGuard& operator=(ScopeGuard&&) = delete;
282
283 ::morph::backend::RemoteServer& server;
284 ::morph::backend::ConnectionId cid;
285 } const guard{_server, conn->cid};
286
287 std::string leftover;
288 try {
289 leftover = ::morph::net::detail::performServerHandshake(conn->socket);
290 } catch (const std::exception&) {
291 conn->closed.store(true);
292 return;
293 }
294 ::morph::net::detail::WsFrameReader reader;
295 reader.feed(leftover);
296 char buf[4096];
297 for (;;) {
298 if (!drainFrames(conn, reader)) {
299 return;
300 }
301 std::size_t got = 0;
302 try {
303 got = conn->socket.recvSome(buf, sizeof(buf));
304 } catch (const std::exception&) {
305 conn->closed.store(true);
306 return;
307 }
308 if (got == 0) {
309 conn->closed.store(true);
310 return;
311 }
312 reader.feed(std::string_view{buf, got});
313 }
314 }

316 /// Best-effort control-frame write (a Close echo or a Pong). Failure to
317 /// send one is never worth propagating: the connection is either already
318 /// going away or will be noticed as gone by the next read.
319 static void sendControlFrame(const std::shared_ptr<ClientConnection>& conn, ::morph::net::detail::WsOpcode opcode,
320 std::string_view payload) {
321 std::scoped_lock const lock{conn->writeMtx};
322 if (conn->closed.load() || !conn->socket.valid()) {
323 return;
324 }
325 try {
326 std::string const frame = ::morph::net::detail::encodeWsFrame(opcode, payload, /*mask=*/false);
327 conn->socket.sendAll(frame.data(), frame.size());
328 } catch (const std::exception&) { // NOLINT(bugprone-empty-catch) — see the doc comment above
329 }
330 }

332 // Returns false when the connection should stop reading (peer sent
333 // Close, or a protocol error was detected).
334 bool drainFrames(const std::shared_ptr<ClientConnection>& conn, ::morph::net::detail::WsFrameReader& reader) {
335 using ::morph::net::detail::WsOpcode;
336 for (;;) {
337 std::optional<::morph::net::detail::WsFrame> frame;
338 try {
339 frame = reader.tryExtractFrame();
340 } catch (const std::exception&) {
341 conn->closed.store(true);
342 return false;
343 }
344 if (!frame) {
345 return true;
346 }
347 if (frame->opcode == WsOpcode::kClose) {
348 sendControlFrame(conn, WsOpcode::kClose, "");
349 conn->closed.store(true);
350 return false;
351 }
352 if (frame->opcode == WsOpcode::kPing) {
353 sendControlFrame(conn, WsOpcode::kPong, frame->payload);
354 continue;
355 }
356 if (frame->opcode == WsOpcode::kPong) {
357 continue;
358 }
359 if (frame->opcode == WsOpcode::kText) {
360 std::weak_ptr<ClientConnection> weak = conn;
361 _server.handle(
362 frame->payload,
363 [weak](const std::string& reply) {
364 if (auto locked = weak.lock()) {
365 locked->sendText(reply);
366 }
367 },
368 conn->cid);
369 }
370 }
371 }

373 /// RAII owner of the self-pipe `acceptLoop()` polls alongside the listener
374 /// and `close()` writes one byte to (morph#437).
375 ///
376 /// A pipe rather than an `eventfd`: `eventfd` is Linux-only, and the whole
377 /// point of this mechanism is that it is the *same* mechanism on every
378 /// platform `morph::net` supports. A `#ifdef`-selected wakeup would be a
379 /// macOS-only code path that CI -- which is Linux-only -- could never
380 /// execute, guarding a bug CI could never observe.
381 class WakeupPipe {
382 public:
383 WakeupPipe() = default;
384 ~WakeupPipe() { reset(); }
385 WakeupPipe(const WakeupPipe&) = delete;
386 WakeupPipe& operator=(const WakeupPipe&) = delete;
387 WakeupPipe(WakeupPipe&&) = delete;
388 WakeupPipe& operator=(WakeupPipe&&) = delete;
389
390 /// Creates a fresh pipe, discarding any previous one -- and with it any
391 /// byte an earlier close() left undrained, which would otherwise make
392 /// the next accept loop exit the moment it started.
393 /// @return `true` if the pipe was created.
394 bool open() {
395 reset();
396 std::array<int, 2> fds{-1, -1};
397 if (::pipe(fds.data()) != 0) {
398 return false;
399 }
400 _readFd = fds.front();
401 _writeFd = fds.back();
402 configure(_readFd);
403 configure(_writeFd);
404 return true;
405 }
406
407 /// Closes both ends if open.
408 void reset() noexcept {
409 closeFd(_readFd);
410 closeFd(_writeFd);
411 }
412
413 /// @return The read end, for the accept loop's `poll()` set (`-1` when closed).
414 [[nodiscard]] int readFd() const noexcept { return _readFd; }
415
416 /// Makes the read end readable, waking a parked `poll()`.
417 ///
418 /// Best effort by construction, and never blocking: the pipe carries no
419 /// information beyond "readable", so a write that fails because the
420 /// buffer is already full has already achieved what it was for.
421 // NOLINTNEXTLINE(readability-make-member-function-const) — mutates the pipe this object owns
422 void signal() noexcept {
423 if (_writeFd < 0) {
424 return;
425 }
426 char const byte = 1;
427 ssize_t const written = ::write(_writeFd, &byte, 1);
428 static_cast<void>(written); // a bare (void) cast does not silence GCC's warn_unused_result
429 }
430
431 private:
432 // Return values deliberately unchecked: neither call can fail on a
433 // descriptor pipe() has just handed back, and a branch per fcntl would
434 // add arms to this header that no test could drive. ::fcntl is
435 // variadic by POSIX's design.
436 // NOLINTBEGIN(cppcoreguidelines-pro-type-vararg)
437 static void configure(int pipeFd) noexcept {
438 ::fcntl(pipeFd, F_SETFL, ::fcntl(pipeFd, F_GETFL, 0) | O_NONBLOCK);
439 ::fcntl(pipeFd, F_SETFD, FD_CLOEXEC);
440 }
441 // NOLINTEND(cppcoreguidelines-pro-type-vararg)
442
443 static void closeFd(int& pipeFd) noexcept {
444 if (pipeFd >= 0) {
445 ::close(pipeFd);
446 pipeFd = -1;
447 }
448 }
449
450 int _readFd{-1};
451 int _writeFd{-1};
452 };

454 /// Size of `acceptLoop()`'s poll set: the listening socket and the wakeup
455 /// pipe's read end. Typed `nfds_t` rather than converted at the call site
456 /// because neither spelling of the conversion is portable: `nfds_t` is
457 /// `unsigned long` on Linux, where an explicit cast trips GCC's
458 /// `-Wuseless-cast`, and `unsigned int` on macOS, where an implicit one
459 /// trips Clang's `-Wshorten-64-to-32`.
460 static constexpr nfds_t kPollFdCount = 2;
461
462 ::morph::backend::RemoteServer& _server;
463 std::uint16_t _requestedPort;
464 Config _cfg;
465 ::morph::net::detail::TcpSocket _listenSocket;
466 /// Written by `close()`, polled by `acceptLoop()`. Created by `listen()`
467 /// before the accept thread starts and released by `close()` after it has
468 /// joined, so the two never touch it concurrently.
469 WakeupPipe _wakeup;
470 /// Serializes `close()` against itself so only one caller ever reaches
471 /// `_acceptThread.join()` (morph#451). Not taken anywhere else.
472 std::mutex _closeMtx;
473 std::atomic<bool> _closing{true};
474 std::thread _acceptThread;
475 std::mutex _clientsMtx;
476 std::vector<std::shared_ptr<ClientConnection>> _clients;
477 std::vector<std::thread> _clientThreads;
478};
