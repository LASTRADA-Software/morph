26/// @brief Raw-socket WebSocket server front for `morph::backend::RemoteServer`.
27///
28/// Mirrors `morph::qt::QtWebSocketServer`'s behavior without Qt: for each
29/// accepted connection it performs the RFC 6455 handshake, then reads framed
30/// text messages and forwards each to `RemoteServer::handle()`; the reply is
31/// written back to the originating socket, or dropped if the connection is
32/// gone by the time the reply is ready.
33///
34/// @par Threading
35/// Owns one accept thread plus one thread per accepted connection — there is
36/// no shared event loop. `RemoteServer::handle()` replies arrive on the
37/// server's worker-pool thread and are marshalled back onto the owning
38/// connection's own write path, serialized by a per-connection mutex.
39///
40/// @par Lifetime
41/// Holds `RemoteServer& _server` by reference, exactly like
42/// `QtWebSocketServer` — the server's owning `shared_ptr` must outlive this
43/// object (see `docs/spec/core/backend.md`'s Lifetime & ownership section).
44class SocketServer {
45public:
46 /// @brief Alias for the config struct.
47 using Config = SocketServerConfig;
48
49 /// @brief Constructs the server; does not start listening.
50 /// @param server `RemoteServer` instance that processes incoming messages.
51 /// Borrowed, not owned: it must outlive this server (see
52 /// `docs/spec/concurrency_and_lifetimes.md`, "Destruction
53 /// ordering").
54 /// @param port TCP port to listen on. Pass 0 to let the OS pick a free port.
55 /// @param cfg Backlog tuning. Default: 64-connection backlog.
56 // Copy/move are implicitly deleted by the non-copyable/non-movable
57 // std::mutex/std::thread members below — no explicit `= delete` needed
58 // (matches the rest of the codebase's convention, e.g. `LocalBackend`).
59 SocketServer(::morph::backend::RemoteServer& server MORPH_LIFETIMEBOUND, std::uint16_t port, Config cfg = {})
60 : _server{server}, _requestedPort{port}, _cfg{cfg} {}
61
62 /// @brief Stops accepting and closes every client connection.
63 ~SocketServer() { close(); }
64
65 /// @brief Starts listening for incoming WebSocket connections.
66 /// @return `true` if the server successfully bound to the requested port.
67 bool listen() {
68 try {
69 _listenSocket = ::morph::net::detail::TcpSocket::listen(_requestedPort, _cfg.backlog);
70 } catch (const std::exception&) {
71 return false;
72 }
73 _closing.store(false);
74 _acceptThread = std::thread{[this] { acceptLoop(); }};
75 return true;
76 }
77
78 /// @brief Returns the port the server is currently bound to.
79 /// @return Bound TCP port (OS-assigned when constructed with port 0), or `0` before `listen()` succeeds.
80 [[nodiscard]] std::uint16_t port() const { return _listenSocket.boundPort(); }

82 /// @brief Stops accepting new connections and closes every client connection.
83 ///
84 /// Idempotent: safe to call more than once (including implicitly, via the
85 /// destructor, after an explicit call).
86 void close() {
87 bool const wasAlreadyClosing = _closing.exchange(true);
88 if (wasAlreadyClosing && !_acceptThread.joinable()) {
89 return;
90 }
91 _listenSocket.shutdownBoth();
92 if (_acceptThread.joinable()) {
93 _acceptThread.join();
94 }
95 std::vector<std::shared_ptr<ClientConnection>> clients;
96 std::vector<std::thread> threads;
97 {
98 std::scoped_lock lock{_clientsMtx};
99 clients = _clients;
100 threads.swap(_clientThreads);
101 _clients.clear();
102 }
103 for (auto& client : clients) {
104 // `closed` first, so no client thread starts a *new* send; then
105 // shutdown without taking writeMtx. Taking it here would deadlock
106 // exactly when the shutdown is most needed: a client thread blocked
107 // in sendAll against a stalled peer's full socket buffer holds
108 // writeMtx for as long as that send is stuck, and close() has no
109 // timeout -- it is reached from the destructor. shutdownBoth() is
110 // documented safe from any thread and is itself the mechanism that
111 // unblocks that send (it fails, sendText catches, the thread exits
112 // and releases the lock). Locking to "protect" the socket would
113 // therefore wait on the very thing it is trying to interrupt.
114 client->closed.store(true);
115 client->socket.shutdownBoth();
116 }
117 for (auto& t : threads) {
118 if (t.joinable()) {
119 t.join();
120 }
121 }
122 }

124private:
125 struct ClientConnection {
126 ClientConnection(::morph::net::detail::TcpSocket sock, ::morph::backend::ConnectionId connectionId)
127 : socket{std::move(sock)}, cid{connectionId} {}
128 ::morph::net::detail::TcpSocket socket;
129 /// Scope every `register` on this connection belongs to, so dropping
130 /// the connection reclaims its models. See `RemoteServer::openConnection`.
131 ::morph::backend::ConnectionId cid{0};
132 std::mutex writeMtx;
133 std::atomic<bool> closed{false};
134
135 void sendText(const std::string& payload) {
136 std::scoped_lock lock{writeMtx};
137 if (closed.load() || !socket.valid()) {
138 return;
139 }
140 try {
141 std::string frame = ::morph::net::detail::encodeWsFrame(::morph::net::detail::WsOpcode::kText, payload,
142 /*mask=*/false);
143 socket.sendAll(frame.data(), frame.size());
144 } catch (const std::exception&) {
145 closed.store(true);
146 }
147 }
148 };

150 void acceptLoop() {
151 for (;;) {
152 ::morph::net::detail::TcpSocket clientSocket;
153 try {
154 clientSocket = _listenSocket.accept();
155 } catch (const std::exception&) {
156 return; // listener was shut down (server closing)
157 }
158 if (_closing.load()) {
159 return;
160 }
161 auto conn = std::make_shared<ClientConnection>(std::move(clientSocket), _server.openConnection());
162 std::thread clientThread{[this, conn] { clientLoop(conn); }};
163 {
164 std::scoped_lock lock{_clientsMtx};
165 _clients.push_back(conn);
166 _clientThreads.push_back(std::move(clientThread));
167 }
168 }
169 }

171 void clientLoop(const std::shared_ptr<ClientConnection>& conn) {
172 // Reclaim this connection's models however the loop exits — failed
173 // handshake, peer close, read error, or shutdown via close(). Without
174 // it every model registered over this transport outlived its connection
175 // forever: the scope machinery was wired into QtWebSocketServer only,
176 // and this server dispatched through the unscoped two-argument
177 // handle(). closeConnection() is idempotent, so the redundant call
178 // during close() (which joins these threads) is harmless.
179 struct ScopeGuard {
180 ScopeGuard(::morph::backend::RemoteServer& srv MORPH_LIFETIMEBOUND,
181 ::morph::backend::ConnectionId connectionId)
182 : server{srv}, cid{connectionId} {}
183 ~ScopeGuard() { server.closeConnection(cid); }
184 ScopeGuard(const ScopeGuard&) = delete;
185 ScopeGuard& operator=(const ScopeGuard&) = delete;
186 ScopeGuard(ScopeGuard&&) = delete;
187 ScopeGuard& operator=(ScopeGuard&&) = delete;
188
189 ::morph::backend::RemoteServer& server;
190 ::morph::backend::ConnectionId cid;
191 } const guard{_server, conn->cid};
192
193 std::string leftover;
194 try {
195 leftover = ::morph::net::detail::performServerHandshake(conn->socket);
196 } catch (const std::exception&) {
197 conn->closed.store(true);
198 return;
199 }
200 ::morph::net::detail::WsFrameReader reader;
201 reader.feed(leftover);
202 char buf[4096];
203 for (;;) {
204 if (!drainFrames(conn, reader)) {
205 return;
206 }
207 std::size_t got = 0;
208 try {
209 got = conn->socket.recvSome(buf, sizeof(buf));
210 } catch (const std::exception&) {
211 conn->closed.store(true);
212 return;
213 }
214 if (got == 0) {
215 conn->closed.store(true);
216 return;
217 }
218 reader.feed(std::string_view{buf, got});
219 }
220 }

222 /// Best-effort control-frame write (a Close echo or a Pong). Failure to
223 /// send one is never worth propagating: the connection is either already
224 /// going away or will be noticed as gone by the next read.
225 static void sendControlFrame(const std::shared_ptr<ClientConnection>& conn, ::morph::net::detail::WsOpcode opcode,
226 std::string_view payload) {
227 std::scoped_lock const lock{conn->writeMtx};
228 if (conn->closed.load() || !conn->socket.valid()) {
229 return;
230 }
231 try {
232 std::string const frame = ::morph::net::detail::encodeWsFrame(opcode, payload, /*mask=*/false);
233 conn->socket.sendAll(frame.data(), frame.size());
234 } catch (const std::exception&) { // NOLINT(bugprone-empty-catch) — see the doc comment above
235 }
236 }
237
238 // Returns false when the connection should stop reading (peer sent
239 // Close, or a protocol error was detected).
240 bool drainFrames(const std::shared_ptr<ClientConnection>& conn, ::morph::net::detail::WsFrameReader& reader) {
241 using ::morph::net::detail::WsOpcode;
242 for (;;) {
243 std::optional<::morph::net::detail::WsFrame> frame;
244 try {
245 frame = reader.tryExtractFrame();
246 } catch (const std::exception&) {
247 conn->closed.store(true);
248 return false;
249 }
250 if (!frame) {
251 return true;
252 }
253 if (frame->opcode == WsOpcode::kClose) {
254 sendControlFrame(conn, WsOpcode::kClose, "");
255 conn->closed.store(true);
256 return false;
257 }
258 if (frame->opcode == WsOpcode::kPing) {
259 sendControlFrame(conn, WsOpcode::kPong, frame->payload);
260 continue;
261 }
262 if (frame->opcode == WsOpcode::kPong) {
263 continue;
264 }
265 if (frame->opcode == WsOpcode::kText) {
266 std::weak_ptr<ClientConnection> weak = conn;
267 _server.handle(
268 frame->payload,
269 [weak](const std::string& reply) {
270 if (auto locked = weak.lock()) {
271 locked->sendText(reply);
272 }
273 },
274 conn->cid);
275 }
276 }
277 }
