32/// @brief Configuration for `SocketServer`.
33struct SocketServerConfig {
34 /// @brief Pending-connection backlog passed to the listening socket.
35 int backlog = 64;
36
37 /// @brief Bound on completing the RFC 6455 Upgrade handshake, from
38 /// accept to the request's terminating `\r\n\r\n`, as a single deadline
39 /// across the whole read. Zero disables it.
40 std::chrono::milliseconds handshakeTimeout{10000};
41
42 /// @brief Bound on writing one chunk (up to 64 KiB) of a reply frame.
43 ///
44 /// A loop timer armed around each chunk's write closes the connection when
45 /// it runs out, so a peer that stops reading is retired — its models
46 /// reclaimed, its requests no longer dispatched — instead of accumulating
47 /// replies it will never read. Zero disables it.
48 std::chrono::milliseconds sendTimeout{30000};
49};

51/// @brief Raw-socket WebSocket server front for `morph::backend::RemoteServer`.
52///
53/// Mirrors `morph::qt::QtWebSocketServer`'s behavior without Qt: for each
54/// accepted connection it performs the RFC 6455 handshake, then reads framed
55/// text messages and forwards each to `RemoteServer::handle()`; the reply is
56/// written back to the originating socket, or dropped if the connection is
57/// gone by the time the reply is ready.
58///
59/// @par Threading
60/// Owns no thread. The listener, every accepted connection and every reply
61/// write live on an `exec::IoLoop` — the application's one loop, shared with
62/// every other socket and timer built on it. `RemoteServer::handle()` replies
63/// arrive on the server's worker pool and are posted to the loop, which queues
64/// them on the connection's writer.
65///
66/// Cross-thread surface: `listen()` and `close()`, which run on the loop and
67/// wait for it (inline when already there), and `port()`, an atomic read.
68///
69/// @par Lifetime
70/// Holds `RemoteServer& _server` by reference, exactly like
71/// `QtWebSocketServer` — the server's owning `shared_ptr` must outlive this
72/// object (see `docs/spec/core/backend.md`'s Lifetime & ownership section).
73class SocketServer {
74public:
75 /// @brief Alias for the config struct.
76 using Config = SocketServerConfig;
77
78 /// @brief Constructs the server on @p loop; does not start listening.
79 /// @param loop The application's I/O loop. Borrowed: it must outlive this
80 /// server.
81 /// @param server `RemoteServer` instance that processes incoming messages.
82 /// Borrowed, not owned: it must outlive this server (see
83 /// `docs/spec/concurrency_and_lifetimes.md`, "Destruction
84 /// ordering").
85 /// @param port TCP port to listen on. Pass 0 to let the OS pick a free port.
86 /// @param cfg Backlog and timeout tuning. Default: 64-connection
87 /// backlog, 10s handshake timeout, 30s send timeout.
88 SocketServer(::morph::exec::IoLoop& loop MORPH_LIFETIMEBOUND,
89 ::morph::backend::RemoteServer& server MORPH_LIFETIMEBOUND, std::uint16_t port, Config cfg = {})
90 : _loop{&loop}, _core{std::make_shared<Core>(loop, server, cfg)}, _requestedPort{port} {}
91
92 /// @brief Constructs the server on a loop of its own, created by the first
93 /// `listen()`; does not start listening.
94 /// @param server `RemoteServer` instance that processes incoming messages.
95 /// Borrowed, not owned: it must outlive this server.
96 /// @param port TCP port to listen on. Pass 0 to let the OS pick a free port.
97 /// @param cfg Backlog and timeout tuning.
98 SocketServer(::morph::backend::RemoteServer& server MORPH_LIFETIMEBOUND, std::uint16_t port, Config cfg = {})
99 : _server{&server}, _requestedPort{port}, _cfg{cfg} {}

106 /// @brief Stops accepting and closes every client connection.
107 ///
108 /// Closes on the loop and waits for it (inline when already there), then
109 /// cuts the loop's tie to `RemoteServer`, so a connection's flow that
110 /// resumes later touches neither this object nor the server.
111 // NOLINTNEXTLINE(bugprone-exception-escape): teardown has no recovery if handing the close to the loop fails, so terminating is the outcome.
112 ~SocketServer() {
113 if (_core) {
114 _loop->runAndWait([core = _core] {
115 core->close();
116 core->server = nullptr;
117 });
118 }
119 }
120
121 /// @brief Starts listening for incoming WebSocket connections.
122 ///
123 /// Binds `127.0.0.1:port` and starts the accept flow, on the loop. With no
124 /// loop of its own yet, creates one first; a process out of descriptors
125 /// cannot, and gets `false`.
126 /// @return `true` if the server bound the requested port and is accepting.
127 bool listen() {
128 if (!_core) {
129 try {
130 _ownedLoop = std::make_unique<::morph::exec::IoLoop>();
131 } catch (const std::exception&) {
132 return false;
133 }
134 _loop = _ownedLoop.get();
135 _core = std::make_shared<Core>(*_ownedLoop, *_server, _cfg);
136 }
137 bool listening = false;
138 _loop->runAndWait([core = _core, port = _requestedPort, &listening] { listening = core->listen(port); });
139 return listening;
140 }
141
142 /// @brief Returns the port the server is currently bound to.
143 /// @return Bound TCP port (OS-assigned when constructed with port 0), or
144 /// `0` when not listening.
145 [[nodiscard]] std::uint16_t port() const { return _core ? _core->port.load() : std::uint16_t{0}; }
146
147 /// @brief Stops accepting new connections and closes every client connection.
148 ///
149 /// Releases the listening socket, so `port()` reads `0` afterwards and the
150 /// bound port is free to rebind — the same post-close observation
151 /// `QtWebSocketServer` makes. Reclaims every connection's models
152 /// (`RemoteServer::closeConnection`) before it returns.
153 ///
154 /// Runs on the loop and waits for it, inline when already there.
155 /// Idempotent, and safe to call from several threads at once on a live
156 /// object: each call is one loop task, and the loop runs them one at a
157 /// time. Racing a call against the *destructor* is still the caller's
158 /// problem (see `docs/spec/concurrency_and_lifetimes.md`, "Destruction
159 /// ordering").
160 void close() {
161 if (_core) {
162 _loop->runAndWait([core = _core] { core->close(); });
163 }
164 }

167 /// One accepted connection.
168 struct Client {
169 std::shared_ptr<::morph::net::detail::LoopConnection> conn;
170 /// Scope every `register` on this connection belongs to, so dropping
171 /// the connection reclaims its models. See `RemoteServer::openConnection`.
172 ::morph::backend::ConnectionId cid{0};
173 /// Set once its models are reclaimed; the flow touches nothing after.
174 bool done{false};

177 /// @brief Everything the loop owns, touched only in its tasks.
178 ///
179 /// Held by `shared_ptr` from the server and from every flow, so a flow
180 /// that resumes after the server is gone finds it closed and ends.
181 struct Core : std::enable_shared_from_this<Core> {
182 Core(::morph::exec::IoLoop& ioLoop, ::morph::backend::RemoteServer& remote, Config config)
183 : loop{ioLoop}, server{&remote}, cfg{config} {}

185 void note(char const* site) const noexcept {
186 ::morph::exec::detail::noteOwner(site, loop.loop(), loop.runningHere());
187 }

189 bool listen(std::uint16_t requestedPort) {
190 note("SocketServer::listen");
191 if (listener || server == nullptr) {
192 return false;
193 }
194 ::morph::net::detail::TcpSocket bound;
195 try {
196 bound = ::morph::net::detail::TcpSocket::listen(requestedPort, cfg.backlog);
197 } catch (const std::exception&) {
198 return false;
199 }
200 if (!bound.setNonBlocking()) {
201 return false;
202 }
203 auto adopted = ::core::net::adoptListener(loop.loop(), bound.nativeHandle());
204 if (!adopted) {
205 return false; // `bound` still owns the descriptor and closes it
206 }
207 static_cast<void>(bound.release());
208 listener = std::move(*adopted);
209 port.store(listener->boundPort());
210 ++generation;
211 loop.loop().spawn(acceptFlow(shared_from_this(), listener.get(), generation));
212 return true;
213 }
214
215 void close() {
216 note("SocketServer::close");
217 ++generation;
218 if (listener) {
219 // The accept flow parked on it resumes, in a later turn, with
220 // `Cancelled`, sees the generation moved on, and ends without
221 // touching the listener it no longer owns.
222 listener->close();
223 listener.reset();
224 }
225 port.store(0);
226 for (auto const& client : std::exchange(clients, {})) {
227 finish(*client, /*flush=*/false);
228 }
229 }
230
231 /// Reclaims @p client's models and closes its socket. Idempotent.
232 void finish(Client& client, bool flush) const {
233 if (client.done) {
234 return;
235 }
236 client.done = true;
237 if (server != nullptr) {
238 server->closeConnection(client.cid);
239 }
240 if (flush) {
241 ::morph::net::detail::closeAfterFlush(client.conn);
242 } else {
243 ::morph::net::detail::closeConnection(*client.conn);
244 }
245 }
246
247 void forget(Client const& client) {
248 std::erase_if(clients, [&client](std::shared_ptr<Client> const& entry) { return entry.get() == &client; });
249 }

251 void accepted(std::unique_ptr<::core::net::ISocket> socket) {
252 note("SocketServer::accept");
253 if (server == nullptr) {
254 return;
255 }
256 auto client =
257 std::make_shared<Client>(Client{.conn = std::make_shared<::morph::net::detail::LoopConnection>(
258 loop.loop(), std::move(socket), cfg.sendTimeout),
259 .cid = server->openConnection(),
260 .done = false});
261 clients.push_back(client);
262 loop.loop().spawn(clientFlow(shared_from_this(), client));
263 }

265 /// Accepts until the listener it was started with is closed.
266 static ::core::async::Task<void> acceptFlow(std::shared_ptr<Core> self, ::core::net::IListener* accepting,
267 std::uint64_t startedAt) {
268 for (;;) {
269 auto next = co_await accepting->accept();
270 if (self->generation != startedAt) {
271 co_return;
272 }
273 if (!next) {
274 if (next.error().code == ::core::net::NetErrorCode::Cancelled) {
275 co_return;
276 }
277 // Out of descriptors, or another failure the next attempt
278 // may not repeat: wait a little rather than spin on a
279 // listener that stays readable.
280 co_await self->loop.loop().delay(std::chrono::milliseconds{50});
281 if (self->generation != startedAt) {
282 co_return;
283 }
284 continue;
285 }
286 self->accepted(std::move(*next));
287 }
288 }

290 /// One connection: handshake, then read frames until it ends.
291 static ::core::async::Task<void> clientFlow(std::shared_ptr<Core> self, std::shared_ptr<Client> client) {
292 auto const connection = client->conn;
293 auto header = co_await ::morph::net::detail::readHeaderBlockAsync(connection, self->cfg.handshakeTimeout);
294 if (client->done) {
295 co_return;
296 }
297 std::optional<std::string> leftover;
298 if (header) {
299 try {
300 auto const request = ::morph::net::detail::parseClientHandshakeRequest(header->header);
301 static_cast<void>(::morph::net::detail::enqueueFrame(
302 connection, ::morph::net::detail::buildServerHandshakeResponse(request.key)));
303 leftover = std::move(header->leftover);
304 } catch (const std::exception&) { // NOLINT(bugprone-empty-catch)
305 // A malformed Upgrade request: the connection ends below.
306 }
307 }
308 bool flush = false;
309 if (leftover) {
310 // Server role: RFC 6455 §5.1 requires a client to mask every
311 // frame it sends, so this reader must reject an unmasked one.
312 ::morph::net::detail::WsFrameReader reader{/*expectMasked=*/true};
313 reader.feed(*leftover);
314 std::array<std::byte, 4096> buf{};
315 for (;;) {
316 auto const outcome = self->drainFrames(client, reader);
317 if (outcome != Drain::More) {
318 flush = outcome == Drain::Closed;
319 break;
320 }
321 auto const got = co_await connection->socket->read(buf);
322 if (client->done || !got || *got == 0) {
323 break;
324 }
325 // NOLINTNEXTLINE(cppcoreguidelines-pro-type-reinterpret-cast): the socket yields bytes, the reader takes text; same object representation.
326 reader.feed(std::string_view{reinterpret_cast<char const*>(buf.data()), *got});
327 }
328 }
329 if (client->done) {
330 co_return;
331 }
332 self->finish(*client, flush);
333 self->forget(*client);
334 }

336 enum class Drain : std::uint8_t {
337 More, ///< Read on.
338 Closed, ///< The peer sent Close; the echo is queued.
339 Failed, ///< A protocol error, or the connection is done.
340 };
341
342 Drain drainFrames(std::shared_ptr<Client> const& client, ::morph::net::detail::WsFrameReader& reader) const {
343 using ::morph::net::detail::WsOpcode;
344 for (;;) {
345 std::optional<::morph::net::detail::WsFrame> frame;
346 try {
347 frame = reader.tryExtractFrame();
348 } catch (const std::exception&) {
349 return Drain::Failed;
350 }
351 if (!frame) {
352 return Drain::More;
353 }
354 if (frame->opcode == WsOpcode::kClose) {
355 static_cast<void>(::morph::net::detail::enqueueFrame(
356 client->conn, ::morph::net::detail::encodeWsFrame(WsOpcode::kClose, frame->payload, false)));
357 return Drain::Closed;
358 }
359 if (frame->opcode == WsOpcode::kPing) {
360 static_cast<void>(::morph::net::detail::enqueueFrame(
361 client->conn, ::morph::net::detail::encodeWsFrame(WsOpcode::kPong, frame->payload, false)));
362 continue;
363 }
364 if (frame->opcode == WsOpcode::kText && server != nullptr) {
365 dispatch(client, frame->payload);
366 if (client->done) {
367 return Drain::Failed;
368 }
369 }
370 }
371 }

373 /// Hands one request to the server. Its reply comes back on whichever
374 /// thread produces it — the server's pool, usually — and is posted
375 /// here, to be queued on this connection if it is still open.
376 void dispatch(std::shared_ptr<Client> const& client, std::string const& payload) const {
377 std::weak_ptr<Client> const weak = client;
378 server->handle(
379 payload,
380 [loopHandle = loop.weak(), weak](const std::string& reply) {
381 // Encoded on the replying thread, so the loop only queues it.
382 std::string frame = ::morph::net::detail::encodeWsFrame(::morph::net::detail::WsOpcode::kText,
383 reply, /*mask=*/false);
384 static_cast<void>(loopHandle.post([weak, frame = std::move(frame)]() mutable {
385 auto const target = weak.lock();
386 if (!target || target->done) {
387 return;
388 }
389 ::morph::exec::detail::noteOwner("SocketServer::reply", target->conn->loop,
390 ::morph::exec::runningOn(target->conn->loop));
391 static_cast<void>(::morph::net::detail::enqueueFrame(target->conn, std::move(frame)));
392 }));
393 },
394 client->cid);
395 }
