11/// @file
12/// `TimeoutScheduler` — "run this callback once, in N milliseconds, unless
13/// cancelled first" — in two builds of the same public API.
14///
15/// @par Why two builds
16/// The ordinary build owns a dedicated `std::thread`. A **single-threaded
17/// Emscripten** build cannot: Qt for WebAssembly is installed here as
18/// `wasm_singlethread` (`.github/workflows/wasm-ladder.yml`) and
19/// `cmake/morph_add_rung.cmake` passes no `-pthread`, so Emscripten's
20/// non-pthread `pthread_create` stub fails and `std::thread`'s constructor
21/// throws `std::system_error` ("thread constructor failed") — from inside
22/// whatever completion callback happened to enable the deadline. Every WASM
23/// client in this repository is Qt-event-loop driven and would hit this the
24/// moment it called `Bridge::setExecuteDeadline` (which
25/// `examples/common/gui/event_poller.hpp`'s constructor does
26/// unconditionally, on every poll open).
27///
28/// So under `__EMSCRIPTEN__` without `__EMSCRIPTEN_PTHREADS__` this class is
29/// built on `emscripten_async_call` — the browser's own `setTimeout` — and
30/// fires its callbacks on the single main thread, i.e. on the same thread the
31/// Qt event loop and every `QtExecutor`-posted completion callback already
32/// run on. Deadlines still fire; nothing is silently disabled.
33///
34/// @par What differs between the two builds
35/// - **Callback thread.** Threaded build: a private background thread, so a
36/// callback must be prepared to run concurrently with the caller (the one
37/// real callback in this codebase, `executeVia`'s, only touches a
38/// `CompletionState`, which is itself mutex-guarded). Browser build: the
39/// main thread, never concurrently with anything.
40/// - **Cancellation.** Threaded build: the entry, its callback and everything
41/// the callback captured are erased immediately. Browser build: identical
42/// for the callback and its captures (the map entry is erased at once), but
43/// the underlying browser timer is not itself cleared — it still fires at
44/// its original deadline and finds nothing to do. Only a small ticket
45/// allocation outlives `cancel()`, until that point.
46/// - **Destruction.** Threaded build: the destructor joins its thread, so no
47/// callback can be in flight afterwards. Browser build: nothing to join;
48/// pending browser timers observe an expired `std::weak_ptr` to the
49/// scheduler's state and return without invoking anything.
50///
51/// @warning The browser build has never been compiled or run in this
52/// repository — no Emscripten toolchain is available where it was written.
53/// Its only verification is the `ladder-wasm` CI compile gate. Stated plainly
54/// here rather than smoothed over, exactly like `examples/TESTING.md`'s note
55/// on the WASM clients themselves.

77/// @brief Background scheduler that invokes a callback once after a delay, unless cancelled first.
78///
79/// Neither `Bridge` nor `RemoteServer` is bound to a specific `IExecutor`
80/// with a delayed-post primitive, so a single dedicated thread per instance
81/// tracks pending deadlines and fires callbacks when they elapse. Used by
82/// `RemoteServer` to enforce `LimitPolicy::executeTimeout` (server-side —
83/// see `docs/spec/core/backend.md`) and by `Bridge::setExecuteDeadline`
84/// (client-side — see `docs/spec/core/completion.md`). See this file's `@file`
85/// comment for the single-threaded-WASM build of the same API.
86class TimeoutScheduler {
87public:
88 /// @brief Opaque identifier for one scheduled callback.
89 using Handle = std::uint64_t;
90
91 /// @brief Starts the background thread.
92 TimeoutScheduler() : _thread{[this] { run(); }} {}
93
94 /// @brief Stops the background thread and joins it.
95 ~TimeoutScheduler() {
96 {
97 std::scoped_lock const lock{_mtx};
98 _stop = true;
99 }
100 _cv.notify_all();
101 _thread.join();
102 }
103
104 TimeoutScheduler(const TimeoutScheduler&) = delete;
105 TimeoutScheduler& operator=(const TimeoutScheduler&) = delete;
106 TimeoutScheduler(TimeoutScheduler&&) = delete;
107 TimeoutScheduler& operator=(TimeoutScheduler&&) = delete;
108
109 /// @brief Schedules @p callback to run after @p delay on the scheduler's
110 /// background thread, unless cancelled first via `cancel()`.
111 /// @param delay Time to wait before firing.
112 /// @param callback Invoked on the scheduler thread if not cancelled in time.
113 /// Exceptions it throws are logged and swallowed.
114 /// @return Handle usable with `cancel()`.
115 Handle schedule(std::chrono::milliseconds delay, std::function<void()> callback) {
116 auto const deadline = std::chrono::steady_clock::now() + delay;
117 std::scoped_lock const lock{_mtx};
118 Handle const handle = ++_nextHandle;
119 auto iter = _entries.emplace(deadline, Entry{handle, std::move(callback)});
120 _index[handle] = iter;
121 _cv.notify_all();
122 return handle;
123 }
124
125 /// @brief Cancels a previously scheduled callback immediately.
126 ///
127 /// If @p handle has not fired yet, its entry (and anything its callback
128 /// captured) is erased right away — the caller does not have to wait for
129 /// the original deadline for that memory to be released. A no-op if
130 /// @p handle already fired or was already cancelled.
131 /// @param handle Handle returned by a prior `schedule()` call.
132 void cancel(Handle handle) {
133 std::scoped_lock const lock{_mtx};
134 auto found = _index.find(handle);
135 if (found == _index.end()) {
136 return;
137 }
138 _entries.erase(found->second);
139 _index.erase(found);
140 }
141
142private:
143 struct Entry {
144 Handle handle;
145 std::function<void()> callback;
146 };
147
148 void run() {
149 std::unique_lock lock{_mtx};
150 while (!_stop) {
151 if (_entries.empty()) {
152 _cv.wait(lock);
153 continue;
154 }
155 auto const nextDeadline = _entries.begin()->first;
156 _cv.wait_until(lock, nextDeadline);
157 if (_stop) {
158 break;
159 }
160 auto now = std::chrono::steady_clock::now();
161 while (!_entries.empty() && _entries.begin()->first <= now) {
162 auto iter = _entries.begin();
163 Entry entry = std::move(iter->second);
164 _index.erase(entry.handle);
165 _entries.erase(iter);
166 lock.unlock();
167 try {
168 entry.callback();
169 } catch (const std::exception& exc) {
170 ::morph::log::logError("[timeout-scheduler] callback threw: " + std::string{exc.what()});
171 } catch (...) {
172 ::morph::log::logError("[timeout-scheduler] callback threw unknown exception");
173 }
174 lock.lock();
175 now = std::chrono::steady_clock::now();
176 }
177 }
178 }
179
180 std::mutex _mtx;
181 std::condition_variable _cv;
182 std::multimap<std::chrono::steady_clock::time_point, Entry> _entries;
183 std::unordered_map<Handle, std::multimap<std::chrono::steady_clock::time_point, Entry>::iterator> _index;
184 Handle _nextHandle{0};
185 bool _stop{false};
186 std::thread _thread;
187};

191/// @brief Single-threaded-Emscripten build of the same API, backed by the
192/// browser's `setTimeout` (`emscripten_async_call`) instead of a
193/// thread. See this file's `@file` comment for why it exists and
194/// exactly how its behaviour differs.
195class TimeoutScheduler {
196public:
197 /// @brief Opaque identifier for one scheduled callback.
198 using Handle = std::uint64_t;
199
200 /// @brief Creates the scheduler. Starts no thread — there is none to start.
201 TimeoutScheduler() = default;
202
203 /// @brief Drops every still-pending callback without firing it.
204 ///
205 /// Browser timers already queued outlive this object; each holds only a
206 /// `std::weak_ptr` to `_state` and returns immediately once it expires,
207 /// which is precisely at this destructor. Matches the threaded build's
208 /// "`~TimeoutScheduler` drops pending entries without firing them".
209 ~TimeoutScheduler() = default;
210
211 TimeoutScheduler(const TimeoutScheduler&) = delete;
212 TimeoutScheduler& operator=(const TimeoutScheduler&) = delete;
213 TimeoutScheduler(TimeoutScheduler&&) = delete;
214 TimeoutScheduler& operator=(TimeoutScheduler&&) = delete;
215
216 /// @brief Schedules @p callback to run after @p delay on the main
217 /// (browser) thread, unless cancelled first via `cancel()`.
218 /// @param delay Time to wait before firing.
219 /// @param callback Invoked on the main thread if not cancelled in time.
220 /// Exceptions it throws are logged and swallowed.
221 /// @return Handle usable with `cancel()`.
222 Handle schedule(std::chrono::milliseconds delay, std::function<void()> callback) {
223 Handle const handle = ++_state->nextHandle;
224 _state->pending.emplace(handle, std::move(callback));
225 // Owned by the browser timer, deleted by `fire` below whether or not
226 // the entry is still live by then. A raw `new` rather than a
227 // `unique_ptr` because the ownership genuinely crosses a C callback
228 // boundary that cannot carry a smart pointer.
229 auto* ticket = new Ticket{_state, handle};
230 ::emscripten_async_call(&TimeoutScheduler::fire, ticket, clampMillis(delay));
231 return handle;
232 }
233
234 /// @brief Cancels a previously scheduled callback immediately.
235 ///
236 /// If @p handle has not fired yet, its callback (and anything that
237 /// callback captured) is released right away, exactly like the threaded
238 /// build. The browser timer itself is left to elapse and find nothing —
239 /// see the `@file` comment. A no-op if @p handle already fired or was
240 /// already cancelled.
241 /// @param handle Handle returned by a prior `schedule()` call.
242 void cancel(Handle handle) { _state->pending.erase(handle); }
243
244private:
245 struct State {
246 std::unordered_map<Handle, std::function<void()>> pending;
247 Handle nextHandle{0};
248 };
249
250 struct Ticket {
251 std::weak_ptr<State> state;
252 Handle handle;
253 };
254
255 /// @brief @p delay as the `int` milliseconds `emscripten_async_call`
256 /// takes, saturating rather than wrapping (a `std::chrono`
257 /// duration can hold far more than an `int` can).
258 ///
259 /// @note This is a real, documented behavioural asymmetry from the
260 /// threaded build, which honours the full `std::chrono::milliseconds`
261 /// range unconditionally: a delay beyond `INT_MAX` ms (~24.85 days) fires
262 /// at ~24.85 days here instead of at its true, much later requested time.
263 /// `emscripten_async_call`'s `int` parameter is a hard platform
264 /// constraint with no larger-range alternative to fall back to, so this
265 /// is accepted rather than worked around. No caller in this codebase
266 /// currently requests a deadline anywhere near that range.
267 /// @param delay The requested delay.
268 /// @return A non-negative millisecond count that fits in an `int`.
269 [[nodiscard]] static int clampMillis(std::chrono::milliseconds delay) noexcept {
270 auto const count = delay.count();
271 if (count <= 0) {
272 return 0;
273 }
274 if (count > static_cast<decltype(count)>(std::numeric_limits<int>::max())) {
275 return std::numeric_limits<int>::max();
276 }
277 return static_cast<int>(count);
278 }
279
280 /// @brief The C callback the browser timer invokes.
281 /// @param arg The `Ticket*` handed to `emscripten_async_call`; always
282 /// deleted here, whether or not its entry is still live.
283 static void fire(void* arg) {
284 std::unique_ptr<Ticket> const ticket{static_cast<Ticket*>(arg)};
285 auto state = ticket->state.lock();
286 if (!state) {
287 return;
288 }
289 auto found = state->pending.find(ticket->handle);
290 if (found == state->pending.end()) {
291 return; // cancelled before this timer elapsed
292 }
293 std::function<void()> callback = std::move(found->second);
294 state->pending.erase(found);
295 try {
296 callback();
297 } catch (const std::exception& exc) {
298 ::morph::log::logError("[timeout-scheduler] callback threw: " + std::string{exc.what()});
299 } catch (...) {
300 ::morph::log::logError("[timeout-scheduler] callback threw unknown exception");
301 }
302 }
303
304 /// @brief Held by `shared_ptr` so a browser timer that outlives this
305 /// object detects that fact instead of writing to freed storage —
306 /// the same weak-token pattern as `morph::bridge::Bridge::_liveness`.
307 std::shared_ptr<State> _state{std::make_shared<State>()};
308};

310#endif
