25/// @file
26/// `TimeoutScheduler` — "run this callback once, in N milliseconds, unless
27/// cancelled first" — over core-cpp's event-loop timers.
28///
29/// @par One class, two ways of driving its loop
30/// The deadlines live in a `core::net::PlatformLoop`, and the one thing that
31/// differs between builds is who turns it:
32/// - **Native:** the scheduler owns a `std::thread` that runs the loop.
33/// `schedule()` and `cancel()` may be called from any thread; they record
34/// the request under a mutex and hand the loop a batch to arm or disarm.
35/// Callbacks run on that thread, so a callback must be prepared to run
36/// concurrently with the caller.
37/// - **Single-threaded WebAssembly** (`__EMSCRIPTEN__` without
38/// `__EMSCRIPTEN_PTHREADS__`): there is no thread to start — Qt for
39/// WebAssembly is `wasm_singlethread` here and no `-pthread` is passed, so
40/// `std::thread`'s constructor would throw. The loop is host-driven: the
41/// browser's own timer pumps it, and `schedule()` and `cancel()` arm and
42/// retire the timer directly. Callbacks run on the main thread, the one the
43/// Qt event loop and every `QtExecutor`-posted completion already run on.
44///
45/// @par What the two builds share
46/// - `cancel()` releases the callback, and everything it captured, before it
47/// returns — the entry lives in this class's own `pending` map, not in the
48/// loop — and retires the loop's timer as well.
49/// - The destructor drops every pending callback without firing it.
50/// - A callback that throws is logged through `morph::log` and swallowed.
51///
52/// @par What differs
53/// - **Cancelling a callback that has already started.** Native: `cancel()`
54/// cannot stop it and does not wait for it — the entry is taken out of
55/// `pending` before the callback is invoked, so a racing `cancel()` finds
56/// nothing and returns while the callback is still running. WebAssembly:
57/// the case cannot arise, because the callback and `cancel()` run on the
58/// same thread. A caller that must work in both builds cannot rely on the
59/// second.
60/// - **Destruction.** Native: the destructor stops the loop and joins its
61/// thread, so no callback is in flight afterwards. WebAssembly: nothing to
62/// join; the timers are retired and the loop is destroyed. A browser timer
63/// already scheduled to pump it finds it gone and runs nothing: from
64/// core-cpp 0.3.0 each pump carries a weak reference to the loop, where
65/// 0.2.1's wrote into the freed loop.
66

69/// @brief Invokes a callback once after a delay, unless cancelled first.
70///
71/// Neither `Bridge` nor `RemoteServer` is bound to a specific `IExecutor`
72/// with a delayed-post primitive, so each owns one of these. Used by
73/// `RemoteServer` to enforce `LimitPolicy::executeTimeout` (server-side —
74/// see `docs/spec/core/backend.md`) and by `Bridge::setExecuteDeadline`
75/// (client-side — see `docs/spec/core/completion.md`). See this file's
76/// `@file` comment for how the native and single-threaded WebAssembly builds
77/// drive it.
78class TimeoutScheduler {
79public:
80 /// @brief Opaque identifier for one scheduled callback.
81 using Handle = std::uint64_t;
82
83#ifndef MORPH_TIMEOUT_SCHEDULER_HOST_DRIVEN
84 /// @brief Starts the thread that runs the scheduler's event loop.
85 TimeoutScheduler() : _thread{[this] { _loop.run(); }} {}
86
87 /// @brief Drops every pending callback without firing it, stops the loop
88 /// and joins its thread.
89 ///
90 /// The timers are retired on the loop's own thread, before it stops, so
91 /// the loop is destroyed with nothing armed.
92 ~TimeoutScheduler() {
93 auto dropped = takePending();
94 _loop.post([this] {
95 disarmAll();
96 _loop.stop();
97 });
98 _thread.join();
99 // `dropped` is destroyed here, with the loop's thread gone and every
100 // member still alive: a capture whose destructor calls back into this
101 // scheduler finds it whole.
102 }
103#else
104 /// @brief Creates the scheduler. Starts no thread: the browser's timer
105 /// pumps the loop.
106 TimeoutScheduler() = default;
107
108 /// @brief Drops every pending callback without firing it and retires the
109 /// loop's timers.
110 ~TimeoutScheduler() {
111 auto dropped = takePending();
112 disarmAll();
113 // `dropped` is destroyed here, with every member still alive.
114 }
115#endif
116
117 TimeoutScheduler(const TimeoutScheduler&) = delete;
118 TimeoutScheduler& operator=(const TimeoutScheduler&) = delete;
119 TimeoutScheduler(TimeoutScheduler&&) = delete;
120 TimeoutScheduler& operator=(TimeoutScheduler&&) = delete;
121
122 /// @brief Schedules @p callback to run after @p delay, unless cancelled
123 /// first via `cancel()`.
124 /// @param delay Time to wait before firing.
125 /// @param callback Invoked on the loop's thread (the main thread under
126 /// single-threaded WebAssembly) if not cancelled in time.
127 /// Exceptions it throws are logged and swallowed.
128 /// @return Handle usable with `cancel()`.
129 Handle schedule(std::chrono::milliseconds delay, std::function<void()> callback) {
130 auto const deadline = _loop.clock().now() + delay;
131 Handle handle{};
132 {
133 std::scoped_lock const lock{_mtx};
134 handle = ++_nextHandle;
135 _pending.emplace(handle, std::move(callback));
136 }
137 submit(Request{.handle = handle, .deadline = deadline});
138 return handle;
139 }
140
141 /// @brief Cancels a previously scheduled callback: releases one that has
142 /// not started, and returns without waiting for one that has.
143 ///
144 /// Two cases, and telling them apart is the caller's business because the
145 /// scheduler cannot:
146 ///
147 /// - **@p handle has not started.** Its callback — and anything it
148 /// captured — is released before this returns, the callback never runs,
149 /// and the loop's timer is retired.
150 /// - **@p handle is already running** (native build only). The callback
151 /// was taken out of `pending` before it was invoked, so this call finds
152 /// nothing and **returns while the callback is still executing** on the
153 /// loop's thread. It is neither interrupted nor waited for.
154 ///
155 /// So `cancel()` returning does **not** mean "no callback is in flight".
156 /// The only thing in this class that means that is `~TimeoutScheduler`,
157 /// which joins the loop's thread. A caller must therefore keep every
158 /// scheduled callback safe to run *after* its `cancel()`: both callbacks in
159 /// this repository (`Bridge::executeVia`'s deadline and `RemoteServer`'s
160 /// `LimitPolicy::executeTimeout`) capture a `shared_ptr` to the state they
161 /// settle and settle it write-once, so a late run is an ignored duplicate
162 /// rather than a use-after-free.
163 ///
164 /// Blocking here until the callback finished would be the wrong contract
165 /// rather than a missing feature: a callback that posts back to the
166 /// cancelling thread would deadlock it — the hazard
167 /// `docs/spec/concurrency_and_lifetimes.md` names, and the same reason
168 /// `CallbackScope` deliberately offers no block-until-drained.
169 ///
170 /// A no-op if @p handle already fired, is firing, or was already cancelled.
171 /// @param handle Handle returned by a prior `schedule()` call.
172 void cancel(Handle handle) {
173 // Moved out under the lock and destroyed after it, so a capture whose
174 // destructor calls back into this scheduler cannot deadlock on _mtx.
175 std::function<void()> released;
176 {
177 std::scoped_lock const lock{_mtx};
178 auto found = _pending.find(handle);
179 if (found == _pending.end()) {
180 return;
181 }
182 released = std::move(found->second);
183 _pending.erase(found);
184 }
185 submit(Request{.handle = handle, .deadline = std::nullopt});
186 }
187
188private:
189 /// One change for the loop to apply: arm @p handle's timer at a deadline,
190 /// or, with none, retire it.
191 struct Request {
192 Handle handle;
193 std::optional<::core::platform::SteadyTimePoint> deadline;
194 };
195
196 /// The state an armed loop timer hands back to `fire()`. Held in
197 /// `_timers`, whose nodes do not move, so its address is stable for as
198 /// long as the timer is armed.
199 struct Timer {
200 TimeoutScheduler* owner;
201 Handle handle;
202 ::core::net::TimerId id;
203 };
204
205 /// Applies @p request on the loop's thread. Natively that means queueing
206 /// it and waking the loop once per batch rather than once per request —
207 /// `Bridge` schedules and cancels one deadline per call.
208 void submit(Request request) {
209#ifndef MORPH_TIMEOUT_SCHEDULER_HOST_DRIVEN
210 bool wake = false;
211 {
212 std::scoped_lock const lock{_mtx};
213 wake = _requests.empty();
214 _requests.push_back(request);
215 }
216 if (wake) {
217 _loop.post([this] { applyRequests(); });
218 }
219#else
220 apply(request);
221#endif
222 }
223
224#ifndef MORPH_TIMEOUT_SCHEDULER_HOST_DRIVEN
225 /// Loop thread: applies every request queued since the last batch.
226 void applyRequests() {
227 std::vector<Request> batch;
228 {
229 std::scoped_lock const lock{_mtx};
230 batch.swap(_requests);
231 }
232 for (auto const& request : batch) {
233 apply(request);
234 }
235 }
236#endif
237
238 /// Loop thread: arms or retires one timer.
239 void apply(Request const& request) {
240 if (!request.deadline) {
241 disarm(request.handle);
242 return;
243 }
244 {
245 // Cancelled before the loop got to it: nothing to arm.
246 std::scoped_lock const lock{_mtx};
247 if (!_pending.contains(request.handle)) {
248 return;
249 }
250 }
251 auto [slot, inserted] =
252 _timers.try_emplace(request.handle, Timer{.owner = this, .handle = request.handle, .id = {}});
253 if (inserted) {
254 slot->second.id = _loop.addTimer(*request.deadline, &TimeoutScheduler::fire, &slot->second);
255 }
256 }
257
258 /// Loop thread: retires @p handle's timer, if it is still armed.
259 void disarm(Handle handle) {
260 auto found = _timers.find(handle);
261 if (found == _timers.end()) {
262 return;
263 }
264 static_cast<void>(_loop.cancelTimer(found->second.id));
265 _timers.erase(found);
266 }
267
268 /// Empties `_pending` under the lock, so no timer can fire what it held,
269 /// and hands its callbacks to the destructor to release outside it.
270 std::unordered_map<Handle, std::function<void()>> takePending() {
271 std::unordered_map<Handle, std::function<void()>> taken;
272 std::scoped_lock const lock{_mtx};
273 taken.swap(_pending);
274 return taken;
275 }
276
277 /// Loop thread: retires every armed timer.
278 void disarmAll() {
279 for (auto const& entry : _timers) {
280 static_cast<void>(_loop.cancelTimer(entry.second.id));
281 }
282 _timers.clear();
283 }
284
285 /// The loop's timer callback, on the loop's thread.
286 /// @param state The `Timer` this timer was armed with.
287 static void fire(void* state) {
288 auto const& timer = *static_cast<Timer const*>(state);
289 TimeoutScheduler& self = *timer.owner;
290 Handle const handle = timer.handle;
291 self._timers.erase(handle); // `timer` is gone from here on
292
293 std::function<void()> callback;
294 {
295 std::scoped_lock const lock{self._mtx};
296 auto found = self._pending.find(handle);
297 if (found == self._pending.end()) {
298 return; // cancelled after the timer came due
299 }
300 callback = std::move(found->second);
301 self._pending.erase(found);
302 }
303 try {
304 callback();
305 } catch (const std::exception& exc) {
306 ::morph::log::logError("[timeout-scheduler] callback threw: {}", exc.what());
307 } catch (...) {
308 ::morph::log::logError("[timeout-scheduler] callback threw unknown exception");
309 }
310 }
311
312 /// Guards `_pending`, `_nextHandle` and, natively, `_requests`.
313 std::mutex _mtx;
314 std::unordered_map<Handle, std::function<void()>> _pending;
315 Handle _nextHandle{0};
316#ifndef MORPH_TIMEOUT_SCHEDULER_HOST_DRIVEN
317 std::vector<Request> _requests;
318#endif
319 /// Loop thread only.
320 std::unordered_map<Handle, Timer> _timers;
321 /// Declared after everything its timers point into, so it is destroyed
322 /// first.
323 ::core::net::PlatformLoop _loop;
324#ifndef MORPH_TIMEOUT_SCHEDULER_HOST_DRIVEN
325 std::thread _thread;
326#endif
327};
