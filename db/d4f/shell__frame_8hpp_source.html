17/// @file
18/// @brief `morph::ui::ShellFrame`: the frame a client shows its screens in — a sidebar with the navigator, a top bar,
19/// and the page the workspace's tabs sit on. Built from backend widgets, so every frontend shows it.
20///
21/// Specified in `docs/spec/ui/shell_frame.md`.

23namespace morph::ui {

25/// @brief The frame's width for the sidebar, in units.
26inline constexpr int kShellSidebarWidth = 28;
27
28/// @brief The chat column's width, in units: enough for a message to read as prose beside the page.
29inline constexpr int kShellChatWidth = 32;
30static_assert(kShellSidebarWidth + 80 <= kExpandedWidthUnits,
31 "the page beside the sidebar must not be narrower than a compact window");
32
33/// @brief The narrowest the top bar's title and head lead are given before the head's trail wraps to a line of its
34/// own, in units; the same bound as the in-page head's.
35inline constexpr int kShellHeadLeadMinimum = 32;
36
37/// @brief What ends the line of a head's trail, which decides how far a compact top bar's trailing end stands from the
38/// window's edge: a button that draws no frame pads itself, so a frame leaves it to the button; anything else
39/// reaches the edge of its box.
40enum class HeadTrailEnd : std::uint8_t {
41 None, ///< Nothing is in the trail: the lead's text or badges end the line.
42 Unframed, ///< A ghost or icon-only action, or the head menu's trigger, which carries its own padding.
43 Framed, ///< A framed action, or a head tool such as a select.
44};

46/// @brief What a `ShellFrame` shows and reports.
47struct ShellFrameOptions {
48 /// @brief The application's name, at the head of the sidebar.
49 std::string applicationName;
50 /// @brief The navigator's entries and handlers.
51 NavigatorOptions navigator;
52 /// @brief Called when the person asks to sign out; empty hides the sign-out button.
53 std::function<void()> onSignOut;
54};

56/// @brief The frame around a client's screens.
57///
58/// A row filling the window: the sidebar (surface `sidebar`, `kShellSidebarWidth` units wide), holding the
59/// application's name and a `Navigator`, and the main column holding the top bar (surface `toolbar`) above the page
60/// (surface `page`), where the client places the workspace. The top bar is one line that wraps only when it is out of
61/// room: the shown screen's title with the active head's lead (subtitle and badges) under it, and the head's trail
62/// (tools and actions, then the principal while it sits in the head) at the trailing edge, dropping to a line of its
63/// own where it does not fit. The principal and a sign-out button sit at the sidebar's foot. Signed out, the sidebar
64/// and the top bar are hidden and the page is all there is. Compact (`setCompact(true)`), the sidebar is hidden and
65/// the top bar starts with a navigation menu over the navigator's entries, ending with the principal's sign-out entry,
66/// so the page has nearly the whole window. Neither copyable nor movable; owned by the owner thread of its backend.

409 /// @brief Takes the frame's compact form, or its full one again. Compact, the sidebar, with the principal and the
410 /// sign-out button at its foot, is hidden, and the top bar starts with the navigation menu
411 /// (`shell_nav_menu`, the framework icon `nav.menu`, named "Navigation"), which lists the navigator's
412 /// entries (`Navigator::menuEntries`) and opens the one chosen, as the navigator does; with a sign-out
413 /// handler it ends with a separator and a "Sign out" entry described by the principal, which signs out.
414 /// A new frame is not compact.
415 /// @param compact Whether the frame is compact.
416 void setCompact(bool compact) {
417 _compact = compact;
418 // The compact top bar is tighter: a touch-sized menu square beside the title leaves no room for the full
419 // steps, so the bar pads its line by step 1 above and below (more for a head that is not one plain line, see
420 // showBarInset), and the line pads its ends by step 1 and spaces its items by step 2.
421 showBarInset();
422 _topLine->setGap(compact ? 1 : 4);
423 _topMain->setGap(compact ? 2 : 3);
424 showChrome();
425 }

427 /// @brief Tells the frame what ends the head's trail. Compact, the line's ends are padded by step 1, which is the
428 /// whole inset a button that pads itself needs; anything else that ends the line (text, a badge, a select
429 /// or a button that draws a frame) stands step 4 from the edge, the page's gutter, so the frame adds step 3
430 /// after the trail. A trail that ends in a framed button or a tool also takes the larger vertical padding
431 /// of the bar. The expanded line's own end pad is that gutter already.
432 /// @param end What ends the trail. A new frame has `None`.
433 void setTrailEnd(HeadTrailEnd end) {
434 _trailEnd = end;
435 showBarInset();
436 showTrail();
437 }

473 // The navigation menu lists the navigator's entries, and, with a sign-out handler, a separator and a sign-out
474 // entry described by the principal.
475 void showNavEntries() {
476 std::vector<MenuEntry> entries = _navigator->menuEntries();
477 _navEntryCount = entries.size();
478 if (_onSignOut) {
479 entries.push_back(MenuEntry{.separator = true});
480 entries.push_back(MenuEntry{.label = "Sign out", .description = _principalName});
481 }
482 _navMenu->setItems(entries);
483 }

497 // What the trail holds shows: the principal while it sits in the head, the frame is expanded and there is a name,
498 // the slot while the head shows, and the row while either does. Every call that changes one of those ends here,
499 // so no stale visibility outlives it.
500 void showTrail() {
501 bool const principalShown = _principalInHead && !_compact && !_principalName.empty();
502 _topPrincipal->setVisible(principalShown);
503 _trail->setVisible(_headShown || principalShown);
504 _trailInset->setVisible(_compact && _trailEnd != HeadTrailEnd::Unframed);
505 }
