22/// @brief The frame's width for the sidebar, in units.
23inline constexpr int kShellSidebarWidth = 28;

25/// @brief What a `ShellFrame` shows and reports.
26struct ShellFrameOptions {
27 /// @brief The application's name, at the head of the sidebar.
28 std::string applicationName;
29 /// @brief The navigator's entries and handlers.
30 NavigatorOptions navigator;
31 /// @brief Called when the person asks to sign out; empty hides the sign-out button.
32 std::function<void()> onSignOut;

35/// @brief The frame around a client's screens.
36///
37/// A row filling the window: the sidebar (surface `sidebar`, `kShellSidebarWidth` units wide), holding the
38/// application's name and a `Navigator`, and the main column holding the top bar (surface `toolbar`: the shown
39/// screen's title, a slot for its active tab's head row, the principal and a sign-out button) above the page
40/// (surface `page`), where the client places the workspace. Signed out, the sidebar and the top bar are hidden and
41/// the page is all there is. Neither copyable nor movable; owned by the owner thread of its backend.
42class ShellFrame {
43public:
44 /// @brief The frame, appended to @p parent.
45 /// @param backend The backend its widgets are made with. Borrowed.
46 /// @param parent The container it is appended to; null makes it a backend root.
47 /// @param options What it shows and reports.
48 ShellFrame(IViewBackend& backend MORPH_LIFETIMEBOUND, ContainerWidget* parent, ShellFrameOptions options)
49 : _backend{backend},
50 _root{backend.createStack(parent, Axis::Horizontal)},
51 _sidebar{backend.createStack(_root.get(), Axis::Horizontal)},
52 _sideStart{backend.createSpacer(_sidebar.get())},
53 _side{backend.createStack(_sidebar.get(), Axis::Vertical)},
54 _sideTop{backend.createSpacer(_side.get())},
55 _name{backend.createText(_side.get())},
56 _navigator{std::make_unique<Navigator>(backend, _side.get(), std::move(options.navigator))},
57 _sideEnd{backend.createSpacer(_sidebar.get())},
58 _main{backend.createStack(_root.get(), Axis::Vertical)},
59 _topBar{backend.createStack(_main.get(), Axis::Horizontal)},
60 _topStart{backend.createSpacer(_topBar.get())},
61 _title{backend.createText(_topBar.get())},
62 _headSlot{backend.createStack(_topBar.get(), Axis::Horizontal)},
63 _principal{backend.createText(_topBar.get())},
64 _signOut{backend.createButton(_topBar.get())},
65 _topEnd{backend.createSpacer(_topBar.get())},
66 _page{backend.createStack(_main.get(), Axis::Vertical)},
67 _top{backend.createSpacer(_page.get())},
68 _padded{backend.createStack(_page.get(), Axis::Horizontal)},
69 _left{backend.createSpacer(_padded.get())},
70 _content{backend.createStack(_padded.get(), Axis::Vertical)},
71 _right{backend.createSpacer(_padded.get())} {
72 _root->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
73 _root->setTestId("shell");
74
75 _sidebar->setSurface("sidebar");
76 _sidebar->setLayout({.width = Sizing::fixed(kShellSidebarWidth), .height = Sizing::stretch()});
77 _sidebar->setGap(3);
78 _sidebar->setTestId("shell_sidebar");
79 // Spacers either side and above pad the sidebar's content by its gaps.
80 _sideStart->setLayout({.width = Sizing::fixed(0), .height = {}});
81 _side->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
82 _side->setGap(3);
83 _sideTop->setLayout({.width = {}, .height = Sizing::fixed(0)});
84 _sideEnd->setLayout({.width = Sizing::fixed(0), .height = {}});
85 _name->setRole(TextRole::Heading);
86 _name->setText(options.applicationName);
87 _name->setTestId("shell_application");
88
89 _main->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
90 _topBar->setSurface("toolbar");
91 _topBar->setGap(3);
92 // Spacers at both ends pad the bar by its gap.
93 _topStart->setLayout({.width = Sizing::fixed(0), .height = {}});
94 _topEnd->setLayout({.width = Sizing::fixed(0), .height = {}});
95 _topBar->setTestId("shell_topbar");
96 // Centred across the bar, so the principal's text sits level with the sign-out button's label.
97 _topBar->setAlign(Align::Center);
98 _title->setRole(TextRole::Heading);
99 _title->setLayout({.width = Sizing::stretch(), .height = {}});
100 _title->setTestId("shell_title");
101 // Its own stretch, independent of whether the row mounted inside it is currently visible: the mounted
102 // content's own visibility (hidden while a screen has no head) must not collapse the slot's width, since
103 // MorphStack.qml reads the *slot's* widthKind to decide whether the bar as a whole stretches or packs.
104 _headSlot->setLayout({.width = Sizing::stretch(), .height = {}});
105 _headSlot->setTestId("shell_head");
106 _principal->setRole(TextRole::Muted);
107 _principal->setTestId("shell_principal");
108 _signOut->setLabel("Sign out");
109 _signOut->setTestId("shell_signOut");
110 _signOut->setVisible(static_cast<bool>(options.onSignOut));
111 _signOut->setOnClick(std::move(options.onSignOut));
112
113 // The page is padded by spacing step 4: zero-sized spacers above and either side, each a gap away.
114 _page->setSurface("page");
115 _page->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
116 _page->setGap(4);
117 _page->setTestId("shell_page");
118 _top->setLayout({.width = {}, .height = Sizing::fixed(0)});
119 _padded->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
120 _padded->setGap(4);
121 _left->setLayout({.width = Sizing::fixed(0), .height = {}});
122 _content->setLayout({.width = Sizing::stretch(), .height = Sizing::stretch()});
123 _right->setLayout({.width = Sizing::fixed(0), .height = {}});
124 }

152 /// @brief Shows or hides the sidebar and the top bar: signed out, the page is all there is.
153 /// @param signedIn Whether someone is signed in.
154 void showSignedIn(bool signedIn) {
155 _sidebar->setVisible(signedIn);
156 _topBar->setVisible(signedIn);
157 }
158
159 /// @brief Shows a passive, transient message over the frame's main content area. Made the first time it is
160 /// asked for; a later call replaces its text and tone and restarts its dismiss interval, whether or not
161 /// it is still showing.
162 /// @param text The message, UTF-8.
163 /// @param tone The tone, mapped to the backend's theme.
164 void showToast(std::string_view text, Tone tone) {
165 if (_toast == nullptr) {
166 _toast = _backend.createToast(_main.get());
167 }
168 _toast->show(text, tone);
169 }
