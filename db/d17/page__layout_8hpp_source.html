23/// @brief A length in millimetres, held as an exact count of micrometres.
24///
25/// A page's geometry is compared in tests and travels in a document; an integer count compares exactly where a
26/// floating-point length would make "the footer's bottom edge equals the margin box's" depend on rounding.
27class Mm {
28public:
29 /// @brief Micrometres in a millimetre.
30 static constexpr std::int64_t kPerMillimetre = 1000;
31
32 /// @brief Zero.
33 constexpr Mm() noexcept = default;
34
35 /// @brief A length in millimetres, rounded to the nearest micrometre.
36 /// @param millimetres The length.
37 /// @throws std::invalid_argument when it is not finite or its micrometre count does not fit an `int32`.
38 // NOLINTNEXTLINE(google-explicit-constructor,hicpp-explicit-conversions): `.width = 210` writes a length.
39 Mm(double millimetres) {
40 double const micrometres = std::round(millimetres * static_cast<double>(kPerMillimetre));
41 if (!std::isfinite(micrometres) ||
42 std::abs(micrometres) > static_cast<double>(std::numeric_limits<std::int32_t>::max())) {
43 throw std::invalid_argument{"morph::ui::Mm: a length must be finite and fit 2147 m"};
44 }
45 _micrometres = static_cast<std::int32_t>(micrometres);
46 }
47
48 /// @brief A length from its micrometre count.
49 /// @param micrometres The count.
50 /// @return The length.
51 [[nodiscard]] static constexpr Mm fromMicrometres(std::int32_t micrometres) noexcept {
52 Mm out;
53 out._micrometres = micrometres;
54 return out;
55 }
56
57 /// @brief The length in micrometres.
58 /// @return The count.
59 [[nodiscard]] constexpr std::int32_t micrometres() const noexcept { return _micrometres; }
60
61 /// @brief The length in millimetres.
62 /// @return It as a `double`.
63 [[nodiscard]] constexpr double millimetres() const noexcept {
64 return static_cast<double>(_micrometres) / static_cast<double>(kPerMillimetre);
65 }
66
67 /// @brief The length in the shortest decimal form of millimetres: `210`, `148.5`, `98.46`.
68 /// @return The text.
69 [[nodiscard]] std::string text() const {
70 std::int64_t const magnitude = _micrometres < 0 ? -static_cast<std::int64_t>(_micrometres) : _micrometres;
71 std::string out = _micrometres < 0 ? "-" : "";
72 out += std::to_string(magnitude / kPerMillimetre);
73 std::int64_t fraction = magnitude % kPerMillimetre;
74 if (fraction != 0) {
75 std::string digits = std::to_string(fraction);
76 digits.insert(0, 3 - digits.size(), '0');
77 while (digits.back() == '0') {
78 digits.pop_back();
79 }
80 out += '.' + digits;
81 }
82 return out;
83 }
84
85 /// @brief Compares lengths.
86 /// @return The order.
87 [[nodiscard]] friend constexpr auto operator<=>(Mm const&, Mm const&) noexcept = default;
88
89 /// @brief The sum of two lengths, held at the largest or smallest length instead of overflowing.
90 /// @param left The first.
91 /// @param right The second.
92 /// @return The sum.
93 [[nodiscard]] friend constexpr Mm operator+(Mm left, Mm right) noexcept {
94 return saturated(static_cast<std::int64_t>(left._micrometres) + right._micrometres);
95 }
96
97 /// @brief The difference of two lengths, held at the largest or smallest length instead of overflowing.
98 /// @param left The first.
99 /// @param right The second.
100 /// @return The difference.
101 [[nodiscard]] friend constexpr Mm operator-(Mm left, Mm right) noexcept {
102 return saturated(static_cast<std::int64_t>(left._micrometres) - right._micrometres);
103 }
104
105private:
106 static constexpr Mm saturated(std::int64_t micrometres) noexcept {
107 constexpr std::int64_t kLowest = std::numeric_limits<std::int32_t>::lowest();
108 constexpr std::int64_t kHighest = std::numeric_limits<std::int32_t>::max();
109 return fromMicrometres(static_cast<std::int32_t>(std::clamp(micrometres, kLowest, kHighest)));
110 }
111
112 std::int32_t _micrometres = 0;
113};

157 /// @brief The sheet's height, 10 mm to 5000 mm.

167 /// @return The sheet's width less both side margins.

169

171 /// @return The sheet's height less both vertical margins.
172 [[nodiscard]] Mm marginHeight() const noexcept { return height - margins.top - margins.bottom; }
173
174 /// @brief Why this geometry is refused.
175 /// @return The reason, or nothing when it is valid: a side outside 10 mm to 5000 mm, a negative margin, or an
176 /// empty margin box.
177 [[nodiscard]] std::optional<std::string> problem() const {
178 if (width < kMinPageSide || width > kMaxPageSide || height < kMinPageSide || height > kMaxPageSide) {
179 return "a page side is 10 mm to 5000 mm";
180 }
181 for (Mm const margin : {margins.top, margins.right, margins.bottom, margins.left}) {
182 if (margin < Mm{}) {
183 return "a page margin is not negative";
184 }
185 }
186 if (marginWidth() <= Mm{} || marginHeight() <= Mm{}) {
187 return "a page's margins leave no margin box";
188 }
189 return std::nullopt;

229/// @brief A region's place on a sheet.
230struct RegionGeometry {
231 /// @brief The anchor.
232 PageAnchor anchor = PageAnchor::Top;
233 /// @brief The distance from the sheet's left edge; only an `Offset` region takes one.
234 Mm x{};
235 /// @brief The distance from the sheet's top edge; only an `Offset` region takes one.
236 Mm y{};
237 /// @brief The width; absent: the margin box's width for `Top` and `Bottom`, and the distance from `x` to the
238 /// margin box's right edge for `Offset`.
239 std::optional<Mm> width;
240 /// @brief The height; absent: the content's. A given height is the region's extent and clips its content.
241 std::optional<Mm> height;
242
243 /// @brief Compares geometries.
244 /// @return Whether every member is equal.
245 [[nodiscard]] friend bool operator==(RegionGeometry const&, RegionGeometry const&) noexcept = default;
246
247 /// @brief Why this region is refused on @p page.
248 /// @param page The sheet it is on, valid.
249 /// @return The reason, or nothing when it is valid: a negative or zero extent, an offset on a pinned region, or a
250 /// stated rectangle that leaves the sheet.
251 [[nodiscard]] std::optional<std::string> problem(PageGeometry const& page) const {
252 if (x < Mm{} || y < Mm{}) {
253 return "a page region's offset is not negative";
254 }
255 if (anchor != PageAnchor::Offset && (x != Mm{} || y != Mm{})) {
256 return "a pinned page region has no offset";
257 }
258 if ((width && *width <= Mm{}) || (height && *height <= Mm{})) {
259 return "a page region's width and height are above zero";
260 }
261 Mm const left = anchor == PageAnchor::Offset ? x : page.margins.left;
262 if (anchor == PageAnchor::Offset && !width && x >= page.width - page.margins.right) {
263 return "an offset page region with no width starts at or past the margin box's right edge";
264 }
265 if (width && left + *width > page.width) {
266 return "a page region leaves the sheet's width";
267 }
268 if (height) {
269 Mm const top = anchor == PageAnchor::Offset ? y
270 : anchor == PageAnchor::Top ? page.margins.top
271 : page.height - page.margins.bottom - *height;
272 if (top < Mm{} || top + *height > page.height) {
273 return "a page region leaves the sheet's height";
274 }
275 }
276 return std::nullopt;
277 }
278};

297/// @brief A sheet laid out: its scale, its displayed size and its regions.
298struct PageLayout {
299 /// @brief `min(1, available / width)`; 1 for an unlimited width.
300 double scale = 1.0;
301 /// @brief The sheet's width times the scale.
302 Mm displayedWidth{};
303 /// @brief The sheet's height times the scale.
304 Mm displayedHeight{};
305 /// @brief One entry per region, in node order.
306 std::vector<RegionLayout> regions;
307};
308
309/// @brief Lays a page out.
310/// @param page The sheet, valid.
311/// @param regions The regions in node order, each valid on @p page.
312/// @param availableWidth The width the container gives the sheet; absent is unlimited.
313/// @return The scale, the displayed size and each region's rectangle.
314[[nodiscard]] inline PageLayout layoutPage(PageGeometry const& page, std::span<RegionSpec const> regions,
315 std::optional<Mm> availableWidth) {
316 PageLayout out;
317 if (availableWidth) {
318 out.scale = std::clamp(availableWidth->millimetres() / page.width.millimetres(), 0.0, 1.0);
319 }
320 out.displayedWidth = Mm{page.width.millimetres() * out.scale};
321 out.displayedHeight = Mm{page.height.millimetres() * out.scale};
322 Mm const right = page.width - page.margins.right;
323 out.regions.reserve(regions.size());
324 for (RegionSpec const& spec : regions) {
325 RegionGeometry const& geometry = spec.geometry;
326 RegionLayout region;
327 Mm const height = geometry.height.value_or(spec.contentHeight);
328 if (geometry.anchor == PageAnchor::Offset) {
329 region.rect = {geometry.x, geometry.y, geometry.width.value_or(std::max(Mm{}, right - geometry.x)),
330 height};
331 } else {
332 Mm const top =
333 geometry.anchor == PageAnchor::Top ? page.margins.top : page.height - page.margins.bottom - height;
334 region.rect = {page.margins.left, top, geometry.width.value_or(page.marginWidth()), height};
335 }
336 region.overflow = geometry.height ? spec.contentHeight > *geometry.height
337 : region.rect.y < Mm{} || region.rect.y + height > page.height;
338 out.regions.push_back(region);
339 }
340 return out;
341}

369/// @brief A length as a cell edge: `round(length · cells / 25.4 mm)`, halves away from zero.
370/// @param length The length.
371/// @param cells The cells to 25.4 mm.
372/// @return The cell count.
373[[nodiscard]] constexpr int cellEdge(Mm length, int cells) noexcept {
374 constexpr std::int64_t kInch = 25'400;
375 std::int64_t const scaled = static_cast<std::int64_t>(length.micrometres()) * cells;
376 std::int64_t const half = kInch / 2;
377 return static_cast<int>(scaled >= 0 ? (scaled + half) / kInch : -((-scaled + half) / kInch));
378}
379
380} // namespace detail

398/// @brief A rectangle in cells, by rounding its edges and not its sizes: two rectangles that touch in millimetres
399/// touch in cells, and a rounded size never drifts from a neighbour's edge.
400/// @param rect The rectangle in the sheet's millimetres.
401/// @param pitch The terminal's pitch.
402/// @return The cells.
403[[nodiscard]] constexpr CellRect toCells(PageRect const& rect, CellPitch pitch) noexcept {
404 int const left = detail::cellEdge(rect.x, pitch.columns);
405 int const right = detail::cellEdge(rect.x + rect.width, pitch.columns);
406 int const top = detail::cellEdge(rect.y, pitch.rows);
407 int const bottom = detail::cellEdge(rect.y + rect.height, pitch.rows);
408 return {left, top, right - left, bottom - top};
409}

411/// @brief The cells of a region of no stated height whose content is @p contentRows rows tall: the anchored edge (the
412/// top of a `Top` or `Offset` region, the bottom of a `Bottom` one) is the rounded edge and the other edge is
413/// @p contentRows away, so a footer of three rows is three rows after rounding.
414/// @param rect The rectangle in millimetres, its height the content's.
415/// @param anchor The region's anchor.
416/// @param pitch The terminal's pitch.
417/// @param contentRows The content's height in rows.
418/// @return The cells.
419[[nodiscard]] constexpr CellRect toCells(PageRect const& rect, PageAnchor anchor, CellPitch pitch,
420 int contentRows) noexcept {
421 CellRect cells = toCells(rect, pitch);
422 cells.height = contentRows;
423 if (anchor == PageAnchor::Bottom) {
424 cells.row = detail::cellEdge(rect.y + rect.height, pitch.rows) - contentRows;
425 }
426 return cells;
427}
