19/// @brief Runs @p action on @p model as a dispatch would: recompute, the validator, the guard, then `execute`.
20///
21/// For a caller that holds a model directly — a unit test of a guard's outcome, server-side code running a model in
22/// process. A plain `model.execute(action)` runs no gate. There is no holder, so nothing is journaled and the guard is
23/// never skipped; a refusal reaches `model.onGuardRefused` when the model declares it.
24/// @tparam Model The model type; registered, so its id names a `ValidationError`.
25/// @tparam Action The action type; its handler is an ordinary one.
26/// @param model The model.
27/// @param action The action, by value: recomputing its computed fields changes it.
28/// @return What `execute` returned.
29/// @throws ValidationError if the recomputed action fails its validator.
30/// @throws std::exception whatever the guard, `onGuardRefused` or `execute` throws.
31template <typename Model, typename Action>
32auto executeChecked(Model& model, Action action) {
33 static_assert(!isTaskHandler<decltype(model.execute(action))>,
34 "executeChecked runs ordinary handlers only: a Task would outlive the action it refers to");
35 ::morph::forms::recomputeAll(action);
36 if (!ActionValidator<Action>::ready(action)) {
37 throw ValidationError{ModelTraits<Model>::typeId(), ActionTraits<Action>::typeId()};
38 }
39 if constexpr (DeclaresGuard<Action>) {
40 try {
41 detail::guardAction(std::as_const(model), action);
42 } catch (const std::exception& refusal) {
43 if constexpr (GuardRefusalObserver<Model, Action>) {
44 model.onGuardRefused(action, refusal);
45 }
46 throw;
47 }
48 }
49 return model.execute(action);
50}

53/// `BRIDGE_REGISTER_ACTION` registers — declares a guard.
54/// @param modelType The model's registered id.
55/// @param actionType The action's registered id.
56/// @return `ActionDispatcher::isGuarded` on `detail::defaultDispatcher()`.
57[[nodiscard]] inline bool isGuarded(std::string_view modelType, std::string_view actionType) {
58 return detail::defaultDispatcher().isGuarded(modelType, actionType);
